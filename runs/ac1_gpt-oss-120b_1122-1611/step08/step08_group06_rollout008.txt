"""
Propose a step‑function with a low value of

    2·n·max(conv(a,a)) / (sum(a)^2)

The optimiser works with a soft‑max (log‑weight) parametrisation,
therefore the variables are always non‑negative and sum to one.
Three phases are performed:

1. **Variance flattening** – minimise the L²‑distance of the self‑convolution
   to the constant value 1/(2·n‑1).  This drives the convolution towards a flat
   shape.

2. **Lp‑smoothing** – minimise the p‑norm of the convolution
   (∑cᵢᵖ)^{1/p} with p increasing from 2 → 30.  This is a smooth surrogate for
   the max and gives a stronger pull on the highest peaks.

3. **High‑γ log‑sum‑exp** – a very sharp log‑sum‑exp (γ up to 100) is used as a
   final surrogate of the true max.

Adam is used for optimisation, with a linear learning‑rate decay and a
restart mechanism if no improvement is seen for 30 s.  The support size is
chosen to be 10 000 (larger than the previous 3 800) – more degrees of freedom
allow a flatter convolution and a smaller objective.  If a previous solution
`height_sequence_1` exists it is interpolated to the required length and used as
a warm start.

The function respects the global time budget (default 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    # width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _interpolate_sequence(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linearly interpolate a 1‑D sequence to a new length."""
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == target_len:
        return seq.copy()
    x_old = np.arange(seq.size)
    x_new = np.linspace(0, seq.size - 1, target_len)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The best sequence discovered before the time budget expires is
    returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    total_budget = deadline - start_time

    # --------------------------------------------------------------
    # 1) Choose a support length.
    #    Larger n gives a lower theoretical bound 2n/(2n‑1) ≈ 1.
    # --------------------------------------------------------------
    TARGET_N = 10_000                     # feel free to tweak if needed
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            # Interpolate / truncate to TARGET_N
            if a0.size != TARGET_N:
                a0 = _interpolate_sequence(a0, TARGET_N)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full(TARGET_N, 1.0 / TARGET_N)
            n = TARGET_N
        else:
            n = TARGET_N
            a0 = None
    else:
        n = TARGET_N
        a0 = None

    # --------------------------------------------------------------
    # 2) Initialise log‑weights (soft‑max parametrisation)
    # --------------------------------------------------------------
    if a0 is not None:
        x = np.log(a0 + 1e-12)               # logits
    else:
        x = np.zeros(n, dtype=np.float64)    # uniform start

    # --------------------------------------------------------------
    # 3) Pre‑compute FFT length for self‑convolution (used repeatedly)
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)     # for a*a

    # --------------------------------------------------------------
    # 4) Adam optimiser state (shared by all phases)
    # --------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # --------------------------------------------------------------
    # 5) Helper: convert logits to probability vector
    # --------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # --------------------------------------------------------------
    # 6) Track the best solution seen so far
    # --------------------------------------------------------------
    a = _a_from_x(x)
    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * c.max() / (a.sum() ** 2)   # a.sum() == 1
    last_improve = time.monotonic()

    # --------------------------------------------------------------
    # 7) Phase schedule
    # --------------------------------------------------------------
    phase2_start = start_time + 0.55 * total_budget   # variance → Lp
    phase3_start = start_time + 0.85 * total_budget   # Lp → sharp max
    in_phase2 = False
    in_phase3 = False

    # --------------------------------------------------------------
    # 8) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- phase switches -------------------------------------------------
        if (not in_phase2) and (now >= phase2_start):
            in_phase2 = True
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] Switch to Phase 2 (Lp‑smoothing)")

        if (not in_phase3) and (now >= phase3_start):
            in_phase3 = True
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] Switch to Phase 3 (high‑γ max)")

        # ----- current a and its self‑convolution -----------------------------
        a = _a_from_x(x)                     # probability vector, sum == 1
        c = _self_conv(a, fft_len_self)      # length 2·n‑1

        # ----- exact objective (for logging and tracking) --------------------
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- compute gradient according to the active phase -----------------
        if not in_phase2 and not in_phase3:
            # Phase 1 – variance flattening (MSE to constant)
            avg = 1.0 / (2 * n - 1)               # because sum(a)=1
            r = c - avg
            g_a = 2.0 * _gradient_wrt_a(a, r)     # ∇ Σ (c‑avg)²
        elif in_phase2 and not in_phase3:
            # Phase 2 – Lp‑smooth surrogate, p grows from 2 → 30
            phase_len = max(phase3_start - phase2_start, 1e-12)
            phase_frac = (now - phase2_start) / phase_len
            p_cur = 2.0 + (30.0 - 2.0) * phase_frac      # 2 → 30 linearly
            # weight w_i = c_i^{p‑1}
            eps_c = 1e-12
            w = np.maximum(c, eps_c) ** (p_cur - 1.0)
            g_a = p_cur * _gradient_wrt_a(a, w)          # ∇ Σ c_i^p
        else:
            # Phase 3 – very sharp log‑sum‑exp (γ up to 100)
            phase_len = max(deadline - phase3_start, 1e-12)
            phase_frac = (now - phase3_start) / phase_len
            gamma = 30.0 + (100.0 - 30.0) * phase_frac   # 30 → 100
            c_shift = c - np.max(c)                      # for numerical stability
            w = np.exp(gamma * c_shift)
            w_sum = w.sum()
            if w_sum > 0.0:
                w /= w_sum
            else:
                w = np.full_like(c, 1.0 / c.size)
            g_a = _gradient_wrt_a(a, w)                 # ∇ Σ w_i c_i

        # ----- chain rule: gradient w.r.t. logits x (soft‑max Jacobian) -----
        avg_g = np.dot(a, g_a)                # ⟨a, ∂F/∂a⟩
        g_x = a * (g_a - avg_g)                # ∂F/∂x

        # ----- Adam update ---------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # linear learning‑rate decay over the whole budget
        lr_start = 0.4
        lr_end = 0.02
        lr = lr_start * (1.0 - elapsed / total_budget) + lr_end * (elapsed / total_budget)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- stagnation handling – restart from best + noise -------------
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # ----- occasional global jitter --------------------------------------
        if rng.random() < 0.02:   # 2 % chance each iteration
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # --------------------------------------------------------------
    # 9) Return the best sequence (clip to the limits used by evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
