Propose a step‑function (non‑negative sequence) that minimizes

        2·n·max(conv(a,a)) / (sum(a)**2)

The search is performed in three phases inside the total time budget
(≈1000 s, 5 s safety margin).

1. **Variance flattening** – minimise the L² distance of the
   self‑convolution to the constant value `1/(2·n‑1)`.  This drives the
   convolution towards a flat shape.

2. **Smoothed‑max refinement** – minimise a smooth approximation of the
   max using a log‑sum‑exp surrogate with an increasing temperature
   (`γ`).  The surrogate becomes sharper as the optimisation proceeds.

3. **Hard‑max sub‑gradient** – finally switch to the true max (a
   one‑hot weight vector) and continue optimisation with a much smaller
   learning‑rate schedule.  This phase removes the remaining tiny
   excess over the theoretical lower bound.

The optimiser works on the log‑softmax parametrisation
`a = softmax(x)`, which automatically enforces non‑negativity and a unit
sum.  Adam is used for all three phases, with phase‑specific learning‑rate
schedules and occasional random “jitter’’ restarts to escape shallow
local minima.  The best sequence seen during the run is returned.
"""

import time
import numpy as np

# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa                      # element‑wise product (â)²
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k  with respect to ``a``.
    ``w`` has length 2·len(a)‑1.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()

    # work in the log‑softmax (x) space
    x = np.log(a0 + 1e-12)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state (shared by all phases)
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Book‑keeping for the best solution
    # ------------------------------------------------------------------
    a = _softmax(x)
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = float(2.0 * n * conv.max())
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 5) Phase schedule (percentage of total budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * total_time   # variance flattening
    phase2_end = start_time + 0.75 * total_time   # smooth‑max
    phase3_end = deadline - 2.0                   # hard‑max sub‑gradient
    # a tiny margin before the final deadline to finish cleanly

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # -------- obtain current a and its self‑convolution -------------
        a = _softmax(x)                     # probability vector, sum = 1
        conv = _self_conv(a, fft_len_self)  # length 2·n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # -------- record improvement ------------------------------------
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # -------- decide which loss to use -------------------------------
        if now < phase1_end:
            # ---- Phase 1 – variance flattening -------------------------
            avg = 1.0 / (2 * n - 1)               # target constant convolution
            r = conv - avg                         # residual vector
            g_a = 2.0 * _gradient_wrt_a(a, r)     # ∂/∂a Σ (c‑avg)²
        elif now < phase2_end:
            # ---- Phase 2 – smooth max (log‑sum‑exp) -------------------
            # linearly increase γ from a small to a large value
            phase_frac = (now - phase1_end) / (phase2_end - phase1_end)
            gamma_start = 0.5
            gamma_end = 200.0                      # much sharper than before
            gamma = gamma_start + (gamma_end - gamma_start) * phase_frac

            c_shift = conv - np.max(conv)          # shift for numerical stability
            w = np.exp(gamma * c_shift)
            w /= w.sum()                           # probability weights
            g_a = _gradient_wrt_a(a, w)            # ∂/∂a (smooth max)
        else:
            # ---- Phase 3 – hard max (sub‑gradient) --------------------
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            g_a = _gradient_wrt_a(a, w)            # ∂/∂a max(conv)

        # -------- chain rule back to log‑weights x (softmax Jacobian) ---
        avg_g = np.dot(a, g_a)                     # ⟨a,∂F/∂a⟩
        g_x = a * (g_a - avg_g)                    # ∂F/∂x

        # -------- Adam update -------------------------------------------
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # phase‑specific learning‑rate schedule (linear decay)
        if now < phase1_end:
            lr_start, lr_end = 0.9, 0.07
        elif now < phase2_end:
            lr_start, lr_end = 0.5, 0.02
        else:
            lr_start, lr_end = 0.2, 0.001
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # -------- occasional stochastic jitter (helps escape plateaus) --
        if rng.random() < 0.015:          # 1.5 % chance per iteration
            # take a block of the current best, perturb it multiplicatively,
            # re‑normalise and jump to that region.
            width = rng.integers(1, max(2, n // 8))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.18))
            a_jit = best.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a_jit = np.clip(a_jit, 0.0, None)
            s = a_jit.sum()
            if s == 0.0:
                a_jit[:] = 1.0 / n
            else:
                a_jit /= s
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] random jitter applied")

        # -------- stagnation handling – restart near the current best -----
        if now - last_improve > 30.0:
            # add a modest Gaussian noise around the best solution and start
            # afresh; this often shakes the optimiser out of a flat region.
            noise = rng.normal(0.0, 0.4, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clip to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
