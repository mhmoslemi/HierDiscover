    rng = np.random.default_rng(seed)
    # initialization
    init_seq = globals().get('height_sequence_1')
    if init_seq is not None:
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)   # log weights
        else:
            n = 5000
            x = np.zeros(n)

    fft_len_self = _next_pow2(2 * n - 1)
    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # keep best
    a = _softmax(x)
    c = _self_conv(a, fft_len_self)
    best_score = 2.0 * n * c.max() / (a.sum() ** 2)
    best = a.copy()
    # time budget
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start = time.monotonic()
    total = deadline - start
    # define phase thresholds
    phase1_end = start + 0.2 * total
    phase2_end = start + 0.8 * total   # Phase2 covers up to 80%
    # start
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        # compute a and conv
        a = _softmax(x)
        c = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now-start:6.2f}s] new best {best_score:.6f}")

        # choose gradient
        if now < phase1_end:
            # variance flatten
            avg = 1.0 / (2 * n - 1)
            r = c - avg
            w = r * 2.0  # gradient is 2 * grad_wrt_a(r)

            gamma_end = 200.0
            gamma = gamma_start + (gamma_end - gamma_start) * phase_frac
            c_shift = c - c.max()

"""
Propose a step‑function that (approximately) minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The method is a three‑phase mirror‑descent on the soft‑max
parameterisation `a = softmax(x)`:

1. **Variance‑flattening** – minimise Σ (conv–average)² (the same loss as
   the reference code).  This quickly drives the self‑convolution towards
   a flat shape.

2. **Smoothed‑max** – for the bulk of the budget we use a log‑sum‑exp
   surrogate of the true max with a linearly increasing temperature
   (`γ` from 1 to 200).  The gradient of this surrogate is computed by
   convolving the current sequence with the soft‑max weights; Adam updates
   adapt the step‑size automatically.

3. **Hard‑max refinement** – in the final stage we replace the surrogate
   by an exact sub‑gradient of the max (weights are the arg‑max
   locations).  A small learning‑rate finish fine‑tunes the solution.

Stagnation (no improvement for >30 s) triggers a random restart around
the current best point, and occasional “block jitter’’ perturbations
help escaping shallow minima.  The routine respects a 1000 s time budget
(5 s safety margin) and returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    ``a`` is a 1‑D array of length n, ``w`` has length 2·n‑1.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that gives a
    low value of `evaluate_sequence`.  The search runs for the given time
    budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Initialise from the best known construction if available
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)          # log‑weights for softmax
        else:
            n = 5000
            x = np.zeros(n, dtype=np.float64)
    else:
        n = 5000
        x = np.zeros(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) Pre‑compute FFT length for the self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Adam optimiser state (shared across all phases)
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 3) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4) Keep track of the best solution observed so far
    # ------------------------------------------------------------------
    a = _softmax(x)
    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * c.max() / (a.sum() ** 2)
    last_improve = time.monotonic()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase split (percentage of the total budget)
    # ------------------------------------------------------------------
    #   0‑20 % : variance flattening
    #  20‑80 % : smoothed‑max (γ ramps 1→200)
    #  80‑100 %: hard‑max refinement
    phase1_end = start_time + 0.20 * total_time
    phase2_end = start_time + 0.80 * total_time

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------------
        # current a and self‑convolution
        # ----------------------------------------------------------------
        a = _softmax(x)               # probability vector, sum≈1
        c = _self_conv(a, fft_len_self)

        # ----------------------------------------------------------------
        # exact score (the quantity we ultimately care about)
        # ----------------------------------------------------------------
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # ----------------------------------------------------------------
        # record improvement
        # ----------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------------
        # choose loss and its gradient
        # ----------------------------------------------------------------
        if now < phase1_end:
            # ---- Phase 1 : minimise variance of the convolution ----
            avg = 1.0 / (2 * n - 1)    # because sum(a)=1
            r = c - avg
            # gradient of Σ (c‑avg)²  = 2·_gradient_wrt_a(r)
            g_a = 2.0 * _gradient_wrt_a(a, r)
        elif now < phase2_end:
            # ---- Phase 2 : smooth approximation of max ----
            phase_frac = (now - phase1_end) / (phase2_end - phase1_end)
            gamma_start, gamma_end = 1.0, 200.0
            gamma = gamma_start + (gamma_end - gamma_start) * phase_frac

            # stable log‑sum‑exp weighting
            c_shift = c - np.max(c)               # max becomes 0
            w = np.exp(gamma * c_shift)           # always ≤1, max entry =1
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                # fall back to hard max if numerical trouble
                w = np.zeros_like(c)
                w[np.argmax(c)] = 1.0
            else:
                w = w / w_sum
            g_a = _gradient_wrt_a(a, w)
        else:
            # ---- Phase 3 : exact sub‑gradient of the max ----
            max_val = c.max()
            # treat all indices within a tiny relative tolerance as active
            tol = max_val * 1e-8
            active = np.where(c >= max_val - tol)[0]
            if active.size == 0:
                active = np.array([np.argmax(c)], dtype=int)
            w = np.zeros_like(c)
            w[active] = 1.0 / active.size          # normalise (optional)
            g_a = _gradient_wrt_a(a, w)

        # ----------------------------------------------------------------
        # Chain rule to get gradient w.r.t. log‑weights x (softmax Jacobian)
        # ----------------------------------------------------------------
        avg_g = np.dot(a, g_a)                     # ⟨a,∂F/∂a⟩
        g_x = a * (g_a - avg_g)                    # ∂F/∂x

        # ----------------------------------------------------------------
        # Adam step on the log‑weights
        # ----------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (linear decay)
        lr_start, lr_end = 0.6, 0.01
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----------------------------------------------------------------
        # Stagnation handling – if no improvement for 30 s, restart near best
        # ----------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # ----------------------------------------------------------------
        # Occasional global jitter to escape shallow minima
        # ----------------------------------------------------------------
        if rng.random() < 0.02:
            pert = _random_block_perturb(best, rng)
            x = np.log(pert + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clip to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
