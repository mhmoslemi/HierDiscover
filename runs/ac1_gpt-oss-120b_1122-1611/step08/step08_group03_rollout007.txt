    n = a.size
    # compute conv via FFT
    fft_len = _next_pow2(3*n)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
    max_val = conv.max()
    max_idx = int(conv.argmax())
    # compute gradient: g_i = 2*a[max_idx - i] if in range else 0
    grad = np.zeros_like(a)
    i_start = max(0, max_idx - (n-1))
    i_end = min(n-1, max_idx)
    # vector of i indices
    idx = np.arange(i_start, i_end+1)
    grad[idx] = a[max_idx - idx]
    grad *= 2.0
    # update
    a_new = a - lr * grad
    # projection onto simplex
    a_new = _project_onto_simplex(a_new)
    return a_new, max_val

def lp_step(a, M, t=0.1):
    n = len(a)
    # Construct C matrix
    C = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        C[i:i+n, i] = a  # each column i gets a at rows starting i
    # Define variable
    b = cp.Variable(n, nonneg=True)
    constraints = [C @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    # solve
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)  # maybe limit
    if b.value is None:
        # fallback: return a unchanged
        return a
    b_val = np.asarray(b.value).flatten()
    # Scale to sum 1
    sum_b = b_val.sum()
    if sum_b <= 1e-12:
        # fallback to a
        return a
    g = b_val / sum_b
    # convex combination
    new_a = (1.0 - t) * a + t * g
    # due to convex combination sum = 1, but might have rounding drift; project onto simplex to ensure
    new_a = _project_onto_simplex(new_a)
    return new_a

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σx = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to avoid numerical drift
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _conv_max(a: np.ndarray) -> float:
    """Exact convolution peak (no clipping, Σa = 1)."""
    conv = np.convolve(a, a)
    return conv.max()


def _score_from_a(a: np.ndarray) -> float:
    """Exact objective value for a (Σa = 1, all entries ≤1000)."""
    return 2.0 * a.size * _conv_max(a)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_phase(a: np.ndarray,
                phase_deadline: float,
                target_score: float = 1.5025) -> tuple[np.ndarray, float]:
    """
    Smooth‑max Adam optimisation.
    Returns the best vector seen together with its exact score.
    """
    n = a.size
    fft_len = _next_pow2(3 * n)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # annealing schedules (more aggressive than the reference script)
    TAU_START, TAU_END = 0.5, 5e-08          # temperature
    LR_START, LR_END = 0.5, 5e-05          # learning rate

    best = a.copy()
    best_score = _score_from_a(best)

    it = 0
    while time.monotonic() < phase_deadline:
        now = time.monotonic()
        frac = (now - (phase_deadline - (phase_deadline - (phase_deadline - phase_deadline)))) / (
            phase_deadline - (phase_deadline - (phase_deadline - phase_deadline))
        )  # always 0‑1, just a dummy to keep compatibility

        # linear annealing
        tau = TAU_START * (1.0 - frac) + TAU_END * frac
        lr = LR_START * (1.0 - frac) + LR_END * frac

        # --- convolution via FFT ---
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # --- smooth‑max weighting ---
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)   # prevent overflow
        w = exp_shifted / exp_shifted.sum()          # soft‑max, sums to 1

        # --- gradient of the surrogate (correlation) ---
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        start = n - 1
        grad = (2.0 / tau) * corr_full[start:start + n]

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        it += 1

        # occasional exact evaluation
        if it % 300 == 0:
            sc = _score_from_a(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                elapsed = now - (phase_deadline - (phase_deadline - (phase_deadline - phase_deadline)))
                print(f"[{elapsed:6.2f}s] Adam improvement → {best_score:.6f} (iter={it})")
                if best_score <= target_score:
                    print("[early stop] target reached during Adam phase")
                    break

    return best, best_score


def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         start_temp: float = 2e-2,
                         end_temp: float = 1e-7,
                         target_score: float = 1.5025) -> tuple[np.ndarray, float]:
    """
    Stochastic local refinement with a (slowly) decreasing temperature.
    Tiny mass‑transfer moves are proposed; worsening moves are occasionally
    accepted according to the current temperature.
    """
    n = a.size
    rng = np.random.default_rng()
    cur = a.copy()
    cur_score = _score_from_a(cur)
    best = cur.copy()
    best_score = cur_score

    it = 0
    while time.monotonic() < deadline:
        # linear temperature schedule
        now = time.monotonic()
        temp = start_temp * (deadline - now) / (deadline - (deadline - (deadline - deadline))) + \
               end_temp * (now - (deadline - (deadline - (deadline - deadline)))) / \
               (deadline - (deadline - (deadline - deadline)))
        # pick source/target indices
        i = rng.integers(0, n)
        if cur[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # fraction of source mass to move (log‑uniform 0.1 % – 5 %)
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = cur[i] * frac_move

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _score_from_a(cand)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cand_score - cur_score) / temp):
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best = cand.copy()
                best_score = cand_score
                elapsed = now - (deadline - (deadline - (deadline - deadline)))
                print(f"[{elapsed:6.2f}s] SA new best → {best_score:.6f} (move #{it})")
                if best_score <= target_score:
                    print("[early stop] target reached during SA phase")
                    break

        # occasional jitter to keep diversity
        it += 1
        if it % 4000 == 0:
            width = rng.integers(1, max(2, n // 50))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            jitter = cur.copy()
            jitter[start_idx:start_idx + width] *= factor
            cur = _project_onto_simplex(jitter)

    return best, best_score


def _hardmax_refine(a: np.ndarray,
                    deadline: float,
                    target_score: float = 1.5025) -> tuple[np.ndarray, float]:
    """
    Few steps of sub‑gradient descent on the *exact* max‑convolution term.
    Uses a decreasing step size and projects onto the simplex after each step.
    """
    n = a.size
    cur = a.copy()
    cur_score = _score_from_a(cur)
    best = cur.copy()
    best_score = cur_score

    it = 0
    while time.monotonic() < deadline:
        conv = np.convolve(cur, cur)
        max_idx = int(conv.argmax())
        # gradient = 2 * shift(cur, max_idx)
        grad = np.zeros_like(cur)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = cur[max_idx - idx]
        grad *= 2.0

        # step size schedule (simple 1/√k)
        it += 1
        lr = 0.05 / math.sqrt(it)

        cand = _project_onto_simplex(cur - lr * grad)
        cand_score = _score_from_a(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best = cand.copy()
                best_score = cand_score
                elapsed = time.monotonic()
                print(f"[{elapsed:6.2f}s] hard‑max improvement → {best_score:.6f} (iter={it})")
                if best_score <= target_score:
                    print("[early stop] target reached during hard‑max phase")
                    break
        # else keep current point (no line‑search for simplicity)

    return best, best_score

# Main entry point

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.
    The routine performs several independent restarts with Adam optimisation,
    followed by simulated‑annealing local refinement and a final hard‑max
    gradient finetuning.
    """
    rng = np.random.default_rng(seed)

    # ---- 0) time bookkeeping -------------------------------------------------
    start_time = time.monotonic()
    # leave a 5‑second safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ---- 1) bookkeeping of the global best ----------------------------------
    global_best = None
    global_best_score = float('inf')
    target = 1.5025

    # ---- 2) decide on a few restart configurations ---------------------------
    # we use three different lengths; larger lengths give a larger search space
    # but are slower, therefore we keep them modest.
    lengths = [1024, 2048, 4096]
    # allocate roughly equal time to each restart (except the final refinement)
    remaining_budget = deadline - time.monotonic()
    n_restarts = len(lengths)
    per_restart = remaining_budget * 0.58 / n_restarts   # 58 % for Adam + SA

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # ---- initialise a random Dirichlet start (sparse) --------------------
        alpha = 0.5  # makes the vector moderately sparse
        a0 = rng.dirichlet([alpha] * n).astype(np.float64)  # already sums to 1
        # ---- 3) Adam optimisation (70 % of the per‑restart budget) ----------
        phase1_deadline = time.monotonic() + per_restart * 0.70
        a_adam, score_adam = _adam_phase(a0, phase1_deadline, target_score=target)

        # ---- 4) Simulated‑annealing local refinement (30 % of the budget) -----
        phase2_deadline = time.monotonic() + per_restart * 0.30
        a_sa, score_sa = _simulated_annealing(
            a_adam, phase2_deadline, start_temp=2e-2, end_temp=1e-7, target_score=target
        )

        # keep the best of this restart
        if score_sa < global_best_score - 1e-12:
            global_best = a_sa.copy()
            global_best_score = score_sa
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] restart n={n} best = {global_best_score:.6f}")

        if global_best_score <= target:
            break

    # ---- 5) final hard‑max refinement on the overall best --------------------
    if global_best is None:
        # fallback – uniform vector (should never happen)
        global_best = np.full(4096, 1.0 / 4096, dtype=np.float64)

    final_deadline = deadline
    a_final, score_final = _hardmax_refine(
        global_best, final_deadline, target_score=target
    )
    if score_final < global_best_score - 1e-12:
        global_best = a_final
        global_best_score = score_final

    # -------------------------------------------------------------------------
    # Return the best sequence – clip to the bounds used by the evaluator.
    # -------------------------------------------------------------------------
    best_clipped = np.clip(global_best, 0.0, 1000.0)
    final_score = _score_from_a(best_clipped)  # quick recomputation
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_clipped]
