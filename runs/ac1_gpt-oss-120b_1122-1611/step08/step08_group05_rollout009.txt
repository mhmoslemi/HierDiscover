def simplex_projection(v):
    # Project vector v onto probability simplex sum=1, v >=0.
    # Use algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    end = start + n

    out = np.clip(out, 0.0, None)
    s = out.sum()

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projects v onto the probability simplex {x>=0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector")
    # Algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.arange(1, len(v) + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho_arr[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

search for a step‑function that minimises ``evaluate_sequence``.

The algorithm has three stages:

1️⃣  Smoothed‑max (log‑sum‑exp) optimisation with Adam.
    A softmax parametrisation (log‑weights) guarantees the simplex
    constraint.  The surrogate becomes sharper (γ→∞) as the run
    proceeds.

2️⃣  Sub‑gradient descent on the *exact* max‑convolution.
    The active set of indices attaining the maximum is used as a
    probability vector; the resulting gradient is applied with a
    linearly decaying step‑size and the iterate is projected onto the
    probability simplex.

3️⃣  A short stochastic hill‑climb that perturbs the current best
    candidate by a random block‑scale or a tiny pair‑wise transfer
    and keeps any improvement.

All helper functions are defined at module level, no closures are used.
The routine respects a 5‑second safety margin and prints progress.
"""


import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of ``∑_k w_k (a*a)_k`` with respect to ``a``.
    ``w`` must have length ``2·len(a)-1`` (the length of a*a).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to the unit simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 8))           # ≤≈12 % of the vector
    start = rng.integers(0, n - width + 1)

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implementation of Duchi‑Shalev‑Shwartz‑Singer‑Chandra (2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        m_hat = m_adam / (1 - beta1 ** t_it)
        v_hat = v_adam / (1 - beta2 ** t_it)

        lr = lr_start * (1 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # exact objective (using the same FFT‑based max – virtually exact)
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase‑1 new best = {best_score:.6f}")

        # stagnation handling – soft restart after 60 s without improvement
        if now - last_improve > 60.0:
            noise = rng.normal(0.0, 0.8, size=n)
            x = np.log(best_a + 1e-12) + noise
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_it = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase‑1 restart (stagnation) – best {best_score:.6f}")

        # occasional global jitter (helps escape plateaus)
        if rng.random() < 0.02:
            jittered = _random_block_perturb(best_a, rng)
            x = np.log(jittered + 1e-12)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            t_it = 0

    # ------------------------------------------------------------------
    # 5) Phase 2 – Sub‑gradient descent on the *exact* max‑convolution
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.9 * total_time
    # start from the best vector obtained in Phase 1
    a = best_a.copy()
    step_start = 0.8
    step_end = 0.005

    while time.monotonic() < phase2_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # fraction of Phase 2 (0→1)
        frac = (elapsed - 0.6 * total_time) / (0.3 * total_time)
        frac = max(0.0, min(1.0, frac))

        # full self‑convolution (FFT – still cheap)
        c_full = _self_conv_fft(a, fft_len_self)
        max_c = c_full.max()

        # active set (indices attaining the maximum, within tolerance)
        active = np.isclose(c_full, max_c, atol=1e-12)
        w = active.astype(float)
        if w.sum() == 0.0:                     # fall‑back (should never happen)
            w = np.ones_like(c_full) / c_full.size
        else:
            w /= w.sum()

        # sub‑gradient of the true max
        g_a = _gradient_wrt_a(a, w)

        # decreasing step size
        step = step_start * (1 - frac) + step_end * frac
        a = a - step * g_a
        a = _simplex_projection(a)

        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{elapsed:6.2f}s] Phase‑2 new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Phase 3 – Simple stochastic hill‑climb (exact evaluator)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # propose a small random perturbation of the current best
        cand = _random_block_perturb(best_a, rng)

        # with some probability also perform a tiny pair‑wise transfer
        if rng.random() < 0.3:
            i, j = rng.integers(0, n, size=2)
            if i != j and best_a[i] > 0.0:
                transfer = rng.random() * best_a[i] * 0.1
                cand[i] -= transfer
                cand[j] += transfer
                cand = _simplex_projection(cand)

        # evaluate with the official (exact) routine
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score - 1e-12:
            best_score = score
            best_a = cand.copy()
            now = time.monotonic()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Phase‑3 new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
