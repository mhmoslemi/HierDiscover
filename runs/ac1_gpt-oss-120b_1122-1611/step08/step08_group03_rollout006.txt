def _adam_optimize(a_init, time_left, start_time):
    # runs Adam until current time exceeds target: target = start_time + time_left
    # Return best a and best score.
    a = a_init.copy()
    n = a.size
    fft_len = _next_pow2(2*n-1)  # enough for convolution
    best_local = a.copy()
    best_local_score = _objective(a)
    # Adam state:
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1,beta2 = 0.9,0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0
    # schedule:
    # We can base iteration number for tau schedule using fraction inside this phase.
    # We'll compute start_time_phase = time.monotonic()
    # For each iteration, compute frac = (now - start_time_phase) / time_left

    start_phase = time.monotonic()

    while time.monotonic() < start_time + time_left:
        now = time.monotonic()
        elapsed_phase = now - start_phase
        frac = elapsed_phase / time_left
        # Temperature schedule: do from 0.3 to 5e-5
        tau = 0.30 * (1 - frac) + 5e-05 * frac
        lr = 0.25 * (1 - frac) + 5e-04 * frac

        # compute convolution
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]

        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)
        w = exp_shifted / exp_shifted.sum()

        # gradient
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # evaluate occasionally
        if np.random.rand() < 0.005:  # approximate every 200 steps
            sc = _objective(a)
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local = a.copy()
                # print progress
                total_elapsed = now - start_time
                print(f"[{total_elapsed:6.2f}s] Adam interim best = {sc:.6f}")

        # early termination inside: if best_local_score below target maybe break.
        if best_local_score <= 1.5025:
            break

    return best_local, best_local_score

def _simulated_annealing(a_init, time_remaining, start_time):
    a = a_init.copy()
    n = a.size
    best_local = a.copy()
    best_local_score = _objective(a)
    # Annealing parameters
    start_temp = 0.02
    end_temp = 1e-5
    start = time.monotonic()
    # We can predefine schedule: T = start_temp * (1 - frac) + end_temp * frac
    # We'll decide move type: occasionally (every few steps) do block reallocation, otherwise tiny move.
    move_counter = 0
    while time.monotonic() < start_time + time_remaining:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / time_remaining
        T = start_temp * (1 - frac) + end_temp * frac

        # random choose move type
        if np.random.rand() < 0.1:
            # block reallocation
            max_width = max(2, n // 30)
            w = rng.integers(2, max_width + 1)
            start_idx = rng.integers(0, n - w + 1)
            block = a[start_idx:start_idx + w]
            block_sum = block.sum()
            # propose new random distribution within block
            # small Dirichlet
            new_block = rng.dirichlet(np.full(w, 0.5)) * block_sum
            a_candidate = a.copy()
            a_candidate[start_idx:start_idx + w] = new_block
        else:
            # tiny mass transfer
            i = rng.integers(0, n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            # fraction move between 0.5% and 20%
            frac_move = np.exp(rng.uniform(np.log(0.005), np.log(0.2)))
            delta = a[i] * frac_move
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta

        # evaluate candidate
        sc = _objective(a_candidate)
        # acceptance
        if sc < best_local_score - 1e-12:
            accept = True
        else:
            # compute acceptance probability
            delta_score = sc - best_local_score
            accept = np.exp(-delta_score / max(T, 1e-12)) > rng.random()
        if accept:
            a = a_candidate
            # update best if improved
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local = a.copy()
                total_elapsed = now - start_time
                print(f"[{total_elapsed:6.2f}s] SA new best = {sc:.6f}")

        move_counter += 1
        # occasional jitter injection
        if move_counter % 5000 == 0:
            width = rng.integers(1, max(2, n // 50))
            idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.02))
            a_j = a.copy()
            a_j[idx:idx+width] *= factor
            a_j = _project_onto_simplex(a_j)
            # acceptance? just set a = a_j
            a = a_j
    return best_local, best_local_score

def _max_conv_and_grad(a):
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    max_idxs = np.where(np.abs(conv - max_val) < 1e-12)[0] # tolerance
    grad = np.zeros_like(a)
    for idx in max_idxs:
        s = idx - (n - 1)
        d = abs(s)
        if d == 0:
            grad += 2 * a
        else:
            # forward: a shifted left by d
            forward = np.concatenate((a[d:], np.zeros(d)))
            backward = np.concatenate((np.zeros(d), a[:n - d]))
            grad += forward + backward
    return max_val, grad

def _adam_optimize(a_init, rng, start_time, deadline):
    a = a_init.copy()
    n = a.size
    L = _next_pow2(2 * n - 1)
    # Adam states
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    best = a.copy()
    best_score = _objective(a)
    iter_cnt = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        # schedule
        tau = 0.30 * (1.0 - frac) + 5e-05 * frac
        lr = 0.25 * (1.0 - frac) + 5e-04 * frac

        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)
        w = exp_shifted / exp_shifted.sum()
        w_fft = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=L)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        iter_cnt += 1

        # periodic exact objective evaluation (cheap via _objective)
        if iter_cnt % 200 == 0:
            sc = _objective(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Adam new best = {best_score:.6f}")

        if best_score <= 1.5025:
            break
    return best, best_score

def _simulated_annealing(a_init, rng, start_time, deadline):
    a = a_init.copy()
    n = a.size
    best = a.copy()
    best_score = _objective(a)
    start_phase = time.monotonic()
    phase_len = deadline - start_phase
    move_counter = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_phase) / phase_len
        T = 0.02 * (1 - frac) + 1e-5 * frac

        # compute current conv and max shift to guide targeted moves
        conv = _full_convolution(a)
        max_idx = conv.argmax()
        s = max_idx - (n - 1)
        d = abs(s)

        # choose move type
        if rng.random() < 0.5:
            # target mass transfer
            if d == 0:
                # pick random source
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                while i == j:
                    j = rng.integers(0, n)
            else:
                # pick source from participating indices
                if s >= 0:
                    i = rng.integers(0, n - d)
                else:
                    i = rng.integers(d, n)
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
            # move fraction larger maybe
            frac_move = np.exp(rng.uniform(np.log(0.01), np.log(0.3)))
            delta = a[i] * frac_move
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
        else:
            # random block reshape
            max_width = max(2, n // 30)
            w = rng.integers(2, max_width + 1)
            start_idx = rng.integers(0, n - w + 1)
            block = a[start_idx:start_idx + w]
            block_sum = block.sum()
            # Dirichlet with alpha=0.3 to allow sparsity
            new_block = rng.dirichlet(np.full(w, 0.3)) * block_sum
            a_candidate = a.copy()
            a_candidate[start_idx:start_idx + w] = new_block

        sc = _objective(a_candidate)

        if sc < best_score:
            accept = True
        else:
            accept = np.exp(-(sc - best_score) / max(T, 1e-12)) > rng.random()
        if accept:
            a = a_candidate
            if sc < best_score:
                best_score = sc
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] SA new best = {best_score:.6f}")

        move_counter += 1
        if move_counter % 4000 == 0:
            # jitter smoothing
            width = rng.integers(1, max(2, n // 50))
            idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.02))
            a_j = a.copy()
            a_j[idx:idx + width] *= factor
            a_j = _project_onto_simplex(a_j)
            # evaluate jitter
            scj = _objective(a_j)
            if scj < best_score:
                a = a_j
                best = a_j.copy()
                best_score = scj
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] jitter improvement = {best_score:.6f}")
            else:
                # maybe accept with low prob
                if np.exp(-(scj - best_score) / max(T, 1e-12)) > rng.random():
                    a = a_j

        if best_score <= 1.5025:
            break

    return best, best_score

def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # choose length n
    n = 8192
    # create initial a
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        sum_prev = prev.sum()
        if sum_prev > 0:
            prev = prev / sum_prev
        # embed
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        embed_len = min(prev.size, n)
        mix = 0.6
        a0[:embed_len] = mix * prev[:embed_len] + (1 - mix) * a0[:embed_len]
        a0 = _project_onto_simplex(a0)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    best = a0.copy()
    best_score = _objective(best)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # Phase 1: Adam optimisation (multiple restarts)
    phase1_deadline = start_time + 0.60 * (deadline - start_time)
    # run a few restarts
    restarts = 4
    for r in range(restarts):
        # generate a random perturbation of the base vector
        a_start = a0 + rng.normal(scale=0.02, size=n)
        a_start = np.maximum(a_start, 0.0)
        a_start = _project_onto_simplex(a_start)
        # allocate time slice
        slice_deadline = start_time + (r + 1) / restarts * (phase1_deadline - start_time)
        if slice_deadline > deadline:
            slice_deadline = deadline
        cand, cand_score = _adam_optimize(a_start, rng, start_time, slice_deadline)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] post-Adam best = {best_score:.6f}")
        if best_score <= 1.5025:
            break

    # Phase 2: subgradient descent
    phase2_deadline = start_time + 0.85 * (deadline - start_time)
    cand, cand_score = _subgradient_descent(best, rng, start_time, phase2_deadline)
    if cand_score < best_score:
        best_score = cand_score
        best = cand.copy()
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] after subgradient best = {best_score:.6f}")

    # Phase 3: simulated annealing / local refinement
    cand, cand_score = _simulated_annealing(best, rng, start_time, deadline)
    if cand_score < best_score:
        best_score = cand_score
        best = cand.copy()
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] final best = {best_score:.6f}")

    # Ensure compliance with evaluator limits
    best_clipped = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best_clipped.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_clipped]

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

The search consists of three phases:

1. **Adam optimisation with a smooth‑max surrogate** – multiple short
   restarts from a perturbed uniform start, each run until its time slice
   expires.  The surrogate replaces the hard max by a temperature‑controlled
   log‑sum‑exp; gradients are obtained via FFT and the iterate is projected
   onto the probability simplex.

2. **Sub‑gradient descent on the exact max** – using the true max‐convolution
   gradient (the sum of the two shifted copies of the vector that attain the
   current maximum).  A linearly decaying step‑size is employed together with
   simplex projection.

3. **Simulated‑annealing refinement** – alternating between targeted mass
   transfers (aimed at the shift that attains the current maximum) and random
   block re‑distribution moves.  A temperature schedule controls occasional
   uphill moves, and periodic jitter helps escape plateaus.

All phases share the same length (default 8192, larger than the previous
construction) and maintain the normalisation ∑a_i = 1, which makes the
objective scale‑invariant.  The best vector found is finally clipped to the
evaluator limits and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (aperiodic) convolution a * a using an FFT of sufficient length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` (fast version)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * a.shape[0] * max_b / (s * s))


def _max_conv_and_grad(a: np.ndarray):
    """Return (max_convolution, gradient of that max w.r.t. a)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_idx = int(conv.argmax())
    shift = max_idx - (n - 1)          # signed shift producing the max
    d = abs(shift)
    if d == 0:
        grad = 2.0 * a
    else:
        # forward part (a shifted by +d)
        forward = np.concatenate((a[d:], np.zeros(d)))
        # backward part (a shifted by -d)
        backward = np.concatenate((np.zeros(d), a[: n - d]))
        grad = forward + backward
    return conv[max_idx], grad


def _adam_optimize(a_init: np.ndarray,
                   rng: np.random.Generator,
                   start_time: float,
                   deadline: float):
    """Smooth‑max Adam optimisation for a fixed time slice."""
    a = a_init.copy()
    n = a.size
    L = _next_pow2(2 * n - 1)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0

    best = a.copy()
    best_score = _objective(a)

    it = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)

        tau = 0.30 * (1.0 - frac) + 5e-05 * frac
        lr = 0.25 * (1.0 - frac) + 5e-04 * frac

        a_fft = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)
        w = exp_shifted / exp_shifted.sum()

        w_fft = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=L)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        it += 1
        # occasional exact evaluation (cheap)
        if it % 200 == 0:
            sc = _objective(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Adam interim best = {best_score:.6f}")

        if best_score <= 1.5025:
            break

    return best, best_score


def _subgradient_descent(a_init: np.ndarray,
                         rng: np.random.Generator,
                         start_time: float,
                         deadline: float):
    """Sub‑gradient descent on the exact max‑convolution."""
    a = a_init.copy()
    best = a.copy()
    best_score = _objective(a)
    n = a.shape[0]

    phase_start = time.monotonic()
    phase_len = deadline - phase_start
    it = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - phase_start) / phase_len
        lr = 1.0 * (1.0 - frac) + 0.001 * frac     # linear decay

        _, grad = _max_conv_and_grad(a)
        a = a - lr * grad
        a = _project_onto_simplex(a)

        it += 1
        if it % 250 == 0:
            sc = _objective(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Subgrad interim best = {best_score:.6f}")
            if best_score <= 1.5025:
                break

    return best, best_score


def _simulated_annealing(a_init: np.ndarray,
                         rng: np.random.Generator,
                         start_time: float,
                         deadline: float):
    """Simulated‑annealing refinement with targeted and block moves."""
    a = a_init.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    phase_start = time.monotonic()
    phase_len = deadline - phase_start
    move_counter = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - phase_start) / phase_len
        T = 0.02 * (1.0 - frac) + 1e-5 * frac   # temperature schedule

        # current convolution to locate the active shift
        conv = _full_convolution(a)
        max_idx = int(conv.argmax())
        shift = max_idx - (n - 1)
        d = abs(shift)

        # ----- propose a move ------------------------------------------------
        if rng.random() < 0.6:          # targeted mass transfer
            if d == 0:
                i = rng.integers(0, n)
            else:
                if shift >= 0:
                    i = rng.integers(0, n - d)
                else:
                    i = rng.integers(d, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            # move a noticeable but not too large fraction
            frac_move = np.exp(rng.uniform(np.log(0.01), np.log(0.3)))
            delta = a[i] * frac_move
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
        else:                           # random block reshuffle
            max_width = max(2, n // 30)
            w = rng.integers(2, max_width + 1)
            start_idx = rng.integers(0, n - w + 1)
            block = a[start_idx:start_idx + w]
            block_sum = block.sum()
            # small Dirichlet to allow sparse fresh shapes
            new_block = rng.dirichlet(np.full(w, 0.3)) * block_sum
            a_candidate = a.copy()
            a_candidate[start_idx:start_idx + w] = new_block

        cand_score = _objective(a_candidate)

        # ----- acceptance test ---------------------------------------------
        if cand_score < best_score:
            accept = True
        else:
            accept = np.exp(-(cand_score - best_score) / max(T, 1e-12)) > rng.random()

        if accept:
            a = a_candidate
            if cand_score < best_score:
                best_score = cand_score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] SA new best = {best_score:.6f}")

        move_counter += 1

        # occasional jitter to shake the system
        if move_counter % 4000 == 0:
            width = rng.integers(1, max(2, n // 50))
            idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.02))
            a_j = a.copy()
            a_j[idx:idx + width] *= factor
            a_j = _project_onto_simplex(a_j)
            jitter_score = _objective(a_j)
            if jitter_score < best_score:
                a = a_j
                best = a_j.copy()
                best_score = jitter_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] jitter improvement = {best_score:.6f}")
            else:
                if np.exp(-(jitter_score - best_score) / max(T, 1e-12)) > rng.random():
                    a = a_j

        if best_score <= 1.5025:
            break

    return best, best_score
