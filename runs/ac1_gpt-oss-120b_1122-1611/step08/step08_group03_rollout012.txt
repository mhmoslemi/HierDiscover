```
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

def solve_g(f, M):
    n = len(f)
    # Build sparse matrix A of shape (2n-1,n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows i to i+n-1 with value f[i]
        rows.extend(range(i, i+n))
        cols.extend([i]*n)
        data.extend([f[i]]*n)
    A = sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    # Right-hand side vector
    b = np.full(2*n-1, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = res.x
    else:
        # fallback
        g = np.full_like(f, 1.0/f.size)
    return g
```

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm (≈1000 s time budget)

0. **Initialisation**
   - Use the historic construction ``height_sequence_1`` if available,
     otherwise start from a uniform probability simplex.
   - Normalise the vector to sum = 1 (the objective is scale‑invariant).

1. **Random exploration**
   - Sample a few Dirichlet‑distributed vectors and keep the best
     (quick cheap way to escape a possible very poor start).

2. **Linear‑programming refinement**
   - Repeatedly solve the LP
        maximise   ∑ g_j
        subject to (f * g)_k ≤ M   for a selected set of k,
                   g_j ≥ 0,
     where `f` is the current candidate, `M = max(f*f)`.
   - Only the indices with the largest convolution values are used
     (at most ``MAX_ACTIVE`` rows) – this keeps the LP small.
   - The LP returns a direction `g`; we normalise it to sum = 1,
     then mix it with the current `f`:
        f ← (1‑t)·f + t·g   with a mixing weight `t` that decays.
   - After mixing we project back onto the simplex.
   - The exact score is evaluated each iteration; the best seen
     vector is retained.

3. **Stochastic local polishing**
   - Tiny mass‑transfer moves between random indices are accepted
     only if they improve the exact score (standard “hill‑climbing”).
   - Occasionally a small jitter is applied to avoid stagnation.

The whole routine stops a few seconds before the supplied budget
so the harness can safely read the result.  The final sequence is
clipped to the limits of ``evaluate_sequence`` before returning.
"""

import time
import numpy as np
from scipy.optimize import linprog

# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a 1‑D vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _build_active_constraint_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build a dense matrix A such that (A @ g)[k] = (f * g)[active_idx[k]].
    Only rows for the supplied active indices are created.
    """
    n = f.shape[0]
    m = active_idx.shape[0]
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        A[row, i_low:i_high + 1] = f[i_low:i_high + 1]
    return A


def _solve_lp_direction(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve the linear programme that finds a direction g which maximises
    its total mass while respecting (f * g)_k ≤ M for the chosen rows.
    Returns the optimal g (unnormalised) or ``None`` on failure.
    """
    A = _build_active_constraint_matrix(f, active_idx)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(f.shape[0], dtype=np.float64)          # maximise sum(g) == minimise -sum(g)
    bounds = [(0.0, None)] * f.shape[0]

    # Using the high‑performance HiGHS solver (available in recent SciPy).
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if result.success:
        return result.x
    return None


def _mix_vectors(a: np.ndarray, b: np.ndarray, t: float) -> np.ndarray:
    """Linear mix of two probability vectors followed by simplex projection."""
    mixed = (1.0 - t) * a + t * b
    mixed = np.maximum(mixed, 0.0)
    mixed = _project_onto_simplex(mixed)
    return mixed

    random_phase_deadline = start_time + 30.0  # first 30 s for random tries
    while time.monotonic() < random_phase_deadline and time.monotonic() < deadline:
        cand = rng.gamma(shape=1.0, scale=1.0, size=n)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] random improvement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase – Linear‑Programming refinement
    # ------------------------------------------------------------------
    # Parameters
    MAX_ITER_LP = 30           # safety cap; loop will stop earlier if time runs out
    MAX_ACTIVE = 500           # at most this many constraints are fed to LP
    mix_t = 0.5                # initial mixing weight
    mix_decay = 0.93           # geometric decay per iteration
    iter_lp = 0

    while time.monotonic() < deadline and iter_lp < MAX_ITER_LP:
        iter_lp += 1
        # current candidate
        f = best.copy()

        # convolution of f with itself and current bound
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # Select active rows – the indices with the largest convolution values
        # (take up to MAX_ACTIVE rows)
        if conv_ff.size <= MAX_ACTIVE:
            active_idx = np.arange(conv_ff.size, dtype=np.int64)
        else:
            # argpartition gives the indices of the largest entries efficiently
            top_idx = np.argpartition(-conv_ff, MAX_ACTIVE)[:MAX_ACTIVE]
            active_idx = np.sort(top_idx)  # sorting is not necessary but keeps order stable

        # Solve the LP for the direction g
        g = _solve_lp_direction(f, M, active_idx)
        if g is None:
            # LP failed – fall back to a tiny random perturbation and continue
            g = rng.gamma(1.0, 1.0, size=n)
            g_sum = g.sum()
            if g_sum > 0.0:
                g /= g_sum
            else:
                g = np.full_like(g, 1.0 / n)

        # Normalise g to the simplex
        g = _project_onto_simplex(g)

        # Mix the new direction into the current solution
        new_candidate = _mix_vectors(f, g, mix_t)

        # Evaluate exact score
        new_score = evaluate_sequence(new_candidate.tolist())
        elapsed = time.monotonic() - start_time
        if new_score < best_score - 1e-12:
            best = new_candidate.copy()
            best_score = new_score
            print(f"[{elapsed:6.2f}s] LP iteration {iter_lp} improvement => {best_score:.6f}")
            # optional early stop if we have already beaten the target
            if best_score <= 1.5025:
                print("[early stop] target reached during LP phase")
                break

        # decay mixing weight – we become more conservative later
        mix_t = max(0.05, mix_t * mix_decay)

    # ------------------------------------------------------------------
    # 5) Phase – stochastic local refinement (tiny mass transfers)
    # ------------------------------------------------------------------
    a = best.copy()
    iter_local = 0
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a small, log‑uniform fraction of the source mass
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[i] * frac_move

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta    # sum stays 1, non‑negativity guaranteed

        cand_score = evaluate_sequence(cand.tolist())
        iter_local += 1

        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score
            a = cand
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local move #{iter_local} improved => {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached during local phase")
                break

        # occasional jitter to escape tiny plateaus
        if iter_local % 4000 == 0:
            width = rng.integers(1, max(2, n // 50))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            jitter = a.copy()
            jitter[start:start + width] *= factor
            a = _project_onto_simplex(jitter)

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clip to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
