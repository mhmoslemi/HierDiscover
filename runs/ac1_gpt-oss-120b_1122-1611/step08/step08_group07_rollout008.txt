def gradient_maxconv(a, idx):
    n = a.size
    # compute convolution index contributions for gradient
    # For each i, contribution where j = idx - i
    g = np.zeros_like(a)
    # valid i where 0 <= idx-i < n
    # That is i in [idx - (n-1), idx]
    lo = max(0, idx - (n - 1))
    hi = min(idx, n - 1)
    i_range = np.arange(lo, hi + 1)
    j_range = idx - i_range
    g[i_range] = a[j_range]
    return 2.0 * g
```

"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the bound on ``evaluate_sequence``.

Algorithm (≈1000 s budget)
---------------------------

1. **Representation** – a is a probability vector (sum = 1, a_i ≥ 0).  The
   exact value of ``evaluate_sequence`` is
   ``score(a) = 2·n·max_k (a*a)_k`` where ``*`` is the linear convolution.
   The scale invariance allows us to enforce the simplex constraint.

2. **Two complementary moves**
   * **Projected sub‑gradient step** – the current peak index ``k*`` is
     identified, the sub‑gradient of ``max_k (a*a)_k`` is
     ``g_i = 2·a_{k*‑i}`` (zero when the index is out of range).  A small
     step in the direction ``-g`` is taken and the result is projected back
     onto the simplex (Euclidean projection, Duchi et al.).
   * **Random mass‑transfer move** – pick two distinct positions ``i, j``,
     move a random fraction of ``a_i`` to ``j``.  This explores the
     neighbourhood more globally.

   The first move quickly flattens the dominant peak, the second avoids
   getting stuck in a local minimum.  Both moves are accepted
   deterministically when they improve the true score and otherwise with a
   Metropolis probability using a slowly decreasing temperature.

3. **Stagnation handling** – if no improvement has been seen for 30 s the
   current best solution is jittered (log‑normal perturbation of a random
   block) and the search continues.

4. **Initialisation** – start from the previously found sequence
   ``height_sequence_1`` (if available) otherwise from a random Dirichlet
   draw.  The vector is normalised to sum = 1.

5. **FFT‑based convolution** – the self‑convolution and the sub‑gradient are
   computed with a single FFT of length ``_next_pow2(2·n‑1)`` for speed.

The combination of a fast exact gradient on the active constraint and a
simple‑yet‑effective simulated‑annealing scheme typically pushes the score
below the previous best (≈1.5132) within the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value 2·n·max(a*a)."""
    n = a.size
    conv = _self_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_of_maxconv(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    For the peak at ``peak_idx`` the gradient is
        g_i = 2 * a[peak_idx - i]   (if the index is inside the vector)
    otherwise 0.
    """
    n = a.size
    g = np.zeros_like(a)
    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    i_range = np.arange(lo, hi + 1, dtype=np.int64)
    j_range = peak_idx - i_range
    g[i_range] = 2.0 * a[j_range]
    return g


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.3))   # moderate noise
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_to_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine sequence length and initialise a
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64).reshape(-1)
        if a0.size == 0:
            n = 4096
            a = np.full(n, 1.0 / n)
        else:
            n = a0.size
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0.0:
                a = a0 / s
            else:
                a = np.full(n, 1.0 / n)
    else:
        # default length – a power of two gives fast FFTs
        n = 4096
        # Dirichlet sample (α=0.5 gives moderately sparse start)
        a = rng.dirichlet(np.full(n, 0.5))

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT length for the self‑convolution
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 3) Track the best solution seen so far
    # --------------------------------------------------------------
    cur_a = a.copy()
    cur_score = _score(cur_a, fft_len_self)
    best_a = cur_a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 4) Hyper‑parameters
    # --------------------------------------------------------------
    # step‑size schedule for projected sub‑gradient
    step_start = 0.5
    step_end = 0.001

    # temperature schedule for random moves (Metropolis)
    T_start = 0.05
    T_end = 0.0001

    # probability of taking a sub‑gradient step
    prob_subgrad = 0.65

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # recompute convolution and peak index only when needed
        conv = _self_conv(cur_a, fft_len_self)
        peak_idx = int(np.argmax(conv))

        # ----------------------------------------------------------
        # 5a) Choose move type
        # ----------------------------------------------------------
        if rng.random() < prob_subgrad:
            # ------- projected sub‑gradient step -------
            grad = _gradient_of_maxconv(cur_a, peak_idx)
            # step size decreasing linearly
            step = step_start * (1.0 - frac) + step_end * frac
            a_candidate = cur_a - step * grad
            a_candidate = _project_to_simplex(a_candidate)
        else:
            # ------- random mass‑transfer move -------
            i = rng.integers(n)
            # ensure j != i
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            # move a random fraction of the mass at i towards j
            max_move = cur_a[i] * rng.uniform(0.0, 0.2)
            delta = max_move * rng.random()
            if delta <= 0.0:
                continue
            a_candidate = cur_a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            # projection is NOT required here because the move preserves sum
            # and non‑negativity (we ensured delta ≤ cur_a[i]).

        # ----------------------------------------------------------
        # 5b) Evaluate candidate
        # ----------------------------------------------------------
        cand_score = _score(a_candidate, fft_len_self)

        # ----------------------------------------------------------
        # 5c) Acceptance test
        # ----------------------------------------------------------
        if cand_score < cur_score - 1e-14:
            # deterministic acceptance
            cur_a = a_candidate
            cur_score = cand_score
            if cand_score < best_score - 1e-14:
                best_score = cand_score
                best_a = a_candidate.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best score = {best_score:.6f}")
        else:
            # Metropolis acceptance for uphill moves
            T = T_start * (1.0 - frac) + T_end * frac
            if T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / T):
                cur_a = a_candidate
                cur_score = cand_score

        # ----------------------------------------------------------
        # 5d) Stagnation handling
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # jitter the best solution to try to escape plateau
            cur_a = _random_block_perturb(best_a, rng)
            cur_score = _score(cur_a, fft_len_self)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stagnation jitter – best {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (already satisfies the evaluator limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
