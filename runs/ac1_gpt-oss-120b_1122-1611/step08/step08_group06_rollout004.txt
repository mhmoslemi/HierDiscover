"""
Propose a candidate step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)^2)

using a three‑phase optimisation on the simplex (soft‑max
parameterisation).  

Phase 1    – minimise the variance of the self‑convolution (drive it
            towards the constant 1/(2·n‑1)).  
Phase 2    – minimise a smooth approximation of the max using a
            log‑sum‑exp surrogate with an annealed temperature (γ).  
Phase 3    – fine‑tune the true max with a sub‑gradient (one‑hot weight).  

The search restarts on stagnation (no improvement for ~20 s) by
perturbing the current best solution, and performs occasional global
jitter.  Several simple parametric initial shapes (uniform, half‑sine,
cosine‑raised) are tried and the one with the lowest objective is used
as the start point.  The best sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _objective(a: np.ndarray, n: int, fft_len_self: int) -> float:
    """Exact objective value used for reporting / best‑track."""
    c = _self_conv(a, fft_len_self)
    max_c = c.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s ** 2)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.18))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# ----------------------------------------------------------------------

    # ------------------------------------------------------------------

            n = a0.size
            candidates = [a0]
        else:
            n = 5000
            candidates = []
    else:
        n = 5000
        candidates = []

    # Generate a few cheap parametric candidates (uniform, half‑sine,
    # cosine‑raised).  All are normalised to unit L1‑norm.
    idx = np.arange(n, dtype=np.float64)
    # half‑sine
    s = np.sin(np.pi * (idx + 0.5) / n)
    candidates.append(s / s.sum())
    # cosine raised (non‑negative)
    c = np.cos(np.pi * (idx - (n - 1) / 2) / n)
    c = np.clip(c, 0.0, None)
    if c.sum() > 0:
        candidates.append(c / c.sum())
    # power‑law of the sine shape
    for p in (0.5, 2.0, 3.0, 4.0):
        sp = np.power(s, p)
        if sp.sum() > 0:
            candidates.append(sp / sp.sum())

    # ------------------------------------------------------------------
    # 2) Pick best initial candidate (cheap evaluation)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)
    best_init = None
    best_score = np.inf
    for cand in candidates:
        score = _objective(cand, n, fft_len_self)
        if score < best_score:
            best_score = score
            best_init = cand
    if best_init is None:
        best_init = np.full(n, 1.0 / n)

    # Initialise log‑weights (soft‑max parameters)
    x = np.log(best_init + 1e-12)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state (shared across phases)
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Tracking the best solution
    # ------------------------------------------------------------------
    a = _softmax(x)
    cur_score = _objective(a, n, fft_len_self)
    best_a = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 5) Phase divisions (percent of time budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * total_time
    phase2_end = start_time + 0.85 * total_time
    # phase3 lasts until deadline

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ------------------------------------------------------------------
        # current point, convolution and exact score
        # ------------------------------------------------------------------
        a = _softmax(x)                     # simplex point
        c = _self_conv(a, fft_len_self)     # length 2·n‑1
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # ------------------------------------------------------------------
        # record improvement
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # choose loss & gradient
        # ------------------------------------------------------------------
        if now < phase1_end:
            # Phase 1 – variance flattening
            avg = 1.0 / (2 * n - 1)               # target constant value
            w = c - avg
            g_a = 2.0 * _gradient_wrt_a(a, w)    # factor 2 from d/d a of Σ(c‑avg)²
        elif now < phase2_end:
            # Phase 2 – smoothed‑max (log‑sum‑exp)
            phase_frac = (now - phase1_end) / max(phase2_end - phase1_end, 1e-12)
            gamma_start = 0.5
            gamma_end = 80.0                     # larger end temperature – tighter max
            gamma = gamma_start + (gamma_end - gamma_start) * phase_frac
            # stable softmax of scaled convolution
            c_shift = c - np.max(c)
            w = np.exp(gamma * c_shift)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(c, 1.0 / c.size)
            else:
                w /= w_sum
            g_a = _gradient_wrt_a(a, w)          # already includes the factor 2
        else:
            # Phase 3 – exact max sub‑gradient (one‑hot)
            k_max = int(np.argmax(c))
            w = np.zeros_like(c)
            w[k_max] = 1.0
            g_a = _gradient_wrt_a(a, w)

        # ------------------------------------------------------------------
        # Chain rule to obtain gradient w.r.t. logits x (softmax Jacobian)
        # ------------------------------------------------------------------
        avg_g = np.dot(a, g_a)                # <a, ∂F/∂a>
        g_x = a * (g_a - avg_g)               # ∂F/∂x

        # ------------------------------------------------------------------
        # Gradient clipping (helps stability)
        # ------------------------------------------------------------------
        g_norm = np.linalg.norm(g_x)
        max_norm = 0.8
        if g_norm > max_norm:
            g_x = g_x * (max_norm / g_norm)

        # ------------------------------------------------------------------
        # Adam step (global learning‑rate schedule)
        # ------------------------------------------------------------------
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Linear decay of the base learning‑rate from 0.45 → 0.02
        lr_start = 0.45
        lr_end = 0.02
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ------------------------------------------------------------------
        # Stagnation handling – restart from the best known point if
        # no improvement for ~20 s.
        # ------------------------------------------------------------------
        if now - last_improve > 20.0:
            print(f"[{elapsed:6.2f}s] stagnation – restarting from best")
            # add a little isotropic Gaussian noise to logits
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now

        # ------------------------------------------------------------------
        # Occasional global jitter (2 % chance)
        # ------------------------------------------------------------------
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best_a, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ----------------------------------------------------------------------
    # 7) Return the best sequence (clip to the limits imposed by the
    #    evaluation routine)
    # ----------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
