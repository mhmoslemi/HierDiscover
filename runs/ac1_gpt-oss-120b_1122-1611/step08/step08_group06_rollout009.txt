"""
Hybrid variance‑maximisation optimiser for the step‑function bound.

The algorithm proceeds in three stages:

1. **Coarse multi‑scale warm‑start** – a short optimisation on a small
   sequence (≈512 points) using a combined variance / smoothed‑max loss.
   The resulting shape is interpolated to the target length.

2. **Main optimisation (symmetrised soft‑max parametrisation)** – the
   variables are the log‑weights `x` of a probability vector
   `a = softmax(x)`.  Symmetry (`x[i]=x[n-1-i]`) is enforced by
   projection after each Adam update, which reduces the search space
   and keeps the convolution shape even.

   The loss is a convex combination of
   * variance of the self‑convolution (`α·‖c‑avg‖²`) and
   * a smooth approximation of the maximum (`(1‑α)·log‑sum‑exp`).

   `α` linearly decays from 1 to 0, while the temperature `γ` of the
   log‑sum‑exp grows from 0.5 to 80, so the optimiser first flattens the
   convolution and then drives the true peak down.

3. **Fine‑tuning & escape mechanisms** – occasional random block
   perturbations, stochastic restarts on stagnation and an Adam learning‑
   rate that linearly decays over the whole budget.

The routine respects a 1000 s budget (5 s safety margin) and returns the
best sequence found, clipped to the limits used by `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution `a*a` using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.

    Returns a vector of length n (the length of ``a``).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _interpolate_sequence(src: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    src_n = src.size
    src_x = np.linspace(0.0, 1.0, src_n)
    tgt_x = np.linspace(0.0, 1.0, target_n)
    tgt = np.interp(tgt_x, src_x, src)
    s = tgt.sum()
    if s == 0.0:
        tgt[:] = 1.0 / target_n
    else:
        tgt /= s
    return tgt


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds (5 s safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global timing (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    total_budget = max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    deadline = start_time + total_budget

    # ------------------------------------------------------------------
    # 1) Determine target length and initialise
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        init_arr = np.asarray(init_seq, dtype=np.float64)
        if init_arr.ndim == 1 and init_arr.size > 0:
            init_arr = np.clip(init_arr, 0.0, None)
            s = init_arr.sum()
            if s > 0.0:
                init_arr /= s
            else:
                init_arr = np.full_like(init_arr, 1.0 / init_arr.size)
            target_n = init_arr.size
            # we'll use this as the final warm‑start after the small‑scale phase
            warm_start = init_arr
        else:
            target_n = 4096
            warm_start = None
    else:
        target_n = 4096
        warm_start = None

    # ------------------------------------------------------------------
    # 2) Small‑scale pre‑training (helps escape bad local minima)
    # ------------------------------------------------------------------
    # Use a tiny fraction of the total budget for a short run on a
    # much smaller dimension (default 512).  The result is interpolated
    # back to ``target_n``.
    SMALL_FRAC = 0.07  # 7 % of the budget
    small_deadline = start_time + SMALL_FRAC * total_budget
    small_n = min(512, max(128, target_n // 8))

    if small_n != target_n:
        # initialise log‑weights for the small problem
        x_small = np.zeros(small_n, dtype=np.float64)
        m_small = np.zeros_like(x_small)
        v_small = np.zeros_like(x_small)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        t_small = 0

        fft_len_self_small = _next_pow2(2 * small_n - 1)

        # simple schedule: α linearly from 1→0, γ from 0.5→30
        while time.monotonic() < small_deadline:
            a_small = _softmax(x_small)                     # already symmetric because we will enforce it
            a_small = (a_small + a_small[::-1]) / 2.0        # projection onto the symmetric sub‑space
            c_small = _self_conv(a_small, fft_len_self_small)
            avg_small = 1.0 / (2 * small_n - 1)

            # ----- loss components -----
            r_small = c_small - avg_small
            grad_var = 2.0 * _gradient_wrt_a(a_small, r_small)

            # smooth‑max part
            now = time.monotonic()
            elapsed = now - start_time
            phase_frac = (elapsed - 0.0) / total_budget
            gamma = 0.5 + (30.0 - 0.5) * phase_frac ** 2
            c_shift = c_small - np.max(c_small)
            w = np.exp(gamma * c_shift)
            w /= w.sum()
            grad_max = _gradient_wrt_a(a_small, w)

            # blend the two gradients
            α = max(0.0, 1.0 - 2.0 * phase_frac)   # α = 1 for first half of *overall* budget, 0 afterwards
            grad_a = α * grad_var + (1.0 - α) * grad_max

            # ----- back‑prop through soft‑max -----
            avg_g = np.dot(a_small, grad_a)
            grad_x = a_small * (grad_a - avg_g)

            # ----- Adam update -----
            t_small += 1
            m_small = beta1 * m_small + (1.0 - beta1) * grad_x
            v_small = beta2 * v_small + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m_small / (1.0 - beta1 ** t_small)
            v_hat = v_small / (1.0 - beta2 ** t_small)

            # simple linear decay of the learning rate
            lr = 1.0 * (1.0 - phase_frac) + 0.05 * phase_frac
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            x_small -= step

            # keep symmetry of the logits
            x_small = (x_small + x_small[::-1]) / 2.0

            # occasional jitter to avoid stagnation
            if rng.random() < 0.02:
                a_jit = _random_block_perturb(a_small, rng)
                a_jit = (a_jit + a_jit[::-1]) / 2.0
                x_small = np.log(a_jit + 1e-12)

        # upscale the learned small shape to the target length
        a_small = _softmax(x_small)
        a_small = (a_small + a_small[::-1]) / 2.0
        warm_start = _interpolate_sequence(a_small, target_n)

    # ------------------------------------------------------------------
    # 3) Initialise the main optimisation (log‑weights `x`)
    # ------------------------------------------------------------------
    if warm_start is not None:
        a0 = warm_start
    else:
        # fall‑back: uniform distribution
        a0 = np.full(target_n, 1.0 / target_n)

    # small epsilon prevents log(0)
    x = np.log(a0 + 1e-12)
    x = (x + x[::-1]) / 2.0                     # enforce symmetry from the start

    n = target_n
    fft_len_self = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Tracking the best solution
    # ------------------------------------------------------------------
    a = _softmax(x)
    a = (a + a[::-1]) / 2.0
    c = _self_conv(a, fft_len_self)
    best_a = a.copy()
    best_score = float(2 * n * c.max() / (a.sum() ** 2))
    last_improve = time.monotonic()

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        phase_frac = elapsed / total_budget

        # ---- obtain current a (symmetric) and its self‑convolution ----
        a = _softmax(x)
        a = (a + a[::-1]) / 2.0          # symmetry projection
        c = _self_conv(a, fft_len_self)

        # ---- exact score (the quantity we finally care about) ----
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # ---- record improvement ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- loss construction (combined variance + smooth‑max) ----
        avg = 1.0 / (2 * n - 1)
        r = c - avg                                   # residual for variance term
        grad_var = 2.0 * _gradient_wrt_a(a, r)

        # smooth‑max term
        gamma = 0.5 + (80.0 - 0.5) * phase_frac ** 2   # start gentle, end sharp
        c_shift = c - np.max(c)
        w = np.exp(gamma * c_shift)
        w /= w.sum()
        grad_max = _gradient_wrt_a(a, w)

        # blending weight α : fade variance out in the first half of the run
        if phase_frac < 0.5:
            α = 1.0 - 2.0 * phase_frac    # 1 → 0 linearly in first half
        else:
            α = 0.0

        grad_a = α * grad_var + (1.0 - α) * grad_max

        # ---- back‑prop through soft‑max (Jacobian of the simplex) ----
        avg_g = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg_g)

        # ---- Adam update -------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (linear from 1 → 0.02)
        lr = 1.0 * (1.0 - phase_frac) + 0.02 * phase_frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # enforce symmetry of the logits after the step
        x = (x + x[::-1]) / 2.0

        # ---- stagnation handling (soft restart) -------------------------
        if now - last_improve > 20.0:          # 20 s without improvement
            noise = rng.normal(0.0, 0.3, size=n)
            x = np.log(best_a + 1e-12) + noise
            x = (x + x[::-1]) / 2.0
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart due to stagnation – best {best_score:.6f}")

        # ---- occasional global jitter ------------------------------------
        if rng.random() < 0.015:               # ~1.5 % of iterations
            a_jit = _random_block_perturb(best_a, rng)
            a_jit = (a_jit + a_jit[::-1]) / 2.0
            x = np.log(a_jit + 1e-12)
            x = (x + x[::-1]) / 2.0
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 6) Return the best sequence found (apply the same clipping as the
    #    evaluator to stay within its safe region).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
