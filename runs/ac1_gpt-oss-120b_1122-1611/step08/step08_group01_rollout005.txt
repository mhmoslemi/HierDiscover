"""
Propose a step‑function (non‑negative heights on the simplex) that beats the
previous best bound (~1.5132).  

The algorithm consists of three stages

1️⃣ **Exploratory shape sweep** –  a small fixed set of parametric families
   (beta‑type, Gaussian, cosine/Hann windows, a simple triangular shape,
   and the previously known sequence) is generated, each candidate is
   normalised to sum = 1 and its exact objective
   ``2·n·max(conv)`` is evaluated with an FFT based self‑convolution.
   The best vector from this sweep becomes the starting point for the next
   phase.

2️⃣ **Simulated‑annealing hill‑climb** – starting from the best vector,
   repeated random “mass‑move’’ operations are performed:
   * transfer a small fraction of probability mass between two random
     indices,
   * multiply a random contiguous block by a log‑normal factor,
   * occasionally replace the whole vector by a jittered version of the
     current best.
   Moves that improve the exact objective are always accepted; worsening
   moves are accepted with probability ``exp(-Δ/T)`` where the temperature
   ``T`` decays linearly with the remaining time budget.  The step size
   (fraction of moved mass) also decays so that the search becomes finer
   near the end.

3️⃣ **Finalisation** – the best vector found is clipped to the limits
   imposed by ``evaluate_sequence`` (non‑negative, ≤ 1000) and returned.

All heavy work (self‑convolution, score) is performed with a pre‑computed
FFT length, making the inner loop fast enough to perform tens of thousands
of move proposals within the 1000 s budget.

The routine is completely self‑contained; it only relies on ``numpy`` and the
already‑imported ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a 1‑D array using an FFT of size ``fft_len``.
    The returned array has length ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _objective(a: np.ndarray, fft_len: int) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when the sum of ``a`` is 1:
    ``2·n·max(conv)``.
    ``a`` may be any non‑negative vector; it is normalised internally.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = _self_conv_fft(a, fft_len)
    return float(2.0 * a.size * conv.max())


def _move_mass(a: np.ndarray, rng: np.random.Generator, step_frac: float) -> np.ndarray:
    """
    Transfer a small fraction (≤ ``step_frac``) of the probability mass from a
    random source index to a random destination index.
    The output is re‑normalised to have sum = 1.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = step_frac * rng.random() * a[i]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and re‑normalise.
    """
    n = a.size
    # block width at most 10 % of the whole vector (but at least 1)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑type shape: (i+1)^α · (n‑i)^β, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + 1.0, alpha) * np.power(n - i, beta)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian, normalised."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64) - centre
    a = np.exp(-(i ** 2) / (2.0 * sigma ** 2))
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _cosine_window(n: int) -> np.ndarray:
    """Hann (cosine) window, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a_sum = a.sum()
    a /= a_sum
    return a


def _hamming_window(n: int) -> np.ndarray:
    """Hamming window, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a_sum = a.sum()
    a /= a_sum
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Simple symmetric triangle (peak in the centre), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    a_sum = a.sum()
    a /= a_sum
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the 1000 s budget.
    Returns the best sequence found (as a plain list of Python floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time handling – reserve a few seconds for the harness
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a comfortable length (power‑of‑two for FFT speed)
    # ------------------------------------------------------------------
    # If a previous construction is available, reuse its length;
    # otherwise fall back to 3840 (≈ 2³ × 480) which is a good compromise.
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)):
        n = len(prev)
        if n < 256:                # guard against absurdly short vectors
            n = 3840
    else:
        n = 3840

    # Ensure n is a power of two (helps FFT); if not, round up.
    if (n & (n - 1)) != 0:
        n = 1 << ((n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution
    # ------------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) ----------------------------------------------------------------
    #   Stage 1 – systematic exploration of simple parametric families
    # ------------------------------------------------------------------
    print(f"[init] sequence length = {n}, FFT length = {fft_len}")

    candidates = []

    # a) Uniform (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # b) Simple triangle
    candidates.append(_triangular_shape(n))

    # c) Cosine/Hann window
    candidates.append(_cosine_window(n))

    # d) Hamming window
    candidates.append(_hamming_window(n))

    # e) Gaussian family (σ ranging from 0.1 · n to 0.5 · n)
    for factor in (0.12, 0.15, 0.18, 0.22, 0.28, 0.35, 0.45):
        sigma = factor * n
        candidates.append(_gaussian_shape(n, sigma))

    # f) Beta‑type shapes with a small integer grid of exponents
    exponents = (0.0, 0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0)
    for a_exp in exponents:
        for b_exp in exponents:
            # skip the trivial (0,0) – already added as uniform
            if a_exp == 0.0 and b_exp == 0.0:
                continue
            candidates.append(_beta_shape(n, a_exp, b_exp))

    # g) If a previous good construction is present, add it as a candidate
    if isinstance(prev, (list, tuple, np.ndarray)):
        arr = np.asarray(prev, dtype=np.float64).ravel()
        if arr.size != n:
            # resize by linear interpolation to the current length
            x_old = np.linspace(0, 1, arr.size)
            x_new = np.linspace(0, 1, n)
            arr = np.interp(x_new, x_old, arr)
        arr = np.clip(arr, 0.0, None)
        s = arr.sum()
        if s > 0.0:
            arr /= s
        else:
            arr[:] = 1.0 / n
        candidates.append(arr)

    # ------------------------------------------------------------------
    # 4) Evaluate all candidates, keep the best
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    start = time.monotonic()
    total_budget = deadline - start

    for idx, vec in enumerate(candidates):
        # stop early if we run out of time
        if time.monotonic() >= deadline:
            break
        sc = _objective(vec, fft_len)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
            print(f"[stage 1] new best from candidate {idx} = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Stage 2 – simulated‑annealing / hill‑climbing from the best vector
    # ------------------------------------------------------------------
    if best_vec is None:                     # safety fallback
        best_vec = np.full(n, 1.0 / n)

    cur_vec = best_vec.copy()
    cur_score = best_score
    best_vec = best_vec.copy()
    best_score = best_score

    # timing helpers
    phase2_start = time.monotonic()
    last_improve = phase2_start

    # annealing schedule parameters
    T0 = max(1e-4, best_score * 0.05)        # initial temperature
    # step‑size (fraction of moved mass) decays from ~0.3 to 0.01
    def _step_fraction(frac_elapsed: float) -> float:
        return 0.3 * (1.0 - frac_elapsed) + 0.01

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        frac_elapsed = elapsed / (deadline - phase2_start)

        # linear temperature decay
        temperature = max(1e-12, T0 * (1.0 - frac_elapsed))

        # ------------------------------------------------------------------
        # propose a move (three possible move types, weighted)
        # ------------------------------------------------------------------
        move_type = rng.random()
        if move_type < 0.45:          # simple mass transfer
            cand = _move_mass(cur_vec, rng, _step_fraction(frac_elapsed))
        elif move_type < 0.85:        # block‑wise log‑normal perturbation
            cand = _random_block_perturb(cur_vec, rng)
        else:                         # occasional global jitter around the best so far
            jitter = rng.normal(0.0, 0.4, size=n)
            cand = np.exp(np.log(best_vec + 1e-12) + jitter)
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()

        cand_score = _objective(cand, fft_len)

        delta = cand_score - cur_score
        accept = False
        if delta < 0.0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-delta / temperature):
                accept = True

        if accept:
            cur_vec = cand
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{now - start:6.2f}s] new best (SA) = {best_score:.6f}")

        # ------------------------------------------------------------------
        # stagnation handling – restart locally around the current best
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            # re‑initialise a fresh candidate by adding moderate noise to best_vec
            noise = rng.normal(0.0, 0.4, size=n)
            cur_vec = np.exp(np.log(best_vec + 1e-12) + noise)
            cur_vec = np.clip(cur_vec, 0.0, None)
            cur_vec /= cur_vec.sum()
            cur_score = _objective(cur_vec, fft_len)
            last_improve = now
            print(f"[{now - start:6.2f}s] local‑restart (stagnation) – score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence found (apply the same clipping that
    #    ``evaluate_sequence`` would apply – non‑negative, ≤ 1000)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]
