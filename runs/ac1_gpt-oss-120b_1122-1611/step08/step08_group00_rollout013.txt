def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    # time budget
    deadline = time.monotonic() + max(0., budget_s - 5.)

    rng = np.random.default_rng(seed)

    # Determine n
    if "height_sequence_1" in globals():
        seq0 = globals()["height_sequence_1"]
        try:
            n0 = len(seq0)
            n = min(max(n0, 512), 4096)
        except Exception:
            n = 4096
    else:
        n = 4096

    # Precompute FFT length for self-conv (next_pow2 of 2n-1)
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    # Helper to compute full convolution via FFT
    def self_conv(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        return conv

    # Objective J
    def objective(a):
        conv = self_conv(a)
        max_b = conv.max()
        # sum_a = a.sum()
        return float(2.0 * n * max_b / (a.sum()**2))

    # Initialize candidates list
    candidates = []

    # Uniform
    candidates.append(np.full(n, 1.0/n, dtype=np.float64))

    # Exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n)
        v = v / v.sum()
        candidates.append(v)

    # Linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v = v / v.sum()
    candidates.append(v)

    # Two spikes at ends (both mass)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    candidates.append(v)

    # Half heavy, half zero
    v = np.zeros(n, dtype=np.float64)
    half = n//2
    v[:half] = 2.0/n  # ensures sum = 1
    candidates.append(v)

    # Dirichlet draws with different concentration
    for alpha in [0.1, 0.3, 1.0, 5.0]:
        v = rng.dirichlet(np.full(n, alpha))
        candidates.append(v)

    # If existing height_sequence_1, add it normalized
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(height_sequence_1, dtype=np.float64)
            v = np.clip(v, 0.0, None)
            v_sum = v.sum()
            if v_sum == 0.0:
                v = np.full_like(v, 1.0 / len(v))
            else:
                v = v / v_sum
            candidates.append(v)
        except Exception:
            pass

    # Deduplicate candidates after projection onto simplex (ensuring sum=1)
    candidates = [proj_simplex(c) for c in candidates]

    # Evaluate candidates, keep best
    best_vec = None

    n = a.shape[0]
    idx = k - np.arange(n)
    g = np.zeros_like(a)
    mask = (idx >= 0) & (idx < n)
    g[mask] = a[idx[mask]]
    idx2 = np.arange(n) - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] += a[idx2[mask2]]
    return g

"""
Hybrid random‑restart sub‑gradient optimizer for the step‑function problem.

* 1️⃣  Build a modest portfolio of diverse seeds (uniform, exponential,
     linear, heavy‑half and Dirichlet draws).  The best seed is taken as
     the starting point.
* 2️⃣  Repeatedly minimise the *exact* max‑convolution objective
     (no soft‑max smoothing) by a projected sub‑gradient method.
     For the current vector `a` we find the index `k` where the
     self‑convolution `c = a*a` attains its maximum and use the
     sub‑gradient

        g_i = a_{k-i}  (if 0≤k-i<n)  +  a_{i-k} (if 0≤i-k<n)

     which is the exact derivative of the active linear term.
* 3️⃣  After each step the iterate is projected onto the probability
     simplex (non‑negative, sum = 1) using the Euclidean projection.
* 4️⃣  An ever‑decreasing stepsize `η = η₀ / sqrt(t)` together with a
     simple simulated‑annealing acceptance rule lets the search escape
     shallow plateaux.
* 5️⃣  If no improvement is seen for 30 s the algorithm restarts from a
     fresh Dirichlet sample (or injects a tiny multiplicative noise).

The whole routine respects a 1000 s budget (5 s safety margin) and
returns the best coefficient list found.  The returned vector is already
clipped to the limits imposed by ``evaluate_sequence`` (0 ≤ x ≤ 1000)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the linear term (a*a)[k] with respect to a.
    Implements g_i = a_{k-i} (if valid) + a_{i-k} (if valid).
    """
    n = a.size
    i = np.arange(n)
    idx1 = k - i
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx2 = i - k
    mask2 = (idx2 >= 0) & (idx2 < n)

    g = np.zeros_like(a)
    g[mask1] = a[idx1[mask1]]
    g[mask2] += a[idx2[mask2]]
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = min(max(n, 512), 4096)          # clamp to a comfortable range
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # length for FFTs

    # ------------------------------------------------------------------
    # 2) Helper that returns the exact objective (sum of a is 1 → J = 2·n·max)
    # ------------------------------------------------------------------
    def objective(a: np.ndarray) -> float:
        conv = _self_convolution(a, fft_len)
        return float(2.0 * n * conv.max())      # sum(a)=1 after projection

    # ------------------------------------------------------------------
    # 3) Build an initial portfolio of seeds
    # ------------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        seeds.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    seeds.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    seeds.append(v)

    # half‑heavy, half‑zero
    v = np.zeros(n, dtype=np.float64)
    v[: n // 2] = 2.0 / n
    seeds.append(v)

    # Dirichlet draws with a few concentration parameters
    for a in [0.1, 0.3, 1.0, 5.0]:
        seeds.append(rng.dirichlet(np.full(n, a)).astype(np.float64))

    # existing construction, if any
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(height_sequence_1, dtype=np.float64)
            v = np.clip(v, 0.0, None)
            if v.sum() == 0.0:
                v = np.full_like(v, 1.0 / v.size)
            else:
                v = v / v.sum()
            seeds.append(v)
        except Exception:
            pass

    # normalise everything onto the simplex (guards against tiny rounding issues)
    seeds = [proj_simplex(s) for s in seeds]

    # ------------------------------------------------------------------
    # 4) Pick the best seed as starting point
    # ------------------------------------------------------------------
    best_vec = None
    best_val = float("inf")
    for s in seeds:
        val = objective(s)
        if val < best_val:
            best_val = val
            best_vec = s.copy()
    print(f"[init] best seed J = {best_val:.6f}")

    # current iterate for the sub‑gradient loop
    cur_vec = best_vec.copy()
    cur_val = best_val

    # ------------------------------------------------------------------
    # 5) Sub‑gradient descent with random restarts
    # ------------------------------------------------------------------
    eta0 = 0.30                      # initial stepsize
    iter_cnt = 0
    last_improve = time.monotonic()
    # a small pool of fresh Dirichlet restart vectors
    restart_pool = [proj_simplex(rng.dirichlet(np.full(n, 0.3))) for _ in range(12)]

    while time.monotonic() < deadline:
        iter_cnt += 1

        # ----- convolution and current objective -----
        conv = _self_convolution(cur_vec, fft_len)
        max_b = conv.max()
        cur_val = float(2.0 * n * max_b)   # sum(cur_vec) ≡ 1

        # ----- keep global best -----
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic():.2f}s] new best J = {best_val:.6f}")

        # ----- pick a random active index (break ties randomly) -----
        max_idxs = np.flatnonzero(conv == max_b)
        k = rng.choice(max_idxs)

        # ----- sub‑gradient of the active linear term -----
        grad = _subgradient(cur_vec, k)

        # ----- stepsize (1/√t decay) and update -----
        eta = eta0 / np.sqrt(iter_cnt)
        candidate = proj_simplex(cur_vec - eta * grad)

        # ----- evaluate candidate -----
        cand_val = objective(candidate)

        # ----- simple simulated‑annealing acceptance -----
        if cand_val <= cur_val:
            cur_vec = candidate
            cur_val = cand_val
        else:
            # temperature slowly goes to zero, encouraging fewer uphill moves later
            remaining = max(0.0, deadline - time.monotonic())
            temp = remaining / (budget_s - 5.0)   # linear schedule in [0,1]
            prob = np.exp(-(cand_val - cur_val) * temp * 0.5)
            if rng.random() < prob:
                cur_vec = candidate
                cur_val = cand_val

        # ----- stagnation handling -----
        if time.monotonic() - last_improve > 30.0:
            if restart_pool:
                cur_vec = restart_pool.pop()
                print(f"[{time.monotonic():.2f}s] restart from fresh Dirichlet")
            else:
                # inject a tiny multiplicative noise and re‑project
                noise = np.exp(rng.normal(0.0, 0.07, size=n))
                cur_vec = proj_simplex(cur_vec * noise)
                print(f"[{time.monotonic():.2f}s] noise‑restart")
            iter_cnt = 0
            last_improve = time.monotonic()

        # ----- occasional diagnostic output (once per ~2 s) -----
        if iter_cnt % 200 == 0:
            now = time.monotonic()
            if now < deadline:
                print(f"[{now:.1f}s] iter {iter_cnt:5d}, cur J={cur_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence found (clipped to the bounds used by evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
