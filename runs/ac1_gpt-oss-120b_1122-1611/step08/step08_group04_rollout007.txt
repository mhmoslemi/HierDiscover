def _subgradient_opt(a, rng, deadline, max_iters=2000):
    n = len(a)
    step = 0.1
    best_local_a = a.copy()
    best_local_score = _objective_score(a)
    it = 0

        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # Determine active set: values within tolerance
        tol = max_val * 1e-6 + 1e-12  # tolerance relative
        active = np.where(conv >= max_val - tol)[0]
        # Compute gradient as average of contributions from each active offset
        grad = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        grad = grad / len(active)  # average
        # Update via projection onto simplex
        a_cand = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_cand)
        if cand_score < best_local_score - 1e-12:

            best_local_score = cand_score
            best_local_a = a.copy()
            step = step * 1.05

            step = step * 0.7

                a = _project_simplex(a + jitter)
                step = max(step, 0.01)

"""
Hybrid optimisation for the self‑convolution (max‑autocorrelation) problem.

The routine mixes several complementary heuristics within the time budget:

  * Phase 1 – Adam on a smooth log‑sum‑exp surrogate of the convolution
    max‑norm.  A slowly rising β brings the surrogate ever closer to the true
    infinity‑norm while Adam with a soft‑max parametrisation guarantees that
    the iterate stays on the probability simplex.

  * Phase 2 – (optional, for moderate sizes) a Frank‑Wolfe style LP step.
    For a fixed bound M = max(a∗a) we solve a linear programme that maximises
    ‖b‖₁ under the linear constraints (a∗b) ≤ M.  The resulting b is projected
    back onto the simplex and mixed with the current a.  This moves mass towards
    indices where the interaction with a is small and usually reduces the peak
    of the self‑convolution.

  * Phase 3 – Projected sub‑gradient descent on the exact max‑convolution.
    The sub‑gradient is taken as the average of the gradients of all
    “active’’ offsets that are within a tiny tolerance of the maximum,
    which yields a smoother descent direction than the single‑offset version.

  * Phase 4 – Simple hill‑climbing on the simplex: repeatedly transfer a
    random amount of probability mass from one coordinate to another and keep
    the change if the objective improves.  This locally refines the solution.

Multiple random restarts with different lengths (1200 – 8000) are performed,
occasionally jittering the current best solution to escape plateaus.
The best sequence seen over the whole run is returned.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (real‑valued)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (sum(a) is not forced to 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------


def _adam_opt(a0: np.ndarray, rng: np.random.Generator,
              deadline: float, max_iters: int = 4000) -> tuple[np.ndarray, float]:
    n = a0.shape[0]
    eps = 1e-12
    x = np.log(a0 + eps)                     # logits → simplex via soft‑max

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    lr = 0.35
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    beta_start = 5.0
    beta_end = 2000.0
    beta_ramp_iters = max_iters // 2

    best_a = a0.copy()
    best_score = _objective_score(a0)

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        a = _softmax_logits(x)

        # exact self‑convolution
        conv = _conv_self(a)

        cur_score = _objective_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        # --- smooth surrogate (log‑sum‑exp) ---------------------------------
        beta = beta_end if it > beta_ramp_iters else \
            beta_start + (beta_end - beta_start) * it / beta_ramp_iters

        scaled = beta * conv
        lse = logsumexp(scaled)
        w = np.exp(scaled - lse)                     # soft‑max over offsets

        # gradient wrt a : 2 * (w ⋆ a)   (cross‑correlation)
        fft_len = _next_pow2(3 * n - 2)
        w_fft = np.fft.rfft(w, n=fft_len)
        a_fft = np.fft.rfft(a, n=fft_len)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft), n=fft_len)
        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to shake out of flat regions
        if it % 300 == 0:
            x += rng.normal(0.0, 0.012, size=n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – LP (Frank‑Wolfe) step (only for moderate n)
# ----------------------------------------------------------------------


def _lp_step(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_i   s.t.   (a ∗ b)_k ≤ M  (∀k),  b_i ≥ 0.
    Returns the optimiser b (not normalised).
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz constraint matrix: each column i is a shifted copy of a
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a

    c = -np.ones(n)                     # maximise Σb  → minimise –Σb
    ub = np.full(rows, M)
    bounds = [(0.0, None) for _ in range(n)]

    # Using the high‑performance HiGHS solver (available in recent SciPy)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – uniform distribution (should be reachable)
        return np.full(n, 1.0 / n, dtype=np.float64)
    return res.x


def _frank_wolfe_opt(a0: np.ndarray, rng: np.random.Generator,
                    deadline: float, max_rounds: int = 3) -> tuple[np.ndarray, float]:
    """
    One or several Frank‑Wolfe style updates based on the LP above.
    Each round mixes the current a with the LP solution using a small step t.
    """
    a = a0.copy()
    best_score = _objective_score(a)
    best_a = a.copy()
    t = 0.15                                    # initial mixing coefficient

    for _ in range(max_rounds):
        if time.monotonic() >= deadline:
            break
        conv = _conv_self(a)
        M = conv.max()
        b = _lp_step(a, M)

        sum_b = b.sum()
        if sum_b < 1e-12:
            break
        g = b / sum_b                           # project onto simplex

        # convex combination (preserves sum = 1 and non‑negativity)
        a_new = (1.0 - t) * a + t * g
        new_score = _objective_score(a_new)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            t = min(t * 1.2, 0.45)              # slowly enlarge step on success
            print(f"[FW] improvement to {best_score:.6f}")
        else:
            t *= 0.6                              # shrink step on failure

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_opt(a0: np.ndarray, rng: np.random.Generator,
                     deadline: float, max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using an active‑set of near‑max offsets."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    step = 0.12
    n = a.shape[0]

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = _conv_self(a)
        max_val = conv.max()
        # active set: all offsets within 1e-6 relative of the maximum
        tol = max_val * 1e-6 + 1e-12
        active = np.where(conv >= max_val - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            idx = np.arange(n)
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        grad /= max(active.size, 1)               # average over the active set

        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            a = a_candidate
            best_score = cand_score
            best_a = a_candidate.copy()
            step *= 1.07
        else:
            step *= 0.68
            # occasional jitter to get out of a flat region
            if step < 1e-7 or rng.random() < 0.001:
                jitter = rng.normal(0.0, 0.0015, size=n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.02)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Simple hill‑climbing on the simplex
# ----------------------------------------------------------------------


def _hill_climb_opt(a0: np.ndarray, rng: np.random.Generator,
                    deadline: float, max_iters: int = 1500) -> tuple[np.ndarray, float]:
    """Random mass‑transfer moves that are kept only when they improve the score."""
    a = a0.copy()
    best_score = _objective_score(a)
    best_a = a.copy()
    n = a.shape[0]

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        i = rng.integers(0, n)
        if a[i] == 0.0:
            continue
        # pick a different index j
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the mass from i to j
        frac = rng.uniform(0.0, 0.5)
        delta = frac * a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective_score(a_new)
        if new_score < best_score - 1e-12:
            a = a_new
            best_score = new_score
            best_a = a_new.copy()
            # occasional larger jitter after a successful improvement
            if rng.random() < 0.05:
                jitter = rng.normal(0.0, 0.001, size=n)
                a = _project_simplex(a + jitter)

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine mixes Adam optimisation, an LP‑based
    Frank‑Wolfe update, projected sub‑gradient descent and a final hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # --------- time handling -------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)   # reserve a few seconds
    deadline = start + total_budget

    # --------- initialise from the previous best if available ---------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.sum() == 0.0:
            prev[:] = 1.0 / prev.size
        best_a = prev / prev.sum()
        best_score = _objective_score(best_a)
        print(f"starting from previous best: {best_score:.6f}")
    else:
        best_a = None
        best_score = np.inf

    # --------- main optimisation loop (random restarts) --------------------
    while time.monotonic() < deadline:
        # pick a length – bias towards larger vectors but keep some diversity
        n = int(rng.integers(1200, 8000 + 1))

        # initialise either from the global best (jittered) or from scratch
        if best_a is not None and rng.random() < 0.35:
            # interpolate the current best onto the new size and add a tiny jitter
            old_x = np.linspace(0.0, 1.0, best_a.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a0 = np.interp(new_x, old_x, best_a)
            a0 = np.clip(a0, 0.0, None)
            a0 += rng.normal(0.0, 0.0008, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            # fresh Dirichlet‑like draw (highly non‑uniform)
            a0 = rng.exponential(scale=1.0, size=n)
            a0 /= a0.sum()

        # --------------------------------------------------------------
        # Phase 1 – Adam on a smooth surrogate
        # --------------------------------------------------------------
        a_adam, score_adam = _adam_opt(a0, rng, deadline, max_iters=3500)
        if score_adam < best_score - 1e-12:
            best_score = score_adam
            best_a = a_adam.copy()
            print(f"[{time.monotonic() - start:6.2f}s] new best after Adam {best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Optional LP / Frank‑Wolfe step (only for moderate n)
        # --------------------------------------------------------------
        if n <= 2000:
            a_fw, score_fw = _frank_wolfe_opt(a_adam, rng, deadline, max_rounds=2)
            if score_fw < best_score - 1e-12:
                best_score = score_fw
                best_a = a_fw.copy()
                print(f"[{time.monotonic() - start:6.2f}s] new best after FW {best_score:.6f}")
        else:
            a_fw, score_fw = a_adam, score_adam  # keep the Adam result for the next phases

        # --------------------------------------------------------------
        # Phase 3 – Projected sub‑gradient descent
        # --------------------------------------------------------------
        a_sub, score_sub = _subgradient_opt(a_fw, rng, deadline, max_iters=1800)
        if score_sub < best_score - 1e-12:
            best_score = score_sub
            best_a = a_sub.copy()
            print(f"[{time.monotonic() - start:6.2f}s] new best after sub‑grad {best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 4 – Hill‑climbing refinement
        # --------------------------------------------------------------
        a_hc, score_hc = _hill_climb_opt(a_sub, rng, deadline, max_iters=1300)
        if score_hc < best_score - 1e-12:
            best_score = score_hc
            best_a = a_hc.copy()
            print(f"[{time.monotonic() - start:6.2f}s] new best after hill‑climb {best_score:.6f}")

        # sanity check – stop early if the budget is nearly exhausted
        if time.monotonic() > deadline - 0.7:
            break

    # ------------------------------------------------------------------
    # Return the best sequence (safeguarded against the evaluator limits)
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – uniform distribution of a modest size
        n = 3000
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)

    return [float(v) for v in best_a]
