    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5

    # Choose n based on computational tradeoff
    n = 8000
    init_seq = height_sequence_1 if exists else None
    if init_seq:
        old = np.asarray(init_seq, dtype=np.float64)
        old_n = old.size
        if old_n != n:
            # upsample to n using linear interpolation
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, old)
        else:
            a = old.copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    else:
        a = np.full(n, 1.0 / n)

    # Precompute FFT length for convolution
    fft_len = _next_pow2(2 * n - 1)

    # Define projection onto simplex
    def _project_simplex(v):
        # Implementation using sorting descending
        # returns array project onto nonnegative simplex sum=1
        v_sorted = np.sort(v)[::-1]
        cssv = np.cumsum(v_sorted) - 1
        rho = np.where(v_sorted - cssv / np.arange(1, n + 1) > 0)[0]
        if rho.size == 0:
            theta = 0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        # Ensure sum=1 due to rounding errors
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0 / n)
        return w

    # Main loop
    best = a.copy()
    # compute best score
    conv = _self_conv(best, fft_len)
    best_score = 2.0 * n * conv.max()
    # track best history
    last_improve = time.monotonic()
    # parameters
    start = time.monotonic()
    total = deadline - start
    # Step size schedule
    step_init = 0.5   # maybe larger
    step_end = 0.001
    # Norm of gradient may vary.

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total
        step = step_init * (1 - frac) + step_end * frac

        # convolution and max index
        conv = _self_conv(a, fft_len)
        idx = int(np.argmax(conv))

        # compute gradient
        grad = np.zeros_like(a)
        # shift indices where idx - j are valid
        js = np.arange(n)
        ks = idx - js
        mask = (ks >= 0) & (ks < n)
        grad[mask] = 2.0 * a[ks[mask]]

        # Subgradient descent
        a_new = a - step * grad
        # Project onto simplex
        a = _project_simplex(a_new)

        # Evaluate score every maybe 10 iterations? but we can evaluate each time
        conv = _self_conv(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:

            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (sgd) = {best_score:.6f}")

        # occasional random perturbation to escape local minima
        if rng.random() < 0.01:
            # add small random noise and reproject
            noise = rng.normal(0.0, 0.1, size=n)
            a_pert = _project_simplex(a + noise * step)
            a = a_pert

        # restarts if stagnated
        if now - last_improve > 30.0:

            jitter = rng.normal(0.0, 0.5, size=n)

            last_improve = now
            print(f"[{elapsed:6.2f}s] restart from best due to stagnation, score {best_score:.6f}")

    # Return best
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

"""
propose_candidate
-----------------
A two‑phase optimizer for the “max‑convolution” objective.

Phase I – Adam on a *sub‑gradient* of the true max‑convolution.
  * a is represented as soft‑max of unrestricted log‑weights `x`.
  * For the current `a` we compute the self‑convolution `c`.
  * The index where `c` attains its maximum gives a sub‑gradient
    `g_a = 2 * a shifted by that index` (the exact gradient of the max term).
  * This gradient is transferred to the log‑weights (`g_x`) using the
    usual soft‑max chain rule and an Adam update is performed.
  * If no improvement for ~20 s the run is jittered (log‑normal noise)
    to escape stagnation.

Phase II – greedy hill‑climbing on the exact objective.
  * Small amounts of mass are moved between random coordinates;
    a move is kept only if it strictly lowers the true bound.
  * Occasionally a larger contiguous‑block perturbation is tried.
  * After a long stagnation a local restart around the current best is done.

The routine respects the supplied time budget (≈ 1000 s) and works
with any initial sequence stored in the global `height_sequence_1`.
The final sequence is clipped to the limits imposed by `evaluate_sequence`
and returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a 1‑D array using an FFT of length ``fft_len``.
    Result length is ``2*len(a) - 1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {w | w_i ≥ 0, Σ w_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to guard against rounding errors
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and re‑normalise.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _move_mass(a: np.ndarray, rng: np.random.Generator, step_frac: float) -> np.ndarray:
    """
    Transfer a small fraction of the mass from one index to another.
    The returned vector is re‑normalised to sum to 1.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = step_frac * rng.random() * a[i]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling – keep a few seconds for the harness
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a length and an initial point
    # ------------------------------------------------------------------
    # A somewhat larger length gives more freedom; 8000 is a good compromise.
    n = 8000

    # Try to initialise from a previously discovered sequence, otherwise use uniform.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        old = np.asarray(init_seq, dtype=np.float64)
        if old.ndim == 1 and old.size > 0:
            # Simple linear interpolation to the new length
            old_n = old.size
            if old_n != n:
                x_old = np.linspace(0, 1, old_n)
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, old)
            else:
                a = old.copy()
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s
        else:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    # Work with log‑weights to keep the simplex constraints implicit
    x = np.log(a + 1e-12)          # log‑weights for soft‑max

    # ------------------------------------------------------------------
    # 2) FFT length (pre‑computed for speed)
    # ------------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # ------------------------------------------------------------------
    # 4) Book‑keeping of the best solution
    # ------------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    a = _a_from_x(x)
    c = _self_conv(a, fft_len)
    cur_score = 2.0 * n * c.max()          # exact objective (sum(a)=1)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 5) Hyper‑parameters (simple linear schedules)
    # ------------------------------------------------------------------
    lr_start = 0.5
    lr_end = 0.01

    # ------------------------------------------------------------------
    # 6) Phase I – sub‑gradient Adam on the true max‑convolution
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.60 * total_time   # ~60 % of the budget

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # current probability vector
        a = _a_from_x(x)

        # convolution and the index where it peaks
        conv = _self_conv(a, fft_len)
        max_val = conv.max()
        idx = int(np.argmax(conv))

        # --- sub‑gradient of the max‑convolution w.r.t. a -----------------
        # For each j with 0 ≤ idx‑j < n we have ∂c_idx/∂a_j = 2 * a_{idx‑j}
        grad_a = np.zeros_like(a)
        start = max(0, idx - (n - 1))
        end = min(n, idx + 1)          # idx inclusive
        # Populate the slice in one vectorised operation
        src = a[idx - np.arange(start, end)]
        grad_a[start:end] = 2.0 * src

        # Chain rule to obtain the gradient for the log‑weights
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # --- Adam update -------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # exact objective – keep the best ever seen
        score = 2.0 * n * max_val
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (phase I) = {best_score:.6f}")

        # ---- stagnation handling ----------------------------------------
        if now - last_improve > 20.0:
            # jitter around the best solution (log‑normal noise)
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # occasional global jitter (small chance each iteration)
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0

    # ------------------------------------------------------------------
    # 7) Phase II – greedy hill‑climbing on the exact bound
    # ------------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    phase2_start = time.monotonic()
    phase2_total = deadline - phase2_start

    # Linear decay of the fraction of mass that may be moved in a single step
    def _step_fraction(elapsed):
        return 0.30 * (1.0 - elapsed / phase2_total) + 0.01

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        step_frac = _step_fraction(elapsed)

        # ---- simple mass move -------------------------------------------
        cand = _move_mass(a, rng, step_frac)
        conv_cand = _self_conv(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best (phase II) = {best_score:.6f}")

        # ---- block‑perturbation (larger exploratory jumps) -------------
        if rng.random() < 0.015:
            a_block = _random_block_perturb(a, rng)
            conv_block = _self_conv(a_block, fft_len)
            block_score = 2.0 * n * conv_block.max()
            if block_score < cur_score - 1e-12:
                a = a_block
                cur_score = block_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = now
                    print(f"[{now - start_time:6.2f}s] new best (block) = {best_score:.6f}")

        # ---- stagnation handling (local restart) -------------------------
        if now - last_improve > 30.0:
            # re‑initialise locally around the current best
            noise = rng.normal(0.0, 0.4, size=n)
            x = np.log(best + 1e-12) + noise
            a = _a_from_x(x)
            cur_score = 2.0 * n * _self_conv(a, fft_len).max()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] local‑restart (stagnation) – score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 8) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
