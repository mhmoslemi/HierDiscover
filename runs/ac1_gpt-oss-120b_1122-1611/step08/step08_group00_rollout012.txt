import time
import numpy as np
from scipy.optimize import linprog

def normalize(v):
    """Scale vector v to have L1 norm 1. If sum is zero return uniform."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

def compute_J(a):
    """Return the exact objective value defined by evaluate_sequence,
    but without clipping. a must be a 1‑D numpy array."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(f):
    """Return the dense (2n‑1)×n matrix that maps a vector g
    to the linear convolution f * g."""
    n = f.shape[0]
    conv_len = 2 * n - 1
    A = np.empty((conv_len, n), dtype=np.float64)
    # column i of A is f shifted by i positions
    for i in range(n):
        A[i:i + n, i] = f
    return A

def _random_block_perturb(a, rng):

    followed by an L1‑projection."""

    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return normalize(out)

    The method builds a linear‐program to increase the L1 norm while keeping the
    current worst self‑convolution unchanged, then mixes the resulting vector
    back into the current iterate.  It alternates LP steps with cheap
    random‑block perturbations and restarts when progress stalls.

    # 0) Choose problem size (try to reuse a known construction when present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)

    f = normalize(base)

    best_f = f.copy()
    best_J = compute_J(best_f)
    print(f"initial n={n}, J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 1) Main optimisation loop – LP expansion + mixing
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    last_improve = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # --------------------------------------------------------------
        # a) current max self‑convolution M
        # --------------------------------------------------------------
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # --------------------------------------------------------------
        # b) linear program: maximise 1ᵀg  st   f * g ≤ M ,  g ≥ 0
        # --------------------------------------------------------------

        conv_len = 2 * n - 1
        b = np.full(conv_len, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # minimise –sum(g)
        bounds = [(0.0, None)] * n

        # we use the HiGHS interior‑point solver – it copes well with dense data
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})

            print(f"[{iteration}] LP failed ({res.message}); applying jitter")
            f = _random_block_perturb(f, rng)
            continue

        sum_g = g.sum()

        if sum_g <= 1.0 + 1e-9:

            if time.monotonic() - last_improve > 20.0:

                f = _random_block_perturb(f, rng)
                last_improve = time.monotonic()

        # --------------------------------------------------------------

        # d) decide whether to accept the LP point or a convex blend
        # --------------------------------------------------------------

        J_mix = compute_J(f_mix)

        if J_g < best_J:
            best_J = J_g
            best_f = g_norm.copy()
            f = g_norm.copy()
            last_improve = time.monotonic()

        # --------------------------------------------------------------

            f = _random_block_perturb(f, rng)

    # ------------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)

"""
LP‑driven refinement for the “step‑function” challenge.

The algorithm works on the scale‑invariant formulation of the objective:
    J(a) = 2·n·max(a*a)   (we keep L1(a)=1, so the denominator disappears).

Given a current vector `f` we compute its worst self‑convolution `M`.
We then solve the linear programme

        maximise   1ᵀg
        subject to  f * g ≤ M         (element‑wise, linear convolution)
                    g ≥ 0

which finds a vector `g` that can carry a larger L1 mass while still respecting
the same convolution bound.  Scaling `g` back to L1=1 yields a new candidate
`ĝ`.  We either adopt `ĝ` directly or move a fraction `t` towards it
(`t` grows with the L1‑gain).  The best vector seen so far is kept.

When the LP stops providing a gain we inject a small random block‑perturbation
to escape plateaus; after a period of stagnation we restart with another
jitter.  The whole search respects the 1000 s budget (5 s safety margin) and
returns the best sequence found.

The implementation uses only NumPy and SciPy (no file or network I/O) and
contains only top‑level functions, as required.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def normalize(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the L1‑simplex (non‑negative, sum = 1)."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def compute_J(a: np.ndarray) -> float:
    """
    Exact value of the objective used by ``evaluate_sequence``.
    The vector a must be non‑negative; the function does *not* clip the values.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(f: np.ndarray) -> np.ndarray:
    """
    Return the dense (2n‑1)×n matrix that maps a vector g to the linear
    convolution ``f * g``.
    Column i of the matrix is the vector f shifted by i places.
    """
    n = f.size
    conv_len = 2 * n - 1
    A = np.empty((conv_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block,
    followed by an L1‑simplex projection."""

    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return normalize(out)

# Main optimisation routine
# ----------------------------------------------------------------------

                     **kwargs) -> list[float]:

    ``evaluate_sequence``.  The routine runs a sequence of linear‑programming
    expansions followed by simple mixing and occasional random jitter.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Choose the problem size – reuse a known construction if possible
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        # keep the vector at a size that stays affordable for dense LPs
        if base.shape[0] > 3000:

    else:

            print(f"[{iteration:4d}] blended   J={best_J:.6f}")

    # --------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)
    return [float(x) for x in best_f.tolist()]
