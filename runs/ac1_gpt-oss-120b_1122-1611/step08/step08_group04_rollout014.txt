"""
Hybrid optimiser for the self‑convolution (max‑autocorrelation) problem.

The routine runs several independent restarts.  Each restart consists of:

  1️⃣  Adam optimisation on a smooth (log‑sum‑exp) surrogate of the
      max‑convolution.  The surrogate is annealed by slowly increasing the
      temperature β so that we approach the true ℓ∞‑norm.

  2️⃣  Projected sub‑gradient descent on the exact max‑convolution.
      The sub‑gradient of the active offset is cheap to compute;
      an adaptive step‑size together with occasional Gaussian jitter
      keeps the iterate moving.

  3️⃣  A lightweight random‑pair local search (move a tiny amount of mass
      from one index to another) that can squeeze out the last few
      percent of improvement.

All iterates stay on the probability simplex (non‑negative, unit sum) by
projection.  The best sequence ever seen across *all* restarts is returned.
The implementation respects the 1000 s time‑budget (5 s safety margin)
and works with any initial sequence stored in the global variable
`height_sequence_1`.
"""

import time
import numpy as np
from scipy.special import logsumexp

# --------------------------------------------------------------
# Helper utilities – plain functions, no closures
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full a * a convolution using real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (sum(a) is assumed 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())   # denominator is 1 because sum(a)=1


# --------------------------------------------------------------
# Optimisation building blocks
# --------------------------------------------------------------

def _adam_phase(a0: np.ndarray,
                rng: np.random.Generator,
                time_limit: float,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(conv).  Returns the best
    vector encountered and its exact score.
    """
    n = a0.shape[0]
    # logits (log‑softmax) representation – always stays in ℝⁿ
    eps = 1e-12
    x = np.log(a0 + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # Adam hyper‑parameters
    lr = 0.45
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # surrogate temperature schedule
    beta_start = 5.0
    beta_end = 2000.0
    beta_ramp = 4000          # iterations over which β is linearly increased

    fft_len_conv = _next_pow2(2 * n - 1)
    it = 0
    best_a = np.exp(x - np.max(x))
    best_a = best_a / best_a.sum()
    best_score = _objective_score(best_a)

    while time.monotonic() < start_time + time_limit:
        it += 1

        # ---- current probability vector
        a = _softmax_logits(x)

        # ---- exact convolution (needed for evaluation)
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        # ---- log‑sum‑exp surrogate of the max
        beta = beta_end if it > beta_ramp else beta_start + (beta_end - beta_start) * it / beta_ramp
        scaled = beta * conv
        lse = logsumexp(scaled)          # robust log‑sum‑exp
        w = np.exp(scaled - lse)         # soft‑max weights, sum to 1

        # ---- gradient of the surrogate w.r.t. a (correlation w ⋆ a)
        fft_len_corr = _next_pow2(3 * n - 2)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)

        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # ---- chain‑rule to obtain gradient w.r.t. logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # ---- Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to shake out plateaus
        if it % 500 == 0:
            x += rng.normal(0.0, 0.02, size=n)

    return best_a, best_score


def _subgradient_phase(a0: np.ndarray,
                       rng: np.random.Generator,
                       time_limit: float,
                       start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector seen and its exact score.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    # start with a modest step size
    step = 0.12
    min_step = 1e-8

    while time.monotonic() < start_time + time_limit:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active offset (see analysis)
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # try a basic projected step
        cand = _project_simplex(a - step * grad)
        cand_score = _objective_score(cand)

        if cand_score < best_score - 1e-12:
            # successful step – keep the move and accelerate a little
            a = cand
            best_a = cand.copy()
            best_score = cand_score
            step = min(step * 1.07, 1.0)
        else:
            # failed step – shrink the learning rate
            step = max(step * 0.65, min_step)

        # occasional random jitter to escape flat plateaux
        if step < 2e-4 or rng.random() < 0.001:
            jitter = rng.normal(0.0, 0.003, size=n)
            a = _project_simplex(a + jitter)
            step = max(step, 0.01)

    return best_a, best_score


def _local_random_search(a0: np.ndarray,
                         rng: np.random.Generator,
                         time_limit: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple pair‑wise mass‑transfer hill‑climbing.  Very cheap, it can
    improve the final objective by a few per‑myriad.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)

    # start with a relatively large transfer amount, then decay
    eps = 0.04
    decay = 0.995

    while time.monotonic() < start_time + time_limit:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] <= 1e-15:
            continue

        # move a small fraction from i to j (preserving the sum)
        delta = min(eps * a[i], a[i] * 0.5)
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        cand_score = _objective_score(a_candidate)
        if cand_score < best_score - 1e-13:
            a = a_candidate
            best_a = a_candidate.copy()
            best_score = cand_score
            # keep epsilon as is – successful moves usually mean we are still in a good neighbourhood
        else:
            # unsuccessful – shrink the move size a little
            eps = max(eps * decay, 1e-6)

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point required by the harness
# --------------------------------------------------------------

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Several restarts are performed; each restart runs
    an Adam‑phase on a smooth surrogate, a projected sub‑gradient phase on the
    true max‑convolution and a final random‑pair local search.  The best
    sequence found within the time budget is returned (clipped to the limits
    of ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)          # 5 s safety margin
    deadline = start + total_budget

    # --------------------------------------------------------------
    # Global best across all restarts
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        base = base / base.sum()
        n0 = max(3800, base.size)               # guarantee a reasonably large vector
        if base.size != n0:
            # upscale by linear interpolation
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n0, endpoint=False)
            a0 = np.interp(new_x, old_x, base)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
        else:
            a0 = base.copy()
    else:
        n0 = 3840
        a0 = np.full(n0, 1.0 / n0, dtype=np.float64)

    best_global_a = a0.copy()
    best_global_score = _objective_score(best_global_a)
    print(f"[{time.monotonic() - start:6.2f}s] start score {best_global_score:.6f}")

    # --------------------------------------------------------------
    # Determine number of restarts (we keep it modest – time is limited)
    # --------------------------------------------------------------
    max_restarts = 4
    for restart_idx in range(max_restarts):
        now = time.monotonic()
        remaining_time = deadline - now
        if remaining_time < 30.0:        # not enough time for another full restart
            break

        # allocate a fair share of the remaining budget to this restart
        # (half for Adam, half for sub‑gradient + local search)
        phase1_budget = remaining_time * 0.55
        phase2_budget = remaining_time * 0.45

        # ----------------------------------------------------------
        # Choose a different initialisation for each restart
        # ----------------------------------------------------------
        if restart_idx == 0:
            a_start = best_global_a.copy()               # warm start from current global best
        elif restart_idx == 1:
            # uniform start (same length as the best so far)
            n = best_global_a.size
            a_start = np.full(n, 1.0 / n, dtype=np.float64)
        elif restart_idx == 2:
            # random Dirichlet (continuous “random” distribution)
            n = best_global_a.size
            a_start = rng.random(n)
            a_start = a_start / a_start.sum()
        else:
            # simple triangular shape – more mass in the centre
            n = best_global_a.size
            half = n // 2
            left = np.linspace(1.0, 2.0, half, endpoint=False)
            right = np.linspace(2.0, 1.0, n - half, endpoint=False)
            a_start = np.concatenate([left, right])
            a_start = np.clip(a_start, 0.0, None)
            a_start = a_start / a_start.sum()

        # ----------------------------------------------------------
        # Phase 1 – Adam on a smooth surrogate
        # ----------------------------------------------------------
        a_phase1, score_phase1 = _adam_phase(
            a_start, rng, phase1_budget, time.monotonic()
        )
        if score_phase1 < best_global_score - 1e-12:
            best_global_score = score_phase1
            best_global_a = a_phase1.copy()
            print(f"[{time.monotonic() - start:6.2f}s] restart {restart_idx} Adam improvement {best_global_score:.6f}")

        # ----------------------------------------------------------
        # Phase 2 – Projected sub‑gradient descent on the exact max
        # ----------------------------------------------------------
        a_phase2, score_phase2 = _subgradient_phase(
            a_phase1, rng, phase2_budget * 0.6, time.monotonic()
        )
        if score_phase2 < best_global_score - 1e-12:
            best_global_score = score_phase2
            best_global_a = a_phase2.copy()
            print(f"[{time.monotonic() - start:6.2f}s] restart {restart_idx} subgrad improvement {best_global_score:.6f}")

        # ----------------------------------------------------------
        # Phase 3 – Light‑weight random pair swaps (local hill‑climb)
        # ----------------------------------------------------------
        a_phase3, score_phase3 = _local_random_search(
            a_phase2, rng, phase2_budget * 0.4, time.monotonic()
        )
        if score_phase3 < best_global_score - 1e-12:
            best_global_score = score_phase3
            best_global_a = a_phase3.copy()
            print(f"[{time.monotonic() - start:6.2f}s] restart {restart_idx} local search improvement {best_global_score:.6f}")

        # Safety check – do not overshoot the global deadline
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # Finalise – clip to the limits imposed by evaluate_sequence()
    # --------------------------------------------------------------
    best_global_a = np.clip(best_global_a, 0.0, 1000.0)
    return [float(v) for v in best_global_a]
