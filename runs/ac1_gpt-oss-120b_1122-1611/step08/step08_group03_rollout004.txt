"""
Propose a step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – start from the given ``height_sequence_1`` if
   available, otherwise from a uniform vector.  The vector is always
   normalised to sum = 1 (the evaluation is scale invariant).

2. **Smooth‑max Adam** – replace the hard max by a temperature‑controlled
   log‑sum‑exp, compute the exact gradient via FFTs and perform Adam
   updates while projecting onto the probability simplex.  The temperature
   and learning‑rate follow an exponential annealing schedule.  This stage
   quickly lowers the large‑peak region of the autocorrelation.

3. **Exact‑max sub‑gradient descent** – after the surrogate has been
   annealed the true peak location is known.  The sub‑gradient of the
   max‑convolution is obtained analytically (a simple shift of the current
   sequence).  A diminishing step‑size gradient step is taken and the
   iterate is projected back onto the simplex.  This stage sharpens the
   solution and often pushes the bound below the target.

4. **Stochastic local refinement** – final fine‑tuning by tiny mass‑transfer
   moves (pick two indices, move a small fraction of the source mass to the
   target) and occasional jitter.  Exact evaluation via ``evaluate_sequence``
   is used to keep the best solution seen.

The routine respects a 1000 s budget (with a 5 s safety margin) and prints
progress.  The returned list is clipped to the evaluator limits.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full convolution of a with itself using FFT.
    Returns array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv_full[: 2 * a.size - 1]


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the evaluation objective assuming the sequence sum = 1."""
    return 2.0 * n * np.max(conv)

# Main search routine
# ----------------------------------------------------------------------

    Search for a step‑function (list of non‑negative heights) that

    The algorithm uses a smooth‑max Adam optimiser, an exact‑max
    sub‑gradient phase and a stochastic local refinement.

        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[i] * frac_move

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta   # sum stays 1, non‑negativity guaranteed

        # fast objective
        conv_cand = _conv_fft(a_candidate, fft_len)
        cand_obj = _objective_from_conv(conv_cand, n)

        local_iter += 1
        if cand_obj < best_obj - 1e-12:
            best_obj = cand_obj
            best = a_candidate.copy()
            a = a_candidate
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_obj:.6f} "
                  f"(move #{local_iter})")
            if best_obj <= 1.5025:
                print("[early stop] target reached during local phase")
                break

        # occasional jitter to keep diversity
        if local_iter % 4000 == 0:
            width = rng.integers(1, max(2, n // 50))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            a_j = a.copy()
            a_j[start:start + width] *= factor
            a = _project_onto_simplex(a_j)

    # --------------------------------------------------------------
    # 6) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = float(evaluate_sequence(best.tolist()))  # sanity check
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
