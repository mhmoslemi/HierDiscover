"""
Improved search for a low‑value step‑function.
================================================

The goal is to minimise

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

over non‑negative vectors a (the ”heights”).  The function is
scale‑invariant as long as no entry is clipped at the limits
[0,1000] imposed by ``evaluate_sequence``, therefore we keep the
vector normalised to a unit sum.

The algorithm consists of three stages:

1. **Diverse initialisation** –  several Dirichlet draws with a small
   concentration (α≈0.5) produce sparse candidates.  The best of them
   becomes the start for the long optimisation run.

2. **Smoothed‑max Adam optimisation** – we minimise a smooth
   approximation of the supremum using a log‑sum‑exp surrogate.
   The temperature ``γ`` is raised from 0.1 to 200, while the Adam
   learning‑rate decays from 0.5 to 0.001.  Whenever progress stalls
   for a while we restart around the current best (Gaussian jitter).

3. **Simulated‑annealing hill‑climbing** – the exact objective is
   evaluated after every move.  Mass is transferred between two
   random indices (the *step fraction* shrinks linearly).  Moves that
   improve the score are always kept; worsening moves are accepted
   with a temperature‑controlled probability, which helps to escape
   shallow local minima.  Periodic block‑perturbations create larger
   jumps.

All work is done on the simplex (sum = 1) and all numbers stay far
below the clipping bound, so ``evaluate_sequence`` would return the
same value as the cheap internal computation.
The routine respects the 1000 s budget (with a 5 s safety margin) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – must stay at top‑level
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution using an FFT of length ``fft_len``.
    Result length is 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate with respect to the probability
    vector ``a``.  ``w`` are the soft‑max weights over the 2·n‑1 convolution
    positions (∑w = 1).  The gradient formula is 2·corr(a_rev, w).
    """
    n = a.size
    a_rev = a[::-1]
    fa = np.fft.rfft(a_rev, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    corr = np.fft.irfft(fa * fw, n=fft_len)
    # Only the slice that aligns with the convolution indices is relevant
    start = n - 1
    return 2.0 * corr[start:start + n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_dirichlet(rng: np.random.Generator, n: int, alpha: float = 0.5) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector and normalise to unit sum."""
    draws = rng.gamma(alpha, 1.0, size=n)
    s = draws.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return draws / s


def _move_mass(a: np.ndarray, rng: np.random.Generator, step_frac: float) -> np.ndarray:
    """
    Transfer a small share of mass from one index to another.
    Returns a new normalised vector.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = step_frac * rng.random() * a[i]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out /= out.sum()
    return out


def _block_perturb(a: np.ndarray, rng: np.random.Generator, width_max: int = None) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor,
    then re‑normalise.  ``width_max`` limits the block size (default: n//5).
    """
    n = a.size
    if width_max is None:
        width_max = max(2, n // 5)
    width = rng.integers(1, width_max + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) time handling – keep a small safety margin for the harness
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 1) choose a length – a few thousand entries give a good trade‑off
    # --------------------------------------------------------------
    n = 4096                                      # power‑of‑two for fast FFTs

    # --------------------------------------------------------------
    # 2) pre‑compute FFT lengths (self‑convolution and gradient step)
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for self‑convolution
    fft_len_grad = _next_pow2(3 * n - 2)          # for the gradient step

    # --------------------------------------------------------------
    # 3) initial population – take the best of a few Dirichlet draws
    # --------------------------------------------------------------
    best = None
    best_score = np.inf
    for _ in range(8):                           # cheap multi‑start
        a0 = _random_dirichlet(rng, n, alpha=0.5)
        c0 = _self_conv(a0, fft_len_self)
        sc0 = 2.0 * n * c0.max()                 # exact objective (sum=1)
        if sc0 < best_score:
            best_score = sc0
            best = a0.copy()
    # keep a reference to the current iterate
    a = best.copy()

    # --------------------------------------------------------------
    # 4) Adam optimiser state (log‑weights)
    # --------------------------------------------------------------
    x = np.log(a + 1e-15)                         # log‑softmax parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # --------------------------------------------------------------
    # 5) hyper‑parameter schedules
    # --------------------------------------------------------------
    gamma_start = 0.1
    gamma_end = 200.0
    lr_start = 0.5
    lr_end = 0.001

    # --------------------------------------------------------------
    # 6) Phase I – smooth‑max Adam optimisation (≈ 70 % of the budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.70 * total_time
    last_improve = start_time

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # current probability vector
        a = _softmax(x)

        # convolution
        conv = _self_conv(a, fft_len_self)

        # smooth‑max surrogate (log‑sum‑exp)
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        shift = conv - conv.max()                # numerical stability
        w = np.exp(gamma * shift)
        w /= w.sum()                             # soft‑max over 2·n‑1 positions

        # gradient wrt a (surrogate)
        g_a = _gradient_wrt_a(a, w, fft_len_grad)

        # chain rule to obtain gradient wrt log‑weights x
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # exact objective – update global best
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (phase I) = {best_score:.6f}")

        # restart if we are stuck for ~12 s
        if now - last_improve > 12.0:
            # Gaussian jitter around the current best
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # occasional global block perturbation (≈2 % of iterations)
        if rng.random() < 0.02:
            a_jit = _block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)

    # --------------------------------------------------------------
    # 7) Phase II – simulated‑annealing hill‑climbing (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    phase2_start = time.monotonic()
    phase2_total = deadline - phase2_start

    # temperature schedule – start relatively hot, finish cold
    def _temperature(elapsed):
        """Linear cooling from T_start to T_end."""
        T_start = 0.12
        T_end = 1e-5
        return T_start * (1.0 - elapsed / phase2_total) + T_end

    # step‑size schedule for mass moves
    def _step_fraction(elapsed):
        """From 0.3 down to 0.01 as time proceeds."""
        return 0.3 * (1.0 - elapsed / phase2_total) + 0.01

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        T = _temperature(elapsed)
        step_frac = _step_fraction(elapsed)

        # ------- propose a cheap mass‑transfer move -----------------
        cand = _move_mass(a, rng, step_frac)
        conv_cand = _self_conv(cand, fft_len_self)
        cand_score = 2.0 * n * conv_cand.max()

        # ------- Metropolis acceptance ----------------------------
        improve = cand_score < cur_score
        accept = improve or (rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)))

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best (phase II) = {best_score:.6f}")

        # ------- once in a while try a larger block perturb --------
        if rng.random() < 0.015:
            a_blk = _block_perturb(a, rng)
            conv_blk = _self_conv(a_blk, fft_len_self)
            blk_score = 2.0 * n * conv_blk.max()
            if blk_score < cur_score:
                a = a_blk
                cur_score = blk_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = now
                    print(f"[{now - start_time:6.2f}s] new best (block) = {best_score:.6f}")

        # ------- stagnation handling – local restart ---------------
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.4, size=n)
            x = np.log(best + 1e-12) + noise
            a = _softmax(x)                       # fresh start around the best
            cur_score = 2.0 * n * _self_conv(a, fft_len_self).max()
            last_improve = now
            print(
                f"[{now - start_time:6.2f}s] local‑restart (stagnation) – score {cur_score:.6f}"
            )

    # --------------------------------------------------------------
    # 8) Return the best sequence (already within the evaluation limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
