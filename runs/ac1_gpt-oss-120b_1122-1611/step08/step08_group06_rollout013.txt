from scipy.optimize import linprog

def solve_lp_max_sum(a, M):
    n = len(a)
    A_ub = build_conv_matrix(a)  # csr
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None

The algorithm is a two‑stage optimiser.

1. **Adam stage** – we minimise a smooth surrogate of the true
   objective.  
   * Phase 1 flattens the self‑convolution (minimise its variance).  
   * Phase 2 replaces the hard `max` by a log‑sum‑exp with a
     temperature that slowly rises, driving the solution towards the
     smallest possible maximum value.  
   The parameters (learning‑rate schedule, Adam β’s, temperature
   schedule, restart rules) are tuned to explore the search space
   aggressively while staying inside the time budget.

2. **Simulated‑annealing finetuning** – after the Adam phase the
   optimisation landscape is already very flat.  A cheap Monte‑Carlo
   local search that perturbs the log‑weights and accepts uphill moves
   with a decaying temperature is able to squeeze out the last
   fractions of improvement that the gradient descent cannot see.

Both phases keep the *best* sequence ever observed and return it at the
end.  All numbers are kept non‑negative, the vector is normalised to a
unit sum (the objective is scale invariant) and finally clipped to the
range required by `evaluate_sequence`.

"""
Hybrid Adam + simulated‑annealing optimiser for the step‑function
problem.

The objective is

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear convolution of `a` with itself.
The optimiser works in three stages:

* Phase 1 – variance flattening (minimise Σ (conv – avg)²).
* Phase 2 – smooth‑max minimisation using a log‑sum‑exp surrogate.
* Phase 3 – local simulated‑annealing that perturbs the log‑weights
  and accepts uphill moves with a temperature that decays to zero.

All updates are performed in the log‑softmax (probability‑simplex)
parameterisation, guaranteeing non‑negativity and unit sum.
The best sequence seen during the whole run is returned.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = rfft(a, n=fft_len)
    fc = fa * fa
    conv = irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements the formula   ∂/∂a_i Σ_k w_k (a*a)_k = 2·(a_rev * w)_i .
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = rfft(a_rev, n=N)
    fw = rfft(w, n=N)
    conv = irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max yielding a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that gives a
    low value of `evaluate_sequence`.  The routine respects the time
    budget (seconds) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose sequence length and initialise log‑weights
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64).ravel()
        if a0.size == 0:
            n = 5000
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
        x = np.log(a0 + 1e-12)                     # log‑weights
    else:
        n = 5000
        x = np.zeros(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT size for the self‑convolution
    # ------------------------------------------------------------------

    # 3) Adam optimiser state (shared by the two gradient phases)
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Helper converting log‑weights to a probability vector
    # ------------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # ------------------------------------------------------------------
    # 5) Tracking the best solution seen so far
    # ------------------------------------------------------------------
    a = _a_from_x(x)
    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * c.max() / (a.sum() ** 2)   # exact objective
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 6) Phase split (percentages of the remaining budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * total_time   # variance flattening
    phase2_end = start_time + 0.80 * total_time   # smoothed‑max
    # the remaining ~20 % will be used for simulated‑annealing

    phase = 1
    print("[0.00s] start optimisation – n =", n)

    # ------------------------------------------------------------------
    # 7) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # phase handling
        # --------------------------------------------------------------
        if phase == 1 and now >= phase1_end:
            phase = 2
            # reset Adam momentum for the new loss landscape
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] switch to smoothed‑max phase")

        if phase == 2 and now >= phase2_end:
            phase = 3
            # no Adam state needed any more
            print(f"[{elapsed:6.2f}s] switch to simulated‑annealing finetune")
            # initialise simulated‑annealing temperature
            temp = 0.1
            temp_decay = 0.9995                # per iteration decay
            sigma = 0.05                       # proposal standard deviation
            # keep a copy of x for the SA stage
            x_sa = x.copy()
            a_sa = a.copy()
            cur_score = best_score
            continue  # jump straight to the SA block below

        # --------------------------------------------------------------
        # common: obtain current a and its self‑convolution
        # --------------------------------------------------------------
        a = _a_from_x(x)                     # probability vector (sum = 1)

        if cur_score < best_score - 1e-12:

            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # choose loss and its gradient
        # --------------------------------------------------------------
        if phase == 1:
            # ------- Phase 1 : variance flattening (target = constant) --------
            avg = 1.0 / (2 * n - 1)                 # because Σ a = 1
            r = c - avg
            g_a = 2.0 * _gradient_wrt_a(a, r)       # gradient of Σ r^2
        elif phase == 2:
            # ------- Phase 2 : smoothed max (log‑sum‑exp) --------------------
            phase_frac = (now - phase1_end) / max(phase2_end - phase1_end, 1e-12)
            gamma_start = 0.5
            gamma_end = 80.0                         # a larger final temperature
            gamma = gamma_start + (gamma_end - gamma_start) * phase_frac

            # stable log‑sum‑exp weights
            c_shift = c - np.max(c)
            w = np.exp(gamma * c_shift)
            w /= w.sum()
            g_a = _gradient_wrt_a(a, w)              # gradient of smooth max
        else:
            # ----------------------------------------------------------------
            # Phase 3 : simulated‑annealing (handled separately, break loop)
            # ----------------------------------------------------------------
            break

        # --------------------------------------------------------------
        # chain rule: from a‑gradient to x‑gradient (soft‑max Jacobian)
        # --------------------------------------------------------------
        avg_g = np.dot(a, g_a)                      # ⟨a,∂F/∂a⟩
        g_x = a * (g_a - avg_g)                     # ∂F/∂x

        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1 - beta1) * g_x

        lr_end = 0.02

        x -= step

        # --------------------------------------------------------------

        if now - last_improve > 30.0:

            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now

        if rng.random() < 0.02:

            m.fill(0.0)
            v.fill(0.0)
            t = 0

        temp_decay = 0.9995

            a_prop = _a_from_x(x_prop)
            c_prop = _self_conv(a_prop, fft_len_self)
            prop_score = 2.0 * n * c_prop.max() / (a_prop.sum() ** 2)

            # Metropolis acceptance
            if prop_score < cur_score - 1e-12:
                accept = True
            else:
                # uphill move with probability exp(-(Δ)/temp)
                accept = rng.random() < np.exp(-(prop_score - cur_score) / max(temp, 1e-12))

                a_sa = a_prop

                    last_improve = now
                    print(f"[{elapsed:6.2f}s] SA new best = {best_score:.6f}")

            sigma *= 0.9998   # slowly shrink proposal size

            if now - last_improve > 40.0:

                a_sa = _a_from_x(x_sa)
                cur_score = best_score
                last_improve = now
                print(f"[{elapsed:6.2f}s] SA restart (stagnation)")

    # ------------------------------------------------------------------
    # 8) Return the best sequence (clip to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
