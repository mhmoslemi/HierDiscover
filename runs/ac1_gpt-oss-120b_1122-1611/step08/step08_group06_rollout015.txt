"""
search for a non‑negative step function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm has two stages.

Stage 1 – fast Adam optimisation on a log‑softmax parametrisation.
The loss is a log‑sum‑exp surrogate of the true maximum; the
temperature γ is raised from 0.5 to 30 so the optimiser first
flattens the whole convolution and then focuses on the true peak.

Stage 2 – a very cheap simulated‑annealing “mass‑transfer’’ walk.
Starting from the best vector of stage 1 we repeatedly move a small
fraction of the mass from a source index to a random destination
(indexes are sampled proportionally to the current mass).  A linear
temperature schedule decides whether a worsening move is accepted.
Sudden large perturbations (Dirichlet draws) are injected occasionally
to escape shallow basins.

Both stages respect the global 1000 s budget (5 s safety margin) and
the best sequence seen so far is returned at the end.  All numbers are
clamped to the limits used by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (step‑function) that gives a low
    value of ``evaluate_sequence``.  The routine spends roughly half of the
    time performing Adam optimisation on a smooth max‑surrogate and the
    rest on a simple simulated‑annealing mass‑transfer walk.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initialise a probability vector (soft‑max domain)
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0.0 else np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        n = 5000                       # default size – a compromise between speed and quality
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a0
        n = a.size

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 3) Adam optimiser state (used in phase 1)
    # --------------------------------------------------------------
    x = np.log(a + 1e-12)               # log‑weights (soft‑max domain)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.85
    beta2 = 0.995
    eps = 1e-8
    t = 0

    # --------------------------------------------------------------
    # 4) Tracking the best solution
    # --------------------------------------------------------------
    a = _softmax(x)
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = float(2 * n * conv.max() / (a.sum() ** 2))
    last_improve = time.monotonic()

    # --------------------------------------------------------------
    # 5) Phase split (50 % Adam → 50 % simulated annealing)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase2_start = start_time + 0.5 * total_time
    phase2 = False

    # --------------------------------------------------------------
    # 6) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # switch to phase 2 when the time budget says so
        # ----------------------------------------------------------
        if not phase2 and now >= phase2_start:
            phase2 = True
            print(f"[{elapsed:6.2f}s] ── entering simulated‑annealing phase ──")

        # ----------------------------------------------------------
        # Phase 1 – Adam on a smooth max surrogate
        # ----------------------------------------------------------
        if not phase2:
            a = _softmax(x)                     # probability vector, sum = 1
            conv = _self_conv(a, fft_len_self)  # length 2·n‑1

            # current exact score (for reporting)
            cur_score = float(2 * n * conv.max() / (a.sum() ** 2))

            # logging of improvement
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f}")

            # ---- smooth‑max loss -------------------------------------------------
            # linearly increase γ from 0.5 to 30 during the whole phase 1
            phase_frac = (now - start_time) / max(phase2_start - start_time, 1e-12)
            gamma = 0.5 + (30.0 - 0.5) * phase_frac

            c_shift = conv - np.max(conv)
            w = np.exp(gamma * c_shift)
            w /= w.sum()                         # probability weights over the 2n‑1 bins

            # gradient of the smooth max w.r.t. a
            g_a = _gradient_wrt_a(a, w)          # shape (n,)

            # ------------------------------------------------------
            # chain rule to obtain gradient w.r.t. log‑weights x
            # ------------------------------------------------------
            avg_g = np.dot(a, g_a)                # ⟨a,∂F/∂a⟩
            g_x = a * (g_a - avg_g)               # ∂F/∂x (softmax Jacobian)

            # ------------------------------------------------------
            # Adam step
            # ------------------------------------------------------
            t += 1
            m = beta1 * m + (1 - beta1) * g_x
            v = beta2 * v + (1 - beta2) * (g_x * g_x)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            # linear decay of the learning rate over the whole budget
            lr_start = 0.4
            lr_end = 0.02
            lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            x -= step

            # ------------------------------------------------------
            # stagnation handling – restart if no improvement for 30 s
            # ------------------------------------------------------
            if now - last_improve > 30.0:
                noise = rng.normal(0.0, 0.5, size=n)
                x = np.log(best + 1e-12) + noise
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                last_improve = now
                print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

            continue   # to the next iteration of the outer loop

        # ----------------------------------------------------------
        # Phase 2 – simple simulated‑annealing mass‑transfer walk
        # ----------------------------------------------------------
        # a already holds the best vector from phase 1 (sum = 1)
        # Propose a tiny transfer of probability mass from a source index
        # to a destination index.
        src = rng.choice(n, p=a / a.sum())
        dst = rng.integers(0, n)
        if src != dst and a[src] > 0.0:
            max_delta = a[src]
            delta = rng.random() * 0.2 * max_delta           # move up to 20 % of the source mass
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta

            score_new = evaluate_sequence(a_new.tolist())

            phase2_elapsed = now - phase2_start
            phase2_total = deadline - phase2_start

                cur_score = score_new

                if score_new < best_score:
                    best_score = score_new
                    best = a.copy()
                    print(f"[{now-start_time:6.2f}s] SA new best = {best_score:.6f}")

        # occasional large random jump (helps escape deep minima)

            a = rng.dirichlet(np.ones(n))
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{now-start_time:6.2f}s] global jump improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clip to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
