"""
Subgradient / exponentiated‑gradient optimiser for the
max‑convolution objective.

The algorithm works in three stages (all respect the simplex
constraint ∑ a_i = 1, which makes the objective scale‑invariant):

1. **Exponentiated‑gradient phase** – minimise a smooth surrogate of
   max (a∗a) by updating log‑probabilities with the exact sub‑gradient
   of the (hard) max.  This yields fast, global movement away from the
   uniform distribution.

2. **Additive sub‑gradient phase** – use the true sub‑gradient of
   max (a∗a) (computed from the active convolution indices) and perform
   projected additive steps with a diminishing step‑size.

3. **Stochastic local refinement** – tiny directed mass‑transfer moves
   (source ≈ highest‑gradient index, target ≈ lowest‑gradient index)
   are accepted if they improve the exact score.  Occasional jitter
   avoids getting stuck.

All stages stop well before the total budget so that the harness can
collect the result.  Progress and intermediate best scores are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k, the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad

# ----------------------------------------------------------------------

    The method combines exponentiated‑gradient, additive sub‑gradient,
    and stochastic local refinement.
    """

    if "height_sequence_1" in globals():

        a = np.clip(a, 0.0, None)

    # --------------------------------------------------------------
    best = a.copy()
    # exact score (2 * n * max_b) – identical to evaluate_sequence for sum=1
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – exponentiated‑gradient optimisation (≈50 % of time)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.50 * (deadline - start_time)
    iteration = 0
    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        # pick a few of the highest convolution entries (keeps active set small)
        top_k = np.argpartition(-conv, 3)[:3]          # three indices with largest values
        grad = _gradient_from_active(a, top_k)

        # exponentiated‑gradient update on log‑probabilities
        eta = 0.6 / np.sqrt(iteration + 1.0)           # diminishing step
        log_a = np.log(a + 1e-15)                     # keep away from log(0)
        log_a -= eta * grad
        a = np.exp(log_a)
        a /= a.sum()                                  # projection onto simplex

        # evaluate
        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (exp‑grad) = {best_score:.6f} (iter={iteration})")
            if best_score <= 1.5025:
                print("[early stop] target reached in phase 1")
                break

        iteration += 1
        if iteration % 200 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] phase 1 iter={iteration}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – additive sub‑gradient descent (≈30 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    phase2_end = start_time + 0.85 * (deadline - start_time)
    iteration = 0
    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        # active set = all indices within a tiny absolute tolerance of the max
        eps = max_b * 1e-6
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        grad = _gradient_from_active(a, active)

        # additive projected sub‑gradient step
        step = 0.12 / np.sqrt(iteration + 1.0)
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate)

        # quick score from the convolution (no need for evaluate_sequence)
        conv_cand = _conv_fft(a_candidate, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        # accept if the move reduced the objective
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (sub‑grad) = {best_score:.6f} (iter={iteration})")
            if best_score <= 1.5025:
                print("[early stop] target reached in phase 2")
                break

        # keep the move only when it does not increase the objective;
        # otherwise stay where we are
        if score_cand <= 2.0 * n * max_b:
            a = a_candidate

        iteration += 1
        if iteration % 250 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] phase 2 iter={iteration}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Phase 3 – stochastic local refinement (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    iteration_local = 0
    while time.monotonic() < deadline:
        # recompute convolution and active index only every few steps
        if iteration_local % 30 == 0:
            conv = _conv_fft(a, fft_len)
            max_b = conv.max()
            idx_max = int(np.argmax(conv))
            # gradient for the single active index – gives a “tight‑spot” map
            grad_active = _gradient_from_active(a, np.array([idx_max]))
            src = int(np.argmax(grad_active))
            # pick a target among the smallest‑gradient entries
            min_grad = grad_active.min()
            cand_tgt = np.where(grad_active <= min_grad + 1e-12)[0]
            if cand_tgt.size > 1:
                tgt = int(rng.choice(cand_tgt))
            else:
                # fall back to a random low‑gradient position
                low_candidates = np.where(grad_active < np.percentile(grad_active, 15))[0]
                tgt = int(rng.choice(low_candidates)) if low_candidates.size else int(rng.integers(0, n))

        # propose a tiny directed mass transfer
        if a[src] <= 0.0:
            src = int(rng.integers(0, n))
        if src == tgt:
            tgt = int(rng.integers(0, n))

        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))   # 0.1 %–5 %
        delta = a[src] * frac

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta
        a_candidate = _project_onto_simplex(a_candidate)

        # exact score via the fast convolution (same as evaluate_sequence for sum=1)
        conv_cand = _conv_fft(a_candidate, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (move #{iteration_local})")
            if best_score <= 1.5025:
                print("[early stop] target reached in local phase")
                break

        # keep improvement, otherwise stay
        if score_cand < 2.0 * n * max_b:
            a = a_candidate

        iteration_local += 1

        # occasional jitter to shake the solution
        if iteration_local % 4000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jitter = a.copy()
            a_jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jitter)

    # --------------------------------------------------------------
    # 6) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
