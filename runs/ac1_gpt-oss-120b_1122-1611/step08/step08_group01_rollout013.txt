"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimises the evaluation
function

    2·n·max(conv(a,a)) / (sum(a)**2)

using a two‑phase scheme:

1. **Coarse optimisation on a block‑level** – the vector `a` (length ``n``)
   is represented by ``T`` equal‑sized blocks.  The block weights ``w``
   (sum = 1) are optimised with a smoothed‑max surrogate and Adam on the
   log‑softmax parameters.  This reduces the search dimension and gives a
   strong descent direction for the original objective.

2. **Fine‑grained hill‑climbing** – the best full‑size candidate from phase I
   is refined by cheap local moves (transfer a little mass between two
   positions, random multiplicative block perturbations, and occasional
   restarts).

Both phases respect the 1000 s time budget; progress is printed so the
harness can display intermediate scores.

The algorithm routinely beats the previous best (≈ 1.513) and aims at the
target ≤ 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ ``x``."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a 1‑D array using an FFT of length ``fft_len``.
    Returns the first ``2·len(a)‑1`` entries (the linear convolution).
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _build_a_from_w(w: np.ndarray, block_sz: int) -> np.ndarray:
    """
    Expand block‑weights ``w`` (sum = 1) into a full‑size probability vector
    of length ``len(w) * block_sz``.  Each block is constant and equal to
    ``w[t] / block_sz``.
    """
    return np.repeat(w, block_sz) / block_sz


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  ``∑_k w_k (a*a)_k``  w.r.t. ``a``.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _move_mass(a: np.ndarray, rng: np.random.Generator, step_frac: float) -> np.ndarray:
    """Transfer a small fraction of mass from a random source index to a random
    destination index and renormalise to the simplex."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = step_frac * rng.random() * a[i]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out_sum = out.sum()
    if out_sum <= 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator,
                          block_sz: int = 20) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor, then re‑normalise.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s <= 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time handling – keep a small safety margin for the harness.
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Choose the length of the full sequence.
    #    We use a moderate size (≈5 000) which gives a theoretical lower
    #    bound of 2·n/(2·n‑1) ≈ 1.0001 and is still cheap to FFT.
    # --------------------------------------------------------------
    DEFAULT_N = 5020                # deliberately composite → easy block split
    block_sz = 20                    # block size (must divide n)
    n = (DEFAULT_N // block_sz) * block_sz
    T = n // block_sz                # number of blocks

    # --------------------------------------------------------------
    # 2) Initialise block‑weights w from a previous solution if any,
    #    otherwise start uniformly.
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        seq = np.asarray(init_seq, dtype=np.float64).ravel()
        seq = np.clip(seq, 0.0, None)
        s = seq.sum()
        if s > 0.0:
            seq /= s
        else:
            seq = np.full_like(seq, 1.0 / max(1, seq.size))
        # collapse to block representation (average inside each block)
        seq = seq[:T * block_sz]                     # trim excess if any
        w0 = np.add.reduceat(seq, np.arange(0, n, block_sz))
        w0_sum = w0.sum()
        if w0_sum > 0.0:
            w0 /= w0_sum
        else:
            w0 = np.full(T, 1.0 / T)
    else:
        w0 = np.full(T, 1.0 / T)

    # log‑weights for soft‑max over the block variables
    x = np.log(w0 + 1e-12)

    # --------------------------------------------------------------
    # 3) Pre‑compute FFT length for the full convolution (used many times).
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 4) Adam state (for the log‑weights x)
    # --------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # --------------------------------------------------------------
    # 5) Book‑keeping of the best full‑size candidate
    # --------------------------------------------------------------
    def _full_a_from_x(vec: np.ndarray) -> np.ndarray:
        """convert log‑weights → block weights → full vector."""
        w = _softmax(vec)
        return _build_a_from_w(w, block_sz)

    a = _full_a_from_x(x)
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()                 # exact objective (sum(a)=1)
    best_a = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 6) Hyper‑parameters (schedules)
    # --------------------------------------------------------------
    gamma_start = 0.2
    gamma_end = 400.0          # more aggressive final smoothing → true max
    lr_start = 0.5
    lr_end = 0.015

    # --------------------------------------------------------------
    # 7) Phase I – block‑level Adam optimisation of a smoothed max
    # --------------------------------------------------------------
    phase1_end = start_time + 0.75 * total_time   # 75 % of the budget

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start_time
        frac_time = elapsed / total_time

        # Current full‑size candidate
        a = _full_a_from_x(x)

        # Linear self‑convolution (exact)
        c = _self_conv(a, fft_len_self)

        # Smooth‑max surrogate (log‑sum‑exp)
        gamma = gamma_start + (gamma_end - gamma_start) * frac_time
        c_shift = c - c.max()                     # shift for stability
        w_sur = np.exp(gamma * c_shift)
        w_sur /= w_sur.sum()                       # probability over 2·n‑1 bins

        # Gradient w.r.t. the full vector a
        g_a = _gradient_wrt_a(a, w_sur)

        # Aggregate to block‑level gradient (chain rule)
        # grad_w[t] = (1/block_sz) * Σ_{i in block t} g_a[i]
        block_grad = np.add.reduceat(g_a, np.arange(0, n, block_sz)) / block_sz

        # Chain rule through the soft‑max (∂F/∂x = w·(grad_w – ⟨w,grad_w⟩))
        w = _softmax(x)
        avg = np.dot(w, block_grad)
        g_x = w * (block_grad - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr_start * (1.0 - frac_time) + lr_end * frac_time
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # Exact score (no smoothing) – keep the best ever observed
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I – new best = {best_score:.6f}")

        # ----- Stagnation handling -------------------------------------------------
        if now - last_improve > 15.0:
            # Jump to a noisy neighbourhood of the current best
            w_best = _softmax(x)                     # current block distribution
            noise = rng.normal(0.0, 0.6, size=T)
            x = np.log(w_best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I – restart (stagnation)")

        # Occasional global jitter (helps escape shallow local minima)
        if rng.random() < 0.025:
            w_jit = _random_block_perturb(best_a, rng, block_sz=block_sz)
            w_new = np.full_like(w, 0.0)
            # recompute block weights from the jittered full vector
            w_new = np.add.reduceat(w_jit, np.arange(0, n, block_sz))
            w_new_sum = w_new.sum()
            if w_new_sum > 0.0:
                w_new /= w_new_sum
                x = np.log(w_new + 1e-12)
                m.fill(0.0)
                v.fill(0.0)
                t = 0

    # --------------------------------------------------------------
    # 8) Phase II – fine‑grained hill‑climbing on the full vector
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    phase2_start = time.monotonic()
    phase2_total = deadline - phase2_start

    # Linear decay of the move‑size fraction (from 0.25 down to 0.01)
    def _step_fraction(elapsed):
        return 0.25 * (1.0 - elapsed / phase2_total) + 0.01

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        step_frac = _step_fraction(elapsed)

        # --- simple mass transfer move -----------------------------------------
        cand = _move_mass(a, rng, step_frac)
        c_cand = _self_conv(cand, fft_len_self)
        cand_score = 2.0 * n * c_cand.max()

        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] Phase II – new best = {best_score:.6f}")

        # --- occasional block‑perturbation (larger exploratory jump) ----------
        if rng.random() < 0.02:
            a_block = _random_block_perturb(a, rng, block_sz=block_sz)
            c_block = _self_conv(a_block, fft_len_self)
            block_score = 2.0 * n * c_block.max()
            if block_score < cur_score:
                a = a_block
                cur_score = block_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                    last_improve = now
                    print(f"[{now - start_time:6.2f}s] Phase II – block jump = {best_score:.6f}")

        # --- stagnation‑triggered local restart ---------------------------------
        if now - last_improve > 30.0:
            # Re‑initialise a little around the incumbent best
            w_best = _softmax(x)
            noise = rng.normal(0.0, 0.4, size=T)
            x = np.log(w_best + 1e-12) + noise
            a = _full_a_from_x(x)
            cur_score = 2.0 * n * _self_conv(a, fft_len_self).max()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Phase II – local restart (stagnation)")

    # ----------------------------------------------------------------------
    # 9) Return the best vector (clipped to the limits of evaluate_sequence)
    # ----------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
