def softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float,
                     n: int, fft_len_grad: int) -> np.ndarray:
    # compute weights w (softmax over conv)
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # reverse w
    w_rev = w[::-1]
    # convolution a * w_rev (linear conv)
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1
    grad = 2.0 * full[start_idx:start_idx + n]
    return grad
```

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – Use the length of the previously found sequence
   (``height_sequence_1``) if it exists, otherwise fall back to 4096.
2. **Global search** – A small Differential‑Evolution (DE) population
   explores the simplex.  A diverse portfolio of seeds (uniform, several
   exponentials, linear ramps, Dirichlet draws, the previous best) is used.
3. **Local refinement** – The best DE individual is refined with a
   projected Adam optimiser that minimises a soft‑max surrogate of the
   max‑convolution objective.  The temperature is increased from a low
   value (smooth) to a very high value (sharp) so that the optimiser
   converges to a true sub‑gradient step on the max.
4. **Fine‑grained hill‑climbing** – A cheap pair‑wise mass‑transfer
   procedure performs small “coin‑flip” moves that can still improve the
   objective after the Adam phase.
5. **Time‑budget handling** – The routine respects the 1000 s budget
   (5 s safety margin) and always returns the best vector seen.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    then project onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Portfolio of diverse starting points (all already normalised to sum = 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays (coarse sweep, later the optimiser will fine‑tune)
    for r in [0.999, 0.997, 0.995, 0.99, 0.975, 0.95, 0.9]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(8):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    return cand


# ----------------------------------------------------------------------
# Objective helpers (all arguments are explicit – no closures)
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """Return the true objective J = 2·n·max(a*a) for a normalised vector."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def compute_conv_and_J(a: np.ndarray,
                       n: int,
                       fft_len_self: int,
                       conv_len: int):
    """
    Return the full self‑convolution (length 2·n‑1) and the objective J.
    """
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    maxv = conv.max()
    J = float(2.0 * n * maxv)
    return conv, J


def softmax_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     n: int,
                     fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the temperature‑controlled soft‑max surrogate:
    g = 2·a * w_rev , where w is the soft‑max over the convolution values.
    """
    # soft‑max weights
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # reversed weights
    w_rev = w[::-1]

    # convolution a * w_rev (linear)
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start = w_rev.size - 1                # = 2·n‑2
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock budget
    (5 s safety margin) and always returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = min(max(n, 512), 4096)          # keep FFTs cheap
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for a * w_rev

    # --------------------------------------------------------------
    # 2) Differential Evolution (global exploration)
    # --------------------------------------------------------------
    popsize = 50
    # ---- initialise a diverse portfolio --------------------------------
    base_cands = _initial_candidates(n, rng)

    # include the previous best if it exists
    if "height_sequence_1" in globals():
        base_cands.append(
            proj_simplex(
                np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
            )
        )

    # fill the pool up to popsize
    while len(base_cands) < popsize:
        base_cands.append(rng.dirichlet(np.ones(n) * 0.5))

    # random selection of distinct initial individuals
    idxs = rng.choice(len(base_cands), size=popsize, replace=False)
    population = [proj_simplex(base_cands[i].astype(np.float64)) for i in idxs]
    scores = [compute_J(v, n, fft_len_self, conv_len) for v in population]

    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]

    # DE hyper‑parameters
    F = 0.8
    CR = 0.9

    # allocate ≈55 % of the remaining budget to DE
    time_left = deadline - time.monotonic()
    de_budget = time_left * 0.55
    de_deadline = time.monotonic() + de_budget

    while time.monotonic() < de_deadline:
        for i in range(popsize):
            # three distinct donors (excluding i)
            donors = rng.choice([j for j in range(popsize) if j != i],
                                size=3, replace=False)
            r1, r2, r3 = donors
            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = np.clip(mutant, 0.0, None)
            s = mutant.sum()
            if s == 0.0:
                mutant[:] = 1.0 / n
            else:
                mutant /= s

            # binomial crossover
            mask = rng.random(n) < CR
            if not np.any(mask):                     # guarantee at least one gene
                mask[rng.integers(n)] = True
            trial = np.where(mask, mutant, population[i])
            trial = proj_simplex(trial)

            trial_score = compute_J(trial, n, fft_len_self, conv_len)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] DE new best J = {best_score:.6f}")

        # safety break
        if time.monotonic() >= de_deadline:
            break

    # --------------------------------------------------------------
    # 3) Projected Adam refinement (soft‑max surrogate)
    # --------------------------------------------------------------
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.monotonic()

    # allocate the rest of the budget to Adam (plus a tiny final hill‑climb)
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)          # ∈[0,1]

        # learning‑rate schedule (exponential decay)
        lr_start, lr_end = 0.30, 0.001
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # temperature schedule (soft‑max becomes sharper)
        alpha_start, alpha_end = 5.0, 5000.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # compute current convolution and Jacobian weights
        conv, cur_J = compute_conv_and_J(a, n, fft_len_self, conv_len)

        # gradient of the smoothed surrogate
        grad = softmax_gradient(a, conv, alpha, n, fft_len_grad)

        # Adam update + projection onto the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # evaluate the true (non‑smoothed) objective
        _, cur_J = compute_conv_and_J(a, n, fft_len_self, conv_len)

        if cur_J < best_score - 1e-12:
            best_score = cur_J
            best_vec = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Adam new best J = {best_score:.6f}")

        # occasional jitter to escape plateaus
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # restart if no improvement for a while
        if now - last_improve > 20.0:
            a = best_vec.copy()
            # tiny multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Fine‑grained pairwise hill‑climbing (local polishing)
    # --------------------------------------------------------------
    a = best_vec.copy()
    cur_score = best_score
    # run for at most 10 s or until the global budget is exhausted
    hill_start = time.monotonic()
    while time.monotonic() < deadline and (time.monotonic() - hill_start) < 10.0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small amount of mass from i to j
        max_transfer = min(a[i], 0.005)               # keep steps modest
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        cand_score = compute_J(a_candidate, n, fft_len_self, conv_len)
        if cand_score < cur_score - 1e-12:
            a = a_candidate
            cur_score = cand_score
            best_vec = a_candidate.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] pairwise improvement J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
