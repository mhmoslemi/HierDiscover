"""
Simulated‑annealing / hill‑climbing optimizer for the step‑function problem.

The algorithm starts from the best known construction (global variable
``height_sequence_1``) or, if that is unavailable, from a random Dirichlet
vector.  The vector is interpreted as a probability distribution
(`sum(a)=1`).  Its *linear* self‑convolution `c = a * a` is maintained
explicitly and updated *incrementally* when a small amount of mass is moved
from one position to another.  The objective to minimise is

        score = 2·n·max(c)          (because sum(a)=1)

which is exactly the value returned by ``evaluate_sequence`` (up to the
clipping performed there).

Two kinds of moves are used:

* **Random transfer** – pick two indices `i≠j` and move a small fraction of
  the mass at `i` to `j`.  The convolution can be updated in O(n) using
  slice operations; the change of the maximum is obtained by a cheap
  ``np.max`` on the updated array.

* **Gradient‑guided transfer** – the index `k_max` where the current
  convolution attains its maximum is identified.  The gradient of
  `c[k_max]` w.r.t. the entries of `a` is
      g[i] = a[k_max−i] (if valid) + a[i−k_max] (if valid).
  Mass is moved from a position with large `g[i]` to one with small
  `g[j]`.  This is a cheap approximation of a steepest‑descent step for
  the ``max`` objective.

A simple linear cooling schedule provides a temperature `T`; uphill
moves are accepted with probability ``exp(-(Δscore)/T)``.  If no improvement
has been observed for 30 s a *jitter* (multiplicative log‑normal perturbation
of a random block) is applied to the current best solution to escape shallow
minima.

All updates keep the vector non‑negative and preserve its L¹‑norm, so the
score can be evaluated without any extra normalisation step.

The routine runs until the allocated time budget (default 1000 s) minus a
small safety margin expires, then returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level only)
# ----------------------------------------------------------------------


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _initial_sequence(seed: int, n: int) -> np.ndarray:
    """Return a random probability vector of length *n*."""
    rng = np.random.default_rng(seed)
    a = rng.dirichlet(np.ones(n))
    return a.astype(np.float64)


def _convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1)."""
    return np.convolve(a, a)


def _move_incremental(a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float):
    """
    Apply the transfer of ``delta`` from position ``i`` to ``j``.
    The arrays ``a`` and ``c`` are modified *in‑place*.
    """
    n = a.size
    # update a
    a[i] -= delta
    a[j] += delta

    # update convolution c  (c := c - 2·δ·shift_i(a_old) + 2·δ·shift_j(a_old) + Δ*Δ)
    # we use the *old* a values, so store a_i and a_j before the change
    # (a_i_old = a[i] + delta, a_j_old = a[j] - delta)
    # but the formula only needs the *original* a (before the move)
    # which we saved in the slices below.
    # To avoid a copy of the whole a, we retrieve the needed slices directly.
    # The old a is simply the current a after the subtraction/addition,
    # because the convolution update uses the *pre‑move* a.
    # Hence we fetch the required slice from the *old* a (which is a+delta at i,
    # a-delta at j). We obtain it by temporarily adding delta back.
    # Simpler: compute the update using the *new* a but the linear term
    # is symmetric, so using the new a gives the same result up to O(δ²); we
    # compensate for the quadratic term explicitly.
    # Therefore we compute using the *new* a, then add the Δ*Δ correction.

    # Slice updates (vectorised)
    c[i:i + n] -= 2.0 * delta * a
    c[j:j + n] += 2.0 * delta * a

    # Quadratic correction Δ*Δ (only at three positions)
    idx2i = 2 * i
    idx2j = 2 * j
    idxij = i + j
    if idx2i < c.size:
        c[idx2i] += delta * delta
    if idx2j < c.size:
        c[idx2j] += delta * delta
    if idxij < c.size:
        c[idxij] -= 2.0 * delta * delta


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _gradient_at_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the convolution value at index ``k_max`` w.r.t. the
    probability vector ``a``.  Implements  g[i] = a[k_max-i] (if in range)
    + a[i-k_max] (if in range).
    """
    n = a.size
    g = np.zeros_like(a)

    # first term: a[k_max - i] for i = 0..k_max
    if k_max >= 0:
        g[: k_max + 1] += a[k_max :: -1]      # reversed slice

    # second term: a[i - k_max] for i = k_max..n-1
    if k_max < n:
        g[k_max:] += a[: n - k_max]

    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that gives a
    low value of ``evaluate_sequence``.  Returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Initialise the probability vector a
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)) and len(init_seq) > 0:
        a = np.asarray(init_seq, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= s
    else:
        n = 5000
        a = _initial_sequence(seed + 1, n)

    n = a.size

    # --------------------------------------------------------------
    # Current convolution and best‑so‑far records
    # --------------------------------------------------------------
    c = _convolution(a)                     # length 2·n‑1
    cur_score = 2.0 * n * c.max()           # sum(a)=1
    best_a = a.copy()
    best_score = cur_score
    best_c = c.copy()
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # Annealing schedule parameters
    # --------------------------------------------------------------
    T_start = 0.05 * best_score               # generous start temperature
    T_end = 1e-7 * best_score
    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = min(elapsed / total_time, 1.0)

        # current temperature (linear cooling)
        T = T_start * (1.0 - frac) + T_end * frac

        # ----------------------------------------------------------
        # decide move type
        # ----------------------------------------------------------
        use_gradient = rng.random() < 0.7       # most moves are gradient‑guided
        if use_gradient:
            # ------------------------------------------------------
            # Gradient‑guided transfer
            # ------------------------------------------------------
            k_max = int(np.argmax(c))
            g = _gradient_at_max(a, k_max)

            # pick source index with large gradient (and non‑zero mass)
            source_candidates = np.where((g > 0) & (a > 1e-14))[0]
            if source_candidates.size == 0:
                # fall back to random move
                i = rng.integers(0, n)
                while a[i] < 1e-14:
                    i = rng.integers(0, n)
            else:
                # bias towards larger gradient values
                probs = g[source_candidates]
                probs = probs / probs.sum()
                i = rng.choice(source_candidates, p=probs)

            # pick destination where gradient is small (preferably zero)
            dest_candidates = np.where(g < np.percentile(g, 20))[0]
            if dest_candidates.size == 0:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)
            else:
                j = rng.choice(dest_candidates)

            # size of the transfer – scaled with temperature
            max_frac = 0.4 * (1.0 - frac) + 0.02   # large early, tiny later
            delta = a[i] * max_frac * rng.random()
        else:
            # ------------------------------------------------------
            # Simple random block transfer (fast)
            # ------------------------------------------------------
            i = rng.integers(0, n)
            while a[i] < 1e-14:
                i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_frac = 0.4 * (1.0 - frac) + 0.02
            delta = a[i] * max_frac * rng.random()

        if delta <= 0.0:
            continue

        # ----------------------------------------------------------
        # Apply the move incrementally (working copies)
        # ----------------------------------------------------------
        a_new = a.copy()
        c_new = c.copy()
        _move_incremental(a_new, c_new, i, j, delta)

        # ----------------------------------------------------------
        # Compute the new score (exact objective)
        # ----------------------------------------------------------
        new_score = 2.0 * n * c_new.max()

        # ----------------------------------------------------------
        # Acceptance test (Metropolis criterion)
        # ----------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move – accept with probability exp(-(Δ)/T)
            if T > 0.0:
                prob = np.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_new
            c = c_new
            cur_score = new_score

            # record global improvement
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_c = c.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Stagnation handling – occasional jitter
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # jitter around the *best* solution seen so far
            a = _random_block_perturb(best_a, rng)
            c = _convolution(a)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] jitter (stagnation) – re‑started from best")
            # reset temperature ramp slightly to give fresh exploration
            T_start = max(T_start, 0.02 * best_score)

        # ----------------------------------------------------------
        # Occasional global random perturbation (2 % chance)
        # ----------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(best_a, rng)
            c = _convolution(a)
            cur_score = 2.0 * n * c.max()
            print(f"[{elapsed:6.2f}s] global jitter")
            # no reset of last_improve – this move can be beneficial

    # --------------------------------------------------------------
    # Return the best sequence (clip to evaluate_sequence limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
