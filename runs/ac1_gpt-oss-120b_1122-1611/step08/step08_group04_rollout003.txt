"""
Hybrid multi‑restart optimiser for the self‑convolution L∞ problem.

* Phase 0 – generate a few diverse seeds (uniform, triangular,
  arcsine‑like, random Dirichlet and the previous best if available).
* Phase 1 – for each seed run a short Adam optimiser on a
  log‑sum‑exp surrogate of the max‑convolution.
  The surrogate parameter β is slowly annealed up to a large value
  (≈2000) so the optimiser sees an increasingly sharper approximation
  of the true L∞‑norm.
* Phase 2 – take the best candidate from phase 1 and polish it with
  a projected sub‑gradient method on the exact max‑convolution.
  A simple back‑tracking line‑search adapts the step size, and occasional
  jitter helps to escape flat plateaux.
* The whole routine respects the global time budget (default 1000 s) and
  prints progress messages.  The final sequence is clipped to the limits
  enforced by `evaluate_sequence` and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping logits → probability simplex."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """Full self‑convolution a*a using real FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (no clipping performed)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------


def _adam_optimize(
    init_a: np.ndarray,
    time_limit: float,
    *,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
    beta_ramp_iters: int = 3000,
    lr: float = 0.4,
    beta1: float = 0.9,
    beta2: float = 0.999,
    adam_eps: float = 1e-8,
    jitter_sigma: float = 0.015,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Run Adam on the soft‑max surrogate until `time_limit` seconds pass.
    Returns the best candidate (as a probability vector) and its score."""
    n = init_a.size
    eps = 1e-12
    x = np.log(init_a + eps)                     # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # pre‑allocated FFT lengths
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n - 2)

    best_a = init_a.copy()
    best_score = _objective_score(best_a)
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_limit:
        it += 1
        # ---- probability vector from logits
        a = _softmax_logits(x)

        # ---- exact convolution (needed for the objective)
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max                      # denominator = 1 (sum(a)=1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---- smooth surrogate (log‑sum‑exp with temperature β)
        beta = beta_end if it > beta_ramp_iters else (
            beta_start + (beta_end - beta_start) * it / beta_ramp_iters
        )
        scaled = beta * conv_full
        lse = logsumexp(scaled)          # robust log‑sum‑exp
        w = np.exp(scaled - lse)         # shape (2n‑1,), sums to 1

        # ---- gradient of surrogate w.r.t. a  (correlation w ⋆ a)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)
        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx : start_idx + n]   # ∂/∂a of the surrogate

        # ---- chain rule: logits → a (soft‑max)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # ---- Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            x += rng.normal(0.0, jitter_sigma, size=n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_refine(
    init_a: np.ndarray,
    time_limit: float,
    *,
    rng: np.random.Generator,
    step_init: float = 0.12,
    step_inc: float = 1.05,
    step_dec: float = 0.70,
    jitter_prob: float = 0.001,
    jitter_sigma: float = 0.002,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with simple back‑tracking line search."""
    n = init_a.size
    a = init_a.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    step = step_init
    start = time.monotonic()
    fft_len = _next_pow2(2 * n - 1)

    while time.monotonic() - start < time_limit:
        # exact convolution
        conv = _conv_self(a, fft_len=fft_len)
        kmax = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # --- try a step with back‑tracking
        improved = False
        trial_step = step
        for _ in range(12):                     # limited number of back‑track steps
            cand = _project_simplex(a - trial_step * grad)
            cand_score = _objective_score(cand)
            if cand_score < best_score - 1e-12:
                # success
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                step = trial_step * step_inc   # grow a bit
                improved = True
                break
            else:
                trial_step *= step_dec          # shrink and retry

        if not improved:
            # no improvement – keep current a and shrink step permanently
            step = max(step * step_dec, 1e-9)

        # occasional random jitter to shake out of flat regions
        if step < 1e-8 or rng.random() < jitter_prob:
            jitter = rng.normal(0.0, jitter_sigma, size=n)
            a = _project_simplex(a + jitter)
            step = max(step, step_init)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine performs several random restarts of a smooth‑Adam phase
    followed by a projected sub‑gradient polishing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Global time management (5 s safety margin for the harness)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start_time + total_budget

    # --------------------------------------------------------------
    # Choose a problem size -------------------------------------------------
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        base = base / base.sum()
        n = max(3800, base.size)          # ensure a sufficiently large vector
        if base.size != n:
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            base = np.interp(new_x, old_x, base)
            base = np.clip(base, 0.0, None)
            base = base / base.sum()
        warm_start = base.copy()
    else:
        n = 3840
        warm_start = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Prepare a collection of diverse seeds
    # --------------------------------------------------------------
    seeds = []

    # 1) the previous best (if any)
    seeds.append(warm_start)

    # 2) uniform distribution
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) triangular (symmetric hat)
    tri = np.arange(n, dtype=np.float64)
    tri = np.minimum(tri, n - 1 - tri) + 1.0
    tri = tri / tri.sum()
    seeds.append(tri)

    # 4) arcsine‑like – heavy near the edges, ≈ 1/√(x(1‑x))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    seeds.append(arcs)

    # 5) random Dirichlet‑like (gamma draw)
    gam = rng.gamma(shape=0.5, scale=1.0, size=n)
    gam = gam / gam.sum()
    seeds.append(gam)

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on several seeds
    # --------------------------------------------------------------
    # allocate roughly 55 % of the budget to phase 1, split evenly
    phase1_budget = total_budget * 0.55
    per_restart = phase1_budget / len(seeds)

    best_overall_a = None
    best_overall_score = np.inf

    print(f"[{time.monotonic() - start_time:6.2f}s] start phase‑1, n={n}")

    for idx, seed_a in enumerate(seeds):
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        limit = min(per_restart, remaining)
        a_opt, score_opt = _adam_optimize(
            seed_a,
            limit,
            rng=rng,
        )
        print(
            f"[{time.monotonic() - start_time:6.2f}s] restart {idx+1}/{len(seeds)} "
            f"phase‑1 best score {score_opt:.6f}"
        )
        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_a = a_opt.copy()

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient refinement
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 0.5 and best_overall_a is not None:
        # give at most ~40 % of total time to sub‑gradient, but do not exceed remaining
        phase2_budget = min(total_budget * 0.40, remaining)
        print(f"[{time.monotonic() - start_time:6.2f}s] start phase‑2 (sub‑grad)")

        a_refined, score_refined = _subgradient_refine(
            best_overall_a,
            phase2_budget,
            rng=rng,
        )
        if score_refined < best_overall_score:
            best_overall_score = score_refined
            best_overall_a = a_refined.copy()
        print(
            f"[{time.monotonic() - start_time:6.2f}s] phase‑2 finished, best score {best_overall_score:.6f}"
        )

    # ------------------------------------------------------------------
    # Final sanity check & conversion to plain list
    # ------------------------------------------------------------------
    if best_overall_a is None:
        # fallback – return uniform (should never happen)
        best_overall_a = np.full(n, 1.0 / n, dtype=np.float64)

    best_overall_a = np.clip(best_overall_a, 0.0, 1000.0)

    return [float(v) for v in best_overall_a]
