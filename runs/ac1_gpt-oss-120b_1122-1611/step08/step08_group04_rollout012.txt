def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):

    import numpy as np, time

    start = time.monotonic()

    remaining_budget = max(0.0, float(budget_s) - 2.0) # 2s safety

    deadline = start + remaining_budget

    # Determine n

    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)

        base = np.clip(base, 0.0, None)

        if base.sum() == 0.0: base[:] = 1.0 / base.size

        base = base / base.sum()
        n = base.size

        a = base.copy()
    else:
        n = 3840
        a = np.full(n, 1.0/n, dtype=np.float64)

    # Keep best

    best_a = a.copy()
    best_score = _objective_score(a)

    print(f"Initial score {best_score}")

    # Phase 1: Adam

    # allocate fraction, e.g., 0.55 of time
    phase1_deadline = start + remaining_budget * 0.55
    # Convert a to logits
    eps = 1e-12
    x = np.log(a + eps)   # logits

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    beta_start = 5.0
    beta_end = 2000.0
    beta_ramp_iters = 5000

    it = 0
    last_improve = start

    # Precompute FFT length for convolution (larger for w)
    fft_len_conv = _next_pow2(2*n-1)

    while time.monotonic() < phase1_deadline:
        it += 1

        # Compute a from logits
        a = _softmax_logits(x)

        # Compute convolution via FFT
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[:2*n-1]

        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max   # sum=1 case

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve-start:.2f}s] Phase1 new best {best_score:.6f}")

        # Softmax surrogate
        beta = beta_end if it > beta_ramp_iters else beta_start + (beta_end-beta_start) * it / beta_ramp_iters

        scaled = beta * conv_full
        # stable logsumexp
        lse = np.max(scaled) + np.log(np.sum(np.exp(scaled - np.max(scaled))))
        w = np.exp(scaled - lse)

        # compute gradient of surrogate with respect to a:
        # g = 2 * (w * a) convolution reversed?

        # The gradient is correlation of w with a as earlier:

        fft_len_corr = _next_pow2(3*n-2)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)
        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx+n]

        # gradient w.r.t logits via softmax Jacobian
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # jitter occasional
        if it % 500 == 0:
            x += rng.normal(scale=0.01, size=n)

        # restart if idle
        if time.monotonic() - last_improve > 30.0:   # 30 sec no improve
            x = np.log(best_a + eps) + rng.normal(scale=0.12, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve-start:.2f}s] Phase1 restart")

    # Phase 2: subgradient descent

    phase2_deadline = deadline - (deadline - start) * 0.2   # allocate about 20% time for phase3

    a = best_a.copy()
    step = 0.2
    cur_score = _objective_score(a)
    last_improve = time.monotonic()

    while time.monotonic() < phase2_deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # compute subgradient
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step *= 1.05
            last_improve = time.monotonic()
            print(f"[{last_improve-start:.2f}s] Phase2 improvement {best_score:.6f}")
        else:
            step *= 0.7
            # jitter occasionally
            if step < 1e-6 or rng.random() < 0.001:
                jitter = rng.normal(scale=0.002, size=n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.01)

    # Phase 3: Simulated annealing local search

    phase3_deadline = deadline
    a = best_a.copy()
    cur_score = best_score
    # schedule
    init_sigma = 0.02
    final_sigma = 0.001
    # Determine number of iterations based on time left
    while time.monotonic() < phase3_deadline:
        # compute fraction of remaining time
        t_now = time.monotonic()
        frac = (t_now - start) / remaining_budget
        # linearly decay sigma and temperature
        sigma = init_sigma * (1 - frac) + final_sigma * frac
        temp = 0.01 * (1 - frac) + 0.0001 * frac

        cand = a + rng.normal(scale=sigma, size=n)
        # project to nonnegative simplex
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)

        delta = cand_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            a = cand
            cur_score = cand_score

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{t_now-start:.2f}s] SA new best {best_score:.6f}")

    # Return best
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]

"""
Hybrid optimizer for the self‑convolution problem.

The routine consists of three stages:

1.  **Random warm‑start** – several cheap random candidates (uniform,
    Dirichlet, and random sparse supports) are evaluated and the best
    is kept.

2.  **Adam on a smooth surrogate** – the max‑convolution is approximated
    by a log‑sum‑exp with a gradually increasing temperature (β).  The
    simplex constraint is enforced by a soft‑max re‑parameterisation.
    Adam adapts the learning‑rate per coordinate and a simple restart
    scheme avoids stagnation.

3.  **Projected sub‑gradient descent** – after the surrogate has
    converged the exact maximum of the convolution is used.  A
    projected sub‑gradient step on the simplex reduces the peak
    value.  The step size is increased on success and shrunk on failure.

4.  **Simulated annealing fine‑tuning** – a final short stochastic
    search (Gaussian perturbations + projection onto the simplex)
    with a linear cooling schedule explores nearby basins and can
    escape shallow plateaus.

The whole algorithm respects a 1000 s budget (a 5 s safety margin) and
returns the best sequence found so far.  All helper functions are
defined at top level to satisfy the harness requirements.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """
    Exact value of evaluate_sequence for a (the sum is not forced to 1).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method combines a cheap random warm‑start,
    Adam optimisation of a smooth max‑convolution surrogate, projected
    sub‑gradient descent on the exact max, and a final simulated–annealing
    refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)   # safety margin
    deadline = start + total_budget

    # --------------------------------------------------------------
    # Choose a length n and an initial distribution.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        base = base / base.sum()
        n = base.size
        a0 = base.copy()
    else:
        n = 3840
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 0. Random warm‑start: evaluate a few cheap candidates
    # --------------------------------------------------------------
    candidates = [a0]

    # uniform (already in a0 for the fallback case)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet with concentration 0.5 (encourages sparsity)
    candidates.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # Random sparse supports of size m≈100 (equal weights)
    m_sparse = int(np.sqrt(4 * n) * 0.9)  # a bit below the theoretical limit
    for _ in range(5):
        idx = rng.choice(n, size=m_sparse, replace=False)
        a_sparse = np.zeros(n, dtype=np.float64)
        a_sparse[idx] = 1.0 / m_sparse
        candidates.append(a_sparse)

    # select the best of the above
    best_a = None
    best_score = np.inf
    for cand in candidates:
        score = _objective_score(cand)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
    print(f"Warm‑start best score {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on a smooth surrogate of max‑convolution
    # --------------------------------------------------------------
    eps = 1e-12
    x = np.log(best_a + eps)                # logits → simplex via soft‑max

    m = np.zeros_like(x)                    # Adam first moment
    v = np.zeros_like(x)                    # Adam second moment

    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    beta_start = 5.0
    beta_end = 2000.0           # higher final temperature for a sharper max
    beta_ramp_iters = 6000      # longer ramp

    RESTART_WAIT = 30.0         # seconds without improvement ⇒ restart
    last_improve = start

    fft_len_conv = _next_pow2(2 * n - 1)
    it = 0
    phase1_deadline = start + total_budget * 0.55   # ~55 % of the budget

    while time.monotonic() < phase1_deadline:
        it += 1

        # probabilities from logits
        a = _softmax_logits(x)

        # self‑convolution (exact)
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]

        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max                     # sum(a)=1 ⇒ denominator =1

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase1 new best {best_score:.6f}")

        # smooth surrogate – log‑sum‑exp with temperature β
        beta = (beta_end if it > beta_ramp_iters
                else beta_start + (beta_end - beta_start) * it / beta_ramp_iters)

        scaled = beta * conv_full
        lse = logsumexp(scaled)          # stable
        w = np.exp(scaled - lse)          # softmax over the convolution (sums to 1)

        # gradient of the surrogate w.r.t. a = correlation w⋆a
        fft_len_corr = _next_pow2(3 * n - 2)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)
        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # back‑prop through the soft‑max
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            x += rng.normal(0.0, 0.015, size=n)

        # hard restart if we have been stagnant for a while
        if time.monotonic() - last_improve > RESTART_WAIT:
            x = np.log(best_a + eps) + rng.normal(0.0, 0.12, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase1 restart")

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    a = best_a.copy()
    step = 0.2
    cur_score = _objective_score(a)
    subgrad_deadline = start + total_budget * 0.90   # leave ~10 % for SA

    while time.monotonic() < subgrad_deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of max‑convolution at the active offset
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step *= 1.05                     # enlarge step after success
            cur_score = cand_score
            print(f"[{time.monotonic() - start:6.2f}s] subgrad improvement {best_score:.6f}")
        else:
            step *= 0.7                      # shrink step after failure
            # occasional jitter to shake out of flat regions
            if step < 1e-8 or rng.random() < 0.001:
                jitter = rng.normal(0.0, 0.002, size=n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.01)

    # --------------------------------------------------------------
    # Phase 3 – Simulated annealing fine‑tuning (short stochastic search)
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    sa_deadline = deadline
    init_sigma = 0.02
    final_sigma = 0.001
    init_temp = 0.02
    final_temp = 1e-4

    while time.monotonic() < sa_deadline:
        # linear schedules based on elapsed fraction
        elapsed = time.monotonic() - start
        frac = elapsed / total_budget
        sigma = init_sigma * (1.0 - frac) + final_sigma * frac
        temp = init_temp * (1.0 - frac) + final_temp * frac

        cand = a + rng.normal(scale=sigma, size=n)
        cand = _project_simplex(cand)
        cand_score = _objective_score(cand)

        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / temp):
            a = cand
            cur_score = cand_score

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{time.monotonic() - start:6.2f}s] SA new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
