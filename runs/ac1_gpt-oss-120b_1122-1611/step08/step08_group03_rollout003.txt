"""
search for a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
* Multiple random restarts (different initial shapes and lengths).
* For each restart:
    1.  **Adam on a smooth‑max surrogate** – the hard max in the
        objective is replaced by a temperature‑controlled soft‑max.
        Gradients are obtained analytically via FFTs and a standard
        Adam update is applied.  The temperature and learning‑rate are
        annealed slowly; the iterates are always projected onto the
        probability simplex.
    2.  **Simulated‑annealing local refinement** – after the gradient
        phase we perform thousands of tiny mass‑transfer moves (single–
        and block‑moves).  Worse moves are accepted with a temperature
        that decays to ~1e‑7, allowing the search to escape local minima.
* The best sequence seen over the whole budget is returned.
* If the target 1.5025 is reached the search stops early.

The implementation is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for the final sanity check, and respects
the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of ``evaluate_sequence`` for a normalised vector a.
    a is assumed to be non‑negative; the function returns
    2·n·max(conv) where n = len(a).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s                     # enforce exact normalisation
    n = a.size
    fft_len = _next_pow2(2 * n)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random non‑negative vector of length n, already normalised.
    Several families of distributions are tried to increase diversity.
    """
    choice = rng.integers(0, 4)
    if choice == 0:                     # uniform
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif choice == 1:                   # Dirichlet with random concentration
        alpha = rng.uniform(0.2, 0.8)
        a = rng.gamma(alpha, 1.0, size=n)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)
    elif choice == 2:                   # exponential decay
        rate = rng.uniform(0.5, 5.0) / n
        r = np.exp(-rate)
        idx = np.arange(n, dtype=np.float64)
        a = (1.0 - r) * (r ** idx)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)
    else:                               # truncated Gaussian
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(n / 10.0, n / 3.0)
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)
    return a


def _adam_phase(a: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                start_time: float,
                target: float = 1.5025) -> tuple[np.ndarray, float]:
    """
    Smooth‑max Adam optimiser.
    The loop stops either when the time budget for this phase expires
    or when the improvement becomes negligible.
    """
    n = a.size
    fft_len = _next_pow2(2 * n)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # annealing schedules (slowly decreasing)
    TAU_START, TAU_END = 0.5, 5e-6
    LR_START, LR_END = 0.5, 5e-4

    # we allocate at most 40 % of the *remaining* budget to Adam
    phase_deadline = time.monotonic() + 0.40 * (deadline - time.monotonic())

    best_local = a.copy()
    best_score = _score(a)

    while time.monotonic() < phase_deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)

        # linear annealing
        tau = TAU_START * (1.0 - frac) + TAU_END * frac
        lr = LR_START * (1.0 - frac) + LR_END * frac

        # convolution via FFT
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # smooth‑max weights
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)      # stability shift
        w = exp_shifted / exp_shifted.sum()              # length 2n‑1, sums to 1

        # gradient of the smooth surrogate (correlation)
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # occasional exact evaluation
        if int(now * 10) % 5 == 0:          # roughly every 0.5 s
            sc = _score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_local = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] Adam improvement → {best_score:.6f}")
                if best_score <= target:
                    print("[early stop] target reached inside Adam")
                    return best_local, best_score
        # tiny‑step convergence test
        if np.linalg.norm(step) < 1e-13:
            break
    return best_local, best_score


def _annealing_phase(a: np.ndarray,
                     current_score: float,
                     deadline: float,
                     rng: np.random.Generator,
                     start_time: float,
                     target: float = 1.5025) -> tuple[np.ndarray, float]:
    """
    Stochastic local search with a simulated‑annealing acceptance rule.
    Moves include single‑mass transfers, block scaling and small Dirichlet
    reshuffles of a random subset.
    """
    n = a.size
    best = a.copy()
    best_score = current_score

    # temperature schedule (linear in the remaining fraction)
    T_start, T_end = 0.12, 1e-7
    phase_start = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        remaining = deadline - now
        if remaining <= 0:
            break
        frac = (now - phase_start) / (deadline - phase_start) if deadline > phase_start else 1.0
        T = T_start * (1.0 - frac) + T_end * frac

        # ----- propose a move -------------------------------------------------
        move_type = rng.integers(0, 3)
        if move_type == 0:                                   # single transfer
            i = rng.integers(0, n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            # log‑uniform fraction between 0.1 % and 7 %
            frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.07)))
            delta = a[i] * frac_move
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
        elif move_type == 1:                                 # block scaling
            width = rng.integers(2, max(4, n // 200))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            cand = a.copy()
            cand[start_idx:start_idx + width] *= factor
        else:                                                # subset Dirichlet reshuffle
            sz = rng.integers(2, min(12, n))
            idxs = rng.choice(n, size=sz, replace=False)
            tot = a[idxs].sum()
            if tot <= 1e-12:
                continue
            new_vals = rng.dirichlet(np.ones(sz))
            cand = a.copy()
            cand[idxs] = new_vals * tot

        # re‑normalise and keep non‑negativity
        cand_sum = cand.sum()
        if cand_sum <= 0:
            continue
        cand /= cand_sum
        cand = np.clip(cand, 0.0, None)      # safety

        # ----- evaluate -------------------------------------------------------
        cand_score = _score(cand)

        # ----- acceptance ------------------------------------------------------
        delta = cand_score - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = np.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] NEW best (anneal) = {best_score:.6f}")
                if best_score <= target:
                    print("[early stop] target reached during annealing")
                    break
    return best, best_score

    Search for a step‑function (list of non‑negative heights) that

    a simulated‑annealing local search and performs several random restarts.
    """

    # bookkeeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    best_seq = None
    best_score = np.inf

    # a few different lengths – larger n gives a lower theoretical bound
    length_options = [2048, 4096, 8192]

    # ------------------------------------------------------------------
    # main loop – keep launching new restarts while time permits
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (biased towards larger sizes as time goes on)
        if time.monotonic() - start_time > 0.5 * (deadline - start_time):
            n = rng.choice(length_options[1:])        # favour 4096 / 8192 later
        else:
            n = rng.choice(length_options)

        # initialise a candidate (use the known construction if lengths match)
        if "height_sequence_1" in globals() and len(height_sequence_1) == n:
            a = np.asarray(height_sequence_1, dtype=np.float64)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)
        else:
            a = _initial_sequence(n, rng)

        # ----------------------------------------------------------------
        # Phase 1 – Adam on smooth‑max surrogate
        # ----------------------------------------------------------------
        a, cur_score = _adam_phase(a, deadline, rng, start_time)

        # ----------------------------------------------------------------
        # Phase 2 – Simulated‑annealing local refinement
        # ----------------------------------------------------------------
        a, cur_score = _annealing_phase(a, cur_score, deadline, rng, start_time)

        # ----------------------------------------------------------------
        # update the global best
        # ----------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST = {best_score:.6f}")

        # stop early if we already beat the target
        if best_score <= 1.5025:
            break

    # ------------------------------------------------------------------
    # final sanitisation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_seq]
