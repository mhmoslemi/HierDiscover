"""
Hybrid optimizer for the step‑function problem.

The objective that `evaluate_sequence` computes (with a simplex normalisation
∑a_i = 1) reduces to

    J(a) = 2·n·max(conv(a,a))

where “conv” is the full linear convolution.  The algorithm consists of three
phases:

1. **Diverse initialisation** – a few simple candidates (uniform,
   Dirichlet draws, a decreasing linear shape and a truncated previous
   construction) are evaluated and the best one is taken as start.

2. **Soft‑max gradient descent** – a smooth approximation of the max
   (soft‑max with a temperature β) yields a dense gradient
   `∇J ≈ 2·ifft(conj(FFT(w))·FFT(a))`.  An exponentiated‑gradient step keeps the
   iterate on the probability simplex.  β is increased linearly to focus the
   optimisation on the true maximum, while the step‑size decays like 1/t.

3. **Linear‑programming mixing** – given a current vector `a`,
   we solve the LP

        maximise   Σ b_i
        subject to  conv(a,b) ≤ M   (M = max(conv(a,a)))
                    b_i ≥ 0

   The LP exploits the linearity of the cross‑convolution `conv(a,b)`.  The
   solution `b` typically has a larger ℓ₁‑norm than 1; after normalising it to
   the simplex we mix it with `a` (`a←(1‑t)·a + t·b`) and keep the mixture that
   reduces the objective.  This step is repeated a few times or until no
   improvement is observed.

The two phases complement each other: gradient descent quickly finds a
reasonable basin, while the LP mixing provably reduces the maximal
convolution without breaking the simplex constraint.  The whole search runs
within the allocated time budget (default 1000 s) and returns the best
sequence found.
"""

import time
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    return np.convolve(a, a)


def _make_constraint_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
    (A @ b)[i] = conv_i(a, b) for all i.
    """
    n = a.shape[0]
    # number of non‑zero entries = n * n
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for col in range(n):
        # rows where a_{i-col} contributes: i = col … col+n‑1
        rows[idx:idx + n] = np.arange(col, col + n, dtype=np.int32)
        cols[idx:idx + n] = col
        data[idx:idx + n] = a  # a is constant across this block
        idx += n
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best normalised sequence found within the
    time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size (keep it modest for the LP phase)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n_candidate = len(globals()["height_sequence_1"])
            n = max(256, min(n_candidate, 1024))   # cap at 1024 for speed
        except Exception:
            n = 1024
    else:
        n = 1024

    # --------------------------------------------------------------
    # 2. Build a diverse set of starting points (all lie on the simplex)
    # --------------------------------------------------------------
    init_vectors = []

    # uniform
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws – from spiky to smooth
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            vec = rng.dirichlet(np.full(n, alpha))
            init_vectors.append(vec)

    # decreasing linear shape  a_i ∝ (n‑i)
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin = lin / lin.sum()
    init_vectors.append(lin)

    # previous construction (if any) – truncated / interpolated to length n
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= n:
            prev = prev[:n]
        else:
            # pad with zeros and renormalise
            prev = np.concatenate([prev, np.zeros(n - prev.size)])
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s == 0.0:
            prev[:] = 1.0 / n
        else:
            prev = prev / s
        init_vectors.append(prev)

    # --------------------------------------------------------------
    # 3. Choose the best initial vector (objective = 2·n·max(conv))
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for vec in init_vectors:
        conv = _conv_full(vec)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Soft‑max gradient descent (dense phase)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 30.0)   # leave a 30 s safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    beta = 1.0
    beta_max = 500.0
    beta_inc = (beta_max - beta) / max(1, int(0.6 * (deadline - start_time) / 0.01))

    iteration = 0
    while time.monotonic() < deadline:
        # recompute convolution and soft‑max weights
        conv = _conv_full(cur_vec)
        max_c = conv.max()
        cur_J = float(2.0 * n * max_c)

        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best J (grad) = {best_J:.6f}")

        # update temperature for the soft‑max
        beta = min(beta_max, beta + beta_inc)

        # soft‑max weights (stabilised)
        w_raw = np.exp(beta * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of the smoothed objective (Fourier‑domain formula)
        # g = 2 * ifft( conj(FFT(w)) * FFT(cur_vec) )
        A_fft = np.fft.rfft(cur_vec)
        W_fft = np.fft.rfft(w)
        g = 2.0 * np.fft.irfft(np.conj(W_fft) * A_fft, n=2 * n - 1)[:n]

        # exponentiated‑gradient step (step size decays as 1/√t)
        step = 0.5 / np.sqrt(iteration + 1.0)
        factor = np.exp(-step * g)
        proposal = cur_vec * factor
        proposal = proj_simplex(proposal)

        # evaluate proposal
        prop_conv = _conv_full(proposal)
        prop_J = float(2.0 * n * prop_conv.max())

        # Metropolis acceptance
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(1e-12, 0.01 * (deadline - time.monotonic())))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1
        if best_J < 1.45:          # early stop if we are already good
            break

    # --------------------------------------------------------------
    # 5. Linear‑programming mixing phase (sparse optimisation)
    # --------------------------------------------------------------
    # Prepare for LP: keep the best vector from the dense phase as start.
    cur_vec = best_vec.copy()
    cur_J = best_J
    lp_deadline = start_time + float(budget_s)  # full remaining budget
    lp_iter = 0
    max_lp_iters = 30          # safety bound

    while time.monotonic() < lp_deadline and lp_iter < max_lp_iters:
        conv = _conv_full(cur_vec)
        M = conv.max()
        cur_J = float(2.0 * n * M)

        # Build constraint matrix A such that (A @ b)_i = conv_i(cur_vec, b)
        A = _make_constraint_matrix(cur_vec)

        # Linear program: maximise Σ b_i  subject to A @ b ≤ M and b ≥ 0
        c = -np.ones(n)                     # minimise – Σb_i  == maximise Σb_i
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        # Solve with HiGHS (fast interior‑point)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as e:
            print(f"LP failed at iteration {lp_iter}: {e}")
            break

        if not res.success:
            # no feasible improvement – exit the LP phase
            print(f"LP did not converge at iteration {lp_iter}, status: {res.message}")
            break

        b_opt = res.x
        sum_b = b_opt.sum()

        if sum_b <= 1.001:            # essentially no increase – stop mixing
            print(f"[LP] No increase in ℓ₁ (sum_b={sum_b:.6f}); terminating LP phase.")
            break

        # Scale to keep the simplex normalisation
        g_vec = b_opt / sum_b

        # Try a few mixing ratios and keep the best improvement
        improved = False
        best_local_J = cur_J
        best_mix = cur_vec

        for t in (0.5, 0.3, 0.7):
            mix = (1.0 - t) * cur_vec + t * g_vec
            mix = proj_simplex(mix)
            mix_conv = _conv_full(mix)
            mix_J = float(2.0 * n * mix_conv.max())
            if mix_J < best_local_J - 1e-12:
                best_local_J = mix_J
                best_mix = mix
                improved = True

        if not improved:
            # If none of the trial mixing ratios improved the objective,
            # we reduce the mixing weight and try again.
            t_small = 0.1
            mix = (1.0 - t_small) * cur_vec + t_small * g_vec
            mix = proj_simplex(mix)
            mix_J = float(2.0 * n * _conv_full(mix).max())
            if mix_J < cur_J - 1e-12:
                best_mix = mix
                best_local_J = mix_J
                improved = True

        if not improved:
            # No progress – exit LP phase
            print(f"[LP] No improvement after mixing at iteration {lp_iter}.")
            break

        # Accept improvement
        cur_vec = best_mix
        cur_J = best_local_J
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best J after LP = {best_J:.6f}")

        lp_iter += 1

        # Very early exit if we have already crossed the target
        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached (J={best_J:.6f}) – terminating.")
            break

    # --------------------------------------------------------------
    # 6. Final clean‑up – clip and re‑normalise (evaluate_sequence does this itself)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    print(f"Finished search. Best J = {best_J:.6f}  (length = {n})")
    return [float(v) for v in best_vec]
