            prop_sw_J = compute_J(prop_sw, n, fft_len, conv_len)

                    best_vec = cur_vec.copy()

"""
Hybrid Adam‑gradient + guided exchange + swap optimiser.

* Works directly on the probability simplex (∑a = 1, a_i ≥ 0).
* Adam‑style adaptive learning rates are used on the sub‑gradient of
  max(conv(a,a)).  The sub‑gradient is the sum of shifted copies of a over
  the active set of convolution entries that are within a tiny tolerance
  of the current maximum.
* Three complementary move families are tried at each iteration:
    – *Gradient move*: Adam update followed by Euclidean projection onto
      the simplex.
    – *Guided exchange*: mass is moved from a source index that has a
      large gradient (i.e. contributes heavily to the maximal convolution)
      to a target index with a small gradient.
    – *Swap move*: two positions are swapped – this rearranges the same
      multiset of weights and can dramatically change the convolution.
* An exponential cooling schedule controls a Metropolis acceptance rule.
  The temperature starts at the current best objective value and decays
  to 1e‑8, allowing occasional uphill moves early on.
* If no improvement has been seen for ≈ 25 s a jitter (multiplicative
  log‑normal noise) is applied centred on the best solution to escape
  plateaus.
* The routine respects a 1000 s time budget, prints intermediate bounds,
  and finally returns the best vector found (clipped to [0, 1000] and
  re‑normalised).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal (or v is zero); return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Return the objective J = 2·n·max(conv(a,a)) for a vector a that
    already lives on the simplex (sum(a) == 1).
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgradient_active_set(
    a: np.ndarray, conv: np.ndarray, tol: float
) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) at the current active set.
    The result is a vector of the same length as a.
    """
    n = a.size
    max_conv = conv.max()
    active = np.where(conv >= max_conv - tol)[0]
    grad = np.zeros_like(a)
    for p in active:
        start = max(0, p - (n - 1))
        end = min(p, n - 1)
        idx = np.arange(start, end + 1)
        grad[idx] += 2.0 * a[p - idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 10000))          # keep it reasonable
        except Exception:
            n = 4096
    else:

    # ------------------------------------------------------------------
    init_cands = []

    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()

            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # a safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.3          # learning‑rate scale
    m = np.zeros_like(cur_vec)
    v_ = np.zeros_like(cur_vec)

    # probability of trying the auxiliary moves
    prob_exchange = 0.30
    prob_swap = 0.20

    # jitter parameters
    jitter_interval = 25.0
    last_improve = start_time

        iteration += 1

        v_ = beta2 * v_ + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_ / (1.0 - beta2 ** iteration)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        prop = cur_vec - step
        prop = proj_simplex(prop)
        prop_J = compute_J(prop, n, fft_len, conv_len)

        # ---- temperature schedule (exponential cooling) -------------------------
        T_start = best_J
        T_end = 1e-8
        total_time = deadline - start_time

        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = prop
            cur_J = prop_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.2 Guided exchange move (occasionally)
        # ------------------------------------------------------------------
        if rng.random() < prob_exchange:
            # source index proportional to gradient magnitude (large contribution)
            grad_abs = np.abs(grad)

            else:

                    accept_sw = True
