"""
Hybrid Adam‑gradient + simulated‑annealing optimiser for
minimising  J(a)=2·n·max(conv(a,a)) / (∑a)² .
The algorithm

* starts from the best known construction (height_sequence_1) if it exists,
* runs a projected Adam optimiser on a smooth log‑sum‑exp surrogate of the
  sup‑norm (the smoothing parameter p is increased from 1 to 200),
* enforces symmetry a[i]=a[-i‑1] after every update (the optimum is known
  to be symmetric),
* injects a multiplicative jitter if no improvement is seen for 20 s,
* finishes with a cheap random‑move local search around the best solution.

The whole search respects the 1000 s time budget and returns a list of
non‑negative heights (already normalised to sum = 1, which is scale‑invariant
for the evaluation function).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def gradient_smooth(
    a: np.ndarray,
    conv: np.ndarray,
    p: float,
    fft_len_g: int,
    conv_len: int,
) -> np.ndarray:
    """
    Gradient of the smoothed maximum:
        φ(a) = log‑sum‑exp(p·conv(a,a)) / p .
    Returns the gradient w.r.t. a (same size as a).
    """
    # soft‑max weights (max weight = 1)
    w = np.exp(p * (conv - conv.max()))
    # reverse a for the cross‑correlation needed in the gradient
    a_rev = a[::-1]
    # convolution w (len 2n‑1) with a_rev (len n)
    grad_full = np.fft.irfft(
        np.fft.rfft(w, fft_len_g) * np.fft.rfft(a_rev, fft_len_g),
        n=fft_len_g,
    )
    # extract the n‑length gradient slice
    grad = 2.0 * grad_full[conv_len - 1 : conv_len - 1 + a.size]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises the
    provided evaluation function.  Returns the best sequence found within the
    allotted time budget (≈ 1000 s).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size – reuse the length of a previously found construction
    #    if it exists, otherwise fall back to a comfortable default.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)
            total = base.sum()
            if total == 0.0:
                base[:] = 1.0 / base.size
            else:
                base /= total
            n = base.size
            a = base.copy()
        except Exception:
            n = 2048
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Enforce symmetry from the start (the optimum is symmetric)
    a = (a + a[::-1]) * 0.5
    a = proj_simplex(a)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for a*a
    fft_len_g = _next_pow2(conv_len + n - 1)           # for gradient (w * a_rev)

    # --------------------------------------------------------------
    # 2. Helper to compute the objective (without the costly sum‑square part
    #    because we keep the sum normalised to 1).
    # --------------------------------------------------------------
    def compute_J(vec: np.ndarray) -> float:
        conv_vec = conv_full_fft(vec, fft_len, conv_len)
        return float(2.0 * n * conv_vec.max())

    # --------------------------------------------------------------
    # 3. Initialise best known solution.
    # --------------------------------------------------------------
    cur_J = compute_J(a)
    best_J = cur_J
    best_a = a.copy()
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Adam optimiser on the smoothed surrogate.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # small safety margin

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.08                     # base step size (additive)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature (soft‑max) schedule
    p_start = 0.5
    p_end = 200.0

    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)

        # ---- current convolution & objective
        conv = conv_full_fft(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ---- keep global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- stagnation handling – jitter if no improvement for >20 s
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = proj_simplex(best_a * noise)
            a = (a + a[::-1]) * 0.5
            a = proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            iteration = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ---- smoothed gradient (log‑sum‑exp)
        p = p_start + (p_end - p_start) * frac
        grad = gradient_smooth(a, conv, p, fft_len_g, conv_len)

        # ---- Adam update (additive)
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / np.sqrt(iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip very large steps to keep the update numerically stable
        step = np.clip(step, -0.5, 0.5)

        a = a - step
        a = proj_simplex(a)

        # Enforce symmetry after every projection
        a = (a + a[::-1]) * 0.5
        a = proj_simplex(a)

    # --------------------------------------------------------------
    # 5. Fine‑grained random‑move local search around the best solution.
    # --------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)

    cur = best_a.copy()
    cur_J = best_J

    while time.monotonic() < fine_deadline:
        # pick two distinct positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # amount to transfer (up to 10 % of the source mass)
        max_delta = cur[i]
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.1
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection is not needed – the move preserves the sum and non‑negativity
        cand_J = compute_J(cand)
        if cand_J < cur_J:
            cur = cand
            cur_J = cand_J
            if cand_J < best_J:
                best_J = cand_J
                best_a = cand.copy()
                print(f"[Fine] Improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best sequence (already normalised and non‑negative).
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
