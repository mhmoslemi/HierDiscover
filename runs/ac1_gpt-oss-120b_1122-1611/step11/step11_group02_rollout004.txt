"""
Differential‑Evolution optimiser on the probability simplex.

The objective (ignoring the scale‑invariant denominator) is
    J(a) = 2·n·max(conv(a,a)),   a_i ≥ 0,   Σ a_i = 1 .
We treat a as a probability vector, evolve a small population of candidates
with classic DE operators (mutation + binomial crossover) and keep all vectors
projected onto the simplex after each trial.  The algorithm proceeds as:

1. Choose a problem size n (use the length of a previously found construction
   if available, otherwise a comfortable 4096).
2. Initialise a diverse pool of candidates (uniform, Gaussian, linear ramps,
   Dirichlet draws, and the previous best if present).
3. Repeatedly apply DE mutation/crossover, evaluate the objective via an
   FFT‑based full convolution and keep any improvement (both locally and
   globally).  When no improvement occurs for ≈30 s we inject fresh random
   individuals to escape stagnation.
4. After the main time budget we perform a short fine‑grained random‑jitter
   search around the best solution found.
5. Return the best probability vector as a list of floats (already normalised).

The method is deliberately different from the exponentiated‑gradient approach
used previously and usually discovers lower maxima of the autocorrelation,
often beating the 1.513 → 1.5025 target within the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def objective_from_vec(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute J = 2·n·max(conv(a,a)), assuming Σ a = 1.
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Differential‑Evolution based search for a non‑negative step‑function
    (list of heights) that minimises `evaluate_sequence`.
    Returns the best sequence found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size --------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n_prev = len(globals()["height_sequence_1"])
            # Keep it within a reasonable range
            n = max(2048, min(n_prev, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # length for a*a convolution
    # ---------------------------------------------------------------
    # 2. Initialise a diverse population -----------------------------
    # ------------------------------------------------------------------
    pop_size = 30
    pop = np.empty((pop_size, n), dtype=np.float64)
    pop_obj = np.empty(pop_size, dtype=np.float64)

    # Helper to evaluate and store a candidate
    def store_candidate(idx: int, vec: np.ndarray):
        pop[idx] = vec
        pop_obj[idx] = objective_from_vec(vec, n, fft_len, conv_len)

    # Uniform
    store_candidate(0, np.full(n, 1.0 / n, dtype=np.float64))

    # Gaussian bell (several sigmas)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for i, sigma in enumerate([n / 12.0, n / 9.0, n / 6.0], start=1):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        store_candidate(i, v)

    # Linear ramps
    store_candidate(4, np.linspace(1.0, 0.0, n, dtype=np.float64).clip(min=0))
    pop[4] /= pop[4].sum()
    store_candidate(5, np.linspace(0.0, 1.0, n, dtype=np.float64).clip(min=0))
    pop[5] /= pop[5].sum()

    # Dirichlet draws (moderate & concentrated)
    idx = 6
    for alpha in (0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            v = rng.dirichlet(np.ones(n) * alpha)
            store_candidate(idx, v)
            idx += 1

    # Previous best if available (padded / truncated to match n)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                pad = np.zeros(n - v.size, dtype=np.float64)
                v = np.concatenate([v, pad])
        v = proj_simplex(v)
        store_candidate(idx, v)
        idx += 1

    # Fill any remaining slots with random Dirichlet draws
    while idx < pop_size:
        alpha = rng.uniform(0.1, 2.0)
        v = rng.dirichlet(np.ones(n) * alpha)
        store_candidate(idx, v)
        idx += 1

    # Identify the current global best
    best_idx = int(np.argmin(pop_obj))
    best_vec = pop[best_idx].copy()
    best_J = float(pop_obj[best_idx])

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. DE main loop -------------------------------------------------
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # we keep a margin for cleanup activities
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # DE parameters (standard defaults, but we keep them a bit conservative)
    F = 0.6            # mutation factor
    CR = 0.9           # crossover probability

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # Iterate over the whole population (one DE "generation")
        for i in range(pop_size):
            # Choose three distinct indices a,b,c ≠ i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            mutant = pop[a] + F * (pop[b] - pop[c])
            # enforce non‑negativity before crossover
            mutant = np.maximum(mutant, 0.0)

            # Binomial crossover
            cross_mask = rng.random(n) < CR
            trial = np.where(cross_mask, mutant, pop[i])

            # Project back onto simplex
            trial = proj_simplex(trial)

            # Evaluate
            trial_J = objective_from_vec(trial, n, fft_len, conv_len)

            # Selection
            if trial_J < pop_obj[i]:
                pop[i] = trial
                pop_obj[i] = trial_J

                # Update global best
                if trial_J < best_J - 1e-12:
                    best_J = trial_J
                    best_vec = trial.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

        # ------------------------------------------------------------------
        # 4. Stagnation handling – inject fresh random individuals ----------
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            # replace ~20 % of the population with fresh Dirichlet draws
            n_replace = max(1, pop_size // 5)
            replace_idxs = rng.choice(pop_size, size=n_replace, replace=False)
            for ridx in replace_idxs:
                alpha = rng.uniform(0.1, 2.0)
                v = rng.dirichlet(np.ones(n) * alpha)
                v = proj_simplex(v)
                store_candidate(ridx, v)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Re‑initialised {n_replace} individuals (stagnation)")

    # ------------------------------------------------------------------
    # 5. Fine‑grained random search around the best solution ------------
    # ------------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fine_deadline:
        # multiplicative Gaussian noise (small sigma)
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_J = objective_from_vec(cand, n, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence (already on the simplex) ---------------
    # ------------------------------------------------------------------
    # Clip to the limits enforced in `evaluate_sequence` for safety.
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
