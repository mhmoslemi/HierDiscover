def propose_move(a):
    # copy a (maybe new copy)
    new = a.copy()
    # pick move_type
    move_type = rng.integers(0,3) # 0: transfer, 1: swap, 2: random perturb
    if move_type == 0:
        # pick source i with a_i > 0
        positive = np.where(new > 1e-12)[0]
        if len(positive) == 0:
            return new
        i = rng.choice(positive)
        # pick target j (could be any index)
        j = rng.integers(0, n)
        # transfer delta
        delta = new[i] * rng.uniform(0.0, 0.5)  # up to half
        new[i] -= delta
        new[j] += delta
    elif move_type == 1:
        # swap index: move mass from i to new location j
        positive = np.where(new > 1e-12)[0]
        if len(positive) == 0:
            return new
        i = rng.choice(positive)
        j = rng.integers(0, n)
        # move entire mass from i to j
        # but keep some mass for i maybe not zero?
        mass = new[i] * rng.uniform(0.0, 1.0)  # fraction
        new[i] -= mass
        new[j] += mass
    elif move_type == 2:
        # jitter: add small random noise to each entry
        noise = rng.normal(0.0, 0.02, size=n)
        new = new + noise
        new = np.maximum(new, 0.0)
        # renormalize
        total = new.sum()
        if total > 0:
            new = new / total
        else:
            new = np.full(n, 1.0/n)
    # ensure sum not too close zero due to rounding
    total = new.sum()
    if total > 0:
        new = new / total
    else:
        new = np.full(n, 1.0/n)
    return new

"""Candidate generator for the step‑function challenge.

The algorithm builds a probability vector (∑ aᵢ = 1, aᵢ ≥ 0) and
optimises the quantity

    J(a) = 2·n·max_k (a * a)[k]

by a simulated‑annealing style search.
A diverse set of initial vectors (uniform, Dirichlet, exponential,
geometric, cosine‑shaped) is evaluated, the best one is taken as a start,
and then a large number of cheap proposals (mass transfer, swaps,
global jitter) are generated.  Proposals are accepted with the usual
Metropolis rule using a linearly‑cooled temperature, and the best
objective seen so far is returned.

The implementation uses FFT‑based convolution for O(n log n) evaluation,
projects every candidate onto the probability simplex, and obeys the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures or lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _obj_value(a: np.ndarray, n: int, conv_len: int, fft_len: int) -> float:
    """Objective 2 n · max(conv) for a probability vector a."""
    conv = _conv_fft(a, fft_len, conv_len)
    return 2.0 * n * conv.max()

# Main search routine
# ----------------------------------------------------------------------

    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)  # for a*a

    # --------------------------------------------------------------
    # 2. Build a set of diverse initial probability vectors
    # --------------------------------------------------------------
    init = []

    # uniform
    init.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws – various concentrations
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            init.append(rng.dirichlet(np.ones(n) * alpha))

    # geometric tails
    for r in [0.90, 0.95, 0.975, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init.append(v)

    # exponential (very close to uniform but gently decreasing)
    eps = 1.0 / n
    r = 1.0 - eps
    v = r ** np.arange(n, dtype=np.float64)
    v /= v.sum()
    init.append(v)

    # cosine‑shaped (to break the symmetry with the reversal)
    x = np.arange(n, dtype=np.float64)
    v = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))  # non‑negative “bump”
    v /= v.sum()
    init.append(v)

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (fast FFT evaluation)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init:
        J = _obj_value(cand, n, conv_len, fft_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    total_time = deadline - start_time

    # temperature schedule
    T_start = best_J * 0.10   # start fairly hot
    T_end = 1e-8

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # pre‑compute list of indices that currently have non‑zero mass
    positive_idx = lambda v: np.nonzero(v > 1e-15)[0]

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear cooling
        T = (T_start * (deadline - now) + T_end * (now - start_time)) / total_time

        # ----------------------------------------------------------
        # 4a – make a proposal
        # ----------------------------------------------------------
        proposal = None
        move_type = rng.integers(0, 3)  # 0: small transfer, 1: larger swap, 2: global jitter

        if move_type == 0:
            # tiny mass transfer between two random positions
            pos = positive_idx(cur_vec)
            if pos.size == 0:
                continue
            i = rng.choice(pos)
            j = rng.integers(0, n)
            if i == j:
                continue
            max_transfer = cur_vec[i]
            if max_transfer <= 0.0:
                continue
            delta = max_transfer * rng.random() * 0.5  # up to 50 % of the source mass
            new = cur_vec.copy()
            new[i] -= delta
            new[j] += delta
            proposal = proj_simplex(new)

        elif move_type == 1:
            # move a substantial fraction from one index to another
            pos = positive_idx(cur_vec)
            if pos.size == 0:
                continue
            i = rng.choice(pos)
            j = rng.integers(0, n)
            if i == j:
                continue
            # move a random fraction of the source mass (20‑80 %)
            frac = 0.2 + 0.6 * rng.random()
            delta = cur_vec[i] * frac
            new = cur_vec.copy()
            new[i] -= delta
            new[j] += delta
            proposal = proj_simplex(new)

        else:
            # global jitter – multiplicative noise on every coordinate
            noise = rng.normal(0.0, 0.08, size=n)   # fairly strong jitter
            new = cur_vec * np.exp(noise)          # stay positive
            proposal = proj_simplex(new)

        # ----------------------------------------------------------
        # 4b – evaluate the proposal
        # ----------------------------------------------------------
        prop_J = _obj_value(proposal, n, conv_len, fft_len)

        # ----------------------------------------------------------
        # 4c – Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ----------------------------------------------------------
        # 4d – keep the best solution seen so far
        # ----------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4e – stagnation handling: occasional restart around the best
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # small Gaussian jitter around the current best
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = _obj_value(cur_vec, n, conv_len, fft_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best vector (clip for safety, renormalise)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
