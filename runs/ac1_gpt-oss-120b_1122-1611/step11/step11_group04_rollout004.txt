    fft_len_grad = _next_pow2(grad_len)      # for gradient convolution

    # ------------------------------------------------------------
    # 2. Initialise a feasible probability vector
    # ------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s
        # pad / truncate to the chosen length
        if len(a) < n:
            a = np.pad(a, (0, n - len(a)), constant_values=0.0)
        elif len(a) > n:
            a = a[:n]
    else:
        # a smooth Gaussian‑like start (better than plain uniform)
        x = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        sigma = n / 6.0
        a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        a = a / a.sum()

    # ------------------------------------------------------------
    # 3. Adam state and schedule parameters
    # ------------------------------------------------------------
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    beta_start = 0.5                         # smooth‑max temperature start
    beta_end   = 20.0                        # target temperature
    beta_steps = 4000                        # iterations over which beta grows
    beta_inc = (beta_end - beta_start) / max(1, beta_steps)

    lr0 = 0.8                                # base learning rate for Adam

    # ------------------------------------------------------------
    # 4. Track the best exact solution seen so far
    # ------------------------------------------------------------
    best_a = a.copy()
    best_J = evaluate_sequence(best_a.tolist())
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin
    last_improve = start_time

    print(f"[0.00s] Initial J = {best_J:.6f}")

    it = 0
    while time.monotonic() < deadline:
        # --------------------------------------------------------
        # 4.1  Compute convolution a*a (FFT) and smooth‑max
        # --------------------------------------------------------
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, fft_len)[:conv_len]

        cur_max = conv.max()
        beta = min(beta_end, beta_start + beta_inc * it)

        # stable soft‑max
        shift = cur_max
        exp_term = np.exp(beta * (conv - shift))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp                     # soft‑max weights (len = conv_len)

        # --------------------------------------------------------
        # 4.2  Gradient of the smooth‑max (via convolution)
        # --------------------------------------------------------
        a_rev = a[::-1]
        Af = np.fft.rfft(a_rev, fft_len_grad)
        Wf = np.fft.rfft(w, fft_len_grad)
        grad_full = np.fft.irfft(Af * Wf, fft_len_grad)
        # slice that corresponds to indices 0 … n‑1
        grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # --------------------------------------------------------
        # 4.3  Adam moment updates
        # --------------------------------------------------------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        # step size (decreases like 1/√t)
        lr = lr0 / np.sqrt(it1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------
        # 4.4  Multiplicative (exponentiated) update on the simplex
        # --------------------------------------------------------
        a_new = a * np.exp(-step)
        # avoid all‑zero vector
        a_sum = a_new.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / a_sum

        # --------------------------------------------------------
        # 4.5  Occasionally jitter to escape stagnation
        # --------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > 30.0 and rng.random() < 0.2:
            # Gaussian jitter around the best point seen so far
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            a_new = best_a * noise
            a_new = a_new / a_new.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Jitter applied.")

        a = a_new

        # --------------------------------------------------------
        # 4.6  Periodic exact evaluation (every 100 iterations)
        # --------------------------------------------------------
        if it % 100 == 0:
            J = evaluate_sequence(a.tolist())
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] New best J = {best_J:.6f}")

        it += 1

    # -----------------------------------------------------------------
    # 5. Final exact evaluation and safe clipping
    # -----------------------------------------------------------------
    final_J = evaluate_sequence(best_a.tolist())
    print(f"Final best J = {final_J:.6f}")

    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a = best_a / total

    return [float(v) for v in best_a.tolist()]
