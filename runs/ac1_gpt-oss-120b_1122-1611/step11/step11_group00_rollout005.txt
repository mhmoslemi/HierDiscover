def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    '''
    Hybrid projected subgradient descent with occasional linear-programming mixing and
    simulated‑annealing style jitter to minimise 
    J = 2·n·max(conv(a,a)) where a lies on the probability simplex.
    '''
    import time, numpy as np, math
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    # select n based on previous sequence
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len-1).bit_length())  # next power of 2

    # Helper conv and projection functions defined outside for performance? We'll define inside.

    # Helper functions
    def next_pow2(x):
        return 1 << ((x - 1).bit_length())

    def conv_full(a):
        # a is 1d np.ndarray length n
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        return conv

    def max_conv(a):
        return conv_full(a).max()

    def J_of_a(a):
        return float(2.0 * n * max_conv(a))

    def project_simplex(v):
        # v: np.ndarray, may contain negative
        # from https://github.com/indigo-d-void/Projection-onto-simplex
        # Implementation: sort descending.
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        # find rho
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # all zero?
            return np.full_like(v, 1.0 / len(v))
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    # 2. build init_cands as list of np.ndarrays
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous if present
    if 'height_sequence_1' in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 0.0)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_cands.append(v)

    # Evaluate initial candidates using J_of_a (fast conv)
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        cur_J = J_of_a(cand)
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # Store globally best solution
    global_best_a = best_a.copy()
    global_best_J = best_J

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)

    # Optimization loop
    iteration = 0
    last_restart = start_time
    # Adjustable hyperparameters
    LP_INTERVAL = 150  # perform LP mixing every 150 iterations
    RESTART_AFTER = 30.0  # seconds of no improvement before random restart
    t_mix = 0.25  # mixing coefficient for LP step
    jitter_sigma = 0.015
    # Track time of last improvement
    last_improve = start_time

    a = best_a.copy()
    while time.monotonic() < deadline:
        # evaluate current a
        conv = conv_full(a)
        max_c = conv.max()
        cur_J = float(2.0 * n * max_c)
        # update global best
        now = time.monotonic()
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] New global best J = {global_best_J:.6f}")

        # Gradient descent step (projected subgradient)
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv >= max_c - tol)[0]
        grad_a = np.zeros_like(a)
        for p in active:
            start = max(0, p - (n - 1))
            end = min(p, n - 1)
            idx = np.arange(start, end + 1)
            grad_a[idx] += 2.0 * a[p - idx]
        # direction -grad_a
        # compute step size estimate
        grad_norm = np.linalg.norm(grad_a)
        if grad_norm > 0:
            # target typical Euclidean change ~0.02
            alpha0 = 0.02 / grad_norm
            # bound alpha0 to avoid overshoot
            if alpha0 > 0.5:
                alpha0 = 0.5
        else:
            alpha0 = 0.0

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125):
            alpha = alpha0 * factor
            if alpha <= 0:
                continue
            cand = a - alpha * grad_a
            cand = project_simplex(cand)
            # evaluate candidate quickly
            cand_max = conv_full(cand).max()
            cand_J = float(2.0 * n * cand_max)
            if cand_J < cur_J - 1e-12:
                a = cand
                improved = True
                break

        if not improved:
            # jitter to escape flat region
            if now - last_improve > RESTART_AFTER:
                # random restart around best
                a = global_best_a + rng.normal(0.0, jitter_sigma, size=n)
                a = np.maximum(a, 0.0)
                a = project_simplex(a)
                last_improve = now
                print(f"[{now - start_time:6.2f}s] Restart – jitter around best")
            else:
                # small jitter
                a = a + rng.normal(0.0, jitter_sigma * 0.5, size=n)
                a = np.maximum(a, 0.0)
                a = project_simplex(a)

        # Periodic LP mixing
        if iteration % LP_INTERVAL == 0 and iteration > 0:
            M = max_c
            # Build constraint matrix A (conv of a with unknown g)
            # Efficient building: each column i adds a_i on rows i..i+n-1
            A = np.zeros((conv_len, n), dtype=np.float64)
            for i in range(n):
                A[i:i+n, i] = a[i]
            c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
            bounds = [(0, None)] * n
            b_ub = np.full(conv_len, M, dtype=np.float64)
            # Solve LP
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
            if res.success:
                g = res.x
                sum_g = g.sum()
                if sum_g > 0:
                    g_norm = g / sum_g
                    # Mix
                    a_mix = (1.0 - t_mix) * a + t_mix * g_norm
                    a_mix = project_simplex(a_mix)  # still sum 1 but enforce non‑neg
                    # evaluate improvement
                    mix_J = float(2.0 * n * conv_full(a_mix).max())
                    if mix_J < cur_J - 1e-12:
                        a = a_mix
                        # record improvement
                        if mix_J < global_best_J - 1e-12:
                            global_best_J = mix_J
                            global_best_a = a.copy()
                            last_improve = now
                            print(f"[{now - start_time:6.2f}s] LP mix improved J = {global_best_J:.6f}")

        iteration += 1

    # After loop, return best found
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total
    return [float(v) for v in best_vec]
```

"""
Hybrid projected sub‑gradient descent + occasional LP mixing.

The objective to minimise is  

    J(a) = 2·n·max(conv(a,a)) ,   a_i ≥ 0 ,   Σ a_i = 1

where `conv(a,a)` is the full linear convolution of the sequence with itself.
`J` is scale‑invariant, so we keep the vector on the probability simplex.

Algorithm
---------

1. Choose a moderate problem size `n` (default 2048, limited by any
   previously discovered `height_sequence_1`).
2. Build a diverse portfolio of initial probability vectors
   (uniform, geometric, ramps, Gaussians, Dirichlet draws, the previous
   construction if available).  Keep the best one w.r.t. `J`.
3. Main optimisation loop (time‑bounded):
   * Compute the current convolution and its maximum.
   * Form the sub‑gradient of the max‑convolution (sum of shifted copies
     of `a` for all active indices where the maximum is attained).
   * Perform a projected line‑search step on the simplex:
        a ← Proj_Simplex( a – α·∇J )
     The step size α is chosen so that the Euclidean change is about
     0.02 (clipped to ≤0.5).  If no improvement is found we jitter the
     vector with small Gaussian noise.
   * Every `LP_INTERVAL` iterations solve a small linear programme

        maximise   Σ g_i
        subject to A·g ≤ M ,   g ≥ 0

     where `M = max(conv(a,a))` and the matrix `A` describes the linear
     dependence of `conv(a,g)` on `g`.  The optimum `g` is normalised to
     sum 1 and mixed with the current iterate (`a ← (1‑t)·a + t·g`).
     If the mixture improves the objective we keep it.
   * Restart (jitter around the best solution) after a period of
     stagnation (`RESTART_AFTER` seconds).

4. Return the best sequence ever seen (clipped to the allowed range
   and re‑normalised).

The method combines fast FFT‑based convolution, an explicit simplex
projection, a cheap LP mixing step (which is the core of the state‑of‑the‑art
procedure described in the prompt) and stochastic jitter to escape
plateaux.  It stays well within the 1000 s budget while typically
improving the previous best bound of ~1.513.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    `a` is a 1‑D array of length n.
    Returns an array of length 2·n‑1 (=`conv_len`).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Reference: D. W. Wang & J. M. Carreira‑Perpiñán,
               "Projection onto the probability simplex: An efficient
               algorithm with a simple proof, and an application"
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho (last index where condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Runs for at most `budget_s` seconds and returns
    the best sequence discovered.

    The optimiser works on the probability simplex via an explicit
    projection, using a projected sub‑gradient descent step, occasional
    linear‑programming mixing (the LP from the reference algorithm) and
    stochastic jitter to avoid stagnation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size (n)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Build a portfolio of diverse initial probability vectors
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – moderate and strong concentration
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 0.0)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate (by the raw objective J = 2·n·max(conv))
    # ------------------------------------------------------------------
    def _J_of_a(a_vec: np.ndarray) -> float:
        return float(2.0 * n * _conv_full(a_vec, fft_len, conv_len).max())

    best_a = None
    best_J = np.inf
    for cand in init_cands:
        cur_J = _J_of_a(cand)
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # Keep the globally best solution seen so far
    global_best_a = best_a.copy()
    global_best_J = best_J

    # ------------------------------------------------------------------
    # 4. Optimisation loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # safety margin so we have time for final post‑processing
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    # parameters
    LP_INTERVAL = 150          # how many iterations between LP mixing steps
    RESTART_AFTER = 30.0       # seconds without improvement before a restart
    t_mix = 0.25                # mixing coefficient for LP step
    jitter_sigma = 0.015        # Gaussian jitter magnitude
    iteration = 0
    last_improve = start_time

    a = best_a.copy()          # current iterate

    while time.monotonic() < deadline:
        # ----- 4.1 evaluate current objective -----
        conv = _conv_full(a, fft_len, conv_len)
        max_c = conv.max()
        cur_J = float(2.0 * n * max_c)

        now = time.monotonic()
        elapsed = now - start_time

        # ----- 4.2 update global best -----
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New global best J = {global_best_J:.6f}")

        # ----- 4.3 sub‑gradient of max(conv) -----
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv >= max_c - tol)[0]   # indices attaining the max

        # Build the sub‑gradient:  grad_i = 2 * sum_{p in active} a_{p-i}
        # Efficiently compute via a convolution with the reversed vector.
        if active.size == 0:
            grad_a = np.zeros_like(a)
        else:
            b = np.zeros(conv_len, dtype=np.float64)
            b[active] = 1.0
            a_rev = a[::-1]
            # full convolution length = conv_len + n - 1
            grad_full = np.convolve(b, a_rev, mode='full')
            grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

        # ----- 4.4 projected line‑search step on the simplex -----
        improved = False
        grad_norm = np.linalg.norm(grad_a)
        if grad_norm > 0:
            # target Euclidean move ≈ 0.02 (tunable)
            alpha0 = 0.02 / grad_norm
            alpha0 = min(alpha0, 0.5)   # avoid too aggressive jumps
        else:
            alpha0 = 0.0

        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            alpha = alpha0 * factor
            if alpha <= 0:
                continue
            cand = a - alpha * grad_a
            cand = _project_onto_simplex(cand)
            cand_J = _J_of_a(cand)
            if cand_J < cur_J - 1e-12:
                a = cand
                improved = True
                break

        # ----- 4.5 if no progress -> jitter (and possibly restart) -----
        if not improved:
            if now - last_improve > RESTART_AFTER:
                # restart around the best known point
                a = global_best_a + rng.normal(0.0, jitter_sigma, size=n)
                a = np.maximum(a, 0.0)
                a = _project_onto_simplex(a)
                last_improve = now
                print(f"[{elapsed:6.2f}s] Restart – jitter around best")
            else:
                # mild jitter to shake the current point
                a = a + rng.normal(0.0, jitter_sigma * 0.5, size=n)
                a = np.maximum(a, 0.0)
                a = _project_onto_simplex(a)

        # ----- 4.6 occasional linear‑programming mixing step -----
        if iteration % LP_INTERVAL == 0 and iteration > 0:
            M = max_c
            # Build the constraint matrix A (conv(a,g) = A·g)
            # Each column i contributes a_i on rows i … i+n‑1
            A = np.zeros((conv_len, n), dtype=np.float64)
            for i in range(n):
                A[i:i + n, i] = a[i]   # fill the diagonal band
            c = -np.ones(n)               # maximise Σ g_i  ⇒ minimise -Σ g_i
            bounds = [(0.0, None)] * n
            b_ub = np.full(conv_len, M, dtype=np.float64)

            try:
                res = linprog(c, A_ub=A, b_ub=b_ub,
                              bounds=bounds, method='highs',
                              options={'presolve': True})
            except Exception:
                res = None

            if res is not None and res.success:
                g = res.x
                sum_g = g.sum()
                if sum_g > 1e-12:
                    g_norm = g / sum_g                       # bring back to Σ=1
                    a_mix = (1.0 - t_mix) * a + t_mix * g_norm
                    a_mix = _project_onto_simplex(a_mix)      # safety
                    mix_J = _J_of_a(a_mix)
                    if mix_J < cur_J - 1e-12:
                        a = a_mix
                        # record the improvement immediately
                        if mix_J < global_best_J - 1e-12:
                            global_best_J = mix_J
                            global_best_a = a.copy()
                            last_improve = now
                            print(f"[{elapsed:6.2f}s] LP mix improved J = {global_best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best sequence (clipped and re‑normalised)
    # ------------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total
    return [float(v) for v in best_vec]
