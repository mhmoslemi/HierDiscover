"""
Propose a candidate step‑function using a smoothed‐max gradient descent
combined with simulated‑annealing style acceptance and occasional random
pair exchanges.  The algorithm works directly on the probability simplex
(∑a_i = 1, a_i ≥ 0) so the objective

    J(a) = 2·n·max(conv(a,a))

is scale‑invariant.  At each iteration we

1. compute the full self‑convolution (via a real FFT);
2. form a soft‑max weighting of the convolution values
   w_k = exp(α·(conv_k – max(conv)))  (max weight = 1);
   as α → ∞ this weight concentrates on the active set so the
   gradient approaches the true sub‑gradient of the max‑operator;
3. obtain the gradient of the weighted sum
   ∇J_smooth = 2·conv(w, a_rev)  restricted to the original length,
   where a_rev = a[::-1];
4. take a projected gradient step (with a decreasing step size and a
   momentum term), optionally perform a small random pair‑exchange,
   and project the result onto the simplex;
5. accept the move if it improves the true J, otherwise accept it with
   a Metropolis probability that decays exponentially with time;
6. if no improvement is seen for a while, jitter the current solution
   by multiplicative log‑normal noise centred on the best vector found
   so far.

The loop runs until the time budget (≈1000 s) is exhausted and returns
the best sequence observed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal (or v is zero); return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two 1‑D real arrays using a real FFT.
    Returns the full convolution of length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    z = np.fft.irfft(X * Y, n=fft_len)
    return z[: len(x) + len(y) - 1]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Return the objective J = 2·n·max(conv(a,a)) for a vector that already
    lives on the simplex (i.e. sum(a) == 1).  The computation uses FFT.
    """
    conv = conv_full_fft(a, a, fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for roughly ``budget_s`` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size -------------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep n in a reasonable range for the FFT‑based solver
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)                 # for a*a
    grad_len = 3 * n - 2
    fft_len_grad = _next_pow2(grad_len)                 # for w * a_rev

    # ------------------------------------------------------------------
    # Initial point ------------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= s
        if a0.size != n:
            # fall back to uniform if the stored vector has a different size
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure we truly sit on the simplex
    a0 = proj_simplex(a0)

    # ------------------------------------------------------------------
    # Book‑keeping -------------------------------------------------------
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_J = compute_J(best_a, n, fft_len_conv, conv_len)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin

    # algorithmic parameters
    beta = 0.90                     # momentum decay
    momentum = np.zeros_like(a0)    # momentum buffer
    step_base = 0.55                # base step size (will be scaled by 1/sqrt(t))
    alpha_base = 0.8                # initial soft‑max sharpening parameter
    jitter_sigma = 0.07             # log‑normal jitter magnitude
    last_improve = start_time
    iteration = 0

    # exponential temperature schedule (from current best down to 1e‑8)
    total_time = deadline - start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ------------------------------------------------------------------
        # 1) current convolution and soft‑max weights
        # ------------------------------------------------------------------
        conv = conv_full_fft(a0, a0, fft_len_conv)[:conv_len]
        max_conv = conv.max()

        # gradually sharpen the soft‑max
        alpha = alpha_base * (1.0 + iteration / 1500.0)

        # w_k = exp(alpha * (conv_k - max_conv)) ∈ (0, 1]
        w = np.exp(alpha * (conv - max_conv))

        # ------------------------------------------------------------------
        # 2) gradient of the weighted sum (smooth surrogate of max)
        #    grad_i = 2 * Σ_j w_{i+j} * a_j   → 2 * conv(w, a_rev) slice
        # ------------------------------------------------------------------
        a_rev = a0[::-1]
        conv_w = conv_full_fft(w, a_rev, fft_len_grad)  # length = grad_len
        grad = 2.0 * conv_w[n - 1 : n - 1 + n]  # keep only the n entries

        # ------------------------------------------------------------------
        # 3) momentum + projected gradient step
        # ------------------------------------------------------------------
        step = step_base / np.sqrt(iteration + 1.0)
        momentum = beta * momentum - step * grad
        cand = a0 + momentum

        # ------------------------------------------------------------------
        # 4) occasional random pair‑exchange to explore invisible directions
        # ------------------------------------------------------------------
        if rng.random() < 0.18:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(cand[i], rng.random() * 0.012)   # move ≤ 1.2 % of total mass
            cand[i] -= delta
            cand[j] += delta

        # ------------------------------------------------------------------
        # 5) projection back onto the simplex
        # ------------------------------------------------------------------
        cand = proj_simplex(cand)

        # ------------------------------------------------------------------
        # 6) evaluate true objective
        # ------------------------------------------------------------------
        cand_J = compute_J(cand, n, fft_len_conv, conv_len)

        # ------------------------------------------------------------------
        # 7) simulated‑annealing acceptance
        # ------------------------------------------------------------------
        # exponential cooling: T goes from best_J to ~1e‑8
        ratio = elapsed / total_time
        T = best_J * (1e-8 / best_J) ** ratio
        accept = False
        if cand_J < best_J:
            accept = True
        else:
            prob = np.exp(-(cand_J - best_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a0 = cand
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_a = cand.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 8) stagnation handling – multiplicative jitter around the best
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a0 = best_a * noise
            a0 = proj_simplex(a0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
