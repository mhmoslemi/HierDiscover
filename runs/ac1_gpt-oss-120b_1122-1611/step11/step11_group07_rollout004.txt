    # Determine n

    # Build initial candidates

    candidates.append(v)

        candidates.append(v)

    # Optimization parameters

    # smoothing schedule
    beta_min = 0.1

    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

"""
Hybrid smooth‑max gradient + stochastic exchange optimiser.

* Works on the probability simplex (∑a = 1, a_i ≥ 0) – the objective
  J(a) = 2·n·max(conv(a,a)) is invariant under scaling.
* A smooth approximation of the max (soft‑max with a temperature β) is
  minimised by (projected) gradient descent using FFT‑based convolutions.
* β is increased exponentially from 0.1 to 1000 so that the optimisation
  starts with a very smooth landscape and ends up targeting the true
  maximum.
* The raw gradient step is projected onto the simplex; a fraction of
  iterations an “exchange” move (transfer a tiny amount of mass between
  two coordinates) is performed to explore directions invisible to the
  linearised gradient.
* A simple Metropolis acceptance rule with an exponential cooling schedule
  lets the search escape shallow plateaus while still converging.
* After 15 s without improvement a short random jitter around the current
  best solution is applied.
* The final vector is clipped to the allowed range [0, 1000] and re‑normalised.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal (or v is zero); return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Return the objective J = 2·n·max(conv(a,a)) for a vector a that
    already lives on the simplex (∑a == 1).
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – use existing construction if available,
    #    otherwise a moderate power‑of‑two size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048
    # enforce a power of two for fast FFTs
    n = _next_pow2(n)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial points (all on the simplex).
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Hann / raised‑cosine windows (zero at the ends)
    v = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    v /= v.sum()
    init_cands.append(v)

    # Dirichlet draws – moderate and strong concentration
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v = v / s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate (cheap objective evaluation).
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation loop (smooth‑max gradient + stochastic exchange +
    #    simulated annealing).
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    total_time = deadline - start_time

    # temperature schedule (exponential cooling)
    T_start = best_J
    T_end = 1e-8

    # smoothing schedule for the soft‑max
    beta_min = 0.1
    beta_max = 1000.0

    # step‑size schedule
    step0 = 0.3

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # cooling & smoothing parameters
        # --------------------------------------------------------------
        T = T_start * (T_end / T_start) ** (elapsed / total_time)
        beta = beta_min * (beta_max / beta_min) ** (elapsed / total_time)

        # --------------------------------------------------------------
        # stagnation handling – jitter after 15 s without improvement
        # --------------------------------------------------------------
        if now - last_improve > 15.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = compute_J(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – cur J = {cur_J:.6f}")

        # --------------------------------------------------------------
        # decide which move to propose
        # --------------------------------------------------------------
        if rng.random() < 0.70:
            # ---- gradient move (smooth max) -------------------------
            # compute full self‑convolution
            conv = _conv_full(cur_vec, fft_len, conv_len)

            # soft‑max weights w_i = exp(beta*conv_i) / sum_j exp(beta*conv_j)
            scaled = beta * conv
            # stable log‑sum‑exp
            lse = logsumexp(scaled)
            w = np.exp(scaled - lse)  # sum(w) == 1

            # gradient of the smoothed objective:
            #   grad = 4 * n * (w * conv a_rev) slice
            a_rev = cur_vec[::-1]
            w_fft = np.fft.rfft(w, n=fft_len)
            a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
            grad_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)
            # slice that corresponds to indices 0 … n‑1
            grad_slice = grad_full[n - 1:n - 1 + n]

            grad = 4.0 * n * grad_slice

            # normalise the direction
            norm = np.linalg.norm(grad)
            if norm > 0:
                direction = grad / norm
            else:
                direction = np.zeros_like(grad)

            step = step0 / np.sqrt(iteration + 1.0)
            prop = cur_vec - step * direction
        else:
            # ---- exchange move --------------------------------------
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(cur_vec[i], rng.random() * 0.02)   # ≤ 2 % of mass
            prop = cur_vec.copy()
            prop[i] -= delta
            prop[j] += delta

        # --------------------------------------------------------------
        # projection onto the simplex
        # --------------------------------------------------------------
        prop = proj_simplex(prop)

        # --------------------------------------------------------------
        # evaluate actual objective
        # --------------------------------------------------------------
        prop_J = compute_J(prop, n, fft_len, conv_len)

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = prop
            cur_J = prop_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
