def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 1.0

    n = 8192   # (maybe adjust based on time)
    conv_len = 2*n -1
    fft_len_conv = _next_pow2(conv_len)
    fft_len_corr = _next_pow2(conv_len + n -1)

    # Generate initial vectors
    init_vecs = []
    # uniform
    init_vecs.append(np.full(n, 1.0/n, dtype=np.float64))
    # sine bell
    x = np.arange(n, dtype=np.float64)
    v = np.sin(np.pi * (x + 0.5) / n)  # positive
    v = np.maximum(v, 0)
    v /= v.sum()
    init_vecs.append(v)
    # geometric
    for r in [0.95, 0.97, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_vecs.append(v)
    # random Dirichlet draws
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(2):
            init_vecs.append(rng.dirichlet(np.ones(n) * alpha))

    # choose best initial according to actual J
    best_vec = None
    best_J = np.inf
    for v in init_vecs:
        conv = _conv_full(v, fft_len_conv, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = v.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # set up Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5
    # log-space
    y = np.log(best_vec + 1e-30)   # to avoid log zero
    m = np.zeros_like(y)
    v_ = np.zeros_like(y)
    t = 0
    cur_best_vec = best_vec.copy()
    cur_best_J = best_J
    last_improve = start_time

    # schedule for beta (softmax temperature)
    beta = 2.0
    beta_max = 2000.0
    beta_inc_factor = 1.005

    while time.monotonic() < deadline:
        t += 1
        # compute current a
        a = np.exp(y)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0/n)
            a_sum = 1.0
        a = a / a_sum

        # compute conv and max
        conv = _conv_full(a, fft_len_conv, conv_len)
        max_conv = conv.max()
        J = float(2.0 * n * max_conv)
        # keep global best
        if J < cur_best_J - 1e-12:
            cur_best_J = J
            cur_best_vec = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {cur_best_J:.6f}")

        # stagnation detection - jitter
        now = time.monotonic()
        if now - last_improve > 20.0:
            # jitter by random log noise
            y = np.log(cur_best_vec + 1e-30)
            noise = rng.normal(0.0, 0.1, size=n)
            y += noise
            # reset Adam moments
            m.fill(0.0)
            v_.fill(0.0)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {cur_best_J:.6f}")

        # compute gradient of smoothed max w.r.t a
        grad_a = _gradient_softmax(a, conv, beta, fft_len_corr)   # shape (n,)

        # gradient w.r.t y
        grad_y = a * grad_a

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v_ = beta2 * v_ + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v_ / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y -= step

        # increase beta gradually
        if beta < beta_max:
            beta = min(beta * beta_inc_factor, beta_max)

    # after loop, assign best vector
    final_vec = cur_best_vec
    # Clip for safety
    final_vec = np.clip(final_vec, 0.0, 1000.0)
    total = final_vec.sum()
    if total == 0.0:
        final_vec[:] = 1.0 / n
    else:
        final_vec /= total
    return [float(v) for v in final_vec]

"""
Simple yet powerful Adam‑based mirror‑descent optimizer for the
self‑convolution L∞ problem.

The algorithm works on the probability simplex (sum of elements = 1)
by representing the vector in log‑space and applying Adam updates to
the log‑variables.  The true objective  J = 2·n·max(conv)  is smoothened
with a temperature‑controlled soft‑max (log‑sum‑exp) so that the
gradient is well‑behaved early on and gradually approximates the true
max‑gradient.  The temperature β is exponentially increased up to a
large value (≈2000) and the learning‑rate is annealed as 1/√t.
Periodical “jitter” restarts prevent the iterate from getting stuck.
Multiple diverse initialisations (uniform, geometric, Gaussian, Dirichlet
draws and the previous solution if present) are tried; the best
candidate found before the time budget expires is returned.

The implementation uses only NumPy, FFT‑based convolutions and
projecting onto the simplex, and comfortably fits into the 1000 s
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to one.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns the first ``conv_len`` entries (the full linear convolution).
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Stable soft‑max weights for a vector ``conv``.
    Returns a probability vector w with Σw = 1.
    """
    shift = conv.max()
    scaled = beta * (conv - shift)
    # ``scaled`` ≤ 0, overflow is impossible
    exp_vals = np.exp(scaled)
    s = exp_vals.sum()
    if s == 0.0:          # extreme β → only the max survives
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
        return w
    return exp_vals / s


def _gradient_softmax(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_corr: int) -> np.ndarray:
    """
    Gradient of the smoothed‑max objective
        f(a) = (1/β)·log Σ_k exp(β·conv_k)
    w.r.t. ``a``.
    The gradient equals 2·(w ⋆ a) where w are the soft‑max weights
    and ⋆ denotes linear cross‑correlation.
    """
    w = _softmax_weights(conv, beta)               # shape (2n‑1,)
    L = fft_len_corr
    A_hat = np.fft.rfft(a, n=L)
    W_hat = np.fft.rfft(w, n=L)
    # cross‑correlation via conj(FFT(a))·FFT(w)
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)
    n = a.size
    start = n - 1                                   # zero‑lag index
    grad = 2.0 * corr[start:start + n]              # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine – the function required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – we pick a power‑of‑two for cheap FFTs.
    #    Larger n gives a lower theoretical floor, but 8192 is a good
    #    compromise for the time budget.
    # ------------------------------------------------------------------
    n = 8192
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)                     # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)             # for gradient

    # ------------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors.
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells (a few widths)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 5.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderately / strongly concentrated)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Sparse masks – a few equally weighted spikes
    for k in (int(np.sqrt(n)), int(np.sqrt(n) * 2), max(2, n // 10)):
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        init_cands.append(mask)

    # Random binary mask (Bernoulli 0.5)
    bin_mask = rng.binomial(1, 0.5, size=n).astype(np.float64)
    if bin_mask.sum() > 0:
        bin_mask /= bin_mask.sum()
        init_cands.append(bin_mask)

    # If a previous construction exists, use it (padded/truncated).
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size > n:
            prev = prev[:n]
        elif prev.size < n:
            prev = np.concatenate([prev, np.zeros(n - prev.size, dtype=np.float64)])
        s = prev.sum()
        if s == 0.0:
            prev = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            prev = prev / s
        init_cands.append(prev)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate according to the **exact**
    #    objective (no surrogate).
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len_conv, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    start = time.monotonic()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam‑based mirror descent on the probability simplex.
    # ------------------------------------------------------------------
    # Log‑space representation: a = softmax(y)
    y = np.log(best_vec + 1e-30)          # avoid log(0)
    m = np.zeros_like(y)                 # first moment
    v_hat = np.zeros_like(y)             # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # learning‑rate schedule (base 0.5, decays as 1/√t)
    base_lr = 0.5

    # soft‑max temperature schedule
    beta = 2.0               # start gentle
    beta_max = 2000.0
    beta_inc = 1.005        # multiplicative increase each iteration

    # bookkeeping
    cur_best_vec = best_vec.copy()
    cur_best_J = best_J
    last_improve = start
    iteration = 0

    # safety margin – leave a second for clean‑up
    deadline = start + max(0.0, float(budget_s) - 1.0)

    while time.monotonic() < deadline:
        iteration += 1

        # ----- 4.1 obtain current probability vector -----------------
        a = np.exp(y)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
            s = 1.0
        a /= s

        # ----- 4.2 evaluate exact objective -------------------------
        conv = _conv_full(a, fft_len_conv, conv_len)
        max_conv = conv.max()
        J = float(2.0 * n * max_conv)

        # keep globally best
        if J < cur_best_J - 1e-12:
            cur_best_J = J
            cur_best_vec = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] New best J = {cur_best_J:.6f}")

        # ----- 4.3 stagnation handling – jitter restart -------------
        now = time.monotonic()
        if now - last_improve > 20.0:
            # inject log‑space Gaussian noise around the best solution
            y = np.log(cur_best_vec + 1e-30)
            y += rng.normal(0.0, 0.09, size=n)   # modest jitter
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] Jitter – restart, best J = {cur_best_J:.6f}")

        # ----- 4.4 gradient of the smoothed‑max objective ------------
        grad_a = _gradient_softmax(a, conv, beta, fft_len_corr)   # shape (n,)

        # gradient w.r.t. log‑variables (mirror‑descent)
        grad_y = a * grad_a

        # ----- 4.5 Adam update ---------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_y ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** iteration)
        v_corr = v_hat / (1.0 - beta2 ** iteration)

        # learning‑rate decays as 1/√t
        lr = base_lr / math.sqrt(iteration)

        # additive update in log‑space
        y -= lr * m_hat / (np.sqrt(v_corr) + eps)

        # ----- 4.6 increase temperature (sharpen soft‑max) ----------
        if beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

    # ------------------------------------------------------------------
    # 5. Return the best vector discovered (clip & renormalise for safety)
    # ------------------------------------------------------------------
    final_vec = cur_best_vec
    final_vec = np.clip(final_vec, 0.0, 1000.0)
    tot = final_vec.sum()
    if tot == 0.0:
        final_vec[:] = 1.0 / n
    else:
        final_vec /= tot

    # Convert to plain Python floats (the evaluator expects list[float])
    return [float(v) for v in final_vec]
