"""
Hybrid sparse‑search + adaptive Adam‑gradient optimizer.

1. Choose a problem size `n`.  If a previous construction exists its length is
   re‑used, otherwise a medium size (≈4000) is taken – large enough to allow a
   low peak autocorrelation but still cheap to evaluate many candidates.

2. **Phase 1 – random sparse supports**  
   Random binary vectors with a controllable number of non‑zero entries are
   generated.  For a binary vector the objective reduces to  

        J = 2·n·max(conv) / (|S|²)

   (|S| = number of ones).  This phase quickly discovers promising supports
   that already give J ≲ 1.55.

3. **Phase 2 – smooth‑max Adam refinement**  
   Each of the best sparse supports is turned into a dense probability vector
   (`a_i ≥ 0`, Σa_i = 1) and optimised with the smoothed objective  

        f_β(a) = Σ_k w_k·(a*a)_k ,   w_k ∝ exp(β·(a*a)_k).

   The gradient w.r.t. `a` is `∇f = 2·ifft(conj(FFT(w))·FFT(a))`.  An Adam
   optimiser (additive updates) is applied, the step is projected back onto the
   probability simplex and the temperature β is increased linearly up to
   `β_max = 5000` so that the smoothed loss concentrates on the true maximum.
   Whenever no improvement is seen for 30 s a small multiplicative jitter is
   applied to escape possible plateaus.

4. The best vector seen over the whole run (including the previous
   construction, if any) is returned after clipping to the allowed range
   `[0,1000]` and normalising to Σa = 1 (the evaluation function is scale‑invariant).

The whole routine respects the time budget (`budget_s` seconds) and prints
status updates.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

    `evaluate_sequence`.  Returns the best normalised sequence found within the
    allocated time budget.
    """

    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:

        except Exception:
            n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    best_vec = None

    max_sparse_iters = 2500
    sparse_iter = 0
    while time.monotonic() < start_time + budget_s - 20.0 and sparse_iter < max_sparse_iters:
        # pick a random support size – between 2 % and 30 % of n
        low = max(2, int(0.02 * n))
        high = max(low + 1, int(0.30 * n))
        m = rng.integers(low, high + 1)

        idx = rng.choice(n, size=m, replace=False)
        a_bin = np.zeros(n, dtype=np.float64)
        a_bin[idx] = 1.0

        # binary objective (scale invariant, sum = m)
        conv = _conv_full(a_bin, fft_len, conv_len)
        max_conv = conv.max()
        J = float(2.0 * n * max_conv / (m * m))

        _register_candidate(a_bin, J, tag="sparse-")
        sparse_iter += 1

    # --------------------------------------------------------------
    # 6. Phase 2 – Adam refinement of the best sparse patterns
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.15                     # base learning rate (additive)

    # soft‑max temperature schedule
    beta_start = 1.0
    beta_max = 5000.0
    # we will increase β linearly with the number of Adam steps
    beta_inc = (beta_max - beta_start) / max(1, int(0.9 * (budget_s - (time.monotonic() - start_time)) / 0.01))

    # time bookkeeping
    global_best_improve = time.monotonic()

    for pat_idx, (pat_vec, pat_J) in enumerate(best_patterns):
        # Convert binary pattern to a dense probability vector (sum = 1)
        a = pat_vec.astype(np.float64)
        total = a.sum()
        if total == 0.0:
            a[:] = 1.0 / n
        else:
            a /= total

        m1 = np.zeros_like(a)      # first Adam moment
        v1 = np.zeros_like(a)      # second Adam moment

        beta = beta_start
        iteration = 0
        last_improve = time.monotonic()

        while time.monotonic() < start_time + budget_s - 5.0:
            # ------------------------------------------------------
            # 6.1  Current objective (scale invariant – sum = 1)
            # ------------------------------------------------------
            conv = _conv_full(a, fft_len, conv_len)
            max_conv = conv.max()
            cur_J = float(2.0 * n * max_conv)   # sum = 1 → denominator = 1

            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] IMPROVED GLOBAL J = {best_J:.6f}")

            # ------------------------------------------------------
            # 6.2  Stagnation handling – jitter after 30 s without improvement
            # ------------------------------------------------------
            if time.monotonic() - last_improve > 30.0:
                jitter = np.exp(rng.normal(0.0, 0.06, size=n))
                a = a * jitter
                a = proj_simplex(a)
                # re‑initialise Adam moments to avoid stale accumulated gradients
                m1.fill(0.0)
                v1.fill(0.0)
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] Jittered pattern {pat_idx}/{len(best_patterns)}")

            # ------------------------------------------------------
            # 6.3  Soft‑max weighting (β grows slowly)
            # ------------------------------------------------------
            beta = min(beta_max, beta_start + beta_inc * iteration)
            # stability shift – subtract max before exponentiation
            w_raw = np.exp(beta * (conv - max_conv))
            w = w_raw / w_raw.sum()

            # ------------------------------------------------------
            # 6.4  Gradient of the smoothed objective
            # ------------------------------------------------------
            A_fft = np.fft.rfft(a, fft_len)
            W_fft = np.fft.rfft(w, fft_len)
            g_full = np.fft.irfft(np.conj(W_fft) * A_fft, n=fft_len)
            g = 2.0 * g_full[:n]                     # real part only

            # ------------------------------------------------------
            # 6.5  Adam additive update (projected onto simplex)
            # ------------------------------------------------------
            iteration += 1
            lr = lr0 / np.sqrt(iteration)            # 1/√t decay

            m1 = beta1 * m1 + (1.0 - beta1) * g
            v1 = beta2 * v1 + (1.0 - beta2) * (g * g)

            m_hat = m1 / (1.0 - beta1 ** iteration)
            v_hat = v1 / (1.0 - beta2 ** iteration)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            # keep on the simplex (also guarantees non‑negativity)
            a = proj_simplex(a)

            # ------------------------------------------------------
            # 6.6  Early exit for this pattern if time is nearly exhausted
            # ------------------------------------------------------
            if time.monotonic() - start_time > budget_s - 5.0:
                break

    # --------------------------------------------------------------
    # 7. Return the best solution (clip, renormalise)
    # --------------------------------------------------------------
    if best_vec is None:
        # fallback – uniform vector
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    # Convert to plain Python list of floats as required by the evaluator
    return [float(v) for v in best_vec]
