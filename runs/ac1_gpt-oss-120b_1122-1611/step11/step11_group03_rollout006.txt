def propose_candidate(seed=..., budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 2048

    # define fft params
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len-1).bit_length())  # next power of two

    # Pre-allocate maybe we can keep buffer for transforms.

    # Helper: max_conv(vec)
        A = np.fft.rfft(vec, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        return conv.max()

    # Compute J = 2*n*max_conv

    # Initialize population of NP vectors.
    NP = 40
    pop = []
    for i in range(NP):
        # random alphas for Dirichlet
        alpha = 10**rng.uniform(-2, 0.5)  # between 0.01 and ~3
        # random sparse: randomly choose support size s in [5, n//4].
        if rng.random() < 0.3:
            k = rng.integers(2, min(200, n//4))
            idx = rng.choice(n, size=k, replace=False)
            weights = rng.random(k)
            weights = weights / weights.sum()
            vec = np.zeros(n)
            vec[idx] = weights
        else:
            vec = rng.dirichlet(np.ones(n) * alpha)
        # Guarantee min sum > 0.01 (should be 1)
        pop.append(vec)
    # Evaluate J for each:
    Js = np.empty(NP)
    for i in range(NP):
        Js[i] = compute_J(pop[i])

    best_idx = int(np.argmin(Js))
    best_vec = pop[best_idx].copy()
    best_J = Js[best_idx]

    start_time = time.monotonic()
    deadline = start_time + budget_s - 5.0  # 5 sec safety

    # DE parameters
    F = 0.8
    CR = 0.9

    # track stagnation: if no improvement for long, reinitialize half population randomly.
    last_improve_time = start_time

    iteration = 0

    while time.monotonic() < deadline:
        for i in range(NP):
            # Choose distinct r1, r2, r3 != i
            idxs = list(range(NP))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            # Mutation: donor = pop[best_idx] + F*(pop[r1] - pop[r2])
            donor = best_vec + F * (pop[r1] - pop[r2])
            # Ensure non-negative and simplex: project
            donor = proj_simplex(donor)
            # Crossover:
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, donor, pop[i])
            # Project onto simplex
            trial = proj_simplex(trial)

            # Evaluate
            J_trial = compute_J(trial)

            # Selection
            if J_trial < Js[i]:
                pop[i] = trial
                Js[i] = J_trial
                if J_trial < best_J:
                    best_J = J_trial
                    best_vec = trial.copy()
                    last_improve_time = time.monotonic()
                    elapsed = last_improve_time - start_time
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

        # Stagnation handling
        if time.monotonic() - last_improve_time > 30.0:
            # replace half of population with fresh random candidates
            for i in range(NP//2, NP):
                alpha = 10**rng.uniform(-2, 0.5)
                vec = rng.dirichlet(np.ones(n) * alpha)
                pop[i] = proj_simplex(vec)
                Js[i] = compute_J(pop[i])
            print(f"[{time.monotonic() - start_time:6.2f}s] Reinitialized half population")
            last_improve_time = time.monotonic()
    # After time out, return best_vec after clipping clamps
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0/n)
    else:
        best_vec = best_vec / total
    return best_vec.tolist()

"""
Hybrid Differential‑Evolution + Linear‑Program refinement.

* The objective to minimise is  J(a) = 2·n·max(conv(a,a))  (the scale of a does not
  matter, therefore we work on the probability simplex Σa_i = 1, a_i ≥ 0).
* A population of candidate vectors is evolved with a classic DE/best/1
  strategy – mutation, binomial crossover and Euclidean projection onto the simplex.
  Stagnation triggers a partial re‑initialisation.
* When the DE phase finishes (or a few seconds are left) we solve a small linear
  programme:
        maximise   Σ g_i
        subject to  conv(f,g)_k ≤ M   for all k   (M = max(conv(f,f)))
                    g_i ≥ 0
  where the convolution constraints are linear in g.  The solution g provides a
  direction that cannot increase the current worst convolution value.
  We move a small step towards the normalised g and keep the improvement.
* Finally a short stochastic local‑search (multiplicative jitter) uses the
  remaining budget.
* All vectors are projected onto the simplex after every modification.
"""

import time
import numpy as np
import cvxpy as cp


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the linear self‑convolution of *vec*."""
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Full objective 2·n·max(conv(vec,vec))."""
    return 2.0 * n * conv_max(vec, fft_len, conv_len)


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix Q such that Q @ g = conv(f,g) (linear convolution).
    Row k, column j contains f_{k‑j} when the index is in range, otherwise 0.
    """
    n = f.size
    conv_len = 2 * n - 1
    Q = np.zeros((conv_len, n), dtype=np.float64)
    for j in range(n):
        Q[j:j + n, j] = f  # fills the anti‑diagonal with f
    return Q


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The search consists of a Differential Evolution phase,
    an LP‑based refinement, and a final stochastic jitter.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        n = max(256, min(n, 8192))
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)                # length for self‑convolution
    # (no need for a second fft length – the LP works in the time domain)

    # ------------------------------------------------------------------
    # 2. Helpers for the objective
    # ------------------------------------------------------------------
    def objective(v: np.ndarray) -> float:
        return compute_J(v, n, fft_len, conv_len)

    # ------------------------------------------------------------------
    # 3. Build an initial diverse population (size NP)
    # ------------------------------------------------------------------
    NP = 40                      # population size
    population = []
    # a few hand‑crafted seeds
    population.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        population.append(v)
    # linear ramps
    for asc in (True, False):
        v = np.arange(1, n + 1, dtype=np.float64) if asc else np.arange(n, 0, -1, dtype=np.float64)
        v /= v.sum()
        population.append(v)
    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        population.append(v)
    # Dirichlet draws with a range of concentrations
    for alpha in (0.05, 0.1, 0.2, 0.3, 0.6, 1.0):
        for _ in range(2):
            population.append(rng.dirichlet(np.ones(n) * alpha))

    # sparse random draws (small support)
    for _ in range(6):
        k = rng.integers(3, min(200, n // 4))
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.random(k)
        weights /= weights.sum()
        v = np.zeros(n, dtype=np.float64)
        v[idx] = weights
        population.append(v)

    # include the previously found construction if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = 1.0 / n
        else:
            v /= v.sum()
        population.append(v)

    # Trim to exactly NP elements (randomly if we have more)
    if len(population) > NP:
        population = rng.choice(population, size=NP, replace=False).tolist()
    else:
        # pad with extra Dirichlet draws if we have fewer than NP
        while len(population) < NP:
            population.append(rng.dirichlet(np.ones(n)))
    # Ensure all members are on the simplex
    population = [proj_simplex(v) for v in population]

    # ------------------------------------------------------------------
    # 4. Evaluate the initial population
    # ------------------------------------------------------------------
    fitness = np.empty(NP, dtype=np.float64)
    for i, v in enumerate(population):
        fitness[i] = objective(v)

    best_idx = int(np.argmin(fitness))
    best_vec = population[best_idx].copy()
    best_J = fitness[best_idx]

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    total_time = deadline - start_time

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. DE optimisation phase
    # ------------------------------------------------------------------
    F = 0.8                     # differential weight
    CR = 0.9                    # crossover probability
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- stagnation handling -----
        if now - last_improve > 30.0:
            # re‑initialise half the population with fresh random draws
            for i in range(NP // 2, NP):
                alpha = 10 ** rng.uniform(-2, 0.5)
                vec = rng.dirichlet(np.ones(n) * alpha)
                population[i] = proj_simplex(vec)
                fitness[i] = objective(population[i])
            print(f"[{elapsed:6.2f}s] Restart – half population re‑initialised")
            last_improve = now

        # ----- DE generation -----
        for i in range(NP):
            # select three distinct indices different from i
            idxs = list(range(NP))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

            # donor uses the current best (DE/best/1)
            donor = population[i] + F * (best_vec - population[i]) + F * (population[r1] - population[r2])
            donor = proj_simplex(donor)

            # binomial crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, donor, population[i])
            trial = proj_simplex(trial)

            # evaluation
            trial_J = objective(trial)

            # selection
            if trial_J < fitness[i]:
                population[i] = trial
                fitness[i] = trial_J
                if trial_J < best_J - 1e-12:
                    best_J = trial_J
                    best_vec = trial.copy()
                    last_improve = time.monotonic()
                    print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 6. LP‑based refinement (if time permits)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 2.0:      # keep a few seconds for the final jitter
        max_lp_steps = 6
        lp_step = 0
        while lp_step < max_lp_steps and time.monotonic() < deadline - 1.0:
            # current worst convolution value
            M = conv_max(best_vec, fft_len, conv_len)

            # build the linear convolution matrix Q (conv(f,g) = Q @ g)
            Q = np.zeros((conv_len, n), dtype=np.float64)
            for j in range(n):
                Q[j:j + n, j] = best_vec

            # linear programme
            g = cp.Variable(n, nonneg=True)
            constraints = [Q @ g <= M * np.ones(conv_len)]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            except Exception:  # any solver failure → abort LP phase
                break
            if g.value is None:
                break

            g_opt = np.maximum(g.value, 0.0)
            if g_opt.sum() == 0.0:
                break
            g_norm = g_opt / g_opt.sum()

            # step towards the LP solution
            t = 0.2
            new_vec = (1.0 - t) * best_vec + t * g_norm
            new_vec = proj_simplex(new_vec)

            new_J = objective(new_vec)
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_vec = new_vec.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] LP step {lp_step+1} → J = {best_J:.6f}")
                # recompute the next LP with the improved vector
                lp_step += 1
                # keep the best vector as the new “current best” for future DE‑like moves
                best_idx = np.argmin(fitness)
                population[best_idx] = best_vec
                fitness[best_idx] = best_J
            else:
                # no improvement – stop the LP refinement
                break

    # ------------------------------------------------------------------
    # 7. Final stochastic jitter (tiny multiplicative noise)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        # gradually shrink the jitter radius
        sigma = max(0.005, 0.2 * remaining / total_time)
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_vec * noise
        cand = proj_simplex(cand)
        cand_J = objective(cand)
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand
            print(f"[{time.monotonic() - start_time:6.2f}s] Jitter improvement → J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 8. Return the best sequence found (clipped and normalised for safety)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
