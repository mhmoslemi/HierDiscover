"""
Simulated‑annealing search on the simplex.

* The search works directly on probability vectors (∑a_i = 1, a_i ≥ 0).
* A simple Metropolis acceptance rule with a linear temperature schedule
  makes it possible to escape local minima.
* Each iteration proposes a tiny mass transfer from a source index (chosen
  proportionally to its current weight) to a random target index.
* The objective  J = 2·n·max(conv(a,a))  is evaluated with a fast FFT‑based
  linear convolution.  The best vector seen during the run is kept.
* If no improvement is observed for a while we “jitter’’ the current
  state (multiplicative log‑normal noise) and re‑project onto the simplex.
* The algorithm starts from a diverse set of initial candidates
  (uniform, geometric decays, Dirichlet draws, and the previous
  construction if available) and runs until the allotted time budget
  (≈1000 s) expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a 1‑D array onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Simulated‑annealing optimiser for ``evaluate_sequence``.
    Returns the best non‑negative step‑function (list of heights) found
    within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size (respect the size of any previous construction)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # safe for linear convolution

    # --------------------------------------------------------------
    # 2. Build a palette of diverse initial probability vectors
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – a range of concentration parameters
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v /= s
            init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (fast FFT evaluation)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing main loop
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J

    best_vec = cur_vec.copy()
    best_J = cur_J
    start_time = time.monotonic()

    total_time = deadline - start_time

    T_start = best_J * 0.2          # start at 20 % of the current best
    T_end = 1e-6

    iteration = 0

            cur_J = float(2.0 * n * _conv_full(cur_vec, fft_len, conv_len).max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Final safety clipping / normalisation
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # return as a plain Python list for the caller
    return [float(v) for v in best_vec]
