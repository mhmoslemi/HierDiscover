def simplex_projection(v):
    """Projection onto probability simplex."""
    # based on algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
A hybrid random‑search + smoothed‑max gradient descent for the
step‑function optimisation problem.

* First a portfolio of parametric families (geometric, power‑law,
  linear ramps, Dirichlet draws, the previous best construction, …)
  is sampled for a few seconds.  Each candidate is evaluated with a fast
  FFT‑based convolution and the best one (lowest `2·n·max(conv)`) is kept.
* Starting from that best candidate we perform a projected gradient descent
  on the smooth approximation  Jα(a)=2·n/α·log∑exp(α·(a*a)).
  The gradient is obtained by a single FFT per iteration (self‑convolution
  to compute the soft‑max weights and another FFT for the correlation that
  gives the gradient).  The iterate is projected onto the probability simplex
  after every step.
* A simple learning‑rate schedule (1/√t) and an adaptive temperature
  (α slowly increased) are used.  If no progress is seen for ≈30 s a
  Gaussian jitter is applied to escape plateaux.
* The routine respects the supplied time budget and finally returns the
  best sequence discovered, clipped to the allowed range.
"""

from __future__ import annotations
import time
import numpy as np
from typing import List

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_full_fft(a: np.ndarray, fft_len: int, out_len: int) -> np.ndarray:
    """Full linear convolution a * a using real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:out_len]
    return conv

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero, return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(a: np.ndarray, fft_len: int) -> float:
    """Return the evaluation objective 2·n·max(conv) for a (already normalized)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    conv = _conv_full_fft(a, fft_len, conv_len)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)

def _generate_candidates(n: int, rng: np.random.Generator, existing: np.ndarray | None) -> List[np.ndarray]:
    """Return a list of candidate probability vectors of length n."""
    candidates: List[np.ndarray] = []

    # Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Linear decreasing ramp (largest at start, smallest at end)
    ramp_desc = np.arange(n, 0, -1, dtype=np.float64)
    ramp_desc /= ramp_desc.sum()
    candidates.append(ramp_desc)

    # Linear increasing ramp
    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    candidates.append(ramp_inc)

    # Quadratic decreasing ramp
    quad_desc = (np.arange(n, 0, -1, dtype=np.float64) ** 2)
    quad_desc /= quad_desc.sum()
    candidates.append(quad_desc)

    # Geometric families
    for r in (0.90, 0.95, 0.97, 0.98, 0.99, 0.995, 0.998):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # Power‑law decreasing families a_i ∝ (i+1)^(-p)
    for p in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        v = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        v /= v.sum()
        candidates.append(v)

    # Symmetric “U‑shape’’ a_i ∝ (i+1)^(-p) + (n-i)^(-p)
    for p in (0.5, 1.0, 1.5, 2.0):
        left = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        right = left[::-1]
        v = left + right
        v /= v.sum()
        candidates.append(v)

    # Dirichlet draws with different concentration parameters
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            candidates.append(rng.dirichlet(np.full(n, alpha)))

    # Random Gamma draws with a decreasing shape parameter
    for p in (0.5, 1.0, 1.5):
        shape = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        # Avoid zeros
        shape = np.maximum(shape, 1e-8)
        samples = rng.gamma(shape, 1.0)
        v = samples / samples.sum()
        candidates.append(v)

    # Include the previous construction if available
    if existing is not None:
        v = existing.astype(np.float64)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        v = np.clip(v, 0.0, None)
        total = v.sum()
        if total == 0.0:
            v = np.full_like(v, 1.0 / n)
        else:
            v /= total
        candidates.append(v)

    return candidates

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size.
    # ----------------------------------------------------------------------
    global height_sequence_1
    if "height_sequence_1" in globals():
        try:
            n0 = len(height_sequence_1)
        except Exception:
            n0 = 2048
    else:
        n0 = 2048

    # we try a moderately large size, but not larger than 8192 (CPU‑friendly)
    n = int(min(max(2048, n0), 8192))

    # ----------------------------------------------------------------------
    # FFT preparation
    # ----------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)          # for self‑convolution
    fft_len_corr = _next_pow2(3 * n - 2)        # for correlation step

    # ----------------------------------------------------------------------
    # Stage 1 – quick portfolio scan.
    # ----------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + budget_s * 0.4   # 40 % of time for scanning
    best_a = None
    best_J = np.inf

    existing_vec = None
    if "height_sequence_1" in globals():
        try:
            existing_vec = np.asarray(height_sequence_1, dtype=np.float64)
        except Exception:
            existing_vec = None

    candidates = _generate_candidates(n, rng, existing_vec)

    for cand in candidates:
        if time.monotonic() >= deadline:
            break
        # conv via FFT
        conv = _conv_full_fft(cand, fft_len_conv, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    if best_a is None:
        # fallback – uniform
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_J = _objective(best_a, fft_len_conv)

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] Scan finished. Best J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # Stage 2 – smoothed‑max gradient descent.
    # ----------------------------------------------------------------------
    a = best_a.copy()            # probability vector (sum = 1)
    global_best_a = a.copy()
    global_best_J = best_J
    last_improve = time.monotonic()

    # hyper‑parameters
    alpha = 10.0          # start temperature for soft‑max
    alpha_max = 200.0
    alpha_inc = 1.005    # multiplicative increase per iteration
    lr0 = 0.2            # base learning rate (will decay)
    it = 0

    # main optimisation loop
    while True:
        now = time.monotonic()
        if now - start_time >= budget_s:
            break

        # --------------------------------------------------------------
        # 2.1 Self‑convolution (full) – also provides the soft‑max weights.
        # --------------------------------------------------------------
        A = np.fft.rfft(a, fft_len_corr)      # padded transform (used later as well)
        conv_full = np.fft.irfft(A * A, n=fft_len_corr)[:conv_len]
        max_conv = conv_full.max()
        cur_J = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 2.2 Update best solution
        # --------------------------------------------------------------
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # --------------------------------------------------------------
        # 2.3 Check for stagnation → jitter restart
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            sigma = 0.02 * np.mean(a)   # relative jitter
            a = global_best_a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")
            # reset counters
            it = 0
            alpha = 10.0

        # --------------------------------------------------------------
        # 2.4 Soft‑max weights over convolution entries
        # --------------------------------------------------------------
        # Stabilise exponentials by subtracting the maximum
        shifted = conv_full - max_conv
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical fallback – use hard max
            p = np.zeros_like(w)
            p[np.argmax(conv_full)] = 1.0
        else:
            p = w / w_sum

        # --------------------------------------------------------------
        # 2.5 Gradient of the smoothed objective.
        #   grad = 4·n·corr(a, p)   where corr uses the linear correlation
        #   grad_i = Σ_j a_j·p_{j+i}
        # --------------------------------------------------------------
        # Correlation via FFT: compute conv of a with reversed(p)
        p_rev = p[::-1]
        P_rev = np.fft.rfft(p_rev, fft_len_corr)
        corr_full = np.fft.irfft(A * P_rev, n=fft_len_corr)[: (n + conv_len - 1)]

        # indices where gradient_i resides: conv_len‑1‑i
        idx_start = conv_len - 1
        grad = 4.0 * n * corr_full[idx_start - np.arange(n)]

        # --------------------------------------------------------------
        # 2.6 Projected gradient step on the simplex.
        # --------------------------------------------------------------
        it += 1
        lr = lr0 / np.sqrt(it)
        a_new = a - lr * grad
        a = _simplex_projection(a_new)

        # --------------------------------------------------------------
        # 2.7 Update temperature (smoothness) – slowly anneal.
        # --------------------------------------------------------------
        alpha = min(alpha_max, alpha * alpha_inc)

    # ----------------------------------------------------------------------
    # Return the best vector discovered, safely clipped and normalised.
    # ----------------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

* A simple 1/√t learning‑rate schedule, an annealing temperature α and a
  jitter‑restart (Gaussian perturbation if no improvement for ~30 s) help
  escape plateaux.
* The routine respects the supplied time budget and finally returns the
  best sequence discovered, clipped to the allowed range.
"""

from __future__ import annotations
import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int, out_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:out_len]
    return conv


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0.0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _generate_candidates(
    n: int, rng: np.random.Generator, existing: np.ndarray | None
) -> List[np.ndarray]:
    """Yield a collection of diverse probability vectors of length n."""
    candidates: List[np.ndarray] = []

    # 1. Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. Linear ramps
    ramp_desc = np.arange(n, 0, -1, dtype=np.float64)
    ramp_desc /= ramp_desc.sum()
    candidates.append(ramp_desc)

    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    ramp_inc /= ramp_inc.sum()
    candidates.append(ramp_inc)

    # 3. Quadratic decreasing ramp
    quad_desc = (np.arange(n, 0, -1, dtype=np.float64) ** 2)
    quad_desc /= quad_desc.sum()
    candidates.append(quad_desc)

    # 4. Geometric families
    for r in (0.90, 0.95, 0.97, 0.98, 0.99, 0.995, 0.998):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # 5. Power‑law decreasing families a_i ∝ (i+1)^(-p)
    for p in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        v = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        v /= v.sum()
        candidates.append(v)

    # 6. Symmetric “U‑shape’’ a_i ∝ (i+1)^(-p) + (n‑i)^(-p)
    for p in (0.5, 1.0, 1.5, 2.0):
        left = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        right = left[::-1]
        v = left + right
        v /= v.sum()
        candidates.append(v)

    # 7. Dirichlet draws with constant concentration
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            candidates.append(rng.dirichlet(np.full(n, alpha)))

    # 8. Gamma draws with decreasing shape parameter
    for p in (0.5, 1.0, 1.5):
        shape = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        shape = np.maximum(shape, 1e-8)
        sample = rng.gamma(shape, 1.0)
        v = sample / sample.sum()
        candidates.append(v)

    # 9. The previously discovered construction (if any)
    if existing is not None:
        v = existing.astype(np.float64)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        v = np.clip(v, 0.0, None)
        tot = v.sum()
        if tot == 0.0:
            v = np.full_like(v, 1.0 / n)
        else:
            v /= tot
        candidates.append(v)

    return candidates


# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size (n)
    # --------------------------------------------------------------
    global height_sequence_1
    if "height_sequence_1" in globals():
        try:
            n0 = len(height_sequence_1)
        except Exception:
            n0 = 2048
    else:
        n0 = 2048
    n = int(min(max(2048, n0), 8192))          # keep n moderate for speed

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)       # for a*a
    fft_len_corr = _next_pow2(3 * n - 2)      # for correlation step

    # --------------------------------------------------------------
    # 2. Quick portfolio scan (≈40 % of the time budget)
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline_scan = start + 0.40 * budget_s
    existing_vec = None
    if "height_sequence_1" in globals():
        try:
            existing_vec = np.asarray(height_sequence_1, dtype=np.float64)
        except Exception:
            existing_vec = None

    candidates = _generate_candidates(n, rng, existing_vec)

    best_a = None
    best_J = np.inf

    for cand in candidates:
        if time.monotonic() >= deadline_scan:
            break
        conv = _conv_full_fft(cand, fft_len_conv, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()

    if best_a is None:                              # safety fallback
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_J = 2.0 * n * _conv_full_fft(best_a, fft_len_conv, conv_len).max()

    elapsed = time.monotonic() - start
    print(f"[{elapsed:6.2f}s] Scan finished – best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Smoothed‑max gradient descent from the best candidate
    # --------------------------------------------------------------
    a = best_a.copy()                               # probability vector (∑a = 1)
    global_best_a = a.copy()
    global_best_J = best_J
    last_improve = time.monotonic()

    # hyper‑parameters
    alpha = 10.0                                     # start temperature
    alpha_max = 200.0
    alpha_mult = 1.005                               # geometric increase
    lr0 = 0.05                                      # base learning rate
    it = 0

    while True:
        now = time.monotonic()
        if now - start >= budget_s:
            break

        # ----- 3.1 Self‑convolution (full) -----
        A = np.fft.rfft(a, fft_len_corr)            # padded FFT of a
        conv_full = np.fft.irfft(A * A, n=fft_len_corr)[:conv_len]
        max_conv = conv_full.max()
        cur_J = 2.0 * n * max_conv

        # ----- 3.2 Update global best -----
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----- 3.3 Stagnation detection → jitter restart -----
        if now - last_improve > 30.0:
            sigma = 0.02 * np.mean(a)               # modest relative jitter
            a = global_best_a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a)
            last_improve = now
            it = 0                                   # reset iteration counter
            alpha = 10.0
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

        # ----- 3.4 Soft‑max weights over convolution entries -----
        shifted = conv_full - max_conv               # numerical stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:                             # fallback – pure max
            p = np.zeros_like(w)
            p[np.argmax(conv_full)] = 1.0
        else:
            p = w / w_sum

        # ----- 3.5 Gradient of the smoothed objective -----
        # correlation of a with p  (grad_i = Σ_j a_j p_{j+i})
        p_rev = p[::-1]
        P_rev = np.fft.rfft(p_rev, fft_len_corr)
        corr_full = np.fft.irfft(A * P_rev, n=fft_len_corr)[: (n + conv_len - 1)]
        idx_start = conv_len - 1                      # position of lag 0
        grad = 4.0 * n * corr_full[idx_start - np.arange(n)]

        # ----- 3.6 Normalised projected gradient step -----
        it += 1
        lr = lr0 / np.sqrt(it)
        grad_norm = np.linalg.norm(grad) + 1e-12
        direction = grad / grad_norm
        a_new = a - lr * direction
        a = _simplex_projection(a_new)

        # ----- 3.7 Anneal temperature -----
        alpha = min(alpha_max, alpha * alpha_mult)

    # --------------------------------------------------------------
    # 4. Return the best sequence (clipped, normalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
