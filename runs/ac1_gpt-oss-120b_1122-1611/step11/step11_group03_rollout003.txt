"""
Hybrid Frank‑Wolfe + line‑search optimizer for the step‑function problem.

We minimise  J(a) = 2·n·max(conv(a,a))  with  a_i ≥ 0 , Σ a_i = 1 .
For a fixed a we solve a linear programme

    maximise   Σ b_i
    subject to conv(a,b) ≤ M   (pointwise) ,  b ≥ 0

where M = max(conv(a,a)).  Because conv(a,b) is linear in b,
the LP yields a direction b that can carry more total mass while
still respecting the current worst‑case bound M.
We normalise b to a probability vector and then perform a small
convex combination a←(1‑t)·a + t·b.  The mixing factor t is chosen
by a cheap line‑search over a fixed set of candidates.
Whenever progress stalls we inject a multiplicative jitter (log‑normal
noise) and project back onto the probability simplex.

The algorithm runs until the time budget (default 1000 s) is exhausted,
keeping the best vector ever seen.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used for jittering only
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a 1‑D array onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (linear, not circular)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Return the full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Compute the evaluation objective from a pre‑computed convolution."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a few seconds margin for finalisation
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # 1. Choose problem size
    # --------------------------------------------------------------
    # A moderate size keeps the LP memory‑wise feasible.
    DEFAULT_N = 2048
    # If a previous construction exists, keep its length if it is modest.
    if "height_sequence_1" in globals():
        try:
            prev_len = len(height_sequence_1)
            n = min(max(prev_len, 512), 3000)   # cap to avoid huge matrices
        except Exception:
            n = DEFAULT_N
    else:
        n = DEFAULT_N

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)       # length for a*a FFT
    # --------------------------------------------------------------
    # 2. Initialise a probability vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        if tmp.size >= n:
            a = tmp[:n].copy()
        else:
            a = np.zeros(n, dtype=np.float64)
            a[:tmp.size] = tmp
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    s = a.sum()
    if s <= 0.0:
        a[:] = 1.0 / n
    else:
        a /= s

    # --------------------------------------------------------------
    # 3. Evaluate initial objective
    # --------------------------------------------------------------
    conv_a = full_convolution(a, fft_len, conv_len)
    best_J = objective_from_conv(conv_a, n)
    best_vec = a.copy()
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop
    # --------------------------------------------------------------
    # Fixed set of mixing coefficients for line‑search
    t_candidates = [0.0, 0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0]

    last_improve = start_time
    iteration = 0
    eps_nonzero = 1e-12

    # pre‑allocate column indices for the sparse matrix (common for all rows)
    col_base = np.arange(n, dtype=np.int32)

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- current bound -------------------------------------------------
        conv_a = full_convolution(a, fft_len, conv_len)
        M = conv_a.max()

        # ---- build the LP constraint matrix (sparse) -----------------------
        idx = np.where(a > eps_nonzero)[0]
        nnz = idx.size * n
        if nnz == 0:
            # all mass vanished – restart from uniform
            a = np.full(n, 1.0 / n, dtype=np.float64)
            continue

        data = np.empty(nnz, dtype=np.float64)
        rows = np.empty(nnz, dtype=np.int32)
        cols = np.empty(nnz, dtype=np.int32)

        pos = 0
        for i in idx:
            data[pos:pos + n] = a[i]
            rows[pos:pos + n] = i + col_base
            cols[pos:pos + n] = col_base
            pos += n

        A_ub = csr_matrix((data, (rows, cols)), shape=(conv_len, n))

        # ---- solve LP: maximise sum(b)  ⇔ minimise -sum(b) -----------------
        c = -np.ones(n, dtype=np.float64)           # minimise -Σb_i
        b_ub = np.full(conv_len, M, dtype=np.float64)

        try:
            res = linprog(c,
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=(0, None),
                          method='highs',
                          options={'presolve': True})
        except Exception:
            # In the unlikely event the solver crashes, abort the LP step.
            break

        if not res.success:
            # fall back to a simple jitter and continue
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            # Nothing useful found – jitter and continue
            break

        b_norm = b / sum_b                     # bring back to Σ = 1

        # ---- line‑search over convex combinations -------------------------
        best_local_J = np.inf
        best_local_a = None

        for t in t_candidates:
            a_candidate = (1.0 - t) * a + t * b_norm
            # numerical safety – enforce simplex (tiny drift may appear)
            a_candidate = proj_simplex(a_candidate)

            conv_cand = full_convolution(a_candidate, fft_len, conv_len)
            J_cand = objective_from_conv(conv_cand, n)

            if J_cand < best_local_J:
                best_local_J = J_cand
                best_local_a = a_candidate

        # ---- accept if we improved globally -------------------------------
        if best_local_J + 1e-12 < best_J:
            best_J = best_local_J
            best_vec = best_local_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # Move the current point to the best candidate from line‑search
        a = best_local_a

        # ---- stagnation handling -------------------------------------------
        if now - last_improve > 30.0:
            # inject a small multiplicative (log‑normal) jitter
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = best_vec * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        iteration += 1

        # quick exit if the remaining time is insufficient for another LP
        if time.monotonic() + 0.5 > deadline:
            break

    # --------------------------------------------------------------
    # 5. Return the best vector (clipped & re‑normalised for safety)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total <= 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
