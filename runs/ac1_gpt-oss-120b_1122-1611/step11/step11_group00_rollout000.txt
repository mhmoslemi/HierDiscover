"""
Iterative LP‐based improvement for the max‑auto‑convolution problem.

The algorithm works on the probability simplex (∑a = 1) and repeatedly
solves a linear program

    max ∑ g_j subject to g_j ≥ 0, (f * g)_k ≤ M  (∀k),

where f is the current distribution, * denotes the (zero‑padded) linear
convolution and M = max_k (f * f)_k.  The LP maximises the ℓ₁‑norm of a
candidate g while keeping the convolution with the current f bounded.
After solving, g is normalised to ∑g = 1 and replaces f if this reduces
M; otherwise a simple convex blend is performed.  Whenever no improvement
occurs for a while we apply a small soft‑max jitter restart to escape
plateaux.

The routine respects the 1000 s time budget, prints progress, and returns
the best sequence encountered (already normalised and clipped to the
allowed range).  It can start from the previously found construction
`height_sequence_1` if available, otherwise from a set of diverse heuristics.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_max(a: np.ndarray) -> float:
    """
    Compute max_k (a * a)[k] for the zero‑padded full convolution of a with itself.
    Uses a real FFT for O(n log n) time.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2*n‑1, n)) such that
        A @ g = f * g   (full linear convolution).
    Column j contains the vector f placed starting at row j.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill column‑wise: column j is f shifted by +j rows.
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a moderate problem size.
    # --------------------------------------------------------------
    # Prefer the length of a previously discovered construction,
    # but keep it in a tractable range.
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 2048))  # hard limits for speed / memory
        except Exception:
            n = 2048
    else:
        n = 2048

    # --------------------------------------------------------------
    # 2. Assemble a small portfolio of initial probability vectors.
    # --------------------------------------------------------------
    init_cands = []

    # Uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderately concentrated)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:

            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate according to the true objective.
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        J = float(evaluate_sequence(cand.tolist()))
        if J < best_J:
            best_J = J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f} (n={n})")

    # --------------------------------------------------------------
    # 4. Main iterative improvement loop (LP + occasional jitter).
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # Keep a small safety margin for final processing.
    deadline = start_time + max(0.0, float(budget_s) - 8.0)

    # Ensure the current iterate lives on the simplex.
    f = best_a.copy()
    f = np.maximum(f, 0.0)
    if f.sum() == 0.0:

    else:

    global_best_f = f.copy()
    global_best_J = best_J
    last_improve = start_time
    iteration = 0

    def jitter_restart(vec: np.ndarray, sigma: float = 0.07) -> np.ndarray:
        """Apply a soft‑max jitter to the probability vector."""
        eps = 1e-12
        logits = np.log(np.maximum(vec, eps))
        logits += rng.normal(0.0, sigma, size=logits.shape)
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        jittered = exp_logits / exp_logits.sum()
        return jittered

    while time.monotonic() < deadline:

        M = conv_max(f)          # this is max_b for the current f
        # ---- 4.2 Build the convolution matrix A (f * g) ----
        A = build_conv_matrix(f)

        # ---- 4.3 Solve the LP: maximise sum(g) subject to A g ≤ M, g ≥ 0 ----
        c = -np.ones(n, dtype=np.float64)          # minimise -sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # The 'highs' method handles dense problems efficiently.
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.monotonic() - start_time:6.2f}s] LP failed at iter {iteration}: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print(f"[{time.monotonic() - start_time:6.2f}s] LP returned zero mass at iter {iteration}.")
            break

        # ---- 4.4 Normalise the LP solution to a probability vector ----
        g = g0 / sum_g0

        # ---- 4.5 Evaluate whether the candidate improves the bound ----
        M_g = conv_max(g)

        # Accept the new candidate if it strictly reduces the max‑convolution.
        if M_g < M * 0.997:
            f_new = g
            accepted = True
        else:
            # Otherwise perform a modest convex blend.
            blend = 0.25
            f_new = (1.0 - blend) * f + blend * g
            accepted = False

        # Re‑normalise to stay on the simplex.
        f_new = np.maximum(f_new, 0.0)
        tot = f_new.sum()
        if tot == 0.0:
            f_new = np.full(n, 1.0 / n)
        else:
            f_new /= tot

        # ---- 4.6 Compute the true evaluation metric ----
        J_new = float(evaluate_sequence(f_new.tolist()))
        now = time.monotonic()
        elapsed = now - start_time

        if J_new < global_best_J - 1e-12:
            global_best_J = J_new
            global_best_f = f_new.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f} (iter {iteration}, {'accept' if accepted else 'blend'})")

        # ---- 4.7 Prepare next iterate ----
        f = f_new

        # ---- 4.8 Stagnation detection – jitter restart if needed ----
        if now - last_improve > 30.0:
            # Apply a soft‑max jitter to shake out of a plateau.
            f = jitter_restart(global_best_f, sigma=0.09)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart (no improvement for >30 s)")

    # --------------------------------------------------------------
    # 5. Return the best sequence (clipped and normalised for safety).
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_f, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:

    else:
        best_vec /= total

    final_J = float(evaluate_sequence(best_vec.tolist()))
    total_time = time.monotonic() - start_time
    print(f"[{total_time:6.2f}s] Finished – best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
