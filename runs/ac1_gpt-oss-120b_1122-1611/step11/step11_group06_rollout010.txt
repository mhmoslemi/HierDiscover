"""
Gradient‑annealed soft‑max optimizer for the non‑negative step‑function problem.

The objective J = 2·n·max(conv(a,a)) is scale‑invariant, so we work on the
probability simplex (∑a_i = 1).  We maintain logits ℓ with a = softmax(ℓ);
this guarantees non‑negativity and the simplex constraint.

At each iteration we
  1. compute the full linear convolution a∗a by FFT,
  2. build a smooth approximation of the max using a soft‑max weight vector
     w_k ∝ exp(β·(conv_k‑max_conv)),
  3. obtain the gradient of the smoothed objective:
        ∇_a ≈ 2·corr(a,w)   (cross‑correlation of a with w)
     which is also computed by a single FFT,
  4. perform an Adam update on the logits,
  5. slowly raise β (starting from 0.5 up to 200) so the optimisation first
     smooths the landscape and later focuses on the true maximum,
  6. inject occasional random jitter when progress stalls.

The routine runs until the 1000 s budget (minus a short safety margin) and
returns the best vector found, clipped to the limits required by
`evaluate_sequence`.  Simple random permutations are tried at the very end
as a cheap post‑processing step.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_self_fft(vec: np.ndarray, fft_len: int, conv_len: int):
    """
    Full linear convolution a*a using FFT.
    Returns (conv[:conv_len], rfft(a)) where conv is the linear convolution.
    """
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv, A


def cross_corr_grad_fft(A: np.ndarray, rev_w: np.ndarray,
                       fft_len: int, center: int, n: int) -> np.ndarray:
    """
    Compute gradient = 2 * (a cross‑correlated with w).
    A is rfft(a); rev_w is the reversed weight vector w of length 2n‑1.
    The linear convolution a * rev_w is performed via FFT; the slice
    [center:center+n] yields the required values.
    """
    W_rev = np.fft.rfft(rev_w, n=fft_len)
    corr = np.fft.irfft(A * W_rev, n=fft_len)
    grad = 2.0 * corr[center:center + n]
    return grad

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:

    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence discovered within the
    allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (length of the vector).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1                      # length of a*a
    fft_len = _next_pow2(3 * n)               # enough for correlation (3n‑2)

    # --------------------------------------------------------------
    # 2. Initialise the probability vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        init = np.clip(init, 0.0, None)
        if init.sum() == 0.0:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = init / init.sum()
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # Logits for a soft‑max parametrisation (guarantees positivity & sum‑1)
    eps = 1e-30
    logits = np.log(a0 + eps)

    # --------------------------------------------------------------
    # 3. Adam state initialisation.
    # --------------------------------------------------------------
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # --------------------------------------------------------------
    # 4. Book‑keeping.
    # --------------------------------------------------------------
    # evaluate the current (initial) J
    conv0, _ = conv_self_fft(a0, fft_len, conv_len)
    best_J = 2.0 * n * conv0.max()
    best_vec = a0.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    it = 0
    last_improve = start_time

    # Soft‑max temperature schedule (β from small to large)
    beta_start = 0.5
    beta_end = 200.0

    # --------------------------------------------------------------
    # 5. Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- reconstruct a from logits (soft‑max) ----------
        max_logit = np.max(logits)
        exp_log = np.exp(logits - max_logit)
        a = exp_log / exp_log.sum()

        # ----- convolution a*a (FFT) -------------------------
        conv_self, A = conv_self_fft(a, fft_len, conv_len)
        max_self = conv_self.max()
        cur_J = 2.0 * n * max_self

        # ----- update best solution -------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- β (soft‑max sharpness) ----------
        beta_cur = beta_start + (beta_end - beta_start) * (elapsed / (deadline - start_time))

        # ----- smoothed max weights -------------------------
        diff = conv_self - max_self                     # non‑positive
        w = np.exp(beta_cur * diff)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / conv_len)
            w_sum = 1.0
        w /= w_sum

        # ----- gradient of the smoothed objective ------------
        rev_w = w[::-1]                                # reverse for correlation
        center = conv_len - 1
        grad = cross_corr_grad_fft(A, rev_w, fft_len, center, n)

        # ----- Adam update on logits -------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + adam_eps)
        logits -= step

        # ----- occasional jitter to escape plateaus ----------
        if now - last_improve > 30.0:                  # 30 s of stagnation
            jitter = rng.normal(scale=0.05, size=n)
            logits += jitter
            last_improve = now

        # ----- tiny random perturbations (exploration) ------
        if rng.random() < 0.01:
            logits += rng.normal(scale=0.01, size=n)

    # --------------------------------------------------------------
    # 6. Post‑processing: try a few random permutations of the best vector.
    # --------------------------------------------------------------
    extra_deadline = start_time + float(budget_s) - 0.5   # tiny leftover margin
    while time.time() < extra_deadline:
        perm = rng.permutation(best_vec)
        conv_perm = np.convolve(perm, perm)  # direct convolution; cheap for a handful
        J_perm = 2.0 * n * conv_perm.max()
        if J_perm < best_J:
            best_J = J_perm
            best_vec = perm.copy()
            print(f"[{time.time() - start_time:6.2f}s] Improved by permutation J = {best_J:.6f}")
        else:
            break

    # --------------------------------------------------------------
    # 7. Final normalisation & clipping (matches evaluate_sequence's guards).
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
