def propose_candidate(seed=4242, budget_s=1000.0):

    rng = np.random.default_rng(seed)

    # Determine n
    n = 2048
    if 'height_sequence_1' in globals():
        try:
            n_cand = int(len(height_sequence_1))
            if 256 <= n_cand <= 8192:
                n = n_cand
        except:
            pass

    # compute lengths
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(conv_len + n - 1)

    # Prepare helper functions
    def compute_J_vec(v):
        conv = _conv_full_fft(v, v, fft_len)[:conv_len]
        return float(2.0 * n * conv.max())

    # Build initial candidates list
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0/n))

    # random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0]:
        for _ in range(5):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # if prior solution available
    if 'height_sequence_1' in globals():
        v = np.array(height_sequence_1, dtype=np.float64)
        # clip negative
        v = np.clip(v, 0.0, None)
        # renorm
        s = v.sum()
        if s == 0:
            v = np.full(n, 1.0/n)
        else:
            v = v / s
        if len(v) != n:
            v = v[:n] if len(v) > n else np.pad(v, (0, n - len(v)), constant_values=0)
        init_cands.append(v)

    # Choose best initial
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J_vec(cand)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Init best J = {best_J:.6f}")

    # Optimization parameters
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # Adam parameters
    lr0 = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(best_vec)
    vvar = np.zeros_like(best_vec)
    t = 0

    # Smoothed max temperature schedule
    beta_smooth = 0.5  # initial
    beta_max = 200.0   # final smoothing factor

    # Stagnation handling
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # update smoothing factor exponentially
        beta_smooth = beta_smooth * (beta_max / beta_smooth) ** (elapsed / (deadline - start_time))
        # compute convolution a*a
        conv = _conv_full_fft(best_vec, best_vec, fft_len)[:conv_len]
        max_c = conv.max()
        # compute softmax weights
        # avoid overflow: subtract max_c
        exp_arg = beta_smooth * (conv - max_c)
        # clip to avoid large exponentials?
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / conv_len)
        else:
            w = w / w_sum

        # gradient of smoothed max
        a_rev = best_vec[::-1]
        # conv of w and a_rev
        conv_w = _conv_full_fft(w, a_rev, fft_len_grad)[:conv_len + n - 1]  # ensure length
        grad = 2.0 * conv_w[conv_len - 1 : conv_len - 1 + n]  # slice length n

        # Adam update
        t += 1
        lr = lr0 / np.sqrt(t)
        m = beta1 * m + (1 - beta1) * grad
        vvar = beta2 * vvar + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = vvar / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = best_vec - step
        # projection onto simplex
        cand = proj_simplex(cand)

        cand_J = compute_J_vec(cand)
        # accept if improved else occasional acceptance
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
        else:
            # Metropolis with temperature based on current best another schedule
            # Use temperature as 0.01 * best_J maybe decays
            T = max(best_J * 0.01 * (1 - elapsed/(deadline-start_time)), 1e-8)
            if rng.random() < np.exp(-(cand_J - best_J) / T):
                best_vec = cand
                # don't update best_J

        # If no improvement for 30 sec, jitter
        if now - last_improve > 30.0:
            # apply random noise multiplicatively
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            jittered = best_vec * noise
            jittered = proj_simplex(jittered)
            jitter_J = compute_J_vec(jittered)
            if jitter_J < best_J:
                best_J = jitter_J
                best_vec = jittered
                last_improve = now
                print(f"[{elapsed:6.2f}s] Jitter improved J = {best_J:.6f}")
            else:
                # still accept jitter with low probability
                if rng.random() < 0.1:
                    best_vec = jittered

    # final sanitize
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0:
        best_vec = np.full(n, 1.0/n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

"""
Propose a better step‑function using a smoothed‑max (log‑sum‑exp) gradient
and Adam‑style updates on the probability simplex.

Key ideas
---------
* The evaluation metric  J(a)=2·n·max(conv(a,a))/sum(a)² is scale‑invariant.
  By projecting onto the simplex (∑a=1) we can drop the denominator and
  minimise  J(a)=2·n·max(conv(a,a)).
* The non‑differentiable max is replaced by
      fβ(a)= (1/β)·log∑ₖ exp(β·cₖ)      (c = a*a)
  whose gradient is a smooth convex combination of the individual
  sub‑gradients.  β is increased slowly so the optimisation first moves
  quickly (small β) and later focuses on the true max (large β).
* Gradients are obtained efficiently by a single FFT‑based convolution:
        ∇fβ = 2·conv(w , a_rev)[offset : offset+n],
  where wₖ∝exp(β·cₖ) are the soft‑max weights and a_rev is a reversed copy
  of a.  All convolutions use real‑FFT and are O(n log n).
* Adam (first‑ and second‑moment adaptive step size) is used on the
  simplex‑projected variables.  A Metropolis acceptance rule lets the
  walk escape shallow plateaus.
* Stagnation (no improvement for 30 s) triggers a multiplicative jitter
  centred on the current best solution.
* Multiple restarts are performed if the walk stays idle for a longer
  period (≈90 s).  The best vector seen so far is always returned.

The routine respects a 1000 s budget (5 s safety margin) and returns a
list of non‑negative heights ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        # all entries are equal (or v is zero)
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of real vectors x and y using a real FFT.
    The result is truncated to length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    out_len = x.size + y.size - 1
    return conv[:out_len]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Exact objective  J = 2·n·max(conv(a,a))  for a vector already on the simplex.
    """
    conv = _conv_full_fft(a, a, fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Uses a smoothed‑max gradient with Adam updates
    on the simplex.  Returns the best sequence found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len_a = _next_pow2(conv_len)                     # for a*a
    fft_len_grad = _next_pow2(conv_len + n - 1)          # for w * a_rev

    # --------------------------------------------------------------
    # Initial candidate pool (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few Dirichlet draws – low, medium and high concentration
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(4):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # geometric decays (used in the reference code)
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # previous best, if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= s
        if len(v) != n:
            v = v[:n] if len(v) > n else np.pad(v, (0, n - len(v)))
        init_cands.append(v)

    # pick the best initial point (cheap J evaluation)
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand, n, fft_len_a, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    cur_vec = best_vec.copy()
    cur_J = best_J

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # Adam hyper‑parameters
    lr0 = 0.12
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(cur_vec)
    vvar = np.zeros_like(cur_vec)
    t = 0

    # smoothing factor for the log‑sum‑exp approximation
    beta_smooth = 0.6          # start small (very smooth)
    beta_target = 250.0        # final “hard” max

    # stagnation handling
    last_improve = start_time
    last_restart = start_time

    # main optimisation loop
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total_time = deadline - start_time

        # ----- anneal the smoothing parameter (exponential schedule) -----
        beta_smooth = beta_smooth * (beta_target / beta_smooth) ** (elapsed / total_time)

        # ----- compute current convolution and the soft‑max weights -----
        conv = _conv_full_fft(cur_vec, cur_vec, fft_len_a)[:conv_len]
        max_c = conv.max()
        # stable soft‑max
        w_raw = np.exp(beta_smooth * (conv - max_c))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv_len)
        else:
            w = w_raw / w_sum

        # ----- gradient of the smoothed max -----
        a_rev = cur_vec[::-1]
        conv_w_full = _conv_full_fft(w, a_rev, fft_len_grad)
        # slice that corresponds to indices (len(w)-1) … (len(w)-1 + n - 1)
        start = conv_len - 1            # len(w) - 1
        grad = 2.0 * conv_w_full[start:start + n]

        # ----- Adam update (projected onto the simplex) -----
        t += 1
        lr = lr0 / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * grad
        vvar = beta2 * vvar + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = vvar / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = cur_vec - step
        cand = proj_simplex(cand)

        cand_J = compute_J(cand, n, fft_len_a, conv_len)

        # ----- Metropolis acceptance (temperature based on current best) -----
        accept = False
        if cand_J < cur_J:
            accept = True
        else:
            # simple temperature that decays to a tiny value
            T = max(best_J * 0.02 * (1.0 - elapsed / total_time), 1e-8)
            if rng.random() < np.exp(-(cand_J - cur_J) / T):
                accept = True

        if accept:
            cur_vec = cand
            cur_J = cand_J

            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation: jitter or full restart -----
        if now - last_improve > 30.0:
            # multiplicative jitter centred on the best solution seen so far
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            jittered = best_vec * jitter
            jittered = proj_simplex(jittered)
            jitter_J = compute_J(jittered, n, fft_len_a, conv_len)
            if jitter_J < best_J:
                best_J = jitter_J
                best_vec = jittered.copy()
                cur_vec = jittered.copy()
                cur_J = jitter_J
                last_improve = now
                print(f"[{elapsed:6.2f}s] Jitter improved J = {best_J:.6f}")
            else:
                # occasional acceptance of pure jitter to escape plateaus
                if rng.random() < 0.05:
                    cur_vec = jittered
                    cur_J = jitter_J

        # after a longer idle period we completely restart from a fresh Dirichlet
        if now - last_restart > 90.0 and now - last_improve > 60.0:
            cur_vec = rng.dirichlet(np.full(n, 0.6))
            cur_J = compute_J(cur_vec, n, fft_len_a, conv_len)
            m.fill(0.0)
            vvar.fill(0.0)
            t = 0
            last_restart = now
            print(f"[{elapsed:6.2f}s] Restarted from a random point (J≈{cur_J:.6f})")

    # --------------------------------------------------------------
    # Final sanitisation (clip to allowed interval & re‑normalise)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
```

The schedule for the smoothing parameter, the Metropolis acceptance rule and the jitter/re‑start heuristics together give the search a good chance of escaping local plateaus and improving the current best bound (target < 1.5025) within the allotted 1000 s.
