"""
Simulated‑annealing on the probability simplex.

* The objective J(a) = 2·n·max(conv(a,a)) is scale‑invariant, therefore we
  keep the vector on the simplex (∑a_i = 1, a_i ≥ 0).  A simple
  simulated‑annealing (SA) loop proposes to move a small amount of mass from
  one index to another; the move is accepted if it reduces J or with a
  probability exp(-(ΔJ)/T) where T cools linearly with the remaining time.
* For each SA run we start from a diverse random seed (uniform, geometric,
  Gaussian, Dirichlet, sparse spikes, or the previously discovered best
  sequence) and keep the best ever seen.
* The convolution is evaluated by a real FFT (O(n log n)) and the maximum
  element is extracted – this is the exact value used by
  `evaluate_sequence`.
* Several problem sizes (powers of two up to ~8000) are tried; the best
  sequence over all sizes is returned.
* The whole routine respects the 1000 s budget and prints occasional progress.
"""

import time
import numpy as np
from scipy.optimize import linprog  # not used but allowed; keep import for sanity

######################################################################
# Helper utilities
######################################################################

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Exact objective 2·n·max(conv(vec,vec)).
    `vec` is assumed to sum to 1 (scale‑invariant), length n = len(vec).
    """
    n = vec.size
    # real FFT of vec padded to fft_len
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


######################################################################
# Core local optimiser – simulated annealing
######################################################################

def _sa_optimize(start_vec: np.ndarray,
                 rng: np.random.Generator,
                 deadline: float,
                 global_start: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing for a single problem size.
    Returns the best vector found and its J‑value.
    """
    n = start_vec.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    cur = start_vec.copy()
    cur_J = compute_J(cur, fft_len, conv_len)

    best = cur.copy()
    best_J = cur_J

    # temperature schedule (linear cooling)
    T_start = cur_J * 0.1   # 10 % of the initial objective
    T_end = 1e-8

    last_improve = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - global_start) / (deadline - global_start)
        # linear interpolation between T_start and T_end
        T = (1.0 - frac) * T_start + frac * T_end

        # ----- propose a move -------------------------------------------------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        while j == i:
            j = rng.integers(0, n)

        # move a fraction (≤20 %) of the mass at i to j
        if cur[i] <= 0.0:
            continue
        step_factor = max(0.01, 0.5 / np.sqrt(it + 1.0))
        delta = cur[i] * step_factor * rng.random()
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # ----- evaluate --------------------------------------------------------
        new_J = compute_J(new, fft_len, conv_len)

        # ----- Metropolis acceptance -------------------------------------------
        if new_J < cur_J or rng.random() < np.exp(-(new_J - cur_J) / max(T, 1e-12)):
            cur = new
            cur_J = new_J
            if new_J < best_J:
                best = new.copy()
                best_J = new_J
                last_improve = now
                print(f"[{now - global_start:.2f}s] SA new local best J={best_J:.6f} (n={n})")

        # ----- stagnation handling ---------------------------------------------
        if now - last_improve > 5.0:
            # jitter around the current best to escape plateaus
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur = best * noise
            cur = proj_simplex(cur)
            cur_J = compute_J(cur, fft_len, conv_len)
            last_improve = now
        it += 1

    return best, best_J

# Top‑level search driver

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  A simulated‑annealing (SA) optimiser runs many short
    SA tours on several sequence lengths; the best sequence encountered within
    the time budget is returned.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.01)   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a collection of problem sizes.
    # ------------------------------------------------------------------
    possible_n = [256, 512, 1024, 2048, 4096, 8192]
    if "height_sequence_1" in globals():
        try:
            n0 = len(height_sequence_1)
            if 256 <= n0 <= 8192:
                possible_n.append(n0)
        except Exception:
            pass
    possible_n = sorted(set(possible_n))

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    global_best_vec = None
    global_best_J = np.inf

    # ------------------------------------------------------------------
    # Main search loop (spends the whole budget)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length at random (biased towards larger sizes)
        n = rng.choice(possible_n)
        # --------------------------------------------------------------
        # Initialise a candidate vector with a random 'palette' choice
        # --------------------------------------------------------------
        init_type = rng.integers(0, 6)
        if init_type == 0:                     # uniform
            vec = np.full(n, 1.0 / n, dtype=np.float64)

        elif init_type == 1:                   # geometric decay
            r = rng.uniform(0.80, 0.999)
            v = r ** np.arange(n, dtype=np.float64)
            vec = v / v.sum()

        elif init_type == 2:                   # Gaussian bell
            sigma = rng.uniform(n / 12.0, n / 4.0)
            x = np.arange(n, dtype=np.float64)
            centre = (n - 1) / 2.0
            v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
            vec = v / v.sum()

        elif init_type == 3:                   # Dirichlet draw
            alpha = rng.uniform(0.05, 0.8)
            vec = rng.dirichlet(np.full(n, alpha))

        elif init_type == 4:                   # sparse random spikes
            k = rng.integers(2, min(12, n))
            idx = rng.choice(n, size=k, replace=False)
            vals = rng.random(k)
            vec = np.zeros(n, dtype=np.float64)
            vec[idx] = vals
            vec /= vec.sum()

        else:                                  # use previous best if dimensions match
            if "height_sequence_1" in globals():
                old = np.asarray(height_sequence_1, dtype=np.float64)
                if old.size == n:
                    vec = np.clip(old, 0.0, None)
                    s = vec.sum()
                    if s == 0.0:
                        vec[:] = 1.0 / n
                    else:
                        vec /= s
                else:
                    # fall back to uniform if size mismatch
                    vec = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                vec = np.full(n, 1.0 / n, dtype=np.float64)

        # --------------------------------------------------------------
        # Run a short SA optimisation on this candidate.
        # --------------------------------------------------------------
        # allocate a modest fraction of the remaining time for this SA run
        now = time.monotonic()
        remaining = deadline - now
        # guarantee at least a few seconds per SA run; otherwise break
        if remaining < 2.0:
            break
        sa_deadline = now + min(5.0, remaining)   # each SA run ≤ 5 s

        candidate_vec, candidate_J = _sa_optimize(vec, rng, sa_deadline, start_time)

        # --------------------------------------------------------------
        # Update the global best
        # --------------------------------------------------------------
        if candidate_J < global_best_J:
            global_best_J = candidate_J
            global_best_vec = candidate_vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] Global best J={global_best_J:.6f} (n={n})")

        # quick exit if we already beat the target (safety)
        if global_best_J < 1.5025:
            break

    # ------------------------------------------------------------------
    # Safety post‑processing: clip, renormalise, convert to list.
    # ------------------------------------------------------------------
    if global_best_vec is None:
        # As a fallback return a uniform vector of length 256
        global_best_vec = np.full(256, 1.0 / 256, dtype=np.float64)
        global_best_J = compute_J(global_best_vec,
                                  _next_pow2(2 * 256 - 1),
                                  2 * 256 - 1)

    global_best_vec = np.clip(global_best_vec, 0.0, 1000.0)
    total = global_best_vec.sum()
    if total == 0.0:
        global_best_vec[:] = 1.0 / global_best_vec.size
    else:
        global_best_vec /= total

    return [float(v) for v in global_best_vec]
