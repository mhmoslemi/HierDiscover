"""Propose a better step‑function using a Frank‑Wolfe style LP improvement.

The algorithm works on a normalized vector `a` (∑a = 1, a_i ≥ 0).  
For the current vector `f` we compute  

    M = max_t (f * f)[t]      (full linear convolution)

and then solve the linear program

    maximize   𝟙ᵀ g
    subject to conv(f,g) ≤ M   (element‑wise, 2·n‑1 constraints)
               g ≥ 0

The LP tells us how much total mass can be placed in a new direction `g`
while keeping the convolution with `f` below the current worst value `M`.
After normalising `g` to a probability vector we move a short step towards it.
If the objective (the evaluation function) improves we keep the move,
otherwise we apply a tiny random jitter and continue.

The LP is solved with SciPy’s HiGHS interior‑point method, which is very
fast for the moderate problem sizes we use (n = 256…1024).  The whole
procedure runs until the time budget (≈ 1000 s) is exhausted or the target
value 1.5025 is reached.  The best vector found is returned as a plain
list of Python floats, ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2n‑1) × n matrix A such that A·g = conv(f,g) (full)."""
    n = f.shape[0]
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n), dtype=np.float64)
    # column j contributes f to rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A


def evaluate(seq):
    """Thin wrapper around the provided evaluate_sequence (global)."""
    # The environment already supplies evaluate_sequence; we just call it.
    return evaluate_sequence(seq)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes `evaluate_sequence`.
    Returns the best sequence found within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a modest problem size (fast LP solves)
    # --------------------------------------------------------------
    # If a previous construction exists, we try to keep roughly its scale,
    # otherwise we default to n = 1024.
    target_n = 1024
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        # down‑sample / interpolate to the desired length
        if old.size != target_n:
            x_old = np.arange(old.size, dtype=np.float64)
            x_new = np.linspace(0, old.size - 1, target_n, dtype=np.float64)
            old = np.interp(x_new, x_old, old)
        init_from_old = np.clip(old, 0.0, None)
        if init_from_old.sum() == 0.0:
            init_from_old = np.full(target_n, 1.0 / target_n)
        else:
            init_from_old /= init_from_old.sum()
    else:
        init_from_old = None

    n = target_n
    conv_len = 2 * n - 1

    # --------------------------------------------------------------
    # 2. Build a diverse set of starting points (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws – various concentrations
    for alpha in [0.1, 0.3, 0.7, 1.5, 3.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # exponential (positive) samples
    for _ in range(3):
        v = rng.exponential(scale=1.0, size=n)
        v /= v.sum()
        init_cands.append(v)

    # previous construction (if present)
    if init_from_old is not None:
        init_cands.append(init_from_old)

    # --------------------------------------------------------------
    # 3. Evaluate the initial candidates and keep the best
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = evaluate(cand.tolist())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop – LP based Frank‑Wolfe steps
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # parameters for the stochastic parts
    jitter_sigma = 0.02
    restart_interval = 30.0  # seconds without improvement → restart
    alphas = [0.1, 0.2, 0.4, 0.6, 0.8]   # candidate step sizes

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # 4.1 current max‑convolution value M
        # ----------------------------------------------------------
        conv_ff = np.convolve(cur_vec, cur_vec)          # full convolution
        M = conv_ff.max()
        cur_J = 2.0 * n * M   # exact objective for a sum‑=1 vector

        # ------------------------------------------------------------------
        # 4.2 Linear program: maximise total mass while keeping conv(f,g) ≤ M
        # ------------------------------------------------------------------
        A = conv_matrix(cur_vec)          # (2n‑1) × n matrix
        c = -np.ones(n, dtype=np.float64)  # minimise –∑g  ⇔ maximise ∑g
        b_ub = np.full(conv_len, M, dtype=np.float64)

        # Solve the LP
        lp_res = linprog(c, A_ub=A, b_ub=b_ub,
                         bounds=(0, None), method='highs',
                         options={'presolve': True})
        if not lp_res.success:
            print(f"[{elapsed:6.2f}s] LP failed (iter {iteration}); terminating.")
            break

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print(f"[{elapsed:6.2f}s] LP returned zero mass; terminating.")
            break

        # Normalise g to a probability vector (sum = 1)
        g = g0 / sum_g0

        # ----------------------------------------------------------
        # 4.3 Line‑search on mixing coefficient α
        # ----------------------------------------------------------
        best_local_J = cur_J
        best_local_vec = cur_vec
        improved = False

        for alpha in alphas:
            trial = (1.0 - alpha) * cur_vec + alpha * g   # already sums to 1
            trial_J = evaluate(trial.tolist())
            if trial_J < best_local_J - 1e-12:
                best_local_J = trial_J
                best_local_vec = trial.copy()
                improved = True

        # ----------------------------------------------------------
        # 4.4 Accept improvement or fallback to a tiny jitter
        # ----------------------------------------------------------
        if improved:
            cur_vec = best_local_vec
            cur_J = best_local_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
        else:
            # jitter to escape a possible plateau
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            jittered = cur_vec * noise
            jittered = proj_simplex(jittered)
            jitter_J = evaluate(jittered.tolist())
            if jitter_J < cur_J - 1e-12:
                cur_vec = jittered
                cur_J = jitter_J
                if jitter_J < best_J:
                    best_J = jitter_J
                    best_vec = jittered.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] Jitter improved J = {best_J:.6f}")

        iteration += 1

        # ----------------------------------------------------------
        # 4.5 Restart if we have been idle for a while
        # ----------------------------------------------------------
        if now - last_improve > restart_interval:
            # start from a fresh Dirichlet draw
            new_start = rng.dirichlet(np.full(n, 0.5))
            cur_vec = new_start
            cur_J = evaluate(new_start.tolist())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – J = {cur_J:.6f}")

        # ----------------------------------------------------------
        # 4.6 Early exit if target reached
        # ----------------------------------------------------------
        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # ------------------------------------------------------------------
    # 5. Return the best solution (clipped and renormalised)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() < 0.01:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec = best_vec / best_vec.sum()
    return [float(v) for v in best_vec]
