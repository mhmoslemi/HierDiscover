        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0:
            v /= s
            init_vec = v.copy()
    if init_vec is None:
        init_vec = rng.dirichlet(np.ones(n) * 0.5)
    cur_vec = init_vec.copy()
    # maintain best
    best_vec = cur_vec.copy()
    # compute initial J
    cur_conv = conv_full_fft(cur_vec, fft_len)
    cur_J = 2.0 * n * cur_conv.max()

    # schedule parameters
    alpha = 0.5
    alpha_max = 20.0
    alpha_inc_rate = (alpha_max / alpha) ** (1.0 / (deadline - start_time))  # approx exponential growth per second

"""
Hybrid smoothed‑gradient + greedy exchange optimiser.

* Works on the probability simplex (∑a = 1, a_i ≥ 0).  The objective
  J(a) = 2·n·max(conv(a,a)) is invariant under scaling.
* A continuation scheme is used: the non‑smooth max is replaced by a
  soft‑max with temperature α, which is slowly increased from a small
  value (smooth landscape) to a large value (close to the true max).
* The gradient of the smoothed objective is obtained analytically:
      ∇f(a) = 2·(a_rev * w)[n‑1 : n‑1+n],
  where w are the soft‑max weights of the convolution entries.
  Both the convolution and the correlation are evaluated with real FFTs.
* Three move families are mixed:
      – “smooth‑gradient” step (≈60 % of the time) using momentum
        and Euclidean projection onto the simplex.
      – “greedy exchange” (≈30 %) which moves a small amount of mass
        from the index with the largest sub‑gradient to the index with
        the smallest, directly targeting the active constraints.
      – occasional random jitter (≈10 %) to escape plateaus.
* A Metropolis acceptance rule with an exponential cooling schedule
  lets the search accept occasional uphill moves early on.
* Stagnation (no improvement for 30 s) triggers a jitter‑restart
  centred on the best solution found so far.
* The best vector is clipped to the allowed range [0, 1000] and
  re‑normalised before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) via a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * a.size - 1]


def gradient_softmax(a: np.ndarray,
                    conv: np.ndarray,
                    alpha: float,
                    fft_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv).

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector (sum = 1, non‑negative).
    conv : np.ndarray, shape (2n‑1,)
        Full self‑convolution of a (already computed).
    alpha : float
        Soft‑max temperature – larger values make the approximation tighter.
    fft_len : int
        Power‑of‑two length used for the FFT of the convolution (≥2n‑1).

    Returns
    -------
    grad : np.ndarray, shape (n,)
        Gradient of f_α(a) = (1/α)·log∑_k exp(α·conv_k) w.r.t. a.
    """
    # soft‑max weights (numerically stable)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    w /= w.sum()

    # correlation a_rev ★ w  (linear, not circular)
    a_rev = a[::-1]
    L = w.size
    pad = _next_pow2(a_rev.size + L - 1)

    A_rev = np.fft.rfft(a_rev, pad)
    W = np.fft.rfft(w, pad)
    corr = np.fft.irfft(A_rev * W, n=pad)[: a_rev.size + L - 1]

    n = a.size
    # grad_i = 2 * (a_rev * w)[i + n - 1]
    grad = 2.0 * corr[n - 1: n - 1 + n]
    return grad


def subgradient_active_set(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) at the current active set.

    Returns a vector of the same length as a.
    """
    n = a.size
    max_conv = conv.max()
    active = np.where(conv >= max_conv - tol)[0]

    grad = np.zeros_like(a)
    # each active shift contributes a shifted copy of a
    for p in active:
        start = max(0, p - (n - 1))
        end = min(p, n - 1)
        idx = np.arange(start, end + 1)
        grad[idx] += 2.0 * a[p - idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the final sanitisation
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1. Problem size – reuse the length of the previous construction if
    #    available, otherwise pick a moderate power of two.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Initial point (simplex, sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        v0 = np.asarray(height_sequence_1, dtype=np.float64)
        v0 = np.clip(v0, 0.0, None)
        s0 = v0.sum()
        if s0 > 0.0:
            v0 /= s0
        else:
            v0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        v0 = np.full(n, 1.0 / n, dtype=np.float64)

    cur_vec = v0.copy()
    best_vec = cur_vec.copy()

    # initial objective
    cur_conv = conv_full_fft(cur_vec, fft_len)
    cur_J = 2.0 * n * cur_conv.max()
    best_J = cur_J
    last_improve = start_time
    print(f"[0.00s] start n={n}, initial J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 3. Optimisation parameters
    # --------------------------------------------------------------
    # soft‑max temperature (continuation)
    alpha = 0.5
    alpha_max = 20.0
    # exponential α‑growth (≈ one e‑fold per second)
    total_time = deadline - start_time
    alpha_growth = (alpha_max / alpha) ** (1.0 / total_time)

    # step‑size schedule (Euclidean)
    eta0 = 0.5

    # momentum for the gradient moves
    momentum = np.zeros_like(cur_vec)
    beta = 0.9

    # Metropolis temperature schedule
    T_start = cur_J
    T_end = 1e-8

    iteration = 0

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- update continuation parameter α -----------------------
        alpha = min(alpha_max, alpha * alpha_growth)

        # ---- recompute convolution and exact objective -------------
        cur_conv = conv_full_fft(cur_vec, fft_len)
        max_conv = cur_conv.max()
        cur_J = 2.0 * n * max_conv

        # ---- record global best -----------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best J = {best_J:.6f}")

        # ---- decide which move to propose -------------------------
        move_type = rng.random()
        if move_type < 0.60:
            # ---- smooth‑gradient move (majority) ------------------
            grad = gradient_softmax(cur_vec, cur_conv, alpha, fft_len)
            # momentum
            momentum = beta * momentum + (1.0 - beta) * grad
            step = eta0 / np.sqrt(iteration + 1.0)
            prop = cur_vec - step * momentum
        elif move_type < 0.90:
            # ---- greedy exchange move (sub‑gradient based) -------
            tol = max(1e-12, max_conv * 1e-12)
            subgrad = subgradient_active_set(cur_vec, cur_conv, tol)
            i_max = int(np.argmax(subgrad))
            i_min = int(np.argmin(subgrad))
            # transfer a modest amount of mass
            delta = min(cur_vec[i_max],
                        rng.random() * 0.01 * cur_vec[i_max] + 1e-12)
            prop = cur_vec.copy()
            prop[i_max] -= delta
            prop[i_min] += delta
        else:
            # ---- random jitter ------------------------------------
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            prop = cur_vec * jitter

        # ---- simplex projection ------------------------------------
        prop = proj_simplex(prop)

        # ---- evaluate proposal -------------------------------------
        prop_conv = conv_full_fft(prop, fft_len)
        prop_J = 2.0 * n * prop_conv.max()

        # ---- Metropolis acceptance ---------------------------------
        # exponential cooling (linear in log‑time)
        T = T_start * (T_end / T_start) ** (elapsed / total_time)
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(prop_J - cur_J) / (T + 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_vec = prop
            # cur_J will be recomputed at the top of the loop

        # ---- stagnation handling ----------------------------------
        if now - last_improve > 30.0:
            # jitter around the best solution to escape a plateau
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * jitter
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] jitter restart – best J = {best_J:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Final sanitisation (clip & re‑normalise)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
