    else:

    return [float(v) for v in best_vec]

"""
propose_candidate – hybrid search for a non‑negative step function.

The routine works in three stages:

1. **Broad parametric sweep** – builds a diverse portfolio of
   analytically‑parameterised candidates (geometric, power‑law,
   exponential·polynomial, Dirichlet draws, random permutations and the
   previously discovered `height_sequence_1`).  All candidates are
   normalised to sum = 1 and the one with the smallest objective
   `J = 2·n·max(conv(a,a))` is kept.

2. **Simulated‑annealing local search** – starting from the best
   candidate a small “mass‑transfer” move (move ε from index i to j)
   is proposed.  The move is accepted if it improves the objective or
   with a Metropolis probability that decays with a linear temperature
   schedule.  Stagnation for >30 s triggers a cheap jitter (mix with a
   tiny Dirichlet noise) to escape plateaus.

3. **Short Adam refinement** – the final candidate is polished by a few
   Adam steps on the soft‑max logits, using the exact sub‑gradient of
   `max(conv)` (identical to the gradient used in the reference code).

Fast convolution is performed with a real‑FFT (rfft/irfft), which makes
each evaluation O(n log n) and allows thousands of iterations within the
1000 s budget.  The routine returns a list of floats that can be fed
directly into `evaluate_sequence`.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_conv(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Return max of the full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Compute J = 2·n·max(conv(a,a))."""
    return 2.0 * n * _max_conv(a, fft_len, conv_len)

    ad_end = deadline
    t = 0
    while time.monotonic() < ad_end:
        t += 1
        # soft‑max
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()

        # convolution & active set
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_c = conv.max()
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv >= max_c - tol)[0]

        # sub‑gradient of max(conv) w.r.t. a
        grad_a = np.zeros_like(a)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=int)
            grad_a[idx] += 2.0 * a[p - idx]

        # chain rule to logits space
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)

        # Adam update
        lr = lr0 / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        logits = np.clip(logits, -500.0, 500.0)

    # final vector from logits
    max_log = np.max(logits)
    exp_logits = np.exp(logits - max_log)
    final_a = exp_logits / exp_logits.sum()
    final_a = np.clip(final_a, 0.0, 1000.0)
    total = final_a.sum()
    if total == 0.0:
        final_a = np.full_like(final_a, 1.0 / n)
    else:
        final_a /= total

    print(f"Final best J ≈ {compute_J(final_a):.6f}")
    return [float(v) for v in final_a]
