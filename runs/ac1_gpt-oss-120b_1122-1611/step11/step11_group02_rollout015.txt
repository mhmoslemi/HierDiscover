"""
AdaGrad‑Adam hybrid optimiser for the step‑function bound.

The objective (after projecting the vector onto the simplex, Σa=1) is

    J(a) = 2·n·max(conv(a,a)) ,

with a_i ≥ 0.  It is scale‑invariant, so we can safely work on the
probability simplex.  The algorithm below combines a *soft‑max*
approximation of the sup‑norm (temperature p) with an Adam style
adaptive step–size, followed by a Euclidean projection onto the simplex.
Periodically a small random “jitter’’ is injected when the search stalls.
A final fine‑grained random search around the best solution is performed
before returning the candidate sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def gradient_smooth(a: np.ndarray, conv: np.ndarray,
                    p: float, fft_len_g: int, conv_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of the max‑norm:

        φ(a) = log‑sum‑exp(p·conv(a,a)) / p .

    The gradient is
        g = 2 * (w ⋆ a_rev)
    where w_k = exp(p·conv_k) / Σ_j exp(p·conv_j)
    and ⋆ denotes linear convolution.
    """
    # stable soft‑max weights
    w = np.exp(p * (conv - conv.max()))
    # reverse a for the cross‑correlation needed in the gradient
    a_rev = a[::-1]
    grad_full = np.fft.irfft(
        np.fft.rfft(w, fft_len_g) * np.fft.rfft(a_rev, fft_len_g),
        n=fft_len_g
    )
    # slice the part that corresponds to the gradient entries
    grad = 2.0 * grad_full[conv_len - 1: conv_len - 1 + a.size]
    return grad

# Main search routine
# ----------------------------------------------------------------------

    for cand in init_cands:
        conv = conv_full_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam‑style optimiser with a smooth‑max surrogate.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # temperature (soft‑max) schedule
    p_start = 2.0
    p_end = 300.0

    # Adam hyper‑parameters
    alpha = 0.12                     # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(best_vec)     # first moment estimate
    v_hat = np.zeros_like(best_vec) # second moment estimate
    t = 0

    cur_vec = best_vec.copy()
    cur_J = best_J
    best_seen_vec = best_vec.copy()
    best_seen_J = best_J
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # --------------------------------------------------------------
        # 3.1 Update the smooth‑max temperature.
        # --------------------------------------------------------------
        p = p_start + (p_end - p_start) * frac

        # --------------------------------------------------------------
        # 3.2 Compute convolution and current objective.
        # --------------------------------------------------------------
        conv = conv_full_fft(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # --------------------------------------------------------------
        # 3.3 Track the global best.
        # --------------------------------------------------------------
        if cur_J < best_seen_J - 1e-12:
            best_seen_J = cur_J
            best_seen_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_seen_J:.6f}")

        # --------------------------------------------------------------
        # 3.4 Stagnation handling – inject jitter after 25 s without progress.
        # --------------------------------------------------------------
        if now - last_improve > 25.0:
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            cur_vec = proj_simplex(best_seen_vec * jitter)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_seen_J:.6f}")

        # --------------------------------------------------------------
        # 3.5 Gradient of the smoothed max.
        # --------------------------------------------------------------
        grad = gradient_smooth(cur_vec, conv, p, fft_len_g, conv_len)

        # --------------------------------------------------------------
        # 3.6 Adam update (projected onto simplex).
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat_corr) + eps)

        # clip steps to avoid overflow in the exponentials later on
        max_step = 0.8
        if np.max(np.abs(step)) > max_step:
            step = step * (max_step / np.max(np.abs(step)))

        proposal = cur_vec - step
        proposal = np.clip(proposal, 0.0, None)
        proposal = proj_simplex(proposal)

        # --------------------------------------------------------------
        # 3.7 Evaluate the proposal.
        # --------------------------------------------------------------
        prop_conv = conv_full_fft(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # --------------------------------------------------------------
        # 3.8 Simple Metropolis acceptance.
        # --------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # a tiny probability to escape flat regions
            if rng.random() < 1e-8:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

    # ------------------------------------------------------------------
    # 4. Fine‑grained random search around the best solution.
    # ------------------------------------------------------------------
    fin_start = time.monotonic()
    fin_deadline = fin_start + min(30.0, (deadline - start_time) * 0.04)

    while time.monotonic() < fin_deadline:
        # multiplicative Gaussian noise (σ = 0.015)
        noise = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = proj_simplex(best_seen_vec * noise)
        conv = conv_full_fft(cand, fft_len, conv_len)
        cand_J = float(2.0 * n * conv.max())
        if cand_J < best_seen_J:
            best_seen_J = cand_J
            best_seen_vec = cand
            print(f"[Fine] Improved J = {best_seen_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (already on the simplex, non‑negative).
    # ------------------------------------------------------------------
    best_seen_vec = np.clip(best_seen_vec, 0.0, 1000.0)
    total = best_seen_vec.sum()
    if total == 0.0:
        best_seen_vec[:] = 1.0 / n
    else:
        best_seen_vec /= total

    return [float(v) for v in best_seen_vec]
