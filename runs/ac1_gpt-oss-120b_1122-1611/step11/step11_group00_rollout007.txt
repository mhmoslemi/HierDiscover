def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5

    # Choose n = 4096 (maybe adjust if height_sequence_1 length available).
    # If height_sequence_1 exists, set n to its length (capped) else n=4096.
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())  # next pow2

    # Helper functions for projection onto simplex
    def proj_simplex(v):
        # Projects v onto probability simplex (nonnegative sum=1)
        # Implementation using sorting method
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    def evaluate_a(a):
        # Compute J=2*n*max(conv) (sum a =1)
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_conv = conv.max()
        return float(2.0 * n * max_conv), conv

    # Build initial candidates
    init_candidates = []
    # uniform
    init_candidates.append(np.full(n, 1.0 / n))
    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        v = np.power(r, np.arange(n, dtype=np.float64))
        v /= v.sum()
        init_candidates.append(v)
    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_candidates.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_candidates.append(v)
    # Gaussian bump
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n/8, n/6, n/4):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_candidates.append(v)
    # Dirichlet draws
    for alpha in (0.3, 0.5, 0.8, 1.0):
        for _ in range(3):
            init_candidates.append(rng.dirichlet(np.ones(n) * alpha))

    # If previous best available, include flattened/trimmed version
    if 'height_sequence_1' in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                pad = np.full(n - v.size, 0.0)
                v = np.concatenate([v, pad])
        v = np.clip(v, 0.0, None)
        if v.sum() == 0:
            v = np.full(n, 1.0 / n)
        else:
            v /= v.sum()
        init_candidates.append(v)

    # Choose best initial based on objective
    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        J, _ = evaluate_a(cand)
        if J < best_J:
            best_J = J
            best_a = cand.copy()
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # Keep global best
    global_best_a = best_a.copy()
    global_best_J = best_J

    # Phase 1: Projected subgradient descent (ADAM style)
    # We'll use simple Adam on logits as before but also allow zeros via projection.
    # We'll run a limited number of steps.
    # However we will incorporate this phase only if time left.

    # Let's compute an initial step size schedule.
    # Use logits representation for stability as before but we will also maintain raw a for direct projection.
    # Actually we'll just use simple projected gradient descent (no logits) for stability.

    # Define parameters
    max_iter = 200000  # high bound, but will break on time

    iteration = 0
    last_improve = start
    # We'll keep momentum (Adam) but on a directly: Actually we can maintain first and second moments of gradient for Adam (like RMSprop) but we also need projection after step.
    m = np.zeros_like(best_a)
    v = np.zeros_like(best_a)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.5

    while time.monotonic() < deadline:
        # compute gradient of max conv
        J_cur, conv = evaluate_a(best_a)
        # if improvement
        if J_cur < global_best_J - 1e-12:
            global_best_J = J_cur
            global_best_a = best_a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # Stagnation: if no improvement in 30 seconds, restart from a random Dirichlet
        now = time.monotonic()

            # reset to random candidate
            best_a = rng.dirichlet(np.ones(n) * 0.5)
            # reset moments
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now-start:6.2f}s] Restart with random Dirichlet due to stagnation.")
            continue

        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]

        # Compute subgradient w.r.t a
        grad = np.zeros_like(best_a)
        # loop over active indices
        for k in active:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            # vectorized for i range
            i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
            j = k - i_range

            # compute mask
            mask_eq = (i_range == j)

            grad[i_range[mask_eq]] += best_a[i_range[mask_eq]]
            # add 2*a_j for not equal
            grad[i_range[~mask_eq]] += 2.0 * best_a[j[~mask_eq]]

        grad *= 2.0 * n  # because dJ/dmaxconv = 2*n, and gradient of maxconv from conv
        # Adam update on a (before projection)
        iteration += 1
        lr = lr0 / np.sqrt(iteration)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        # gradient step
        a_tent = best_a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        # Project onto simplex
        a_proj = proj_simplex(a_tent)
        best_a = a_proj

        # Also schedule occasional jitter to avoid plateau (like adding small Gaussian)
        if iteration % 5000 == 0:
            sigma = 0.01 * (1.0 - (now - start) / budget_s)
            best_a = proj_simplex(best_a + sigma * rng.normal(size=n))

        # Possibly break if time close to deadline
        if time.monotonic() > deadline - 5.0:
            break

    # Phase 2: Simulated annealing refinement
    # If there is time left, run SA
    remainder = deadline - time.monotonic()
    if remainder > 5.0:
        # temperature schedule: start at fraction of best_J
        T0 = global_best_J * 0.1
        # Linear cooling to near zero at end.
        SA_start_time = time.monotonic()
        while time.monotonic() < deadline:
            # Determine current temperature based on elapsed fraction
            elapsed = time.monotonic() - SA_start_time
            frac = elapsed / remainder
            T = T0 * (1.0 - frac)  # linear decay

            # Propose move: pick i,j
            i = rng.integers(n)
            # ensure a_i > 0 to move from
            if global_best_a[i] < 1e-12:
                continue
            # pick j possibly different
            j = rng.integers(n)
            # transfer amount delta
            delta = rng.random() * min(global_best_a[i], 0.05)  # up to 5% of a_i
            a_new = global_best_a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # renorm (should keep sum=1)
            # evaluate
            J_new, _ = evaluate_a(a_new)
            # accept?
            if J_new < global_best_J:
                global_best_J = J_new
                global_best_a = a_new
                # print improvement
                print(f"[{time.monotonic()-start:6.2f}s] SA improve J = {global_best_J:.6f}")
            else:
                # Metropolis acceptance
                if T > 0 and rng.random() < np.exp(-(J_new - global_best_J) / T):
                    global_best_a = a_new

        # end SA

    # Finally, ensure values within allowed range and sum normalized
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

"""
Propose a step‑function by a two‑phase optimisation.
Phase 1 – projected Adam on the simplex (sub‑gradient of max‑convolution).
Phase 2 – simple simulated‑annealing refinements.
The method works directly on the probability simplex so the returned list
already satisfies the pre‑conditions of `evaluate_sequence`.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    # due to numerical errors the sum may deviate slightly – renormalise
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum

def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int):
    """Return (J, full convolution) where J = 2·n·max(conv) and conv = a*a."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    max_conv = conv.max()
    J = float(2.0 * n * max_conv)
    return J, conv

    v /= v.sum()
    init_candidates.append(v)

        for k in active:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            i_arr = np.arange(start_i, end_i + 1, dtype=np.int64)
            j_arr = k - i_arr
            neq = i_arr != j_arr
            # i != j contribution: 2*a_j
            grad[i_arr[neq]] += 2.0 * best_a[j_arr[neq]]
            # i == j contribution: a_i
            grad[i_arr[~neq]] += best_a[i_arr[~neq]]

        # scale for objective J = 2·n·max_conv
        grad *= 2.0 * n

        # Adam update (before projection)
        iteration += 1
        lr = lr0 / np.sqrt(iteration)
        m = beta1 * m + (1.0 - beta1) * grad
        v_mom = beta2 * v_mom + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_mom / (1.0 - beta2 ** iteration)
        a_tent = best_a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # Projection onto the simplex guarantees feasibility
        best_a = _proj_simplex(a_tent)

        # Occasional jitter to kick the optimiser out of plateaus
        if iteration % 5000 == 0:
            sigma = 0.01 * (1.0 - (now - start_time) / budget_s)
            if sigma > 0:
                best_a = _proj_simplex(best_a + sigma * rng.normal(size=n))

    # ----------------------------------------------------------------------
    # 5. Phase‑2 – lightweight simulated‑annealing refinement (if time left)
    # ----------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        # Simple linear cooling schedule
        SA_start = time.monotonic()
        T0 = max(0.1 * global_best_J, 1e-3)   # initial temperature
        while time.monotonic() < deadline:
            elapsed = time.monotonic() - SA_start
            frac = elapsed / remaining
            T = T0 * (1.0 - frac)   # temperature decays to zero

            # propose a transfer move
            i = rng.integers(n)
            if global_best_a[i] < 1e-12:
                continue
            j = rng.integers(n)
            # transfer a random fraction of the mass at i
            delta = rng.random() * min(global_best_a[i], 0.05)   # at most 5 % of a_i
            a_new = global_best_a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            J_new, _ = _objective(a_new, n, fft_len, conv_len)

            if J_new < global_best_J - 1e-12:
                global_best_J = J_new
                global_best_a = a_new
                elapsed_total = time.monotonic() - start_time
                print(f"[{elapsed_total:6.2f}s] SA improvement → J = {global_best_J:.6f}")
            else:
                # Metropolis acceptance
                if T > 0 and rng.random() < np.exp(-(J_new - global_best_J) / T):
                    global_best_a = a_new

    # ----------------------------------------------------------------------
    # 6. Final normalisation and return
    # ----------------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

            frac = elapsed / remaining
            T = T0 * (1.0 - frac)                     # linear cooling

            # Propose a “mass‑transfer” move
            i = rng.integers(n)
            if global_best_a[i] < 1e-12:
                continue
            j = rng.integers(n)
            # Transfer up to 5 % of the weight at i
            delta = rng.random() * min(global_best_a[i], 0.05)
            a_new = global_best_a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            J_new, _ = _objective(a_new, n, fft_len, conv_len)

            if J_new < global_best_J - 1e-12:
                global_best_J = J_new
                global_best_a = a_new
                elapsed_tot = time.monotonic() - t_start
                print(f"[{elapsed_tot:6.2f}s] SA improvement → J = {global_best_J:.6f}")
            else:
                # Metropolis acceptance
                if T > 0 and rng.random() < np.exp(-(J_new - global_best_J) / T):
                    global_best_a = a_new

    # ------------------------------------------------------------------
    # 6. Final normalisation and return
    # ------------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
