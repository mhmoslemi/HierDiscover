"""
Propose a candidate sequence for `evaluate_sequence`.

Algorithm
---------
* Work on the probability simplex by parametrising the vector
  `a = softmax(z)`.  The objective `J = 2·n·max(conv(a,a))`
  is scale‑invariant for `∑a = 1`.
* The non‑smooth max is approximated by a smooth log‑sum‑exp
  with a temperature parameter `p`.  The gradient of the
  smoothed objective can be written in closed form as a
  convolution of the weight vector `w` (the soft‑max over
  the convolution entries) with the reversed probability
  vector.  This convolution is evaluated by an FFT – O(n log n).
* Adam is used on the logits `z`.  The learning‑rate decays
  like 1/√t, moments are bias‑corrected, and a simple jitter
  restart is performed if no improvement is seen for a while.
* Several heuristic initialisations (uniform, geometric,
  ramps, Gaussian bumps, Dirichlet draws and the previously
  discovered `height_sequence_1`) are tried and the best
  one is taken as the start point.
* The routine runs for at most `budget_s` seconds and returns
  the best sequence found, clipped to the allowed range and
  re‑normalised so that `evaluate_sequence` can be called
  without further processing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT, log‑sum‑exp, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (output length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _conv_weighted(w: np.ndarray, a_rev: np.ndarray, fft_len: int, out_len: int) -> np.ndarray:
    """Convolution w * a_rev (full) using real FFT."""
    W = np.fft.rfft(w, fft_len)
    A = np.fft.rfft(a_rev, fft_len)
    full = np.fft.irfft(W * A, n=fft_len)[:out_len]
    return full


def _smooth_weights(conv: np.ndarray, p: float) -> np.ndarray:
    """
    Return a probability vector `w` that approximates the argmax of `conv`
    using the soft‑max with temperature `p` (larger p → sharper).
    """
    # scale before exponentiation for numerical stability
    max_val = np.max(conv)
    # w_i ∝ exp(p * (conv_i - max_val))
    w = np.exp(p * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:  # safety, should not happen
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the evaluation function `evaluate_sequence`.  The routine
    respects the time budget and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (n).  We keep it a power of two for FFT.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init_len = len(height_sequence_1)
            # clamp to a reasonable range
            n = int(max(256, min(init_len, 8192)))
        except Exception:
            n = 2048
    else:
        n = 2048

    # Round up to the nearest power of two (helps FFT a little)
    n = _next_pow2(n)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                 # for a * a
    # for weighted convolution we need length (2n‑1) + n ‑ 1 = 3n‑2
    weighted_len = 3 * n - 2
    fft_len_w = _next_pow2(weighted_len)

    # --------------------------------------------------------------
    # 2. Build a portfolio of diverse initial probability vectors.
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – moderately and strongly concentrated
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (by raw J = 2·n·max(conv)).
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    start_time = time.monotonic()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop – Adam on logits + smooth‑max gradient.
    # --------------------------------------------------------------
    # Soft‑max parametrisation
    eps = 1e-12
    logits = np.log(np.maximum(best_a, eps))
    m = np.zeros_like(logits)   # first moment
    v = np.zeros_like(logits)   # second moment

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.5  # base learning rate

    iteration = 0
    last_improve = start_time
    global_best_a = best_a.copy()
    global_best_J = best_J

    # temperature (soft‑max sharpness) schedule
    p_start = 6.0
    p_end = 30.0

    # main loop
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    while time.monotonic() < deadline:
        # ---- 4.1 Build current probability vector a from logits ----
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()

        # ---- 4.2 Compute convolution and smooth weights ----
        conv = _conv_full(a, fft_len, conv_len)          # length 2n‑1
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # adapt temperature linearly with time
        elapsed = time.monotonic() - start_time
        p = p_start + (p_end - p_start) * (elapsed / budget_s)
        w = _smooth_weights(conv, p)                     # length 2n‑1, sums to 1

        # ---- 4.3 Gradient of the smoothed objective w.r.t. a ----
        # gradient = 2 * (w * a_rev)[n-1 : n-1+n]
        a_rev = a[::-1]
        full_conv = _conv_weighted(w, a_rev, fft_len_w, weighted_len)
        grad_a = 2.0 * full_conv[n - 1: n - 1 + n]

        # ---- 4.4 Transform gradient to logits space (soft‑max) ----
        avg = np.dot(a, grad_a)          # <a, grad_a>
        grad_z = a * (grad_a - avg)      # ∂J/∂z

        # ---- 4.5 Adam step on logits ----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)    # simple 1/√t decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        logits = np.clip(logits, -500.0, 500.0)   # keep numerics sane

        # ---- 4.6 Update best solution seen so far ----
        now = time.monotonic()
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] New best J = {global_best_J:.6f}")

        # ---- 4.7 Detect stagnation, apply jitter restart ----
        if now - last_improve > 25.0:
            jitter_sigma = 0.15                     # somewhat larger jitter
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

    # --------------------------------------------------------------
    # 5. Return the best sequence (safely clipped and re‑normalised).
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
