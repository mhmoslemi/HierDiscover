Projected Adam on the sub‑gradient of the ‑max‑convolution objective.

The evaluation function is
    J(a) = 2·n·max(conv(a,a)) / (sum(a)²)
with a ≥ 0.  Because the objective is scale‑invariant we keep the
vector on the probability simplex (∑a = 1).  The gradient of the
un‑smoothed max is obtained by taking the binary mask of all indices
where the convolution attains its maximum and using the Fourier‑domain

        ∇_a Σ_k w_k·(a*a)_k = 2·ifft( conj(FFT(w)) · FFT(a) ),

where w is the mask.  An Adam step is taken in the original space,
followed by Euclidean projection onto the simplex.  Whenever the
search stagnates a random jitter is applied to escape shallow plateaus.
The routine runs for the supplied time budget (≈1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (zero‑padded)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best normalised sequence
    found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a safe default for speed vs quality

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Diverse initial candidates (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws – from spiky to smooth
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Gaussian bells of several widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 8.0, n / 5.0, n / 3.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial vector (objective = 2·n·max(conv))
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Adam optimiser on the sub‑gradient of the max‑convolution
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.3                # initial learning‑rate (will decay)

    cur_vec = best_vec.copy()
    m = np.zeros_like(cur_vec)      # first moment estimate
    v_acc = np.zeros_like(cur_vec)  # second moment estimate

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    last_improve = start_time
    iteration = 0
    stagnation_seconds = 20.0
    target_J = 1.5025

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- current objective ----
        conv = _conv_full(cur_vec, fft_len, conv_len)
        maxb = conv.max()
        cur_J = float(2.0 * n * maxb)

        # ---- update global best ----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        if best_J < target_J:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

        # ---- stagnation handling ----
        if now - last_improve > stagnation_seconds:
            # small multiplicative jitter around the current best
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            m.fill(0.0)
            v_acc.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")
            continue

        # ---- mask for the (near) maximal convolution entries ----
        tol = 1e-12
        mask = (conv >= maxb - tol).astype(float)
        mask_cnt = mask.sum()
        if mask_cnt == 0:          # safety fallback – use a soft‑max
            beta = 200.0
            w_raw = np.exp(beta * (conv - maxb))
            mask = w_raw / w_raw.sum()
            mask_cnt = mask.sum()

        # ---- gradient of Σ(mask·conv) w.r.t. a (Fourier domain) ----
        A = np.fft.rfft(cur_vec, fft_len)
        W = np.fft.rfft(mask, fft_len)
        g = 2.0 * np.fft.irfft(np.conj(W) * A, n=fft_len)[:n]
        if mask_cnt > 1:
            g = g / mask_cnt               # average over all tied maxima

        # ---- Adam update (additive) ----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)       # simple decay
        m = beta1 * m + (1.0 - beta1) * g
        v_acc = beta2 * v_acc + (1.0 - beta2) * (g * g)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_acc / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cur_vec = cur_vec - step
        cur_vec = proj_simplex(cur_vec)     # back onto the probability simplex

    # --------------------------------------------------------------
    # 5. Return the best solution (clipped and renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
