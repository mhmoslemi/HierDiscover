"""
Propose a step‑function (non‑negative probability vector) that minimizes
`evaluate_sequence`.  The method is a simple *hard‑max sub‑gradient descent*
on the quantity `max_k (a * a)[k]` with projection onto the simplex.
Random restarts with multiplicative jitter are used to escape plateaus.
The algorithm runs for the allotted time (default 1000 s) and returns the
best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to ``x``."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to one.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (the full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine – the function required by the evaluator
# ----------------------------------------------------------------------

                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (length of the step function)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096   # default – still cheap to FFT

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)                     # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)             # for correlation

    # --------------------------------------------------------------
    # 2. Build a diverse portfolio of initial probability vectors
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws with various concentrations
    for alpha in [0.3, 0.6, 1.0, 2.0, 5.0]:
        for _ in range(4):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Geometric decays (truncated)
    for r in [0.995, 0.998, 0.999, 0.9995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Small random masks – few spikes
    for k in [int(np.sqrt(n)), max(2, n // 10), n // 20]:
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        init_cands.append(mask)

    # If we already have a previous construction, add it (renormalised)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)

            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")
            # continue to next iteration (no gradient this round)
            iteration += 1
            continue

        # ----- sub‑gradient of the hard max -----
        # mask of indices achieving the maximal convolution value
        # (allow a tiny tolerance to catch ties caused by floating point)
        tol = max(1e-14, 1e-12 * max_conv)
        w = (conv >= max_conv - tol).astype(np.float64)          # shape (conv_len,)

        # correlation w ⋆ cur_vec  (linear, not circular)
        A_hat = np.fft.rfft(cur_vec, fft_len_corr)
        W_hat = np.fft.rfft(w, fft_len_corr)
        corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=fft_len_corr)
        start = n - 1
        subgrad = corr[start:start + n]                         # w ⋆ cur_vec

        # ----- update (additive mirror‑descent on the simplex) -----
        step = step0 / np.sqrt(iteration + 1.0)
        # gradient of J is 4*n * (w ⋆ cur_vec); we absorb the constant into step
        cur_vec = cur_vec - step * subgrad
        # keep feasibility
        cur_vec = np.maximum(cur_vec, 0.0)
        cur_vec = proj_simplex(cur_vec)

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best solution (clip and renormalise for safety)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
