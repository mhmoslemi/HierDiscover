"""
Hybrid simulated‑annealing + exponentiated‑gradient optimiser.

* Works on the probability simplex (∑a = 1, a_i ≥ 0) – the objective
  J(a) = 2·n·max(conv(a,a)) is invariant under scaling, so we keep the sum = 1.
* Moves:
    – Projected sub‑gradient step (using the active set of the max‑convolution).
    – Exponentiated‑gradient (multiplicative) step that pushes mass away from
      indices with large sub‑gradient.
    – Simple exchange of a random fraction of mass between two coordinates.
    – Local smoothing by a 3‑point kernel.
    – Periodic jitter after long stagnation.
* Acceptance follows a Metropolis rule with an exponential cooling schedule.
  The temperature starts high (≈10·current best) to allow exploration and
  cools to 1e‑8.
* Multiple random initialisations (uniform, geometric, Dirichlet, Gaussian)
  are tried and the best one is taken as the start.
* The algorithm respects the 1000 s budget and returns the best feasible
  sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all equal or all zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Returns the objective J = 2·n·max(conv(a,a)) for a vector a that
    lives on the simplex (sum(a) == 1).  The factor 2·n makes the
    quantity directly comparable with ``evaluate_sequence`` (which also
    divides by the square of the sum, but our sum is 1).
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgradient_active_set(a: np.ndarray,
                           conv: np.ndarray,
                           active_tol: float) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv)`` at the current active set.
    The implementation follows the reference code: each active
    position contributes a shifted copy of ``a``.
    """
    n = a.size
    max_conv = conv.max()
    active = np.where(conv >= max_conv - active_tol)[0]

    grad = np.zeros_like(a)
    for p in active:
        # the indices i where a_i participates in conv[p]
        start = max(0, p - (n - 1))
        end = min(p, n - 1)
        # indexes of a that are summed to produce conv[p]
        idx = np.arange(start, end + 1)
        # contribution 2 * a_{p‑idx}
        grad[idx] += 2.0 * a[p - idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
