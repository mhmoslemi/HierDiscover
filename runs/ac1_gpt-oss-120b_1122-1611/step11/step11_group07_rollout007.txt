Function `solve_lp_for_b(a, M)`:
   n = len(a)
   conv_len = 2*n - 1
   # Build coefficient matrix
   rows = []
   cols = []
   data = []
   for k in range(conv_len):
       # for each i in [max(0, k-(n-1)), min(k, n-1)]:
       i_start = max(0, k - (n - 1))
       i_end = min(k, n - 1)
       # For each i, j = k - i
       # coefficient = a[j]
       # j = k - i
       for i in range(i_start, i_end + 1):
           j = k - i
           rows.append(k)
           cols.append(i)
           data.append(a[j])
   A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(conv_len, n))
   # Constraints: A_ub @ b <= M
   # b >= 0 will be bound.
   # Objective: maximize sum(b) -> minimize -sum(b)
   c = -np.ones(n)
   bnds = [(0, None)] * n
   res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.full(conv_len, M), bounds=bnds, method='highs', options={'presolve': True})
   if res.success:
       b = np.maximum(res.x, 0.0)
       total = b.sum()
       if total > 0:
           b /= total
       return b
   else:
       # fallback to uniform
       return np.full(n, 1.0 / n)

import scipy.sparse as sp
def solve_lp_max_sum(a, M):
    n = len(a)
    conv_len = 2*n - 1
    # Build A_ub
    # Offsets 0..n-1 each with constant a[d]
    # Create a 2D array where each row corresponds to a constant diagonal
    diag_vals = np.tile(a, (n, 1))  # shape (n, n) each row is a
    # Use diags
    offsets = np.arange(n)
    A = sp.diags(diag_vals, offsets, shape=(conv_len, n), format='csr')
    # Solve LP: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(conv_len, M)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        b = np.maximum(res.x, 0.0)
        if b.sum() > 0:
            b /= b.sum()
        return b
    else:
        return np.full(n, 1.0 / n)
```

def targeted_exchange(a, conv, rng, max_idx):
    # conv is full convolution (linear)
    # max_idx is index of max conv
    n = len(a)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(max_idx, n - 1)
    # compute contributions
    idxs = np.arange(i_start, i_end+1)
    j = max_idx - idxs
    contrib = a[idxs] * a[j]
    i_max = idxs[np.argmax(contrib)]
    # choose random target index not equal
    target = rng.choice(n-1)
    if target >= i_max:
        target += 1
    # transfer small fraction
    delta = min(a[i_max], rng.random() * 0.02) # up to 2% mass
    a2 = a.copy()
    a2[i_max] -= delta
    a2[target] += delta
    return a2
```

"""
Hybrid local‑search + LP‑mix optimiser for the step‑function problem.

* Works on the probability simplex (∑a = 1, a_i ≥ 0); the objective
  J(a)=2·n·max(conv(a,a)) is invariant under scaling.
* Move families:
    – Projected sub‑gradient descent with momentum.
    – Random pair‑exchange (small mass transfer).
    – Targeted exchange that reduces the currently maximal convolution term.
* Every 30 s (if time allows) solves a linear programme:
      maximise ∑b_j subject to b_j≥0 and (a∗b)_k ≤ M,
  where M is the present maximal convolution value.
  The LP solution b is then convex‑combined with the current best vector,
  the mixing ratio being chosen by a cheap line‑search.
* A simple simulated‑annealing acceptance rule (temperature follows an
  exponential cooling schedule) lets the walk escape shallow plateaus.
* Stagnation (no improvement for 40 s) triggers a jitter‑reset centred on
  the best solution found so far.
* The routine respects the supplied time budget (default 1000 s) and
  returns the best sequence discovered.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or zero); return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective J = 2·n·max(conv(a,a)) for a summed to 1."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgradient_active_set(a: np.ndarray,
                           conv: np.ndarray,
                           active_tol: float) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the current active set.
    The result has the same length as a.
    """
    n = a.size
    max_conv = conv.max()
    active = np.where(conv >= max_conv - active_tol)[0]

    grad = np.zeros_like(a)
    # each active position contributes a shifted copy of a
    for p in active:
        start = max(0, p - (n - 1))
        end = min(p, n - 1)
        idx = np.arange(start, end + 1, dtype=int)
        # contribution 2 * a_{p‑idx}
        grad[idx] += 2.0 * a[p - idx]
    return grad


def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise   ∑ b_j
    subject to  b ≥ 0,  (a ∗ b)_k ≤ M   for all k.
    Returns the feasible b projected onto the simplex.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1

    # Build A_ub where each row k is a shifted copy of a.
    # This is a banded matrix – each diagonal offset d (0≤d<n) has constant value a[d].
    diags_data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)               # 0,1,…,n‑1
    A = sp.diags(diags_data, offsets,
                 shape=(conv_len, n), format='csr')

    c = -np.ones(n)                      # minimise –∑b  ⇔ maximise ∑b
    b_ub = np.full(conv_len, M)
    bounds = [(0, None)] * n

    # Linear programming via HiGHS (fast, handles sparse matrices)
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True})
    if res.success:
        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s > 0:
            b /= s
        else:
            b = np.full(n, 1.0 / n)
        return b
    else:
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)


def targeted_exchange(a: np.ndarray,
                      conv: np.ndarray,
                      rng: np.random.Generator,
                      max_idx: int) -> np.ndarray:
    """
    Transfer a small amount of mass from an index participating heavily in the
    maximal convolution entry to a random other index.
    """
    n = a.shape[0]
    i_start = max(0, max_idx - (n - 1))
    i_end = min(max_idx, n - 1)
    idxs = np.arange(i_start, i_end + 1, dtype=int)
    js = max_idx - idxs
    contrib = a[idxs] * a[js]
    i_max = idxs[np.argmax(contrib)]

    # pick a distinct target index
    target = rng.integers(n - 1)
    if target >= i_max:
        target += 1

    delta = min(a[i_max], rng.random() * 0.02)   # up to 2 % of the mass
    a2 = a.copy()
    a2[i_max] -= delta
    a2[target] += delta
    return a2

# Main optimisation routine

def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``. The function respects the optional time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a diverse pool of initial points (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – various concentrations
    for alpha in (0.3, 0.6, 1.0, 2.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (cheap objective evaluation)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    total_time = deadline - start_time
    T_start = best_J
    T_end = 1e-8

    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    last_improve = start_time
    iteration = 0
    momentum = np.zeros_like(cur_vec)
    beta = 0.9
    lp_cooldown = 0.0      # time until next LP attempt

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # exponential temperature schedule
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # recompute convolution for the current vector
        cur_conv = _conv_full(cur_vec, fft_len, conv_len)
        max_idx = int(cur_conv.argmax())

        # ----------------------------------------------------------
        # 4.1  Choose a move
        # ----------------------------------------------------------
        rnd = rng.random()
        if rnd < 0.55:
            # gradient‑type move
            max_conv = cur_conv.max()
            tol = max(1e-12, max_conv * 1e-12)
            grad = subgradient_active_set(cur_vec, cur_conv, tol)
            momentum = beta * momentum + (1.0 - beta) * grad
            step = 0.6 / np.sqrt(iteration + 1.0)
            prop = cur_vec - step * momentum
        elif rnd < 0.80:
            # random pair‑exchange
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(cur_vec[i], rng.random() * 0.02)   # ≤ 2 % of mass
            prop = cur_vec.copy()
            prop[i] -= delta
            prop[j] += delta
        else:
            # targeted exchange based on the active maximal convolution term
            prop = targeted_exchange(cur_vec, cur_conv, rng, max_idx)

        # ----------------------------------------------------------
        # 4.2  Project and evaluate
        # ----------------------------------------------------------
        prop = proj_simplex(prop)
        prop_J = compute_J(prop, n, fft_len, conv_len)

        # ----------------------------------------------------------
        # 4.3  Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = prop
            cur_J = prop_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

        # ----------------------------------------------------------
        # 4.4  Periodic LP‑mix step (≈ every 30 s, if time permits)
        # ----------------------------------------------------------
        if (now - lp_cooldown) > 30.0 and (deadline - now) > 5.0:
            # recompute for the global best (more stable)
            best_conv = _conv_full(best_vec, fft_len, conv_len)
            M = best_conv.max()
            b = solve_lp_max_sum(best_vec, M)

            # try a handful of convex combinations
            t_candidates = np.concatenate([
                np.linspace(0.0, 0.5, 6),
                rng.uniform(0.0, 0.5, size=4),
                rng.uniform(0.5, 1.0, size=2)
            ])
            for t in t_candidates:
                cand = (1.0 - t) * best_vec + t * b
                cand = proj_simplex(cand)
                cand_J = compute_J(cand, n, fft_len, conv_len)
                if cand_J < best_J - 1e-12:
                    best_J = cand_J
                    best_vec = cand.copy()
                    cur_vec = cand.copy()
                    cur_J = cand_J
                    last_improve = time.monotonic()
                    print(f"[{time.monotonic() - start_time:6.2f}s] LP‑mix improved J = {best_J:.6f}")
            lp_cooldown = now

        # ----------------------------------------------------------
        # 4.5  Stagnation handling – jitter around the best solution
        # ----------------------------------------------------------
        if now - last_improve > 40.0:
            # apply a gentle log‑normal jitter
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = compute_J(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter reset – current J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 5. Final sanitisation (clip to allowed range, renormalise)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
