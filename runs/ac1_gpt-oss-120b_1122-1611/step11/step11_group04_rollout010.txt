        # shift for numerical stability

            w_sum = 1.0

        a_rev = a[::-1]

        grad_a = 2.0 * conv_wa_rev[n-1 : n-1 + n]  # length n

        grad_p = a * (grad_a - dot)

        v_hat = v / (1 - b2 ** (iteration + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step
        # Occasionally add jitter if stagnation detected
        if iteration % 2000 == 0 and iteration > 0:

            # naive: use last best improvement time
            if best_time < (time.monotonic() - start_time) - 30:

                noise = rng.normal(0.0, 0.1, size=n)
                p = p + noise
        # increase beta slowly
        beta = min(beta + beta_inc, 50.0)
        iteration += 1
    # Return best vector
    if best_a is None:
        # fallback uniform
        best_a = np.full(n, 1.0/n)
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0/n
    else:
        best_a /= total
    return [float(v) for v in best_a]

"""
Hybrid projected‑gradient optimizer for the step‑function problem.

The objective (for a vector normalised to sum = 1) is

    J(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full linear convolution of a with itself.
This code minimises J by iteratively

* evaluating the convolution with a real FFT (O(n log n)),
* selecting an “active” set of convolution entries that are close to the
  current maximum,
* computing the sub‑gradient of the max‑convolution on that active set,
* taking an additive step opposite to the gradient,
* projecting the result back onto the probability simplex (∑a_i = 1, a_i≥0).

A simple temperature‑like schedule (decreasing step size, shrinking
tolerance) and occasional random “jitter” moves help escape plateaus.
The routine keeps the best vector seen so far and returns it before the
budget expires.  All numbers are clipped to the range accepted by
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT of length ``fft_len``."""
    A = np.fft.rfft(a, n=fft_len)
    c = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return c


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x≥0, sum x=1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:        # numerical safety
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _active_set(conv: np.ndarray, rel_tol: float, abs_tol: float = 1e-14) -> np.ndarray:
    """Indices where conv is within the tolerance of its maximum."""
    max_val = conv.max()
    tol = max(abs_tol, max_val * rel_tol)
    return np.where(conv >= max_val - tol)[0]


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function runs at most ``budget_s`` seconds
    and returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            # keep the size reasonable for the optimisation loop
            n = max(256, min(n, 5000))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # for a*a convolution

    # ------------------------------------------------------------------
    # 2. Initialise the probability vector.
    # ------------------------------------------------------------------
    # Start from the previous construction, if it exists, otherwise from a
    # random Dirichlet draw (moderate concentration) to break symmetry.
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.interp(
                np.linspace(0, len(a0) - 1, n),
                np.arange(len(a0)),
                a0,
            )
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
    else:
        a0 = rng.dirichlet(np.full(n, 0.6))

    cur_a = a0.copy()

    # ------------------------------------------------------------------
    # 3. Book‑keeping.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # tiny safety margin

    best_a = cur_a.copy()
    # J = 2 * n * max(conv) because cur_a sums to 1
    conv = _conv_full_fft(cur_a, fft_len, conv_len)
    best_J = float(2.0 * n * conv.max())
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # Hyper‑parameters (tuned empirically)
    base_lr = 0.25                     # initial additive step size
    min_lr = 5e-4                      # lower bound for the step size
    decay = 0.0005                     # linear decay per iteration
    rel_tol_start = 0.02               # active‑set tolerance start
    rel_tol_end = 0.001                # final active‑set tolerance
    jitter_prob = 0.005                # probability of a jitter move each iteration
    jitter_scale = 0.2                 # standard deviation of multiplicative Gaussian jitter

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 4. Evaluate current state.
        # --------------------------------------------------------------
        conv = _conv_full_fft(cur_a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # --------------------------------------------------------------
        # 5. Record improvement.
        # --------------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = cur_a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 6. Stagnation handling – occasional jitter.
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > 30.0 and rng.random() < jitter_prob:
            # multiplicative Gaussian noise centred at 1.0
            noise = np.exp(rng.normal(0.0, jitter_scale, size=n))
            cur_a = cur_a * noise
            cur_a = _project_onto_simplex(cur_a)
            last_improve = time.monotonic()
            continue

        # --------------------------------------------------------------
        # 7. Build the active set (indices close to the current maximum).
        # --------------------------------------------------------------
        # The tolerance is shrunk linearly with the iteration count.
        rel_tol = max(rel_tol_end,
                      rel_tol_start - (iteration / 20000.0) *
                      (rel_tol_start - rel_tol_end))
        active_idx = _active_set(conv, rel_tol)

        # --------------------------------------------------------------
        # 8. Sub‑gradient of max(conv) on the active set.
        # --------------------------------------------------------------
        grad = np.zeros_like(cur_a)
        # For each active convolution index p we add 2*a[p‑i] to grad[i]
        for p in active_idx:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=np.int64)
            # cur_a[p-idx] is exactly the slice of a that pairs with idx
            grad[idx] += 2.0 * cur_a[p - idx]

        # --------------------------------------------------------------
        # 9. Additive update with simplex projection.
        # --------------------------------------------------------------
        # Step size decays slowly; never go below ``min_lr``.
        lr = max(min_lr, base_lr / (1.0 + decay * iteration))

        # Move opposite to the gradient (we want to reduce the peak).
        proposal = cur_a - lr * grad
        proposal = _project_onto_simplex(proposal)

        # --------------------------------------------------------------
        # 10. Accept the proposal (always – we already projected).
        # --------------------------------------------------------------
        cur_a = proposal

        iteration += 1

    # ------------------------------------------------------------------
    # 11. Return the best discovered sequence (clipped to evaluator limits).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    # Optional – final reporting (commented out to keep output tidy)
    # final_conv = _conv_full_fft(best_a, fft_len, conv_len)
    # final_J = float(2.0 * n * final_conv.max())
    # print(f"Final best J = {final_J:.6f}")

    return [float(v) for v in best_a]
