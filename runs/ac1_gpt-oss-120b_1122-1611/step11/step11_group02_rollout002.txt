"""
Propose a candidate step‑function that minimizes
    J(a) = 2·n·max(conv(a,a))   with a_i ≥ 0,  Σ a_i = 1.

The implementation combines three complementary phases:

1. **Diverse initialisation** – a portfolio of analytic shapes (uniform,
   geometric, ramps, Gaussians, U‑shaped, bimodal, Dirichlet draws and the
   previous best if available). The best of these is taken as the start.

2. **Smoothed mirror descent** – an exponentiated‑gradient update on the
   probability simplex using a soft‑max surrogate of the sup‑norm.
   The temperature *p* for the soft‑max is increased linearly from 1 to 200
   and the step size decays as η ∝ 1/√t.  When the iterate stagnates for
   >15 s a small multiplicative jitter is applied.

3. **Stochastic pairwise mass‑transfer search** – random moves that shift a
   small amount of probability mass from one index to another while keeping
   the total sum = 1.  A simulated‑annealing acceptance rule with a slowly
   cooling temperature allows occasional uphill steps and helps escape
   local minima.  The step magnitude and temperature both decay
   logarithmically.

A short fine‑grained random‑noise phase at the end gives the final polish.
All operations use FFT‑based convolution (O(n log n)) so the routine can
perform millions of iterations within the 1000 s budget.

The returned list is already on the simplex, non‑negative and obeys the
caps enforced by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def compute_obj(a: np.ndarray, n: int,
                fft_len: int, conv_len: int) -> tuple[float, np.ndarray]:
    """
    Compute the objective J = 2·n·max(conv(a,a)) together with the full
    convolution vector (needed by the gradient routine).
    """
    conv = conv_fft(a, fft_len, conv_len)
    J = float(2.0 * n * conv.max())
    return J, conv


def gradient_smooth(a: np.ndarray, conv: np.ndarray,
                    p: float, fft_len_g: int,
                    conv_len: int) -> np.ndarray:
    """
    Gradient of the smoothed sup‑norm
        φ(a) = log‑sum‑exp(p·conv(a,a)) / p .
    The gradient has the closed form
        g = 2 * (w * a_rev)   (cross‑correlation)
    where   w_k = exp(p·conv_k) / Σ_j exp(p·conv_j) .
    """
    # Soft‑max weights (stable: subtract maximum)
    w = np.exp(p * (conv - conv.max()))

    # reverse a for the correlation
    a_rev = a[::-1]

    # Convolution w (length 2n‑1) with a_rev (length n)
    # Result length = (2n‑1)+n‑1 = 3n‑2.
    # Needed slice is [conv_len‑1 : conv_len‑1 + n] (central n entries).
    grad_full = np.fft.irfft(
        np.fft.rfft(w, fft_len_g) * np.fft.rfft(a_rev, fft_len_g),
        n=fft_len_g
    )
    grad = 2.0 * grad_full[conv_len - 1: conv_len - 1 + a.size]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the evaluation function. Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size.
    # ------------------------------------------------------------------
    # Use the length of a previously found construction if present,
    # otherwise fall back to a comfortable size.
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)          # remove negatives
            s = prev.sum()
            if s == 0.0:
                prev[:] = 1.0
                s = prev.sum()
            prev /= s                                 # normalise to sum‑1
            n = prev.size
            # clamp n to a reasonable interval (power‑of‑two not required)
            n = max(1024, min(n, 8192))
        except Exception:
            n = 2048
            prev = None
    else:
        n = 2048
        prev = None

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_g = _next_pow2(conv_len + n - 1)      # for gradient (w * a_rev)

    # ------------------------------------------------------------------
    # 2. Build a portfolio of initial probability vectors.
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # U‑shaped (concave) polynomials
    for p in [0.5, 1.0, 2.0]:
        v = (x * (n - 1 - x)) ** p
        v += 1e-12                       # avoid all‑zero
        v /= v.sum()
        init_cands.append(v)

    # Bimodal (two separated Gaussians)
    sigma = n / 30.0
    c1, c2 = n * 0.25, n * 0.75
    v = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    v /= v.sum()
    init_cands.append(v)

    # Dirichlet draws (moderately concentrated)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best (if present)
    if prev is not None:
        v = prev.copy()
        # if lengths differ, truncate or pad with zeros
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            v = proj_simplex(v)
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Select the best initial candidate (based on the true objective).
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J, _ = compute_obj(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Phase‑1: Smoothed mirror descent (exponentiated‑gradient).
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # keep a safety margin for the final polishing phase
    deadline = start_time + max(0.0, float(budget_s) - 50.0)

    # schedule for the soft‑max temperature
    p_start, p_end = 1.0, 200.0
    eta0 = 0.6  # base step size for the EG update

    cur_vec = best_vec.copy()
    cur_J = best_J
    best_vec_phase1 = best_vec.copy()
    best_J_phase1 = best_J
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)

        # temperature for soft‑max and jitter magnitude
        p = p_start + (p_end - p_start) * frac

        # full convolution and objective for the current iterate
        conv = conv_fft(cur_vec, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())

        # track global best
        if cur_J < best_J_phase1 - 1e-12:
            best_J_phase1 = cur_J
            best_vec_phase1 = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase‑1 new best J = {best_J_phase1:.6f}")

        # stagnation handling – jitter after 15 s without improvement
        if now - last_improve > 15.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = proj_simplex(best_vec_phase1 * noise)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase‑1 jitter – best J = {best_J_phase1:.6f}")

        # smoothed gradient of the surrogate sup‑norm
        grad = gradient_smooth(cur_vec, conv, p, fft_len_g, conv_len)

        # adaptive step size η ∝ 1/√t, clipped for numeric stability
        eta = eta0 / np.sqrt(iteration + 1.0)
        max_abs = np.max(np.abs(grad))
        if max_abs * eta > 30.0:
            eta = 30.0 / max_abs

        # exponentiated‑gradient proposal
        factor = np.exp(-eta * grad)
        proposal = cur_vec * factor
        proposal = proj_simplex(proposal)

        # evaluate proposal
        prop_conv = conv_fft(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # simple Metropolis acceptance (mostly greedy)
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # tiny uphill probability helps escape flat plateaus
            if rng.random() < 5e-7:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Phase‑2: Stochastic pairwise mass‑transfer search (simulated annealing).
    # ------------------------------------------------------------------
    cur_vec = best_vec_phase1.copy()
    cur_J = best_J_phase1
    best_vec = best_vec_phase1.copy()
    best_J = best_J_phase1
    start_phase2 = time.monotonic()
    deadline_phase2 = start_time + float(budget_s) - 5.0   # leave a few seconds for polishing

    # annealing schedule parameters
    T0 = 0.015                     # initial temperature
    T_end = 1e-6
    step0 = 0.12                   # initial fraction of mass moved
    step_end = 0.003

    it2 = 0
    while time.monotonic() < deadline_phase2:
        now = time.monotonic()
        elapsed = now - start_phase2
        total = deadline_phase2 - start_phase2
        frac = elapsed / total

        # temperature decreasing geometrically
        T = T0 * (T_end / T0) ** frac
        # step magnitude decreasing similarly
        step = step0 * (step_end / step0) ** frac

        # propose a random pairwise transfer
        i = rng.integers(n)
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)

        # the amount we can transfer is limited by the source mass
        max_transfer = cur_vec[i]
        if max_transfer <= 1e-12:
            continue
        delta = rng.random() * step * max_transfer

        cand = cur_vec.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate candidate (fast FFT)
        cand_J, _ = compute_obj(cand, n, fft_len, conv_len)

        # acceptance rule (Metropolis)
        if cand_J < cur_J:
            accept = True
        else:
            # probability ~ exp(-(Δ)/T)
            accept = rng.random() < np.exp(-(cand_J - cur_J) / T)

        if accept:
            cur_vec = cand
            cur_J = cand_J

            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{now - start_time:6.2f}s] Phase‑2 new best J = {best_J:.6f}")

        it2 += 1

    # ------------------------------------------------------------------
    # 6. Fine‑grained random jitter around the best solution.
    # ------------------------------------------------------------------
    fin_start = time.monotonic()
    fin_deadline = fin_start + min(30.0, (budget_s - (time.monotonic() - start_time)) * 0.1)

    while time.monotonic() < fin_deadline:
        noise = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_J, _ = compute_obj(cand, n, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            print(f"[Fine] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 7. Final normalisation and safety clipping.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
