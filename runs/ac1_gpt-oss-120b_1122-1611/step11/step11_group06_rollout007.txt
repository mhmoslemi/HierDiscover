"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence(sequence).  The search combines:

* A wide initial sweep over analytic families (geometric, power‑law,
  Tukey, Kaiser, Gaussian, Chebyshev, Dirichlet draws, etc.).
* A fast evaluation of the objective J = 2·n·max(conv(a,a)) / sum(a)^2
  using FFT based convolution and a Euclidean projection onto the
  probability simplex (∑a_i = 1).  Scaling is irrelevant for the
  objective as long as no entry exceeds 1000, therefore all candidates
  are kept normalised to unit sum.
* A refinement loop that alternates
  – an exponentiated‑gradient step using the sub‑gradient of the
    active (tight) convolution constraints,
  – a small “linear‑programming” step that solves for a direction g
    with max(conv(f,g)) ≤ current maximum, maximising the ℓ₁‑norm of g;
    mixing f and the scaled solution g provably reduces the bound.
  – occasional stochastic jitter and Dirichlet injections to escape
    plateaus, with a simple Metropolis acceptance rule and a slowly
    cooled temperature schedule.

The routine respects the 1000‑second time budget, prints progress,
and finally returns the best vector it found as a plain list of floats.
"""

import time
import numpy as np
from scipy import signal, optimize

# ----------------------------------------------------------------------
# Helper utilities – must be top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to one).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Maximum entry of the full (linear) self‑convolution of ``vec``.
    Computed with FFT for O(n log n) speed.
    """
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def objective_from_vec(vec: np.ndarray, n: int,
                       fft_len: int, conv_len: int) -> float:
    """
    Return the evaluation objective J = 2·n·max(conv(vec,vec)).
    The vector is assumed to be normalised to sum = 1,
    therefore the denominator in ``evaluate_sequence`` is 1.
    """
    return 2.0 * n * full_conv_max(vec, fft_len, conv_len)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found
    within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size – use the length of an existing construction
    #    if present, otherwise fall back to a moderate size.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Helper to evaluate a candidate and maintain the global best.
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    start_time = time.monotonic()

    def evaluate_and_update(arr: np.ndarray) -> None:
        """Normalise ``arr`` to a probability vector, compute J and update best."""
        nonlocal best_vec, best_J
        arr = np.maximum(arr, 0.0)
        s = arr.sum()
        if s == 0.0:
            return
        arr = arr / s
        J = objective_from_vec(arr, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = arr.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Broad initial portfolio (analytic families + random draws).
    # --------------------------------------------------------------

    # uniform
    evaluate_and_update(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (including values extremely close to 1)
    for r in np.logspace(np.log10(0.8), np.log10(0.9999), num=30):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        evaluate_and_update(v)
        evaluate_and_update(v[::-1])

    # power‑law shapes
    for p in np.linspace(0.05, 5.0, 30):
        i = np.arange(1, n + 1, dtype=np.float64)
        v = 1.0 / (i ** p)
        v /= v.sum()
        evaluate_and_update(v)

    # Tukey windows (dense sweep)
    for alpha in np.linspace(0.0, 1.0, 51):
        w = signal.tukey(n, alpha)
        if w.sum() > 0.0:
            evaluate_and_update(w / w.sum())

    # Kaiser windows (dense sweep over beta)
    for beta in np.linspace(0.0, 20.0, 41):
        w = signal.kaiser(n, beta)
        if w.sum() > 0.0:
            evaluate_and_update(w / w.sum())

    # Gaussian windows – sigma expressed as fraction of the length
    for sigma_frac in np.linspace(0.05, 0.5, 41):
        sigma = sigma_frac * n
        w = signal.gaussian(n, std=sigma, sym=True)
        if w.sum() > 0.0:
            evaluate_and_update(w / w.sum())

    # Chebyshev windows – attenuation in dB
    for att in np.linspace(10, 150, 30):
        w = signal.chebwin(n, at=att)
        if w.sum() > 0.0:
            evaluate_and_update(w / w.sum())

    # Classic windows
    for w in (signal.hann(n), signal.hamming(n), signal.blackman(n),
              signal.bartlett(n), signal.parzen(n), signal.triang(n)):
        if w.sum() > 0.0:
            evaluate_and_update(w / w.sum())

    # Random Dirichlet draws with a variety of concentration parameters
    for conc in [0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0]:
        for _ in range(15):
            evaluate_and_update(rng.dirichlet(np.full(n, conc)))

    # Exponential edge blends (uniform + symmetric exponential)
    for _ in range(80):
        alpha = 10 ** rng.uniform(-3.0, 1.0)          # 0.001 … 10
        mix = rng.random()
        exp_vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
        exp_vec += np.exp(-alpha * (n - 1 - np.arange(n, dtype=np.float64)))
        exp_vec /= exp_vec.sum()
        v = (1.0 - mix) * np.full(n, 1.0 / n) + mix * exp_vec
        v /= v.sum()
        evaluate_and_update(v)

    # previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v = v / v.sum()
        evaluate_and_update(v)

    print(f"[{time.monotonic() - start_time:6.2f}s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Local refinement – exponentiated‑gradient + occasional LP step.
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J

    # temperature schedule (linear cooling)
    T_start = max(best_J * 0.2, 1e-4)
    T_end = 1e-8

    iteration = 0
    last_improve = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a small safety margin

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear temperature schedule
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * elapsed / (deadline - start_time))

        # ----------------------------------------------------------
        # 4.1 Full convolution of the current vector
        # ----------------------------------------------------------
        conv_vals = np.fft.irfft(
            np.fft.rfft(cur_vec, n=fft_len) ** 2,
            n=fft_len)[:conv_len]
        max_c = conv_vals.max()
        cur_J = 2.0 * n * max_c

        # record a new global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4.2 Stagnation handling – jitter around the best solution.
        # ----------------------------------------------------------
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = best_vec * noise
            cur_vec /= cur_vec.sum()
            last_improve = now
            continue

        # ----------------------------------------------------------
        # 4.3 Sub‑gradient of max(conv) at the active set.
        # ----------------------------------------------------------
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv_vals >= max_c - tol)[0]

        grad = np.zeros_like(cur_vec)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=np.int64)
            # contribution of each a_i to the active maximal convolution term
            grad[idx] += 2.0 * cur_vec[p - idx]

        # ----------------------------------------------------------
        # 4.4 Propose a multiplicative (exponentiated‑gradient) update.
        # ----------------------------------------------------------
        step = 0.7 / np.sqrt(iteration + 1.0)   # slightly larger initial steps
        factor = np.exp(-step * grad)
        proposal = cur_vec * factor

        # occasional Dirichlet injection to increase diversity
        if rng.random() < 0.12:
            eps = rng.uniform(0.0, 0.25)
            rand_vec = rng.dirichlet(np.ones(n))
            proposal = (1.0 - eps) * proposal + eps * rand_vec

        proposal = proj_simplex(proposal)

        # ----------------------------------------------------------
        # 4.5 Evaluate the proposal.
        # ----------------------------------------------------------
        prop_J = objective_from_vec(proposal, n, fft_len, conv_len)

        # ----------------------------------------------------------
        # 4.6 Metropolis acceptance.
        # ----------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ----------------------------------------------------------
        # 4.7 Periodic LP step (successive linearisation).
        # ----------------------------------------------------------
        if iteration % 250 == 0 and active.size > 0:
            rows = active.size
            A = np.zeros((rows, n), dtype=np.float64)
            for r, k in enumerate(active):
                start = max(0, k - (n - 1))
                end = min(k, n - 1)
                # coefficients are a_{k-j}
                A[r, start:end + 1] = cur_vec[k - np.arange(start, end + 1)]
            b = np.full(rows, max_c, dtype=np.float64)
            c_obj = -np.ones(n)                 # maximise sum(g)  → minimise -sum(g)
            bounds = [(0, None)] * n
            # solve with HiGHS – efficient for the very sparse problem
            res = optimize.linprog(c_obj, A_ub=A, b_ub=b,
                                  bounds=bounds, method='highs',
                                  options={'presolve': True})
            if res.success:
                g0 = res.x
                sum_g0 = g0.sum()
                if sum_g0 > 0.0:
                    # scale g0 to have the same ℓ₁‑norm as the current vector
                    g_scaled = g0 * (cur_vec.sum() / sum_g0)
                    # mix a modest fraction of the LP direction
                    mix = 0.2
                    new_vec = (1.0 - mix) * cur_vec + mix * g_scaled
                    new_vec = proj_simplex(new_vec)
                    new_J = objective_from_vec(new_vec, n, fft_len, conv_len)
                    if new_J < cur_J:
                        cur_vec = new_vec
                        cur_J = new_J

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution found (clip, renormalise, cast).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
