    a_fft = np.fft.rfft(a, fft_len)

def propose_candidate(seed=12345, budget_s=1000.0, **kwargs)->list[float]:
    rng = np.random.default_rng(seed)
    # Determine n
    try:
        n = len(globals()["height_sequence_1"])
        n = max(256, min(n, 8192))
    except Exception:
        n = 4096

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    corr_len = _next_pow2(conv_len + n - 1)

    # function evaluate_J
    def eval_J(a):
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        m = conv.max()
        s = a.sum()
        if s < 0.01:
            return np.inf, m, conv
        J = 2.0 * n * m / (s * s)
        return J, m, conv

    # initialize best
    # start with uniform
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_J, best_m, _ = eval_J(best_a)

    # optionally incorporate previous sequence as starting candidate
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s > 0:
            prev = prev / s
        else:
            prev = best_a.copy()
        prev_J, _, _ = eval_J(prev)
        if prev_J < best_J:
            best_J, best_a = prev_J, prev.copy()
    # more initial candidates via Dirichlet draws
    alphas = [0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for alpha in alphas:
        cand = rng.dirichlet(np.full(n, alpha))
        cand_J, _, _ = eval_J(cand)
        if cand_J < best_J:
            best_J, best_a = cand_J, cand.copy()
    # print initial best
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    total_time = deadline - start_time

    # schedule parameters
    beta_start = 0.5
    beta_end = 2000.0
    base_lr = 0.2  # learning rate base
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    cur_a = best_a.copy()
    cur_J = best_J
    cur_m = best_m
    # compute a random small jitter and track improvement
    last_improve = start_time
    iteration = 0
    # for Adam
    m_m = np.zeros_like(cur_a)
    v_m = np.zeros_like(cur_a)
    # main loop
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # evaluate current a if needed
        cur_J, cur_max, conv = eval_J(cur_a)
        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = cur_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # handle stagnation
        if now - last_improve > 30.0:
            # random jitter
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            cur_a = best_a * noise
            cur_a = proj_simplex(cur_a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # schedule beta
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total_time)
        # compute softmax weights
        # avoid overflow: subtract max before exp
        exp_beta = np.exp(beta * (conv - cur_max))
        w = exp_beta / exp_beta.sum()
        # compute gradient via weighted correlation
        grad = _weighted_gradient(cur_a, w, corr_len, n)  # returns gradient vector (2*corr)
        # Adam update
        iteration += 1
        t_iter = iteration
        m_m = beta1 * m_m + (1 - beta1) * grad
        v_m = beta2 * v_m + (1 - beta2) * (grad ** 2)
        m_hat = m_m / (1 - beta1 ** t_iter)
        v_hat = v_m / (1 - beta2 ** t_iter)
        lr = base_lr / np.sqrt(t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        new_a = cur_a - step
        # ensure non-negativity before projection
        new_a = np.maximum(new_a, 0.0)
        new_a = proj_simplex(new_a)

        # evaluate proposal
        new_J, _, _ = eval_J(new_a)
        # accept if better OR with some probability (simulated annealing)
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            # temperature schedule (linear)
            T_start = best_J * 0.02
            T_end = 1e-6
            T = (T_start * (deadline - now) + T_end * (now - start_time)) / total_time
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_a = new_a
            cur_J = new_J

    # final return
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total
    return [float(v) for v in best_a]

"""
Propose a step‑function (non‑negative list) that minimizes the given
`evaluate_sequence` objective.

Algorithm
---------

1. **Problem size** – Use the length of the previous solution if
   available, otherwise pick a power‑of‑two size (`n = 4096`) that
   keeps FFTs cheap.

2. **Fast objective** – The true objective is
   J(a) = 2·n·max(conv(a,a)) / (sum(a)²).  A fast version is
   implemented with real FFTs and a Euclidean simplex projection
   (∑a = 1, a_i ≥ 0).  The scaling‐invariance of the objective lets us
   keep the sum fixed to 1 during optimisation.

3. **Exploration phase (≈30 % of the time)** – Sample many candidates
   from diverse Dirichlet distributions, a few sparse “block‑sparse”
   patterns and the previous best solution.  Keep the best few.

4. **Refinement phase** – Starting from the current best vector perform
   a projected Adam optimiser on the smooth approximation of the max
   (soft‑max with a temperature β that grows from 0.5 to 2000).  The
   gradient of the smoothed max is obtained via a weighted correlation
   (FFT‑based) and the update is projected back onto the simplex.
   An annealed Metropolis step occasionally accepts uphill moves.

5. **Stagnation handling** – If no improvement for 30 s a multiplicative
   log‑normal jitter is applied to the best vector to jump out of flat
   regions.

6. **Return** – Clip to the allowed range, renormalise and output as a
   plain Python list.

The routine respects the 1000 s budget, prints progress, and never
uses closures or lambda expressions.  Only NumPy and the standard
library are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear (full) convolution a*a via a real FFT.
    Returns the first ``conv_len`` entries (the linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return c


def evaluate_J(a: np.ndarray, fft_len: int, conv_len: int, n: int):
    """
    Fast version of ``evaluate_sequence`` (no security checks).
    Returns (objective J, maximal convolution value, full convolution array).
    """
    conv = conv_full(a, fft_len, conv_len)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf, max_conv, conv
    J = float(2.0 * n * max_conv / (s * s))
    return J, max_conv, conv


def weighted_corr(a: np.ndarray, weight: np.ndarray,
                  corr_len: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently using FFT.
    ``weight`` has length 2·n‑1, ``a`` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, corr_len)
    A_rev_hat = np.fft.rfft(a_rev, corr_len)
    full = np.fft.irfft(W_hat * A_rev_hat, n=corr_len)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  Returns the best sequence found within the time
    budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (power of two, not too small, not too large)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                        # for a*a
    corr_len = _next_pow2(conv_len + n - 1)               # for weighted correlation

    # --------------------------------------------------------------
    # 2. Initial diverse candidates (including the previous solution)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    # uniform
    cand = np.full(n, 1.0 / n, dtype=np.float64)
    J, _, _ = evaluate_J(cand, fft_len, conv_len, n)
    if J < best_J:
        best_J, best_vec = J, cand.copy()

    # previous solution (if any)
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s > 0.0:
            prev /= s
        else:
            prev = np.full(n, 1.0 / n, dtype=np.float64)
        J, _, _ = evaluate_J(prev, fft_len, conv_len, n)
        if J < best_J:
            best_J, best_vec = J, prev.copy()

    # Dirichlet draws with several concentration parameters
    alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for alpha in alphas:
        cand = rng.dirichlet(np.full(n, alpha))
        J, _, _ = evaluate_J(cand, fft_len, conv_len, n)
        if J < best_J:
            best_J, best_vec = J, cand.copy()

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    total_time = deadline - start_time

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.20

    # Soft‑max temperature schedule
    beta_start = 0.5
    beta_end = 2000.0

    # Metropolis temperature schedule (linear cooling)
    T_start = best_J * 0.03   # small fraction of current best
    T_end = 1e-8

    # State for the optimiser
    cur_vec = best_vec.copy()
    m_vec = np.zeros_like(cur_vec)   # first moment
    v_vec = np.zeros_like(cur_vec)   # second moment
    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # Exploration phase – random sampling (first ~30 % of the budget)
        # ----------------------------------------------------------
        if elapsed < 0.30 * total_time:
            # generate a small batch of random candidates
            for _ in range(12):
                # pick a random Dirichlet concentration
                alpha = rng.choice(alphas)
                cand = rng.dirichlet(np.full(n, alpha))
                Jcand, _, _ = evaluate_J(cand, fft_len, conv_len, n)
                if Jcand < best_J:
                    best_J, best_vec = Jcand, cand.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] New best (rand) J = {best_J:.6f}")

            # occasional sparse “block‑sparse” candidates
            if rng.random() < 0.05:
                k = rng.integers(2, min(50, n // 2))
                idx = rng.choice(n, size=k, replace=False)
                cand = np.zeros(n, dtype=np.float64)
                cand[idx] = 1.0 / k
                Jcand, _, _ = evaluate_J(cand, fft_len, conv_len, n)
                if Jcand < best_J:
                    best_J, best_vec = Jcand, cand.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] New best (sparse) J = {best_J:.6f}")

            # continue to next iteration (no gradient step yet)
            continue

        # ----------------------------------------------------------
        # Refinement phase – projected Adam optimiser with soft‑max
        # ----------------------------------------------------------

        # evaluate current vector (if not already up‑to‑date)
        cur_J, cur_max, cur_conv = evaluate_J(cur_vec, fft_len, conv_len, n)

        # global best update
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # stagnation handling – inject multiplicative jitter
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # Soft‑max weights (smooth approximation of the max)
        # ----------------------------------------------------------
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total_time)
        # shift by max for numerical stability
        exp_beta = np.exp(beta * (cur_conv - cur_max))
        w = exp_beta / exp_beta.sum()

        # ----------------------------------------------------------
        # Gradient via weighted correlation
        # ----------------------------------------------------------
        grad = weighted_corr(cur_vec, w, corr_len, n)   # contains the factor 2

        # ----------------------------------------------------------
        # Adam update (projected onto the simplex)
        # ----------------------------------------------------------
        iteration += 1
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)

        m_hat = m_vec / (1.0 - beta1 ** iteration)
        v_hat = v_vec / (1.0 - beta2 ** iteration)

        lr = base_lr / np.sqrt(iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        proposal = cur_vec - step
        proposal = np.maximum(proposal, 0.0)
        proposal = proj_simplex(proposal)

        prop_J, _, _ = evaluate_J(proposal, fft_len, conv_len, n)

        # ----------------------------------------------------------
        # Metropolis acceptance (linear temperature schedule)
        # ----------------------------------------------------------
        T = (T_start * (deadline - now) + T_end * (now - start_time)) / total_time
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

    # --------------------------------------------------------------
    # 5. Return best solution (clip and renormalise for safety)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
