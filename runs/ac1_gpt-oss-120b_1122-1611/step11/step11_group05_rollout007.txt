def _gradient_max(a, mask, fft_len_corr):
    # mask length conv_len (2n-1). But mask entries sum to 1, but can be any nonnegative weights.
    A = np.fft.rfft(a, fft_len_corr)
    W = np.fft.rfft(mask, fft_len_corr)
    corr = np.fft.irfft(np.conj(A) * W, n=fft_len_corr)
    n = a.size
    start = n - 1
    grad = 2.0 * corr[start:start+n]
    return grad

"""
Iterative mirror‑descent on the (soft) maximum of the auto‑convolution.

The routine builds a diverse portfolio of probability vectors (uniform,
geometric, ramps, Gaussians, Dirichlet draws, sparse masks and a
perturbed version of the previous best).  The best of those seeds is
selected and refined for the full time budget.

Two phases are used:

*  soft‑max objective with a temperature β that is increased
   (β up to 5000) – this gives a smooth surrogate of the L∞‑norm
   while preserving gradients.
*  once β hits the maximum the exact max‑gradient is used (weights are
   a uniform distribution over the indices where the convolution attains
   its current maximum).

Updates are performed with an exponentiated‑gradient step
   a←a·exp(−η·∇)  (mirrored‑descent on the simplex)  and the vector
is renormalised onto the probability simplex.
A linear cooling schedule provides a Metropolis acceptance rule that
allows occasional uphill moves; after 30 s without improvement a small
jitter is applied to escape plateaus.

All calculations use real FFTs (O(n log n)).  The algorithm works with a
larger problem size (up to 8192 entries) than the previous run,
giving the optimizer more degrees of freedom to flatten the
convolution and push the evaluation below the 1.5025 target.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (the full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights for a vector of convolutions."""
    shift = conv.max()
    exp_vals = np.exp(beta * (conv - shift))
    return exp_vals / exp_vals.sum()


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float,
                     fft_len_corr: int) -> np.ndarray:
    """
    Gradient of the smoothed‑max objective
        f(a) = (1/β)·log Σ exp(β·conv)
    with respect to ``a``.
    The gradient is 2·(w ⋆ a) where w are the soft‑max weights.
    """
    w = _softmax_weights(conv, beta)                 # shape (2n‑1,)
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    W_hat = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)   # linear correlation
    n = a.size
    start = n - 1
    grad = 2.0 * corr[start:start + n]
    return grad


def _gradient_max(a: np.ndarray, mask: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """
    Exact gradient of the L∞‑objective.
    ``mask`` contains non‑negative weights that sum to 1,
    e.g. a uniform distribution over the indices where the convolution
    attains its maximum.
    """
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    W_hat = np.fft.rfft(mask, L)
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)
    n = a.size
    start = n - 1
    grad = 2.0 * corr[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine – required entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  Returns the best vector found
    before the supplied time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – try to be larger than the previous run
    # ------------------------------------------------------------------
    # Upper limit (adjustable if memory becomes an issue)
    MAX_N = 8192

    if "height_sequence_1" in globals():
        try:
            prev_len = len(height_sequence_1)
            # aim for a size at least twice the previous one, but not beyond MAX_N
            n = min(max(prev_len * 2, 2048), MAX_N)
        except Exception:
            n = 4096
    else:
        n = 4096

    n = int(n)
    conv_len = 2 * n - 1
    fft_len_corr = _next_pow2(conv_len + n - 1)   # length for correlation / gradient
    # for convolution we can reuse the same length (no loss of accuracy)
    fft_len_conv = fft_len_corr

    # ------------------------------------------------------------------
    # 2. Build a diverse set of initial candidates (all are probability vectors)
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.linspace(1, 0, n, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.linspace(0, 1, n, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells (few widths)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 12.0, n / 9.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Arcsine‑shaped distribution (a_i ∝ sin²)
    v = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))) ** 2
    v /= v.sum()
    init_cands.append(v)

    # Dirichlet draws (moderately / strongly concentrated)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Sparse masks – a few equally weighted spikes
    for k in [int(np.sqrt(n)), int(np.sqrt(n) * 2), max(2, n // 10)]:
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        init_cands.append(mask)

    # Slightly perturbed uniform (adds a tiny amount of randomness)
    pert = np.full(n, 1.0 / n) + rng.normal(0.0, 0.001, size=n)
    pert = np.clip(pert, 0.0, None)
    pert /= pert.sum()
    init_cands.append(pert)

    # previous construction (if present) – normalise to a probability vector
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v = v / s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate using the exact objective
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len_conv, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # reserve a few seconds

    cur_vec = best_vec.copy()
    cur_J = best_J
    best_global = best_vec.copy()
    best_J_global = best_J
    last_improve = start_time

    # Temperature schedule for Metropolis acceptance (linear cooling)
    T_start = best_J * 0.05
    T_end = 1e-8

    # Soft‑max temperature schedule
    beta = 5.0
    beta_max = 5000.0
    beta_inc = 1.05          # multiplicative increase every few iterations

    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --- compute exact convolution and objective for the current point ---
        conv = _conv_full(cur_vec, fft_len_conv, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        # --- keep the best ever found ---
        if cur_J < best_J_global - 1e-12:
            best_J_global = cur_J
            best_global = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J_global:.6f}")

        # ------------------------------------------------------------------
        # Stagnation handling – jitter after 30 s without improvement
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_global * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J_global:.6f}")
            continue

        # ------------------------------------------------------------------
        # 4.1   Determine the weight vector for the gradient
        # ------------------------------------------------------------------
        if beta < beta_max:
            # Smoothed‑max (soft‑max) weights
            w = _softmax_weights(conv, beta)                # shape (conv_len,)
            grad = _gradient_softmax(cur_vec, conv, beta, fft_len_corr)
        else:
            # Exact max – uniform weight over all attaining indices
            mask = np.zeros_like(conv)
            idx = np.where(conv == max_conv)[0]
            mask[idx] = 1.0 / len(idx)
            grad = _gradient_max(cur_vec, mask, fft_len_corr)

        # ------------------------------------------------------------------
        # 4.2   Propose a new point – exponentiated‑gradient (mirror descent)
        # ------------------------------------------------------------------
        step = 0.5 / np.sqrt(iteration + 1.0)          # decaying step size
        exp_arg = -step * grad
        # keep the exponent numerically stable
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        factor = np.exp(exp_arg)
        proposal = cur_vec * factor
        proposal = np.clip(proposal, 0.0, None)
        s = proposal.sum()
        if s == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= s

        # ------------------------------------------------------------------
        # 4.3   Evaluate the proposal (exact J)
        # ------------------------------------------------------------------
        prop_conv = _conv_full(proposal, fft_len_conv, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ------------------------------------------------------------------
        # 4.4   Metropolis acceptance (linear cooling)
        # ------------------------------------------------------------------
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * (elapsed / (deadline - start_time)))
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ------------------------------------------------------------------
        # 4.5   Update the soft‑max temperature β (slowly increase)
        # ------------------------------------------------------------------
        if iteration % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        iteration += 1

    # ----------------------------------------------------------------------
    # 5. Optional lightweight local refinement (pairwise mass transfers)
    # ----------------------------------------------------------------------
    # a few random transfers that may squeeze a tiny extra improvement
    for _ in range(2000):
        i, j = rng.choice(n, size=2, replace=False)
        # move a small amount from the higher‑gradient index to the lower‑gradient one
        delta = rng.uniform(0.0, min(cur_vec[i], 0.001))
        if delta <= 0.0:
            continue
        cand = best_global.copy()
        cand[i] -= delta
        cand[j] += delta
        # the sum stays exactly 1, no need to renormalise
        cand_conv = _conv_full(cand, fft_len_conv, conv_len)
        cand_J = float(2.0 * n * cand_conv.max())
        if cand_J < best_J_global - 1e-12:
            best_J_global = cand_J
            best_global = cand

    # ----------------------------------------------------------------------
    # 6. Return the best found vector (clip and renormalise for safety)
    # ----------------------------------------------------------------------
    best_global = np.clip(best_global, 0.0, 1000.0)
    total = best_global.sum()
    if total == 0.0:
        best_global[:] = 1.0 / n
    else:
        best_global /= total

    return [float(v) for v in best_global]
