```
def project_to_simplex(v):
    # v is np array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid gradient / linear‑programming optimizer for the step‑function problem.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0).

* Exponentiated‑gradient steps driven by a sub‑gradient of the
  max‑convolution (active set enlarged by a relative tolerance).
* Periodic exact LP step: maximise ∑b_i subject to the current
  convolution bound f∗b ≤ M.  The LP solution supplies a direction g
  that is then blended with the current iterate by a line‑search
  over the mixing parameter t∈[0,0.5].
* Simple simulated‑annealing style temperature for occasional uphill
  moves and a jitter restart after prolonged stagnation.
* All operations are performed with FFT‑based convolutions for speed.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a∗a (or a∗b) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _conv_two(a: np.ndarray, b: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Convolution a∗b using real FFT (both real, same length)."""
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:conv_len]
    return conv


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix K with
    K[p,i] = f[p‑i]  when 0≤p‑i<n else 0.
    This matrix encodes the linear map b → f∗b.
    """
    n = f.size
    conv_len = 2 * n - 1
    # each diagonal d (offset d≥0) contains the constant value f[d]
    diagonals = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)               # 0,…,n‑1
    K = sp.diags(diagonals, offsets, shape=(conv_len, n), format="csc")
    return K


def _solve_lp(f: np.ndarray, M: float, max_time: float = 5.0):
    """
    Solve   max ∑b_i
    s.t.    f∗b ≤ M   (componentwise)
            b_i ≥ 0
    Returns the optimal b (numpy array) or None on failure/timeout.
    """
    n = f.size
    K = _build_conv_matrix(f)                 # sparse (2n‑1)×n
    c = -np.ones(n)                           # minimise -∑b_i  → maximise ∑b_i
    bounds = [(0, None)] * n
    ub = np.full(2 * n - 1, M)

    # linprog may spend a long time on a difficult LP; we give it a modest
    # time limit via the `options` dict (highs supports `time_limit` in seconds).
    try:
        res = opt.linprog(
            c,
            A_ub=K,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": max_time, "presolve": True},
        )
    except Exception:          # any unexpected error – treat as failure
        return None

    if res.success and res.x is not None:
        return res.x
    return None


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size (length of the sequence)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # FFT length for convolutions

    # ------------------------------------------------------------------
    # 2. Build a set of basis vectors (each already normalised to sum‑1)
    # ------------------------------------------------------------------
    basis = []

    # uniform
    basis.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.98, 0.985, 0.99, 0.995, 0.999]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        basis.append(v)

    # linear ramps
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    basis.append(v)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    basis.append(v)

    # Gaussian bells (different widths)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0, n / 4.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        basis.append(v)

    # half‑uniform (mass concentrated on left/right half)
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 1.0 / half
    basis.append(v / v.sum())
    v = np.zeros(n, dtype=np.float64)
    v[half:] = 1.0 / (n - half)
    basis.append(v / v.sum())

    # previous best sequence (if available)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        basis.append(v)

    B = np.column_stack(basis)          # shape (n, k)
    k = B.shape[1]

    # ------------------------------------------------------------------
    # 3. Initialise the current vector (start from the previous best)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cur_vec = np.asarray(height_sequence_1, dtype=np.float64)
        cur_vec = np.clip(cur_vec, 0.0, None)
        s = cur_vec.sum()
        cur_vec = cur_vec / s if s > 0 else np.full(n, 1.0 / n)
    else:
        cur_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # best‑so‑far tracker
    best_vec = cur_vec.copy()
    conv_cur = _conv_full(cur_vec, fft_len, conv_len)
    best_J = float(2.0 * n * conv_cur.max())
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a small margin
    iteration = 0
    last_improve = start_time
    T_start = best_J * 0.15          # initial temperature (15 % of current J)
    T_end = 1e-9
    lp_calls = 0
    max_lp_calls = 12               # limit the number of LP solves

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- current objective ------------------------------------------------
        conv_cur = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv_cur.max()
        cur_J = float(2.0 * n * max_conv)

        # ---- update global best -----------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- stagnation handling ------------------------------------------------
        if now - last_improve > 22.0:           # restart after ~22 s idle
            noise = np.exp(rng.normal(0.0, 0.18, size=n))
            cur_vec = best_vec * noise
            cur_vec = _project_to_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ---- adaptive active‑set tolerance ------------------------------------
        if iteration < 3000:
            rel_tol = 0.015      # 1.5 % of peak early on
        elif iteration < 15000:
            rel_tol = 0.008
        else:
            rel_tol = 0.004
        tol = max(1e-14, max_conv * rel_tol)
        active = np.where(conv_cur >= max_conv - tol)[0]

        # ---- sub‑gradient of the max‑convolution (active set version) ---------
        grad = np.zeros_like(cur_vec)
        # loop over active indices – cheap because active set is small
        for p in active:
            start = max(0, p - (n - 1))
            end = min(n - 1, p)
            idx = np.arange(start, end + 1, dtype=int)
            grad[idx] += 2.0 * cur_vec[p - idx]

        # ---- propose a candidate ------------------------------------------------
        # with probability 0.72 use exponentiated‑gradient,
        # otherwise perform a random mass‑transfer move.
        if rng.random() < 0.72:
            # exponentiated‑gradient step
            step = 0.75 / np.sqrt(iteration + 1.0)   # aggressive early, then decay
            factor = np.exp(-step * grad)
            proposal = cur_vec * factor
            proposal = _project_to_simplex(proposal)   # renormalise to sum 1
        else:
            # cheap random transfer between two indices guided by the gradient
            if grad.sum() > 0.0:
                donor = rng.choice(n, p=grad / grad.sum())
            else:
                donor = rng.integers(n)
            inv = grad.max() - grad + 1e-12
            receiver = rng.choice(n, p=inv / inv.sum())
            delta = min(cur_vec[donor], 0.07 * cur_vec[donor])   # up to 7%
            proposal = cur_vec.copy()
            proposal[donor] -= delta
            proposal[receiver] += delta
            # projection not needed – sum unchanged and all entries stay ≥0

        # ---- evaluate the proposal ---------------------------------------------
        conv_prop = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * conv_prop.max())

        # ---- Metropolis acceptance -----------------------------------------------
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal

        # -----------------------------------------------------------------------
        # Periodic LP‑based tightening step
        # -----------------------------------------------------------------------
        if (iteration % 3500 == 0) and (lp_calls < max_lp_calls) and (now - last_improve > 8.0):
            # solve the LP for the current iterate
            M = max_conv
            b_opt = _solve_lp(cur_vec, M, max_time=4.5)
            if b_opt is not None and b_opt.sum() > 0.0:
                g_vec = b_opt / b_opt.sum()                # normalize to sum = 1
                # compute the three convolutions needed for a line‑search
                conv_fg = _conv_two(cur_vec, g_vec, fft_len, conv_len)
                conv_gg = _conv_full(g_vec, fft_len, conv_len)

                # helper arrays for the quadratic interpolation in t
                d1 = 2.0 * (conv_fg - conv_cur)                # linear term coefficient
                d2 = conv_gg - 2.0 * conv_fg + conv_cur        # quadratic term coefficient

                # try a decent set of mixing parameters
                t_vals = np.linspace(0.0, 0.5, 51)              # 0 … 0.5 in steps of 0.01
                max_vals = conv_cur + np.outer(t_vals, d1) + np.outer(t_vals ** 2, d2)
                # max over convolution entries, then over t
                max_over_t = max_vals.max(axis=1)
                best_idx = np.argmin(max_over_t)
                best_t = t_vals[best_idx]
                new_J = float(2.0 * n * max_over_t[best_idx])

                if new_J < cur_J - 1e-12:                     # accept only genuine improvement
                    # convex combination f←(1‑t)f + t g
                    cur_vec = (1.0 - best_t) * cur_vec + best_t * g_vec
                    cur_vec = _project_to_simplex(cur_vec)    # safeguard against numerical drift
                    lp_calls += 1
                    print(f"[{elapsed:6.2f}s] LP step #{lp_calls} – t={best_t:.3f}, J={new_J:.6f}")

        iteration += 1

    # ----------------------------------------------------------------------
    # 5. Return the best solution found (clip to allowed range and renorm)
    # ----------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"Finished after {time.monotonic() - start_time:.1f}s – best J = {best_J:.6f}")
    return [float(v) for v in best_vec]
