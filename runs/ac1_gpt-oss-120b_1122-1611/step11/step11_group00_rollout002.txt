"""
Propose a step‑function minimizing `evaluate_sequence` by
1. a small linear‑programming (LP) phase that repeatedly mixes the
   current distribution with the maximally feasible one (maximising the L1‑norm
   under the current convolution bound);
2. a fast simulated‑annealing / hill‑climbing refinement that transfers tiny
   amounts of probability mass between two positions.
Both phases respect the simplex (non‑negative, sum = 1) and use FFTs for
convolution, so each objective evaluation is O(n log n).  The routine adapts
its work to the given time budget (default 1000 s) and prints progress.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using real FFT (output length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The function runs for at most `budget_s` seconds and returns the best
    sequence discovered (as a list of floats).  The algorithm consists of a
    linear‑programming mixing phase followed by a simulated‑annealing fine‑tune.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a moderate problem size (keep memory & FFT cheap)
    # ------------------------------------------------------------------
    # Use the length of a previously found construction if available,
    # otherwise fall back to a default of 2048.  Clamp to a safe upper bound.
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
        except Exception:
            n = 2048
    else:
        n = 2048
    n = max(256, min(n, 2048))          # keep n ≤ 2048 for memory safety

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Initialise a probability vector a on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.concatenate([a0,
                                      np.full(n - a0.size, 0.0)])
        a0 = np.clip(a0, 0.0, None)
    else:
        a0 = np.full(n, 0.0)

    if a0.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a0 / a0.sum()

    # ------------------------------------------------------------------
    # 3. Helper for evaluating the objective J = 2·n·max(conv(a,a))
    # ------------------------------------------------------------------
    def _objective(vec: np.ndarray) -> float:
        conv = _conv_full(vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    def _max_conv(vec: np.ndarray) -> float:
        conv = _conv_full(vec, fft_len, conv_len)
        return float(conv.max())

    cur_J = _objective(a)
    best_a = a.copy()
    best_J = cur_J

    start_time = time.monotonic()
    deadline = start_time + budget_s - 1.0   # leave a second margin

    # ------------------------------------------------------------------
    # 4. Phase 1 – LP based mixing (aggressive descent)
    # ------------------------------------------------------------------
    # Allocate roughly 30 % of the budget to the LP phase.
    lp_deadline = start_time + min(300.0, budget_s * 0.30)
    t_mix = 0.5                     # convex combination weight
    lp_iter = 0

    # Pre‑allocate the constraint matrix (re‑used each iteration)
    A_ub = np.zeros((conv_len, n), dtype=np.float64)

    while time.monotonic() < lp_deadline:
        M = _max_conv(a)                     # current convolution bound

        # Build the matrix A_ub where row k, column j gets coefficient a[k‑j]
        # (zero when index out of range).  This is effectively the convolution
        # operator f * b ≤ M written as linear constraints in b.
        # The simple column‑wise filling is fast and memory‑friendly.
        for j in range(n):
            A_ub[j:j + n, j] = a

        # Solve: maximise sum(b)  ⇔ minimise -sum(b)  under A_ub b ≤ M, b ≥ 0
        c = -np.ones(n, dtype=np.float64)    # objective for minimise
        b_ub = np.full(conv_len, M, dtype=np.float64)

        res = optimize.linprog(c,
                               A_ub=A_ub,
                               b_ub=b_ub,
                               bounds=(0, None),
                               method='highs',
                               options={'presolve': True, 'disp': False})

        if not res.success:
            # If the solver fails, break early – we will continue with SA.
            print(f"[LP] iteration {lp_iter}: linprog failed – {res.message}")
            break

        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 <= 0.0:
            break

        # Scale to keep the sum equal to 1 (preserves the convolution bound)
        g = b0 / sum_b0

        # Mix the new vector into the current one
        a = (1.0 - t_mix) * a + t_mix * g

        # Evaluate new objective
        cur_J = _objective(a)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] LP iter {lp_iter} new best J = {best_J:.6f}")

        lp_iter += 1

    # ------------------------------------------------------------------
    # 5. Phase 2 – Simulated annealing / hill climbing fine‑tuning
    # ------------------------------------------------------------------
    # Parameters for the annealing schedule
    T0 = 0.015                     # initial temperature (controls uphill moves)
    T = T0
    decay = 0.9995                 # exponential decay per iteration
    move_scale = 0.02              # max fraction of mass that can be moved

    # Track last improvement time to trigger occasional jitter restarts
    last_improve = time.monotonic()
    cur_J = _objective(a)

    while time.monotonic() < deadline:
        # Randomly pick a source and a destination index
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1                      # ensure i != j

        if a[i] <= 1e-14:
            continue                    # nothing to move from this index

        # Propose moving a small amount of probability mass
        delta = rng.uniform(0.0, min(a[i], move_scale))
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_J = _objective(a_new)

        # Accept if improvement, otherwise accept with prob. exp(-Δ/T)
        if new_J < cur_J:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_J - cur_J) / max(T, 1e-12))

        if accept:
            a = a_new
            cur_J = new_J
            # Record global best
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] SA improved best J = {best_J:.6f}")

        # Temperature and move‑scale decay
        T *= decay
        move_scale *= decay

        # Restart if we have been stuck for a while (no improvement)
        if time.monotonic() - last_improve > 30.0:
            # Jitter the current solution towards a fresh Dirichlet draw
            jitter = rng.dirichlet(np.ones(n) * 0.5)
            a = 0.7 * best_a + 0.3 * jitter
            a /= a.sum()
            cur_J = _objective(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] SA restart – jitter applied (best J = {best_J:.6f})")

    # ------------------------------------------------------------------
    # 6. Return the best sequence (safely clipped & normalised)
    # ------------------------------------------------------------------
    final_vec = np.clip(best_a, 0.0, 1000.0)
    total = final_vec.sum()
    if total == 0.0:
        final_vec = np.full_like(final_vec, 1.0 / n)
    else:
        final_vec /= total

    return [float(v) for v in final_vec]
