"""
search using a two‑phase scheme.

Phase 1 – a broad portfolio of analytical windows and random Dirichlet draws,
keeping the best normalised vector (∑a_i = 1).

Phase 2 – projected gradient descent on the L2 distance between the full
auto‑convolution a∗a and the flat target 1/(2n‑1).  The gradient can be
computed efficiently via a single FFT, the iterate is projected onto the
probability simplex after each step and the step size follows a 1/√t schedule.
The best vector seen during the descent (measured by the true objective
J = 2·n·max(conv)) is retained.

The routine stops after the allocated wall‑clock budget (≈ 1000 s) and
returns the best sequence as a plain Python list.
"""

import time
import numpy as np
from typing import List

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex (sum=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("Simplex proj expects 1‑D array")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the full linear convolution vec * vec."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())

def objective_from_vec(vec: np.ndarray, n: int,
                       fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(vec,vec)) for a vector already normalised (∑=1)."""
    return 2.0 * n * full_conv_max(vec, fft_len, conv_len)

        except Exception:

    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

        nonlocal best_vec, best_J
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return

        if J < best_J:

        v = 1.0 / (i ** p)
        eval_update(v, f"pow p={p}")

    for _ in range(60):

        exp_vec += np.exp(-alpha * (n - 1 - np.arange(n, dtype=np.float64)))
        exp_vec /= exp_vec.sum()
        v = (1.0 - mix) * np.full(n, 1.0 / n) + mix * exp_vec
        eval_update(v, f"expblend a={alpha:.3g}")

    wdict = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),

    for name, w in wdict.items():
        if w.sum() > 0:
            eval_update(w, f"window {name}")

    for beta in [0.5, 2.0, 5.0, 10.0, 15.0]:
        w = np.kaiser(n, beta)
        if w.sum() > 0:
            eval_update(w, f"kaiser β={beta}")

    for alpha in np.linspace(0.0, 1.0, 11):

        def tukey_window(n_: int, alpha_: float) -> np.ndarray:
            if alpha_ <= 0.0:
                return np.ones(n_, dtype=np.float64)
            if alpha_ >= 1.0:
                return np.hanning(n_)

            w = np.ones_like(t)

            last = t > 1.0 - alpha_ / 2.0
            w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha_ - 2.0 / alpha_ + 1.0)))
            return w
        w = tukey_window(n, alpha)
        if w.sum() > 0:
            eval_update(w, f"tukey a={alpha}")

    # random Dirichlet draws
    for conc in [0.3, 0.6, 1.0, 2.0]:
        for _ in range(16):
            eval_update(rng.dirichlet(np.full(n, conc)), f"dirichlet conc={conc}")

    # previous construction if available
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = 1.0 / n
        else:
            v /= v.sum()
        eval_update(v, "prev")

    print(f"[0.00s] Portfolio best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected gradient descent on the flat‑target loss
    # ------------------------------------------------------------------
    # target constant for the convolution (average value).  Since sum(vec)=1,
    # the total mass of the auto‑convolution is 1, therefore the flat target is
    # 1/(2n‑1) at every index.
    target_val = 1.0 / conv_len
    target = np.full(conv_len, target_val, dtype=np.float64)

    # start from the current best vector
    cur_vec = best_vec.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a short safety margin

    step0 = 0.5   # initial step size multiplier
    iter_count = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        # ------------------------------------------------------------------
        # compute convolution and objective for the current iterate
        # ------------------------------------------------------------------
        conv = np.fft.irfft(np.fft.rfft(cur_vec, n=fft_len) ** 2, n=fft_len)[:conv_len]
        cur_J = 2.0 * n * conv.max()
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # gradient of the L2 loss L = Σ (conv - target)^2
        # ------------------------------------------------------------------
        residual = conv - target                     # shape conv_len
        # convolution of residual with reversed cur_vec gives gradient (up to a factor 4)
        rev_vec = cur_vec[::-1]
        grad_full = np.fft.irfft(
            np.fft.rfft(residual, n=fft_len) *
            np.fft.rfft(rev_vec, n=fft_len),
            n=fft_len)
        grad = 4.0 * grad_full[n-1:n-1+n]           # slice length n
        # ------------------------------------------------------------------
        # projected gradient step (simple 1/√t schedule)
        # ------------------------------------------------------------------
        step = step0 / np.sqrt(iter_count + 1.0)
        new_vec = cur_vec - step * grad
        new_vec = proj_simplex(new_vec)            # keep on simplex
        # ------------------------------------------------------------------
        # accept the move (always) – this is pure descent on the smooth loss,
        # but we keep the external best based on the true objective.
        # ------------------------------------------------------------------
        cur_vec = new_vec
        iter_count += 1

        # occasional random restart if stagnating for >30 s
        if time.monotonic() - last_improve > 30.0:
            cur_vec = proj_simplex(rng.random(n))
            last_improve = time.monotonic()
            print(f"[{elapsed:6.2f}s] restart due to stagnation")

    # ------------------------------------------------------------------
    # 5. Final polishing – a few steps of exponentiated‑gradient (optional)
    # ------------------------------------------------------------------
    # (the earlier reference used a sophisticated EG+annealing loop; we add a tiny
    #  version here because it is cheap and sometimes improves the tail.)
    cur_vec = best_vec.copy()
    cur_J = best_J
    for _ in range(200):
        # compute active set of maximal convolution entries
        conv_vals = np.fft.irfft(np.fft.rfft(cur_vec, n=fft_len) ** 2,
                                 n=fft_len)[:conv_len]
        max_c = conv_vals.max()
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv_vals >= max_c - tol)[0]
        grad = np.zeros_like(cur_vec)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=np.int64)
            grad[idx] += 2.0 * cur_vec[p - idx]
        # multiplicative EG update
        step = 0.3 / np.sqrt(_ + 1.0)
        factor = np.exp(-step * grad)
        cur_vec = cur_vec * factor
        cur_vec = proj_simplex(cur_vec)
        # check objective
        conv_max = full_conv_max(cur_vec, fft_len, conv_len)
        cur_J = 2.0 * n * conv_max
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cur_vec.copy()

    # ------------------------------------------------------------------
    # 6. Return best vector (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"Final best J = {best_J:.6f}")
    return [float(v) for v in best_vec]

            print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")

        # gradient of L2 loss L = Σ (conv - target)^2
        residual = conv - target
        rev = cur_vec[::-1]
        grad_full = np.fft.irfft(
            np.fft.rfft(residual, n=fft_len) *
            np.fft.rfft(rev, n=fft_len),
            n=fft_len)
        grad = 4.0 * grad_full[n-1:n-1+n]      # slice of length n

        # projected gradient step
        step = step0 / np.sqrt(it + 1.0)
        cand = cur_vec - step * grad
        cur_vec = proj_simplex(cand)

        it += 1

        # stagnation restart
        if time.monotonic() - last_improve > 30.0:
            cur_vec = proj_simplex(rng.random(n))
            last_improve = time.monotonic()
            print(f"[{elapsed:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # 5. Light polishing with an exponentiated‑gradient loop.
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    for t in range(200):
        conv_vals = np.fft.irfft(np.fft.rfft(cur_vec, n=fft_len) ** 2,
                                 n=fft_len)[:conv_len]
        max_c = conv_vals.max()
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv_vals >= max_c - tol)[0]

        g = np.zeros_like(cur_vec)
        for p in active:
            start_i = max(p - (n - 1), 0)
            end_i = min(p, n - 1)
            idx = np.arange(start_i, end_i + 1, dtype=np.int64)
            g[idx] += 2.0 * cur_vec[p - idx]

        step = 0.3 / np.sqrt(t + 1.0)
        factor = np.exp(-step * g)
        cur_vec = proj_simplex(cur_vec * factor)

        cur_max = full_conv_max(cur_vec, fft_len, conv_len)
        cur_J = 2.0 * n * cur_max
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cur_vec.copy()

    # ------------------------------------------------------------------
    # 6. Return the best vector (clipped, renormalised, list conversion).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    tot = best_vec.sum()
    if tot == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= tot

    print(f"Final best J = {best_J:.6f}")
    return [float(v) for v in best_vec]
