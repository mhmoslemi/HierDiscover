def propose_candidate(seed=12345, budget_s=1000.0):
    import time
    import numpy as np
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog
    rng = np.random.default_rng(seed)
    # Determine n
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = max(512, min(len(init_seq), 2000))
        seq = init_seq[:n]
        seq = np.clip(seq, 0, None)
        total = seq.sum()
        if total == 0:
            seq = np.full(n, 1.0 / n)
        else:
            seq /= total
        f = seq.copy()
    else:
        n = 1024
        f = np.full(n, 1.0 / n)
    # ensure sum f =1
    f = np.clip(f, 0, None)
    f_sum = f.sum()
    if f_sum == 0:
        f = np.full(n, 1.0 / n)
    else:
        f /= f_sum

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len-1).bit_length())
    def conv_full(a):
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        return conv

    best_f = f.copy()
    best_M = conv_full(f).max()
    best_J = 2.0 * n * best_M
    print(f"[0.0s] Initial J={best_J:.6f}, n={n}")

    start = time.monotonic()
    deadline = start + budget_s - 5.0  # safety margin
    iteration = 0
    last_improve = start
    t = 0.2
    eps_lp = 1e-12

    while time.monotonic() < deadline:
        iteration += 1
        # Compute M for current f
        conv_f_f = conv_full(f)
        M = conv_f_f.max()
        # Build convolution matrix A using toeplitz
        c = np.concatenate([f, np.zeros(n-1)])
        r = np.concatenate([f[:1], np.zeros(n-1)])
        A = toeplitz(c, r)   # dense matrix shape (conv_len, n)

        # Solve LP: maximize sum(g) i.e., minimize -sum(g)
        # constraints: A @ g <= M + eps_lp
        b = np.full(conv_len, M + 1e-12)  # slight tolerance
        # Use scipy.linprog
        # bounds: g_i >= 0
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True})
        if not res.success:
            print(f"[{time.monotonic()-start:.2f}s] LP failed at iter {iteration}, message: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            # trivial solution; break
            break
        # Scale g to same total sum as f (which is 1)
        g = g / sum_g   # now sum(g) =1

        # Mix
        f_new = (1 - t) * f + t * g
        # Re-normalise to sum=1
        f_new = np.clip(f_new, 0, None)
        total = f_new.sum()
        if total == 0:
            f_new = np.full_like(f_new, 1.0 / n)
        else:
            f_new /= total

        # Evaluate new J
        conv_new = conv_full(f_new)
        M_new = conv_new.max()
        J_new = 2.0 * n * M_new

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_f = f_new.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve-start:.2f}s] Iter {iteration} new best J={best_J:.6f}")

        # Replace f with new one
        f = f_new

        # Stagnation detection
        if time.monotonic() - last_improve > 30.0:
            # random jitter: reinitialize f from Dirichlet with alpha around current f magnitude
            alpha = np.full(n, 0.3) + 0.1 * rng.random(n)
            f = rng.dirichlet(alpha)
            f /= f.sum()
            last_improve = time.monotonic()
            print(f"[{last_improve-start:.2f}s] Restart jitter after stagnation")
            continue

    # Return best_f as list after clipping to [0,1000] (range fine)
    best_f = np.clip(best_f, 0.0, 1000.0)
    total = best_f.sum()
    if total == 0:
        best_f = np.full_like(best_f, 1.0 / n)
    else:
        best_f /= total
    return [float(v) for v in best_f]

"""
Propose a step‑function using an *alternating linear‑program* scheme.

Idea
----
Given a current probability vector `f` (∑f = 1, f ≥ 0) we compute
`M = max(conv(f,f))`.  The next vector `g` is obtained by solving the
linear program

    maximize   ∑ g_j
    subject to 0 ≤ g_j
                (f * g)_k ≤ M   for every convolution index k

which is linear because for a fixed `f` the convolution `f * g` is a
matrix–vector product `T(f) @ g`.  The LP pushes mass of `g` into
positions where `f` is small while respecting the same peak bound `M`.
After solving we renormalise `g` to unit sum and form a convex combination

    f_new = (1‑t)·f + t·g

with a step size `t` that is reduced until the new self‑convolution
does not exceed the previous bound `M`.  This guarantees a monotone
decrease of the evaluation value `J = 2·n·max(conv(f,f))`.

The procedure is repeated until the time budget expires.  To avoid
stagnation we occasionally jitter the current vector with a Dirichlet
sample.  The best vector found is returned (clipped to the allowed
range and re‑normalised).

The algorithm works entirely with NumPy, SciPy (for sparse matrices and
`linprog`) and FFT‑based convolutions, and comfortably fits into the
1000 s budget.
"""

import time
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT padding)."""
    return 1 << ((x - 1).bit_length())

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (probability vector) that
    minimises ``evaluate_sequence``.  The search runs for at most
    ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a moderate problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = max(512, min(len(init_seq), 2000))
        f = init_seq[:n].copy()
    else:
        n = 1024
        f = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise to a probability vector (sum = 1)
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s == 0.0:
        f = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        f /= s

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # Helper: convolution of a vector with itself (full linear conv)
    # ------------------------------------------------------------------
    def _conv_self(a: np.ndarray) -> np.ndarray:
        A = np.fft.rfft(a, fft_len)
        c = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        return c

    # ------------------------------------------------------------------
    # 2. Pre‑compute the index pattern of the convolution matrix
    #    (sparse Toeplitz – maps g -> f * g)
    # ------------------------------------------------------------------
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n, dtype=np.int32)
        cols[idx:idx + n] = j
        idx += n
    # shape of the matrix: (conv_len, n)
    shape_T = (conv_len, n)

    # ------------------------------------------------------------------
    # 3. Initial evaluation
    # ------------------------------------------------------------------
    cur_conv = _conv_self(f)
    M = cur_conv.max()
    best_f = f.copy()
    best_J = float(2.0 * n * M)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin
    last_improve = start_time

    print(f"[0.00s] start n={n}, initial J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main alternating LP loop
    # ------------------------------------------------------------------
    ITER_MAX = 10_000_000  # just a huge upper bound
    iter_cnt = 0
    while time.monotonic() < deadline and iter_cnt < ITER_MAX:
        iter_cnt += 1

        # ----- 4.1 Build the convolution matrix for the current f  -----
        data = np.tile(f, n)                # each column repeats the whole f
        T = csr_matrix((data, (rows, cols)), shape=shape_T)

        # ----- 4.2 Linear program: maximise sum(g)  -----
        #    constraints: T @ g <= M   (one for each convolution index)
        #    bounds:       g >= 0
        b_ub = np.full(conv_len, M + 1e-12)   # tiny tolerance for numerical safety
        lp_res = linprog(c=-np.ones(n), A_ub=T, b_ub=b_ub,
                         bounds=(0, None), method='highs',
                         options={'presolve': True, 'time_limit': 20})
        if not lp_res.success:
            # If the LP fails (rare), break out – we keep the best found so far.
            print(f"[{time.monotonic()-start_time:.2f}s] LP failed: {lp_res.message}")
            break

        g = lp_res.x
        # Renormalise to unit sum (the LP enforces sum(g) ≤ 1, we want equality)
        g_sum = g.sum()
        if g_sum == 0.0:      # degenerate, abort
            break
        g = g / g_sum

        # ----- 4.3 Line‑search mixing to keep the bound monotone -----
        # start with a moderate step size and shrink until the new bound
        # does not exceed the old one.
        t_try = 0.5
        accepted = False
        while t_try > 1e-5:
            f_candidate = (1.0 - t_try) * f + t_try * g
            # Small numeric clean‑up
            f_candidate = np.clip(f_candidate, 0.0, None)
            f_candidate /= f_candidate.sum()

            conv_cand = _conv_self(f_candidate)
            M_cand = conv_cand.max()
            if M_cand <= M + 1e-12:          # non‑increasing bound
                f = f_candidate
                M = M_cand
                accepted = True
                break
            t_try *= 0.5                     # halve step size and retry

        # If no step size succeeded we keep the old `f` unchanged.
        # The current `f` is still feasible for the LP in the next round.

        # ----- 4.4 Update global best -----
        cur_J = float(2.0 * n * M)
        now = time.monotonic()
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_f = f.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best J = {best_J:.6f}")

        # ----- 4.5 Stagnation detection -----
        if now - last_improve > 30.0:
            # jitter the current vector with a fresh Dirichlet sample
            alpha = np.full(n, 0.3) + 0.1 * rng.random(n)
            f = rng.dirichlet(alpha)
            f /= f.sum()
            # recompute current bound after jitter
            M = _conv_self(f).max()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] stagnation – jitter restart")

    # ------------------------------------------------------------------
    # 5. Return the best vector found (clipped to the allowed range)
    # ------------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)
    total = best_f.sum()
    if total == 0.0:
        best_f = np.full_like(best_f, 1.0 / n)
    else:
        best_f /= total

    return [float(v) for v in best_f]
