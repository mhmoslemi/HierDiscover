    conv_len = 2*n-1
    # Build matrix A_ub where each row is C_k with negative M

def solve_g(f):
    n = f.shape[0]
    conv_len = 2*n - 1

    # Build sparse matrix C
    # Using coo
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    # fill
    offset = 0
    for j in range(n):
        rows[offset:offset+n] = np.arange(j, j+n)
        cols[offset:offset+n] = j
        data[offset:offset+n] = f
        offset += n

    C = coo_matrix((data, (rows, cols)), shape=(conv_len, n)).tocsr()
    # Column for -M
    M_col = -np.ones((conv_len, 1), dtype=np.float64)
    A_ub = hstack([C, M_col], format='csr')
    b_ub = np.zeros(conv_len, dtype=np.float64)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimize M (last variable)
    c = np.zeros(n+1, dtype=np.float64)
    c[-1] = 1.0

    # bounds
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 10})
    if not res.success:
        return None, None
    x = res.x
    g = x[:n]
    M_opt = x[-1]
    # ensure sum close to 1
    g_sum = g.sum()
    if g_sum > 0:
        g = g / g_sum
    return g, M_opt

"""
Hybrid alternating linear‑program optimisation + local refinement.

* Stage 1 – Alternating minimisation:
    Given a current non‑negative vector f (∑f=1) we solve the linear programme
    minimise   M
    s.t.       (f * g)[k] ≤ M  for all k (full convolution)
               g_i ≥ 0 ,   ∑ g_i = 1 .
    The solution g is the best partner for f.  We replace f by a convex
    combination (1‑t)·f + t·g (t≈1) and repeat.  Each LP reduces the bound
    on the maximal convolution, driving the objective J = 2·n·max(conv) down.

* Stage 2 – Fast stochastic pair‑exchange polishing:
    When the LP phase stagnates we switch to a simple simulated‑annealing
    walk that repeatedly transfers a small amount of mass between two
    coordinates and accepts improvements (or occasional worse moves).

The whole routine respects the 1000 s budget, works on any length n
(we pick a moderate size, 1024 … 2048, to keep the LP tractable) and
returns a list of non‑negative heights that minimise ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, hstack

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(a: np.ndarray) -> float:
    """Return the objective J = 2·n·max(conv(a,a)) for a sum‑normalised vector."""
    n = a.size
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def build_conv_matrix(f: np.ndarray) -> coo_matrix:
    """
    Build the (2n‑1) × n sparse matrix C such that C·g = conv(f,g) (full convolution).
    Row k contains the shifted copy of f that participates in the k‑th entry of the
    convolution.
    """
    n = f.size
    conv_len = 2 * n - 1

    # rows[i,j] = i + j  (i indexes entry of f, j indexes column of g)
    rows = (np.arange(n)[:, None] + np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)

    return coo_matrix((data, (rows, cols)), shape=(conv_len, n), dtype=np.float64)


def solve_for_partner(f: np.ndarray) -> tuple[np.ndarray, float] | tuple[None, None]:
    """
    Solve the linear programme:
        minimise   M
        s.t.       conv(f,g) ≤ M
                   g ≥ 0 ,  ∑ g = 1 .
    Returns the optimal g and the achieved bound M, or (None,None) on failure.
    """
    n = f.size
    conv_len = 2 * n - 1

    # sparse matrix C with C·g = conv(f,g)
    C = build_conv_matrix(f).tocsr()

    # Append the column for -M (one per inequality)
    M_col = -np.ones((conv_len, 1), dtype=np.float64)
    A_ub = hstack([C, M_col], format="csr")
    b_ub = np.zeros(conv_len, dtype=np.float64)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise the last variable (M)
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    # Bounds: g_i ≥ 0,   M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Solve with HiGHS (fast for moderate sized LPs)
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None, None

    x = res.x
    g = x[:n]
    M_opt = x[-1]

    # Numerical cleanup – enforce the simplex exactly
    g = np.maximum(g, 0.0)
    if g.sum() > 0:
        g /= g.sum()
    else:
        g = np.full_like(g, 1.0 / n)

    return g, M_opt


def stochastic_refine(vec: np.ndarray, time_deadline: float, rng: np.random.Generator):
    """
    Simple simulated‑annealing style pair‑exchange polish.
    Operates until the supplied deadline and returns the best vector found.
    """
    n = vec.size
    cur = vec.copy()
    cur_J = compute_J(cur)
    best = cur.copy()
    best_J = cur_J

    # exponential temperature schedule (starts from current J)
    T_start = cur_J
    T_end = 1e-8
    total = time_deadline - time.monotonic()

    iteration = 0
    while time.monotonic() < time_deadline:
        now = time.monotonic()
        elapsed = now - (time_deadline - total)
        T = T_start * (T_end / T_start) ** (elapsed / total)

        # propose an exchange move
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] == 0:
            continue
        delta = rng.random() * min(cur[i], 0.02)          # move up to 2 % of source mass
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = proj_simplex(cand)                         # stay on the simplex
        cand_J = compute_J(cand)

        accept = False
        if cand_J < cur_J:
            accept = True
        else:
            if rng.random() < np.exp(-(cand_J - cur_J) / max(T, 1e-12)):
                accept = True

        if accept:
            cur = cand
            cur_J = cand_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best = cur.copy()
        iteration += 1
    return best, best_J

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm first runs an alternating LP
    optimisation on a moderate size (≈1500) and then finishes with a fast
    stochastic pair‑exchange polish.  The best vector found within the time
    budget is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # leave a small safety margin for the final conversion
    deadline = start_time + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Choose a problem size.
    # If an old construction is present we keep its length modest to stay fast.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = int(len(globals()["height_sequence_1"]))
            # Use a size between 1024 and 2048 for tractability.
            n = max(1024, min(2048, n0))
        except Exception:
            n = 1500
    else:
        n = 1500

    # ------------------------------------------------------------------
    # Build a small pool of diverse initial vectors on the simplex.
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws with various concentrations
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best (down‑sampled or padded)
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        if old.size >= n:
            # simple block‑averaging down‑sampling
            factor = old.size // n
            down = old[: factor * n].reshape(n, factor).mean(axis=1)
            down = np.clip(down, 0, None)
            if down.sum() > 0:
                down /= down.sum()
                init_cands.append(down)
        else:
            pad = np.zeros(n, dtype=np.float64)
            pad[: old.size] = old
            if pad.sum() > 0:
                pad /= pad.sum()
                init_cands.append(pad)

    # pick the best start according to the cheap J‐computation
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[{time.monotonic() - start_time:.2f}s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Alternating LP optimisation
    # ------------------------------------------------------------------
    cur_f = best_vec.copy()
    cur_J = best_J
    global_best_f = cur_f.copy()
    global_best_J = cur_J

    iteration = 0
    last_improve_iter = 0
    max_iters = 500                       # generous upper bound; loop stops earlier by time

    # we reserve about 80 % of the budget for the LP phase
    lp_deadline = start_time + 0.8 * (deadline - start_time)

    while time.monotonic() < lp_deadline and iteration < max_iters:
        # Solve the partner‑LP
        g, M_opt = solve_for_partner(cur_f)
        if g is None:
            # fall back to a small random jitter and continue
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_f = proj_simplex(cur_f * noise)
            cur_J = compute_J(cur_f)
            iteration += 1
            continue

        # Convex blend (we start with almost full replacement)
        t = 0.9
        new_f = (1.0 - t) * cur_f + t * g
        new_f = proj_simplex(new_f)               # stay exactly on the simplex
        new_J = compute_J(new_f)

        # Accept the new point unconditionally (it never makes the bound worse)
        cur_f = new_f
        cur_J = new_J

        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_f = cur_f.copy()
            last_improve_iter = iteration
            print(f"[{time.monotonic() - start_time:.2f}s] LP iter {iteration} – new best J = {global_best_J:.6f}")

        # stagnation handling – small jitter after a while without improvement
        if iteration - last_improve_iter > 40:
            jitter = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_f = proj_simplex(global_best_f * jitter)
            cur_J = compute_J(cur_f)
            last_improve_iter = iteration
            print(f"[{time.monotonic() - start_time:.2f}s] LP jitter – J = {cur_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Stage 2 – Stochastic pair‑exchange polishing (simulated annealing)
    # ------------------------------------------------------------------
    refine_deadline = deadline
    polished_f, polished_J = stochastic_refine(global_best_f, refine_deadline, rng)

    if polished_J < global_best_J:
        global_best_f = polished_f
        global_best_J = polished_J
        print(f"[{time.monotonic() - start_time:.2f}s] Polishing yielded J = {global_best_J:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation (clip to allowed range and re‑normalise)
    # ------------------------------------------------------------------
    best_seq = np.clip(global_best_f, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / best_seq.size
    else:
        best_seq /= total

    final_J = compute_J(best_seq)
    print(f"Final best J = {final_J:.6f} (length {best_seq.size})")
    return [float(v) for v in best_seq]
