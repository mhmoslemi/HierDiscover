"""
Hybrid random‑search + Adam‑softmax optimizer for the step‑function problem.

* Phase 1 – fast random exploration:
    – Sample many Dirichlet‑distributed probability vectors with a wide
      range of concentration parameters.
    – Keep the best vector (lowest 2·n·max(conv)) as a warm‑start.

* Phase 2 – smooth‑max Adam descent on the simplex (via a softmax
  parametrisation y → a = softmax(y)):
    – The objective is the smooth approximation
          fβ(a) = log‑sum‑exp(β·(a*a))   (β grows from 1 to a large value).
    – Its gradient w.r.t. a is 2·corr(w,a) where w is the softmax weight.
    – Using the chain‑rule we obtain the gradient w.r.t. y:
          ∇y fβ = a·(∇a fβ – ⟨a,∇a fβ⟩),
      which is fed to an Adam optimiser.
    – β is increased exponentially during the run (continuation method).

* Throughout both phases the current best sequence (in terms of the true
  evaluation J = 2·n·max(conv)) is tracked and finally returned.

All operations are performed with FFT‑based convolutions (O(n log n)),
so the routine easily fits into the 1000 s budget for n up to ≈ 8000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a numerically stable soft‑max over the convolution values.
    Returns a probability vector of the same length as `conv`.
    """
    # shift by the maximum to avoid overflow
    shifted = beta * (conv - conv.max())
    # exp may underflow to 0, but that's fine
    w = np.exp(shifted)
    s = w.sum()
    if s == 0.0:
        # fallback to uniform – should never happen for finite beta
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _log_sum_exp_max(conv: np.ndarray, beta: float) -> float:
    """
    Compute the smooth max value: (1/β)·log Σ exp(β·conv).
    This is not needed for the optimisation itself, but handy for debugging.
    """
    shifted = conv - conv.max()
    return conv.max() + np.log(np.exp(beta * shifted).sum()) / beta

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  The algorithm consists of a fast random‑search
    warm‑start followed by an Adam optimiser on a smooth‑max surrogate.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size (use existing sequence length if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # keep n in a practical range for FFT speed
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # length for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # length for correlations

    # --------------------------------------------------------------
    # 2. Utility: evaluate J = 2·n·max(conv) for a probability vector
    # --------------------------------------------------------------
    def _eval_J(a_vec: np.ndarray) -> float:
        conv = _conv_full(a_vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 3. Phase 1 – random Dirichlet exploration
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 0.5   # keep a small safety margin

    # initialise best with a simple uniform candidate
    best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    best_J = _eval_J(best_vec)

    # deterministic seeds (geometric, linear ramps, Gaussian)
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        J = _eval_J(v)
        if J < best_J:
            best_J, best_vec = J, v.copy()

    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    J = _eval_J(v)
    if J < best_J:
        best_J, best_vec = J, v.copy()

    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    J = _eval_J(v)
    if J < best_J:
        best_J, best_vec = J, v.copy()

    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        J = _eval_J(v)
        if J < best_J:
            best_J, best_vec = J, v.copy()

    # Random Dirichlet draws for a limited time (≈ 15 % of total budget)
    random_phase = min(30.0, float(budget_s) * 0.15)
    phase_end = start_time + random_phase
    alphas = np.array([0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 2.0])
    while time.monotonic() < phase_end:
        alpha = rng.choice(alphas)
        cand = rng.dirichlet(np.full(n, alpha))
        J = _eval_J(cand)
        if J < best_J:
            best_J, best_vec = J, cand.copy()

    print(f"[{time.monotonic() - start_time:6.2f}s] After random phase best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Phase 2 – Adam descent on a soft‑max surrogate (continuation on β)
    # --------------------------------------------------------------
    # Convert the best probability vector into log‑space parameters y
    eps = 1e-12
    y = np.log(best_vec + eps)                     # shape (n,)

    # Adam state
    m = np.zeros_like(y)
    v_acc = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5                                     # initial learning rate

    # β‑schedule (exponential from 1 → 1500)
    beta_start = 1.0
    beta_end = 1500.0

    iteration = 0
    last_improve = start_time
    global_best_vec = best_vec.copy()
    global_best_J = best_J

    # Main optimisation loop
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # Update β and learning rate
        # ----------------------------------------------------------
        progress = elapsed / (deadline - start_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr0 * (1.0 - progress) + 0.01 * progress   # slight decay, never below 0.01

        # ----------------------------------------------------------
        # Current probability vector a = softmax(y)
        # ----------------------------------------------------------
        y_shift = y - y.max()          # improve numerical stability
        a = np.exp(y_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ----------------------------------------------------------
        # Compute convolution and smooth‑max weight vector
        # ----------------------------------------------------------
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # Track global best
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_vec = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New global best J = {global_best_J:.6f}")

        # ----------------------------------------------------------
        # Soft‑max weights over the convolution values
        # ----------------------------------------------------------
        w = _softmax_weights(conv, beta)

        # ----------------------------------------------------------
        # Gradient w.r.t. a (smooth‑max surrogate)
        # ----------------------------------------------------------
        grad_a = _weighted_gradient(a, w, fft_len_corr, n)   # shape (n,)

        # ----------------------------------------------------------
        # Convert to gradient w.r.t. y (softmax parametrisation)
        # ----------------------------------------------------------
        avg = np.dot(a, grad_a)               # scalar = ⟨a, grad_a⟩
        grad_y = a * (grad_a - avg)           # shape (n,)

        # ----------------------------------------------------------
        # Adam update on y
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v_acc = beta2 * v_acc + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_acc / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        y -= step

        # ----------------------------------------------------------
        # Simple stagnation escape: occasional Gaussian jitter
        # ----------------------------------------------------------
        if (iteration % 2000) == 0:
            # If we haven't improved for a while, add a tiny random perturbation
            if now - last_improve > 30.0:
                jitter = rng.normal(0.0, 0.02, size=n)
                y += jitter

    # --------------------------------------------------------------
    # 5. Return the best sequence found (clip + renormalise for safety)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"[{time.monotonic() - start_time:6.2f}s] Finished – best J = {global_best_J:.6f}")

    return [float(v) for v in best_vec]
