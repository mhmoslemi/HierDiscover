"""
Propose a monotone power‑law step‑function.

The evaluation function is
    J = 2·n·max(conv(a,a)) / (sum(a)²)
and is invariant under scaling of `a`.  After normalising the vector to
unit L¹‑norm we only have to minimise the maximum entry of the linear
convolution.  For a discrete density of the form

        a_i ∝ (i+1) ** p   (i = 0 … n‑1)

the convolution is a smooth, unimodal function whose peak occurs near
the centre (offset n‑1).  By choosing the exponent `p` ≈ log₄(n) the
central value and the tail value (where a single large entry is squared)
become comparable, giving a very small objective.  The code evaluates a
few candidate exponents and returns the best vector found.

The algorithm runs in a few milliseconds, well within the 1000 s budget.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper – fast convolution using FFT (more robust for n up to ~8192)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    n = a.size
    conv_len = 2 * n - 1
    # next power‑of‑two length ≥ conv_len for exact linear convolution
    L = 1 << ((conv_len - 1).bit_length())
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, n=L)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative sequence that minimises ``evaluate_sequence``.

    The method builds a family of power‑law ramps a_i ∝ (i+1)^p,
    normalises each to unit L¹‑norm and evaluates the exact objective.
    The best exponent (typically around log₄(n)) is selected.
    """
    rng = np.random.default_rng(seed)            # kept for reproducibility

    # --------------------------------------------------------------
    # 1. Choose the problem size
    # --------------------------------------------------------------
    # Default to a fairly large length that is still cheap to FFT.
    # If a previous solution exists we reuse its length (capped).
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    # --------------------------------------------------------------
    # 2. Build candidate power‑law ramps
    # --------------------------------------------------------------
    # Exponent range centred around log4(n).  Adding a small margin
    # guarantees we check the optimum even if n is not a perfect power.
    log4_n = np.log(n) / np.log(4)
    p_min = max(1, int(np.floor(log4_n)) - 2)
    p_max = int(np.ceil(log4_n)) + 3
    exponents = list(range(p_min, p_max + 1))

    best_vec = None
    best_J = np.inf

    start = time.monotonic()
    for p in exponents:
        # a_i = (i+1)^p  (power‑law ramp, increasing)
        idx = np.arange(1, n + 1, dtype=np.float64)    # 1 … n
        vec = idx ** p
        vec /= vec.sum()                               # L¹‑normalise (sum = 1)

        # Compute the exact objective J = 2·n·max(conv)
        conv = _conv_full(vec)
        J = float(2.0 * n * conv.max())

        # Keep the best
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
        # (optional) print progress – useful when debugging
        # print(f"p={p:<2d} -> J={J:.6f}")

    elapsed = time.monotonic() - start
    print(f"[{elapsed:.2f}s] Chosen exponent p={p_min}..{p_max}, best J={best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Final safety – clip to allowed range and re‑normalise
    # --------------------------------------------------------------
    # The evaluator clamps values to [0,1000]; we stay safely inside.
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        # fallback – uniform distribution
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
