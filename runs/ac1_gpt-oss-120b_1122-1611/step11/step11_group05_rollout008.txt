"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.

Algorithm
---------

1. **Problem size** – use the length of the previously found
   `height_sequence_1` (default 4096).  The objective
   `evaluate_sequence` is homogeneous, so we normalise the sequence to
   sum‑to‑one and minimise the maximal entry of the auto‑convolution.

2. **Combinatorial construction** – we look for a *sparse* support
   `S ⊂ {0,…,n‑1}` such that for the binary mask `b[i]=1_{i∈S}` the
   full convolution `c = b * b` never exceeds a small bound `M`
   (typically `M=2`).  With `|S|=k` and `max(c)=M` the true objective is

       J = 2·n·M / k² .

   For `M=2` the target `J≤1.5025` requires `k ≥ sqrt(2·n·M/target)`,
   i.e. about 100 indices for `n≈3800`.  The theoretical capacity is
   `k ≤ 2·sqrt(n) ≈ 123`, so a solution should exist.

   We build `S` by a *greedy random* algorithm: scan a random permutation
   of the indices, add an index if none of the resulting sums would
   exceed `M`.  Many independent trials are performed; the best set
   (largest `k`, smallest `J`) is kept.

3. **Optional refinement** – with the remaining time a few
   exponentiated‑gradient steps (soft‑max surrogate of the max‑norm)
   are applied to the full vector (uniform weight on the chosen support)
   to smooth out any remaining peaks.

The whole procedure respects the 1000 s budget, contains no I/O,
and returns a list of floats ready for `evaluate_sequence`.  
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT (first `conv_len` entries)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights for a vector of convolution values."""
    shift = conv.max()
    exp_vals = np.exp(beta * (conv - shift))
    return exp_vals / exp_vals.sum()


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float,
                     fft_len_corr: int) -> np.ndarray:
    """
    Gradient of the smoothed max objective
        f(a) = (1/β)·log Σ exp(β·conv)
    w.r.t. a.  Result = 2·(w ⋆ a) where w are the soft‑max weights
    and ⋆ denotes linear cross‑correlation.
    """
    w = _softmax_weights(conv, beta)                 # shape (2n‑1,)
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    W_hat = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)   # linear correlation
    n = a.size
    start = n - 1
    return 2.0 * corr[start:start + n]


def _greedy_build(n: int, bound: int, rng: np.random.Generator):
    """
    Greedy random construction of a set S ⊂ {0,…,n‑1} such that the
    convolution of its binary mask never exceeds `bound`.
    Returns the set (as a Python ``set``) and the full integer convolution.
    """
    conv_len = 2 * n - 1
    conv = np.zeros(conv_len, dtype=np.int16)   # 0 ≤ conv ≤ bound ≤ 4
    positions = []               # list for ordered iteration
    pos_set = set()

    order = rng.permutation(n)
    for idx in order:
        # self‑contribution would be 1
        if conv[2 * idx] + 1 > bound:
            continue
        feasible = True
        # check all pair contributions (two ordered pairs per distinct pair)
        for j in positions:
            s = idx + j
            if conv[s] + 2 > bound:
                feasible = False
                break
        if not feasible:
            continue

        # accept idx – update convolution counts
        for j in positions:
            s = idx + j
            conv[s] += 2
        conv[2 * idx] += 1
        positions.append(idx)
        pos_set.add(idx)

    return pos_set, conv


def _refine_softmax(vec: np.ndarray, time_budget: float,
                    n: int, rng: np.random.Generator) -> np.ndarray:
    """
    A light‑weight exponentiated‑gradient optimiser that works on the
    soft‑max surrogate of the max‑norm.  It stops after ``time_budget``
    seconds and returns the best vector seen.
    """
    # pre‑compute FFT lengths
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)
    fft_len_corr = _next_pow2(conv_len + n - 1)

    best_vec = vec.copy()
    best_J = float('inf')
    cur_vec = vec.copy()
    cur_J = best_J

    start = time.monotonic()
    deadline = start + time_budget

    # scheduling parameters
    beta = 5.0
    beta_max = 300.0
    beta_inc = 1.02
    iteration = 0

    while time.monotonic() < deadline:
        # exact objective (no smoothing)
        conv = _conv_full(cur_vec, fft_len_conv, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()

        # gradient of soft‑max surrogate
        grad = _gradient_softmax(cur_vec, conv, beta, fft_len_corr)

        # exponentiated‑gradient proposal
        step = 0.5 / np.sqrt(iteration + 1.0)
        factor = np.exp(-step * grad)
        proposal = cur_vec * factor
        proposal = np.clip(proposal, 0.0, None)
        s = proposal.sum()
        if s == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= s

        # evaluate proposal
        prop_conv = _conv_full(proposal, fft_len_conv, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # accept if better (no simulated‑annealing here)
        if prop_J < cur_J:
            cur_vec = proposal
            cur_J = prop_J

        # slowly increase smoothing
        if iteration % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        iteration += 1

    return best_vec


# ----------------------------------------------------------------------
# Main optimisation routine – function required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine first builds a sparse support
    with a hard bound on the integer convolution, then (optionally)
    refines the weights by a few gradient steps.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine problem size `n`
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 4096
    n = int(max(256, min(n, 16384)))

    # ------------------------------------------------------------------
    # 2. Settings
    # ------------------------------------------------------------------
    target_J = 1.5025                       # we want to beat this value
    bounds = [2, 3, 4]                      # allowed convolution bound M
    best_vec = None
    best_J = float('inf')

    start_time = time.monotonic()
    total_deadline = start_time + float(budget_s)

    # allocate 80 % of the budget for the combinatorial search
    search_deadline = start_time + 0.80 * budget_s

    # ------------------------------------------------------------------
    # 3. Greedy random search for a low‑max‑convolution support
    # ------------------------------------------------------------------
    for bound in bounds:
        # required size to beat the target (derived from J = 2·n·M/k²)
        k_needed = int(np.ceil(np.sqrt(2.0 * n * bound / target_J)))
        # a small safety cushion
        k_needed = max(k_needed, bound + 5)

        attempts = 0
        while time.monotonic() < search_deadline:
            attempts += 1
            positions, conv = _greedy_build(n, bound, rng)
            k = len(positions)
            max_cnt = int(conv.max())
            if k == 0:
                continue

            # exact objective for this support (uniform weights)
            J = 2.0 * n * max_cnt / (k * k)

            if J < best_J:
                # construct the probability vector
                vec = np.zeros(n, dtype=np.float64)
                vec[list(positions)] = 1.0 / k
                best_vec = vec
                best_J = J
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] bound={bound} size={k} max={max_cnt} → J={J:.6f}")

            # early exit if we already satisfy the target
            if J <= target_J:
                break

        if best_J <= target_J:
            break   # good enough – stop trying larger bounds

    # ------------------------------------------------------------------
    # 4. If no decent support was found use a fallback (uniform)
    # ------------------------------------------------------------------
    if best_vec is None:
        # Uniform distribution – guarantees J = 2 (the worst case)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_J = 2.0
        print("[fallback] using uniform sequence (J=2)")

    # ------------------------------------------------------------------
    # 5. Optional soft‑max refinement with remaining time
    # ------------------------------------------------------------------
    remaining_time = total_deadline - time.monotonic()
    if remaining_time > 5.0:
        try:
            refined = _refine_softmax(best_vec, remaining_time, n, rng)
            # compute true J for the refined vector
            conv_len = 2 * n - 1
            fft_len_conv = _next_pow2(conv_len)
            conv = _conv_full(refined, fft_len_conv, conv_len)
            J_refined = float(2.0 * n * conv.max())
            if J_refined < best_J:
                best_vec = refined
                best_J = J_refined
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] refined J={J_refined:.6f}")
        except Exception as exc:          # pragma: no cover – safety net
            print(f"[refine] exception ignored: {exc}")

    # ------------------------------------------------------------------
    # 6. Final safety – clip, renormalise and return
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # (optional) final verification – does not affect the result
    try:
        J_final = evaluate_sequence(best_vec.tolist())
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] final J (via evaluate_sequence) = {J_final:.6f}")
    except Exception:                       # pragma: no cover
        pass

    return [float(v) for v in best_vec]
