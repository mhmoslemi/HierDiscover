import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a * a via real FFT, length conv_len."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

"""
Projected‑gradient optimiser for the max‑auto‑convolution problem.

What it does
------------
* Works on the probability simplex (∑a_i = 1, a_i ≥ 0).
* Approximates the objective   max_k (a*a)_k   by the smooth
  soft‑max   f_α(a) = (1/α)·log(∑_k exp(α·(a*a)_k)).
  Its gradient is a weighted cross‑correlation and never suffers from
  overflow because we always subtract the current maximum before the
  exponential.
* Uses a simple decaying step size together with a short back‑tracking
  line‑search and a Euclidean projection onto the simplex.
* If no improvement is seen for a while a multiplicative Gaussian jitter
  (≈ random restart) is applied to the current best solution.
* The routine runs for at most ``budget_s`` seconds and returns the best
  vector found (as a plain list of floats).

The algorithm is deliberately different from the exponentiated‑gradient
approach in the reference implementation; the soft‑max gradient provides
information from *all* convolution entries, not only the few that are
within a fixed tolerance of the peak.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Fallback: uniform distribution (should hardly happen)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a list of floats).
    """

    # 1. Choose a problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    fft_len = _next_pow2(conv_len)

    # 2. Build a small collection of initial probability vectors.
    # ------------------------------------------------------------------
    init_vecs = []

    # uniform
    init_vecs.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_vecs.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_vecs.append(v)

    #    iteration count (sharpens the approximation of the max).
    # ------------------------------------------------------------------

            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final clean‑up (clip, renormalise) and return the best vector.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
