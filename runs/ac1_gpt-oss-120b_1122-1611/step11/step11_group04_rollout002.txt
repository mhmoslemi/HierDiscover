"""
Propose a step‑function using a smooth–max gradient descent on the simplex.

Algorithm
---------
* Work on the probability simplex via a soft‑max parametrisation:
      a_i = exp(x_i) / Σ_j exp(x_j)   (a_i ≥ 0, Σ a_i = 1).
* Replace the non‑smooth objective  max(conv(a,a)) by a smooth
  log‑sum‑exp surrogate:
      F_λ(a) = (1/λ)·log Σ_k exp( λ·(a*a)_k )
  with a schedule λ : 5 → 400.  For large λ the surrogate
  approaches the true maximum.
* The gradient of F_λ w.r.t. a is
      ∇_a F = 2·corr( w , a )   where   w_k ∝ exp(λ·(a*a)_k)
  (corr denotes the “valid’’ cross‑correlation).
* Convert the gradient to the soft‑max space:
      ∇_x F = a ⊙ (∇_a F – (a·∇_a F)).
* Apply Adam (adapted learning‑rate, bias‑correction) to the
  logits x.  Every 1 000 iterations a tiny Gaussian jitter is added
  to escape plateaus.
* Multiple restarts are performed until the time budget is exhausted.
  The best vector (according to the exact objective
  J = 2·n·max(conv(a,a))) is kept and returned.

The routine respects the 1000 s budget, prints progress and finally
returns a list of floats ready for ``evaluate_sequence``.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, n: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Length of the result is 2*n‑1.
    """
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _smooth_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Return the probability weights w_k = exp(lam*conv_k) / Σ exp(lam*conv).
    Computed in a numerically stable way.
    """
    vals = lam * conv
    maxv = np.max(vals)
    exp_vals = np.exp(vals - maxv)
    sum_exp = np.sum(exp_vals)
    return exp_vals / sum_exp


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / np.sum(e)

        # Inner optimisation loop
        # ----------------------------------------------------------
        for it in range(1, max_inner_iters + 1):
            # time check – exit early if we are out of budget
            if time.monotonic() >= deadline:
                break

            # ---- soft‑max to obtain a valid probability vector ----
            a = _softmax(x)

            # ---- convolution a * a (FFT version, O(n log n)) ----
            conv = _conv_full_fft(a, n)

            # ---- smoothing parameter λ (linear schedule) ----
            lam = lam_start + (lam_end - lam_start) * (it / max_inner_iters)

            # ---- compute the smooth‑max weights w_k ----
            w = _smooth_max_weights(conv, lam)

            # ---- gradient w.r.t. a : 2 * correlation(w, a) (valid) ----
            # np.correlate with mode='valid' yields exactly the needed size n
            g_a = 2.0 * np.correlate(w, a, mode='valid')

            # ---- chain rule to obtain gradient w.r.t. logits x ----
            dot = np.dot(g_a, a)                     # a·∇_aF
            g_x = a * (g_a - dot)                    # ∇_xF

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * g_x
            v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            lr = lr0 * np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it)
            x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # ---- occasional jitter to boost exploration ----
            if it % 1000 == 0:
                x += rng.normal(0.0, 0.02, size=n)

            # ---- evaluate the true (non‑smooth) objective from time to time ----
            if it % 200 == 0:
                J = _exact_objective(a, n)
                if J < best_J - 1e-12:
                    best_J = J
                    best_vec = a.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

            iteration_counter += 1

        # end of inner loop – final check for the last point of this restart
        J = _exact_objective(a, n)
        if J < best_J - 1e-12:
            best_J = J
            best_vec = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Post‑processing: make absolutely sure the vector satisfies
    #    the constraints of ``evaluate_sequence``.
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # optional sanity print using the provided evaluate_sequence (if available)
    try:
        from __main__ import evaluate_sequence  # the function is injected by the runner
        val = evaluate_sequence(best_vec.tolist())
        print(f"Final evaluate_sequence value = {val:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 5. Return the best solution as a plain Python list.
    # --------------------------------------------------------------
    return [float(v) for v in best_vec]
