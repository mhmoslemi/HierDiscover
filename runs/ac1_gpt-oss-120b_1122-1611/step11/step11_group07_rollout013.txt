"""
Hybrid smooth‑max gradient descent with simplex projection.

* Starts from the best known sequence (global ``height_sequence_1``) if
  available, otherwise from a uniform vector.
* Uses a differentiable soft‑max approximation of the ``max(conv)`` term.
  For a smoothing parameter α the soft‑max gradient is a weighted sum of
  the individual convolution‑gradients and can be computed by a single
  FFT‑based convolution.
* An Adam‑style adaptive step is applied to the gradient; the step is
  projected onto the probability simplex (∑a_i = 1, a_i ≥ 0).
* A simple simulated‑annealing‐style acceptance rule with an exponential
  temperature schedule (starting at the current best value and cooling
  towards 1e‑8) allows occasional uphill moves.
* If no improvement is seen for 20 s a small random jitter centred on
  the best solution is performed to escape plateaus.
* The whole optimisation runs for the allotted time budget (≈ 1000 s)
  and returns the best vector found, clipped to the allowed range
  [0, 1000] and re‑normalised.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | sum(x)=1, x_i ≥ 0 }.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries equal or zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT of length ``fft_len``."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of ``x`` and ``y`` using an FFT of length ``L``."""
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[: len(x) + len(y) - 1]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute the objective J = 2·n·max(conv(a,a)) for a vector already on the
    simplex (∑a = 1).  The convolution is performed via FFT for speed.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – reuse a known good length if possible.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep a reasonable upper bound to avoid excessive FFT sizes
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # for a*a convolution
    L_grad = _next_pow2(conv_len + n - 1)        # for the weighted gradient

    # ------------------------------------------------------------------
    # 2. Initialise the search vector on the simplex.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            # truncate or pad with zeros to fit the chosen size
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
        s = a0.sum()
        if s == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= s
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)
    cur_J = best_J
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    total_time = deadline - start_time

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12                # initial learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max smoothing schedule
    alpha_start, alpha_end = 12.0, 120.0

    # Temperature schedule (simulated annealing)
    T_start = best_J
    T_end = 1e-8

    last_improve = start_time
    iteration = 1

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # 3.1 Temperature and smoothing parameters (time‑based)
        # --------------------------------------------------------------
        T = T_start * (T_end / T_start) ** (elapsed / total_time)
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total_time)

        # --------------------------------------------------------------
        # 3.2 Stagnation handling – jitter after 20 s without improvement
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            a = proj_simplex(best_a * noise)
            cur_J = compute_J(a, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – current J = {cur_J:.6f}")

        # --------------------------------------------------------------
        # 3.3 Compute convolution and soft‑max weights
        # --------------------------------------------------------------
        conv_a = _conv_full(a, fft_len, conv_len)
        max_conv = conv_a.max()
        # soft‑max (weights sum to 1)
        shift = conv_a - max_conv                     # ≤ 0
        w_raw = np.exp(alpha * shift)
        w_sum = w_raw.sum()
        if w_sum == 0.0:                              # fallback – active set
            # use the sub‑gradient on the active set (same as previous code)
            tol = max(1e-12, max_conv * 1e-12)
            active = np.where(conv_a >= max_conv - tol)[0]
            grad_smooth = np.zeros_like(a)
            for p in active:
                start = max(0, p - (n - 1))
                end = min(p, n - 1)
                idx = np.arange(start, end + 1)
                grad_smooth[idx] += 2.0 * a[p - idx]
        else:
            w = w_raw / w_sum
            # gradient of the smooth max: 2 * sum_k w_k * a_{k-i}
            a_rev = a[::-1]
            conv2 = conv_fft(w, a_rev, L_grad)        # length 3n‑2
            grad_smooth = 2.0 * conv2[n - 1:n - 1 + n]   # shape (n,)

        # Gradient of the smooth objective J_smooth = 2*n*softmax
        grad_J = 2.0 * n * grad_smooth

        # --------------------------------------------------------------
        # 3.4 Adam update and projection onto the simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_J
        v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_proposed = a - step
        a_proposed = proj_simplex(a_proposed)

        # --------------------------------------------------------------
        # 3.5 Occasional exchange move (large‑scale re‑balancing)
        # --------------------------------------------------------------
        if rng.random() < 0.08:                       # 8 % chance
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(a_proposed[i], rng.random() * 0.008)   # ≤ 0.8 % of mass
            a_proposed[i] -= delta
            a_proposed[j] += delta
            a_proposed = proj_simplex(a_proposed)

        # --------------------------------------------------------------
        # 3.6 Evaluate the proposal
        # --------------------------------------------------------------
        new_J = compute_J(a_proposed, n, fft_len, conv_len)

        # --------------------------------------------------------------
        # 3.7 Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_proposed
            cur_J = new_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
