"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm (≈ 1000 s budget)
--------------------------------
1. Choose a problem size n (use the length of the previously found
   `height_sequence_1` if it exists, otherwise a power‑of‑two ≤ 8192).
2. Initialise a set of diverse probability vectors on the simplex,
   keep the one with the smallest current objective J = 2·n·max(conv).
3. Run a long Adam‑style optimisation on a smooth approximation of the
   max‑operator:
       * soft‑max weight w_k ∝ exp(β·conv_k), β increases from 1 to
         ≈2000 during the run;
       * gradient = 2·corr(w, a) (computed by FFT);
       * Adam steps are projected back onto the probability simplex.
   An adaptive temperature‑controlled Metropolis rule allows occasional
   uphill moves.
4. If the optimiser stalls for >30 s a small multiplicative jitter is
   applied to escape the basin.
5. After the main loop a cheap local‑search (pairwise mass swaps) is
   performed with a simple simulated‑annealing schedule to polish the
   result.
6. The best vector found is clipped to the limits of `evaluate_sequence`
   and returned as a plain Python list.

All heavy work is done with real FFTs (O(n log n)), so even n≈8000
fits easily into the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    weight length = 2·n‑1, a length = n.
    Returns a gradient vector of length n.
    """
    # reverse a for correlation
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

    best_J = np.inf
    for cand in init_cands:
        J = _objective(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam optimisation (soft‑max smoothing + Metropolis acceptance)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 40.0)   # keep 40 s for polishing
    total_time = deadline - start_time

    # Adam hyper‑parameters
    lr0 = 0.6          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Soft‑max schedule
    beta_start = 1.0
    beta_end = 2000.0

    # Temperature schedule for Metropolis (linear decay)
    T_initial = best_J * 0.1   # 10 % of the current objective
    T_final = 1e-8

    cur_vec = best_vec.copy()
    cur_J = best_J
    best_vec = cur_vec.copy()
    best_J = cur_J

    m = np.zeros_like(cur_vec)
    v = np.zeros_like(cur_vec)

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / total_time  # in [0,1]

        # ----- temperature (linear cooling) -----
        T = (T_initial * (deadline - now) + T_final * (now - start_time)) / total_time

        # ----- compute convolution & soft‑max weights -----
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ----- update global best -----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation handling (jitter after 30 s without improvement) -----
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = _objective(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ----- soft‑max weight vector -----
        beta = beta_start * (beta_end / beta_start) ** t_frac
        # stable soft‑max
        shifted = conv - max_conv   # max_conv is the max entry
        w_exp = np.exp(beta * shifted)
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)

        # ----- Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        lr = lr0 * (1.0 - t_frac) + 1e-3   # slowly decay learning rate
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        proposal = cur_vec + step
        proposal = proj_simplex(proposal)

        # ----- evaluate proposal -----
        prop_J = _objective(proposal, n, fft_len, conv_len)

        # ----- Metropolis acceptance -----
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = proposal
            cur_J = prop_J

    # ------------------------------------------------------------------
    # 5. Local polishing: pairwise mass swaps with simulated annealing
    # ------------------------------------------------------------------
    polish_deadline = start_time + float(budget_s) - 2.0  # leave a couple seconds for finalisation
    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    # Simple temperature schedule for polishing
    T0_polish = cur_J * 0.01
    T_end_polish = 1e-9

    while time.monotonic() < polish_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / total_time
        T = T0_polish * (1.0 - t_frac) + T_end_polish * t_frac

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # move a small amount of mass from i to j
        eps = rng.random() * cur_vec[i] * 0.5  # at most half of the donor mass
        if eps <= 0.0:
            continue

        cand = cur_vec.copy()
        cand[i] -= eps
        cand[j] += eps

        # due to rounding we may get negative tiny values – project quickly
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        if cand_J < cur_J:
            cur_vec = cand
            cur_J = cand_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{elapsed:6.2f}s] Polishing improvement: J = {best_J:.6f}")
        else:
            # Metropolis‑style uphill acceptance
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur_vec = cand
                cur_J = cand_J

    # ------------------------------------------------------------------
    # 6. Final safety normalisation & conversion to Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
