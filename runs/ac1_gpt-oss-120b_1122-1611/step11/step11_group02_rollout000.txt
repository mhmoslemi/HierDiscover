"""
Differential Evolution on the probability simplex with final local refinement.

The algorithm works on a probability vector `a` (non‑negative, sum = 1) of
length *n*.  For such a vector the evaluation function reduces to  

    J(a) = 2·n·max(conv(a, a))

where `conv` is the discrete linear convolution.  The goal is to minimise
`J`.  Differential Evolution (DE) is well‑suited for this highly non‑convex
problem because it explores the space with cheap, population‑based
updates while respecting the simplex constraint via a Euclidean projection.
A small final random‑noise search around the best solution refines the
result.

Key components
--------------
* `proj_simplex(v)`: Euclidean projection onto the probability simplex.
* `conv_full_fft(a, fft_len, conv_len)`: fast full convolution using a real
  FFT (size is a power of two for speed).
* `eval_J(a)`: returns the objective `J(a)` for a probability vector.
* DE loop: population size ≈ 15 – 20, mutation factor `F` (≈0.5) and
  crossover probability `CR` (≈0.9).  After each generation the best
  candidate is kept, and if no improvement occurs for 30 s the half of the
  population is reinitialised with fresh Dirichlet draws.
* Final fine‑grained random perturbations (multiplicative log‑normal noise)
  around the best vector for a few seconds.

The routine respects the 1000 s time‑budget, printing progress and finally
returning the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def eval_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute the objective J = 2·n·max(conv(a, a)).
    The vector `a` is assumed to lie on the simplex (sum = 1, a_i ≥ 0).
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The method uses Differential Evolution on the
    probability simplex, followed by a short local random search.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – reuse a previous construction if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n0 = old.size
            # Clamp to a comfortable range for FFT speed/memory
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # for a*a
    # ------------------------------------------------------------------
    # 2. Initialise a modest population on the simplex.
    # ------------------------------------------------------------------
    pop: list[np.ndarray] = []
    pop_J: list[float] = []

    # Helper to add a candidate if we still need more members.
    def add_candidate(v: np.ndarray):
        v = np.maximum(v, 0.0)
        v = proj_simplex(v)
        J = eval_J(v, n, fft_len, conv_len)
        pop.append(v)
        pop_J.append(J)

    # Uniform distribution.
    add_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # Geometric decays.
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        add_candidate(v)

    # Linear ramps (decreasing / increasing).
    add_candidate(np.arange(n, 0, -1, dtype=np.float64))
    add_candidate(np.arange(1, n + 1, dtype=np.float64))

    # Gaussian bells (different widths).
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        add_candidate(v)

    # Dirichlet draws with various concentrations.
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            v = rng.dirichlet(np.full(n, alpha))
            add_candidate(v)

    # Include the previous best construction if present (truncated/padded).
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if v.size != n:
                if v.size > n:
                    v = v[:n]
                else:
                    v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            add_candidate(v)
        except Exception:
            pass

    # Fill up to a target population size (≈15‑20).
    target_pop = max(15, min(30, n // 300))
    while len(pop) < target_pop:
        v = rng.dirichlet(np.ones(n))
        add_candidate(v)

    popsize = len(pop)

    # ------------------------------------------------------------------
    # 3. Track the global best.
    # ------------------------------------------------------------------
    best_idx = int(np.argmin(pop_J))
    best_vec = pop[best_idx].copy()
    best_J = pop_J[best_idx]
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # leave margin
    last_improve = start_time

    print(f"[0.00s] Initial best J = {best_J:.6f} (popsize = {popsize}, n = {n})")

    # ------------------------------------------------------------------
    # 4. Differential Evolution loop.
    # ------------------------------------------------------------------
    F = 0.5            # differential weight
    CR = 0.9           # crossover probability
    generation = 0

    while time.monotonic() < deadline:
        # One generation – try to improve each target vector.
        for i in range(popsize):
            # Choose three distinct indices different from i.
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            donor = pop[a] + F * (pop[b] - pop[c])
            donor = np.maximum(donor, 0.0)
            donor = proj_simplex(donor)

            # Binomial crossover.
            j_rand = rng.integers(n)
            mask = (rng.random(n) < CR) | (np.arange(n) == j_rand)
            trial = np.where(mask, donor, pop[i])
            trial = proj_simplex(trial)

            trial_J = eval_J(trial, n, fft_len, conv_len)

            # Selection.
            if trial_J <= pop_J[i]:
                pop[i] = trial
                pop_J[i] = trial_J
                if trial_J < best_J - 1e-12:
                    best_J = trial_J
                    best_vec = trial.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        generation += 1

        # Optional gradual decay of the mutation factor (helps fine‑tuning).
        F = 0.5 * (1.0 - generation / (deadline - start_time + 1.0))

        # ------------------------------------------------------------------
        # 4.1 Stagnation handling – re‑initialise part of the population.
        # ------------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > 30.0:
            reinit_cnt = max(1, popsize // 2)
            for _ in range(reinit_cnt):
                idx = rng.integers(popsize)
                v = rng.dirichlet(np.ones(n))
                pop[idx] = v
                pop_J[idx] = eval_J(v, n, fft_len, conv_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Re‑initialised {reinit_cnt} individuals")

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random search around the best solution.
    # ------------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fine_deadline:
        # small multiplicative log‑normal perturbation
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_J = eval_J(cand, n, fft_len, conv_len)
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Normalise and return the result (ensure legal range).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s

    return [float(v) for v in best_vec]
