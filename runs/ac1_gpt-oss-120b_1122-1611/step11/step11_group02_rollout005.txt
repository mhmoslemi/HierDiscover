"""
Hybrid stochastic optimiser for the non‑negative step‑function bound.

The routine builds an initial simplex‑normalised candidate (uniform,
geometric, Dirichlet draws or the supplied `height_sequence_1`) and then
alternates two complementary phases:

1. **Smooth‑max exponentiated‑gradient phase** – a soft‑max surrogate of
   the sup‑norm is differentiable; we take an exponentiated‑gradient step
   with a decaying learning rate while the temperature *p* of the
   soft‑max grows from 1 to 200.  This quickly drives the large peaks of
   the convolution down.

2. **Simulated‑annealing refinement** – after the first phase a classic
   Metropolis‑Hastings walk on the probability simplex is performed.
   Proposals are convex combinations of the current best vector and a
   Dirichlet‑sampled noise vector; the temperature decays exponentially.
   Interleaved occasional gradient refinements keep the walk directed
   towards the true (non‑smoothed) objective.

A short *fine‑grained* random‑perturbation search is finally carried out
to squeeze a little extra improvement.  All vectors stay on the simplex,
so the objective
    J(a)=2·n·max(conv(a,a))/ (∑a)²
is scale‑invariant and can be evaluated cheaply via FFT.

The function returns the best sequence found before the time budget
expires.  It works for any length (the code automatically picks the
length of `height_sequence_1` if present, otherwise defaults to 2048).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def gradient_smooth(a: np.ndarray, conv: np.ndarray,
                    p: float, fft_len_g: int, conv_len: int) -> np.ndarray:
    """
    Gradient of the smooth max
        φ(a) = log‑sum‑exp(p·conv(a,a)) / p .
    The gradient is
        g = 2 * (w * a_rev)   (convolution),
    where w_k = exp(p·conv_k) / Σ_j exp(p·conv_j) and a_rev is a reversed.
    """
    # soft‑max weights (shifted for numerical stability)
    w = np.exp(p * (conv - conv.max()))
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    a_rev = a[::-1]

    # convolution w (length 2n‑1) with a_rev (length n)
    grad_full = np.fft.irfft(
        np.fft.rfft(w, fft_len_g) * np.fft.rfft(a_rev, fft_len_g),
        n=fft_len_g
    )
    # slice the part that corresponds to the gradient (length n)
    grad = 2.0 * grad_full[conv_len - 1: conv_len - 1 + a.size]
    return grad


def compute_J(a: np.ndarray) -> float:
    """
    Direct evaluation of the objective without the security wrappers.
    a must be a 1‑D non‑negative array; the function returns np.inf for
    infeasible inputs (sum < 0.01).
    """
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = conv_full_fft(a, fft_len, conv_len)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Stochastic search for a non‑negative step‑function that minimises the
    evaluation function.  The implementation follows the hybrid strategy
    described in the module docstring and respects the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size and build a portfolio of start points.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        n0 = base.size
        # Keep the length reasonable (too large → very slow history)
        n = max(256, min(n0, 4096))
        if n != n0:
            # truncate or pad
            if n0 > n:
                base = base[:n]
            else:
                pad = np.zeros(n - n0, dtype=np.float64)
                base = np.concatenate([base, pad])
        base /= base.sum() if base.sum() > 0 else 1.0 / n
    else:
        n = 2048
        base = None

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fft_len_g = _next_pow2(conv_len + n - 1)

    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws with different concentration
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # previous best if available
    if base is not None:
        init_cands.append(base)

    # --------------------------------------------------------------
    # 2. Pick the best initial candidate (cheap evaluation).
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Phase‑1 : smooth‑max exponentiated‑gradient optimisation.
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)      # keep 5 s safety margin
    phase1_end = start + (deadline - start) * 0.55          # 55 % of the budget

    # schedule parameters
    p_start, p_end = 1.0, 200.0          # soft‑max temperature
    eta0 = 0.5                           # base learning rate for EG update
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start
    it = 0

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start
        frac = (now - start) / (phase1_end - start)

        # temperature for the smooth max
        p = p_start + (p_end - p_start) * frac

        # convolution and objective (exact, not smoothed)
        conv = conv_full_fft(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv / (cur_vec.sum() ** 2))

        # track global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best after EG J = {best_J:.6f}")

        # stagnation handling – add a gentle jitter if nothing improves for 25 s
        if now - last_improve > 25.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = proj_simplex(best_vec * noise)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter after stagnation")

        # gradient of the smoothed sup‑norm
        grad = gradient_smooth(cur_vec, conv, p, fft_len_g, conv_len)

        # step size η ∝ 1/√t  (with clipping to avoid overflow)
        eta = eta0 / np.sqrt(it + 1.0)
        max_abs = np.max(np.abs(grad))
        if max_abs * eta > 30.0:          # keep exp(-η·grad) numerically safe
            eta = 30.0 / max_abs

        # exponentiated‑gradient proposal
        factor = np.exp(-eta * grad)
        proposal = cur_vec * factor
        proposal = proj_simplex(proposal)

        # evaluate proposal (exact objective)
        prop_conv = conv_full_fft(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max() / (proposal.sum() ** 2))

        # accept only improvements (no Metropolis noise in this phase)
        if prop_J < cur_J:
            cur_vec = proposal
            cur_J = prop_J

        it += 1

    # --------------------------------------------------------------
    # 4. Phase‑2 : simulated‑annealing walk with occasional gradient nudges.
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    global_best_vec = best_vec.copy()
    global_best_J = best_J
    last_imp = time.monotonic()
    it2 = 0

    # temperature schedule (exponential decay)
    T0 = 0.35
    Tmin = 1e-5
    total_phase2 = deadline - time.monotonic()
    phase2_start = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        frac = elapsed / total_phase2
        T = max(Tmin, T0 * (0.5 ** frac))   # halve every half‑life

        # convex‑combination proposal: (1‑α)*cur + α*noise,  α≈T
        alpha = T
        noise = rng.dirichlet(np.full(n, 0.5))
        cand = (1.0 - alpha) * cur_vec + alpha * noise
        cand = proj_simplex(cand)

        cand_J = compute_J(cand)

        # Metropolis acceptance
        if cand_J < cur_J:
            accept = True
        else:
            # probability exp(−Δ/T)
            accept = rng.random() < np.exp(-(cand_J - cur_J) / T)

        if accept:
            cur_vec = cand
            cur_J = cand_J

        # update global best
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_vec = cur_vec.copy()
            last_imp = now
            print(f"[{now - start:6.2f}s] SA new global best J = {global_best_J:.6f}")

        # stagnation – restart from a fresh random Dirichlet if no progress for 45 s
        if now - last_imp > 45.0:
            cur_vec = rng.dirichlet(np.full(n, 0.5))
            cur_vec = proj_simplex(cur_vec)
            cur_J = compute_J(cur_vec)
            last_imp = now
            print(f"[{now - start:6.2f}s] SA restart due to stagnation")

        # occasional cheap gradient refinement (helps escape shallow plateaus)
        if it2 % 3000 == 0:
            conv = conv_full_fft(cur_vec, fft_len, conv_len)
            grad = gradient_smooth(cur_vec, conv, p_end, fft_len_g, conv_len)
            eta = 0.15
            factor = np.exp(-eta * grad)
            refined = proj_simplex(cur_vec * factor)
            refined_J = compute_J(refined)
            if refined_J < cur_J:
                cur_vec = refined
                cur_J = refined_J
                if cur_J < global_best_J:
                    global_best_J = cur_J
                    global_best_vec = cur_vec.copy()
                    print(f"[{time.monotonic() - start:6.2f}s] Refine improved J = {global_best_J:.6f}")

        it2 += 1

    # --------------------------------------------------------------
    # 5. Final fine‑grained random perturbations around the best.
    # --------------------------------------------------------------
    fin_start = time.monotonic()
    fin_deadline = fin_start + min(30.0, (deadline - phase2_start) * 0.06)
    while time.monotonic() < fin_deadline:
        noise = np.exp(rng.normal(0.0, 0.018, size=n))
        cand = proj_simplex(global_best_vec * noise)
        cand_J = compute_J(cand)
        if cand_J < global_best_J:
            global_best_J = cand_J
            global_best_vec = cand
            print(f"[Fine] Improved J = {global_best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best sequence (already non‑negative and normalised).
    # --------------------------------------------------------------
    # Guarantee the output respects the security clamps of evaluate_sequence
    global_best_vec = np.clip(global_best_vec, 0.0, 1000.0)
    total = global_best_vec.sum()
    if total == 0.0:
        global_best_vec[:] = 1.0 / n
    else:
        global_best_vec /= total

    return [float(v) for v in global_best_vec]
