def propose_candidate(seed=..., budget_s=1000):
    rng = np.random.default_rng(seed)

    # Choose n: maybe 3809 default or try a few lengths
    n = 3809

    # Initialize with previously found height_sequence_1 if present; else random Dirichlet
    if 'height_sequence_1' in globals():
        vec = np.asarray(globals()['height_sequence_1'], dtype=float)
    else:
        vec = rng.dirichlet(np.ones(n))

    # Clip to [0, 1000] and renormalize
    vec = np.clip(vec, 0.0, 1000.0)
    if vec.sum() == 0:
        vec = np.full(n, 1.0/n)
    else:
        vec = vec / vec.sum()

    # Precompute FFT parameters
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    # define function to compute J quickly
    def compute_J(v):
        max_c = full_conv_max(v, fft_len, conv_len)
        return 2.0 * n * max_c

    best_vec = vec.copy()
    best_J = compute_J(best_vec)
    cur_vec = best_vec.copy()
    cur_J = best_J

    start = time.time()
    deadline = start + budget_s - 5.0 # small safety

    # Define schedule for temperature
    T0 = best_J * 0.2
    T_end = 1e-9

    iteration = 0
    while time.time() < deadline:
        # Determine temperature linearly decreasing
        elapsed = time.time() - start
        T = (T0 * (1 - elapsed / (deadline - start)) + T_end * (elapsed / (deadline - start)))
        # Sample a move: choose k indices to modify
        k = rng.integers(2, 5) # 2-4 indices
        idx = rng.choice(n, size=k, replace=False)
        # compute net mass available to be redistributed
        # propose perturbation direction via random Gaussian on log-space
        delta = rng.normal(0.0, 0.05, size=k)
        # apply log-space scaling: new_vec[idx] = cur_vec[idx] * exp(delta)
        proposal = cur_vec.copy()
        mul = np.exp(delta)
        proposal[idx] *= mul
        # Clip to nonnegative
        proposal = np.clip(proposal, 0.0, None)

        # Renormalize to sum 1
        s = proposal.sum()
        if s == 0:
            continue
        proposal /= s

        # Compute J
        prop_J = compute_J(proposal)

        if prop_J < cur_J or rng.random() < np.exp(-(prop_J - cur_J) / max(T, 1e-12)):
            cur_vec = proposal
            cur_J = prop_J
            if cur_J < best_J:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{elapsed:.2f}s] New best J = {best_J:.6f}")

        # Occasionally random restart: jitter around best
        if iteration % 2000 == 0 and iteration > 0:
            # small random noise around best
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            cur_vec = best_vec * noise
            cur_vec /= cur_vec.sum()
            cur_J = compute_J(cur_vec)

        iteration += 1

    # return best vector
    return [float(v) for v in best_vec]

def subgradient_descent(start_vec, n, fft_len, conv_len, time_budget, rng):
    vec = start_vec.copy()
    best_vec = vec.copy()
    best_J = compute_J(vec,...)
    start_time = time.time()
    iteration = 0
    # step schedule param
    eta0 = 0.2
    while time.time() - start_time < time_budget:
        # compute conv max and active set
        conv_vals = np.fft.irfft(np.fft.rfft(vec, n=fft_len) ** 2, n=fft_len)[:conv_len]
        max_c = conv_vals.max()
        tol = max_c * 1e-12
        active = np.where(conv_vals >= max_c - tol)[0]

        # compute gradient
        grad = np.zeros_like(vec)
        for p in active:
            start_i = max(0, p - (n - 1))
            end_i = min(p, n - 1)
            idx = np.arange(start_i, end_i + 1)
            grad[idx] += 2.0 * vec[p - idx]

        # step size
        eta = eta0 / np.sqrt(iteration + 1.0)
        # gradient descent step on simplex
        candidate = proj_simplex(vec - eta * grad)
        # evaluate
        Jcand = compute_J(candidate, n, fft_len, conv_len)
        # accept if improvement else maybe keep as is?
        if Jcand < best_J:
            best_J = Jcand
            best_vec = candidate.copy()
            # Optionally set vec = candidate
            vec = candidate
        else:
            # optionally with probability? but not needed
            vec = candidate

        iteration += 1
    return best_vec, best_J

"""
Hybrid sub‑gradient optimiser for the step‑function inequality.

* Generates a diverse portfolio of candidate vectors (uniform, geometric,
  power‑law, Gaussian, classic window functions, random Dirichlet draws,
  and the previously discovered construction if present) and keeps the best
  one on the probability simplex (∑a_i = 1).

* Refines the best vector with a projected sub‑gradient method.
  The sub‑gradient of J = 2·n·max(conv(a,a)) is obtained from the active
  indices of the full linear auto‑convolution.
  An additive update with a diminishing step‑size η_t = η₀ / √t
  (η₀≈0.2) together with a Euclidean projection onto the simplex provides
  a monotone descent, while occasional random jitter (multiplicative noise)
  helps to escape plateaux.

* The routine runs until the supplied time budget (≈1000 s) expires and
  returns the best sequence discovered.
"""

import time
import numpy as np
from typing import List, Tuple


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0,  ∑ w_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are the same; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum value of the full linear convolution vec * vec."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Return the objective J = 2·n·max(conv(vec,vec)).
    The vector is assumed to be on the simplex (sum = 1).
    """
    return 2.0 * n * full_conv_max(vec, fft_len, conv_len)


def generate_initial_pool(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """
    Produce a list of diverse initial probability vectors.
    All vectors are already normalised to sum‑to‑one.
    """
    pool: List[np.ndarray] = []

    # uniform
    pool.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (both directions)
    for r in [0.905, 0.92, 0.94, 0.96, 0.98, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        pool.append(v)
        pool.append(v[::-1])

    # power‑law shapes
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1, dtype=np.float64)
        v = 1.0 / (i ** p)
        v /= v.sum()
        pool.append(v)

    # Gaussian bells centred in the middle
    xs = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    for sigma in [n / 12.0, n / 8.0, n / 6.0, n / 4.0]:
        v = np.exp(-0.5 * (xs / sigma) ** 2)
        v /= v.sum()
        pool.append(v)

    # classic window functions
    windows = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),
    }
    for w in windows.values():
        if w.sum() > 0.0:
            pool.append(w / w.sum())

    # Kaiser windows (several betas)
    for beta in [0.5, 2.0, 5.0, 10.0, 15.0]:
        w = np.kaiser(n, beta)
        if w.sum() > 0.0:
            pool.append(w / w.sum())

    # Tukey windows (α from 0 to 1)
    for alpha in np.linspace(0.0, 1.0, 11):
        t = np.arange(n, dtype=np.float64) / (n - 1)
        w = np.ones_like(t)
        if alpha > 0.0:
            first = t < alpha / 2.0
            w[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
            last = t > 1.0 - alpha / 2.0
            w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
        pool.append(w / w.sum())

    # random Dirichlet draws with various concentration parameters
    for conc in [0.3, 0.6, 1.0, 2.0]:
        for _ in range(12):
            pool.append(rng.dirichlet(np.full(n, conc)))

    # previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = 1.0 / n
        else:
            v /= v.sum()
        pool.append(v)

    return pool


def subgradient_descent(
    start_vec: np.ndarray,
    n: int,
    fft_len: int,
    conv_len: int,
    time_budget: float,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient optimisation starting from *start_vec*.
    Returns the best vector found and its objective value.
    """
    vec = start_vec.copy()
    best_vec = vec.copy()
    best_J = compute_J(vec, n, fft_len, conv_len)

    start_time = time.monotonic()
    deadline = start_time + time_budget
    iteration = 0
    last_improve = start_time
    eta0 = 0.20                     # initial step‑size multiplier

    while time.monotonic() < deadline:
        # full linear convolution (via FFT)
        A = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]

        max_c = conv.max()
        cur_J = 2.0 * n * max_c

        # record improvement
        if cur_J + 1e-13 < best_J:
            best_J = cur_J
            best_vec = vec.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Sub‑grad new best J = {best_J:.6f}")

        # stagnation handling – jitter around the best after 20 s without progress
        now = time.monotonic()
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.025, size=n))
            vec = best_vec * noise
            vec = proj_simplex(vec)
            last_improve = now
            continue

        # --- active set of the max convolution ---
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv >= max_c - tol)[0]

        # --- sub‑gradient of max(conv) at the active set ---
        grad = np.zeros_like(vec)
        for p in active:
            # indices i with 0 ≤ i < n and 0 ≤ p‑i < n
            i_start = max(0, p - (n - 1))
            i_end = min(p, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad[idx] += 2.0 * vec[p - idx]

        # --- additive update on the simplex ---
        eta = eta0 / np.sqrt(iteration + 1.0)
        proposal = proj_simplex(vec - eta * grad)

        # occasional random blend with a fresh Dirichlet draw
        if rng.random() < 0.07:
            eps = rng.uniform(0.0, 0.12)
            rand_vec = rng.dirichlet(np.ones(n))
            proposal = proj_simplex((1.0 - eps) * proposal + eps * rand_vec)

        vec = proposal
        iteration += 1

    return best_vec, best_J


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered within the
    allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a power‑of‑two → fast FFTs

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Global best bookkeeping.
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    # ------------------------------------------------------------------
    # 3. Generate and evaluate a broad portfolio of candidates.
    # ------------------------------------------------------------------
    init_pool = generate_initial_pool(n, rng)

    for idx, cand in enumerate(init_pool):
        J = compute_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
            print(f"[init {idx:03d}] New best J = {best_J:.6f}")

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine the best candidate with projected sub‑gradient descent.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    remaining = max(0.0, float(budget_s) - (time.monotonic() - start_time) - 5.0)

    refined_vec, refined_J = subgradient_descent(
        start_vec=best_vec,
        n=n,
        fft_len=fft_len,
        conv_len=conv_len,
        time_budget=remaining,
        rng=rng,
    )

    if refined_J < best_J:
        best_J = refined_J
        best_vec = refined_vec

    # ------------------------------------------------------------------
    # 5. Finalise – clip, renormalise and convert to plain list.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"[final] Best J = {best_J:.6f}, length = {len(best_vec)}")
    return [float(v) for v in best_vec]
