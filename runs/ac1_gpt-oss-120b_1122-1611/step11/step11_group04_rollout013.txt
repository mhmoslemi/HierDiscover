"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

The algorithm combines three ideas:

1. **Simplex parametrisation** – the vector is always normalised
   (∑a_i = 1).  The objective then reduces to ``J = 2·n·max(conv(a,a))``.

2. **Smoothed max‑convolution** – instead of the non‑differentiable
   ``max`` we minimise the smooth approximation  
   ``f_λ(a)=log‑sum‑exp(λ·conv) / λ``.
   Its gradient can be written as a convolution and is obtained
   efficiently with FFTs.

3. **Exponentiated‑gradient descent on the simplex** – a multiplicative
   update ``a ← a·exp(-α·grad)`` keeps the vector non‑negative and
   preserves the L₁‑norm after a normalisation step.
   The step size ``α`` shrinks like 1/√t and the smoothing parameter
   ``λ`` is increased gradually (homotopy).  A light Metropolis
   acceptance rule and occasional “jitter’’ restarts avoid getting
   trapped in local minima.

The routine runs for at most ``budget_s`` seconds and returns the
best vector seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two  ≥  x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution of *a* and *b* using a real FFT.
    The caller must slice the result to the required length:
    ``len(a)+len(b)-1``.
    """
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    return np.fft.irfft(A * B, n=fft_len)


def _conv_self(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Convolution of a vector with itself, trimmed to ``conv_len``."""
    return _conv_fft(a, a, fft_len)[:conv_len]


def _conv_w_a_rev(w: np.ndarray, a_rev: np.ndarray, fft_len: int, out_len: int) -> np.ndarray:
    """Convolution of the weight vector w with the reversed a."""
    return _conv_fft(w, a_rev, fft_len)[:out_len]


def _objective(a: np.ndarray, n: int, conv_len: int, fft_len: int) -> float:
    """
    Compute the exact objective used inside ``evaluate_sequence`` for a
    normalised vector (∑a_i = 1):  J = 2·n·max(conv(a,a)).
    """
    conv = _conv_self(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- restart (jitter) if we have been stuck too long -----
        if now - last_improve > 12.0:
            # multiplicative Gaussian noise around the best point
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            cur_vec = best_vec * noise
            cur_vec /= cur_vec.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ----- smooth approximation of max -----
        # weights w_k = exp(λ·conv_k) / Σ exp(λ·conv)
        # to keep numbers stable we shift by the maximum
        shift = conv.max()
        w = np.exp(lam * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / conv_len
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        # ∂f/∂a_i = 2 * Σ_k w_k a_{k-i}
        a_rev = cur_vec[::-1]
        conv_rev_full = _conv_w_a_rev(w, a_rev, fft_len_grad, grad_len)
        grad = 2.0 * conv_rev_full[n - 1: n - 1 + n]

        # ----- propose a new point (exponentiated gradient step) -----
        step = alpha0 / np.sqrt(iteration + 1.0)   # decreasing step size
        factor = np.exp(-step * grad)            # multiplicative update
        proposal = cur_vec * factor
        s = proposal.sum()
        if s == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= s

        # ----- evaluate proposal -----
        prop_conv = _conv_self(proposal, fft_len_conv, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ----- Metropolis acceptance (linear cooling) -----
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ----- occasional cheap mass‑transfer move (exploration) -----
        if rng.random() < 0.12:
            # move a small fraction from a high‑gradient index to a low‑gradient one
            if grad.sum() > 0.0:
                donor = rng.choice(n, p=grad / grad.sum())
            else:
                donor = rng.integers(n)
            inv = grad.max() - grad + 1e-15
            receiver = rng.choice(n, p=inv / inv.sum())
            delta = min(cur_vec[donor], 0.04 * cur_vec[donor])  # up to 4 %
            cur_vec[donor] -= delta
            cur_vec[receiver] += delta
            cur_vec /= cur_vec.sum()   # re‑normalise (preserves sum)

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution found (final safety clipping)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"Finished – best J = {best_J:.6f} after {iteration} iterations")
    return [float(v) for v in best_vec]
