"""
Propose a step‐function by a simple projected sub‑gradient descent on the
maximum entry of the a *a convolution.

The evaluation function is scale invariant, so we normalise all candidates to
sum = 1 (the simplex).  The objective to minimise is

    J(a) = 2·n·max_k (a * a)[k]

where “*” denotes linear convolution.  For a fixed k the convolution entry is a
quadratic form.  Its (sub‑)gradient with respect to a_i is

    ∂/∂a_i (a*a)[k] = 2·a_{k‑i}      (if the index is in range)

Thus a sub‑gradient of the max‑operator is obtained by summing the above for
all indices k that attain (or are within a tiny tolerance of) the current
maximum.  We perform an exponentiated‑gradient (multiplicative) update,
project back onto the probability simplex and repeat.  If no improvement is
seen for a while we jitter the current iterate (multiply by log‑normal noise)
to escape possible plateaus.

The method works directly on the full objective (no soft‑max smoothing) and
has very little overhead – each iteration costs one FFT‑based convolution.
The routine respects the total time budget (≈1000 s) and returns the best
vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

    for sigma in (n / 10.0, n / 5.0, n / 3.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        if v.sum() > 0:
            v /= v.sum()
        init_cands.append(v)

    # Cosine‑shaped bump (always non‑negative)
    v = np.cos(np.pi * (x - centre) / n)          # values in [-1,1]
    v = v - v.min()                               # shift to ≥0
    if v.sum() > 0:
        v = v / v.sum()
    init_cands.append(v)

    # Half‑sine wave (zero at the ends)
    v = np.sin(np.pi * x / (n - 1))
    v = v - v.min()
    if v.sum() > 0:
        v = v / v.sum()
    init_cands.append(v)

    # previous construction, if any
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = np.full(n, 1.0 / n)
        init_cands.append(v)

    # --------------------------------------------------------------
    # 4. Pick the best initial candidate (lowest J)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        if cand.shape[0] != n:
            continue
        J = compute_J(cand)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Sub‑gradient descent on max‑convolution
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_global_vec = cur_vec.copy()
    best_global_J = cur_J

    start_time = time.monotonic()
    # leave a small safety margin so that the runtime environment can finish cleanly
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # algorithmic parameters
    step0 = 0.7                     # initial step magnitude
    jitter_sigma = 0.05            # std‑dev of log‑normal jitter
    stagnation_limit = 30.0        # seconds without improvement before a restart
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- current convolution and objective ---------------------------------
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_val = conv.max()
        cur_J = 2.0 * n * max_val

        # ---- update the global best -------------------------------------------
        if cur_J < best_global_J - 1e-12:
            best_global_J = cur_J
            best_global_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_global_J:.6f}")

        # ---- early exit if we already reached the target -----------------------
        if best_global_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_global_J:.6f}")
            break

        # ---- stagnation handling ------------------------------------------------
        if now - last_improve > stagnation_limit:
            # jitter around the best known solution to escape a plateau
            noise = rng.normal(0.0, jitter_sigma, size=n)
            cur_vec = best_global_vec * np.exp(noise)
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_global_J:.6f}")

        # ---- active set (indices whose convolution value is (almost) maximal) --
        eps = max_val * 1e-8 + 1e-12
        active_idx = np.where(conv >= max_val - eps)[0]

        # ---- sub‑gradient of the max‑convolution --------------------------------
        g = np.zeros_like(cur_vec)
        for k in active_idx:
            # i ranges such that both i and k‑i lie in [0, n‑1]
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            g[idx_i] += 2.0 * cur_vec[idx_j]

        # ---- multiplicative (exponentiated‑gradient) update --------------------
        step = step0 / np.sqrt(iteration + 1.0)
        factor = np.exp(-step * g)                # always positive
        cur_vec = cur_vec * factor
        cur_vec = proj_simplex(cur_vec)           # stay on the probability simplex

        iteration += 1

    # --------------------------------------------------------------
    # 6. Return the best solution found (clip and renormalise)
    # --------------------------------------------------------------
    best_global_vec = np.clip(best_global_vec, 0.0, 1000.0)
    total = best_global_vec.sum()
    if total == 0.0:
        best_global_vec = np.full(n, 1.0 / n)
    else:
        best_global_vec = best_global_vec / total

    return [float(v) for v in best_global_vec]
