def softmax(z):
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / np.sum(e)
```

Define conv computation:

```
def conv_fft(a):
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]
```

Define gradient function:

```
def grad_phi(z, p):
    a = softmax(z)
    conv = conv_fft(a)
    max_conv = conv.max()
    exp_diff = np.exp(p * (conv - max_conv))
    sumexp = exp_diff.sum()
    phi = max_conv + np.log(sumexp) / p

    w = exp_diff / sumexp
    # compute convolution w * a_rev
    a_rev = a[::-1]
    # w conv a_rev
    W_fft = np.fft.rfft(w, fft_len_g)
    A_rev_fft = np.fft.rfft(a_rev, fft_len_g)
    grad_full = np.fft.irfft(W_fft * A_rev_fft, n=fft_len_g)
    grad_a = 2.0 * grad_full[conv_len - 1 : conv_len - 1 + a.size]

    # chain rule to compute grad_z
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return phi, grad_z

    def __init__(self, n, fft_len, conv_len, fft_len_g):

"""
Propose a step‑function that minimizes
    J(a) = 2·n·max(conv(a,a))   (with a_i ≥ 0, Σ a_i = 1).

Algorithm
---------
1. Choose a problem size n (use the length of ``height_sequence_1`` if it
   exists, otherwise a comfortable default ~2048).
2. Build a small portfolio of diverse initial probability vectors
   (uniform, geometric, Gaussian, Dirichlet draws, previous best).
3. Pick the best of those vectors w.r.t. the exact objective (computed by a
   fast FFT‑based convolution).
4. Optimise the smooth surrogate
        φ_p(a) = (log Σ_i exp(p·conv_i)) / p  ≈ max(conv)
   using the re‑parameterisation a = softmax(z).  The gradient of φ_p
   w.r.t. z is obtained analytically (softmax Jacobian × convolution
   gradient).  A multi‑stage L‑BFGS optimisation is performed with a
   gradually increasing temperature p (5 → 15 → 40 → 120 → 300).  This
   provides fast convergence and avoids being trapped in bad plateaus.
5. A brief random‑jitter phase explores the neighbourhood of the best
   solution found.
6. Return the final probability vector (already non‑negative, sums to 1).

The implementation avoids closures – a small class ``_SmoothMaxFunctor``
carries the problem data and exposes a ``__call__`` method required by
``scipy.optimize.minimize``.  All heavy work (convolutions, gradients) is
done with real FFTs for speed, so the optimiser can explore many
iterations well within the 1000 s budget.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Top‑level helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a via a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


# ----------------------------------------------------------------------
# Smooth‑max objective + gradient (re‑parameterised with softmax)
# ----------------------------------------------------------------------


class _SmoothMaxFunctor:
    """
    Callable object required by ``scipy.optimize.minimize``.
    Holds problem size and FFT parameters; ``p`` (smoothness) can be
    changed between optimisation stages via ``set_p``.
    """

    def __init__(self, n: int, fft_len: int, conv_len: int, fft_len_g: int):
        self.n = n
        self.fft_len = fft_len
        self.conv_len = conv_len
        self.fft_len_g = fft_len_g
        self.p = 10.0  # default, will be overwritten

    def set_p(self, p: float) -> None:
        """Update the soft‑max temperature."""
        self.p = float(p)

    def __call__(self, z: np.ndarray):
        """
        Returns (φ_p(a), ∇_z φ_p) where a = softmax(z).
        φ_p(a) = (log Σ_i exp(p·conv_i)) / p  (smooth approximation of max).
        """
        a = softmax(z)

        # ----- convolution a * a (full) -----
        A_fft = np.fft.rfft(a, self.fft_len)
        conv = np.fft.irfft(A_fft * A_fft, n=self.fft_len)[:self.conv_len]

        # ----- smooth max value -----
        max_conv = conv.max()
        # stable exponentials
        exp_shift = np.exp(self.p * (conv - max_conv))
        sumexp = exp_shift.sum()
        phi = max_conv + np.log(sumexp) / self.p

        # ----- gradient w.r.t. a -----
        # normalized soft‑max weights
        w = exp_shift / sumexp               # shape (conv_len,)
        w_rev = w[::-1]                      # reverse for correct cross‑correlation

        # convolution w_rev * a  (linear, via FFT)
        W_rev_fft = np.fft.rfft(w_rev, self.fft_len_g)
        A_fft_g = np.fft.rfft(a, self.fft_len_g)
        grad_full = np.fft.irfft(W_rev_fft * A_fft_g, n=self.fft_len_g)

        # indices that yield the n gradient components (see derivation)
        idx = self.conv_len - 1 - np.arange(self.n)   # descending slice
        grad_a = 2.0 * grad_full[idx]                # shape (n,)

        # ----- chain rule to obtain ∇_z φ_p -----
        dot = np.dot(a, grad_a)          # scalar = a·∇_a φ_p
        grad_z = a * (grad_a - dot)      # Jacobian of softmax times ∇_a

        return float(phi), grad_z.astype(np.float64)

        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderate concentrations)
    for alpha in [0.2, 0.5, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best (if available)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v = np.full_like(v, 1.0 / v.size)
        else:
            v = v / s
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            v = proj_simplex(v)
        init_cands.append(v)

    # --------------------------------------------------------------
    # Helper: exact objective (fast FFT version)
    # --------------------------------------------------------------
    def exact_J(a: np.ndarray) -> float:
        conv = conv_full_fft(a, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # Pick the best initial candidate
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        Jc = exact_J(cand)
        if Jc < best_J:
            best_J = Jc
            best_a = cand.copy()
    start_time = time.monotonic()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # Optimisation driver (L‑BFGS on the smooth surrogate)
    # --------------------------------------------------------------
    functor = _SmoothMaxFunctor(n=n, fft_len=fft_len,
                                conv_len=conv_len, fft_len_g=fft_len_g)

    # schedule of (p, max_iterations) – p grows gradually (annealing)
    schedule = [
        (5.0, 300),
        (15.0, 300),
        (40.0, 300),
        (120.0, 300),
        (300.0, 200),
    ]

    # initialise z = log‑softmax of the current best vector
    cur_z = np.log(best_a + 1e-12)

    for p_val, max_iter in schedule:
        # respect the overall time budget
        now = time.monotonic()
        remaining = (start_time + budget_s) - now - 5.0  # keep a safety margin
        if remaining <= 0:
            break

        functor.set_p(p_val)

        # L‑BFGS with analytic gradient
        res = _opt.minimize(
            functor,
            cur_z,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": int(max_iter), "gtol": 1e-12, "disp": False},
        )
        cur_z = res.x

        # evaluate on the true (non‑smoothed) objective
        cur_a = softmax(cur_z)
        J_cur = exact_J(cur_a)
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] after p={p_val:4.0f} → J = {J_cur:.6f}")

        if J_cur < best_J:
            best_J = J_cur
            best_a = cur_a.copy()

    # --------------------------------------------------------------
    # Fine‑grained random jitter around the best solution
    # --------------------------------------------------------------
    fin_start = time.monotonic()
    fin_deadline = fin_start + min(30.0, (budget_s - (fin_start - start_time)))
    while time.monotonic() < fin_deadline:
        # multiplicative log‑normal noise (small sigma)
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_a * noise)
        Jc = exact_J(cand)
        if Jc < best_J:
            best_J = Jc
            best_a = cand.copy()
            print(f"[Fine] Improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (already on the simplex, but guard against rounding)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    print(f"Final best J = {best_J:.6f}")
    return [float(v) for v in best_a]
