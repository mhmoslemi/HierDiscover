def propose_candidate(...):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0., float(budget_s) - 5.)  # safety margin

    # Determine n
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    # ensure n is int
    n = int(n)

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    fft_len_g = _next_pow2(conv_len + n - 1)

    # Build initial candidate set
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0/n, dtype=np.float64))

    # geometric sequences
    for r in [0.9, 0.93, 0.95, 0.97, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)
        # mirrored decreasing from right:
        v_m = r ** np.arange(n-1, -1, -1, dtype=np.float64)
        v_m /= v_m.sum()
        init_cands.append(v_m)

    # triangular shape: linear ramp
    v = np.arange(1, n+1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bell
    x = np.arange(n, dtype=np.float64)
    centre = (n-1)/2.0
    for sigma in [n/8.0, n/6.0, n/4.0]:
        v = np.exp(-0.5*((x-centre)/sigma)**2)
        v /= v.sum()
        init_cands.append(v)

    # Random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0]:
        for _ in range(5):
            init_cands.append(rng.dirichlet(np.ones(n)*alpha))

    # Prior best if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0:
            v /= s
        else:
            v = np.full_like(v, 1.0/n)
        if v.size != n:
            if v.size > n:
                v=v[:n]
            else:
                pad = np.zeros(n - v.size, dtype=np.float64)
                v=np.concatenate([v,pad])
        v = proj_simplex(v)
        init_cands.append(v)

    # Evaluate all candidates and keep best
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = conv_full_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J={best_J:.6f}")

    # current state
    cur_vec = best_vec.copy()
    cur_conv = conv_full_fft(cur_vec, fft_len, conv_len)
    cur_J = float(2.0 * n * cur_conv.max())
    best_J = cur_J

    last_improve = start

    iteration = 0
    # hyperparameters

    lr = 0.1
    beta1 = 0.9

    v_vec = np.zeros_like(cur_vec)

            direction = -grad
            # normalize direction to have bounded step
            norm_dir = np.linalg.norm(direction)
            if norm_dir == 0:
                # random nudge
                direction = rng.normal(0.0, 1.0, size=n)
                norm_dir = np.linalg.norm(direction)
            # base step size decreasing with sqrt(iteration)
            base_step = 0.5 / np.sqrt(iteration + 1.0)
            alpha = base_step / (norm_dir + 1e-12)

            accepted = False
            # try few shrink factors
            for factor in [1.0, 0.5, 0.25, 0.125]:
                step_vec = direction * (alpha * factor)
                cand = proj_simplex(cur_vec + step_vec)
                cand_conv = conv_full_fft(cand, fft_len, conv_len)
                cand_J = float(2.0 * n * cand_conv.max())
                if cand_J < cur_J - 1e-12:
                    cur_vec = cand
                    cur_J = cand_J
                    accepted = True
                    break

                step_vec = direction * (alpha * 0.01)
                cand = proj_simplex(cur_vec + step_vec)
                cur_vec = cand

        iteration += 1

    # final fine random search
    fin_deadline = time.monotonic() + min(30.0, (deadline - start) * 0.05)
    while time.monotonic() < fin_deadline:
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_conv = conv_full_fft(cand, fft_len, conv_len)
        cand_J = float(2.0 * n * cand_conv.max())
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            print(f"[Fine] Improved J={best_J:.6f}")

    # Normalize final vector to sum 1 and clip values
    best_vec = np.clip(best_vec, 0.0, 1000.0)

"""
Hybrid Adam + sub‑gradient optimiser for the max‑convolution problem.

The algorithm works in two phases (both respecting the simplex constraint):
  1.  Adam optimiser on the smooth log‑sum‑exp surrogate of the
      ‖a∗a‖∞ norm.  The smoothing parameter `p` is increased from 1 to
      200 over the whole run, while the learning‑rate follows a 1/√t
      schedule.  This phase quickly discovers a good basin of attraction.
  2.  Hard sub‑gradient descent on the true max‑convolution.  The active
      set (all positions within a tiny tolerance of the current maximum)
      defines a piece‑wise‑linear sub‑gradient; a simple back‑tracking line
      search with a Euclidean projection onto the probability simplex is
      used to guarantee monotone improvement.

Occasional “jitter” restarts (if no improvement for 30 s) and a final
fine‑grained random search round the best solution help escape plateaus.
The routine respects the 1000 s time‑budget, works for any `height_sequence_1`
present in the globals, and returns a non‑negative list whose sum is 1
(so the evaluation function is scale‑invariant).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the supplied evaluation function.  Returns the best sequence discovered
    within `budget_s` seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine problem size (vector length)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048
    n = int(n)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_g = _next_pow2(conv_len + n - 1)      # for gradient‑type convolutions

    # ------------------------------------------------------------------
    # 2. Build a diverse set of initial candidates
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (both left‑to‑right and mirrored)
    for r in (0.90, 0.93, 0.95, 0.97, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)
        v = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws with several concentration parameters
    for alpha in (0.5, 1.0, 2.0):
        for _ in range(4):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0:
            v /= s
        else:
            v = np.full_like(v, 1.0 / n)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                pad = np.zeros(n - v.size, dtype=np.float64)
                v = np.concatenate([v, pad])
        v = proj_simplex(v)
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate (quick evaluation)
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = conv_full_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation parameters
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Adam hyper‑parameters
    lr = 0.12
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(best_vec)
    v_vec = np.zeros_like(best_vec)

    # smoothing parameter schedule (log‑sum‑exp surrogate)
    p_start, p_end = 1.0, 200.0

    # switch to hard sub‑gradient after 20 % of the allotted time
    switch_time = start_time + 0.20 * (deadline - start_time)

    cur_vec = best_vec.copy()
    last_improve = start_time
    iteration = 0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop (Adam → sub‑gradient)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - start_time) / (deadline - start_time)

        # ------------------------------------------------------------------
        # 5.1  Common quantities
        # ------------------------------------------------------------------
        conv = conv_full_fft(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ------------------------------------------------------------------
        # 5.2  Track global best
        # ------------------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 5.3  Stagnation handling (jitter)
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = proj_simplex(best_vec * noise)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 5.4  Choose optimisation mode
        # ------------------------------------------------------------------
        if now < switch_time:
            # ----------------------------------------------------------------
            # 5.4a  Adam on the smooth surrogate  φ(a) = log‑sum‑exp(p·conv) / p
            # ----------------------------------------------------------------
            p = p_start + (p_end - p_start) * frac

            # soft‑max weights (unnormalised, max weight = 1)
            w = np.exp(p * (conv - max_conv))

            a_rev = cur_vec[::-1]
            grad_full = np.fft.irfft(
                np.fft.rfft(w, fft_len_g) * np.fft.rfft(a_rev, fft_len_g),
                n=fft_len_g,
            )
            grad = 2.0 * grad_full[conv_len - 1 : conv_len - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** (iteration + 1))
            v_hat = v_vec / (1.0 - beta2 ** (iteration + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            proposal = cur_vec - step
            cur_vec = proj_simplex(proposal)

        else:
            # ----------------------------------------------------------------
            # 5.4b  Hard sub‑gradient descent on the true max‑norm
            # ----------------------------------------------------------------
            # tiny tolerance to collect the active set
            rel_tol = 1e-8 * max_conv
            active = np.where(conv >= max_conv - max(rel_tol, 1e-12), 1.0, 0.0)

            a_rev = cur_vec[::-1]
            grad_full = np.fft.irfft(
                np.fft.rfft(active, fft_len_g) * np.fft.rfft(a_rev, fft_len_g),
                n=fft_len_g,
            )
            grad = 2.0 * grad_full[conv_len - 1 : conv_len - 1 + n]

            direction = -grad
            norm_dir = np.linalg.norm(direction)
            if norm_dir == 0.0:
                direction = rng.normal(0.0, 1.0, size=n)
                norm_dir = np.linalg.norm(direction)

            # step size ~ 0.5/√t, scaled to unit direction length
            base_step = 0.5 / np.sqrt(iteration + 1.0)
            alpha = base_step / (norm_dir + 1e-12)

            improved = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                cand = proj_simplex(cur_vec + factor * alpha * direction)
                cand_conv = conv_full_fft(cand, fft_len, conv_len)
                cand_J = float(2.0 * n * cand_conv.max())
                if cand_J < cur_J - 1e-12:
                    cur_vec = cand
                    improved = True
                    break
            if not improved:
                # tiny step to keep the dynamics alive
                cur_vec = proj_simplex(cur_vec + 0.01 * alpha * direction)

        iteration += 1

    # ------------------------------------------------------------------
    # 6. Fine‑grained random search around the best solution
    # ------------------------------------------------------------------
    fine_deadline = time.monotonic() + min(30.0, 0.05 * (deadline - start_time))
    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_conv = conv_full_fft(cand, fft_len, conv_len)
        cand_J = float(2.0 * n * cand_conv.max())
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            print(f"[Fine] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 7. Final normalisation & return
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total > 0.0:
        best_vec /= total
    else:
        best_vec[:] = 1.0 / n

    return [float(v) for v in best_vec]
