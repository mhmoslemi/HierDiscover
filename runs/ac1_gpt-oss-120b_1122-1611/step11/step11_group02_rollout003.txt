"""
Adam‑based soft‑max optimiser for the l∞‑convolution problem.

We minimise J(a)=2·n·max(conv(a,a)) subject to a_i≥0, Σa_i=1.
The optimisation is performed on the unconstrained log‑probabilities
x (a_i = softmax(x_i)).  A smooth approximation of the max
(log‑sum‑exp) is used: φ_p(a)=log ∑_k exp(p·(a*a)_k)/p.
The gradient of φ_p is obtained analytically (see `gradient_smooth`).

The algorithm:
  • Initialise x from a random Dirichlet (non‑uniform start).
  • Run Adam on x while slowly increasing the smoothness parameter p.
  • Track the best discrete candidate (by the true objective J).
  • If progress stalls, add Gaussian jitter to x.
  • After the main budget a short fine‑grained random‑noise search
    around the best solution is performed.
The routine returns the best probability vector found (scaled to sum=1,
non‑negative, and clipped to the limits enforced by `evaluate_sequence`).

The method differs from the previous exponentiated‑gradient optimiser
by using Adam (adaptive moment estimation), an explicit soft‑max
parameterisation and a simple stagnation‑jitter scheme, which in practice
helps to escape plateaux and achieve a lower bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def gradient_smooth(a: np.ndarray, conv: np.ndarray,
                    p: float, fft_len_g: int, conv_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max
          φ_p(a) = log‑sum‑exp(p·conv(a,a)) / p .
    Returns ∂φ/∂a (size = len(a)).
    """
    # soft‑max weights (max weight = 1)
    w = np.exp(p * (conv - conv.max()))
    a_rev = a[::-1]

    # convolution w (length 2n‑1) with a_rev (length n)
    grad_full = np.fft.irfft(
        np.fft.rfft(w, fft_len_g) * np.fft.rfft(a_rev, fft_len_g),
        n=fft_len_g
    )
    # slice to obtain length‑n gradient
    grad = 2.0 * grad_full[conv_len - 1: conv_len - 1 + a.size]
    return grad


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  Returns the best sequence found within the
    allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            # keep size reasonable (power of two gives fast FFTs)
            n = max(256, min(_next_pow2(n0), 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # for a*a
    fft_len_g = _next_pow2(conv_len + n - 1)  # for gradient (w * a_rev)

    # --------------------------------------------------------------
    # 2. Initialisation on the simplex via a Dirichlet draw.
    # --------------------------------------------------------------
    # a Dirichlet with alpha < 1 gives a spiky start, helping exploration.
    init_a = rng.dirichlet(np.full(n, 0.6))
    x = np.log(init_a + 1e-32)          # log‑probabilities (unconstrained)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.15                      # base learning rate

    # --------------------------------------------------------------
    # 3. Warm‑up / scheduling parameters.
    # --------------------------------------------------------------
    p_start = 0.5
    p_end = 400.0

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time
    best_J = np.inf
    best_a = None
    iteration = 0

    # --------------------------------------------------------------
    # 4. Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # ----- 4.1 temperature (soft‑max smoothing) schedule -----
        p = p_start + (p_end - p_start) * frac

        # ----- 4.2 construct current probability vector -----
        a = softmax(x)                     # guaranteed non‑negative, sum=1

        # ----- 4.3 evaluate true objective J -----
        conv = conv_full_fft(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ----- 4.4 track global best -----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- 4.5 stagnation handling (jitter) -----
        if now - last_improve > 20.0:
            # add moderate Gaussian noise to log‑probabilities
            sigma = 0.25
            x += rng.normal(0.0, sigma, size=n)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter after stagnation (best J = {best_J:.6f})")

        # ----- 4.6 compute smoothed‑max gradient -----
        grad_a = gradient_smooth(a, conv, p, fft_len_g, conv_len)

        # ----- 4.7 back‑prop to log‑probabilities (chain rule) -----
        #   ∂φ/∂x = a * (∂φ/∂a - (∂φ/∂a · a))
        ga_dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - ga_dot)

        # ----- 4.8 Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # learning‑rate decay (∝ 1/√t) built into α0; keep it modestly decreasing
        lr = alpha0 / np.sqrt(iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

    # --------------------------------------------------------------
    # 5. Fine‑grained random search around the best candidate.
    # --------------------------------------------------------------
    if best_a is None:  # fallback – should never happen
        best_a = softmax(x)

    fin_start = time.monotonic()
    # allocate up to 30 s or 5 % of the remaining time whichever is smaller
    remaining = deadline - start_time
    fin_deadline = fin_start + min(30.0, 0.05 * remaining)

    while time.monotonic() < fin_deadline:
        # multiplicative perturbation (log‑normal noise)
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = best_a * noise
        cand = proj_simplex(cand)          # ensure Σ=1, non‑negative

        conv = conv_full_fft(cand, fft_len, conv_len)
        cand_J = float(2.0 * n * conv.max())
        if cand_J < best_J:
            best_J = cand_J
            best_a = cand
            now = time.monotonic() - start_time
            print(f"[Fine] Improved J = {best_J:.6f} at t={now:6.2f}s")

    # --------------------------------------------------------------
    # 6. Final normalisation and return.
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full_like(best_a, 1.0 / best_a.size)
    else:
        best_a = best_a / total

    return [float(v) for v in best_a]
