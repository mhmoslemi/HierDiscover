"""
Propose candidate – gradient‑descent on a smoothed max‑convolution objective.

Key ideas
---------

* Work on the simplex (∑a_i = 1, a_i ≥ 0).  The evaluation
  ``evaluate_sequence`` is invariant under scaling, therefore we keep the
  vector normalised after every update.

* Approximate the non‑smooth objective
        J(a) = 2·n·max(conv(a,a))
  by a *soft‑max* (log‑sum‑exp) with a temperature β.
  The gradient of the soft‑max can be written analytically as a
  correlation between the soft‑max weights and the current vector.
  Both the convolution and the correlation are computed with a real FFT
  (O(n log n) per iteration).

* Use a multiplicative (exponentiated‑gradient) update:
        a_i ← a_i · exp(‑η·∂L/∂a_i)
  where η is a decreasing step‑size.  After the update we renormalise
  to stay on the simplex.

* β is increased linearly with the elapsed time – early iterations are
  smooth (large basin of attraction) and later iterations focus on the
  true max.

* Stagnation is broken by a short *jitter* (multiplicative Gaussian noise)
  around the current best solution.

The routine runs for at most the supplied budget (≈ 1000 s) and returns
the best vector found.  All numbers are clipped to the range accepted by
``evaluate_sequence`` before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    Returns ``len(x)+len(y)-1`` elements.
    """
    size = len(x) + len(y) - 1
    fft_len = _next_pow2(size)
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:size]
    return conv


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective value 2·n·max(conv) given a full convolution array."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  The search is a gradient descent on a
    smoothed version of the max‑convolution objective with occasional jitter.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # keep it within a reasonable interval
            n = max(256, min(n, 16384))
        except Exception:
            n = 2048
    else:
        n = 2048

    # ------------------------------------------------------------------
    # 2. Build a set of diverse initial probability vectors
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderate concentration)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # reuse the previous construction if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate (using the true objective)
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_fft(cand, cand)
        J = _objective_from_conv(conv, n)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (soft‑max + exponentiated‑gradient)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # small safety margin to guarantee return
    deadline = start_time + max(0.0, float(budget_s) - 1.0)

    # hyper‑parameters
    lr0 = 0.5                # initial learning rate (will decay)
    beta_start = 0.8         # start temperature for soft‑max
    beta_end = 15.0          # final temperature (focus on true max)
    jitter_sigma = 0.15      # jitter magnitude (log‑Gaussian)
    stagnation_limit = 30.0  # seconds without improvement before jitter

    cur = best_vec.copy()
    cur = np.maximum(cur, 1e-30)            # avoid exact zeros
    cur /= cur.sum()

    last_improve = start_time
    iteration = 0

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start_time
        # --------------------------------------------------------------
        # evaluate current state
        # --------------------------------------------------------------
        conv_cur = _conv_fft(cur, cur)              # length 2n‑1
        max_conv = conv_cur.max()
        cur_J = float(2.0 * n * max_conv)

        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # stagnation handling – occasional jitter
        # --------------------------------------------------------------
        if now - last_improve > stagnation_limit:
            # perturb the best known solution and continue from there
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cur = best_vec * noise
            cur = np.clip(cur, 1e-30, None)
            cur /= cur.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # schedule β (soft‑max temperature) – linear in elapsed fraction
        # --------------------------------------------------------------
        frac = (now - start_time) / (deadline - start_time)
        beta = beta_start + (beta_end - beta_start) * frac

        # --------------------------------------------------------------
        # compute soft‑max weights on the convolution entries
        # --------------------------------------------------------------
        # stability: subtract max before exponentiation
        exp_arg = beta * (conv_cur - max_conv)
        w = np.exp(exp_arg)
        p = w / w.sum()                # p has length 2n‑1, sums to 1

        # --------------------------------------------------------------
        # gradient of the smoothed max (∂L/∂a) = 2·corr(p, a)
        # --------------------------------------------------------------
        rev_cur = cur[::-1]            # reversed current vector
        conv_corr = _conv_fft(p, rev_cur)   # length 3n‑2
        # slice the part that corresponds to shifts 0 … n‑1
        grad = 2.0 * conv_corr[n - 1:n - 1 + n]

        # --------------------------------------------------------------
        # exponentiated‑gradient update
        # --------------------------------------------------------------
        # learning‑rate schedule (inverse square‑root decay)
        lr = lr0 / np.sqrt(iteration + 1.0)

        # multiplicative update, keep positivity
        factor = np.exp(-lr * grad)
        cur = cur * factor
        # safeguard against numerical collapse
        cur = np.maximum(cur, 1e-30)
        cur /= cur.sum()

        # --------------------------------------------------------------
        # occasional random mass‑transfer move (helps escape plateaus)
        # --------------------------------------------------------------
        if rng.random() < 0.05:
            # pick donor proportionally to gradient magnitude
            donor = rng.choice(n, p=grad / grad.sum())
            # favour indices with small gradient as receivers
            inv = grad.max() - grad + 1e-12
            receiver = rng.choice(n, p=inv / inv.sum())
            delta = min(cur[donor], 0.03 * cur[donor])   # move up to 3 % of donor mass
            cur[donor] -= delta
            cur[receiver] += delta
            cur = np.clip(cur, 1e-30, None)
            cur /= cur.sum()

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (final safety clipping & normalisation)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # optional: final verification with the provided evaluate_sequence
    # try:
    #     import sys
    #     print("Final J (checked) =", evaluate_sequence(best_vec.tolist()))
    # except Exception:
    #     pass

    return [float(v) for v in best_vec]
