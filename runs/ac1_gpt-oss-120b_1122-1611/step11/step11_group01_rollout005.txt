"""
Propose a step‑function (non‑negative vector on the simplex) that minimises
`evaluate_sequence`.  The method is a fast projected Adam optimiser on a
smooth approximation of the objective
    J(a) = 2·n·max(conv(a,a)),
using a soft‑max (log‑sum‑exp) surrogate for the max and a linear‑in‑time
gradient computed via FFT.  A simple simulated‑annealing style jitter
re‑initialises the search after prolonged stagnation.  The optimiser works
directly on the probability simplex (∑a_i = 1) by projecting after each
update.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used in the core loop


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector w such that w_i ≥ 0 and sum(w) = 1.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen for non‑negative inputs, but fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Parameters
    ----------
    a : ndarray, shape (n,)
    fft_len : int – length of the FFT (a power of two ≥ 2·n‑1)
    conv_len : int – length of the full convolution (2·n‑1)

    Returns
    -------
    conv : ndarray, shape (conv_len,)
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The function respects the time budget `budget_s`
    (seconds) and returns the best normalised vector found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # safety margin to guarantee we return before the external timeout
    safety = 5.0
    deadline = start_time + max(0.0, float(budget_s) - safety)

    # ------------------------------------------------------------------
    # 1. Determine problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # bound the size to a tractable range
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a reasonable default

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Initialise the probability vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = a[:n]                     # truncate / pad if needed
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # a small random perturbation to avoid exact symmetry
    a *= np.exp(rng.normal(scale=0.02, size=n))
    a = proj_simplex(a)

    # ------------------------------------------------------------------
    # 3. Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # learning‑rate schedule (linear decay from lr_start to lr_end)
    lr_start = 0.12
    lr_end = 0.001

    # soft‑max temperature schedule (log‑sum‑exp temperature β)
    beta_start = 1.0
    beta_end = 1600.0

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    best_vec = a.copy()
    # true objective: J = 2·n·max(conv)
    conv = full_conv_fft(a, fft_len, conv_len)
    best_J = float(2.0 * n * conv.max())

    last_improve = start_time
    iteration = 0

    print(f"[0.00s] Initial J = {best_J:.6f}")

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # --------------------------------------------------------------
        # 4.1  Compute convolution and current objective
        # --------------------------------------------------------------
        A_fft = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A_fft * A_fft, fft_len)[:conv_len]
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------------
        # 4.2  Stagnation handling – jitter after a while without progress
        # ----------------------------------------------------------------
        if now - last_improve > 25.0:          # 25 s of no improvement
            # multiplicative noise centred at 1
            noise = np.exp(rng.normal(loc=0.0, scale=0.03, size=n))
            a = best_vec * noise
            a = proj_simplex(a)
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 4.3  Soft‑max (log‑sum‑exp) weights for the convolution entries
        # --------------------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        # Numerically stable soft‑max
        w_raw = np.exp(beta * (conv - max_conv))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv_len)
        else:
            w = w_raw / w_sum

        # --------------------------------------------------------------
        # 4.4  Gradient of the smoothed objective (FFT‑based)
        # --------------------------------------------------------------
        W_fft = np.fft.rfft(w, fft_len)
        # g = 2·cross‑correlation(a, w)  (real vector of length n)
        g = 2.0 * np.fft.irfft(np.conj(W_fft) * A_fft, fft_len)[:n]
        # Gradient of J_β = 2·n·Σ w_k·conv_k   (the β‑smooth surrogate)
        grad = 2.0 * n * g

        # --------------------------------------------------------------
        # 4.5  Adam update on the simplex (additive, then project)
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr_start + (lr_end - lr_start) * (elapsed / total)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                     # move downhill
        a = np.maximum(a, 0.0)           # keep non‑negative
        a = proj_simplex(a)              # back onto the simplex

        # --------------------------------------------------------------
        # 4.6  Early exit if target already reached
        # --------------------------------------------------------------
        if best_J <= 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # ------------------------------------------------------------------
    # 5. Return the best solution found (clipped/re‑normalised)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
