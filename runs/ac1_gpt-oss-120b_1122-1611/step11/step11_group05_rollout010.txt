def propose_candidate(seed=..., budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n (use existing height_sequence_1 length else default 3800 maybe)
    # Build candidates as described.
    # Evaluate J for each candidate, pick best as cur (prob vector)
    best_J = float('inf')
    best_vec = None
    for cand in init_cands:
        # normalize sum to 1 then evaluate
        conv = _conv_full(cand, fft_len_conv, conv_len)
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()

    cur_vec = best_vec.copy()
    cur_J = best_J
    best_J = cur_J
    best_vec = cur_vec.copy()
    # parameters
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 0.5) # leaving margin
    last_improve = start_time
    iteration = 0
    base_eta = 0.5 # initial step magnitude
    # Keep track of best objective not only J (based on maxconv)
    while time.monotonic() < deadline:
        # Compute conv and max
        conv = _conv_full(cur_vec, fft_len_conv, conv_len)
        max_val = conv.max()
        cur_J = 2.0 * n * max_val

        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_J:.6f}")

        # stagnation jitter restart
        if time.monotonic() - last_improve > 25.0:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] Jitter restart")
            continue

        # compute gradient for active shifts (approx tolerance)
        # Use tolerance relative to max_val
        tol = 1e-12
        active = np.isclose(conv, max_val, atol=tol)
        active_shifts = np.nonzero(active)[0]
        # compute gradient
        grad = np.zeros_like(cur_vec)
        n_local = n
        for k in active_shifts:
            # for i from max(0, k-(n-1)) to min(n-1, k)
            i0 = max(0, k - (n_local - 1))
            i1 = min(n_local - 1, k)
            # Vectorized approach:
            idx_i = np.arange(i0, i1 + 1, dtype=np.int64)
            idx_j = k - idx_i
            grad[idx_i] += 2.0 * cur_vec[idx_j]

        # Choose step size schedule
        eta = base_eta / np.sqrt(iteration + 1.0)

        # Try a few candidates for step factor
        best_candidate_J = cur_J
        best_candidate_vec = cur_vec
        for factor in [0.5, 1.0, 2.0]:
            step = eta * factor
            # multiplicative update (exponential) with gradient sign - moving away from high gradient
            proposal = cur_vec * np.exp(-step * grad)
            # ensure positivity and renormalize to simplex
            proposal = proj_simplex(proposal)
            # compute J
            prop_conv = _conv_full(proposal, fft_len_conv, conv_len)
            prop_J = 2.0 * n * prop_conv.max()
            if prop_J < best_candidate_J:
                best_candidate_J = prop_J
                best_candidate_vec = proposal

        # Also try additive step if gradient not too large
        # additive: a - step*grad then project (ensuring not negative)
        step = eta
        add_proposal = cur_vec - step * grad
        add_proposal = np.clip(add_proposal, 0.0, None)
        add_proposal = proj_simplex(add_proposal)
        add_conv = _conv_full(add_proposal, fft_len_conv, conv_len)
        add_J = 2.0 * n * add_conv.max()
        if add_J < best_candidate_J:
            best_candidate_J = add_J
            best_candidate_vec = add_proposal

        # Accept best candidate as new cur
        cur_vec = best_candidate_vec
        cur_J = best_candidate_J

        iteration += 1

    # final best_vec may be out-of-range (some entries <0?), clip
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

Propose a step‑function (non‑negative probability vector) that minimizes
`evaluate_sequence`.  The algorithm consists of three phases:

1. Build a small portfolio of diverse initial probability vectors
   (uniform, geometric decays, ramps, Gaussians, Dirichlet draws,
   sparse masks and the previously found best `height_sequence_1`).

2. Run a short sub‑gradient (exponentiated‑gradient) descent that uses the
   exact sub‑gradient of the *maximum* convolution entry.  This quickly
   drives the objective down from the random starts.

3. Finish with a simulated‑annealing “mass‑transfer” walk.  In each step a
   small amount of probability mass is moved from an index that contributes
   heavily to the current worst convolution peak to an index that contributes
   little.  The convolution is updated *incrementally* (O(n) work) and a
   Metropolis acceptance rule with a linear cooling schedule is used.
   Periodic full FFT recomputation removes accumulated rounding error.

The whole search respects the 1000 s budget and returns the best vector
found (normalised to sum = 1, clipped to [0,1000]).

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negative entries that sum to one.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (the full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Exact objective value 2·n·max(conv) for a probability vector ``vec``.
    """
    conv = _conv_full(vec, fft_len, conv_len)
    return float(2.0 * n * conv.max())
