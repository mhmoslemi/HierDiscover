"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm works on a probability vector
`a` (non‑negative entries summing to one – therefore the global
scale is irrelevant).  It uses a *soft‑max* (log‑sum‑exp) surrogate for
the `max` in the objective, and performs Adam optimisation on the
log‑its of the probability vector.  The soft‑max temperature β is
annealed from a gentle value (≈5) to a very sharp one (≈500) so that
the optimisation gradually focuses on the true L∞‑norm.  Throughout the
run simple jitter restarts (random Dirichlet draws) are triggered if
no improvement has been seen for a while, which helps escape plateaus.
All heavy work (convolution, correlation) is done with real FFTs, so
the method scales to several‑thousand‑element vectors well within the
1000 s budget.

The final sequence is clipped to the domain limits required by
`evaluate_sequence` and returned as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _gradient_softmax(
    a: np.ndarray,
    conv: np.ndarray,
    beta: float,
    fft_len_corr: int,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute gradient of the smoothed‑max objective
        fβ(a) = (1/β)·log Σ exp(β·conv)
    and also return the soft‑max weights.
    Returns (grad_a, w) where
        grad_a = 2·(w ⋆ a)  (linear correlation)
    """
    # stable soft‑max weights
    shift = conv.max()
    exp_vals = np.exp(beta * (conv - shift))
    w = exp_vals / exp_vals.sum()

    # correlation w ⋆ a  (linear, not circular)
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    W_hat = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)

    n = a.size
    start = n - 1                     # index where shift = 0
    grad_a = 2.0 * corr[start:start + n]
    return grad_a, w


def _logit_grad(a: np.ndarray, grad_a: np.ndarray) -> np.ndarray:
    """Chain rule for a = softmax(v) :  ∂f/∂v = a·(grad_a – (grad_a·a))."""
    dot = np.dot(grad_a, a)
    return a * (grad_a - dot)


# ----------------------------------------------------------------------
# Main optimisation routine – the function required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size (length of the step function)
    # ------------------------------------------------------------------
    # Use a fairly large size (still cheap for an FFT) but allow a
    # previously found sequence to dictate a minimum length.
    if "height_sequence_1" in globals():
        try:
            n0 = int(len(globals()["height_sequence_1"]))
            n = max(4096, min(n0, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)                    # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)            # for correlation

    # ------------------------------------------------------------------
    # 2. Build a cheap portfolio of deterministic initial probability
    #    vectors and pick the best (according to the true objective).
    # ------------------------------------------------------------------
    def _to_prob(v):
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        init_cands.append(_to_prob(v))

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    init_cands.append(_to_prob(v))
    v = np.arange(1, n + 1, dtype=np.float64)
    init_cands.append(_to_prob(v))

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        init_cands.append(_to_prob(v))

    # Dirichlet draws (different concentrations)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(_to_prob(rng.dirichlet(np.ones(n) * alpha)))

    # Sparse masks – a few equally‑weighted spikes
    k_vals = (int(np.sqrt(n)), int(np.sqrt(n) * 2), max(2, n // 10))
    for k in k_vals:
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        init_cands.append(_to_prob(mask))

    # random binary mask (Bernoulli 0.5)
    bin_mask = rng.binomial(1, 0.5, size=n).astype(np.float64)
    if bin_mask.sum() > 0:
        init_cands.append(_to_prob(bin_mask))

    # previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        init_cands.append(_to_prob(v))

    # ------------------------------------------------------------------
    # 3. Pick the best deterministic starter (true objective).
    # ------------------------------------------------------------------
    def _objective(a_vec):
        conv = _conv_full(a_vec, fft_len_conv, conv_len)
        return float(2.0 * n * conv.max())

    best_a = None
    best_J = np.inf
    for cand in init_cands:
        J = _objective(cand)
        if J < best_J:
            best_J = J
            best_a = cand.copy()
    print(f"[0.00s] Initial deterministic best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation – Adam on log‑its, annealed soft‑max.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # a few seconds safety

    # initialise logits from the best deterministic vector
    eps = 1e-12
    cur_logit = np.log(best_a + eps)

    # Adam state
    m = np.zeros_like(cur_logit)
    v2 = np.zeros_like(cur_logit)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05                     # base learning rate for the logits
    t_iter = 1

    # soft‑max temperature schedule
    beta = 5.0
    beta_target = 500.0

    # stagnation handling
    last_improve = start_time
    jitter_interval = 30.0

    while time.monotonic() < deadline:
        # --- construct probability vector from logits ---
        max_logit = cur_logit.max()
        exp_logits = np.exp(cur_logit - max_logit)
        a = exp_logits / exp_logits.sum()                     # guarantees sum=1

        # --- evaluate true objective (no surrogate) ---
        conv = _conv_full(a, fft_len_conv, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # keep global best
        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --- jitter restart if stuck for a while ---
        if now - last_improve > jitter_interval:
            # random Dirichlet restart with moderate concentration
            a_rand = _to_prob(rng.dirichlet(np.ones(n) * 0.8))
            cur_logit = np.log(a_rand + eps)
            m.fill(0.0)
            v2.fill(0.0)
            t_iter = 1
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # --- gradient of the smoothed‑max objective ---
        grad_a, _ = _gradient_softmax(a, conv, beta, fft_len_corr)

        # chain rule to logits
        grad_logit = _logit_grad(a, grad_a)

        # --- Adam update on logits (gradient descent) ---
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_logit ** 2)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v2_hat = v2 / (1.0 - beta2 ** t_iter)

        cur_logit -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

        # --- anneal β (soft‑max sharpness) ---
        # exponential schedule reaching β_target at the end of the budget
        fraction = (now - start_time) / (deadline - start_time)
        beta = beta * (beta_target / beta) ** fraction
        beta = min(beta, beta_target)

        t_iter += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution found (clip & renormalise for safety)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    # sanity: ensure the evaluator will not reject because of tiny sum
    if best_a.sum() < 0.01:
        # fallback to uniform
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_a]
