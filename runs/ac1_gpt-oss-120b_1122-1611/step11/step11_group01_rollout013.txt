"""
Propose a candidate sequence that minimizes `evaluate_sequence`.

Algorithm
---------

* **Problem size** – use the length of the previously found solution if
  available, otherwise fall back to 8192 (capped at 8192 for speed).

* **Initial pool** – a diverse collection of vectors on the simplex:
  uniform, several Dirichlet draws (spiky to smooth), Gaussian bells,
  a few random sparse vectors and the previous construction (if any).  
  The best of these is taken as the starting point.

* **Early random search** – for a few seconds we sample many random
  Dirichlet vectors (α∈{0.1,0.2,0.3}) and keep the best one.  This helps
  to escape the uniform stationary point.

* **Projected gradient descent on the max‑convolution** – the objective
  is `J = 2·n·max(conv(a,a)) / (sum a)²`.  Since `sum a` is fixed to 1,
  minimizing `J` is equivalent to minimizing `max(conv)`.  The max
  entry of the convolution is a piece‑wise linear function; its
  sub‑gradient is the gradient of the active convolution entry (or the
  average gradient of all entries tied for the maximum).

  * Compute the full convolution by a real FFT (zero‑padded to the next
    power of two).
  * Build a hard mask `w` that places weight `1/k` at each index where
    the convolution attains its maximum (`k` is the number of active
    indices).
  * Gradient of the weighted sum `∑ w·conv` w.r.t. `a` is  
    `g = 2·ifft(conj(FFT(w))·FFT(a))` (real part only).
  * Perform an **additive** update `a ← a – η·g` with a step size that
    decays like `1/√t`.  Momentum (`μ≈0.9`) is added for stability.
  * Project back onto the probability simplex (Euclidean projection)
    to enforce non‑negativity and unit L¹‑norm.

* **Metropolis acceptance** – occasionally accept uphill moves with a
  linearly cooling temperature to escape shallow plateaus.

* **Stagnation handling** – if no improvement is seen for 30 s the
  current iterate is jittered by multiplicative log‑normal noise and
  re‑projected.

* **Termination** – stop when the time budget (≈1000 s) is exhausted or
  when the target `J < 1.5025` is reached.

The routine returns the best vector found (clipped to the allowed
range).  All operations are pure NumPy and respect the 2‑CPU limit.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero -> return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT of length `fft_len`."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    Returns the best normalised sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size (length of the sequence)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 8192

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a diverse set of starting points (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws – from spiky to smooth
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Gaussian bell shapes
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 8.0, n / 5.0, n / 3.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Sparse Dirichlet (very spiky)
    for alpha in [0.05, 0.1]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best of the initial deterministic candidates
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        J = float(2.0 * n * conv.max())          # sum(cand) == 1
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial deterministic best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Early random search (few seconds) to escape uniform state
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    random_search_end = start_time + 5.0                       # first 5 s

    while time.monotonic() < random_search_end:
        # randomly pick an alpha for a Dirichlet draw
        alpha = rng.choice([0.1, 0.15, 0.2, 0.3])
        cand = rng.dirichlet(np.full(n, alpha))
        conv = _conv_full(cand, fft_len, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
            # print only when we improve during the short random search
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Random‑search improvement J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Main optimisation loop (projected gradient on the active max)
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # Momentum term for Nesterov‑style updates
    momentum = np.zeros_like(cur_vec)

    # Temperature schedule for Metropolis acceptance
    T_start = best_J * 0.05          # 5 % of current best
    T_end = 1e-8

    # Stagnation handling parameters
    stagnation_limit = 30.0          # seconds
    jitter_std = 0.02                # standard deviation of log‑normal jitter

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ------------------------------------------------------
        # 5.1 Current objective (full recompute needed each iteration)
        # ------------------------------------------------------
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)   # sum(cur_vec) == 1

        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------
        # 5.2 Stagnation handling – jitter after long inactivity
        # ------------------------------------------------------
        if now - last_improve > stagnation_limit:
            noise = np.exp(rng.normal(0.0, jitter_std, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            momentum.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ------------------------------------------------------
        # 5.3 Hard‑max mask (indices attaining the maximum)
        # ------------------------------------------------------
        # Numerical tolerance – treat values within 1e‑15 of the max as active
        eps = 1e-15
        active_idx = np.nonzero(conv >= max_conv - eps)[0]
        w = np.zeros(conv_len, dtype=np.float64)
        if active_idx.size == 0:
            # should never happen, but fall back to uniform weighting
            w[conv_len // 2] = 1.0
        else:
            w[active_idx] = 1.0 / float(active_idx.size)

        # ------------------------------------------------------
        # 5.4 Gradient of the weighted sum of convolutions
        # ------------------------------------------------------
        A = np.fft.rfft(cur_vec, fft_len)
        W = np.fft.rfft(w, fft_len)
        g = 2.0 * np.fft.irfft(np.conj(W) * A, n=fft_len)[:n]

        # ------------------------------------------------------
        # 5.5 Adaptive step size and momentum update (additive)
        # ------------------------------------------------------
        step = 0.5 / np.sqrt(iteration + 1.0)   # similar decay to the older code
        mu = 0.9
        momentum = mu * momentum - step * g
        proposal = cur_vec + momentum
        proposal = proj_simplex(proposal)

        # ------------------------------------------------------
        # 5.6 Evaluate the proposal
        # ------------------------------------------------------
        prop_conv = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ------------------------------------------------------
        # 5.7 Metropolis acceptance (allows occasional uphill moves)
        # ------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # linear cooling for the Metropolis temperature
            T = T_start * max(0.0, 1.0 - elapsed / (deadline - start_time)) + \
                T_end * (elapsed / (deadline - start_time))
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J
            # momentum is already updated; keep it for the next step

        iteration += 1

        # Early exit if target already reached
        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # --------------------------------------------------------------
    # 6. Return the best solution (clipped, renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
