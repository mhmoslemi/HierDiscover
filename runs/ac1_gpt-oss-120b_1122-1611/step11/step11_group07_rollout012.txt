    rng = np.random.default_rng(seed)
    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        n = max(256, min(n, 8192))
    else:
        n = 2048

    # FFT lengths
    conv_len = 2*n-1
    fft_len_conv = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(conv_len + n - 1)

    # Initialize a
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0, None)
        total = a.sum()
        if total == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= total
    else:
        a = np.full(n, 1.0/n)

    # parameters
    beta0 = 5.0
    beta_max = 800.0
    eta0 = 0.5
    t = 0
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)
    best_a = a.copy()
    best_J = 2.0 * n * _conv_full(a, fft_len_conv, conv_len).max()
    last_improvement = start

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        # schedule beta
        beta = beta0 + (beta_max - beta0) * (elapsed / (deadline - start))
        # compute conv
        A = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(A * A, n=fft_len_conv)[:conv_len]
        # smoothed max gradient
        max_c = conv.max()
        # compute softmax weights p
        w = np.exp(beta * (conv - max_c))  # max element becomes exp(0)=1, others <=1
        p = w / w.sum()
        # compute gradient via convolution with a_rev
        a_rev = a[::-1]
        P = np.fft.rfft(p, fft_len_grad)
        AR = np.fft.rfft(a_rev, fft_len_grad)
        conv2 = np.fft.irfft(P * AR, n=fft_len_grad)[:conv_len + n - 1]
        grad = 2.0 * conv2[n-1:n-1+n]
        # multiplicative update
        eta = eta0 / np.sqrt(t + 1)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()

        # occasional exchange move
        if rng.random() < 0.15:
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = 0.05
            delta = min(a[i], rng.random() * max_transfer)
            a[i] -= delta
            a[j] += delta
            # renorm automatically as sum unchanged

        # evaluate objective
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_a = a.copy()
            last_improvement = time.monotonic()
            print(f"[{elapsed:.2f}s] New best J={best_J:.6f}")

        # stagnation check
        if time.monotonic() - last_improvement > 15.0:
            # jitter around best solution
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = best_a * noise
            a = a / a.sum()
            last_improvement = time.monotonic()
            print(f"[{elapsed:.2f}s] Jitter reset")
        t += 1

    # final scaling
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0:
        best_a[:] = 1.0/n
    else:
        best_a /= total
    return [float(v) for v in best_a]

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(A * A).real[:conv_len]   # using standard ifft (complex)
    # Using irfft for real speed
    # conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

def _conv_full_real(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """fast real convolution a*a (full length) using rfft/irfft."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

def gradient_softmax(a: np.ndarray, beta: float,
                     fft_len_conv: int, conv_len: int,
                     fft_len_grad: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed max
        f_beta(a) = (1/β) log Σ_k exp(β·(a*a)[k])
    with respect to a.  The result has the same shape as a.
    """
    # convolution a*a
    A = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(A * A, n=fft_len_conv)[:conv_len]

    # soft‑max weights
    max_c = conv.max()
    # stability: subtract max
    w = np.exp(beta * (conv - max_c))
    p = w / w.sum()

    # gradient = 2 * (p * a_rev) restricted to n entries
    a_rev = a[::-1]
    P = np.fft.rfft(p, fft_len_grad)
    AR = np.fft.rfft(a_rev, fft_len_grad)
    conv2 = np.fft.irfft(P * AR, n=fft_len_grad)[:conv_len + a.size - 1]

    grad = 2.0 * conv2[a.size - 1 : a.size - 1 + a.size]
    return grad, conv.max()

                     budget_s: float = 1000.0,

    """Search for a non‑negative step‑function that minimises evaluate_sequence.

    The method uses a smooth‑max gradient descent with an annealing schedule
    for the temperature β and occasional stochastic exchange moves.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

            # Clamp to a reasonable range to keep FFT cheap

        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(conv_len + n - 1)   # for p * a_rev

    # ------------------------------------------------------------------
    # 2. Initialise a (probability vector)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)[:n]
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. Hyper‑parameters and schedule
    # ------------------------------------------------------------------
    beta_start = 4.0
    beta_end   = 800.0
    eta0 = 0.5               # base step size for multiplicative update
    exchange_prob = 0.12     # probability of a pure exchange move
    max_transfer = 0.08      # max fraction of source mass moved during exchange
    jitter_interval = 20.0   # seconds without improvement before jitter

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep safety margin

    # evaluate initial objective
    conv0 = _conv_full_real(a, fft_len_conv, conv_len)
    best_J = 2.0 * n * conv0.max()
    best_a = a.copy()
    last_improve = start_time
    iteration = 0

    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * (elapsed / (deadline - start_time))

        # ----- 4.1 gradient step (soft‑max) -----
        grad, conv_max = gradient_softmax(a, beta,
                                          fft_len_conv, conv_len,
                                          fft_len_grad)

        # step size decays sqrt
        step = eta0 / np.sqrt(iteration + 1.0)

        # multiplicative (exponential) update that stays on simplex automatically
        a = a * np.exp(-step * grad)
        a = a / a.sum()

        # ----- 4.2 occasional exchange move (large‑scale re‑balancing) -----
        if rng.random() < exchange_prob:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(a[i], rng.random() * max_transfer)
            a[i] -= delta
            a[j] += delta
            # renormalise (the sum is unchanged, but small numerical drift may appear)
            a = a / a.sum()

        # ----- 4.3 evaluate -----
        J = 2.0 * n * conv_max    # because sum(a)=1 by construction
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- 4.4 stagnation handling -----
        if now - last_improve > jitter_interval:
            # reset a to a jittered version of the best seen so far
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = best_a * noise
            a = a / a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – reset around best")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final sanitisation and return
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]

"""
Simulated‑annealing / smooth‑max gradient descent for the step‑function problem.

The objective to minimise is
        J(a) = 2·n·max( a ⊛ a )          (∑ a_i = 1 , a_i ≥ 0)

The non‑smooth max is replaced by the log‑sum‑exp (soft‑max)

        fβ(a) = (1/β)·log Σ_k exp(β·(a⊛a)[k]).

For a fixed temperature β the gradient of fβ is
        ∇fβ(a)_i = 2· Σ_k p_k·a_{k‑i},
where p_k = exp(β·conv_k) / Σ_j exp(β·conv_j) are the soft‑max weights.
The algorithm starts with a low β (very smooth) and raises it
exponentially up to a large value (≈800) – a deterministic annealing
schedule.  At each iteration an exponential (multiplicative) update
a_i ← a_i·exp(−η·∇fβ_i) is performed; normalising the vector restores
the simplex constraint.  Occasionally a large‑scale “exchange” move
re‑balances two random positions, and after a long period without
improvement a small random jitter centred on the current best solution
is applied to escape plateaus.

All operations are performed with real FFTs, giving O(n log n) per
iteration.  The routine respects the 1000 s budget and returns the best
sequence found.
"""

        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]

def _conv_full_general(x: np.ndarray, y: np.ndarray,
                       fft_len: int) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    return np.fft.irfft(X * Y, n=fft_len)


def gradient_softmax(a: np.ndarray, beta: float,
                     fft_len_conv: int, conv_len: int,
                     fft_len_grad: int) -> tuple[np.ndarray, float]:
    """
    Gradient of the smooth‑max (log‑sum‑exp) w.r.t. a and the current max
    convolution value.
    Returns (grad, max_conv), where grad has the same shape as a.
    """
    # 1) a ⊛ a
    conv = _conv_full_real(a, fft_len_conv, conv_len)

    # 2) soft‑max weights
    max_c = conv.max()
    # stability: subtract the maximum before exponentiation
    w = np.exp(beta * (conv - max_c))

    grad = 2.0 * grad_full[len(a) - 1: len(a) - 1 + len(a)]
    return grad, max_c

            a[i] -= delta
            a[j] += delta
            a = a / a.sum()

        # ----- 4.5 evaluate the new objective -----
        conv_new = _conv_full_real(a, fft_len_conv, conv_len)
        J = 2.0 * n * conv_new.max()

        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- 4.6 stagnation handling (jitter) -----
        if now - last_improve > jitter_interval:
            # jitter around the best solution seen so far
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = best_a * noise
            a = a / a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – reset around best")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Final sanitisation (clip, re‑normalise) and return
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
