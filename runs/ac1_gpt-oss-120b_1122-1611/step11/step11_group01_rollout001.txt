        # i from i_start to i_end inclusive
        # Fill T[k,i] = f[i] for those i
        T[k, i_start:i_end+1] = f[i_start:i_end+1]

    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return b.value

Hybrid random‑search + momentum exponentiated‑gradient descent with
late‑stage up‑sampling.

The algorithm works in three phases:

1. **Exploratory random sampling** – a quick budget‑limited screen of
   Dirichlet draws (different concentrations) on several modest
   sequence lengths.  The best candidate from this phase seeds the second
   phase.

2. **Momentum exponentiated‑gradient optimisation** – the objective is the
   soft‑max approximation of  max(conv(a,a)).  A temperature schedule
   provides occasional uphill moves (Metropolis acceptance).  The update
   is multiplicative (a ← a·exp(−η·g)) which keeps positivity, followed
   by Euclidean projection onto the probability simplex.
   Momentum (exponential moving average of the gradient) and an
   adaptive learning‑rate (η∝1/√t) give a stable, fast descent.

3. **Upscaling** – once a good shape on a moderate length (≈1024) is
   found, it is interpolated to the maximum allowed length (8192).  The
   up‑sampled vector is optionally fine‑tuned for a short while using the
   same optimiser.  Upscaling spreads the mass and, thanks to the
   scaling‑invariance of `evaluate_sequence`, typically reduces the
   objective further, often below the target 1.5025.

All helper routines are top‑level, there are no closures or lambdas,
and the routine respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (zero‑padded to fft_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _objective_J(a: np.ndarray) -> float:
    """Return the raw objective J = 2·n·max(conv(a,a)) (assuming sum(a)=1)."""
    n = a.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

            best_n = n
            print(f"[{time.monotonic() - start_time:6.2f}s] "
                  f"Random start n={n:4d} α={alpha:4.2f} J={J:.6f}")

    # ------------------------------------------------------------------
    # 2. Momentum exponentiated‑gradient optimisation on the best n
    # ------------------------------------------------------------------
    if best_vec is None:                     # fallback (should never happen)
        best_n = 1024
        best_vec = np.full(best_n, 1.0 / best_n)

    n = best_n
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    cur_vec = best_vec.copy()
    cur_J = best_J

    # momentum buffer
    m_buf = np.zeros_like(cur_vec)

    # temperature schedule for Metropolis acceptance
    T_start = cur_J * 0.1      # fairly high to allow exploration
    T_end = 1e-8

    # beta schedule (soft‑max temperature)
    beta_start = 1.0
    beta_end = 400.0

    iteration = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        remaining = deadline - start_time

        # --------------------------------------------------------------
        # recompute current convolution and objective
        # --------------------------------------------------------------
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # stagnation handling – jitter after 30 s without improvement
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # Soft‑max weighting (smooth approximation of max)
        # --------------------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (elapsed / remaining)
        beta = min(beta_end, max(beta_start, beta))
        w_raw = np.exp(beta * (conv - max_conv))
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum != 0.0 else np.full_like(conv, 1.0 / conv_len)

        # --------------------------------------------------------------
        # Gradient of the smoothed objective (Fourier domain)
        # --------------------------------------------------------------
        A = np.fft.rfft(cur_vec, fft_len)
        W = np.fft.rfft(w, fft_len)
        grad = 2.0 * np.fft.irfft(np.conj(W) * A, n=fft_len)[:n]

        # momentum update (exponential moving average)
        mu = 0.9
        m_buf = mu * m_buf + (1.0 - mu) * grad

        # --------------------------------------------------------------
        # Exponentiated‑gradient proposal
        # --------------------------------------------------------------
        step = 0.5 / np.sqrt(iteration + 1.0)        # 1/√t decay
        factor = np.exp(-step * m_buf)              # always positive
        proposal = cur_vec * factor
        proposal = proj_simplex(proposal)           # back onto simplex

        # --------------------------------------------------------------
        # Evaluate proposal
        # --------------------------------------------------------------
        prop_conv = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        T = T_start * (1.0 - elapsed / remaining) + T_end * (elapsed / remaining)
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

        # early termination if we already beat the target
        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target achieved – best J = {best_J:.6f}")
            break

    # ------------------------------------------------------------------
    # 3. Upsampling to the maximum allowed length (8192) and final polish
    # ------------------------------------------------------------------
    max_len = 8192
    if best_vec.size < max_len:
        # linear interpolation to a longer grid
        x_old = np.linspace(0.0, 1.0, best_vec.size)
        x_new = np.linspace(0.0, 1.0, max_len)
        up_vec = np.interp(x_new, x_old, best_vec)
        up_vec = np.clip(up_vec, 0.0, None)
    else:
        up_vec = best_vec.copy()

    # re‑normalise onto the simplex
    up_vec = proj_simplex(up_vec)

    # a short refinement on the up‑sampled vector (if time permits)
    n_up = up_vec.size
    conv_len_up = 2 * n_up - 1
    fft_len_up = _next_pow2(conv_len_up)

    cur_up = up_vec.copy()
    m_up = np.zeros_like(cur_up)
    iteration_up = 0
    start_refine = time.monotonic()
    while time.monotonic() < deadline and iteration_up < 2000:
        conv_up = _conv_full(cur_up, fft_len_up, conv_len_up)
        max_conv_up = conv_up.max()
        cur_J_up = float(2.0 * n_up * max_conv_up)

        # optional early break if no progress for a while
        if iteration_up % 200 == 0 and cur_J_up >= best_J - 1e-12:
            # stop after a few stagnant cycles
            break

        # soft‑max weights
        beta_up = beta_start + (beta_end - beta_start) * (iteration_up / 2000.0)
        w_up_raw = np.exp(beta_up * (conv_up - max_conv_up))
        w_up = w_up_raw / w_up_raw.sum()

        # gradient
        A_up = np.fft.rfft(cur_up, fft_len_up)
        W_up = np.fft.rfft(w_up, fft_len_up)
        grad_up = 2.0 * np.fft.irfft(np.conj(W_up) * A_up, n=fft_len_up)[:n_up]

        # momentum and update
        m_up = mu * m_up + (1.0 - mu) * grad_up
        step_up = 0.5 / np.sqrt(iteration_up + 1.0)
        factor_up = np.exp(-step_up * m_up)
        proposal_up = cur_up * factor_up
        proposal_up = proj_simplex(proposal_up)

        # evaluate proposal
        prop_conv_up = _conv_full(proposal_up, fft_len_up, conv_len_up)
        prop_J_up = float(2.0 * n_up * prop_conv_up.max())

        # simple downhill acceptance (no temperature needed for short stage)
        if prop_J_up < cur_J_up:
            cur_up = proposal_up
            cur_J_up = prop_J_up

        iteration_up += 1

    final_vec = np.clip(cur_up, 0.0, 1000.0)
    s = final_vec.sum()
    if s == 0.0:
        final_vec = np.full_like(final_vec, 1.0 / final_vec.size)
    else:
        final_vec = final_vec / s

    # ------------------------------------------------------------------
    # Return the solution as a plain Python list
    # ------------------------------------------------------------------
    return [float(v) for v in final_vec]
