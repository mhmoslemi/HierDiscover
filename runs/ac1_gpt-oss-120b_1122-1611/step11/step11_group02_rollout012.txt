"""
A simple multi‑phase stochastic search for a step‑function that minimises
   J(a) = 2·n·max(conv(a,a)) / (sum(a)**2) .
The search works with probability vectors (sum = 1) – scaling does not affect
the score – and proceeds as follows:

* Phase 0 – deterministic seeding: a small portfolio of analytic families
  (uniform, linear, geometric, power‑law and mixed geometric×power) for a
  handful of lengths.  The best of these is kept as the initial seed.

* Phase 1 – random hill‑climbing on the simplex.  Each iteration mutates the
  current vector by multiplicative log‑normal noise, projects onto the
  simplex and accepts the move if it improves J (with a tiny chance of
  accepting a worsening move to escape plateaus).  A long‑no‑improvement
  timeout triggers a jittered restart.

* Phase 2 – fine‑grained random search around the best discovered vector.

All convolutions are performed with a real FFT for speed.  The routine runs
until the supplied time budget (≈ 1000 s) expires and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def eval_J(vec: np.ndarray, fft_len: int) -> float:
    """
    Compute the evaluation function (without the extra clipping done by
    `evaluate_sequence`).  The vector should be non‑negative; the sum does
    not need to be 1 because the ratio is scale‑invariant.
    """
    n = vec.size
    s = vec.sum()
    if s < 1e-12:
        return float("inf")
    conv = conv_full_fft(vec, fft_len, 2 * n - 1)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the evaluation metric.  Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0.  Timing
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # keep a small safety margin for the host environment
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1.  Determine a set of candidate lengths.
    # ------------------------------------------------------------------
    # The previous run used a length of ~3800.  We explore a modest
    # neighbourhood around that size and also a few smaller sizes.
    candidate_lengths = [2000, 3000, 3800, 4600, 5600, 7200, 9600]

    best_vec = None
    best_J = float("inf")

    # ------------------------------------------------------------------
    # 2.  Deterministic seeding – scan a few analytic families.
    # ------------------------------------------------------------------
    print("[0.00s] Starting deterministic scans.")
    for n in candidate_lengths:
        fft_len = _next_pow2(2 * n - 1)

        # uniform
        v = np.full(n, 1.0 / n, dtype=np.float64)
        cur_J = eval_J(v, fft_len)
        if cur_J < best_J:
            best_J, best_vec = cur_J, v.copy()
        # linear decreasing (high at start, zero at end)
        v = np.arange(n, 0, -1, dtype=np.float64)
        v /= v.sum()
        cur_J = eval_J(v, fft_len)
        if cur_J < best_J:
            best_J, best_vec = cur_J, v.copy()
        # linear increasing (mirror of above – usually worse but cheap)
        v = np.arange(1, n + 1, dtype=np.float64)
        v /= v.sum()
        cur_J = eval_J(v, fft_len)
        if cur_J < best_J:
            best_J, best_vec = cur_J, v.copy()
        # geometric families
        for r in np.logspace(np.log10(0.94), np.log10(0.9999), num=30):
            v = r ** np.arange(n, dtype=np.float64)
            s = v.sum()
            if s > 0:
                v /= s
                cur_J = eval_J(v, fft_len)
                if cur_J < best_J:
                    best_J, best_vec = cur_J, v.copy()
        # power‑law families
        for beta in np.linspace(0.5, 3.0, num=20):
            v = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** beta)
            s = v.sum()
            if s > 0:
                v /= s
                cur_J = eval_J(v, fft_len)
                if cur_J < best_J:
                    best_J, best_vec = cur_J, v.copy()
        # mixed geometric × power‑law
        for r in np.logspace(np.log10(0.94), np.log10(0.9995), num=12):
            for beta in np.linspace(0.5, 2.0, num=8):
                v = (r ** np.arange(n, dtype=np.float64)) * (
                    1.0 / (np.arange(1, n + 1, dtype=np.float64) ** beta)
                )
                s = v.sum()
                if s > 0:
                    v /= s
                    cur_J = eval_J(v, fft_len)
                    if cur_J < best_J:
                        best_J, best_vec = cur_J, v.copy()

    # If a previous construction exists, try it as a seed (truncated / padded).
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            # match the current best length
            if prev.size != best_vec.size:
                if prev.size > best_vec.size:
                    prev = prev[: best_vec.size]
                else:
                    pad = np.zeros(best_vec.size - prev.size, dtype=np.float64)
                    prev = np.concatenate([prev, pad])
                prev = proj_simplex(prev)
            prev_sum = prev.sum()
            if prev_sum > 0:
                prev /= prev_sum
                cur_J = eval_J(prev, _next_pow2(2 * prev.size - 1))
                if cur_J < best_J:
                    best_J, best_vec = cur_J, prev.copy()
        except Exception:
            pass

    print(f"[{time.monotonic() - start_time:6.2f}s] "
          f"Deterministic best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3.  Phase‑1: Random hill‑climbing on the simplex.
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    n = cur_vec.size
    fft_len = _next_pow2(2 * n - 1)

    iteration = 0
    last_improve = time.monotonic()
    stagnation_timeout = 30.0          # seconds without improvement → jitter

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # 3.1  Propose a multiplicative perturbation.
        # ------------------------------------------------------------------
        sigma = 0.12 if iteration < 2000 else 0.05
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = cur_vec * noise
        cand = proj_simplex(cand)

        # ------------------------------------------------------------------
        # 3.2  Evaluate.
        # ------------------------------------------------------------------
        cand_J = eval_J(cand, fft_len)

        # ------------------------------------------------------------------
        # 3.3  Accept if better, or with a tiny probability otherwise.
        # ------------------------------------------------------------------
        accept = False
        if cand_J < cur_J:
            accept = True
        elif rng.random() < 1e-6:
            accept = True

        if accept:
            cur_vec, cur_J = cand, cand_J
            if cand_J < best_J:
                best_J, best_vec = cand_J, cand.copy()
                last_improve = time.monotonic()
                print(f"[{time.monotonic() - start_time:6.2f}s] "
                      f"New best J = {best_J:.6f}")

        iteration += 1

        # ------------------------------------------------------------------
        # 3.4  Stagnation handling.
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > stagnation_timeout:
            # jitter around the global best to escape plateaus
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            cur_vec = proj_simplex(best_vec * jitter)
            cur_J = eval_J(cur_vec, fft_len)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] "
                  f"Restart after stagnation – best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4.  Phase‑2: Fine‑grained random search around the best vector.
    # ------------------------------------------------------------------
    fine_deadline = time.monotonic() + min(30.0, (deadline - start_time) * 0.07)
    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_J = eval_J(cand, fft_len)
        if cand_J < best_J:
            best_J, best_vec = cand_J, cand.copy()
            print(f"[Fine] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5.  Normalise the final vector (scale‑invariant, but keeps it tidy).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"[Done] Final best J = {best_J:.6f}  (length {n})")
    return [float(v) for v in best_vec]
