"""
Project‑gradient search for the minimum of
    J(a) = 2·n·max(conv(a,a)) / (sum(a)^2)   (with sum(a)>0)

The objective is scale‑invariant, therefore we keep the vector on the
probability simplex (sum = 1, a_i ≥ 0).  The exact non‑smooth “max’’ is
handled by a sub‑gradient:
    ∂J/∂a = 2·(w ⋆ a)    with  w_k = 1 iff conv_k is maximal,
where “⋆’’ denotes the linear correlation.  The correlation is obtained
by a real FFT (O(n log n)).  At each iteration we take a multiplicative
exponentiated‑gradient step and re‑project onto the simplex.  A simple
stagnation detector injects log‑normal jitter and, if necessary,
re‑starts from a fresh Dirichlet draw.  The best vector seen before the
time budget expires is returned.

The routine works for any length; if a previous construction is
available via the global variable ``height_sequence_1`` it is used as
the initial point, otherwise a uniform distribution is taken.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or negative) → uniform solution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _subgradient(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """
    Sub‑gradient of the (unsmoothed) max‑convolution objective.
    ``w`` is a binary mask (same length as the full convolution) with 1s
    on the positions that achieve the maximal convolution value.
    Returns a vector of length len(a).
    """
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    W_hat = np.fft.rfft(w, L)
    # linear correlation of w with a
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)
    n = a.size
    start = n - 1               # index where shift = 0
    grad = 2.0 * corr[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine – the function required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector)
    that minimises ``evaluate_sequence``.
    Returns the best sequence found before the budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Determine problem size (length of the step function)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)
            if base.size == 0:
                raise ValueError
        except Exception:
            base = None
    else:
        base = None

    if base is not None:
        n = int(base.size)
        # ensure n is not too small or too huge for the time budget
        n = max(256, min(n, 8192))
        a = base.copy()
        if a.sum() == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a.sum()
    else:
        n = 4096                     # fallback size
        a = np.full(n, 1.0 / n, dtype=np.float64)

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)               # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)       # for correlation

    # --------------------------------------------------------------
    # 2. Initialise bookkeeping
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # best solution seen so far (in probability simplex representation)
    best_a = a.copy()
    # compute its objective
    conv = _conv_full(a, fft_len_conv, conv_len)
    best_J = float(2.0 * n * conv.max())
    print(f"[{0.00:6.2f}s] initial J = {best_J:.6f}")

    cur_a = a.copy()
    cur_J = best_J
    last_improve = start_time

    # step‑size schedule (additive decay via 1/√iteration)
    step0 = 0.30

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---------------------------------------------------------
        # 2.1  Convolution and exact objective
        # ---------------------------------------------------------
        conv = _conv_full(cur_a, fft_len_conv, conv_len)
        max_val = conv.max()
        cur_J = float(2.0 * n * max_val)

        # ---------------------------------------------------------
        # 2.2  Keep global best
        # ---------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = cur_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best J = {best_J:.6f}")

        # ---------------------------------------------------------
        # 2.3  Stagnation handling – jitter after 30 s without improvement
        # ---------------------------------------------------------
        if now - last_improve > 30.0:
            # log‑normal jitter keeps the vector on the simplex after re‑projection
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_a = best_a * noise
            cur_a = proj_simplex(cur_a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] jitter restart – best J = {best_J:.6f}")

        # ---------------------------------------------------------
        # 2.4  Identify the active (maximal) convolution positions
        # ---------------------------------------------------------
        eps = 1e-14 * max_val   # tolerance for numerical ties
        max_pos = np.where(conv >= max_val - eps)[0]
        w = np.zeros(conv_len, dtype=np.float64)
        w[max_pos] = 1.0

        # ---------------------------------------------------------
        # 2.5  Sub‑gradient of the max‑convolution objective
        # ---------------------------------------------------------
        grad = _subgradient(cur_a, w, fft_len_corr)

        # ---------------------------------------------------------
        # 2.6  Exponentiated‑gradient step (multiplicative update)
        # ---------------------------------------------------------
        eta = step0 / np.sqrt(iteration + 1.0)   # decreasing step size
        # protect against overflow/under‑flow
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -700, 0)   # np.exp(-700) ≈ 5e-305
        factor = np.exp(log_factor)
        proposal = cur_a * factor
        proposal = proj_simplex(proposal)   # back to simplex

        # ---------------------------------------------------------
        # 2.7  Accept the proposal (always – deterministic descent)
        # ---------------------------------------------------------
        cur_a = proposal

        # ---------------------------------------------------------
        # 2.8  Occasionally perform a *large* random restart to diversify
        # ---------------------------------------------------------
        if iteration % 20000 == 0 and iteration > 0:
            # draw a fresh Dirichlet vector (moderately concentrated)
            alpha = rng.uniform(0.2, 1.0)
            cur_a = rng.dirichlet(np.full(n, alpha))
            # keep the best so far unchanged, just explore
            print(f"[{elapsed:6.2f}s] random restart (α={alpha:.3f})")

        iteration += 1

    # --------------------------------------------------------------
    # 3. Return the best solution (clip to allowed range, renormalise)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
