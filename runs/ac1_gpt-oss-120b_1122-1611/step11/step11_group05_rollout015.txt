def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---
    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _gradient_max(a: np.ndarray, max_mask: np.ndarray, fft_len_corr: int) -> np.ndarray:
    # max_mask is binary float array of length conv_len (2n-1) indicating active sums.
    # Compute gradient 2 * (max_mask ⋆ a)
    w = max_mask  # shape (conv_len,)
    # correlation via FFT:
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    W_hat = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)
    n = a.size
    start = n - 1
    grad = 2.0 * corr[start:start + n]
    return grad
```

def greedy_sidon(n, seed, attempts=10):
    best = []
    rng = np.random.default_rng(seed)
    for _ in range(attempts):
        order = rng.permutation(n)
        S = []
        used = set()
        for x in order:
            ok = True
            for y in S:
                s = x + y
                if s in used:
                    ok = False
                    break
            if ok:
                for y in S:
                    used.add(x + y)
                used.add(2*x)
                S.append(x)
        if len(S) > len(best):

"""
Step‑function optimiser.

The objective is   J(a) = 2·n·max_i (a*a)[i] / (sum a)^2 .
All entries are clipped to [0,1000] before evaluation, therefore we work
with a probability vector (sum = 1) and minimise 2·n·max(conv).

The algorithm:

1. Build a diverse portfolio of initial candidates.
   * Uniform, geometric decays, ramps, Gaussians.
   * Random Dirichlet draws.
   * Sparse “spike’’ masks.
   * Several greedy Sidon‑set constructions (sets whose pairwise sums are
     (almost) distinct – they give a low max‑convolution for equal weights).

2. Keep the best initial candidate (exact J).

3. Refine it with an exponentiated‑gradient on the *hard* max
   (the active set of convolution indices).  The gradient of
   max(conv) w.r.t. a is 2·(w⋆a) where w is a binary mask of the
   active indices.  Updates are multiplicative, projected back onto the
   probability simplex.

4. Simulated‑annealing style acceptance: always accept a strictly
   improving step, otherwise accept with probability exp(−ΔJ/T) with a
   linear cooling schedule.

5. If no improvement for 30 s the iterate is jittered (log‑normal noise
   then re‑projected) to escape plateaux.

The loop runs until the time budget expires (≈ 1000 s).  The best vector
found is returned after clipping to [0,1000] and re‑normalising.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater or equal to x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex requires a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (the full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _gradient_max(a: np.ndarray, mask: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """
    Gradient of the *hard* max‑convolution objective.
    mask – binary (0/1) array of length 2n‑1 indicating the active indices.
    Returns 2·(mask ⋆ a)  (cross‑correlation, length n).
    """
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    M_hat = np.fft.rfft(mask, L)
    corr = np.fft.irfft(np.conj(A_hat) * M_hat, n=L)
    n = a.size
    start = n - 1                     # position of zero shift in linear conv
    grad = 2.0 * corr[start:start + n]
    return grad


def _greedy_sidon(n: int, rng: np.random.Generator, attempts: int = 15) -> list[int]:
    """
    Greedy construction of a (near) Sidon set.
    Repeats the random order ``attempts`` times and returns the largest set found.
    """
    best_set = []
    for _ in range(attempts):
        order = rng.permutation(n)
        cur_set = []
        used = set()                   # stores sums that already appear
        for x in order:
            ok = True
            # check sums with already selected elements
            for y in cur_set:
                if (x + y) in used:
                    ok = False
                    break
            if not ok:
                continue
            # check the self‑sum
            if (2 * x) in used:
                continue
            # accept x
            for y in cur_set:
                used.add(x + y)
            used.add(2 * x)
            cur_set.append(x)
        if len(cur_set) > len(best_set):
            best_set = cur_set
    return best_set


# ----------------------------------------------------------------------
# Main optimisation routine – the function required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size (length of the step function)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep it in a reasonable range
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a comfortable size for FFT

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)                # for a*a
    # for correlation we need size >= conv_len + n - 1
    fft_len_corr = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    # 2. Build a diverse portfolio of initial probability vectors
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells (few widths)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderately / strongly concentrated)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Sparse masks – a few equally weighted spikes
    for k in [int(np.sqrt(n)), int(np.sqrt(n) * 2), max(2, n // 10)]:
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        init_cands.append(mask)

    # Random binary mask (Bernoulli 0.5)
    bin_mask = rng.binomial(1, 0.5, size=n).astype(np.float64)
    if bin_mask.sum() > 0:
        bin_mask /= bin_mask.sum()
        init_cands.append(bin_mask)

    # Greedy Sidon‑set candidates (several attempts)
    for attempt in range(5):
        sid_set = _greedy_sidon(n, rng, attempts=20)
        if sid_set:
            v = np.zeros(n, dtype=np.float64)
            v[sid_set] = 1.0
            v /= v.sum()
            init_cands.append(v)

    # previous construction (if present) – normalised to a probability vector
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v = v / s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best of the initial candidates (true J)
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len_conv, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop – hard‑max exponentiated‑gradient + annealing
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time

    # temperature schedule (linear cooling)
    T_start = best_J * 0.05   # 5 % of the current best
    T_end = 1e-8

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # exact convolution and true objective
        conv = _conv_full(cur_vec, fft_len_conv, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # keep global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
            # early exit if we already beat the target
            if best_J <= 1.5025:
                break

        # ------------------------------------------------------------------
        # Stagnation handling – jitter after 30 s without improvement
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.1 Gradient of the *hard* max‑convolution objective
        # ------------------------------------------------------------------
        mask = (conv >= max_conv - 1e-12).astype(np.float64)   # active set
        grad = _gradient_max(cur_vec, mask, fft_len_corr)

        # ------------------------------------------------------------------
        # 4.2 Propose a new point – exponentiated‑gradient step
        # ------------------------------------------------------------------
        step = 0.6 / np.sqrt(iteration + 1.0)   # decaying learning rate
        factor = np.exp(-step * grad)          # multiplicative update
        proposal = cur_vec * factor
        proposal = np.clip(proposal, 0.0, None)
        s = proposal.sum()
        if s == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= s

        # ------------------------------------------------------------------
        # 4.3 Evaluate proposal (exact J)
        # ------------------------------------------------------------------
        prop_conv = _conv_full(proposal, fft_len_conv, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ------------------------------------------------------------------
        # 4.4 Metropolis acceptance (linear cooling)
        # ------------------------------------------------------------------
        T = (T_start * (1.0 - elapsed / (deadline - start_time)) +
             T_end * (elapsed / (deadline - start_time)))
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (clip and renormalise for safety)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
