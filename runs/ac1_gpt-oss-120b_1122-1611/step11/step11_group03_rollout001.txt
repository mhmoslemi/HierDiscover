def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    # choose n
    if 'height_sequence_1' in globals():
        n = len(globals()['height_sequence_1'])
        n = max(512, min(n, 8192))
    else:
        n = 4096  # maybe 4096

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    fft_len_corr = _next_pow2(conv_len + n - 1)

    # initial a: perhaps mixture of some patterns
    # uniform, peaked edges, random Dirichlet with low alpha
    # we will initialize parameters z from log(a) of randomly chosen candidate.

    def init_candidate():
        # random Dirichlet with alpha randomly chosen from small set.
        alpha = rng.choice([0.02, 0.05, 0.1, 0.2, 0.5, 1.0])
        a = rng.dirichlet(np.ones(n) * alpha)
        # optionally pad with small constant for support
        return a

    # build initial candidates and pick best based on objective
    num_initial = 5
    best_obj = np.inf
    best_a = None
    for i in range(num_initial):
        a = init_candidate()
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv # sum=1
        if obj < best_obj:
            best_obj = obj
            best_a = a

    # Keep best sequence from previous global variable also if available.
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0:
            a0 = np.full_like(a0, 1.0/a0.size)
        else:
            a0 = a0 / a0.sum()
        conv = _conv_full(a0, fft_len, conv_len)
        obj = 2*n*conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a0

    # Initialize log-weights for softmax: z = log(best_a + epsilon)
    eps = 1e-12
    z = np.log(best_a + eps)

    # Adam hyperparameters
    lr0 = 0.1  # initial learning rate in log-space
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # schedule for beta (softmax smoothing)
    beta_start = 1.0
    beta_end = 1000.0  # high
    # schedule for learning rate: maybe decrease linearly to 0.001
    lr_end = 0.001

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 10.0)  # safety margin

    # track best overall
    best_overall_a = best_a.copy()
    best_overall_obj = best_obj
    # compute evaluate_sequence value for best to log
    try:
        best_score = evaluate_sequence(best_overall_a.tolist())
    except Exception:
        best_score = np.inf

    iteration = 0
    last_improve_time = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        # compute schedule parameters
        t_frac = elapsed / (deadline - start)
        beta = beta_start * (beta_end / beta_start) ** t_frac
        lr = lr0 * (lr_end / lr0) ** t_frac

        # compute a from z via softmax
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # compute convolution
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv  # our simplified objective

        # softmax weights

        w_exp = np.exp(beta * (conv - max_conv))
        w_sum = w_exp.sum()
        if w_sum == 0.0:

        else:

        grad_a = _weighted_gradient(a, w, fft_len_corr, n)

        # gradient w.r.t z (softmax Jacobian)
        dot = np.dot(grad_a, a)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # update best if improved
        if obj < best_overall_obj - 1e-12:
            best_overall_obj = obj
            best_overall_a = a.copy()
            last_improve_time = now
            # compute full evaluate
            try:
                best_score = evaluate_sequence(best_overall_a.tolist())
                print(f"[{elapsed:6.2f}s] New best obj = {obj:.6f}, eval = {best_score:.6f}")
            except Exception:
                best_score = np.inf

        # handle stagnation: if no improvement for >30 sec, random restart
        if now - last_improve_time > 30.0:
            # jitter z around best
            z = np.log(best_overall_a + eps) + rng.normal(0.0, 0.1, size=n)

            v = np.zeros_like(z)
            last_improve_time = now
            print(f"[{elapsed:6.2f}s] Stagnation restart")

        iteration += 1

    # final best vector normalized to sum=1 (already)
    best_overall_a = np.clip(best_overall_a, 0.0, 1000.0)
    total = best_overall_a.sum()
    if total == 0.0:
        best_overall_a[:] = 1.0 / n
    else:
        best_overall_a /= total

    # return as list of floats
    return best_overall_a.tolist()

assistantfinal**Solution Overview**

The optimisation goal is  

\[
\min_{a_i\ge 0,\;\sum a_i=1}\;J(a)=2\,n\;\max_k\bigl((a*a)_k\bigr),
\]

because `evaluate_sequence` equals  
\(2\,n\max(a*a)/(\sum a)^2\) and the denominator is constant when the
vector is kept on the probability simplex.  
The problem is non‑convex (quadratic constraints) but the max‑norm can be
smoothly approximated by a soft‑max:

\[
\operatorname{softmax}_\beta(x)_k=
\frac{\exp\bigl(\beta\,(x_k-\max x)\bigr)}{\sum_j\exp\bigl(\beta\,(x_j-\max x)\bigr)} .
\]

For a large \(\beta\) this approaches the true \(\ell_\infty\) norm and
its gradient w.r.t. the vector \(a\) is a *weighted correlation*
that can be computed with a single FFT.

The optimiser works on the **log‑softmax parametrisation**

\[
a_i = \frac{e^{z_i}}{\sum_j e^{z_j}},
\]

so the simplex constraints are satisfied automatically.
An Adam optimiser is used on the unconstrained variable `z`.  
During optimisation the smoothing parameter `β` is increased
exponentially, while the Adam learning rate is decreased
linearly.  Periodically an improvement is recorded using the real
`evaluate_sequence` function; if no improvement occurs for 30 s a
random jitter is added to `z` to escape possible plateaus.

The whole routine runs for the supplied time budget (≤ 1000 s) and
returns the best sequence found.

```python
"""
Hybrid Adam optimiser on the simplex (soft‑max parametrisation) with
exponential soft‑max temperature increase.  The algorithm keeps the
vector on the probability simplex, computes the auto‑convolution by FFT,
uses a soft‑max approximation of the ℓ∞ norm to obtain a smooth
gradient, and updates the log‑probabilities with Adam.  Stagnation is
detected and a random jitter is added to escape local minima.  The best
sequence (scaled to have sum = 1) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    a has length n, result length is 2·n‑1 = conv_len.
    """
    # real FFT of a
    A = np.fft.rfft(a, fft_len)
    # element‑wise square gives the spectrum of the auto‑convolution
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute the gradient of  Σ_k weight_k·(a*a)_k   w.r.t. a.
    For a soft‑max weight this is exactly the sub‑gradient of the smoothed
    ℓ∞‑norm.
    Returns a vector of length n.
    """
    # reverse a for correlation
    a_rev = a[::-1]
    # FFT of weight (length conv_len) and reversed a (length n)
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    # correlation (full length = conv_len + n - 1)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Uses an Adam optimiser on the probability simplex
    together with an exponentially increasing soft‑max temperature.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size (FFT friendly, not too small, not too large)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # clamp to a reasonable interval
            n = max(512, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlation

    # --------------------------------------------------------------
    # 2. Initial candidates (different shapes, Dirichlet, previous best)
    # --------------------------------------------------------------
    def random_dirichlet(alpha):
        """Return a Dirichlet draw with concentration alpha."""
        return rng.dirichlet(np.ones(n) * alpha)

    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_candidates.append(v)

    # linear ramps (up and down)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_candidates.append(v)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_candidates.append(v)

    # Gaussian bells of different widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_candidates.append(v)

    # Dirichlet draws – sparse to dense
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            init_candidates.append(random_dirichlet(alpha))

    # previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s == 0.0:
            prev[:] = 1.0 / n
        else:
            prev /= s
        init_candidates.append(prev)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (cheap objective, no Adam yet)
    # --------------------------------------------------------------
    best_a = None
    best_obj = np.inf
    for cand in init_candidates:
        conv = _conv_full(cand, fft_len, conv_len)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv        # sum=1 → objective equals 2·n·max_conv
        if obj < best_obj:
            best_obj = obj
            best_a = cand.copy()

    # --------------------------------------------------------------
    # 4. Adam optimiser on the log‑softmax parameters
    # --------------------------------------------------------------
    eps = 1e-12
    z = np.log(best_a + eps)              # initialise log‑probabilities

    # Adam hyper‑parameters
    lr_start = 0.12                       # initial learning rate in log‑space
    lr_end = 0.001                        # final learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Soft‑max temperature schedule (β goes from 1 → 1000 exponentially)
    beta_start = 1.0
    beta_end = 1000.0

    start_time = time.monotonic()
    # keep a safety margin (10 s) for clean termination
    deadline = start_time + max(0.0, float(budget_s) - 10.0)
    total_time = deadline - start_time

    # ------------------------------ tracking ------------------------------
    best_overall_a = best_a.copy()
    best_overall_obj = best_obj
    try:
        best_score = evaluate_sequence(best_overall_a.tolist())
    except Exception:
        best_score = np.inf

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        progress = (elapsed / total_time)               # ∈[0,1]

        # ---- schedule for β and learning rate ---------------------------------
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---- current probability vector (soft‑max of z) -----------------------
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()          # sum = 1, a_i ≥ 0

        # ---- auto‑convolution -------------------------------------------------
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv      # our surrogate objective

        # ---- soft‑max weights for a smooth ℓ∞‑approximation ------------------
        # shift by max_conv for numerical stability
        w_exp = np.exp(beta * (conv - max_conv))
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum

        # ---- gradient w.r.t. a (weighted correlation) -----------------------
        grad_a = _weighted_gradient(a, w, fft_len_corr, n)

        # ---- gradient w.r.t. z (soft‑max Jacobian) ---------------------------
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # ---- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- record improvement ------------------------------------------------
        if cur_obj < best_overall_obj - 1e-12:
            best_overall_obj = cur_obj
            best_overall_a = a.copy()
            last_improve = now
            try:
                best_score = evaluate_sequence(best_overall_a.tolist())
            except Exception:
                best_score = np.inf
            print(f"[{elapsed:6.2f}s] New best obj={cur_obj:.6f}, eval={best_score:.6f}")

        # ---- stagnation handling ------------------------------------------------
        if now - last_improve > 30.0:
            # add a moderate Gaussian jitter around the current best
            noise = rng.normal(0.0, 0.09, size=n)
            z = np.log(best_overall_a + eps) + noise
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stagnation – jitter restart")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Final normalisation (sum = 1, clip to allowed range)
    # --------------------------------------------------------------
    best_overall_a = np.clip(best_overall_a, 0.0, 1000.0)
    total = best_overall_a.sum()
    if total == 0.0:
        best_overall_a[:] = 1.0 / n
    else:
        best_overall_a /= total

    # Return as plain Python list (the evaluation routine expects a list)
    return [float(v) for v in best_overall_a]
```
