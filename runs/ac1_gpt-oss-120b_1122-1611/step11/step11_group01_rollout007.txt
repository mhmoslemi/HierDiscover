"""
Hybrid simulated‑annealing + targeted “max‑conv” moves.

The objective J = 2·n·max(conv) (the sum is fixed to 1) is invariant
under scaling, so we keep the sequence on the probability simplex.
At each step we locate the index `k_max` where the self‑convolution
is largest and move a small amount of probability mass from a source
position that contributes heavily to this peak to a destination that
contributes little.  The convolution update can be performed in O(n)
by slicing because a move changes the convolution by
    Δconv = 2·δ·(shift_dst – shift_src)
where `shift_i[k] = a[k‑i]`.  A Metropolis acceptance rule with a
linearly cooling temperature lets the search escape plateaus.
If no improvement occurs for 30 s we jitter the current best solution
and re‑project onto the simplex.  Periodically the full convolution is
re‑computed with an FFT to eliminate accumulated numerical drift.

The routine runs for the supplied time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

    # --------------------------------------------------------------
    conv = _conv_full(a, fft_len, conv_len)
    cur_J = float(2.0 * n * conv.max())
    best_J = cur_J
    best_a = a.copy()
    best_conv = conv.copy()

    print(f"[0.00s] Initial J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 4. Annealing schedule & steps
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    total_time = deadline - start_time

    # temperature (linear cooling)
    T_start = cur_J * 0.08
    T_end = 1e-12

    # step‑size for mass moves (linear cooling)
    step_start = 0.20
    step_end = 0.001

    # stagnation handling
    last_improve = start_time

    iteration = 0
    RECOMPUTE_EVERY = 5000   # full FFT recompute interval

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # 4.1 Temperature & step‑size
        # ----------------------------------------------------------
        T = T_start * (1.0 - elapsed / total_time) + T_end * (elapsed / total_time)
        step_factor = step_start * (1.0 - elapsed / total_time) + step_end * (elapsed / total_time)

        # ----------------------------------------------------------
        # 4.2 Identify the current worst convolution index
        # ----------------------------------------------------------
        k_max = int(conv.argmax())
        max_conv = conv[k_max]

        # ----------------------------------------------------------
        # 4.3 Gradient‑like weights: contribution of each entry to k_max
        # ----------------------------------------------------------
        grad = np.zeros(n, dtype=np.float64)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1)
        grad[idx] = a[k_max - idx]

        # source – pick proportionally to the contribution (larger -> more likely)
        sum_grad = grad.sum()
        if sum_grad > 0:
            src = rng.choice(n, p=grad / sum_grad)
        else:
            # fall back to any positive entry
            pos = np.where(a > 1e-14)[0]
            src = rng.choice(pos) if pos.size else rng.integers(n)

        # ensure source has enough mass
        if a[src] <= 1e-14:
            continue

        # destination – pick where the contribution is small
        inv = (grad.max() - grad) + 1e-12
        sum_inv = inv.sum()
        if sum_inv > 0:
            dst = rng.choice(n, p=inv / sum_inv)
        else:
            dst = rng.integers(n)
        if src == dst:
            continue

        # ----------------------------------------------------------
        # 4.4 Propose moving a small amount of mass from src → dst
        # ----------------------------------------------------------
        delta = step_factor * a[src]
        # enforce a[src] - delta ≥ 0
        delta = min(delta, a[src])
        if delta <= 0:
            continue

        # remember the original values at the two positions
        a_src = a[src]
        a_dst = a[dst]

        # ----- update convolution (use the *old* a for the slice update)
        conv[src:src + n] -= 2.0 * delta * a
        conv[dst:dst + n] += 2.0 * delta * a

        # ----- update the probability vector
        a[src] = a_src - delta
        a[dst] = a_dst + delta

        # ----------------------------------------------------------
        # 4.5 Compute new objective
        # ----------------------------------------------------------
        new_J = float(2.0 * n * conv.max())

        # ----------------------------------------------------------
        # 4.6 Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_J = new_J
            # record global best
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a.copy()
                best_conv = conv.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
        else:
            # ----- revert updates
            a[src] = a_src
            a[dst] = a_dst
            conv[src:src + n] += 2.0 * delta * a
            conv[dst:dst + n] -= 2.0 * delta * a
            # cur_J stays unchanged

        iteration += 1

        # ----------------------------------------------------------
        # 4.7 Periodic full recomputation to kill floating‑point drift
        # ----------------------------------------------------------
        if iteration % RECOMPUTE_EVERY == 0:
            conv = _conv_full(a, fft_len, conv_len)

        # ----------------------------------------------------------
        # 4.8 Stagnation handling – jitter after 30 s without improvement
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = best_a * noise
            a = proj_simplex(a)
            conv = _conv_full(a, fft_len, conv_len)
            cur_J = float(2.0 * n * conv.max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4.9 Early exit if target reached
        # ----------------------------------------------------------
        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # --------------------------------------------------------------
    # 5. Return the best solution (clip & renormalise)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    return [float(v) for v in best_a]
