            v = np.concatenate([v, np.zeros(n - v.size)])
        s = v.sum()
        if s > 0:
            v = v / s
        else:
            v = np.full(n, 1.0 / n)
        init.append(v)

    # evaluate initial candidates quickly using FFT
    best_vec = None
    best_J = np.inf
    for cand in init:
        conv = _conv_full(cand, fft_len, conv_len)
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] initial best J = {best_J:.6f}")

    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)   # keep few seconds buffer

    global_best = best_vec.copy()
    global_best_J = best_J
    last_improve = start

    # parameters
    eta0 = 0.5  # initial step size
    jitter_sigma = 0.05

    iteration = 0
    # we will cycle over candidates
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start

        # pick random candidate from init for a new local run
        cur = rng.choice(init)
        # ensure sum 1
        cur = np.clip(cur, 0, None)
        if cur.sum() == 0:
            cur[:] = 1.0 / n
        else:
            cur /= cur.sum()
        cur_J = None

        # local optimisation loop for this start
        inner_deadline = now + 0.5   # spend up to 0.5 s per start
        inner_iter = 0
        while time.monotonic() < inner_deadline:
            cur_conv = _conv_full(cur, fft_len, conv_len)
            max_idx = int(np.argmax(cur_conv))
            cur_J = 2.0 * n * cur_conv.max()
            # update global best
            if cur_J < global_best_J - 1e-12:
                global_best_J = cur_J
                global_best = cur.copy()
                last_improve = time.monotonic()
                print(f"[{elapsed:6.2f}s] new global best J = {global_best_J:.6f}")

            # gradient at worst index
            g = _grad_wrt_max(cur, max_idx)

            # learning rate schedule
            eta = eta0 / np.sqrt(iteration + inner_iter + 1.0)

            # additive update with projection onto simplex
            new_vec = cur - eta * g
            new_vec = np.maximum(new_vec, 0.0)
            s = new_vec.sum()
            if s == 0.0:
                new_vec[:] = 1.0 / n
            else:
                new_vec /= s

            cur = new_vec
            inner_iter += 1
            iteration += 1

        # Stagnation handling: if we've spent too long without improvement add jitter
        if time.monotonic() - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cur = global_best * noise
            cur = np.maximum(cur, 0)
            s = cur.sum()
            cur /= s if s > 0 else (global_best * 0 + np.full(n, 1.0 / n))
            last_improve = time.monotonic()
            print(f"[{time.monotonic()-start:6.2f}s] jitter restart")

    best = np.clip(global_best, 0.0, 1000.0)
    total = best.sum()
    if total == 0.0:
        best[:] = 1.0 / n
    else:
        best /= total
    return [float(v) for v in best]

"""
Propose a step‑function (non‑negative probability vector) that minimises the
`evaluate_sequence` objective.

Algorithm
---------
1. Choose a length `n` (use the length of the previously found construction if
   available, otherwise 4096 – still cheap for FFT‑based convolutions).
2. Build a diverse portfolio of initial probability vectors (uniform,
   Dirichlet draws, sparse masks, the previous solution, …) and keep the one
   with the smallest surrogate objective
        Ĵ = 2·n·max(conv(a,a))
   where `conv` is computed by a real‑FFT (very fast).
3. Starting from that best seed run a **projected exponentiated‑gradient**
   descent that tries to flatten the self‑convolution:
   * Find the index `k*` where the convolution attains its maximum.
   * The derivative of the peak with respect to `a_i` is
        g_i = 2·a_{k*‑i}   (if the index is in range, otherwise 0).
   * Perform a multiplicative update `a ← a·exp(‑η·g)` with a decaying step
     size `η`, then re‑normalise onto the simplex.
4. If no improvement is seen for 30 s a small random jitter is added
   (`a ← a·exp(σ·N(0,1))`) to escape plateaus.
5. The loop continues until the time budget (≈ 1000 s) is exhausted.
   The best vector found is returned (clipped to the limits imposed by
   `evaluate_sequence`).

The method is simple, fully vectorised, and makes heavy use of FFTs, allowing
hundreds of thousands of objective evaluations within the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _grad_wrt_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the *maximum* convolution entry with respect to the vector `a`.
    For the index `k_max` where the convolution is maximal we have
        ∂/∂a_i  Σ_j a_j a_{k_max‑j} = 2·a_{k_max‑i}
    (zero when the shifted index is out of range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_lo = max(0, k_max - (n - 1))
    i_hi = min(n - 1, k_max)
    idx = np.arange(i_lo, i_hi + 1, dtype=int)
    g[idx] = 2.0 * a[k_max - idx]
    return g


# ----------------------------------------------------------------------
# Main optimisation routine – the function required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------------
    # 1. Problem size (length of the step function)
    # ---------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a comfortable default

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # power‑of‑2 ≥ 2n‑1 for FFTs

    # ---------------------------------------------------------------
    # 2. Build a portfolio of diverse initial probability vectors
    # ---------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws (different concentrations)
    for alpha in [0.1, 0.3, 0.6, 1.0, 2.0]:
        for _ in range(2):
            init_candidates.append(rng.dirichlet(np.full(n, alpha)))

    # Sparse masks – a few equally weighted spikes
    for k in [int(np.sqrt(n)), int(np.sqrt(n) * 2), max(2, n // 20)]:
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        init_candidates.append(mask)

    # Random binary mask (Bernoulli 0.5)
    bin_mask = rng.binomial(1, 0.5, size=n).astype(np.float64)
    if bin_mask.sum() > 0:
        bin_mask /= bin_mask.sum()
        init_candidates.append(bin_mask)

    # Previous construction (if present)
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v = v / s
        init_candidates.append(v)

    # ---------------------------------------------------------------
    # 3. Pick the best initial candidate (exact objective)
    # ---------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_candidates:
        conv = _conv_full(cand, fft_len, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ---------------------------------------------------------------
    # 4. Main optimisation loop – projected exponentiated‑gradient on the
    #    worst convolution entry.
    # ---------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    global_best_vec = best_vec.copy()
    global_best_J = best_J
    last_improve = start_time

    # optimisation parameters
    eta0 = 0.8                 # initial step‑size for the multiplicative update
    jitter_sigma = 0.04        # magnitude of occasional jitter
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # compute full convolution and locate the peak
        conv = _conv_full(cur_vec, fft_len, conv_len)
        k_max = int(conv.argmax())
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # keep the globally best vector
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New global best J = {global_best_J:.6f}")

        # -----------------------------------------------------------------
        # 4.1 Gradient of the *maximum* convolution entry
        # -----------------------------------------------------------------
        g = _grad_wrt_max(cur_vec, k_max)

        # -----------------------------------------------------------------
        # 4.2 Multiplicative (exponentiated‑gradient) update + simplex proj.
        # -----------------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1.0)          # decaying step size
        factor = np.exp(-eta * g)                     # element‑wise factor > 0
        cur_vec = cur_vec * factor
        cur_vec = np.clip(cur_vec, 0.0, None)
        s = cur_vec.sum()
        if s == 0.0:
            cur_vec[:] = 1.0 / n
        else:
            cur_vec /= s

        iteration += 1

        # -----------------------------------------------------------------
        # 4.3 Stagnation handling – periodic jitter when stuck >30 s
        # -----------------------------------------------------------------
        if now - last_improve > 30.0:
            # jitter the *current* iterate around the best solution found so far
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cur_vec = global_best_vec * noise
            cur_vec = np.clip(cur_vec, 0.0, None)
            cur_vec /= cur_vec.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {global_best_J:.6f}")

    # -------------------------------------------------------------------
    # 5. Return the best solution (clip and renormalise for safety)
    # -------------------------------------------------------------------
    best = np.clip(global_best_vec, 0.0, 1000.0)
    total = best.sum()
    if total == 0.0:
        best[:] = 1.0 / n
    else:
        best /= total

    return [float(v) for v in best]
