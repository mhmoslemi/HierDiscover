    start = time.monotonic()
    deadline = start + budget_s

    # Choose length n:
    if 'height_sequence_1' in globals():
        n0 = len(height_sequence_1)
    else:

    # perhaps if n0 > 8192 choose 8192 directly.

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    fft_len_corr = _next_pow2(conv_len + n - 1)

    best_seq = None

    def optimize_from(a_init):

        # convert to logits
        # Avoid zeros: set small epsilon
        eps = 1e-12
        z = np.log(np.maximum(a_init, eps))

        v = np.zeros_like(z)

        beta2 = 0.999

        last_improve_time = time.monotonic()
        while time.monotonic() < deadline:

            max_z = np.max(z)

    if s == 0.0:

    else:

    # Convert to list of floats
    return [float(v) for v in best_seq]

"""
Gradient‑based Adam optimizer on the simplex (soft‑max parameterisation).

* Variables are stored as logits `z`; the probabilities are `a = softmax(z)`,
  guaranteeing non‑negativity and ∑a = 1.
* Objective J(a) = 2·n·max(conv(a,a)).
  A smooth approximation of the max is used:
      smooth(β) = (1/β)·log∑_k exp(β·conv_k)
  whose gradient w.r.t. `a` is
      ∇J = 2·n·(2·corr(w, a)),   w_i ∝ exp(β·conv_i).
* Adam (first‑/second‑moment) updates are applied to the logits.
* β is annealed linearly from ≈1 to ≈2000, turning the smooth max into the
  true max as the search proceeds.
* Several diverse initial points (uniform, geometric, ramps, Gaussians,
  Dirichlet draws, interpolation of any existing `height_sequence_1`) are
  tried; the best solution found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = fft_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Adam‑based optimisation of the step‑function heights.
    Returns the best non‑negative sequence found before the time budget runs out.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (power of two, limited to 8192)
    # --------------------------------------------------------------
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 2048

    n = _next_pow2(base_n)
    if n < 256:
        n = 256
    if n > 8192:
        n = 8192

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlations

    # --------------------------------------------------------------
    # 2. Helper for objective (fast, avoids evaluate_sequence overhead)
    # --------------------------------------------------------------
    def _objective(a_vec: np.ndarray) -> float:
        """Returns 2·n·max(conv(a,a)) for a already normalised (sum=1)."""
        conv = _conv_full(a_vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 3. Assemble a diverse palette of initial probability vectors
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995, 0.999):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0):

            x_new = np.linspace(0, old_len - 1, n)
            v = np.interp(x_new, x_old, old)
        else:

        else:

    lr0 = 0.12                # base learning rate for Adam
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_seq = None
    best_val = np.inf

    start = time.monotonic()
    deadline = start + float(budget_s)

    # --------------------------------------------------------------
    # 5. Main optimisation loops (one inner loop per initial candidate)
    # --------------------------------------------------------------
    for a0 in init_cands:

        a0 /= a0.sum()
        z = np.log(a0 + 1e-30)

        # Adam state
        m = np.zeros_like(z)
        v_state = np.zeros_like(z)

        # timing for stagnation handling
        last_improve = time.monotonic()
        it_local = 0

        while time.monotonic() < deadline:
            # ----- compute current probabilities -----
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)   # shift for numerical stability
            a = exp_z / exp_z.sum()

            # ----- current objective -----
            cur_J = _objective(a)

            # ----- update global best if improved -----
            if cur_J < best_val - 1e-12:
                best_val = cur_J
                best_seq = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start:.2f}s] New best J = {best_val:.6f}")

            # ----- early exit if target reached -----
            if best_val <= target_J:
                # immediate return – we already have a satisfactory solution
                break

            # ----- stagnation handling (jitter) -----
            if time.monotonic() - last_improve > 40.0:
                # add Gaussian noise to logits to escape a plateau
                z += rng.normal(0.0, 0.3, size=n)
                m.fill(0.0)
                v_state.fill(0.0)
                last_improve = time.monotonic()
                continue

            # ----- soft‑max weight (smooth approximation of max) -----
            elapsed = time.monotonic() - start
            total = deadline - start
            # β grows linearly from 1 to ~2000
            beta = 1.0 + (2000.0 - 1.0) * (elapsed / total)

            # full auto‑convolution
            conv = _conv_full(a, fft_len, conv_len)
            max_conv = conv.max()

            # soft‑max over convolution entries
            w_exp = np.exp(beta * (conv - max_conv))   # max shift → no overflow
            w_sum = w_exp.sum()
            if w_sum == 0.0:
                w_norm = np.full_like(w_exp, 1.0 / w_exp.size)
            else:
                w_norm = w_exp / w_sum

            # ----- gradient of the smoothed objective w.r.t. a -----
            # dJ/da = 2·n·∇_a (∑ w_k conv_k) = 2·n·_weighted_gradient(a, w_norm)
            grad_a = 2.0 * n * _weighted_gradient(a, w_norm, fft_len_corr, n)

            # ----- chain rule to logits (soft‑max) -----
            # da/dz = diag(a) - a aᵀ  → grad_z = a * (grad_a - ⟨grad_a, a⟩)
            scalar = np.dot(grad_a, a)
            grad_z = a * (grad_a - scalar)

            # ----- Adam update on logits -----
            it_local += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v_state = beta2 * v_state + (1 - beta2) * (grad_z ** 2)

            m_hat = m / (1 - beta1 ** it_local)
            v_hat = v_state / (1 - beta2 ** it_local)

            # learning‑rate schedule: decay ≈ 1/√t
            lr = lr0 / np.sqrt(it_local + 1e-12)
            z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # If we already met the target, break the outer loop as well
        if best_val <= target_J:
            break

    # --------------------------------------------------------------
    # 6. Return the best sequence (final safety clipping & normalisation)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s == 0.0:
        best_seq[:] = 1.0 / n
    else:
        best_seq /= s

    return [float(v) for v in best_seq]
