"""Local simulated‑annealing for the step‑function problem.

- Starts from the best known sequence (``height_sequence_1``) if available,
  otherwise from a uniform distribution.
- Maintains the full auto‑convolution ``b = a * a`` and updates it
  incrementally when a small amount of mass is transferred from one
  index to another.  This allows millions of cheap proposals per second.
- Proposes moves that shift probability mass from “high‑gradient’’ indices
  (those that contribute most to the current peak of the convolution) to
  low‑gradient ones, mixed with occasional random transfers.
- Accepts moves that improve the objective ``J = 2·n·max(b)`` and,
  otherwise, accepts them with a Metropolis probability that cools
  linearly to zero.
- When no improvement is seen for a while a multiplicative Gaussian
  jitter is applied to the current best vector to escape plateaus.
- The best vector found before the time budget expires is returned.
"""

import time
import numpy as np

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size and initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total < 1e-12:
            n = 2048
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total
            n = a.shape[0]
    else:
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Initial convolution and objective
    # ------------------------------------------------------------------
    b = np.convolve(a, a)                      # full convolution, length 2*n-1
    cur_max = b.max()
    cur_J = 2.0 * n * cur_max

    best_a = a.copy()
    best_J = cur_J

    # ------------------------------------------------------------------
    # 3. Annealing schedule and bookkeeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # tiny safety margin
    total_budget = deadline - start_time

    # initial temperature = 5 % of the current objective; final ≈ 0
    T_start = cur_J * 0.05
    T_end = 1e-9

    last_improve = start_time
    last_jitter = start_time
    iteration = 0

    # Hyper‑parameters (tuned empirically)
    rel_tol_initial = 0.01            # active‑set tolerance at the start
    rel_tol_final = 0.001             # tolerance when we are close to convergence
    recompute_every = 3000            # full convolution recomputed to avoid drift
    jitter_interval = 30.0            # seconds without improvement → jitter
    donor_frac_small = 0.05           # up‑weight for gradient‑guided moves
    donor_frac_large = 0.30           # up‑weight for random moves
    min_delta = 1e-8                  # avoid zero‑moves

    print(f"[0.00s] start n={n}, J={cur_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # periodic full recompute (numerical drift)
        if iteration % recompute_every == 0:
            b = np.convolve(a, a)
            cur_max = b.max()
            cur_J = 2.0 * n * cur_max

        # --------------------------------------------------------------
        # 4.1. Stagnation handling – jitter
        # --------------------------------------------------------------
        if now - last_improve > jitter_interval:
            # multiplicative log‑normal jitter around the best solution
            noise = np.exp(rng.normal(0.0, 0.2, size=n))
            a = best_a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            b = np.convolve(a, a)
            cur_max = b.max()
            cur_J = 2.0 * n * cur_max
            last_improve = now
            last_jitter = now
            print(f"[{elapsed:6.2f}s] jitter restart – J={cur_J:.6f}")

        # --------------------------------------------------------------
        # 4.2. Choose a move (mass transfer)
        # --------------------------------------------------------------
        use_gradient = rng.random() < 0.7

        if use_gradient:
            # ---- active set (indices where convolution is close to its peak) ----
            rel_tol = rel_tol_initial if iteration < 5000 else rel_tol_final
            tol = max(cur_max * rel_tol, 1e-15)
            active = np.where(b >= cur_max - tol)[0]

            # ---- sub‑gradient of the peak (vector of contributions) ----
            grad = np.zeros(n, dtype=np.float64)
            for p in active:
                # range of i such that 0 ≤ i ≤ n‑1 and 0 ≤ p‑i ≤ n‑1
                i_start = max(0, p - (n - 1))
                i_end = min(p, n - 1)
                # a[p‑i] runs over a slice of a in reverse order
                # slice of a that participates:
                a_slice = a[p - i_end : p - i_start + 1]   # increasing i gives decreasing indices
                grad[i_start : i_end + 1] += 2.0 * a_slice[::-1]

            # ---- pick donor (large gradient) and receiver (small gradient) ----
            donor_weights = grad + 1e-12
            donor = rng.choice(n, p=donor_weights / donor_weights.sum())
            inv_weights = grad.max() - grad + 1e-12
            receiver = rng.choice(n, p=inv_weights / inv_weights.sum())
            max_move = min(a[donor], donor_frac_small * a[donor])
        else:
            # ---- purely random move ----
            # donor proportional to current mass (more mass → more to give)
            donor = rng.choice(n, p=a / a.sum())
            # receiver chosen uniformly among the remaining indices
            probs = np.ones(n, dtype=np.float64)
            probs[donor] = 0.0
            probs /= probs.sum()
            receiver = rng.choice(n, p=probs)
            max_move = min(a[donor], donor_frac_large * a[donor])

        # --------------------------------------------------------------
        # 4.3. Sample the transferred amount
        # --------------------------------------------------------------
        max_move = max(max_move, min_delta)          # ensure a tiny move is possible
        delta = rng.uniform(0.0, max_move)
        if delta <= 0.0 or donor == receiver:
            iteration += 1
            continue

        # --------------------------------------------------------------
        # 4.4. Incremental convolution update
        # --------------------------------------------------------------
        # copy current convolution (will be overwritten only if we accept)
        b_new = b.copy()

        # contribution of 2 * a * d (where d = delta⋅(e_receiver - e_donor))
        # subtract 2·delta·a shifted by donor
        s_i = donor
        e_i = donor + n
        b_new[s_i:e_i] -= 2.0 * delta * a

        # add 2·delta·a shifted by receiver
        s_j = receiver
        e_j = receiver + n
        b_new[s_j:e_j] += 2.0 * delta * a

        # add the tiny quadratic term d*d
        d2 = delta * delta
        idx_2i = 2 * donor
        if idx_2i < b_new.size:
            b_new[idx_2i] += d2
        idx_2j = 2 * receiver
        if idx_2j < b_new.size:
            b_new[idx_2j] += d2
        idx_ij = donor + receiver
        if idx_ij < b_new.size:
            b_new[idx_ij] -= 2.0 * d2

        # --------------------------------------------------------------
        # 4.5. Compute new objective
        # --------------------------------------------------------------
        new_max = b_new.max()
        new_J = 2.0 * n * new_max

        # --------------------------------------------------------------
        # 4.6. Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            # linear temperature schedule
            T = T_start * (1.0 - elapsed / total_budget) + T_end * (elapsed / total_budget)
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # ---- update the state ----
            a[donor] -= delta
            a[receiver] += delta
            # ensure non‑negativity (tiny rounding errors)
            a[donor] = max(a[donor], 0.0)
            a[receiver] = max(a[receiver], 0.0)
            # re‑normalise to avoid accumulation of floating‑point drift
            total = a.sum()
            if total > 0.0:
                a /= total
            else:
                a[:] = 1.0 / n

            b = b_new
            cur_max = new_max
            cur_J = new_J

            # ---- record a new global best ----
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (clipped and normalised)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    # final report (optional)
    # print(f"final best J = {best_J:.6f}")

    return [float(v) for v in best_a]

            # ---- donor from large gradient, receiver from small gradient ----
            donor_weights = grad + 1e-12
            donor = rng.choice(n, p=donor_weights / donor_weights.sum())
            inv_weights = grad.max() - grad + 1e-12
            receiver = rng.choice(n, p=inv_weights / inv_weights.sum())
            max_move = min(a[donor], donor_frac_small * a[donor])
        else:
            # ---- purely random move ----
            donor = rng.choice(n, p=a / a.sum())
            probs = np.ones(n, dtype=np.float64)
            probs[donor] = 0.0
            probs /= probs.sum()
            receiver = rng.choice(n, p=probs)
            max_move = min(a[donor], donor_frac_large * a[donor])

        # ---- amount to transfer -------------------------------------------------
        max_move = max(max_move, min_delta)
        delta = rng.uniform(0.0, max_move)
        if delta <= 0.0 or donor == receiver:
            iteration += 1
            continue

        # --------------------------------------------------------------
        # 4.4. Incremental update of the convolution
        # --------------------------------------------------------------
        b_new = b.copy()

        # 2·a·d term (subtract shift by donor, add shift by receiver)
        s_i = donor
        e_i = donor + n
        b_new[s_i:e_i] -= 2.0 * delta * a

        s_j = receiver
        e_j = receiver + n
        b_new[s_j:e_j] += 2.0 * delta * a

        # d*d term (quadratic, very small)
        d2 = delta * delta
        idx_2i = 2 * donor
        if idx_2i < b_new.size:
            b_new[idx_2i] += d2
        idx_2j = 2 * receiver
        if idx_2j < b_new.size:
            b_new[idx_2j] += d2
        idx_ij = donor + receiver
        if idx_ij < b_new.size:
            b_new[idx_ij] -= 2.0 * d2

        # --------------------------------------------------------------
        # 4.5. New objective
        # --------------------------------------------------------------
        new_max = b_new.max()
        new_J = 2.0 * n * new_max

        # --------------------------------------------------------------
        # 4.6. Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            T = T_start * (1.0 - elapsed / total_budget) + T_end * (elapsed / total_budget)
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # ---- update the probability vector ----
            a[donor] -= delta
            a[receiver] += delta
            a[donor] = max(a[donor], 0.0)
            a[receiver] = max(a[receiver], 0.0)

            # renormalise to avoid drift
            total = a.sum()
            if total > 0.0:
                a /= total
            else:
                a[:] = 1.0 / n

            # ---- accept the new convolution ----
            b = b_new
            cur_max = new_max
            cur_J = new_J

            # ---- record a new global best ----
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (clip and normalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    # optional final log
    # print(f"final best J = {best_J:.6f}")

    return [float(v) for v in best_a]
