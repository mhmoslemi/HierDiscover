        try: n = len(globals()['height_sequence_1'])

        n = 2048

    init_cands = []

    init_cands.append(np.full(n, 1.0/n))

    for r in (0.90, 0.95, 0.975,0.985, 0.99):

        """
Alternatively use normalized geometric series.
        """
        v = v / v.sum()
        init_cands.append(v)

    v = np.arange(n, 0, -1)
    v = v / v.sum()
    init_cands.append(v.astype(float))
    v = np.arange(1, n+1)
    v = v / v.sum()
    init_cands.append(v.astype(float))

    # Gaussian
    x = np.arange(n)
    centre = (n-1)/2.0
    for sigma in (n/10.0, n/8.0, n/6.0):

        v = v / v.sum()
        init_cands.append(v)

"""
Hybrid simulated‑annealing optimiser with richer move set.

* Works on the probability simplex (∑a = 1, a_i ≥ 0).  The objective
  J(a) = 2·n·max(conv(a,a)) is invariant under scaling, therefore we
  optimise directly on the simplex.
* Initial pool contains a variety of analytically‑motivated shapes
  (uniform, geometric, linear, Gaussian) and several Dirichlet draws
  with different concentrations.  The best of these seeds the optimiser.
* Moves (chosen each iteration):
    – Gradient step using the sub‑gradient of max(conv) together with
      exponential‑moving‑average momentum.
    – Exchange move: transfer a random fraction (up to 50 %) of mass from
      one index to another.
    – Swap move: exchange two coordinates (pure permutation).
    – Random‑mix move: blend the current vector with a fresh Dirichlet
      sample (up to 10 % new mass).
* Temperature follows an exponential cooling schedule; a Metropolis
  acceptance rule lets the walk escape shallow plateaus.
* If no improvement occurs for 20 s, a targeted mass‑rebalancing move
  tries to reduce the currently active convolution peak, otherwise a
  small jitter is applied.
* The whole search respects the 1000 s budget.  The best vector found is
  finally clipped to the allowed range [0, 1000] and re‑normalised.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # if all entries are zero (or equal) return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Fast evaluation of the objective (without the denominator).
    The returned value is J = 2·n·max(conv(a,a)).
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgradient_active_set(a: np.ndarray,
                           conv: np.ndarray,
                           active_tol: float) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For each index p where conv[p] is within `active_tol` of the maximum,
    the gradient receives a shifted copy of a.
    """
    n = a.size
    max_conv = conv.max()
    active = np.where(conv >= max_conv - active_tol)[0]

    grad = np.zeros_like(a)
    for p in active:
        # contribution of shift p
        start = max(0, p - (n - 1))
        end = min(p, n - 1)
        idx = np.arange(start, end + 1)
        # a[p-idx] is the shifted part
        grad[idx] += 2.0 * a[p - idx]
    return grad


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def exchange_move(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction (up to 50 %) from i to j."""
    n = vec.size
    i, j = rng.choice(n, size=2, replace=False)
    if vec[i] <= 0.0:
        return vec.copy()
    delta = rng.random() * vec[i] * 0.5   # up to 50 % of the source mass
    new_vec = vec.copy()
    new_vec[i] -= delta
    new_vec[j] += delta
    return new_vec


def swap_move(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Pure permutation – exchange two coordinates."""
    n = vec.size
    i, j = rng.choice(n, size=2, replace=False)
    new_vec = vec.copy()
    new_vec[i], new_vec[j] = new_vec[j], new_vec[i]
    return new_vec


def jitter_move(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Blend the current vector with a fresh Dirichlet sample.
    Up to 10 % of the mass is replaced by a random direction.
    """
    n = vec.size
    mix_strength = rng.random() * 0.1  # ≤10 %
    rand_vec = rng.dirichlet(np.ones(n))
    new_vec = (1.0 - mix_strength) * vec + mix_strength * rand_vec
    return new_vec


def targeted_move(vec: np.ndarray,
                  conv: np.ndarray,
                  rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the current maximal convolution peak.
    Identify the index k* with max(conv) and move a chunk of mass
    away from the coordinate that contributes most to that peak.
    """
    n = vec.size
    max_k = int(np.argmax(conv))
    # contribution of each i to conv[max_k] (zero if pair out of range)
    contrib = np.zeros(n)
    for i in range(n):
        j = max_k - i
        if 0 <= j < n:
            contrib[i] = vec[i] * vec[j]
    i_max = int(np.argmax(contrib))
    # pick a target index where contribution is minimal (could be same as i_max)
    i_target = int(np.argmin(contrib))
    if i_max == i_target or vec[i_max] <= 1e-12:
        return vec.copy()   # nothing to do
    delta = 0.5 * vec[i_max]  # move half of the offending mass
    new_vec = vec.copy()
    new_vec[i_max] -= delta
    new_vec[i_target] += delta
    return new_vec


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The function respects the 1000 s
    time budget and returns the best sequence it discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial points (all lie on the simplex)
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.985, 0.99):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – various concentrations
    for alpha in (0.2, 0.5, 1.0):
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # previous best (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # random permutations of a few base shapes
    for base in init_cands[:5]:
        perm = np.arange(n)
        rng.shuffle(perm)
        init_cands.append(base[perm])

    # ------------------------------------------------------------------
    # 3. Select the best among the cheap candidates
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (SA + rich move set)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # temperature schedule – start fairly hot to allow uphill moves
    T_start = best_J * 5.0
    T_end = 1e-8
    total_time = deadline - start_time

    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    momentum = np.zeros_like(cur_vec)      # exponential moving average of sub‑gradient
    beta = 0.9                              # momentum decay
    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # exponential cooling
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # -----------------------------------------------------------------
        # 4.1 Stagnation handling (every ~20 s without improvement)
        # -----------------------------------------------------------------
        if now - last_improve > 20.0:
            # try a targeted reduction of the active peak
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            prop = targeted_move(cur_vec, cur_conv, rng)
            prop = proj_simplex(prop)
            prop_J = compute_J(prop, n, fft_len, conv_len)
            if prop_J < cur_J:
                cur_vec, cur_J = prop, prop_J
                if cur_J < best_J - 1e-12:
                    best_J, best_vec = cur_J, cur_vec.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] Targeted move improved J = {best_J:.6f}")
            else:
                # fallback: small jitter around the current best
                noise = np.exp(rng.normal(0.0, 0.08, size=n))
                prop = best_vec * noise
                prop = proj_simplex(prop)
                cur_vec = prop
                cur_J = compute_J(cur_vec, n, fft_len, conv_len)
                print(f"[{elapsed:6.2f}s] Jitter – cur J = {cur_J:.6f}")
            last_improve = now

        # -----------------------------------------------------------------
        # 4.2 Propose a candidate move
        # -----------------------------------------------------------------
        move_rand = rng.random()
        if move_rand < 0.45:                     # gradient‑type move (majority)
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            max_conv = cur_conv.max()
            tol = max(1e-12, max_conv * 1e-12)
            grad = subgradient_active_set(cur_vec, cur_conv, tol)
            momentum = beta * momentum + (1.0 - beta) * grad
            step = 0.6 / np.sqrt(iteration + 1.0)
            prop = cur_vec - step * momentum
        elif move_rand < 0.75:                   # exchange move
            prop = exchange_move(cur_vec, rng)
        elif move_rand < 0.90:                   # swap move
            prop = swap_move(cur_vec, rng)
        else:                                    # random‑mix / jitter move
            prop = jitter_move(cur_vec, rng)

        # project onto simplex (guarantees feasibility)
        prop = proj_simplex(prop)

        # -----------------------------------------------------------------
        # 4.3 Evaluate and accept/reject (Metropolis rule)
        # -----------------------------------------------------------------
        prop_J = compute_J(prop, n, fft_len, conv_len)

        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # uphill moves are accepted with probability exp(-(Δ)/T)
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec, cur_J = prop, prop_J
            # record global best
            if cur_J < best_J - 1e-12:
                best_J, best_vec = cur_J, cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
