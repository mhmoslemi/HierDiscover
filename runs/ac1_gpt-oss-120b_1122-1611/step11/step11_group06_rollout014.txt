"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.  The search is a hybrid of
  * a broad, diversified portfolio of handcrafted families (uniform,
    geometric, power‑law, window functions, random Dirichlet draws),
  * intensive simulated‑annealing on the probability simplex using
    pairwise mass transfers,
  * occasional restart from a fresh random Dirichlet when progress stalls.

All candidates are evaluated with the *exact* ``evaluate_sequence`` (uses
``np.convolve``) so the returned sequence is guaranteed to be judged by the
same objective the competition uses.  The routine runs for the supplied
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def random_dirichlet(rng: np.random.Generator, n: int, conc: float) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector with concentration ``conc``."""
    g = rng.gamma(conc, 1.0, size=n)
    if g.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return g / g.sum()


def evaluate_candidate(vec: np.ndarray) -> float:
    """
    Call the global ``evaluate_sequence`` on a candidate.
    The function returns ``np.inf`` for invalid vectors, otherwise the
    true objective value.
    """
    # ``evaluate_sequence`` expects a Python list of scalars.
    return evaluate_sequence(vec.tolist())


def perturb_pairwise(rng: np.random.Generator,
                    cur: np.ndarray,
                    step: float) -> np.ndarray:
    """
    Produce a new probability vector by moving a small amount of mass
    from one coordinate to another.
    ``step`` controls the typical magnitude of the transfer (0 < step ≤ 1).
    """
    n = cur.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)

    # amount to transfer, limited by the amount currently at i
    max_transfer = cur[i]
    # Gaussian step scaled by ``step`` and by the source amount
    delta = step * max_transfer * rng.normal()
    delta = np.clip(delta, -max_transfer, cur[j])   # keep both entries non‑negative

    new = cur.copy()
    new[i] -= delta
    new[j] += delta
    # re‑normalise – numerical noise may have introduced a tiny drift
    s = new.sum()
    if s == 0.0:
        # fallback to uniform if something went wrong
        return np.full_like(new, 1.0 / n)
    return new / s


def initial_portfolio(rng: np.random.Generator, n: int) -> List[np.ndarray]:
    """
    Generate a diverse set of starting vectors.
    """
    candidates = []

    # 1) Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) Geometric decays (both directions)
    for r in [0.90, 0.93, 0.95, 0.97, 0.99, 0.995]:
        g = r ** np.arange(n, dtype=np.float64)
        g /= g.sum()
        candidates.append(g)
        candidates.append(g[::-1])

    # 3) Power‑law shapes
    for p in [0.3, 0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1, dtype=np.float64)
        pl = 1.0 / (i ** p)
        pl /= pl.sum()
        candidates.append(pl)

    # 4) Classic windows
    windows = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),
    }
    for w in windows.values():
        if w.sum() > 0:
            candidates.append(w / w.sum())

    # 5) Random Dirichlet draws with a spread of concentrations
    for conc in [0.05, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(8):
            candidates.append(random_dirichlet(rng, n, conc))

    # 6) If a previous construction exists, use it (after normalising)
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        old = np.clip(old, 0.0, None)
        if old.sum() == 0.0:
            old = np.full_like(old, 1.0 / n)
        else:
            old = old / old.sum()
        # If length differs, pad / truncate
        if old.size != n:
            if old.size < n:
                pad = np.zeros(n - old.size, dtype=np.float64)
                old = np.concatenate([old, pad])
                old /= old.sum()
            else:
                old = old[:n]
                old /= old.sum()
        candidates.append(old)

    return candidates

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The routine respects the time
    budget (≈1000 s) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size (length of the vector).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # bound the size to a manageable range
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    # ------------------------------------------------------------------
    # 2. Helper to keep the global best.
    # ------------------------------------------------------------------
    best_vec = None          # numpy array, normalised (sum = 1)
    best_score = np.inf

    def update_best(vec: np.ndarray, score: float) -> None:
        nonlocal best_vec, best_score
        if math.isfinite(score) and score < best_score:
            best_score = score
            best_vec = vec.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Broad initial portfolio.
    # ------------------------------------------------------------------
    for cand in initial_portfolio(rng, n):
        score = evaluate_candidate(cand)
        update_best(cand, score)

    print(f"[{0.0:6.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Simulated‑annealing / pairwise‑transfer local search.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # leave a safety margin
    cur_vec = best_vec.copy() if best_vec is not None else np.full(n, 1.0 / n)
    cur_score = best_score
    iteration = 0
    last_improve = start_time

    # temperature schedule – start relatively high, cool linearly
    T_start = best_score * 0.2 if best_score < np.inf else 1.0
    T_end = 1e-8

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear cooling
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * elapsed / (deadline - start_time))

        # Adaptive step size – small transfers when we are deep in the
        # neighbourhood, larger transfers earlier.
        step = 0.5 / math.sqrt(iteration + 1.0)

        # Propose a neighbour by moving mass between two random indices.
        proposal = perturb_pairwise(rng, cur_vec, step)

        # Occasionally perform a more global perturbation (mix with a fresh Dirichlet)
        if rng.random() < 0.04:
            eps = rng.uniform(0.0, 0.3)
            rand = random_dirichlet(rng, n, conc=1.0)
            proposal = (1.0 - eps) * proposal + eps * rand
            proposal /= proposal.sum()

        # Evaluate the candidate with the *exact* objective.
        prop_score = evaluate_candidate(proposal)

        # Metropolis acceptance.
        accept = False
        if prop_score < cur_score:
            accept = True
        else:
            # uphill moves are allowed with a temperature‑dependent probability
            prob = math.exp(-(prop_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_score = prop_score
            # Record a new global best
            if cur_score < best_score - 1e-12:
                update_best(cur_vec, cur_score)
                last_improve = now

        # Stagnation handling – after 30 s without improvement we jump.
        if now - last_improve > 30.0:
            # re‑initialise from a fresh random Dirichlet (moderately concentrated)
            cur_vec = random_dirichlet(rng, n, conc=0.8)
            cur_score = evaluate_candidate(cur_vec)
            last_improve = now

        iteration += 1

    # ------------------------------------------------------------------
    # 5.  Return the best solution (clip to legal range, renormalise).
    # ------------------------------------------------------------------
    if best_vec is None:
        # fallback – uniform vector (always legal)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # Convert to ordinary Python floats for the platform.
    return [float(v) for v in best_vec]
