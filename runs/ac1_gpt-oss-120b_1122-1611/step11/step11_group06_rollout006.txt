"""
Hybrid evolutionary‑gradient optimizer for the step‑function problem.

* Stage 1 – population based search.
  - Initialise a diverse set of probability vectors (uniform, geometric,
    power‑law, classic windows, Dirichlet draws and the previous best if
    available).
  - Repeatedly evaluate the objective J = 2·n·max(conv(a,a)) using an FFT
    based convolution (O(n log n)).
  - Keep the elite individuals, generate offspring by
        • exponentiated‑gradient step on the active set of the convolution
        • multiplicative Gaussian mutation
        • random permutation (shuffle)
        • simple convex crossover
  - Occasionally replace a fraction of the population by fresh Dirichlet
    draws if no improvement has been seen for a while.

* Stage 2 – permutation annealing.
  - The multiset of values of the best vector is kept fixed; only the
    ordering is changed.
  - Simulated‑annealing on random swaps quickly reduces the peak of the
    full self‑convolution.  The temperature is linearly cooled.
  - The best ordering found is returned.

All operations stay on the probability simplex (∑a_i = 1) so the
evaluation by ``evaluate_sequence`` is scale‑invariant.
The routine respects the 1000 s budget and prints progress.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Utility functions (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum value of the full linear convolution vec * vec (FFT)."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def objective(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective J = 2·n·max(conv(vec,vec)) (assumes sum(vec)=1)."""
    return 2.0 * n * full_conv_max(vec, fft_len, conv_len)


def subgradient(vec: np.ndarray, n: int,
                fft_len: int, conv_len: int) -> np.ndarray:
    """Sub‑gradient of the max‑convolution at the active set."""
    # full convolution via FFT
    conv_vals = np.fft.irfft(
        np.fft.rfft(vec, n=fft_len) ** 2,
        n=fft_len)[:conv_len]
    max_c = conv_vals.max()
    # numerical tolerance for active indices
    tol = max(1e-12, max_c * 1e-12)
    active = np.where(conv_vals >= max_c - tol)[0]

    grad = np.zeros_like(vec)
    for k in active:
        # indices i contributing to conv[k] satisfy
        # 0 ≤ i < n , 0 ≤ k-i < n
        start = k - (n - 1)
        if start < 0:
            start = 0
        end = k if k < n else n - 1
        idx = np.arange(start, end + 1, dtype=np.int64)
        # each pair (i, k-i) contributes a_{k-i} to the derivative of a_i
        grad[idx] += 2.0 * vec[k - idx]
    return grad


def mutate_multiplicative(vec: np.ndarray, sigma: float,
                          rng: np.random.Generator) -> np.ndarray:
    """Gaussian multiplicative mutation (log‑space)."""
    noise = rng.normal(0.0, sigma, size=vec.shape)
    mut = vec * np.exp(noise)
    mut = np.maximum(mut, 0.0)
    s = mut.sum()
    if s == 0.0:
        mut[:] = 1.0 / vec.size
    else:
        mut /= s
    return mut


def exp_gradient_step(vec: np.ndarray, grad: np.ndarray,
                      step: float) -> np.ndarray:
    """Exponentiated‑gradient update followed by renormalisation."""
    factor = np.exp(-step * grad)
    upd = vec * factor
    s = upd.sum()
    if s == 0.0:
        upd[:] = 1.0 / vec.size
    else:
        upd /= s
    return upd


def permutation_anneal(base_vec: np.ndarray, n: int,
                       fft_len: int, conv_len: int,
                       budget_s: float, start_time: float,
                       rng: np.random.Generator) -> np.ndarray:
    """
    Simulated‑annealing on permutations of ``base_vec``.
    The multiset of values stays fixed; only ordering changes.
    """
    cur = base_vec.copy()
    cur_J = objective(cur, n, fft_len, conv_len)
    best = cur.copy()
    best_J = cur_J

    # temperature schedule (linear cooling)
    T_start = cur_J * 0.2
    T_end = 1e-8
    last_improve = time.monotonic()
    while True:
        now = time.monotonic()
        if now - start_time >= budget_s:
            break
        elapsed = now - start_time
        # linear cooling for the permutation phase (the remaining budget)
        T = (T_start * (1.0 - elapsed / budget_s) + T_end * (elapsed / budget_s))

        # propose a random swap
        i, j = rng.integers(0, n, size=2)
        cur[i], cur[j] = cur[j], cur[i]

        new_J = objective(cur, n, fft_len, conv_len)

        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_J = new_J
            if new_J < best_J:
                best_J = new_J
                best = cur.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] Permutation improve J = {best_J:.6f}")
        else:
            # revert the swap
            cur[i], cur[j] = cur[j], cur[i]

        # if stuck, perform a fresh random shuffle
        if now - last_improve > 20.0:
            rng.shuffle(cur)
            cur_J = objective(cur, n, fft_len, conv_len)
            last_improve = now

    return best


def proj_to_simplex(vec: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    (Used only as a safety net; most updates keep the sum = 1 already.)
    """
    if vec.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = vec.size
    u = np.sort(vec)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(vec, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(vec - theta, 0.0)
    return w

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimizes ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Determine problem size (length of the vector)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Helper for evaluating a candidate (fast, without security checks)
    # --------------------------------------------------------------
    def eval_and_normalise(vec: np.ndarray) -> (np.ndarray, float):
        """Clamp, renormalise and return J."""
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            vec = np.full(n, 1.0 / n)
        else:
            vec = vec / s
        J = objective(vec, n, fft_len, conv_len)
        return vec, J

    # --------------------------------------------------------------
    # 3. Initialise a diverse population
    # --------------------------------------------------------------
    popsize = 30
    population: List[np.ndarray] = []

    # uniform
    population.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (forward & reversed)
    for r in [0.90, 0.92, 0.94, 0.96, 0.98, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v = v / v.sum()
        population.append(v)
        population.append(v[::-1])

    # power‑law shapes
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1, dtype=np.float64)
        v = 1.0 / (i ** p)
        v = v / v.sum()
        population.append(v)

    # classic window functions
    windows = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),
    }
    for w in windows.values():
        if w.sum() > 0.0:
            population.append(w / w.sum())

    # Kaiser windows
    for beta in [0.5, 2.0, 5.0, 10.0, 15.0]:
        w = np.kaiser(n, beta)
        if w.sum() > 0.0:
            population.append(w / w.sum())

    # random Dirichlet draws with a variety of concentration parameters
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            population.append(rng.dirichlet(np.full(n, conc)))

    # previous construction, if any
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = 1.0 / n
        else:
            v /= v.sum()
        population.append(v)

    # pad / trim to the exact population size
    while len(population) < popsize:
        conc = rng.uniform(0.1, 4.0)
        population.append(rng.dirichlet(np.full(n, conc)))
    if len(population) > popsize:
        population = population[:popsize]

    # --------------------------------------------------------------
    # 4. Evolutionary‑gradient loop (main search)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    start_time = time.monotonic()
    # reserve ~20 s for the permutation phase
    search_deadline = start_time + max(0.0, float(budget_s) - 20.0)

    iteration = 0
    last_global_improve = start_time

    while time.monotonic() < search_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # simple linear cooling for the acceptance of noisy proposals
        T_start = best_J * 0.15 + 1e-3
        T_end = 1e-9
        T = (T_start * (1.0 - elapsed / (search_deadline - start_time)) +
             T_end * (elapsed / (search_deadline - start_time)))

        # ---- evaluate whole population -------------------------------------------------
        J_vals = []
        for i, vec in enumerate(population):
            vec, J = eval_and_normalise(vec)
            population[i] = vec
            J_vals.append(J)
            if J < best_J - 1e-12:
                best_J = J
                best_vec = vec.copy()
                last_global_improve = now
                print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")

        # ---- select elite ----------------------------------------------------------------
        elite_count = max(2, popsize // 5)
        elite_idx = np.argsort(J_vals)[:elite_count]
        elite = [population[i] for i in elite_idx]

        # ---- create next generation ------------------------------------------------------
        next_pop: List[np.ndarray] = elite.copy()  # keep the elites

        # gradient‑based offspring from elite
        for vec in elite:
            grad = subgradient(vec, n, fft_len, conv_len)
            step = 0.5 / np.sqrt(iteration + 1.0)    # 1/√t schedule
            proposal = exp_gradient_step(vec, grad, step)
            next_pop.append(proposal)

        # multiplicative‑mutation offspring
        sigma = 0.15 * (1.0 - elapsed / (search_deadline - start_time)) + 0.02
        for vec in elite:
            proposal = mutate_multiplicative(vec, sigma, rng)
            next_pop.append(proposal)

        # shuffled copies (pure permutation)
        for vec in elite:
            shuffled = vec.copy()
            rng.shuffle(shuffled)
            next_pop.append(shuffled)

        # simple convex crossover between random elite members
        while len(next_pop) < popsize:
            a, b = rng.choice(elite, size=2, replace=False)
            alpha = rng.random()
            child = alpha * a + (1.0 - alpha) * b
            child = np.maximum(child, 0.0)
            child_sum = child.sum()
            if child_sum == 0.0:
                child[:] = 1.0 / n
            else:
                child /= child_sum
            next_pop.append(child)

        # occasional random injection to maintain diversity
        if rng.random() < 0.07:
            conc = rng.uniform(0.1, 5.0)
            next_pop.append(rng.dirichlet(np.full(n, conc)))
            next_pop = next_pop[:popsize]

        # ---- stagnation handling --------------------------------------------------------
        if now - last_global_improve > 30.0:
            # replace half the population with fresh random draws
            half = popsize // 2
            for _ in range(half):
                conc = rng.uniform(0.1, 4.0)
                next_pop.append(rng.dirichlet(np.full(n, conc)))
            # keep only the first popsize individuals
            next_pop = next_pop[:popsize]
            last_global_improve = now

        population = next_pop[:popsize]
        iteration += 1

    # --------------------------------------------------------------
    # 5. Permutation annealing on the best multiset of values
    # --------------------------------------------------------------
    if best_vec is None:
        # fallback – should never happen
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # keep the values but allow a new ordering
    permuted_best = permutation_anneal(best_vec, n, fft_len,
                                      conv_len,
                                      budget_s,
                                      start_time,
                                      rng)

    # --------------------------------------------------------------
    # 6. Final clean‑up and return
    # --------------------------------------------------------------
    final_vec = np.clip(permuted_best, 0.0, 1000.0)
    total = final_vec.sum()
    if total == 0.0:
        final_vec[:] = 1.0 / n
    else:
        final_vec /= total

    return [float(v) for v in final_vec]
