def propose_candidate(...):
    import time, numpy as np
    from scipy.optimize import linprog

    start = time.monotonic()
    deadline = start + budget_s - 5  # safety margin

    # Determine n
    target_n = 256
    n = target_n

    # Initialize candidates list
    inits = []

    # Uniform
    inits.append(np.full(n, 1.0/n))

    # Dirichlet draws with alphas
    rng = np.random.default_rng(seed)
    for alpha in [0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            vec = rng.dirichlet(np.ones(n) * alpha)
            inits.append(vec)

    # Geometric families
    for r in [0.9, 0.93, 0.95, 0.97, 0.98, 0.99]:

        v = v / v.sum()
        inits.append(v)

    # Gaussian shapes

        except Exception as e:

    best_global_vec = None

"""
Alternating Linear‑Programming (LP) minimiser for the autocorrelation max‑norm.

The objective of evaluate_sequence (with the sum of the coefficients normalised
to 1) is  J = 2·n·max(conv(a,a)).
We iteratively reduce the value of max(conv) as follows:

1.  Choose a modest dimension n (256 – 512) so that a dense LP 
    (variables =n, constraints =2·n‑1) can be solved many times.
2.  Initialise a probability vector a (∑a = 1, a_i≥0) by down‑sampling the
    current best construction if available, otherwise by a fewhand‑crafted
    shapes (uniform, geometric, Gaussian, Dirichlet draws).
3.  Loop while time is left:
     * Build the Toeplitz matrix A_f with entries A_f[k,i]=a_i if
       0≤k‑i<n, else 0.  For any candidate b we have conv(a,b)=A_f·b.
     * Solve the LP   maximise ∑b_i   subject to   A_f·b ≤ M,
       b≥0, where M = max(conv(a,a)).  The LP pushes as much total mass
       as possible into a new direction while keeping the convolution
       under the current peak M.
     * Normalise the LP solution to a probability vector g.
     * If g alone improves the objective, replace a←g.
       Otherwise try a convex combination a←(1−λ)·a+λ·g with several
       λ∈{0.5,0.25,0.125} (projected back onto the simplex).
     * Accept the best improvement; if none is found for a while,
       jitter a by a small multiplicative Gaussian noise to escape plateaus.
4.  Keep the globally best vector seen across all random starts.
5.  Return the best probability vector (clipped to the allowed range).

The method is completely different from the previous mixed multiplicative/
additive gradient optimiser and often reaches a lower bound on the
max‑autocorrelation, thereby lowering the value returned by
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def build_toeplitz_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A such that (A @ b) == np.convolve(a, b)
    for any vector b of length n.
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


def downsample_to(arr: np.ndarray, target_len: int) -> np.ndarray:
    """
    Down‑sample a 1‑D array to the required length by blockwise summation.
    The total mass is preserved.
    """
    L = arr.size
    if L <= target_len:
        out = np.zeros(target_len, dtype=np.float64)
        out[:L] = arr
        return out
    block = L // target_len
    remainder = L % target_len
    out = np.empty(target_len, dtype=np.float64)
    start = 0
    for i in range(target_len):
        sz = block + (1 if i < remainder else 0)
        out[i] = arr[start:start + sz].sum()
        start += sz
    return out


def evaluate_J(vec: np.ndarray) -> float:
    """
    Compute the scalar J = 2·n·max(conv(vec,vec)) for a probability vector.
    """
    n = vec.size
    conv = np.convolve(vec, vec)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size (small enough for fast LP solves)
    # ------------------------------------------------------------------
    # If a previously found sequence exists, down‑sample it, otherwise use a
    # modest default size.
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(height_sequence_1, dtype=np.float64)
            n_target = min(512, max(128, old.size // 5))   # a balanced size
            f0 = downsample_to(old, n_target)
        except Exception:
            n_target = 256
            f0 = np.full(n_target, 1.0 / n_target)
    else:
        n_target = 256
        f0 = np.full(n_target, 1.0 / n_target)

    n = f0.size

    # ------------------------------------------------------------------
    # 2. Assemble a diverse set of initial probability vectors
    # ------------------------------------------------------------------
    init_cands = []

    # Uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Down‑sampled previous best (if any)
    if "height_sequence_1" in globals():
        init_cands.append(f0 / max(f0.sum(), 1e-12))

    # Geometric decays
    for r in [0.90, 0.93, 0.95, 0.97, 0.985, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws with a spread of concentration parameters
    for alpha in [0.05, 0.1, 0.3, 0.6, 1.0]:
        for _ in range(2):
            v = rng.dirichlet(np.full(n, alpha))
            init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop over the different starts
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    for start_idx, cur_vec in enumerate(init_cands):
        if time.monotonic() >= deadline:
            break

        # Normalise and safeguard
        cur_vec = np.maximum(cur_vec, 0.0)
        total = cur_vec.sum()
        if total == 0.0:
            cur_vec[:] = 1.0 / n
        else:
            cur_vec /= total

        cur_J = evaluate_J(cur_vec)
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cur_vec.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] "
                  f"Start {start_idx} initial J = {cur_J:.6f}")

        # ------------------------------------------------------------------
        # 4. Alternating LP optimisation for this start
        # ------------------------------------------------------------------
        stagnation_counter = 0
        iteration = 0
        last_improve = time.monotonic()

        while time.monotonic() < deadline:
            # Re‑build the Toeplitz constraint matrix for the current vector
            A = build_toeplitz_matrix(cur_vec)
            conv_cur = np.convolve(cur_vec, cur_vec)
            M = conv_cur.max()

            # Linear programme: maximise sum(b) subject to A·b ≤ M, b ≥ 0
            c = -np.ones(n)                     # linprog minimises; -1 → maximise
            b_ub = np.full(A.shape[0], M, dtype=np.float64)
            bounds = [(0.0, None)] * n

            # Solve LP – the “highs” method is fast for dense problems of this size
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'disp': False})

            if not res.success:
                # In the rare case the LP fails, abort this start.
                break

            b_opt = res.x
            sum_b = b_opt.sum()
            if sum_b <= 0.0:
                break

            g = b_opt / sum_b                     # Normalised candidate

            # Evaluate candidate alone
            J_g = evaluate_J(g)

            improved = False

            # 4a – direct replacement if it already improves the objective
            if J_g + 1e-12 < cur_J:
                cur_vec = g
                cur_J = J_g
                improved = True
                stagnation_counter = 0
                if cur_J < best_J - 1e-12:
                    best_J = cur_J
                    best_vec = cur_vec.copy()
                    print(f"[{time.monotonic() - start_time:6.2f}s] "
                          f"Start {start_idx} iter {iteration} new best J = {best_J:.6f}")

            else:
                # 4b – try convex combinations with decreasing mixing weights
                for lam in (0.5, 0.25, 0.125, 0.0625):
                    cand = (1.0 - lam) * cur_vec + lam * g
                    cand = proj_simplex(cand)      # enforce sum=1, non‑negativity
                    J_cand = evaluate_J(cand)
                    if J_cand + 1e-12 < cur_J:
                        cur_vec = cand
                        cur_J = J_cand
                        improved = True
                        stagnation_counter = 0
                        if cur_J < best_J - 1e-12:
                            best_J = cur_J
                            best_vec = cur_vec.copy()
                            print(f"[{time.monotonic() - start_time:6.2f}s] "
                                  f"Start {start_idx} iter {iteration} mix λ={lam:.3f} "
                                  f"new best J = {best_J:.6f}")
                        break  # accept first improving λ

            # ------------------------------------------------------------------
            # 5. Stagnation handling
            # ------------------------------------------------------------------
            if not improved:
                stagnation_counter += 1
                if stagnation_counter >= 20:
                    # Small multiplicative jitter to escape flat regions
                    noise = np.exp(rng.normal(0.0, 0.06, size=n))
                    cur_vec = best_vec * noise
                    cur_vec = proj_simplex(cur_vec)
                    cur_J = evaluate_J(cur_vec)
                    stagnation_counter = 0
                    print(f"[{time.monotonic() - start_time:6.2f}s] "
                          f"Start {start_idx} iter {iteration} jitter – J = {cur_J:.6f}")

            iteration += 1
            # Safeguard against infinite loops
            if iteration >= 500:
                break

    # ------------------------------------------------------------------
    # 6. Return the globally best sequence found
    # ------------------------------------------------------------------
    if best_vec is None:
        # Fallback – uniform distribution (should never happen)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s

    # Convert to plain Python list as required by the evaluator
    return [float(v) for v in best_vec]
