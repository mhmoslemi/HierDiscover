    * With probability ``p_grad`` a projected additive step
      ``a ← proj_simplex(a − η·g)`` is taken, otherwise a directed random
      mass‑transfer move that specifically reduces the contribution to the
      maximal lag is performed.
    * A simple exponential cooling schedule provides a Metropolis
      acceptance rule, allowing occasional uphill moves.
    * If no improvement has been seen for 30 s a small multiplicative
      jitter (log‑normal noise) is applied to the best vector to escape
      stagnation.

3.  The whole search runs for the supplied time budget (default 1000 s);
    the best vector found is returned as a plain Python list of floats.

All helper functions are top‑level, use only NumPy, and avoid closures.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to one).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, fft_len: int, conv_len: int):
    """
    Compute the full linear convolution a*a and return both its maximum
    value and the index where the maximum occurs.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return max_val, max_idx


def _subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at index ``k``.
    For each i:  g_i = a_{k‑i} (if in range) + a_{i‑k} (if in range).
    """
    n = a.size
    g = np.zeros_like(a)

    if k < n:
        # part where k-i is valid  (i = 0 … k)
        g[: k + 1] = a[k::-1]                     # a_{k-i}
        # part where i‑k is valid (i = k+1 … n‑1)
        if k + 1 < n:
            g[k + 1 :] = a[1 : n - k]             # a_{i‑k}
    else:
        # k ≥ n : only the first term can be non‑zero
        d = k - (n - 1)                           # smallest i that satisfies 0 ≤ k‑i < n
        # i runs from d … n‑1  →  k‑i runs from n‑1 … d
        g[d:] = a[n - 1 : d - 1 : -1]             # reversed slice
    return g


def _directed_move(a: np.ndarray, k: int, rng: np.random.Generator,
                   eps_factor: float) -> np.ndarray:
    """
    Construct a proposal that explicitly reduces the contribution to the
    currently maximal convolution entry ``k``.
    A source index i is sampled proportionally to a_i * a_{k-i},
    then a small amount of mass is transferred to a target index j that
    does *not* belong to the support of the maximal lag (if possible).
    """
    n = a.size
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)                     # all i with a_i a_{k-i} defined
    contrib = a[idx] * a[k - idx]                  # contribution of each i
    total = contrib.sum()
    if total > 0:
        probs = contrib / total
        i = rng.choice(idx, p=probs)
    else:
        i = rng.integers(0, n)

    # pick a target j outside the support of the maximal lag if we can
    mask = np.ones(n, dtype=bool)
    mask[idx] = False
    cand = np.where(mask)[0]
    if cand.size > 0:
        j = rng.choice(cand)
    else:
        # support covers everything (happens for k = n‑1); pick a low‑weight index
        j = int(np.argmin(a))

    # transfer a fraction of the source mass
    delta = eps_factor * a[i] * rng.random()
    if delta > a[i]:
        delta = a[i] * 0.5
    proposal = a.copy()
    proposal[i] -= delta
    proposal[j] += delta
    # small numerical drift may break the simplex – re‑project
    return proj_simplex(proposal)


# ----------------------------------------------------------------------
# Main optimisation routine – entry point for the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses a projected sub‑gradient / simulated‑annealing hybrid as described
    above.  Returns the best vector found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # reasonable default

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a diversified set of initial candidates (probability vectors)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderate to strong concentration)
    for alpha in [0.3, 0.6, 1.0, 0.2, 0.5]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Sparse masks – a few equally weighted spikes
    for k in [int(np.sqrt(n)), int(np.sqrt(n) * 2), max(2, n // 10)]:
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        init_cands.append(mask)

    # Random binary mask (Bernoulli 0.5)
    bin_mask = rng.binomial(1, 0.5, size=n).astype(np.float64)
    if bin_mask.sum() > 0:
        bin_mask /= bin_mask.sum()
        init_cands.append(bin_mask)

    # previous construction (if present)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v = v / s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Evaluate all initial candidates (exact objective) and pick the best
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        max_val, _ = _conv_max(cand, fft_len, conv_len)
        J = 2.0 * n * max_val
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop – sub‑gradient + SA + directed moves
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # leave a safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time

    # annealing schedule
    T_start = best_J * 0.2          # start temperature (20 % of current best)
    T_end = 1e-8

    # step‑size schedules
    step_start = 0.5                # additive step at the beginning
    step_end = 0.001                # near‑zero at the end
    eps_start = 0.5                 # fraction of source mass moved in directed move
    eps_end = 0.01

    # probability of using a pure gradient step
    p_grad = 0.65

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # progress ratio in [0,1]
        ratio = elapsed / (deadline - start_time)

        # ----------------------------------------------------------
        # recompute exact max‑convolution for the current vector
        # ----------------------------------------------------------
        max_val, max_idx = _conv_max(cur_vec, fft_len, conv_len)
        cur_J = 2.0 * n * max_val

        # keep track of the global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # stagnation handling – jitter after 30 s without improvement
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # generate a proposal
        # ----------------------------------------------------------
        if rng.random() < p_grad:
            # --- pure additive (sub‑gradient) move ---------------------------------
            grad = _subgradient(cur_vec, max_idx)
            step = step_start * (1.0 - ratio) + step_end * ratio
            proposal = cur_vec - step * grad
            proposal = proj_simplex(proposal)
        else:
            # --- directed random mass transfer --------------------------------------
            eps = eps_start * (1.0 - ratio) + eps_end * ratio
            proposal = _directed_move(cur_vec, max_idx, rng, eps)

        # ----------------------------------------------------------
        # evaluate proposal (exact objective)
        # ----------------------------------------------------------
        prop_max, _ = _conv_max(proposal, fft_len, conv_len)
        prop_J = 2.0 * n * prop_max

        # ----------------------------------------------------------
        # Metropolis acceptance (exponential cooling)
        # ----------------------------------------------------------
        T = T_start * (1.0 - ratio) + T_end * ratio
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best solution (clip and renormalise for safety)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
```
