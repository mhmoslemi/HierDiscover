"""
Soft‑max Adam with entropy‑regularised (soft‑max) smoothing of the
max‑convolution objective.

* a is kept on the probability simplex via a = softmax(z).
* The true objective J(a)=2·n·max(conv(a,a)) is nonsmooth.
  We replace max(·) by a smoothed version
      Sα(a)= (1/α)·log Σₖ exp(α·(a*a)ₖ)
  whose gradient is a weighted sum of all convolution entries.
  By slowly increasing α we start with a gentle landscape and
  end with a close approximation of the true max.
* Gradients are obtained by one extra convolution:
      ∂Sα/∂a = 2·( w * a_rev )[n‑1 : n‑1+n] ,
  where wₖ ∝ exp(α·(a*a)ₖ) and a_rev is a reversed.
* Adam updates are performed on the logits z.
* A simple jitter/restart (Gaussian perturbation of z) is triggered
  when no improvement has been seen for ~20 s.
* The routine works for any reasonable n (defaults to a power of two
  ≤8192).  It returns the best simplex‑normalised sequence discovered
  before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    Returns the array of length len(x)+len(y)-1.
    """
    conv_len = len(x) + len(y) - 1
    fft_len = _next_pow2(conv_len)
    fx = np.fft.rfft(x, fft_len)
    fy = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(fx * fy, fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The optimiser uses a smoothed max‑convolution
    objective with Adam on the soft‑max parametrisation and occasional
    jitter restarts.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size – keep it a power of two for FFT speed.
    # --------------------------------------------------------------
    default_n = 4096
    if "height_sequence_1" in globals():
        try:
            n_guess = len(height_sequence_1)
            n = max(1024, min(n_guess, 8192))
        except Exception:
            n = default_n
    else:
        n = default_n

    # force a power of two (helps FFT a little)
    if n & (n - 1):
        n = 1 << ((n - 1).bit_length())

    # --------------------------------------------------------------
    # 2. Build a diverse portfolio of starting probability vectors.
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # inverse‑linear (high at the two ends)
    i = np.arange(n, dtype=np.float64)
    v = 1.0 / (i + 1.0) + 1.0 / (n - i)
    v /= v.sum()
    init_cands.append(v)

    # inverse‑quadratic (sharper ends)
    v = 1.0 / ((i + 1.0) ** 2) + 1.0 / ((n - i) ** 2)
    v /= v.sum()
    init_cands.append(v)

    # linear ramps (increasing / decreasing)
    v = (i + 1) / (n * (n + 1) / 2.0)
    init_cands.append(v / v.sum())
    v = (n - i) / (n * (n + 1) / 2.0)
    init_cands.append(v / v.sum())

    # previous construction, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.zeros(n - v.size)])
        v = np.clip(v, 0.0, None)
        tot = v.sum()
        if tot == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= tot
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best of the candidates (by the raw J = 2·n·max(conv)).
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_fft(cand, cand)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop – Adam on logits + smoothed‑max gradient.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin
    total_allowed = deadline - start_time

    # Soft‑max parametrisation (logits → probability simplex)
    eps = 1e-12
    logits = np.log(np.maximum(best_a, eps))

    # Adam state
    m = np.zeros_like(logits)
    v_adam = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.5                     # base learning rate

    # Smoothing schedule for the soft‑max of the convolution
    alpha_start = 1.0
    alpha_end = 5000.0

    iteration = 0
    last_improve = start_time

    # Global best seen so far
    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # ---- 4.1 Build current a from logits (soft‑max) ----
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)   # shift for numerical stability
        a = exp_logits / exp_logits.sum()       # on the simplex, sum(a)=1

        # ---- 4.2 Full convolution a*a and current objective ----
        conv = _conv_fft(a, a)                  # length 2n‑1
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ---- 4.3 Update global best if improved ----
        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ---- 4.4 Smoothed max‑convolution gradient (entropy‑regularised) ----
        # Linear schedule for α
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total_allowed)

        # Soft‑max weights over convolution entries (numerically stable)
        w_raw = np.exp(alpha * (conv - max_conv))   # max entry gets weight 1
        w = w_raw / w_raw.sum()                     # Σ w = 1

        a_rev = a[::-1]

        # Transform gradient to logits space (soft‑max chain rule)
        avg = np.dot(a, grad_a)                     # scalar = <a,grad_a>
        grad_z = a * (grad_a - avg)                 # ∂J/∂z

        # ---- 4.5 Adam step on logits ----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)               # simple 1/√t decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_adam / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # Keep logits in a numerically reasonable range
        logits = np.clip(logits, -500.0, 500.0)

        # ---- 4.6 Detect stagnation – jitter restart ----
        if now - last_improve > 20.0:
            # jitter around the current global best
            jitter_sigma = 0.5
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v_adam.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

    # --------------------------------------------------------------
    # 5. Return the best sequence (clipped to the allowed range and re‑normalised).
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
