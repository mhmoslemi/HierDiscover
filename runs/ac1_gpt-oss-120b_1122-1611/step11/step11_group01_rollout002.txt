def lp_step(f, M, active_idx):
    # variables g >=0 length n
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_idx:
        # compute conv_f_g at index k linear expression: sum_i f[i] * g[k-i]
        # Equivalent to sum_i f[i] * g_j where j = k-i
        # compute dot product of f and g shifted
        # easier: Use vectorized: f reversed at appropriate indices.
        # We'll compute row = np.zeros(n)
        # row[i] = f[i] if 0 <= k-i < n else 0 => g[k-i] selection.
        # We can express as sum_i f[i] * g[k-i] = sum_j f[k-j] * g[j] where j in bounds.
        # So row[j] = f[k-j] if 0 <= k-j < n else 0.
        row = np.zeros(n)
        low = max(0, k-(n-1))
        high = min(k, n-1)
        # For j in [low, high], set row[j] = f[k-j]
        # Use slice
        # but iterating small set
        for j in range(low, high+1):
            row[j] = f[k - j]
        constraints.append(row @ g <= M)
    # maximize sum(g)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return f  # fallback, no solution
    new_g = np.array(g.value).flatten()
    # renormalize sum to 1.
    total = new_g.sum()
    if total > 0:
        new_g = new_g / total
    else:
        new_g = np.full_like(new_g, 1.0/n)
    return new_g
```

"""
Hybrid optimisation for the max‑convolution objective.

* Initialise a diverse set of probability vectors (uniform, Dirichlet,
  Gaussian bumps, and the previously found construction if available)
  on the simplex ∑a_i = 1, a_i ≥ 0.
* Primary optimiser: soft‑max smoothing of the max‑convolution, gradient
  computed in the Fourier domain and an exponentiated‑gradient update.
  A linear‑cooling Metropolis temperature lets the walk escape plateaus.
* Stagnation handling: every ~20 s of no improvement a log‑normal jitter
  perturbs the current best solution.
* Periodic refinement: every few hundred iterations we solve a small linear
  programme (via SciPy’s linprog) that maximises the total mass of a new
  candidate g under the active convolution constraints
  (conv(f,g) ≤ current max).  The new candidate is blended into the walk.
* The loop runs until the supplied time budget (≈ 1000 s) expires, after
  which the best vector (clipped, renormalised) is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(seq: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Return 2·n·max(conv(seq,seq)) (the evaluation objective when sum(seq)=1)."""
    A = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * seq.size * conv.max())


def lp_refine(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Solve a small linear programme:
        maximise   sum(g)
        s.t.       conv(f,g)[k] ≤ M  for k in active_idx
                   g ≥ 0
    Returns the optimiser g (renormalised to sum 1); if the LP fails,
    the input f is returned unchanged.
    """
    n = f.size
    if active_idx.size == 0:
        return f.copy()

    # Build constraint matrix A_ub * g ≤ b_ub, one row per active index.
    rows = []
    for k in active_idx:
        # row_j = f[k-j] if 0 ≤ k-j < n else 0
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        row = np.zeros(n, dtype=np.float64)
        # fill the slice efficiently
        j_vals = np.arange(low, high + 1, dtype=np.int64)
        row[j_vals] = f[k - j_vals]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)

    # linprog minimises, so maximise sum(g) -> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if not res.success or res.x is None:
        return f.copy()

    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s > 0.0:
        g /= s
    else:
        g = np.full_like(g, 1.0 / n)
    return g

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Hybrid soft‑max gradient descent + occasional LP refinement.
    Returns a non‑negative list of heights (sum≈1) that minimises
    evaluate_sequence within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 4096))
        except Exception:
            n = 2048
    else:
        n = 2048                     # a comfortable size for many iterations

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a diverse pool of initial vectors (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws – from spiky to smooth
    for alpha in [0.05, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Gaussian bumps of several widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 5.0, n / 3.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = np.full(n, 1.0 / n)
        # resize / truncate to the current n if needed
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            pad = np.zeros(n - v.size, dtype=np.float64)
            v = np.concatenate([v, pad])
            v /= v.sum()
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (objective = 2·n·max(conv))
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin

    # Metropolis temperature schedule (linear cooling)
    T_start = best_J * 0.08          # start at a modest fraction of current best
    T_end = 1e-9

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # soft‑max temperature (β) – gradual increase
    beta = 1.0
    beta_max = 600.0
    # Approximate number of iterations we expect: (deadline-start_time)/0.01 ≈ 100k
    beta_inc = (beta_max - beta) / max(1, int(0.6 * (deadline - start_time) / 0.01))

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear cooling of Metropolis temperature
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))

        # ------------------------------------------------------
        # 4.1  Compute convolution and max (FFT – single forward transform)
        # ------------------------------------------------------
        A = np.fft.rfft(cur_vec, fft_len)
        conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_conv = conv_full.max()
        cur_J = float(2.0 * n * max_conv)

        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------
        # 4.2  Stagnation handling – jitter after 20 s without improvement
        # ------------------------------------------------------
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter applied (restart)")

        # ------------------------------------------------------
        # 4.3  Soft‑max weighting of convolution entries
        # ------------------------------------------------------
        beta = min(beta_max, beta + beta_inc)
        # Stabilised soft‑max: use max for numerical stability
        w_raw = np.exp(beta * (conv_full - max_conv))
        w = w_raw / w_raw.sum()

        # ------------------------------------------------------
        # 4.4  Gradient of the soft‑max objective (Fourier‑domain)
        # ------------------------------------------------------
        W = np.fft.rfft(w, fft_len)
        grad = 2.0 * np.fft.irfft(np.conj(W) * A, n=fft_len)[:n]    # real‑valued gradient

        # ------------------------------------------------------
        # 4.5  Exponentiated‑gradient proposal
        # ------------------------------------------------------
        step = 0.7 / np.sqrt(iteration + 1.0)   # slightly larger initial step
        factor = np.exp(-step * grad)          # always positive
        proposal = cur_vec * factor
        proposal = proj_simplex(proposal)

        # ------------------------------------------------------
        # 4.6  Evaluate the proposal
        # ------------------------------------------------------
        A_prop = np.fft.rfft(proposal, fft_len)
        conv_prop = np.fft.irfft(A_prop * A_prop, n=fft_len)[:conv_len]
        prop_J = float(2.0 * n * conv_prop.max())

        # ------------------------------------------------------
        # 4.7  Metropolis acceptance
        # ------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ------------------------------------------------------
        # 4.8  Periodic LP refinement (every 500 iterations)
        # ------------------------------------------------------
        if iteration % 500 == 0 and iteration > 0:
            # Determine “active” constraints – the top few convolution entries
            thr = max_conv * 0.998  # keep only the very largest values
            act_idx = np.where(conv_full >= thr)[0]
            if act_idx.size > 30:
                # Keep the 30 most critical indices only
                order = np.argsort(-conv_full[act_idx])[:30]
                act_idx = act_idx[order]
            if act_idx.size > 0:
                g_new = lp_refine(cur_vec, max_conv, act_idx)
                # blend a fraction of the LP solution into the current walk
                blend = 0.22
                cur_vec = (1.0 - blend) * cur_vec + blend * g_new
                cur_vec = proj_simplex(cur_vec)

        iteration += 1

        # early exit if we already surpass the target
        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # --------------------------------------------------------------
    # 5. Return the best solution (clipped and renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    # Optional sanity check against the provided evaluate_sequence
    try:
        score = evaluate_sequence(best_vec.tolist())
        print(f"Final evaluate_sequence score: {score:.6f}")
    except Exception:
        pass

    return [float(v) for v in best_vec]
