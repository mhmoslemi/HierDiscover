"""
Hybrid exponential‑gradient optimizer with soft‑max smoothing.

The objective to minimise is
    J(a) = 2·n·max(conv(a,a)) / (∑a)² ,
which is scale‑invariant.  We therefore keep the vector on the
probability simplex (∑a = 1) during the whole run.

Key ideas
---------

*  *Soft‑max smoothing* – instead of a hard active set we use the
   weights   w_k ∝ exp(α·conv_k)  (α grows from a small value to a
   large one).  The resulting gradient is a smooth convex combination
   of the sub‑gradients of the individual convolution entries.

*  *Exponentiated‑gradient (mirror descent) update* – a multiplicative
   step keeps all coordinates non‑negative and automatically respects
   the simplex after renormalisation.

*  *Deterministic annealing* – α is increased slowly, so the optimiser
   first sees a very smooth landscape and later focuses on the true
   maximum.

*  *Stagnation handling* – if no improvement is seen for a while the
   current best solution is jittered with multiplicative log‑normal
   noise, providing a cheap restart while preserving useful structure.

*  *Lightweight local refinement* – after the main loop a few random
   mass‑transfer moves are tried to squeeze out the last bits of
   improvement.

The routine runs for at most ``budget_s`` seconds and returns the best
sequence (as a plain Python list) it discovered.  All numbers are
clipped to the range accepted by ``evaluate_sequence`` before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures.
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution of ``x`` and ``y`` using a real FFT.
    Returns the first ``len(x)+len(y)-1`` entries.
    """
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    c = np.fft.irfft(X * Y, n=fft_len)
    return c[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Convolution of ``a`` with itself (full length)."""
    return _conv_fft(a, a, fft_len_self)


def _conv_rev_weight(a_rev: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Convolution of the reversed vector ``a_rev`` with the weight vector ``w``.
    Returned length is ``len(a_rev)+len(w)-1``.
    """
    return _conv_fft(a_rev, w, fft_len_grad)


def _objective_from_conv(n: int, conv_full: np.ndarray) -> float:
    """Compute J = 2·n·max(conv) assuming the vector is already normalised."""
    return float(2.0 * n * conv_full.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)         # for a_rev * w

    # --------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors.
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    x = np.arange(n, dtype=np.float64)

        elapsed = now - start_time

        conv_cur = _conv_self(cur, fft_len_self)
        max_conv = conv_cur.max()
        cur_J = 2.0 * n * max_conv

        # ----- keep global best -----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation handling (jitter) -----
        if now - last_improve > 30.0:

            cur = best_vec * noise
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {best_J:.6f})")

        # ----- soft‑max weights for the current convolution -----
        # stability: shift by the maximum value
        shift = max_conv
        w_raw = np.exp(alpha * (conv_cur - shift))
        w = w_raw / w_raw.sum()  # sum to 1, length = conv_len

        # ----- gradient of the smoothed max (via convolution) -----
        a_rev = cur[::-1]
        grad_full = _conv_rev_weight(a_rev, w, fft_len_grad)   # length 3n‑2
        # slice the part that corresponds to the n original coordinates
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # shape (n,)

        # centre the gradient – this does not change the direction
        grad_centered = grad - grad.mean()

        # ----- exponentiated‑gradient update -----
        eta = eta0 / np.sqrt(iteration + 1.0)      # diminishing step‑size
        factor = np.exp(-eta * grad_centered)
        proposal = cur * factor
        s = proposal.sum()
        if s == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= s

            proposal[receiver] += delta
            # renormalise (sum unchanged, but tiny numerical drift may appear)

        # ----- evaluate the proposal -----

        # Temperature decays linearly from 10 % of current best to a tiny value.
        T_start = best_J * 0.10
        T_end = 1e-8
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))

    # --------------------------------------------------------------

        new_vec[i] -= delta

        # evaluation

        new_J = 2.0 * n * conv_new.max()
        if new_J < best_J - 1e-12:

            best_vec = new_vec
            print(f"[{time.monotonic() - start_time:6.2f}s] Refine – new best J = {best_J:.6f}")

    # 6. Return the best solution (clip to allowed range and renormalise).
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
