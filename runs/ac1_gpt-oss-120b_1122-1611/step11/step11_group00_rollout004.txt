            a = np.exp(logits)
            a = normalize(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] jitter restart due to stagnation")
    # final best sequence
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    best_vec /= best_vec.sum()
    return [float(v) for v in best_vec]

"""
Propose a candidate step‑function using an alternating‐LP scheme.

The algorithm keeps a probability vector `a` (sum = 1).  For the current
vector it computes

    M = max(conv(a, a))

and then solves a *linear* program

    maximise    Σ_j g_j
    subject to  conv(a, g) ≤ M   (pointwise)
                g_j ≥ 0

The LP finds a distribution `g` that can carry *more* total mass than `a`
while respecting the same peak‐convolution bound `M`.  By normalising `g`
to sum = 1 (i.e. `ĝ = g / Σg`) we obtain a new candidate with a
strictly smaller peak (since `conv(a, ĝ) ≤ M / Σg < M`).  The new
candidate is accepted either directly (if its own peak is lower) or via
a convex combination with the old one.
If the LP does not produce any improvement we jitter the current vector
and repeat.  The procedure is repeated until the time budget is exhausted,
and the best vector found is returned (clipped and re‑normalised so that
`evaluate_sequence` can be called without further processing).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (output length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix `A` such that (A @ g) == conv(a, g) (full, length 2n‑1).

    Row `k` contains the shifted copy of `a` that multiplies the appropriate
    entries of `g` when the dot product is taken.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    A = np.empty((conv_len, n), dtype=np.float64)
    A[:] = 0.0
    for shift in range(n):
        A[shift:shift + n, shift] = a
    return A


def _normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    total = v.sum()
    if total <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / total


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds and returns the best vector discovered.

    The search uses an alternating linear‑programming scheme described
    above.  Several heuristic initial vectors are tried and the best
    one is refined.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (choose a moderate n for speed, but honour any
    # existing construction if it is not too large).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        tmp = np.clip(tmp, 0.0, None)
        n = int(min(max(tmp.size, 256), 2048))
        if tmp.size != n:
            if tmp.size > n:
                tmp = tmp[:n]
            else:
                tmp = np.concatenate([tmp, np.full(n - tmp.size, 0.0)])
        init_from_prev = _normalise(tmp)
    else:
        n = 1024
        init_from_prev = None

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # Build a portfolio of diverse initial probability vectors.
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        init_cands.append(_normalise(v))

    # linear ramps
    init_cands.append(_normalise(np.arange(n, 0, -1, dtype=np.float64)))
    init_cands.append(_normalise(np.arange(1, n + 1, dtype=np.float64)))

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        init_cands.append(_normalise(v))

    # Dirichlet draws – assorted concentration
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # previous construction, if present
    if init_from_prev is not None:
        init_cands.append(init_from_prev)

    # --------------------------------------------------------------
    # Choose the best start by raw objective J = 2·n·max(conv).
    # --------------------------------------------------------------
    best_start = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_start = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    a = best_start.copy()
    global_best_a = a.copy()
    global_best_J = best_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    last_improve = start_time
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ----- evaluate current objective -----
        conv_a = _conv_full(a, fft_len, conv_len)
        M = conv_a.max()
        cur_J = float(2.0 * n * M)

        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----- build convolution matrix for the LP -----
        A = _build_conv_matrix(a)

        # ----- solve the linear programme -----
        # maximise Σ g_j   ⇔   minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b = np.full(conv_len, M, dtype=np.float64)

        # Using HiGHS (fast for dense problems)
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method="highs",
                      options={"presolve": True, "time_limit": max(0.0, deadline - now)})

        # ----- if LP failed or gave no extra mass, jitter -----
        if not res.success:
            # a small stochastic restart
            jitter_sigma = 0.07
            logits = np.log(np.maximum(a, 1e-12))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            a = _normalise(np.exp(logits))
            continue

        g0 = res.x
        sum_g0 = g0.sum()

        if sum_g0 <= 1.0 + 1e-9:
            # No useful improvement – inject jitter to escape a plateau
            jitter_sigma = 0.10
            logits = np.log(np.maximum(a, 1e-12))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            a = _normalise(np.exp(logits))
            continue

        # ----- candidate built from the LP solution -----
        g_norm = g0 / sum_g0                # now Σ g_norm = 1
        conv_g = _conv_full(g_norm, fft_len, conv_len)
        J_g = float(2.0 * n * conv_g.max())

        # Accept the pure LP solution if it already improves the objective
        if J_g < cur_J - 1e-12:
            a = g_norm
            continue

        # ----- otherwise perform a convex mixing step -----
        # mixing weight proportional to the extra mass we obtained
        t = min(0.5, (sum_g0 - 1.0) / sum_g0)
        a = (1.0 - t) * a + t * g_norm
        a = _normalise(a)

        # ----- stagnation handling -----
        if now - last_improve > 30.0:
            jitter_sigma = 0.12
            logits = np.log(np.maximum(a, 1e-12))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            a = _normalise(np.exp(logits))
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart (stagnation)")

    # --------------------------------------------------------------
    # Return the best vector found (clip, re‑normalise, cast to list).
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    best_vec = _normalise(best_vec)          # guarantees sum = 1
    return [float(v) for v in best_vec]
