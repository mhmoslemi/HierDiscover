def compute_conv_and_gradient(vec, fft_len, conv_len, rel_tol=1e-6):
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    max_val = conv.max()
    tol = max(max_val * rel_tol, 1e-14)
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(vec)
    n = len(vec)
    for k in active:
        start = max(0, k - (n-1))
        end = min(n-1, k)
        idx = np.arange(start, end+1)
        grad[idx] += 2 * vec[k - idx]  # k-idx yields array of length idx, each entry valid.
    return conv, max_val, grad

"""
Hybrid Simulated‑Annealing / Gradient‑Guided optimiser for the step‑function
bound minimisation problem.

Key ideas
---------

* Work on the probability simplex (∑a_i = 1, a_i ≥ 0).  The evaluation
  objective J = 2·n·max(conv(a,a)) is invariant under scaling, so we keep
  the vector normalised after every update.

* Initialise the search with a diverse pool (uniform, geometric, Dirichlet,
  the previous best if available, a few hand‑crafted “U‑shapes”, …) and keep
  the best seed.

* Main loop = Simulated Annealing:
    – The current objective J and the active set (indices whose convolution
      value is within a tiny relative tolerance of the max) are computed via
      an FFT‑based convolution.
    – From the active set we build a sub‑gradient of max(conv).  Large
      gradient entries indicate positions that contribute most to the peak.
    – Three cheap move types are mixed:
        • *mass‑transfer*: move a fraction of probability from a high‑gradient
          index to a low‑gradient one (guided by the sub‑gradient).
        • *exponentiated‑gradient*: a multiplicative update
          a_i ← a_i·exp(‑η·grad_i) followed by simplex normalisation.
        • *jitter / smoothing*: occasional multiplicative Gaussian noise
          or a tiny convolution with a smoothing kernel to escape plateaus.
    – Moves are accepted if they improve J, otherwise with Metropolis
      probability exp(‑ΔJ / T) where T follows a linear cooling schedule.

* Stagnation handling – if no improvement is seen for a while we
  re‑initialise the current state by applying a stronger jitter around the
  best solution found so far.

The routine respects the 1000 s time budget, prints progress, and finally
returns the best sequence (as a list of Python floats) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns a 1‑D array of length conv_len = 2·len(a)‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _max_conv(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Return max(conv(a,a))."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(conv.max())


def _max_conv_and_grad(
    a: np.ndarray,
    fft_len: int,
    conv_len: int,
    rel_tol: float = 1e-5,
) -> tuple[float, np.ndarray]:
    """
    Compute max(conv(a,a)) and a sub‑gradient of the max‑convolution.

    The sub‑gradient is
        g_i = 2 * Σ_{k∈A} a_{k-i}
    where A = {k | conv(k) ≥ max – rel_tol·max }.
    """
    conv = _conv_full(a, fft_len, conv_len)
    max_val = float(conv.max())
    tol = max(max_val * rel_tol, 1e-14)
    active = np.where(conv >= max_val - tol)[0]

    n = a.shape[0]
    grad = np.zeros_like(a)

    # accumulate contributions from each active convolution index
    for k in active:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        idx = np.arange(start, end + 1)
        # k-idx yields the partner indices inside a that appear in conv(k)
        grad[idx] += 2.0 * a[k - idx]

    return max_val, grad


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx = 1}.
    Returns a new array (the input is not modified).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D vector")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all elements are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]  # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safeguard – renormalise exactly
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The function runs for at most
    ``budget_s`` seconds and returns the best sequence found (as a list of
    floats).  All numbers are clipped to the range accepted by
    ``evaluate_sequence`` before returning.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Generate a diverse set of starting vectors
    # --------------------------------------------------------------
    start_vectors = []

    # uniform
    start_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (forward & reversed)
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        start_vectors.append(v)

        v_rev = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
        v_rev /= v_rev.sum()
        start_vectors.append(v_rev)

    # linear ramps (incl. reversed)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    start_vectors.append(v)

    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    start_vectors.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        start_vectors.append(v)

    # U‑shaped (inverse quadratic) families
    for scale in [0.2, 0.4, 0.6]:
        i = np.arange(n, dtype=np.float64)
        dist = ((i - centre) / (scale * n)) ** 2 + 1.0
        v = 1.0 / dist
        v /= v.sum()
        start_vectors.append(v)

    # Dirichlet draws (moderately concentrated)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            start_vectors.append(rng.dirichlet(np.ones(n) * alpha))

    # reuse the previous best if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        start_vectors.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best among the initial candidates
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in start_vectors:
        max_c = _max_conv(cand, fft_len, conv_len)
        J = 2.0 * n * max_c
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑Annealing main loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)  # 1 s safety margin

    # temperature schedule (linear cooling)
    T_start = best_J * 0.20               # start relatively hot
    T_end = 1e-9

    cur_vec = best_vec.copy()
    cur_max, cur_grad = _max_conv_and_grad(cur_vec, fft_len, conv_len, rel_tol=1e-4)
    cur_J = 2.0 * n * cur_max

    last_improve = start_time
    iteration = 0

    # move‑size schedules
    transfer_scale_start = 0.25
    transfer_scale_end = 0.02
    jitter_sigma_start = 0.25
    jitter_sigma_end = 0.05
    exp_step_start = 2.0
    exp_step_end = 0.2
    smoothing_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        remaining = deadline - now
        if remaining <= 0.0:
            break

        # ----------------------------------------------------------
        # Adaptive parameters (linear decay with elapsed time)
        # ----------------------------------------------------------
        progress = elapsed / (deadline - start_time)
        transfer_scale = (
            transfer_scale_start * (1.0 - progress)
            + transfer_scale_end * progress
        )
        jitter_sigma = (
            jitter_sigma_start * (1.0 - progress)
            + jitter_sigma_end * progress
        )
        exp_step = (
            exp_step_start * (1.0 - progress)
            + exp_step_end * progress
        )
        # tolerance for the active set – start generous, then tighten
        if iteration < 5000:
            rel_tol = 0.02
        elif iteration < 20000:
            rel_tol = 0.01
        else:
            rel_tol = 0.004

        # ----------------------------------------------------------
        # Propose a new candidate
        # ----------------------------------------------------------
        move_type = rng.random()
        if move_type < 0.55:
            # ------------------ mass‑transfer move -------------------
            grad = cur_grad
            # donor – high‑gradient indices
            if grad.sum() > 0.0:
                donor = rng.choice(n, p=grad / grad.sum())
            else:
                donor = rng.integers(n)

            # receiver – low‑gradient indices
            inv = grad.max() - grad + 1e-12
            receiver = rng.choice(n, p=inv / inv.sum())

            max_delta = cur_vec[donor] * transfer_scale
            if max_delta <= 0.0:
                proposal = cur_vec.copy()
            else:
                delta = rng.random() * max_delta
                proposal = cur_vec.copy()
                proposal[donor] -= delta
                proposal[receiver] += delta

        elif move_type < 0.80:
            # ------------------ exponentiated‑gradient -------------
            factor = np.exp(-exp_step * cur_grad)
            proposal = cur_vec * factor
            proposal = _project_to_simplex(proposal)

        elif move_type < 0.90:
            # ------------------ jitter (multiplicative noise) ------
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            proposal = cur_vec * noise
            proposal = _project_to_simplex(proposal)

        else:
            # ------------------ smoothing (tiny convolution) ------
            tmp = np.convolve(cur_vec, smoothing_kernel, mode="same")
            proposal = np.clip(tmp, 0.0, None)
            proposal = _project_to_simplex(proposal)

        # ----------------------------------------------------------
        # Evaluate the proposal
        # ----------------------------------------------------------
        prop_max = _max_conv(proposal, fft_len, conv_len)
        prop_J = 2.0 * n * prop_max

        # ----------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------
        T = T_start * (1.0 - progress) + T_end * progress
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_max = prop_max
            cur_J = prop_J
            # recompute gradient for the new state (used next iteration)
            cur_max, cur_grad = _max_conv_and_grad(
                cur_vec, fft_len, conv_len, rel_tol=rel_tol
            )
        else:
            # even if we reject, we still need a fresh gradient for the next
            # iteration (unless we accept a move that already gave it).
            # Gradient of the unchanged state stays valid, so we keep it.
            pass

        # ----------------------------------------------------------
        # Track the global best
        # ----------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # Stagnation handling – restart with stronger jitter
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # reset to a jittered version of the best solution found so far
            jitter = np.exp(rng.normal(0.0, 0.3, size=n))
            cur_vec = best_vec * jitter
            cur_vec = _project_to_simplex(cur_vec)
            cur_max = _max_conv(cur_vec, fft_len, conv_len)
            cur_J = 2.0 * n * cur_max
            cur_max, cur_grad = _max_conv_and_grad(
                cur_vec, fft_len, conv_len, rel_tol=rel_tol
            )
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (jitter) – best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (safety clipping and normalisation)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # optional: final print
    final_J = 2.0 * n * _max_conv(best_vec, fft_len, conv_len)
    print(f"Finished – final best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
