    – An exponentiated‑gradient step a ← a·exp(−η·g) keeps the iterates

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _objective(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """J = 2·n·max(conv) / (sum^2).  Here vec is already non‑negative."""
    n = vec.size
    conv = _conv_full(vec, fft_len, conv_len)
    return float(2.0 * n * conv.max() / (vec.sum() ** 2))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    Returns the best normalised sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size (use previous construction if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a safe default

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a diverse set of initial vectors (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws – spiky to smooth
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Gaussian bells of several widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 6.0, n / 4.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        s = v.sum()
        if s > 0:
            v = v / s
        init_cands.append(v)

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        total = v.sum()
        if total > 0:
            v = v / total
        else:
            v = np.full(n, 1.0 / n)
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (objective = 2·n·max(conv) / sum²)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = _objective(cand, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Phase 1 – Sub‑gradient descent on the current best vector
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    start_time = time.monotonic()
    # reserve about 30 % of the budget for this phase
    subgrad_deadline = start_time + 0.30 * budget_s
    step0 = 0.8
    it = 0

    while time.monotonic() < subgrad_deadline:
        # convolution and arg‑max
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_val = conv.max()
        k_star = int(conv.argmax())

        # sub‑gradient of the maximal convolution entry
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur_vec)
        # g_i = a_i + a_{k*‑i}
        grad[idx] = cur_vec[idx] + cur_vec[k_star - idx]

        # exponentiated‑gradient step on the simplex
        eta = step0 / np.sqrt(it + 1.0)
        cur_vec = cur_vec * np.exp(-eta * grad)

        # renormalise to the simplex (projection reduces to scaling here)
        s = cur_vec.sum()
        if s > 0:
            cur_vec /= s
        else:
            cur_vec = np.full_like(cur_vec, 1.0 / n)

        cur_J = _objective(cur_vec, fft_len, conv_len)
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cur_vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Sub‑grad improved J = {best_J:.6f}")
            if best_J < 1.5025:
                print(f"[{elapsed:6.2f}s] Target reached during sub‑grad phase.")
                break
        it += 1

    # --------------------------------------------------------------
    # 5. Phase 2 – Targeted simulated annealing
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = time.monotonic()
    iteration = 0

    # temperature schedule (linear cooling)
    T_start = best_J * 0.10   # 10 % of current best as initial temperature
    T_end = 1e-8

    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear annealing
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))

        # ----------------------------------------------------------
        # 5.1  Propose a new candidate
        # ----------------------------------------------------------
        if rng.random() < 0.02:                     # occasional global jitter
            sigma = 0.05
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            proposal = cur_vec * noise
            proposal = proj_simplex(proposal)
        else:
            # compute the current maximal convolution entry and its location
            conv = _conv_full(cur_vec, fft_len, conv_len)
            max_val = conv.max()
            k_star = int(conv.argmax())

            # contributions to the maximal entry: a_i * a_{k*-i}
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            contrib = cur_vec[idx] * cur_vec[k_star - idx]
            total_contrib = contrib.sum()

            # pick i preferably from a heavy contributor
            if total_contrib > 0 and rng.random() < 0.7:
                probs = contrib / total_contrib
                i = rng.choice(idx, p=probs)
            else:
                i = rng.integers(0, n)

            # pick a distinct j uniformly
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move a random fraction of mass from i to j
            max_move = cur_vec[i] * 0.5               # at most 50 % of the source mass
            delta = rng.random() * max_move
            proposal = cur_vec.copy()
            proposal[i] -= delta
            proposal[j] += delta

        # ----------------------------------------------------------
        # 5.2  Evaluate the proposal
        # ----------------------------------------------------------
        prop_J = _objective(proposal, fft_len, conv_len)

        # ----------------------------------------------------------
        # 5.3  Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ----------------------------------------------------------
        # 5.4  Update global best
        # ----------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Anneal improved J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 5.5  Stagnation handling
        # ----------------------------------------------------------
        if now - last_improve > 20.0:            # 20 s without improvement
            sigma = 0.04
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stagnation jitter – best J = {best_J:.6f}")

        iteration += 1

        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # --------------------------------------------------------------
    # 6. Final normalisation & clipping
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total > 0:
        best_vec = best_vec / total
    else:
        best_vec = np.full(n, 1.0 / n)

    # Return as a plain Python list (required by the evaluation harness)
    return [float(v) for v in best_vec]
