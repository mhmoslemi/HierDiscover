"""
Propose a step‑function (non‑negative sequence) that minimizes
`evaluate_sequence`.  The algorithm works on a fixed length *n*
(probably 4096 or the length of the previous construction),
normalises the vector to a probability simplex, and then performs
a projected (multiplicative) sub‑gradient descent on the true
objective  J = 2 n·max(conv(a,a))  (the sum of the vector is 1, so the
denominator of `evaluate_sequence` vanishes).  At each iteration the
index *k* achieving the maximal convolution coefficient is located;
the sub‑gradient of that coefficient is `g_j = 2·a_{k‑j}` (zero outside
the support).  A decaying exponential update `a←a·exp(‑η·g)` is applied
and projected back onto the simplex.  Simple simulated‑annealing,
periodic jitter and a small portfolio of diverse initialisations
help escape plateaus.  The routine runs until the time budget expires
and returns the best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to one).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    # if all false, just return uniform
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (the full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine – the function required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size (length of the step function)
    # ------------------------------------------------------------------
    # Prefer the length of the previous best construction if available,
    # otherwise fall back to a power‑of‑two that is cheap for FFTs.
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # length for a*a

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial probability vectors
    # ------------------------------------------------------------------
    init_vecs = []

    # uniform
    init_vecs.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_vecs.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_vecs.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_vecs.append(v)

    # Gaussian bells (different widths)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 12.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_vecs.append(v)

    # Dirichlet draws (various concentration)
    for alpha in [0.2, 0.5, 1.0]:
        for _ in range(2):
            init_vecs.append(rng.dirichlet(np.ones(n) * alpha))

    # Sparse masks – a few equally weighted spikes
    for k in [max(2, n // 30), max(2, n // 15), max(2, n // 8)]:
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        init_vecs.append(mask)

    # Arcsine‑like shape (large at the ends, small in the middle)
    if n >= 4:
        idx = np.arange(1, n - 1, dtype=np.float64)
        v = 1.0 / np.sqrt(idx * (n - idx))
        v = np.concatenate([[0.0], v, [0.0]])
        v = np.clip(v, 0, None)
        v /= v.sum()
        init_vecs.append(v)

    # Previous construction (if any) – normalised to a simplex
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        if v.sum() == 0.0:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v = v / v.sum()
        init_vecs.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best of the initial candidates (true objective)
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_vecs:
        conv = _conv_full(cand, fft_len, conv_len)
        J = float(2.0 * n * conv.max())          # sum is 1 → denominator =1
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop – projected sub‑gradient descent
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # leave a tiny safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # annealing parameters
    T_start = best_J * 0.05          # 5 % of the current best is the initial temperature
    T_end = 1e-8

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- evaluate current vector -----
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_b = conv.max()
        cur_J = float(2.0 * n * max_b)

        # ----- keep global best -----
        if cur_J < best_J - 1e-14:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation handling: jitter restart -----
        if now - last_improve > 30.0:
            # add log‑normal jitter around the best solution
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * jitter
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ----- sub‑gradient for the active lag -----
        k_star = int(np.argmax(conv))                 # index of the maximal coefficient
        idx = np.arange(n, dtype=np.int64)
        rev_idx = k_star - idx                         # the partner index for each position
        mask = (rev_idx >= 0) & (rev_idx < n)
        grad = np.zeros_like(cur_vec)
        # g_j = 2 * a_{k*-j}
        grad[mask] = 2.0 * cur_vec[rev_idx[mask]]

        # ----- multiplicative update (exponentiated‑gradient) -----
        step = 0.02 / np.sqrt(iteration + 1.0)        # decaying step size
        factor = np.exp(-step * grad)                 # multiplicative factor (≤1)
        proposal = cur_vec * factor
        proposal = proj_simplex(proposal)             # re‑normalise on the simplex

        # ----- evaluate proposal -----
        prop_conv = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ----- Metropolis acceptance with linear cooling -----
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * (elapsed / (deadline - start_time)))
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

        # occasional lightweight progress report
        if iteration % 500 == 0:
            print(f"[{elapsed:6.2f}s] iter {iteration:5d} – cur J = {cur_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Post‑processing – a few sanity checks and final return
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)   # enforce the hard caps used by the evaluator
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total                         # final normalisation (sum ≈ 1)

    # Return as plain Python floats
    return [float(v) for v in best_vec]
