"""
Simulated‑annealing optimiser for the step‑function problem.

The evaluation function is scale‑invariant, so we restrict the search to
probability vectors (sum = 1) and minimise

    J(a) = 2·n·max(conv(a,a))

with a_i ≥ 0, Σ a_i = 1.
A simple simulated‑annealing scheme is used:

* a set of diverse initial candidates (uniform, geometric, ramps,
  Gaussians, Dirichlet draws and the previously found construction) is
  evaluated and the best one is kept;
* the main loop repeatedly moves a small amount of probability mass from
  a random source index to a random target index;
* the move is accepted always if it improves J, otherwise with
  Metropolis probability exp(−ΔJ / T) where the temperature T decays
  linearly to zero;
* if no improvement is seen for 30 s the current state is jittered by
  multiplicative log‑normal noise centred on the best solution found so
  far – this helps escape local minima;
* after the annealing phase a brief fine‑grained random‑search phase
  performs tiny perturbations around the best vector.

All heavy work (the convolution) is performed with a real FFT, which
makes each evaluation O(n log n).  The routine respects the overall
time budget (default 1000 s) and returns the best sequence discovered as
a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def max_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Return the maximum entry of the linear convolution a*a.
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the provided evaluation function.  The routine runs for at most
    `budget_s` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose the problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # for a*a
    # ------------------------------------------------------------------
    # 2. Build a portfolio of diverse initial probability vectors.
    # ------------------------------------------------------------------
    init_cands = []

    # Uniform distribution (baseline)
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Geometric decays (r < 1)
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 12.0, n / 9.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – moderate and concentrated
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(6):
            init_cands.append(rng.dirichlet(np.full(n, alpha, dtype=np.float64)))

    # previous best (if present)
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v /= s
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            v = proj_simplex(v)
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Evaluate initial candidates and keep the best.
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        cur_J = 2.0 * n * max_convolution(cand, fft_len, conv_len)
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time

    # Annealing schedule parameters
    T_start = 0.02                 # initial temperature (controls uphill acceptance)
    T_end = 1e-8                  # final temperature – effectively zero
    stagnation_limit = 30.0       # seconds without improvement → jitter

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)  # in [0,1]

        # linearly decay temperature
        T = T_start * (1.0 - frac) + T_end

        # ------------------------------------------------------------------
        # 4.1 Propose a local move: transfer a small amount of mass from i → j.
        # ------------------------------------------------------------------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1                     # ensure i != j

        # maximum transferable amount is the current mass at i
        max_transfer = cur_vec[i]
        if max_transfer <= 1e-12:
            # source essentially empty – resample
            continue

        # magnitude scales with temperature (larger early, vanishing later)
        base_move = 0.05               # up to 5 % of the total probability
        delta = base_move * T * rng.random()
        delta = min(delta, max_transfer)

        # construct proposal
        proposal = cur_vec.copy()
        proposal[i] -= delta
        proposal[j] += delta

        # ------------------------------------------------------------------
        # 4.2 Evaluate objective for the proposal.
        # ------------------------------------------------------------------
        prop_J = 2.0 * n * max_convolution(proposal, fft_len, conv_len)

        # ------------------------------------------------------------------
        # 4.3 Metropolis acceptance.
        # ------------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # uphill move – accept with probability exp(-Δ/T)
            # guard against overflow with a tiny floor on T
            if T > 0:
                prob = np.exp(-(prop_J - cur_J) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

            # record new global best
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.4 Stagnation handling – random jitter around the best solution.
        # ------------------------------------------------------------------
        if now - last_improve > stagnation_limit:
            sigma = 0.2                # jitter strength
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            cur_vec = best_vec * noise
            cur_vec /= cur_vec.sum()
            cur_J = 2.0 * n * max_convolution(cur_vec, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random search around the best solution.
    # ------------------------------------------------------------------
    fin_start = time.monotonic()
    fin_deadline = fin_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fin_deadline:
        sigma = 0.02
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_vec * noise
        cand /= cand.sum()
        cand_J = 2.0 * n * max_convolution(cand, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            now = time.monotonic() - start_time
            print(f"[Fine] Improved J = {best_J:.6f} at {now:6.2f}s")

    # ------------------------------------------------------------------
    # 6. Normalise and return the best sequence.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # The returned list is ready for evaluate_sequence().
    return [float(v) for v in best_vec]
