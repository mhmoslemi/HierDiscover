"""
Mirror‑descent (Adam) optimizer on the probability simplex.

* Starts from a large, diverse portfolio of candidates (random Dirichlet
  draws, classic window functions, exponential blends and the previously
  discovered construction, if any).

* The best candidate is refined by minimizing the smooth proxy of
  ``J = 2·n·max(conv(a,a))`` using an Adam update on the log‑probability
  (exponentiated‑gradient on the simplex).  The gradient is a sub‑gradient
  of the maximum convolution entry obtained from the active set
  (all indices where the convolution is within a tiny tolerance of the
  maximum).  For large active sets a fast FFT‑based correlation is used
  instead of an O(|S|·n) loop.

* The optimizer runs until the 1000 s budget expires (a small safety
  margin is kept).  Intermediate improvements are printed.  The final
  vector is clipped to [0,1000] and renormalised to unit sum before being
  returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum value of the full linear convolution vec * vec."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def subgradient(vec: np.ndarray,
                conv_vals: np.ndarray,
                max_c: float,
                fft_len: int,
                tol: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of the scalar function ``max(conv(vec,vec))``
    w.r.t. ``vec``.  If the active set (indices attaining the maximum)
    is small, an explicit O(|S|·n) loop is used; otherwise a fast FFT
    based correlation computes the same quantity.

    Returns ``grad`` such that ``max_c + <grad, delta>`` is a first‑order
    upper bound for the perturbed max.
    """
    n = vec.size
    # active set: all indices whose value is within *tol* (absolute or
    # relative) of the maximum.
    active = np.where(conv_vals >= max_c - max(tol, max_c * 1e-12))[0]
    if active.size <= 200:               # cheap explicit loop
        grad = np.zeros_like(vec)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=np.int64)
            grad[idx] += 2.0 * vec[p - idx]      # ∂conv_p/∂vec_i = 2·vec_{p‑i}
        return grad
    # large active set -> use correlation via FFT
    a = np.zeros(conv_vals.shape, dtype=np.float64)
    a[active] = 1.0
    # correlation a ⋆ vec  (real‑valued, non‑circular)
    A = np.fft.rfft(a, n=fft_len)
    V = np.fft.rfft(vec, n=fft_len)
    corr = np.fft.irfft(A * np.conj(V), n=fft_len)   # length = fft_len
    # extract the part needed for the gradient.
    # For i in [0,n) we need Σ_r a[r]·vec[r‑i] = corr[(n‑1)‑i] (see analysis).
    offset = n - 1
    # Use the slice that directly yields the needed values.
    grad = 2.0 * corr[offset:offset + n]
    return grad

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = len(prev)
            # keep the size reasonable for fast FFTs
            n = max(1024, min(n, 4096))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Helper to evaluate a candidate and keep the global best.
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    start_time = time.monotonic()

    def eval_and_update(v: np.ndarray) -> None:
        """Normalise ``v`` to sum‑1, compute J and update the global best."""
        nonlocal best_vec, best_J
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            return
        v = v / s
        max_c = full_conv_max(v, fft_len, conv_len)
        J = 2.0 * n * max_c
        if J < best_J:
            best_J = J
            best_vec = v.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Broad initial portfolio (parameter sweep + random draws).
    # ------------------------------------------------------------------
    # Uniform baseline
    eval_and_update(np.full(n, 1.0 / n, dtype=np.float64))

    # Classic windows (Hann, Hamming, Blackman, Bartlett)
    windows = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),
    }
    for w in windows.values():
        if w.sum() > 0.0:
            eval_and_update(w / w.sum())

    # Kaiser windows – sweep beta in a fairly fine grid
    for beta in np.linspace(0.0, 30.0, 61):
        w = np.kaiser(n, beta)
        if w.sum() > 0.0:
            eval_and_update(w / w.sum())

    # Tukey windows – vary tapering parameter alpha
    for alpha in np.linspace(0.0, 1.0, 21):
        t = np.arange(n, dtype=np.float64) / (n - 1)
        w = np.ones_like(t)
        if alpha > 0.0:
            first = t < alpha / 2.0
            w[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
            last = t > 1.0 - alpha / 2.0
            w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
        eval_and_update(w / w.sum())

    # Exponential edge blends (mixture of uniform & symmetric exponential)
    for _ in range(200):
        alpha = 10 ** rng.uniform(-3.0, 1.0)     # 1e‑3 … 10
        mix = rng.random()
        expo = np.exp(-alpha * np.arange(n, dtype=np.float64))
        expo += np.exp(-alpha * (n - 1 - np.arange(n, dtype=np.float64)))
        expo /= expo.sum()
        v = (1.0 - mix) * np.full(n, 1.0 / n) + mix * expo
        eval_and_update(v)

    # Random Dirichlet draws (wide range of concentration parameters)
    conc_params = np.concatenate([
        np.logspace(-2.0, 1.0, 12),   # 0.01 … 10
        np.array([0.3, 0.6, 1.0, 2.0, 5.0])
    ])
    for conc in conc_params:
        for _ in range(80):
            eval_and_update(rng.dirichlet(np.full(n, conc)))

    # Previous construction, if present (maybe already good)
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if v.size != n:
            # truncate or pad to the chosen length
            if v.size > n:
                v = v[:n]
            else:
                v = np.pad(v, (0, n - v.size), mode='constant')
        eval_and_update(v)

    # ------------------------------------------------------------------
    # 4. Local refinement – Adam (mirror descent) on the simplex.
    # ------------------------------------------------------------------
    # Initialise Adam on the log‑probability (exponentiated‑gradient)
    eps = 1e-30
    if best_vec is None:
        cur_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur_vec = best_vec.copy()
    logits = np.log(np.maximum(cur_vec, eps))

    m = np.zeros_like(logits)          # first moment
    v_m = np.zeros_like(logits)        # second moment
    beta1 = 0.9
    beta2 = 0.99
    lr = 0.25                           # base learning rate on logits
    it = 0

    # Time budget (leave a few seconds margin for tidy‑up)
    deadline = start_time + max(0.0, float(budget_s) - 3.0)

    while time.monotonic() < deadline:
        # ----- probability vector (softmax) -----
        max_log = np.max(logits)
        probs = np.exp(logits - max_log)
        w = probs / probs.sum()

        # ----- convolution & max -----
        A = np.fft.rfft(w, n=fft_len)
        conv_vals = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_c = float(conv_vals.max())

        # ----- evaluate current J for bookkeeping -----
        J_cur = 2.0 * n * max_c
        if J_cur < best_J:
            best_J = J_cur
            best_vec = w.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_J:.6f}")

        # ----- sub‑gradient of max(conv) -----
        grad = subgradient(w, conv_vals, max_c, fft_len)

        # ----- gradient of J w.r.t. w -----
        dJdw = 2.0 * n * grad                       # J = 2·n·max(conv)

        # ----- mirror‑descent (log‑space) gradient -----
        avg = np.dot(w, dJdw)
        dJdl = w * (dJdw - avg)                     # ∂J/∂logits (KL‑mirror)

        # ----- Adam update on logits -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * dJdl
        v_m = beta2 * v_m + (1.0 - beta2) * (dJdl ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_m / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # ----- occasional jitter to escape plateaus -----
        if it % 3000 == 0:
            # restart around the current best with a touch of noise
            noise_factor = np.exp(rng.normal(0.0, 0.03, size=n))
            logits = np.log(np.maximum(best_vec, eps)) + rng.normal(0.0, 0.2, size=n)
            logits += np.log(noise_factor)
            m.fill(0.0)
            v_m.fill(0.0)

        # ----- very occasional large random perturbation -----
        if it % 8000 == 0 and rng.random() < 0.15:
            # completely random Dirichlet point blended with current best
            rand = rng.dirichlet(np.ones(n))
            w_mix = 0.8 * w + 0.2 * rand
            logits = np.log(np.maximum(w_mix, eps))

    # ------------------------------------------------------------------
    # 5. Return the best solution found (clip, renormalise, cast to list).
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(x) for x in best_vec]
