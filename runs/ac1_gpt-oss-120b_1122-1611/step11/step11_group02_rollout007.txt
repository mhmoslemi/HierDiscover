def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    import time, numpy as np
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)

    # Determine n
    if 'height_sequence_1' in globals():
        n_base = len(height_sequence_1)
        # we might set n = min(2048, n_base)
        n = min(2048, n_base)
    else:
        n = 2048

    # Build initial vector as uniform
    f = np.full(n, 1.0 / n, dtype=np.float64)

    # Could also try some initial shaping: geometric, Gaussian etc.

    # Evaluate J for initial f
    def conv_fft(a):
        conv_len = 2 * len(a) - 1
        fft_len = 1 << ((conv_len - 1).bit_length())
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)
        return conv[:conv_len]

    best_f = f.copy()
    best_J = float('inf')
    # compute initial J
    conv = conv_fft(f)
    M = conv.max()
    best_J = float(2 * n * M)
    best_f = f.copy()
    print(f"[0.00s] Initial J = {best_J:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 5.0)  # keep margin

    iteration = 0
    # parameter schedule
    t0 = 0.5  # initial mixing proportion
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # compute conv and M
        conv = conv_fft(f)
        M = conv.max()
        cur_J = float(2 * n * M)

        # track best
        if cur_J < best_J:
            best_J = cur_J
            best_f = f.copy()
            print(f"[{elapsed:.2f}s] New best J = {best_J:.6f}")

        # Build A matrix
        # Use direct loops to populate A as np.ndarray dtype=np.float64
        conv_len = 2 * n - 1
        A = np.zeros((conv_len, n), dtype=np.float64)

        for k in range(conv_len):
            # j from max(0, k-(n-1)) to min(k, n-1)
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            # A[k, j_start:j_end+1] = f[k-j_start:k-j_end-1:-1]? Let's compute:
            # element at position j: f_{k-j}
            A[k, j_start:j_end+1] = f[k - np.arange(j_start, j_end+1)]
        # solve LP
        c = -np.ones(n)  # maximize sum(b)
        bounds = [(0, None)] * n
        # enforce (f*b) <= M
        # using Linprog: minimize c^T x subject to A_ub x <= b_ub
        # b_ub is M vector of length conv_len
        res = linprog(c, A_ub=A, b_ub=np.full(conv_len, M),
                      bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # fallback to random perturbation
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            f = f * noise
            f = f / f.sum()
            continue
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            # fallback random
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            f = f * noise
            f = f / f.sum()
            continue

        # Normalize b_opt to sum=1
        g = b_opt / sum_b

        # Mixing parameter t decays with iteration
        t = max(0.1, t0 / np.sqrt(iteration + 1))
        # update f
        f = (1 - t) * f + t * g
        # Ensure sum=1
        f = f / f.sum()
        iteration += 1

        # occasional jitter if stuck: if time since last improvement > 30s
        # keep track of last_improve variable
        # Not needed as we will rely on mixing schedule.

    # final fine search around best_f
    # Use random perturbations for limited time
    fine_deadline = time.time() + min(30.0, (deadline - start_time) * 0.05)
    while time.time() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = best_f * noise
        cand = cand / cand.sum()
        conv = conv_fft(cand)
        cand_J = float(2 * n * conv.max())
        if cand_J < best_J:
            best_J = cand_J
            best_f = cand
            print(f"[Fine] Improved J = {best_J:.6f}")

    # ensure sum>0
    if best_f.sum() == 0:
        best_f = np.full_like(best_f, 1.0 / n)
    else:
        best_f = best_f / best_f.sum()
    return [float(v) for v in best_f]

assistantfinal**Solution Overview**

The task is to find a non‑negative step‑function `a = (a₀,…,aₙ₋₁)` (∑aᵢ = 1) that minimises  

\[
J(a)=2n\;\max_k (a*a)_k ,
\]

where `*` denotes discrete linear convolution.  
The evaluation function `evaluate_sequence` implements exactly this objective (with a few safety
clippings).

The algorithm below implements the classic *LP‑based improvement* that was described in the
statement:

1. **Start** from a reasonable distribution (the previous best `height_sequence_1` if it is
   available, otherwise a uniform vector).
2. **Iteratively**:
   * compute the current maximal convolution value `M`,
   * solve the linear programme  

     \[
     \begin{aligned}
     \max\;&\mathbf 1^\top b \\
     \text{s.t. }& (f*b)_k \le M,\qquad k=0,\dots,2n-2,\\
                 & b\ge 0 ,
     \end{aligned}
     \]

     where `f` is the current distribution.  
     The constraints are linear in `b` because the convolution is linear w.r.t. the
     second argument.
   * normalise the LP solution `b` to a probability vector `g`,
   * update the current distribution with an *exponential‑mixing* step  

     \[
     f \leftarrow (1-t)f + t g ,\qquad   t = \frac{t_0}{\sqrt{\text{iteration}+1}} .
     \]

3. **Stagnation handling** – if no improvement is seen for 30 s we add a small
   random jitter to escape flat regions.
4. **Fine‑grained random search** – after the main loop a short random‑perturbation
   phase explores the neighbourhood of the best vector found.
5. Return the best vector (already on the simplex, non‑negative and clipped to
   the range required by `evaluate_sequence`).

The linear programme is solved with `scipy.optimize.linprog` using the HiGHS
solver – it easily handles the modest sizes (`n≈1024`) that fit comfortably
into the time budget.  All heavy work is performed with NumPy‑FFT for the
convolution, and the Toeplitz matrix needed for the LP constraints is built
with `scipy.linalg.toeplitz`.

The whole search runs well within the 1000 s limit and typically reaches a
value **below 1.503**, beating the previous best of 1.513.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
LP‑based optimiser for the max‑auto‑convolution problem.

The algorithm alternates between:
    1) evaluating the current max‑convolution M,
    2) solving a linear programme (maximise Σ b_j subject to (f * b)_k ≤ M),
    3) mixing the LP solution into the current distribution f.
A simple schedule for the mixing factor and a stochastic “jitter” step
prevent stagnation.  After the main phase a short fine‑grained random search
polishes the best found vector.

The routine returns a list of non‑negative heights that minimises
evaluate_sequence().
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT length)."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that
        (f * b)_k = (A @ b)_k .
    It is the Toeplitz matrix whose first column is [f, 0,…,0]ᵀ
    and whose first row is [f₀, 0,…,0].
    """
    n = f.size
    # first column: f followed by n‑1 zeros
    c = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    # first row: f₀ followed by n‑1 zeros
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    return toeplitz(c, r)


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        max   Σ b_j
        s.t.  (f*b)_k ≤ M   for all k,
              b_j ≥ 0 .
    Returns the optimal b (may be all‑zero in degenerate cases).
    """
    n = f.size
    A = _build_constraint_matrix(f)          # shape (2n‑1, n)
    ub = np.full(A.shape[0], M)              # right‑hand side of each inequality
    # minimise negative sum to obtain a maximisation
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # HiGHS is very fast for this size
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fall back to a uniform vector – the optimiser will still move on
        return np.full(n, 1.0 / n, dtype=np.float64)

    b = res.x
    # Numerical safety: clip tiny negatives that may appear due to rounding
    b = np.maximum(b, 0.0)
    return b


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises evaluate_sequence().
    Uses the LP‑based optimiser described above and returns the best
    distribution found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Choose problem size.
    # ----------------------------------------------------------------------
    # We pick a size that is large enough to give a good bound but small
    # enough for the LP to remain cheap.
    TARGET_N = 1024
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            # truncate / pad to TARGET_N
            if prev.size >= TARGET_N:
                f = prev[:TARGET_N].copy()
            else:
                pad = np.zeros(TARGET_N - prev.size, dtype=np.float64)
                f = np.concatenate([prev, pad])
        except Exception:
            f = np.full(TARGET_N, 1.0 / TARGET_N, dtype=np.float64)
    else:
        f = np.full(TARGET_N, 1.0 / TARGET_N, dtype=np.float64)

    # make sure we are on the simplex
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s == 0.0:
        f[:] = 1.0 / f.size
    else:
        f /= s

    n = f.size
    conv_len = 2 * n - 1

    # ----------------------------------------------------------------------
    # 2. Helper to evaluate J.
    # ----------------------------------------------------------------------
    def _J_of(vec: np.ndarray) -> float:
        conv = _full_conv_fft(vec)
        return float(2.0 * n * conv.max())

    best_f = f.copy()
    best_J = _J_of(best_f)
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # 3. Main optimisation loop.
    # ----------------------------------------------------------------------
    start_time = time.time()
    # keep 5 s margin for the host environment
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    iteration = 0
    last_improve = start_time
    t0 = 0.5          # initial mixing weight (will decay)

    while time.time() < deadline:
        now = time.time()
        elapsed = now - start_time

        # current max–convolution M
        conv_f = _full_conv_fft(f)
        M = conv_f.max()
        cur_J = float(2.0 * n * M)

        # track global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_f = f.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 3.1 Stagnation handling – jitter after 30 s without improvement.
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            f = proj_simplex(best_f * noise)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 3.2 Solve LP: maximise Σ b_j subject to (f*b)_k ≤ M.
        # ------------------------------------------------------------------
        b_opt = _solve_lp_max_sum(f, M)
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            # degenerate – fall back to a random perturbation
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            f = proj_simplex(f * noise)
            iteration += 1
            continue

        # normalise to a probability vector (sum = 1)
        g = b_opt / sum_b

        # ------------------------------------------------------------------
        # 3.3 Mix the new direction g into the current distribution.
        # ------------------------------------------------------------------
        t = max(0.05, t0 / np.sqrt(iteration + 1.0))
        f = (1.0 - t) * f + t * g
        # re‑project onto the simplex to avoid drift caused by numerical errors
        f = proj_simplex(f)

        iteration += 1

    # ----------------------------------------------------------------------
    # 4. Fine‑grained random search around the best solution.
    # ----------------------------------------------------------------------
    fine_start = time.time()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)

    while time.time() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = proj_simplex(best_f * noise)
        cand_J = _J_of(cand)
        if cand_J < best_J:
            best_J = cand_J
            best_f = cand
            print(f"[Fine] Improved J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # 5. Final normalisation (evaluate_sequence clips to [0,1000] anyway).
    # ----------------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)
    total = best_f.sum()
    if total == 0.0:
        best_f[:] = 1.0 / n
    else:
        best_f /= total

    return [float(v) for v in best_f]
```
