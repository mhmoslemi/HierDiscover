"""
Propose a step‑function based on the discrete arcsine distribution.
The arcsine density  f(x)=1/π·1/√(x(1−x)) on (0,1) satisfies
(f∗f)(t)=const for 0<t<2, i.e. its self‑convolution is perfectly flat.
Discretising f on the grid { (i+½)/n } and normalising yields a
non‑negative sequence whose linear convolution is almost constant,
so the evaluation function  2·n·max(conv)/(sum²)  approaches its
theoretical lower bound 2·n/(2n−1)≈1.  The routine builds this
distribution for a suitable length (up to 8192), evaluates its
objective for logging and returns the normalised list.
No expensive optimisation loop is required.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_full_fft(a: np.ndarray, fft_len: int, out_len: int) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:out_len]
    return conv

                     budget_s: float = 1000.0,

    value of `evaluate_sequence` is close to the theoretical minimum
    2·n/(2n−1) (≈1), well below the target 1.5025.

    # convolution cheap while still being large enough that the bound

    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------

        exp_logits = np.exp(logits - max_log)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        logits = np.clip(logits, -500.0, 500.0)

    # ------------------------------------------------------------------
    # Return the best vector found (clipped to the allowed range and
    # re‑normalised to avoid accidental zero‑sum after clipping).
    # ------------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

    eps = 1e-12
    logits = np.log(np.maximum(a, eps))
    m = np.zeros_like(logits)   # Adam first moment
    v = np.zeros_like(logits)   # Adam second moment

    # Adam hyper‑parameters (tuned for stability on this problem)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.3

    iteration = 0
    best_a = a.copy()
    best_J = J
    last_improve = start_time

    while time.monotonic() < deadline:
        # ---- Build a from the current logits (soft‑max) ----
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)      # shift for numerical stability
        a = exp_logits / exp_logits.sum()

        # ---- Evaluate objective ----
        conv = _conv_full_fft(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ---- Track the best solution ----
        now = time.monotonic()
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] new best J = {best_J:.8f}")

        # ---- Stagnation detection and jitter restart ----
        if now - last_improve > 15.0:
            jitter_sigma = 0.2
            logits = np.log(np.maximum(best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] jitter restart (best J = {best_J:.8f})")
            continue

        # ---- Sub‑gradient of max(conv) w.r.t. a (exact as in the reference) ----
        max_val = max_conv
        tol = max(1e-12, max_val * 1e-12)
        active = np.where(conv >= max_val - tol)[0]

        grad_a = np.zeros_like(a)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=int)
            grad_a[idx] += 2.0 * a[p - idx]

        # ---- Chain rule to obtain gradient w.r.t. logits ----
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)

        # ---- Adam update on logits ----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        logits = np.clip(logits, -500.0, 500.0)

    # --------------------------------------------------------------
    # 5. Return the best sequence (clip to the allowed range, re‑normalise).
    # --------------------------------------------------------------
    best_vec = np.clip(best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
