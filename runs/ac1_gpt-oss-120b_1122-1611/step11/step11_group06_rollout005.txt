"""
Propose a non‑negative step‑function (probability vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Choose a reasonable length *n* (up to 8192).  If a previous solution
  ``height_sequence_1`` exists it is resampled; otherwise a uniform start
  is used.

* Build a diverse portfolio of initial candidates (uniform, classic windows,
  geometric mixes, random Dirichlet draws, the previous construction, …) and keep
  the best one according to the true objective
  J = 2·n·max(conv(a,a)).

* Run a continuous optimisation on the probability simplex:
  – At each iteration compute the full linear convolution `c = a*a`
    using FFT (O(n log n)).
  – Replace the nonsmooth ``max`` by a smooth log‑sum‑exp surrogate.
    For a temperature parameter `γ` we use weights

        w_k ∝ exp(γ·(c_k – max(c))).

    As `γ` grows the weights concentrate on the active indices of the max,
    reproducing a sub‑gradient of the original objective.
  – The gradient of the smoothed objective is

        ∇J_i = 2·Σ_k w_k·a_{k‑i}
               = 2·(w * rev(a))[i + n‑1],

    i.e. a convolution of the weight vector with the reversed current vector.
    This is also evaluated with FFT.

  – An exponentiated‑gradient update keeps the iterate inside the simplex:

        a ← a·exp(‑η·n·∇J) ,    a ← a / Σa

    (the factor `n` normalises the gradient magnitude).

  – Learning rate `η` follows a 1/√t schedule, while `γ` is increased
    linearly in time from a small smoothing value (≈5) to a large one
    (≈4000).  Periodic random jitter (multiplicative log‑normal noise)
    prevents stagnation.

* The best vector seen during the run is returned (clipped to the allowed
  range and renormalised).

The method combines a smooth approximation of the max‑convolution objective
with fast FFT‑based gradients and a mirror‑descent step on the simplex,
which is well suited for the hundred‑second budget and yields a tighter
upper bound than the previous heuristic (target < 1.5025)."""

import time
import numpy as np
from typing import List


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window."""
    if alpha <= 0.0:
        return np.ones(n, dtype=np.float64)
    if alpha >= 1.0:
        return np.hanning(n)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(t)
    first = t < alpha / 2.0
    w[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
    last = t > 1.0 - alpha / 2.0
    w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
    return w


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> List[float]:
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size.
    # ------------------------------------------------------------------
    # Prefer a fairly large n (up to 8192) – larger n gives a smaller theoretical
    # lower bound (≈1) while still being fast enough for many FFTs.
    if "height_sequence_1" in globals():
        base_n = max(1024, min(8192, len(globals()["height_sequence_1"])))
        # try a little larger than the existing one if possible
        n = max(base_n, 4096)
    else:
        n = 8192
    n = min(8192, n)   # hard cap

    conv_len = 2 * n - 1                # length of linear convolution a*a
    fft_len = _next_pow2(conv_len)       # for the main convolution
    fft_len2 = _next_pow2(conv_len + n - 1)   # for the gradient convolution

    # ------------------------------------------------------------------
    # 2. Helper to evaluate J = 2·n·max(conv(a,a)) (sum(a)=1 assumed)
    # ------------------------------------------------------------------
    def max_conv_of(vec: np.ndarray) -> float:
        """Return max(conv(vec,vec)) using FFT."""
        Af = np.fft.rfft(vec, fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[:conv_len]
        # small negative artefacts due to round‑off are ignored
        return float(np.max(conv))

    # ------------------------------------------------------------------
    # 3. Generate a diverse set of initial candidates and keep the best.
    # ------------------------------------------------------------------
    candidates: List[np.ndarray] = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous construction, resampled if necessary
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        old = np.clip(old, 0.0, None)
        if old.size != n:
            x_old = np.linspace(0.0, 1.0, old.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            old = np.interp(x_new, x_old, old)
        s = old.sum()
        if s == 0.0:
            old[:] = 1.0 / n
        else:
            old /= s
        candidates.append(old)

    # classic windows
    for win in (np.hanning(n), np.hamming(n), np.blackman(n), np.bartlett(n)):
        if win.sum() > 0.0:
            candidates.append(win / win.sum())

    # Tukey windows
    for alpha in np.linspace(0.0, 1.0, 7):
        w = tukey_window(n, float(alpha))
        if w.sum() > 0.0:
            candidates.append(w / w.sum())

    # random Dirichlet draws (different concentrations)
    for _ in range(12):
        conc = rng.uniform(0.2, 3.0)
        candidates.append(rng.dirichlet(np.full(n, conc)))

    # exponential edge blends (uniform + symmetric exponential)
    for _ in range(30):
        alpha = 10 ** rng.uniform(-2.0, 1.0)          # 0.01 … 10
        mix = rng.random()
        exp_vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
        exp_vec += np.exp(-alpha * (n - 1 - np.arange(n, dtype=np.float64)))
        exp_vec /= exp_vec.sum()
        vec = (1.0 - mix) * np.full(n, 1.0 / n) + mix * exp_vec
        vec /= vec.sum()
        candidates.append(vec)

    # power‑law shapes
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        idx = np.arange(1, n + 1, dtype=np.float64)
        v = 1.0 / (idx ** p)
        v /= v.sum()
        candidates.append(v)

    # keep the best among the portfolio
    best_vec = None
    best_J = np.inf
    for vec in candidates:
        maxc = max_conv_of(vec)
        J = 2.0 * n * maxc
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Continuous optimisation on the simplex.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # small safety margin
    cur_vec = best_vec.copy()
    last_improve = start_time
    iteration = 0

    # schedule parameters
    eta0 = 0.5            # initial step size multiplier
    gamma_start = 5.0    # start smoothing
    gamma_end = 4000.0   # final (near‑max) smoothing
    stagnation_sec = 30.0
    jitter_sigma = 0.03  # std. dev. for multiplicative log‑normal jitter

    total_time = deadline - start_time

    while time.monotonic() < deadline:
        # ---- compute convolution and current objective ----
        Af = np.fft.rfft(cur_vec, fft_len)
        conv = np.fft.irfft(Af * Af, n=fft_len)[:conv_len]
        maxc = float(conv.max())
        cur_J = 2.0 * n * maxc

        # ---- record global best ----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- handle stagnation (random jitter) ----
        if time.monotonic() - last_improve > stagnation_sec:
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cur_vec = best_vec * noise
            cur_vec /= cur_vec.sum()
            last_improve = time.monotonic()
            continue

        # ---- smoothing parameter γ (linear in elapsed time) ----
        elapsed = time.monotonic() - start_time
        gamma = gamma_start + (gamma_end - gamma_start) * (elapsed / total_time)

        # ---- compute normalized weights for the smoothed max ----
        shifted = conv - maxc                     # max element becomes 0
        weight_raw = np.exp(gamma * shifted)      # ≤ 1
        w_sum = weight_raw.sum()
        if w_sum == 0.0:
            # fallback: pure sub‑gradient from the active set
            active = np.where(conv >= maxc - 1e-12)[0]
            weight_norm = np.zeros_like(conv)
            weight_norm[active] = 1.0 / max(len(active), 1)
        else:
            weight_norm = weight_raw / w_sum

        # ---- gradient of the smoothed objective (FFT‑based) ----
        # grad_i = 2 * Σ_k weight_norm_k * cur_vec_{k-i}
        Af_rev = np.fft.rfft(cur_vec[::-1], fft_len2)
        Wf = np.fft.rfft(weight_norm, fft_len2)
        grad_full = np.fft.irfft(Wf * Af_rev, n=fft_len2)
        # slice the central part (indices n‑1 … n‑1+n‑1)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---- step size (1/√t) ----
        eta = eta0 / np.sqrt(iteration + 1.0)

        # ---- exponentiated‑gradient update (scaled by n) ----
        factor = np.exp(-eta * n * grad)
        # avoid numerical extremes
        factor = np.clip(factor, 1e-12, 1e12)
        cur_vec = cur_vec * factor
        cur_vec_sum = cur_vec.sum()
        if cur_vec_sum == 0.0:
            cur_vec[:] = 1.0 / n
        else:
            cur_vec /= cur_vec_sum

        # ---- occasional random mixing to keep diversity ----
        if rng.random() < 0.05:
            eps = rng.uniform(0.0, 0.10)
            rand_vec = rng.dirichlet(np.ones(n))
            cur_vec = (1.0 - eps) * cur_vec + eps * rand_vec
            cur_vec /= cur_vec.sum()

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best vector (clipped to allowed range and renormalised).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s
    return [float(v) for v in best_vec]
