Now function to solve LP:

```
def lp_optimize(f, M):
    n = len(f)
    conv_len = 2 * n - 1
    # build sparse matrix A_f
    offsets = np.arange(n)
    data = [f_i * np.ones(n) for f_i in f]  # each diagonal
    A = sp.diags(data, offsets, shape=(conv_len, n), format='csc')
    # bound constraints
    b_ub = np.full(conv_len, M)
    # objective to maximize sum(b) -> minimize -sum(b)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        b = res.x
        if b.sum() > 0:
            g = b / b.sum()
        else:
            g = np.full(n, 1.0/n)
        return g
    else:
        # fallback: return f unchanged
        return f
```

"""
Hybrid projected sub‑gradient + periodic LP refinement.

* Works on the probability simplex (∑a_i = 1, a_i ≥ 0) – the evaluation
  function J(a) = 2·n·max(conv(a,a)) is invariant under scaling.
* For a given a we compute the exact sub‑gradient of max(conv) by
  inspecting the rows where the convolution attains its maximum.
* A simple diminishing‑step projected gradient descent (with a Metropolis
  acceptance rule) drives the iterates downhill.
* Every ~30 iterations we solve a linear program:
        maximise   ∑ b_i
        subject to   (a * b)_k ≤ M   (M = current max‑convolution)
                     b_i ≥ 0
  This “LP boost’’ tries to spread the mass as much as possible without
  increasing the current worst convolution entry.
* When no improvement has been seen for a while a small random jitter
  around the current best solution restarts the walk, helping to escape
  plateaus.
* Multiple diverse starting points (uniform, Dirichlet, U‑shaped,
  inverted‑U, previous construction) are tried; the best vector found
  before the time budget expires is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_conv(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _max_conv_and_subgrad(a: np.ndarray, conv: np.ndarray):
    """Return (max(conv), sub‑gradient of max(conv) w.r.t. a)."""
    n = a.shape[0]
    M = conv.max()
    # rows where the maximum is attained (tiny tolerance)
    ks = np.where(conv >= M - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in ks:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i = np.arange(i_min, i_max + 1, dtype=np.int64)
        j = k - i
        neq = i != j
        # i ≠ j  → derivative contribution 2·a_j
        grad[i[neq]] += 2.0 * a[j[neq]]
        # i = j  → contribution a_i
        grad[i[~neq]] += a[i[~neq]]
    return M, grad


def _lp_step(f_vec: np.ndarray, M_bound: float) -> np.ndarray:
    """
    Linear‑programming boost.
    maximise   ∑ b_i
    s.t.       (f * b)_k ≤ M_bound   (for all k)
               b_i ≥ 0
    Returns the normalised solution g (∑g = 1).
    """
    n = f_vec.shape[0]
    conv_len = 2 * n - 1

    # Build the convolution matrix A_f : (conv_len × n) sparse
    offsets = np.arange(n)
    data = [np.full(n, f_vec[i], dtype=np.float64) for i in range(n)]
    A = sp.diags(data, offsets, shape=(conv_len, n), format='csc')

    b_ub = np.full(conv_len, M_bound, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(b) → minimise –sum(b)

    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})

    if res.success:
        b = res.x
        s = b.sum()
        if s > 0.0:
            return b / s
    # fallback – keep the current vector
    return f_vec.copy()
