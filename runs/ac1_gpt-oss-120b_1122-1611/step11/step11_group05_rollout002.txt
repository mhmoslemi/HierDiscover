    T_end = 1e-6

    while time.monotonic() < deadline:
        # compute conv and max and index

        cur_J = 2.0 * n * max_val
        # update best whatever

            last_improve = time.monotonic()
            print(f"[{time.monotonic()-start:6.2f}s] New best J={best_J:.6f}")

        # stagnation handle jitter
        if time.monotonic() - last_improve > 20.0:
            # jitter: random local moves
            cur = best_vec.copy()
            # add lognormal noise
            cur *= np.exp(rng.normal(0.0, 0.02, size=n))
            cur = proj_simplex(cur)
            last_improve = time.monotonic()
            continue

        # compute subgradient at kmax
        g = np.zeros(n, dtype=np.float64)
        # compute reversed slice temporarily
        # For indices i where 0 <= kmax - i < n
        low = max(0, kmax - (n - 1))
        high = min(n - 1, kmax)
        # i in [low, high]
        i_range = np.arange(low, high + 1)
        j_range = kmax - i_range
        g[i_range] = 2.0 * cur[j_range]
        if kmax % 2 == 0:
            mid = kmax // 2
            if 0 <= mid < n:
                g[mid] = cur[mid]  # correct diagonal coefficient

        # step size schedule
        step = 0.1 / np.sqrt(iteration + 1.0)  # or dynamic adapt
        proposal = cur - step * g
        proposal = np.clip(proposal, 0.0, None)
        proposal = proj_simplex(proposal)

        # evaluate proposal
        prop_conv = conv_fft(proposal)
        prop_J = 2.0 * n * prop_conv.max()

        # acceptance with temperature
        elapsed = time.monotonic() - start
        T = T_start * (1.0 - elapsed/(deadline-start)) + T_end * (elapsed/(deadline-start))
        if prop_J < cur_J or rng.random() < np.exp(-(prop_J - cur_J) / max(T, 1e-12)):
            cur = proposal
            cur_J = prop_J

        iteration += 1

    # final return best_vec as list of floats
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total
    return list(best_vec.astype(float))

"""
Sub‑gradient descent on the true max‑convolution objective.

The evaluation function is scale‑invariant, so the search works with a
probability vector (∑a_i = 1).  For such a vector  

    J(a) = 2·n·max( a * a )                     ( * = linear convolution )

is the exact value returned by ``evaluate_sequence``.

The algorithm keeps a single candidate and repeatedly:
  * evaluates the full convolution (FFT, O(n log n));
  * finds the index k* where the convolution attains its maximum;
  * builds a sub‑gradient of the max‑operator at k* (it is simply
    2·a reflected around k* with the diagonal entry halved);
  * takes a projected sub‑gradient step with a decreasing step‑size;
  * occasionally perturbs the best solution (jitter) to escape plateaus
    and accepts worse moves with a simple annealing schedule.

A pool of diverse initial vectors (uniform, geometric, ramps,
Gaussian, Dirichlet draws and a sparse “spike” mask) is tested and the
best one is taken as the start.  The whole search respects the 1000 s
budget and finally returns the best probability vector found,
clipped to the limits imposed by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (required by the platform)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a 1‑D array onto the probability simplex.
    Guarantees non‑negative entries that sum to one.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (the full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper – returns a*a (full linear convolution)."""
    n = a.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    return _conv_full(a, fft_len, conv_len)


def _max_conv_and_index(a: np.ndarray):
    """
    Compute the full convolution a*a, return (max_value, argmax_index).
    """
    conv = _full_convolution(a)
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
# Main optimisation routine – what the platform will call
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (a probability vector)
    that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # stay in a range where FFTs remain cheap
            n = max(1024, min(n, 16384))
        except Exception:
            n = 8192
    else:
        n = 8192                     # a generous size, still fast to FFT

    # --------------------------------------------------------------
    # 2. Build a varied portfolio of initial probability vectors
    # --------------------------------------------------------------
    inits = []

    # uniform
    inits.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.93, 0.95, 0.97, 0.99):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        inits.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    inits.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    inits.append(v)

    # Gaussian bells (various widths)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 5.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        inits.append(v)

    # Dirichlet draws (different concentrations)
    for alpha in (0.2, 0.5, 1.0):
        for _ in range(3):
            inits.append(rng.dirichlet(np.ones(n) * alpha))

    # Sparse masks – a few equally weighted spikes
    for k in (int(np.sqrt(n)), int(np.sqrt(n) * 2), max(2, n // 12)):
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        inits.append(mask)

    # Random Bernoulli mask (density 0.3)
    bern = rng.binomial(1, 0.3, size=n).astype(np.float64)
    if bern.sum() > 0:
        bern /= bern.sum()
        inits.append(bern)

    # If a previous construction exists, use it (normalised to a simplex)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v = v / s
        inits.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (true objective)
    # --------------------------------------------------------------
    def _objective_from_conv(max_conv):
        """Return the true evaluate_sequence value when the L1‑norm = 1."""
        return 2.0 * n * max_conv

    best_vec = None
    best_J = np.inf
    for cand in inits:
        max_conv, *_ = _max_conv_and_index(cand)
        J = _objective_from_conv(max_conv)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop – projected sub‑gradient descent
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # a small safety margin to guarantee we return before timeout
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time

    # annealing temperature (linear cooling)
    T_start = best_J * 0.15      # start relatively high
    T_end = 1e-8

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- evaluate current point -----
        max_conv, k_max, conv = _max_conv_and_index(cur_vec)
        cur_J = _objective_from_conv(max_conv)

        # best‑so‑far bookkeeping
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation handling -----
        if now - last_improve > 25.0:
            # random jitter around the best solution
            noise = np.exp(rng.normal(0.0, 0.025, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – re‑initialised")
            continue

        # ----- sub‑gradient of the max‑convolution -----
        # g_i = 2 * a_{k_max - i}  (if the index is in range)
        # diagonal entry (i == k_max-i) gets weight a_i instead of 2*a_i
        g = np.zeros_like(cur_vec)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        i_range = np.arange(low, high + 1, dtype=np.int64)
        j_range = k_max - i_range                       # guaranteed 0 … n‑1
        g[i_range] = 2.0 * cur_vec[j_range]
        if k_max % 2 == 0:
            mid = k_max // 2
            if 0 <= mid < n:
                g[mid] = cur_vec[mid]                   # correct diagonal coefficient

        # ----- step size schedule -----
        step = 0.12 / np.sqrt(iteration + 1.0)           # decays like 1/√t

        # ----- propose a new point -----
        proposal = cur_vec - step * g
        proposal = np.clip(proposal, 0.0, None)
        proposal = proj_simplex(proposal)

        # ----- evaluate proposal (exact max) -----
        prop_max_conv, _, _ = _max_conv_and_index(proposal)
        prop_J = _objective_from_conv(prop_max_conv)

        # ----- Metropolis acceptance (linear cooling) -----
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * (elapsed / (deadline - start_time)))
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # probability to accept a worse move
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best solution found
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # Convert to plain Python list as required by the evaluator
    return [float(v) for v in best_vec]
