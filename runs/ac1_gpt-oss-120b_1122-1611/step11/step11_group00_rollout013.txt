"""
Evolutionary optimisation of a non‑negative step function.
The objective is  J = 2·n·max(conv(a,a)) / (sum(a)^2) .
Because J is scale‑invariant we keep all candidates on the probability
simplex (∑a_i = 1).  The optimiser maintains a small population of
probability vectors, repeatedly mutates the best members (Gaussian noise
in log‑space), occasionally recombines two parents and inserts any child
that improves the worst member of the population.  The current best
solution is printed whenever it improves.  A simple annealing schedule
reduces the mutation strength over time.  The algorithm works for any
length n (chosen from the previous solution or a default) and respects
the 1000 s time budget.
"""

import time
import bisect
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (output length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _max_conv(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of a*a (full convolution)."""
    return float(_conv_full(a, fft_len, conv_len).max())


def _normalise(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative, normalise to sum 1 (protect against zero‑sum)."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Evolutionary search for a step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size (n)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
        except Exception:
            n = 4096
    else:
        n = 4096

    # enforce a reasonable range
    n = max(256, min(n, 8192))
    if n < 2048:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Helper functions for mutation / crossover
    # --------------------------------------------------------------
    eps = 1e-12

    def mutate(a: np.ndarray, sigma: float) -> np.ndarray:
        """Gaussian jitter in log‑softmax space."""
        logits = np.log(np.maximum(a, eps))
        logits += rng.normal(0.0, sigma, size=n)
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        return exp_logits / exp_logits.sum()

    def crossover(a1: np.ndarray, a2: np.ndarray) -> np.ndarray:
        """Average the log‑probabilities of two parents."""
        l1 = np.log(np.maximum(a1, eps))
        l2 = np.log(np.maximum(a2, eps))
        mixed = (l1 + l2) * 0.5
        max_log = np.max(mixed)
        exp_logits = np.exp(mixed - max_log)
        return exp_logits / exp_logits.sum()

    # --------------------------------------------------------------
    # 3. Initialise a diverse population (sorted by J)
    # --------------------------------------------------------------
    pop_size = 30
    population = []          # list of tuples (J, a)

    def add_to_pop(J, a):
        """Insert (J,a) keeping the list sorted and limited to pop_size."""
        bisect.insort(population, (J, a))
        if len(population) > pop_size:
            population.pop()               # drop the worst

    # a few handcrafted seeds ------------------------------------------------
    # uniform
    add_to_pop(2.0 * n * _max_conv(np.full(n, 1.0 / n), fft_len, conv_len),
                np.full(n, 1.0 / n))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        add_to_pop(2.0 * n * _max_conv(_normalise(v), fft_len, conv_len),
                    _normalise(v))

    # linear ramps
    v = np.arange(1, n + 1, dtype=np.float64)          # increasing
    add_to_pop(2.0 * n * _max_conv(_normalise(v), fft_len, conv_len),
                _normalise(v))
    v = np.arange(n, 0, -1, dtype=np.float64)         # decreasing
    add_to_pop(2.0 * n * _max_conv(_normalise(v), fft_len, conv_len),
                _normalise(v))

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        add_to_pop(2.0 * n * _max_conv(_normalise(v), fft_len, conv_len),
                    _normalise(v))

    # Dirichlet draws – moderate concentration
    for alpha in (0.5, 1.0, 2.0):
        for _ in range(4):
            v = rng.dirichlet(np.ones(n) * alpha)
            add_to_pop(2.0 * n * _max_conv(v, fft_len, conv_len), v)

    # previous construction, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 0.0)])
        add_to_pop(2.0 * n * _max_conv(_normalise(v), fft_len, conv_len),
                    _normalise(v))

    # If the list is still too small, pad with random Dirichlet samples
    while len(population) < pop_size:
        v = rng.dirichlet(np.ones(n))
        add_to_pop(2.0 * n * _max_conv(v, fft_len, conv_len), v)

    # --------------------------------------------------------------
    # 4. Evolution loop
    # --------------------------------------------------------------
    best_J, best_a = population[0]            # current global best
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin

    # schedule for mutation strength (annealing)
    sigma_start = 0.6
    sigma_end = 0.02

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time
        # linear annealing of sigma
        frac = elapsed / (deadline - start_time)
        sigma = sigma_start * (1.0 - frac) + sigma_end * frac
        # ------------------------------------------------------------------
        # 4.1  Mutate a few of the current best individuals
        # ------------------------------------------------------------------
        for parent_J, parent_a in population[:5]:          # top‑5
            child = mutate(parent_a, sigma)
            child_J = 2.0 * n * _max_conv(child, fft_len, conv_len)
            if child_J < best_J - 1e-12:
                best_J, best_a = child_J, child.copy()
                print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")
            if child_J < population[-1][0]:
                add_to_pop(child_J, child)

        # ------------------------------------------------------------------
        # 4.2  Crossover between random pairs
        # ------------------------------------------------------------------
        for _ in range(5):
            a1, a2 = rng.choice(population, size=2, replace=False)
            child = crossover(a1[1], a2[1])
            child = mutate(child, sigma * 0.5)           # slight extra noise
            child_J = 2.0 * n * _max_conv(child, fft_len, conv_len)
            if child_J < best_J - 1e-12:
                best_J, best_a = child_J, child.copy()
                print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")
            if child_J < population[-1][0]:
                add_to_pop(child_J, child)

        # ------------------------------------------------------------------
        # 4.3  Occasionally inject a completely fresh random specimen
        # ------------------------------------------------------------------
        if iteration % 200 == 0:
            fresh = rng.dirichlet(np.ones(n) * 0.5)
            fresh_J = 2.0 * n * _max_conv(fresh, fft_len, conv_len)
            if fresh_J < best_J - 1e-12:
                best_J, best_a = fresh_J, fresh.copy()
                print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")
            if fresh_J < population[-1][0]:
                add_to_pop(fresh_J, fresh)

        # ------------------------------------------------------------------
        # 4.4  Large‑scale jitter of the whole population (to shake plateaux)
        # ------------------------------------------------------------------
        if iteration % 500 == 0:
            for i in range(len(population)):
                J_i, a_i = population[i]
                jittered = mutate(a_i, sigma * 3.0)
                jitter_J = 2.0 * n * _max_conv(jittered, fft_len, conv_len)
                population[i] = (jitter_J, jittered)

    # --------------------------------------------------------------
    # 5. Return the best sequence (clip and re‑normalise for safety)
    # --------------------------------------------------------------
    result = np.clip(best_a, 0.0, 1000.0)
    total = result.sum()
    if total == 0.0:
        result = np.full_like(result, 1.0 / n)
    else:
        result = result / total

    return [float(v) for v in result]
