"""
Gradient‑descent on a smooth approximation of the max‑convolution objective.

* Works on the probability simplex (∑a = 1, a_i ≥ 0).  The true objective
  J(a) = 2·n·max(conv(a,a)) is homogeneous; fixing the sum to 1 is safe.
* The non‑smooth max is replaced by a temperature‑controlled soft‑max:
       S_β(a) = (1/β)·log∑_k exp(β·conv_k(a))
  whose gradient is cheap to compute via a single extra convolution.
* An exponential schedule raises the inverse temperature β from a small
  value (very smooth) up to a large value (≈ 200), turning the smooth
  objective into the true max without getting stuck early on.
* Projected (momentum‑accelerated) gradient steps keep the iterate on the
  simplex.  A simple Metropolis acceptance rule plus occasional random
  jitter lets the search escape shallow plateaus.
* The routine respects the 1000 s budget, prints intermediate best
  values and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex (sum=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all equal (or v is zero)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (full length) using a real FFT."""
    n = a.size
    m = b.size
    out_len = n + m - 1
    fft_len = _next_pow2(out_len)
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:out_len]
    # numerical noise may produce tiny negatives – clip them.
    return np.maximum(conv, 0.0)


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Exact objective J = 2·n·max(conv(a,a)) for a vector that already lives
    on the simplex (∑a = 1).  The value is the same that evaluate_sequence()
    would return up to the 2·n factor, because the denominator is 1².
    """
    conv = _conv_full_fft(a, a)[:conv_len]   # full conv length = 2n‑1
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm performs a smooth‑max gradient
    descent with an annealing schedule for the temperature β and returns the
    best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # Initial point – start from the known best if available, otherwise a
    # simple decreasing geometric shape (which is a good basin for this
    # problem).  Normalise to sum = 1.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= s
        a0 = proj_simplex(a0)               # ensure perfect simplex projection
    else:
        # geometric decay; the ratio is drawn randomly to provide diversity
        r = rng.uniform(0.90, 0.995)
        a0 = r ** np.arange(n, dtype=np.float64)
        a0 /= a0.sum()

    cur_a = a0.copy()
    cur_J = compute_J(cur_a, n, fft_len, conv_len)

    best_a = cur_a.copy()
    best_J = cur_J
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # Annealing / optimisation parameters
    # ------------------------------------------------------------------
    beta = 0.5               # start with a very smooth approximation
    beta_max = 200.0
    beta_mul = 1.015         # multiplicative increase per 10 iterations
    step0 = 0.12             # base step size
    momentum = np.zeros_like(cur_a)
    mom_beta = 0.85          # momentum decay
    iteration = 0
    last_improve = time.monotonic()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep 5 s safety margin

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1. Compute full convolution a*a
        # --------------------------------------------------------------
        conv = _conv_full_fft(cur_a, cur_a)[:conv_len]

        # --------------------------------------------------------------
        # 2. Soft‑max weights for the current temperature β
        #    w_i = exp(β·conv_i) / Σ exp(β·conv)
        # --------------------------------------------------------------
        max_c = conv.max()
        # subtract max for numerical stability (conv ≤ max_c)
        w_raw = np.exp(beta * (conv - max_c))
        w = w_raw / w_raw.sum()

        # --------------------------------------------------------------
        # 3. Gradient of the smooth max: g = Σ_i w_i·∂conv_i/∂a
        #    Using the identity ∂conv_i/∂a_k = 2·a_{i‑k} (if in‑range)
        #    ⇒ g = 2·(a * reverse(w)) restricted to length n
        # --------------------------------------------------------------
        w_rev = w[::-1]                     # reverse for the correlation
        grad_full = _conv_full_fft(cur_a, w_rev)          # length n+conv_len‑1
        # Slice the part that corresponds to indices 0 … n‑1
        start = conv_len - 1
        grad = 2.0 * grad_full[start:start + n]

        # --------------------------------------------------------------
        # 4. Projected (momentum) gradient step
        # --------------------------------------------------------------
        momentum = mom_beta * momentum + (1.0 - mom_beta) * grad
        step = step0 / np.sqrt(iteration + 1.0)
        prop = cur_a - step * momentum
        prop = proj_simplex(prop)

        # --------------------------------------------------------------
        # 5. Occasionally inject a small random jitter (exploration)
        # --------------------------------------------------------------
        if iteration % 2500 == 0 and iteration > 0:
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            prop = prop * jitter
            prop = proj_simplex(prop)

        # --------------------------------------------------------------
        # 6. Evaluate the true (non‑smoothed) objective
        # --------------------------------------------------------------
        prop_J = compute_J(prop, n, fft_len, conv_len)

        # --------------------------------------------------------------
        # 7. Metropolis acceptance (temperature driven by β)
        # --------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # Temperature shrinks as β grows; we map β∈[β₀,β_max] → T∈[1,0]
            T = max(1e-8, 1.0 - (beta - 0.5) / (beta_max - 0.5))
            if rng.random() < math.exp(-(prop_J - cur_J) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_a = prop
            cur_J = prop_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = cur_a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f} (β={beta:.2f})")

        # --------------------------------------------------------------
        # 8. Anneal β (make the approximation less smooth)
        # --------------------------------------------------------------
        if iteration % 10 == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # --------------------------------------------------------------
        # 9. Stagnation handling – if no improvement for ~15 s, jitter.
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > 15.0:
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_a = best_a * jitter
            cur_a = proj_simplex(cur_a)
            cur_J = compute_J(cur_a, n, fft_len, conv_len)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Jitter – best J still {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final sanitisation (clip to allowed range, renormalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
