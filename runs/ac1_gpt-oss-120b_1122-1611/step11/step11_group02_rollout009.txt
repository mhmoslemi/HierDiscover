def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    # Choose n
    if "height_sequence_1" in globals():
        n0 = len(globals()["height_sequence_1"])
        n = max(256, min(n0, 8192))
    else:
        n = 2048
    # Prepare FFT lengths
    conv_len = 2*n-1
    fft_len = _next_pow2(conv_len)
    
    # Helper functions: conv_full_fft (already defined), proj_simplex, gradient_max
    
    # Build init candidates (list of np arrays)
    init_cands = []
    # uniform
    init_cands.append(np.full(n, 1.0/n, dtype=np.float64))
    # geometric decays
    for r in [0.9, 0.95, 0.97, 0.99, 0.995]:
        v = r**np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)
    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n+1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    # dirichlet
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n)*alpha))
    # previous best or height_sequence_1
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        # clip and normalize
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0:
            v = np.full(n, 1.0/n, dtype=np.float64)
        else:
            v = v / s
        if v.size != n:
            if v.size > n: v=v[:n]
            else: v=np.concatenate([v, np.zeros(n-v.size)])
            v = proj_simplex(v)
        init_cands.append(v)

    # Evaluate each candidate quickly; find best
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = conv_full_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print("[0.00s] Init best J =", best_J)
    
    # Main subgradient descent loop
    start = time.monotonic()
    deadline = start + float(budget_s) - 5.0
    eta0 = 0.5
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_iter_vec = cur_vec.copy()
    best_iter_J = best_J
    last_improve = start
    iter_cnt = 0
    
    while time.monotonic() < deadline:
        # compute convolution and objective
        conv = conv_full_fft(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        k_max = conv.argmax()
        cur_J = float(2.0 * n * max_conv)
        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic()-start:.2f}s] New best J = {best_J:.6f}")
        # stagnation detection
        if time.monotonic() - last_improve > 30.0:
            # apply jitter
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = proj_simplex(best_vec * noise)
            last_improve = time.monotonic()
            print(f"[{time.monotonic()-start:.2f}s] Jitter restart, best J = {best_J:.6f}")
        # compute subgradient at max index
        idx = np.arange(n)
        src = k_max - idx
        mask = (src >= 0) & (src < n)
        grad = np.zeros_like(cur_vec)
        grad[mask] = 2.0 * cur_vec[src[mask]]
        # step size
        eta = eta0 / np.sqrt(iter_cnt + 1.0)

        max_abs = np.max(np.abs(grad))
        if max_abs * eta > 30.0:

        # Euclidean step
        proposal = cur_vec - eta * grad
        proposal = proj_simplex(proposal)
        # evaluate proposal
        prop_conv = conv_full_fft(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())
        # accept if better
        if prop_J < cur_J:
            cur_vec = proposal
            cur_J = prop_J
        else:
            # occasional random acceptance
            if rng.random() < 1e-7:

        iter_cnt += 1

    # final fine random perturbations
    fin_start = time.monotonic()
    fin_deadline = fin_start + min(30.0, (deadline - start) * 0.05)
    while time.monotonic() < fin_deadline:

        if cand_J < best_J:

    # final normalization ensure sum approx 1

    total = best_vec.sum()
    if total == 0:

    else:

    print("Final best J =", best_J)
    return [float(v) for v in best_vec]

"""
Minimal‑max‑convolution optimiser.

We minimise
        J(a) = 2·n·max(conv(a,a)) / (∑a)²
on the probability simplex (the objective is scale‑invariant, so we keep
∑a = 1).  The algorithm is a simple (projected) sub‑gradient / mirror‑descent
loop that always uses the *true* active index of the convolution (hard max):
the gradient of max_k conv_k(a) w.r.t. a is obtained by reflecting a about the
max‑index.  At every iteration we try two updates (Euclidean and
exponentiated‑gradient) and keep the better one.  When progress stalls for
30 s we apply a small multiplicative jitter to escape plateaus.  A short
fine‑grained random‑search phase finishes the run.  The routine respects the
global 1000 s budget and returns the best sequence it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013)."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns only the first ``conv_len`` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered within
    the allotted time budget (≈ 1000 s).  The routine does not import
    ``evaluate_sequence``; the caller will evaluate the returned list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size ------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # length for a*a

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial simplex vectors ------------------
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderate & concentrated)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            v = proj_simplex(v)
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate ----------------------------------
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = conv_full_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (hard‑max sub‑gradient + mirror step) -----
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    eta0 = 0.5               # base step size (decays as 1/√t)
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        # ----- 4.1 current convolution and active index -------------------
        conv = conv_full_fft(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        k_max = int(conv.argmax())
        cur_J = float(2.0 * n * max_conv)

        # ----- 4.2 record global best ------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- 4.3 stagnation handling (jitter after 30 s) ---------------
        now = time.monotonic()
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = proj_simplex(best_vec * noise)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ----- 4.4 sub‑gradient of the hard max --------------------------
        #    g_i = 2 * a_{k_max - i}  (zero when the reflected index is out of range)
        idx = np.arange(n)
        src = k_max - idx
        mask = (src >= 0) & (src < n)
        grad = np.zeros_like(cur_vec)
        grad[mask] = 2.0 * cur_vec[src[mask]]

        # ----- 4.5 step‑size (η ∝ 1/√t) ---------------------------------
        eta = eta0 / np.sqrt(iteration + 1.0)
        # protect against overflow in the exponentiated update
        max_abs = np.max(np.abs(grad))
        if max_abs * eta > 30.0:
            eta = 30.0 / max_abs

        # ----- 4.6 two candidate updates ---------------------------------
        # Euclidean (projected gradient)
        cand_euc = proj_simplex(cur_vec - eta * grad)

        # Exponentiated‑gradient (mirror descent with KL divergence)
        factor = np.exp(-eta * grad)
        cand_exp = proj_simplex(cur_vec * factor)

        # ----- 4.7 evaluate both proposals -------------------------------
        conv_euc = conv_full_fft(cand_euc, fft_len, conv_len)
        J_euc = float(2.0 * n * conv_euc.max())

        conv_exp = conv_full_fft(cand_exp, fft_len, conv_len)
        J_exp = float(2.0 * n * conv_exp.max())

        # ----- 4.8 accept the best improvement ---------------------------
        if J_euc < cur_J or J_exp < cur_J:
            # pick the better of the two
            if J_euc <= J_exp:
                cur_vec = cand_euc
                cur_J = J_euc
            else:
                cur_vec = cand_exp
                cur_J = J_exp
        else:
            # occasional tiny random acceptance to escape tiny plateaus
            if rng.random() < 1e-7:
                cur_vec = cand_euc

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Fine‑grained random search around the best solution ---------------
    # ------------------------------------------------------------------
    fin_start = time.monotonic()
    # spend up to 5 % of the total time (or 30 s) on the final refinement
    fin_deadline = fin_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fin_deadline:
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        conv = conv_full_fft(cand, fft_len, conv_len)
        cand_J = float(2.0 * n * conv.max())
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine] Improved J = {best_J:.6f} at {elapsed:6.2f}s")

    # ------------------------------------------------------------------
    # 6. Normalise & return ------------------------------------------------
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"Final best J = {best_J:.6f} (length = {n})")
    return [float(v) for v in best_vec]
