"""
Hybrid simulated‑annealing / exponentiated‑gradient optimizer.

The objective (up to the constant factor 2·n) is
    J(a) = max(conv(a,a))            where a ≥ 0,  Σ a_i = 1.
The code works on the probability simplex (Σ a_i = 1) and
searches for a vector a that minimises J.  It combines three
move types:

* exponentiated‑gradient step using a sub‑gradient of the max‑convolution,
* random “mass‑transfer’’ moves that relocate a small fraction of probability,
* occasional smoothing / jitter moves to escape plateaus.

A simple exponential cooling schedule controls Metropolis acceptance.
Multiple diverse initialisations (uniform, geometric, ramps,
Gaussian, Dirichlet draws, sparse spikes) are tried, and the best ever
found is returned.  All vectors are kept non‑negative, renormalised
after each proposal, and finally clipped to the limits accepted by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int):
    """
    Returns (J, conv, max_conv) where
        J = 2·n·max(conv)   (the factor 2·n is part of the evaluation),
        conv = a*a (full convolution),
        max_conv = max(conv).
    The vector a must be normalised (Σ a = 1).
    """
    conv = _conv_full(a, fft_len, conv_len)
    max_conv = conv.max()
    J = 2.0 * n * max_conv
    return J, conv, max_conv


def _subgradient(a: np.ndarray, conv: np.ndarray, max_conv: float,
                 rel_tol: float):
    """
    Sub‑gradient of max(conv) at a.
    active = indices where conv is within rel_tol·max_conv of the peak.
    Returns (grad, active_indices).
    """
    n = a.shape[0]
    tol = max(rel_tol * max_conv, 1e-14)
    active = np.where(conv >= max_conv - tol)[0]

    grad = np.zeros_like(a)
    for p in active:
        # contributions from a_i * a_{p-i}
        start = p - (n - 1)
        if start < 0:
            start = 0
        end = p if p < n else n - 1
        idx = np.arange(start, end + 1)
        grad[idx] += 2.0 * a[p - idx]          # derivative of a_i * a_{p-i}
    return grad, active


def _expgrad_step(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """Multiplicative (exponentiated‑gradient) update and re‑normalisation."""
    factor = np.exp(-step * grad)
    proposal = a * factor
    s = proposal.sum()
    if s == 0.0:
        proposal[:] = 1.0 / a.shape[0]
    else:
        proposal /= s
    return proposal


def _mass_transfer_step(a: np.ndarray, grad: np.ndarray,
                       rng: np.random.Generator) -> np.ndarray:
    """Move a small fraction of probability from a high‑gradient index
    to a low‑gradient one."""
    n = a.shape[0]
    if grad.sum() > 0.0:
        donor = rng.choice(n, p=grad / grad.sum())
    else:
        donor = rng.integers(n)

    # Prefer low‑gradient indices as receivers
    inv = grad.max() - grad + 1e-12
    receiver = rng.choice(n, p=inv / inv.sum())

    # Transfer between 1 % and 10 % of the donor’s mass
    delta_frac = rng.uniform(0.01, 0.10)
    delta = min(a[donor], delta_frac * a[donor])

    proposal = a.copy()
    proposal[donor] -= delta
    proposal[receiver] += delta
    # sum unchanged → no renormalisation needed
    return proposal


def _smooth_step(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a tiny 3‑point smoothing filter and blend."""
    kernel = np.array([0.25, 0.5, 0.25])
    smooth = np.convolve(a, kernel, mode='same')
    alpha = rng.uniform(0.05, 0.20)          # strength of smoothing
    proposal = (1.0 - alpha) * a + alpha * smooth
    proposal = np.clip(proposal, 0.0, None)
    proposal /= proposal.sum()
    return proposal


def _jitter_step(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative Gaussian jitter around the current vector."""
    noise = np.exp(rng.normal(0.0, 0.20, size=a.shape))
    proposal = a * noise
    proposal = np.clip(proposal, 0.0, None)
    proposal /= proposal.sum()
    return proposal


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a list of floats).

    The method is a hybrid of simulated annealing, exponentiated‑gradient,
    and random mass‑transfer moves with occasional jitter/re‑initialisation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size (try to reuse a previous solution if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            v0 = np.asarray(height_sequence_1, dtype=np.float64)
            v0 = np.clip(v0, 0.0, None)
            s0 = v0.sum()
            if s0 == 0.0:
                v0[:] = 1.0 / v0.size
            else:
                v0 /= s0
            n = max(256, min(v0.size, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors.
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (ascending & descending)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells (different widths)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws with various concentration parameters
    alphas = (0.01, 0.05, 0.10, 0.30, 0.60, 1.00)
    for alpha in alphas:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Very sparse random spikes (≈5 % non‑zero entries)
    k_spike = max(2, int(0.05 * n))
    for _ in range(5):
        spikes = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=k_spike, replace=False)
        spikes[idx] = rng.random(k_spike) + 0.1   # avoid exact zeros
        spikes /= spikes.sum()
        init_cands.append(spikes)

    # reuse the previous construction if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        if v.shape[0] != n:
            # If the old vector has different length, resample it to length n
            # by simple linear interpolation.
            x_old = np.linspace(0, 1, v.shape[0])
            x_new = np.linspace(0, 1, n)
            v = np.interp(x_new, x_old, v)
            v /= v.sum()
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best of the initial candidates.
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J, _, _ = _objective(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (simulated annealing + gradient moves)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # keep a small safety margin for the final normalisation & return
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # temperature schedule (exponential cooling)
    T_start = best_J * 0.15          # start a fraction of the current best
    T_end = 1e-8

    cur_vec = best_vec.copy()
    cur_J, cur_conv, cur_max = _objective(cur_vec, n, fft_len, conv_len)

    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- keep track of the best solution so far ----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- stagnation handling (jitter) ----
        if now - last_improve > 20.0:
            cur_vec = _jitter_step(best_vec, rng)
            cur_J, cur_conv, cur_max = _objective(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ---- dynamic active‑set tolerance ----
        if iteration < 3000:
            rel_tol = 0.02          # 2 % early on
        elif iteration < 15000:
            rel_tol = 0.01
        else:
            rel_tol = 0.004

        # ---- compute sub‑gradient of the max convolution ----
        grad, active = _subgradient(cur_vec, cur_conv, cur_max, rel_tol)

        # ---- propose a new candidate (randomly pick a move type) ----
        move_type = rng.random()
        if move_type < 0.55:
            # exponentiated‑gradient step
            step = 0.6 / np.sqrt(iteration + 1.0)
            proposal = _expgrad_step(cur_vec, grad, step)
        elif move_type < 0.80:
            # random mass‑transfer move
            proposal = _mass_transfer_step(cur_vec, grad, rng)
        else:
            # mild smoothing step
            proposal = _smooth_step(cur_vec, rng)

        # ---- evaluate the proposal ----
        prop_J, prop_conv, prop_max = _objective(proposal, n, fft_len, conv_len)

        # ---- Metropolis acceptance with exponential cooling ----
        # Linear interpolation between T_start and T_end
        total_time = deadline - start_time
        frac = min(1.0, elapsed / total_time)
        T = max(T_end, T_start * (1.0 - frac) + T_end * frac)

        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal
            cur_J = prop_J
            cur_conv = prop_conv
            cur_max = prop_max

        # ----------------------------------------------------------------
        iteration += 1
        # occasional cheap status update (every ~5000 iterations)
        if iteration % 5000 == 0:
            print(f"[{elapsed:6.2f}s] iter {iteration:6d} cur J = {cur_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best solution (clip & re‑normalise for safety)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # final sanity check (optional – can be removed for speed)
    # print(f"Final best J = {best_J:.6f}")

    return [float(v) for v in best_vec]
