"""
Hybrid simulated‑annealing + projected sub‑gradient optimiser.

* Works on the simplex (∑a = 1, a_i ≥ 0) – the objective
  J(a) = 2·n·max(conv(a,a)) is invariant under scaling.
* The algorithm alternates between
    – a momentum‑guided projected sub‑gradient step,
    – a gradient‑guided exchange move (mass transfer from a high‑gradient
      index to a low‑gradient one).
* An exponential cooling schedule controls a Metropolis acceptance rule;
  the temperature starts at the current best J and decays to 1e‑8.
* If no improvement is observed for 30 s we inject a strong multiplicative
  jitter centred on the best solution to escape plateaus.
* After the time budget a short deterministic polishing phase performs a few
  extra gradient steps.
* The search runs on a relatively large problem size (up to 8192 entries)
  – larger n gives a finer‑grained step function and a better chance of
  lowering the bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or v is zero); return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Return the objective J = 2·n·max(conv(a,a)) for a vector a that
    already lives on the simplex (sum(a) == 1).
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgradient_active_set(a: np.ndarray,
                           conv: np.ndarray,
                           active_tol: float) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) at the current active set.
    The result is a vector of the same length as a.
    """
    n = a.size
    max_conv = conv.max()
    active = np.where(conv >= max_conv - active_tol)[0]

    grad = np.zeros_like(a)
    # each active position contributes a shifted copy of a
    for p in active:
        start = max(0, p - (n - 1))
        end = min(p, n - 1)
        idx = np.arange(start, end + 1)
        # contribution 2 * a_{p‑idx}
        grad[idx] += 2.0 * a[p - idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget,
    returns the best sequence found and prints progress information.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – try to use a fairly large vector (up to 8192).
    # ------------------------------------------------------------------
    max_n = 8192
    if "height_sequence_1" in globals():
        try:
            prev_len = len(globals()["height_sequence_1"])
            n = min(max_n, max(256, int(prev_len * 1.5)))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Assemble a diverse pool of initial simplex points.
    # ------------------------------------------------------------------
    init_vectors = []

    # uniform
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_vectors.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_vectors.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_vectors.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_vectors.append(v)

    # Dirichlet draws – moderate concentration
    for _ in range(5):
        init_vectors.append(rng.dirichlet(np.ones(n)))

    # previous best (if any) – padded to the new size at a random offset
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s > 0.0:
            prev /= s
        else:
            prev = np.full_like(prev, 1.0 / prev.size)
        big = np.zeros(n, dtype=np.float64)
        start = rng.integers(0, n - prev.size + 1)
        big[start:start + prev.size] = prev
        if big.sum() == 0.0:
            big[:] = 1.0 / n
        else:
            big /= big.sum()
        init_vectors.append(big)

    # ------------------------------------------------------------------
    # 3. Select the best initial candidate (cheap J evaluation).
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for vec in init_vectors:
        J = compute_J(vec, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation loop (gradient + guided exchange + SA).
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    iteration = 0
    last_improve = start_time
    momentum = np.zeros_like(cur_vec)
    beta = 0.85                                 # momentum decay factor
    T_start = best_J
    T_end = 1e-8
    total_time = deadline - start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # exponential cooling
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # -----------------------------------------------------------------
        # 4.1 Stagnation handling – jitter after 30 s without improvement
        # -----------------------------------------------------------------
        if now - last_improve > 30.0:
            sigma = 0.4
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = compute_J(cur_vec, n, fft_len, conv_len)
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter reset – cur J = {cur_J:.6f}")

        # -----------------------------------------------------------------
        # 4.2 Choose a move type
        # -----------------------------------------------------------------
        if rng.random() < 0.6:                       # projected sub‑gradient step
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            max_conv = cur_conv.max()
            tol = max(1e-12, max_conv * 1e-12)
            grad = subgradient_active_set(cur_vec, cur_conv, tol)

            # momentum update
            momentum = beta * momentum + (1.0 - beta) * grad

            # step size decays like 1/√t
            step = 0.7 / np.sqrt(iteration + 1.0)
            proposal = cur_vec - step * momentum
        else:                                         # gradient‑guided exchange
            # (re‑use the most recent gradient if it is fresh enough)
            if iteration % 2 == 0:
                cur_conv = _conv_full(cur_vec, fft_len, conv_len)
                max_conv = cur_conv.max()
                tol = max(1e-12, max_conv * 1e-12)
                grad = subgradient_active_set(cur_vec, cur_conv, tol)

            # source index – weighted towards large gradient entries
            prob = grad.copy()
            if prob.sum() > 0:
                prob /= prob.sum()
                i = rng.choice(n, p=prob)
            else:
                i = rng.integers(0, n)

            # destination index – preferably a low‑gradient position
            low_thresh = np.median(grad) if grad.size > 0 else 0.0
            attempts = 0
            while True:
                j = rng.integers(0, n)
                if j != i and grad[j] <= low_thresh:
                    break
                attempts += 1
                if attempts > 5:
                    j = (i + rng.integers(1, n)) % n
                    break

            max_transfer = cur_vec[i] * 0.20               # up to 20 % of source mass
            delta = rng.random() * max_transfer
            proposal = cur_vec.copy()
            proposal[i] -= delta
            proposal[j] += delta

        # -----------------------------------------------------------------
        # 4.3 Project onto simplex and evaluate.
        # -----------------------------------------------------------------
        proposal = proj_simplex(proposal)
        prop_J = compute_J(proposal, n, fft_len, conv_len)

        # -----------------------------------------------------------------
        # 4.4 Metropolis acceptance.
        # -----------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final polishing – a few deterministic gradient steps.
    # ------------------------------------------------------------------
    for polish_iter in range(30):
        cur_conv = _conv_full(best_vec, fft_len, conv_len)
        max_conv = cur_conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        grad = subgradient_active_set(best_vec, cur_conv, tol)
        step = 0.2 / np.sqrt(polish_iter + 1.0)
        cand = proj_simplex(best_vec - step * grad)
        cand_J = compute_J(cand, n, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand.copy()
        else:
            break

    # ------------------------------------------------------------------
    # 6. Sanitise and return the result.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"Finished – best J = {best_J:.6f} – length = {n}")
    return [float(v) for v in best_vec]
