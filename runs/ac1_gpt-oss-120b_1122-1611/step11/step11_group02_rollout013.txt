Hybrid exponentiated‑gradient + Adam + simulated‑annealing optimiser.
Starts from a diverse portfolio (including previous best) and iteratively
flattens the convolution using a smoothed soft‑max surrogate of the
sup‑norm.  A per‑coordinate Adam step is applied inside a multiplicative
update (exponentiated‑gradient) and the temperature of the soft‑max is
annealed from p≈1 to p≈300.  When progress stalls we inject a targeted
mass‑redistribution move that directly attacks the currently largest
convolution peak.  After the main phase a short fine‑grained random
search around the best vector is performed.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all equal fallback
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]

def gradient_smooth(a: np.ndarray, conv: np.ndarray,
                    p: float, fft_len_g: int, conv_len: int) -> np.ndarray:
    """
    Gradient of the smoothed max φ(a) = log‑sum‑exp(p·conv)/p.
    Returns ∂φ/∂a as (2 * (w * a_rev))[central part].
    """
    w = np.exp(p * (conv - conv.max()))          # soft‑max weights, max weight =1
    a_rev = a[::-1]
    # Convolution of w with a_rev
    # length of w = conv_len, length of a_rev = a.size
    G = np.fft.irfft(np.fft.rfft(w, fft_len_g) *
                     np.fft.rfft(a_rev, fft_len_g),
                     n=fft_len_g)
    # required slice corresponds to indices (conv_len-1) … (conv_len-1 + n-1)
    grad = 2.0 * G[conv_len - 1: conv_len - 1 + a.size]
    return grad

def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return the evaluation function value, assuming a sums to 1."""
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

    The routine runs for roughly ``budget_s`` seconds and returns the best

    exponentiated‑gradient/Adam updates with a soft‑max surrogate, a

        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    v /= v.sum()
    init_cands.append(v)

    v /= v.sum()
    init_cands.append(v)

        v /= v.sum()
        init_cands.append(v)

        v = np.zeros(n, dtype=np.float64)

        init_cands.append(v)

        init_cands.append(prev)

    # 3. Pick the best initial candidate ------------------------------------
    # ------------------------------------------------------------------

    best_J = np.inf
    for cand in init_cands:

    # ------------------------------------------------------------------

    beta2 = 0.999
    eps = 1e-8

    p_end = 300.0

    cur_J = best_J
    best_vec_global = cur.copy()
    best_J_global = best_J

        maxc = conv.max()

            best_J_global = cur_J
            best_vec_global = cur.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J_global:.6f}")

        # ------------------------------------------------------------------

            print(f"[{elapsed:6.2f}s] Jitter – restart around best J = {best_J_global:.6f}")

        # 4.2 Compute smoothed gradient of the sup‑norm --------------------
        # ------------------------------------------------------------------
        grad = gradient_smooth(cur, conv, p, fft_len_g, conv_len)

        # ------------------------------------------------------------------
        # 4.3 Adam moments update -------------------------------------------
        # ------------------------------------------------------------------

        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v_adam / (1.0 - beta2 ** (iteration + 1))
        # per‑coordinate pre‑conditioned gradient
        scaled_grad = m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------------
        # 4.4 Adaptive multiplicative update (exponentiated‑Adam) ------------
        # ------------------------------------------------------------------
        # Clip the exponent to avoid overflow/underflow
        exponent = - eta0 * scaled_grad / np.sqrt(iteration + 1.0)
        exponent = np.clip(exponent, -30.0, 30.0)
        factor = np.exp(exponent)
        proposal = cur * factor
        proposal = proj_simplex(proposal)

        # ------------------------------------------------------------------
        # 4.5 Metropolis acceptance -------------------------------------------
        # ------------------------------------------------------------------
        prop_J = compute_J(proposal, n, fft_len, conv_len)
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # linear cooling temperature for occasional uphill moves
            T = max(0.001, 0.05 * (1.0 - frac))
            if rng.random() < np.exp(-(prop_J - cur_J) / T):
                accept = True
        if accept:
            cur = proposal
            cur_J = prop_J

        # ------------------------------------------------------------------
        # 4.6 Targeted pair‑wise redistribution when stuck ------------------
        # ------------------------------------------------------------------
        if now - last_improve > 40.0:
            # Identify the index of the largest convolution entry
            k = int(np.argmax(conv))

            j_range = k - i_range
            contributions = cur[i_range] * cur[j_range]
            idx = int(np.argmax(contributions))
            i_star = i_range[idx]
            j_star = j_range[idx]

                new_vec[j_star] -= delta

                rand_idx = rng.choice(n, size=4, replace=False)
                new_vec[rand_idx] += delta / 4.0    # keep the sum unchanged
                new_vec = proj_simplex(new_vec)
                cur = new_vec
                # reset stagnation timer
                last_improve = now

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random search around the best solution ----------
    # ------------------------------------------------------------------
    fin_start = time.monotonic()

    # ------------------------------------------------------------------

    total = best_vec_global.sum()
    if total == 0.0:

    else:
        best_vec_global /= total

"""
Hybrid exponentiated‑gradient / Adam optimiser with a soft‑max surrogate,
periodic target‑peak redistribution and a final fine‑grained random
refinement.  The routine builds a diverse set of initial probability
vectors (including the previous best if available), then repeatedly
flattens the auto‑convolution a*a while staying on the simplex.
It runs for roughly ``budget_s`` seconds and returns the best sequence
found (as a plain ``list[float]``) which can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first ``conv_len`` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def gradient_smooth(a: np.ndarray, conv: np.ndarray,
                    p: float, fft_len_g: int, conv_len: int) -> np.ndarray:
    """
    Gradient of the smoothed sup‑norm
        φ(a) = log‑sum‑exp(p·conv(a,a)) / p .
    Returns ∂φ/∂a = 2·(w * a_rev) evaluated on the central slice.
    """
    w = np.exp(p * (conv - conv.max()))          # soft‑max weights, max=1
    a_rev = a[::-1]
    G = np.fft.irfft(np.fft.rfft(w, fft_len_g) *
                     np.fft.rfft(a_rev, fft_len_g),
                     n=fft_len_g)
    # slice corresponding to the original length of ``a``
    grad = 2.0 * G[conv_len - 1: conv_len - 1 + a.size]
    return grad


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Evaluation function (sum(a) == 1 after projection).
    Equivalent to ``evaluate_sequence`` up to the invariant scaling.
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.  The routine mixes exponentiated‑Adam
    updates with a soft‑max surrogate of the sup‑norm, a targeted
    redistribution move that attacks the current peak, and a final
    random refinement phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    max_n = 8192
    if "height_sequence_1" in globals():
        try:
            prev_len = len(globals()["height_sequence_1"])
            n = min(max(prev_len * 2, 4096), max_n)
            n = max(2048, n)
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for a*a
    fft_len_g = _next_pow2(conv_len + n - 1)           # for gradient (w*a_rev)

    # --------------------------------------------------------------
    # 2. Build a diverse portfolio of initial probability vectors
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderate & concentrated)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Sparse random supports (K ≈ sqrt(n))
    K = max(2, int(np.sqrt(n)))
    for _ in range(3):
        idx = rng.choice(n, size=K, replace=False)
        v = np.zeros(n, dtype=np.float64)
        v[idx] = rng.random(K)
        v /= v.sum()
        init_cands.append(v)

    # previous best (if any), padded / truncated to current length
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s == 0.0:
            prev[:] = 1.0 / prev.size
        else:
            prev /= s
        if prev.size != n:
            if prev.size > n:
                prev = prev[:n]
            else:
                pad = np.zeros(n - prev.size, dtype=np.float64)
                prev = np.concatenate([prev, pad])
            prev = proj_simplex(prev)
        init_cands.append(prev)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        cand = proj_simplex(cand)
        cur_J = compute_J(cand, n, fft_len, conv_len)
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop (Adam + exponentiated‑gradient)
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # safety margin

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.5                # base learning‑rate for the multiplicative step
    p_start = 1.0
    p_end = 500.0             # final soft‑max sharpness

    cur = best_vec.copy()
    cur_J = best_J
    best_global = cur.copy()
    best_global_J = best_J

    m = np.zeros(n, dtype=np.float64)          # first ADAM moment
    v_adam = np.zeros(n, dtype=np.float64)    # second ADAM moment

    iteration = 0
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        frac = elapsed / total

        # ---- anneal soft‑max temperature ---------------------------------
        p = p_start + (p_end - p_start) * frac

        # ---- current convolution & objective -------------------------------
        conv = conv_full_fft(cur, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())   # sum(cur) == 1 after projection

        # ---- track global best --------------------------------------------
        if cur_J < best_global_J - 1e-12:
            best_global_J = cur_J
            best_global = cur.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_global_J:.6f}")

        # ---- stagnation handling – jitter ---------------------------------
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur = proj_simplex(best_global * noise)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – restart around best J = {best_global_J:.6f}")

        # ---- smoothed gradient --------------------------------------------
        grad = gradient_smooth(cur, conv, p, fft_len_g, conv_len)

        # ---- ADAM moment updates -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v_adam / (1.0 - beta2 ** (iteration + 1))
        scaled_grad = m_hat / (np.sqrt(v_hat) + eps)

        # ---- exponentiated‑Adam proposal -----------------------------------
        exponent = - eta0 * scaled_grad / np.sqrt(iteration + 1.0)
        exponent = np.clip(exponent, -30.0, 30.0)   # avoid overflow
        factor = np.exp(exponent)
        proposal = cur * factor
        proposal = proj_simplex(proposal)

        # ---- Metropolis acceptance -----------------------------------------
        prop_J = compute_J(proposal, n, fft_len, conv_len)
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # linear cooling temperature for rare uphill moves
            T = max(0.001, 0.05 * (1.0 - frac))
            if rng.random() < np.exp(-(prop_J - cur_J) / T):
                accept = True
        if accept:
            cur = proposal
            cur_J = prop_J

        # ---- targeted pair‑wise redistribution when stuck -----------------
        if now - last_improve > 40.0:
            k = int(np.argmax(conv))                  # index of current peak
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
            j_range = k - i_range
            contributions = cur[i_range] * cur[j_range]
            idx = int(np.argmax(contributions))
            i_star = i_range[idx]
            j_star = j_range[idx]
            delta = 0.02 * min(cur[i_star], cur[j_star])   # 2 % of the smaller weight
            if delta > 0.0:
                new_vec = cur.copy()
                new_vec[i_star] -= delta
                new_vec[j_star] -= delta
                # redistribute uniformly over a few random locations
                rand_idx = rng.choice(n, size=4, replace=False)
                new_vec[rand_idx] += delta / 4.0
                cur = proj_simplex(new_vec)
                last_improve = now

        iteration += 1

    # --------------------------------------------------------------
    # 5. Final fine‑grained random search around the best solution
    # --------------------------------------------------------------
    fin_start = time.monotonic()
    fin_deadline = fin_start + min(30.0, (deadline - start) * 0.07)

    while time.monotonic() < fin_deadline:
        noise = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = proj_simplex(best_global * noise)
        cand_J = compute_J(cand, n, fft_len, conv_len)
        if cand_J < best_global_J:
            best_global_J = cand_J
            best_global = cand
            print(f"[fine] Improved J = {best_global_J:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best sequence (already on the simplex)
    # --------------------------------------------------------------
    best_global = np.clip(best_global, 0.0, 1000.0)
    total = best_global.sum()
    if total == 0.0:
        best_global[:] = 1.0 / n
    else:
        best_global /= total

    return [float(v) for v in best_global]
