import time, numpy as np
from scipy.fft import rfft, irfft

def _next_pow2(x): ...
def proj_simplex(v): same as before.

def compute_conv(a, fft_len, conv_len):
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

def evaluate_J(a, fft_len, conv_len, n):
    conv = compute_conv(a, fft_len, conv_len)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)

def candidate_geometric(n, r):
    a = np.power(r, np.arange(n, dtype=np.float64))
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0/n)
    return a / a_sum

def candidate_powerlaw(n, p):
    # a_i = (i+1)^(-p)
    idx = np.arange(1, n+1, dtype=np.float64)
    a = np.power(idx, -p)
    a_sum = a.sum()
    return a / a_sum

def candidate_cosine(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2*np.pi * x / (n-1)))  # Hann window maybe
    a_sum = a.sum()
    return a / a_sum

def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
        except Exception:
            n = 4096
    else:
        n = 4096
    # maybe ensure n >= 256
    n = max(256, n)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # Build initial candidates list
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws (20 candidates)
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            cand = rng.dirichlet(np.full(n, alpha))
            candidates.append(cand)

    # geometric family
    for r in np.linspace(0.4, 0.99, 12):
        candidates.append(candidate_geometric(n, r))

    # power law
    for p in np.linspace(0.1, 2.5, 12):
        candidates.append(candidate_powerlaw(n, p))

    # cosine / Hann window
    candidates.append(candidate_cosine(n))

    # previous construction if any
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0:
            v = v / s
        else:
            v = np.full_like(v, 1.0 / n)
        candidates.append(v)

    # Evaluate each candidate using fast compute (2n * max conv)
    best_vec = None
    best_J = np.inf
    for cand in candidates:
        conv = compute_conv(cand, fft_len, conv_len)
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Best initial J = {best_J:.6f}")

    # Setup Adam parameters
    m = np.zeros_like(best_vec)
    v_hat = np.zeros_like(best_vec)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # annealing schedule for softmax temperature (beta) and Metropolis temperature T
    beta_start = 1.0
    beta_end = 800.0
    T_start = best_J * 0.05
    T_end = 1e-9
    iteration = 0
    last_improve = start_time
    cur_vec = best_vec.copy()
    cur_J = best_J

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # Update beta linearly
        beta = beta_start + (beta_end - beta_start) * min(1.0, elapsed / (deadline - start_time))
        # Compute convolution and softmax weights
        conv = compute_conv(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # Stagnation restart
        if now - last_improve > 30.0:
            # Reset to best with random perturbation
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation, best J = {best_J:.6f}")

        # softmax weighting w (length conv_len)
        # To improve numerical stability subtract max_conv
        w_raw = np.exp(beta * (conv - max_conv))
        w = w_raw / w_raw.sum()

        # gradient of weighted sum of convolutions
        A = np.fft.rfft(cur_vec, fft_len)
        W = np.fft.rfft(w, fft_len)
        grad = 2.0 * np.fft.irfft(np.conj(W) * A, n=fft_len)[:n]

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * (grad ** 2)
        m_corr = m / (1 - beta1 ** iteration)
        v_corr = v_hat / (1 - beta2 ** iteration)

        step_size = 0.05 / (np.sqrt(iteration) + 1)  # optional decay
        update = step_size * m_corr / (np.sqrt(v_corr) + eps)
        # Move opposite to gradient direction (since we want to minimize)
        proposal = cur_vec - update
        # Ensure nonnegativity by projecting onto simplex
        proposal = proj_simplex(proposal)

        # Evaluate new proposal
        prop_conv = compute_conv(proposal, fft_len, conv_len)
        prop_J = 2.0 * n * prop_conv.max()

        # Metropolis acceptance
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))
            if rng.random() < np.exp(-(prop_J - cur_J) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached J = {best_J:.6f}")
            break

    # final clipping and output
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0/n)
    else:
        best_vec = best_vec / total

    # Convert to list of floats
    return [float(v) for v in best_vec]

   best candidate.
4. Refine the best candidate with a projected Adam optimiser on a
   soft‑max smoothed version of the max‑convolution objective.
   * β (soft‑max “temperature”) is increased linearly so that the
     optimisation first sees many large entries and later focuses on the
     true maximum.
   * A simple Metropolis acceptance rule with a linearly cooled temperature
     allows occasional uphill moves.
   * If no improvement is seen for 30 s the current iterate is jittered
     (multiplicative log‑normal noise) to escape plateaus.
5. The final vector is clipped to the allowed range, renormalised and
   returned as a plain Python list.

The whole routine respects the 1000 s budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns w such that w >= 0, sum(w) == 1 and ||w - v||_2 is minimal.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT (length = conv_len).
    Returns the first `conv_len` entries of the circular convolution,
    which for a zero‑padded FFT is the exact linear convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Candidate factories
# ----------------------------------------------------------------------


def _geom_candidate(n: int, r: float) -> np.ndarray:
    """Geometric series a_i ∝ r**i, projected onto the simplex."""
    if r <= 0.0:
        r = 1e-12
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}, projected onto the simplex."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, -p)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _sine_candidate(n: int, power: float = 1.0) -> np.ndarray:
    """Half‑sine window a_i = sin(π (i+0.5)/n) ^ power, projected onto simplex."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(np.pi * x) ** power
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _hann_candidate(n: int) -> np.ndarray:
    """Hann window (raised cosine) a_i = 0.5*(1‑cos(2π i/(n‑1)))."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _triangular_candidate(n: int) -> np.ndarray:
    """Symmetric triangular shape a_i = min(i+1, n‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best normalised sequence
    found within the provided time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a convenient power of two

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a rich pool of initial candidates (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # Uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws (different concentrations)
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Geometric families
    for r in np.linspace(0.90, 0.9999, 12):
        init_cands.append(_geom_candidate(n, r))

    # Power‑law families
    for p in np.linspace(0.1, 2.5, 12):
        init_cands.append(_powerlaw_candidate(n, p))

    # Sine‑window with different powers
    for pw in [0.5, 1.0, 2.0, 4.0]:
        init_cands.append(_sine_candidate(n, pw))

    # Classic smooth windows
    init_cands.append(_hann_candidate(n))
    init_cands.append(_triangular_candidate(n))

    # Previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v /= s
        else:
            v = np.full_like(v, 1.0 / n)
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best of the initial candidates (objective = 2·n·max(conv))
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Refine with projected Adam + soft‑max smoothing + annealing
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # Adam state
    m = np.zeros_like(best_vec)
    v_hat = np.zeros_like(best_vec)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # Temperature schedules
    T_start = best_J * 0.05          # 5 % of current best
    T_end = 1e-9
    beta_start = 1.0
    beta_end = 800.0

    cur_vec = best_vec.copy()
    cur_J = best_J
    iteration = 0
    last_improve = start_time
    stagnation_thresh = 30.0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - start_time) / (deadline - start_time)

        # ----------------------------------------------------------
        # 4.1  Current objective (exact)
        # ----------------------------------------------------------
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # global best update
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4.2  Stagnation handling – jitter after a while
        # ----------------------------------------------------------
        if now - last_improve > stagnation_thresh:
            # multiplicative log‑normal jitter
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4.3  Soft‑max weighting of convolution entries
        # ----------------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * frac
        # shift for numerical stability
        w_raw = np.exp(beta * (conv - max_conv))
        w = w_raw / w_raw.sum()

        # ----------------------------------------------------------
        # 4.4  Gradient of the soft‑max objective (Fourier‑domain)
        # ----------------------------------------------------------
        A = np.fft.rfft(cur_vec, fft_len)
        W = np.fft.rfft(w, fft_len)
        grad = 2.0 * np.fft.irfft(np.conj(W) * A, n=fft_len)[:n]

        # ----------------------------------------------------------
        # 4.5  Adam update (projected onto the simplex)
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad ** 2)

        m_corr = m / (1.0 - beta1 ** iteration)
        v_corr = v_hat / (1.0 - beta2 ** iteration)

        # Learning‑rate schedule – modest decay
        step = 0.05 / (np.sqrt(iteration) + 1.0)
        update = step * m_corr / (np.sqrt(v_corr) + eps)

        proposal = cur_vec - update          # descent step (minimise)
        proposal = proj_simplex(proposal)    # keep on simplex

        # ----------------------------------------------------------
        # 4.6  Evaluate the proposal (exact)
        # ----------------------------------------------------------
        prop_conv = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ----------------------------------------------------------
        # 4.7  Metropolis acceptance with a linearly cooling T
        # ----------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            T = T_start * (1.0 - frac) + T_end * frac
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # early exit if target is already beaten
        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # --------------------------------------------------------------
    # 5. Return the best solution (final clipping & renormalisation)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    # Convert to plain Python list of floats
    return [float(v) for v in best_vec]
```

This implementation expands the initial exploration beyond the previous soft‑max walk, adds several analytically‑motivated families of step‑functions, and employs a projected Adam optimiser with a temperature‑controlled soft‑max objective. The stagnation‑restart and Metropolis acceptance help escape local plateaus, and the whole procedure respects the 1000 s time budget.
