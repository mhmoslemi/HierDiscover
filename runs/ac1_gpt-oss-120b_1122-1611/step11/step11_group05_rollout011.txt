def evaluate_sequence(sequence):

    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]

assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the heights of a step function) that minimises  

\[
J(a)=\frac{2\,n\;\max\limits_{k} (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `*` denotes the linear convolution.  
`J` is invariant under scaling, therefore we restrict ourselves to probability
vectors (`a_i ≥ 0`, `∑ a_i = 1`).  The problem is highly non‑convex – the
objective is the maximum of many quadratic forms – so we use a *global
stochastic search* combined with local gradient information.

The algorithm consists of three stages

1. **Diverse initialisation** – many simple shapes (uniform, ramps,
   geometric decays, Gaussians, sparse spikes, Dirichlet draws and the
   previously found best vector) are created and the best of them is kept.
2. **Simulated‑annealing loop** – at each iteration a new candidate is drawn
   either
   * from a Dirichlet distribution centred on the current point (the
     concentration grows as the temperature drops, giving large moves early
     and fine‑grained moves later), or
   * by multiplicative Gaussian noise.
   The new candidate is accepted if it improves `J` or with a Metropolis
   probability that depends on a linearly decreasing temperature.
3. **Periodic sub‑gradient step** – every few hundred iterations the exact
   convolution is examined, the offset that attains the maximum is identified
   and a simple sub‑gradient step (`a ← a – η∇L_k`) is taken, after which the
   vector is projected back onto the probability simplex.

All operations are performed with FFT‑based convolutions, which makes a full
evaluation of `J` cheap (`O(n log n)`).  The search runs for the whole time
budget (≈ 1000 s) and returns the best sequence found.

# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << ((x - 1).bit_length())


def _max_conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Linear convolution a*a using a real FFT.
    Returns the maximal entry of the full convolution (length ``conv_len``).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective  J(a) = 2·n·max(a*a)/(sum(a)²)."""
    max_conv = _max_conv_fft(a, fft_len, conv_len)
    s = a.sum()
    if s == 0.0:          # should never happen, but guard against div‑0
        return np.inf
    return float(2.0 * n * max_conv / (s * s))
