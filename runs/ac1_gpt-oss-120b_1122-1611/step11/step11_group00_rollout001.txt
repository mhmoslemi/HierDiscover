"""
Simple annealed‐softmax Adam optimiser for the step‑function problem.

The algorithm keeps the vector a on the probability simplex via a = softmax(z)
and minimises

    J(a) = 2·n·max(conv(a,a))

(∑a = 1, therefore the denominator of `evaluate_sequence` is 1).

Instead of using the non‑smooth sub‑gradient of the hard maximum we replace
`max` by a temperature‑controlled soft‑max:

    w_i(β) = exp( β·(c_i – c_max) ) / Σ_j exp( β·(c_j – c_max) ),   c = conv(a,a)

For β→∞ the weights concentrate on the true maximisers, while for small β
the loss landscape is smooth and large steps are possible.  β is increased
linearly during the run (annealing).  Adam updates are performed on the
logits z and a cheap jitter‑restart is triggered if no improvement is seen
for a while.

The routine respects the 1000 s time budget, prints progress and finally
returns the best vector (clipped to the allowed range and re‑normalised so
that `evaluate_sequence` can be called directly).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns the first `conv_len` entries of the linear convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i = exp(beta*(conv_i - conv_max)) / Σ.
    The subtraction of conv_max guarantees numerical stability.
    """
    c_max = conv.max()
    # exponent is always ≤ 0, no overflow risk
    exp_part = np.exp(beta * (conv - c_max))
    total = exp_part.sum()
    if total == 0.0:
        # fallback to uniform weights (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / total


def _grad_softmax(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·conv_i  w.r.t. a, where conv = a*a.
    Using the identity ∂conv_i/∂a = shifted copy of a,
    the gradient is 2 * correlation(a, weights) in the appropriate slice.
    """
    n = a.size
    # full linear correlation length = n + len(weights) - 1 = 3n-2
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    # FFT based correlation: corr = ifft( fft(a) * conj(fft(weights)) )
    A = np.fft.rfft(a, fft_len)
    W = np.fft.rfft(weights, fft_len)
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len)[:corr_len]

    # gradient entries correspond to lag = (len(weights)-1) .. (len(weights)-1)+n-1
    start = (2 * n - 2)               # len(weights)-1 = 2n-2
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Main search routine
# ----------------------------------------------------------------------

    best_J = np.inf
    for cand in init_cands:
        conv = _conv_fft(cand, fft_len_conv, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop (Adam on logits + annealed soft‑max)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # keep a tiny safety margin so we always return before the external timeout
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    eps = 1e-12
    logits = np.log(np.maximum(best_a, eps))
    m = np.zeros_like(logits)
    v_m = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.5

    # Temperature schedule (β)
    beta_min = 5.0
    beta_max = 3000.0

    iteration = 0
    last_improve = start_time

    # Global best (on the exact objective)
    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # ----- 4.1 Build current a from logits (soft‑max) -----
        a = _softmax(logits)

        # ----- 4.2 Compute convolution and exact objective -----
        conv = _conv_fft(a, fft_len_conv, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ----- 4.3 Update global best if improved -----
        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----- 4.4 Stagnation detection & jitter restart -----
        if now - last_improve > 20.0:
            # Apply a relatively strong jitter to logits and reset Adam moments
            jitter_sigma = 0.15
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v_m.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

        # ----- 4.5 Determine current temperature β (linear annealing) -----
        progress = min(1.0, (now - start_time) / (deadline - start_time))
        beta = beta_min + (beta_max - beta_min) * progress

        # ----- 4.6 Compute soft‑max weights over convolution entries -----
        weights = _softmax_weights(conv, beta)

        # ----- 4.7 Compute smooth gradient w.r.t a -----
        grad_a = _grad_softmax(a, weights)

        # ----- 4.8 Transform gradient to logits space (soft‑max chain rule) -----
        avg = np.dot(a, grad_a)            # scalar <a,∇_a J_s>
        grad_z = a * (grad_a - avg)        # ∂J_s/∂z

        # ----- 4.9 Adam update on logits -----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)      # simple 1/√t decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_m = beta2 * v_m + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_m / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # Keep logits within a reasonable numeric range to avoid overflow
        logits = np.clip(logits, -500.0, 500.0)

    # --------------------------------------------------------------
    # 5. Return the best sequence (clipped, renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    # Print final result for convenience
    final_J = float(2.0 * n * np.convolve(best_vec, best_vec).max())
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
