"""
Propose candidate step‑function heights using a smooth‑approximation
gradient descent on the simplex.

The objective J = 2·n·max(conv(a,a)) is invariant to scaling, so we
keep all vectors on the probability simplex (∑a_i = 1).  Instead of
optimising the non‑smooth max directly we minimise the smooth
log‑sum‑exp surrogate

    L(a) = (1/α)·log Σ_k exp(α·conv_k(a))    (α → ∞ ⇒ L → max)

Its gradient is a simple cross‑correlation:
    ∇L_i = 2·(w ⋆ a)_i,
where w_k ∝ exp(α·conv_k) are the softmax weights.
We perform a multiplicative (exponentiated‑gradient) update,
project back onto the simplex, and accept the move with a
Metropolis rule that uses a linearly cooling temperature.
Occasional random Dirichlet perturbations and jittering avoid
getting stuck in shallow plateaus.

The routine builds a diverse portfolio of initial vectors
(uniform, geometric, power‑law, sinusoidal, random Dirichlet,
and the previously found construction if available) and starts
the continuous optimisation from the best of them.  It runs until
the supplied time budget (≈1000 s) expires and returns the best
sequence discovered.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_vec(vec: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution of ``vec`` with itself (full length).
    Returned array has length ``conv_len = 2*len(vec)-1``.
    """
    A = np.fft.rfft(vec, n=fft_len)
    return np.fft.irfft(A * A, n=fft_len)[:conv_len]


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the full self‑convolution."""
    return float(full_conv_vec(vec, fft_len, conv_len).max())


def correlation(vec: np.ndarray,
                w: np.ndarray,
                grad_fft_len: int,
                conv_len: int) -> np.ndarray:
    """
    Compute the cross‑correlation (a⋆w) needed for the gradient:
    grad_i = 2·∑_k w_k·a_{k-i}.
    Implemented via FFT: (vec * rev(w)).
    """
    w_rev = w[::-1]
    A = np.fft.rfft(vec, n=grad_fft_len)
    B = np.fft.rfft(w_rev, n=grad_fft_len)
    corr_full = np.fft.irfft(A * B, n=grad_fft_len)
    # slice indices that correspond to offsets 0 … n‑1
    n = vec.shape[0]
    start = conv_len - 1
    return 2.0 * corr_full[start:start + n]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for a*a
    grad_fft_len = _next_pow2(n + conv_len - 1)        # for a⋆w

    # --------------------------------------------------------------
    # 2. Objective helper (J = 2·n·max(conv) because sum = 1)
    # --------------------------------------------------------------
    def objective(vec: np.ndarray) -> float:
        """Assumes ``vec`` lies on the simplex."""
        return 2.0 * n * full_conv_max(vec, fft_len, conv_len)

    # --------------------------------------------------------------
    # 3. Build a diverse portfolio of starting points
    # --------------------------------------------------------------
    candidates: List[np.ndarray] = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (both directions)
    for r in [0.90, 0.92, 0.94, 0.96, 0.98, 0.99, 0.995]:
        g = r ** np.arange(n, dtype=np.float64)
        g /= g.sum()
        candidates.append(g)
        candidates.append(g[::-1])

    # power‑law shapes
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1, dtype=np.float64)
        pl = 1.0 / (i ** p)
        pl /= pl.sum()
        candidates.append(pl)
        candidates.append(pl[::-1])

    # sinusoidal (zero at the ends, peak in the middle)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    s = np.sin(np.pi * t)
    if s.sum() > 0.0:
        candidates.append(s / s.sum())

    # random Dirichlet draws with a few concentration parameters
    for conc in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(6):
            candidates.append(rng.dirichlet(np.full(n, conc)))

    # previous construction, if available
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        total = v.sum()
        if total == 0.0:
            v[:] = 1.0 / n
        else:
            v /= total
        candidates.append(v)

    # --------------------------------------------------------------
    # 4. Evaluate the portfolio and keep the best vector
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for vec in candidates:
        J = objective(vec)
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Continuous optimisation (smooth‑approximation + EG)
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # temperature schedule (linear cooling)
    T_start = best_J * 0.15   # 15 % of current best
    T_end = 1e-8

    # exponentiated‑gradient parameters
    step0 = 0.45               # initial step size (will decay as 1/√t)
    max_log_factor = 50.0      # clip exponent to avoid overflow

    # smoothing (soft‑max) schedule
    α_start = 0.5
    α_end = 800.0

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - start_time) / (deadline - start_time)

        # ----------------------------------------------------------
        # 5.1 Current objective (already on the simplex ⇒ sum = 1)
        # ----------------------------------------------------------
        conv = full_conv_vec(cur_vec, fft_len, conv_len)
        max_c = conv.max()
        cur_J = 2.0 * n * max_c

        # record a new global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 5.2 Stagnation handling – jitter around the best solution
        # ----------------------------------------------------------
        if now - last_improve > 25.0:
            noise = np.exp(rng.normal(0.0, 0.035, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            continue

        # ----------------------------------------------------------
        # 5.3 Soft‑max weights for the smooth surrogate
        # ----------------------------------------------------------
        α = α_start + (α_end - α_start) * frac
        # stable soft‑max (conv - max_c ≤ 0)
        exp_arg = α * (conv - max_c)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / conv_len)
        else:
            w /= w_sum

        # ----------------------------------------------------------
        # 5.4 Gradient of the smooth surrogate (cross‑correlation)
        # ----------------------------------------------------------
        grad = correlation(cur_vec, w, grad_fft_len, conv_len)

        # ----------------------------------------------------------
        # 5.5 Multiplicative (exponentiated‑gradient) proposal
        # ----------------------------------------------------------
        step = step0 / np.sqrt(iteration + 1.0)
        log_factor = -step * grad
        # clipping prevents overflow in the exponential
        log_factor = np.clip(log_factor, -max_log_factor, max_log_factor)
        factor = np.exp(log_factor)
        proposal = cur_vec * factor

        # occasional random Dirichlet mixing to keep diversity
        if rng.random() < 0.07:
            eps = rng.uniform(0.0, 0.14)
            rand_vec = rng.dirichlet(np.ones(n))
            proposal = (1.0 - eps) * proposal + eps * rand_vec

        # re‑project onto the simplex
        proposal = proj_simplex(proposal)

        # ----------------------------------------------------------
        # 5.6 Evaluate the proposal
        # ----------------------------------------------------------
        prop_conv = full_conv_vec(proposal, fft_len, conv_len)
        prop_J = 2.0 * n * prop_conv.max()

        # ----------------------------------------------------------
        # 5.7 Metropolis acceptance (allows uphill moves early on)
        # ----------------------------------------------------------
        T = (T_start * (1.0 - frac) + T_end * frac)
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # ------------------------------------------------------------------
    # 6. Return the best solution (clip, renormalise, cast to list)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
