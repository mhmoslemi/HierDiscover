def propose_candidate(...):
    rng = np.random.default_rng(seed)
    # Determine n
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        # Ensure n >= 256 and <= 8192.
        n = max(256, min(n, 8192))
    else:
        n = 4096

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    # Evaluate conv for a candidate:
    def compute_J(vec):
        conv = _conv_full(vec, fft_len, conv_len)
        max_conv = conv.max()
        return 2.0 * n * max_conv

    # Initialization of population
    pop = []
    scores = []
    # Add uniform vector, previous best, maybe some Dirichlet draws.
    candidate_list = []
    candidate_list.append(np.full(n, 1.0/n))
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0, None)
        if v.shape[0] != n:
            # Pad or truncate
            if v.shape[0] < n:
                pad = np.zeros(n - v.shape[0])
                v = np.concatenate([v, pad])
            else:
                v = v[:n]
        v_sum = v.sum()
        if v_sum > 0:
            v = v / v_sum
        else:
            v = np.full(n, 1.0/n)
        candidate_list.append(v)

    # Dirichlet draws with varied alphas
    for alpha in [0.01, 0.05, 0.1, 0.3, 1.0]:
        for _ in range(2):
            cand = rng.dirichlet(np.ones(n)*alpha)
            candidate_list.append(cand)

    # Add some random Gaussian exponentials (exp of Normal)
    for _ in range(4):
        cand = np.exp(rng.normal(0.0, 1.0, size=n))
        cand = cand / cand.sum()
        candidate_list.append(cand)

    # Fill remaining to population size
    pop_size = 30
    pop = []
    for cand in candidate_list:
        pop.append(cand)
        if len(pop) >= pop_size:
            break
    while len(pop) < pop_size:
        cand = rng.dirichlet(np.ones(n))
        pop.append(cand)

    # Compute initial scores
    scores = np.empty(len(pop), dtype=np.float64)
    for i,vec in enumerate(pop):
        scores[i] = compute_J(vec)
    best_idx = np.argmin(scores)
    best_vec = pop[best_idx].copy()
    best_J = float(scores[best_idx])
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # DE main loop
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    F = 0.5
    Cr = 0.9

    while time.monotonic() < deadline:
        # For each target vector
        for i in range(len(pop)):
            # Mutation: select a,b,c distinct from i
            idxs = list(range(len(pop)))
            idxs.remove(i)
            a,b,c = rng.choice(idxs, size=3, replace=False)
            # Mutation vector
            mutant = pop[a] + F * (pop[b] - pop[c])
            # Non-negativity
            mutant = np.maximum(mutant, 0.0)
            # Normalize to simplex
            s = mutant.sum()
            if s == 0.0:
                mutant = np.full_like(mutant, 1.0 / n)
            else:
                mutant = mutant / s

            # Crossover (binomial)
            cross = rng.random(n) < Cr
            # Ensure at least one component from mutant
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, pop[i])
            # Ensure non-negativity
            trial = np.maximum(trial, 0.0)
            # Normalize
            ts = trial.sum()
            if ts == 0.0:
                trial = np.full_like(trial, 1.0 / n)
            else:
                trial = trial / ts

            # Evaluate
            J_trial = compute_J(trial)

            # Selection
            if J_trial <= scores[i]:
                pop[i] = trial
                scores[i] = J_trial
                if J_trial < best_J:
                    best_J = float(J_trial)
                    best_vec = trial.copy()
                    now = time.monotonic()
                    elapsed = now - start
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
        iteration += 1

        # occasional local refinement using gradient descent (like earlier algorithm)
        if iteration % 10 == 0:
            # refine best_vec by a few gradient steps
            cur = best_vec.copy()
            cur_conv = _conv_full(cur, fft_len, conv_len)
            cur_J = compute_J(cur)
            # compute active mask and gradient
            max_conv = cur_conv.max()
            active_mask = (cur_conv >= max_conv - 1e-12).astype(np.float64)
            grad_sub = _weighted_gradient(cur, active_mask, fft_len, n)
            # param step?
            step_mul = 0.8  # bigger for quick improvement
            factor = np.exp(-step_mul * grad_sub)
            proposal = cur * factor
            proposal = proj_simplex(proposal)
            prop_J = compute_J(proposal)
            if prop_J < best_J:
                best_J = prop_J
                best_vec = proposal.copy()
                print(f"[{time.monotonic() - start:6.2f}s] Refine best J = {best_J:.6f}")
    # Ensure final vector is valid
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

"""
Differential‑Evolution + occasional gradient refinement.

* The search runs on the probability simplex (∑a_i = 1, a_i ≥ 0).
* A small population (≈30) is evolved with classic DE/rand/1/bin.
* Every 10 generations the current global best is given a fast
  multiplicative‑gradient “polish” step (the same soft update used by
  the previous solver) to escape shallow plateaus.
* If the best value has not improved for a while half of the population
  is re‑initialised (jitter) to regain diversity.
* The whole routine respects the 1000 s wall‑clock budget and finally
  returns the best vector found, normalised to sum = 1 and clipped to
  the allowed range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.

    The algorithm combines a differential‑evolution population with occasional
    gradient polishing.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose the problem size (number of steps)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a comfortable size for FFT speed

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlations

    # --------------------------------------------------------------
    # 2. Helper to evaluate the objective (2·n·max(conv))
    # --------------------------------------------------------------
    def _objective(vec: np.ndarray) -> float:
        conv = _conv_full(vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 3. Build an initial population (simplex‑projected candidates)
    # --------------------------------------------------------------
    pop_size = 30
    population = []
    # a few deterministic seeds
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    population.append(uniform)

    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.shape[0] < n:
            v = np.concatenate([v, np.zeros(n - v.shape[0])])
        elif v.shape[0] > n:
            v = v[:n]
        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = uniform.copy()
        population.append(v)

    # Dirichlet draws with several concentration parameters
    for alpha in (0.01, 0.05, 0.1, 0.3, 1.0):
        for _ in range(2):
            population.append(rng.dirichlet(np.ones(n) * alpha))

    # Exponential of normal draws (log‑normal like)
    for _ in range(4):
        cand = np.exp(rng.normal(0.0, 1.0, size=n))
        cand = cand / cand.sum()
        population.append(cand)

    # Fill up to pop_size with pure Dirichlet draws (α = 1)
    while len(population) < pop_size:
        population.append(rng.dirichlet(np.ones(n)))

    population = population[:pop_size]

    # --------------------------------------------------------------
    # 4. Evaluate initial population
    # --------------------------------------------------------------
    scores = np.empty(pop_size, dtype=np.float64)
    for i, vec in enumerate(population):
        scores[i] = _objective(vec)

    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx].copy()
    best_J = float(scores[best_idx])
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Main DE loop with occasional gradient refinements
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time
    stagnation_counter = 0

    F = 0.5            # mutation factor
    Cr = 0.9           # crossover probability

    iteration = 0
    while time.monotonic() < deadline:
        # one DE generation
        for i in range(pop_size):
            # --- mutation -------------------------------------------------
            pool = list(range(pop_size))
            pool.remove(i)
            a, b, c = rng.choice(pool, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.maximum(mutant, 0.0)          # keep non‑negative

            # --- projection onto simplex ----------------------------------
            s = mutant.sum()
            if s == 0.0:
                mutant = np.full_like(mutant, 1.0 / n)
            else:
                mutant = mutant / s

            # --- binomial crossover ---------------------------------------
            cross = rng.random(n) < Cr
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])

            # --- final simplex projection ----------------------------------
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = np.full_like(trial, 1.0 / n)
            else:
                trial = trial / ts

            # --- selection ------------------------------------------------
            J_trial = _objective(trial)
            if J_trial <= scores[i]:
                population[i] = trial
                scores[i] = J_trial
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_vec = trial.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1
        now = time.monotonic()

        # ---- stagnation handling -----------------------------------------
        if now - last_improve > 30.0:
            # re‑inject diversity: replace half of the pop with fresh draws
            for k in range(pop_size // 2):
                cand = rng.dirichlet(np.ones(n) * 0.1)
                population[k] = cand
                scores[k] = _objective(cand)
            last_improve = now
            stagnation_counter += 1
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Stagnation #{stagnation_counter} – jittered population")

        # ---- periodic gradient polish of the global best -----------------
        if iteration % 10 == 0:
            cur = best_vec.copy()
            conv = _conv_full(cur, fft_len, conv_len)
            max_conv = conv.max()
            active = (conv >= max_conv - 1e-12).astype(np.float64)
            grad = _weighted_gradient(cur, active, fft_len_corr, n)

            # multiplicative step (larger step‑size than in the baseline)
            step_mul = 0.8
            factor = np.exp(-step_mul * grad)
            polished = cur * factor
            polished = proj_simplex(polished)

            J_polished = _objective(polished)
            if J_polished < best_J - 1e-12:
                best_J = J_polished
                best_vec = polished.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] Polished best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Final clean‑up: clip, renormalise and convert to python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total
