    # returns projection onto probability simplex
    # Works for any size.
    n = len(v)
    u = np.sort(v)[::-1]  # sort descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety
    # Determine n
    if 'height_sequence_1' in globals():
        v_prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        n = len(v_prev)
        # Normalize to sum 1
        v_prev = np.clip(v_prev, 0.0, None)
        total = v_prev.sum()
        if total > 0:
            v_prev /= total
        else:
            v_prev = np.full(n, 1.0/n)
    else:
        n = 2048
        v_prev = None

    # generate initial candidates
    init_candidates = []
    init_candidates.append(np.full(n, 1.0/n, dtype=np.float64))  # uniform

    # Dirichlet draws
    for alpha in [0.3, 0.6, 1.0]:
        init_candidates.append(rng.dirichlet(np.full(n, alpha)))
    # Gaussian shape
    x = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    for sigma in [n/12.0, n/8.0, n/6.0]:
        g = np.exp(-0.5*((x-center)/sigma)**2)
        init_candidates.append(g / g.sum())
    # if previous exists
    if v_prev is not None:
        init_candidates.append(v_prev)

    # evaluate initial candidates using FFT conv compute (fast)
    best_vec = None
    best_obj = np.inf
    for cand in init_candidates:
        # compute conv via FFT
        L = 1 << ((2*n-2).bit_length()) # next power of 2 for conv length 2n-1 maybe
        A = np.fft.rfft(cand, L)
        conv = np.fft.irfft(A*A, n=L)[:2*n-1]
        maxc = conv.max()
        obj = 2.0*n*maxc
        if obj < best_obj:
            best_obj = obj
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_obj:.6f}")

    # set current vector as best_vec
    cur_vec = best_vec.copy()
    best_solution = best_vec.copy()
    best_J = best_obj
    last_improve_time = start_time

    # Prepare FFT lengths
    L_conv = _next_pow2(2*n-1)
    L_grad = _next_pow2(3*n-2)

    # schedule of beta values
    betas = np.geomspace(1.0, 200.0, num=25)  # geometric progression

    iteration = 0
    while time.monotonic() < deadline:

        # For simplicity cycle through betas
        beta = betas[iteration % len(betas)]

        A = np.fft.rfft(cur_vec, L_conv)
        conv_full = np.fft.irfft(A * A, n=L_conv)[:2*n-1]

            best_J = cur_J
            best_solution = cur_vec.copy()
            last_improve_time = time.monotonic()

    return [float(v) for v in best_solution]

"""
Hybrid smooth‑max gradient descent for the step‑function (self‑convolution) problem.

The objective to minimise is
    J(a) = 2·n·max(conv(a,a))   with   a_i ≥ 0 , Σ a_i = 1 .
Because of the scaling invariance we work on the probability simplex.
A smooth approximation of the max (log‑sum‑exp) gives a differentiable
proxy; its gradient can be written as a convolution and is obtained
efficiently by FFTs.  An exponentiated‑gradient update (multiplicative
step) keeps the vector non‑negative; after each step we project back onto
the simplex.  The algorithm cycles through an increasing “β’’ parameter
(temperature of the smooth max) and uses occasional jitter/re‑start
when progress stalls.

The routine runs for at most ``budget_s`` seconds and returns the best
vector found (clipped to the limits of ``evaluate_sequence``).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few Dirichlet draws with different concentrations
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Gaussian‑shaped bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        init_cands.append(g / g.sum())

    # reuse the previous construction when it exists
    if prev is not None:
        init_cands.append(prev)

    # --------------------------------------------------------------
    # 3. Choose the best of the initial candidates (fast FFT evaluation)
    # --------------------------------------------------------------
    L_conv = _next_pow2(2 * n - 1)          # length for self‑convolution FFT
    best_vec = None
    best_J = np.inf

    for cand in init_cands:
        A = np.fft.rfft(cand, L_conv)
        conv = np.fft.irfft(A * A, n=L_conv)[: 2 * n - 1]
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop (smooth‑max gradient descent)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    cur_vec = best_vec.copy()
    best_solution = best_vec.copy()
    best_obj = best_J
    last_improve = start_time

    # FFT length for the gradient convolution (w * a_rev)
    L_grad = _next_pow2(3 * n - 2)

    # geometric schedule for the smooth‑max temperature β
    betas = np.geomspace(1.0, 200.0, num=25)

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- evaluate current state (fast FFT) ----
        A = np.fft.rfft(cur_vec, L_conv)
        conv_full = np.fft.irfft(A * A, n=L_conv)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_J = 2.0 * n * max_conv

        # ---- keep global best ----
        if cur_J < best_obj - 1e-12:
            best_obj = cur_J
            best_solution = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_obj:.6f}")

        # ---- stagnation handling (jitter) ----
        if now - last_improve > 15.0:
            noise = np.exp(rng.normal(0.0, 0.3, size=n))
            cur_vec = best_solution * noise
            cur_vec = _project_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart")
            continue

        # ----------------------------------------------------------
        # 5. Smooth‑max (log‑sum‑exp) weights
        # ----------------------------------------------------------
        beta = betas[iteration % len(betas)]

        # shift so that the maximum becomes zero (numerical stability)
        shifted = conv_full - max_conv          # ≤ 0
        exp_arg = beta * shifted                # still ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----------------------------------------------------------
        # 6. Gradient of the smooth max (exact formula)
        #    grad_i = 2·β·∑_j w_{i+j}·a_j   (indices outside [0,2n‑2] are ignored)
        #    which is the linear convolution of w with a reversed.
        # ----------------------------------------------------------
        a_rev = cur_vec[::-1]
        W_fft = np.fft.rfft(w, L_grad)
        A_rev_fft = np.fft.rfft(a_rev, L_grad)
        conv_rev_full = np.fft.irfft(W_fft * A_rev_fft, n=L_grad)[: 3 * n - 2]
        grad = 2.0 * beta * conv_rev_full[n - 1 : 2 * n - 1]   # length n

        # ----------------------------------------------------------
        # 7. Exponentiated‑gradient update (multiplicative, keeps positivity)
        # ----------------------------------------------------------
        step = 0.5 / np.sqrt(iteration + 1.0)   # decaying step size
        factor = np.exp(-step * grad)
        proposal = cur_vec * factor
        proposal = _project_simplex(proposal)

        # occasional symmetrisation to encourage a≈a_rev
        if iteration % 50 == 0:
            proposal = (proposal + proposal[::-1]) * 0.5
            proposal = _project_simplex(proposal)

        # ----------------------------------------------------------
        # 8. Greedy (with tiny Metropolis) acceptance
        # ----------------------------------------------------------
        A_prop = np.fft.rfft(proposal, L_conv)
        conv_prop = np.fft.irfft(A_prop * A_prop, n=L_conv)[: 2 * n - 1]
        prop_J = 2.0 * n * conv_prop.max()

        if prop_J < cur_J:
            cur_vec = proposal
        else:
            # a vanishing probability of accepting a worse move; temperature
            # decays linearly with the remaining budget
            T = max(1e-5,
                    (deadline - now) / (deadline - start_time))
            if rng.random() < np.exp(-(prop_J - cur_J) / T):
                cur_vec = proposal

        iteration += 1

    # --------------------------------------------------------------
    # 9. Return the best solution (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_solution = np.clip(best_solution, 0.0, 1000.0)
    total = best_solution.sum()
    if total == 0.0:
        best_solution = np.full(n, 1.0 / n)
    else:
        best_solution /= total

    # (optional) final sanity check – uncomment to see the final objective
    # final_J = evaluate_sequence(best_solution.tolist())
    # print(f"Final J (evaluate_sequence) = {final_J:.6f}")

    return [float(v) for v in best_solution]
