```
def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    # determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        n = max(256, min(n, 8192))
    else:
        n = 4096
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    fft_len_corr = _next_pow2(conv_len + n - 1)
    best_vec = None
    best_J = np.inf
    start_time = time.monotonic()
    deadline = start_time + budget_s - 0.5   # 0.5s buffer
    # function to compute J
    def compute_J(vec):
        conv = _conv_full(vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())
    # optimizer function
    def run_opt(p0, max_iter, beta_start, beta_end, lr_start):
        m = np.zeros_like(p0)
        v = np.zeros_like(p0)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        best_local_vec = p0.copy()
        best_local_J = compute_J(p0)
        for it in range(1, max_iter+1):
            # compute conv and beta schedule
            conv = _conv_full(p0, fft_len, conv_len)
            max_conv = conv.max()
            # softmax weighting
            beta = beta_start * (beta_end / beta_start) ** (it / max_iter)
            # to avoid overflow:
            tmp = beta * (conv - max_conv)
            w_exp = np.exp(tmp)
            w = w_exp / w_exp.sum()
            # gradient
            grad = _weighted_gradient(p0, w, fft_len_corr, n)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            # step size schedule
            lr = lr_start / np.sqrt(it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            p_new = p0 - step
            p_new = proj_simplex(p_new)
            p0 = p_new
            # evaluate
            cur_J = compute_J(p0)
            if cur_J < best_local_J:
                best_local_J = cur_J
                best_local_vec = p0.copy()
        return best_local_vec, best_local_J
    # Main loop of random restarts
    iteration = 0
    while time.monotonic() < deadline:
        # Choose initialization
        init_type = rng.integers(0, 5)
        if init_type == 0:
            p0 = np.full(n, 1.0 / n)
        elif init_type == 1:
            # Dirichlet random
            alpha = rng.choice([0.1, 0.3, 0.5, 1.0])
            p0 = rng.dirichlet(np.ones(n) * alpha)
        elif init_type == 2:
            # geometric
            r = rng.choice([0.9, 0.95, 0.97, 0.99])
            p0 = r ** np.arange(n, dtype=np.float64)
            p0 /= p0.sum()
        elif init_type == 3:
            # sinusoidal modulation
            eps = 0.5
            p0 = 1.0 + eps * np.sin(2.0 * np.pi * np.arange(n) / n)
            p0 = np.clip(p0, 0.0, None)
            p0 /= p0.sum()
        else:
            # random jitter of global best or previous candidate if any
            if best_vec is not None:
                noise = np.exp(rng.normal(0.0, 0.02, size=n))
                p0 = best_vec * noise
                p0 = proj_simplex(p0)
            else:
                p0 = np.full(n, 1.0 / n)
        # Determine iteration count based on remaining time
        remaining = deadline - time.monotonic()
        # allocate at most e.g., 200 iterations per run, but ensure we have enough time
        max_iters = int(min(200, remaining / (0.002 * n / 1024)))  # rough estimate of time per iteration
        if max_iters < 10:

        except Exception as e:

        if J < best_J:
            best_J = J
            best_vec = vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] New global best J = {best_J:.6f}")
        iteration += 1
    # final fallback: if none found, use uniform
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
    # Clip to safe bounds
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total
    return [float(v) for v in best_vec]
```

assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function (a probability vector) that minimises  

```
J(a) = 2·n·max( a * a )                 ( * is the linear convolution )
```

Because the objective is scale invariant we restrict the search to the probability simplex  
`∑ a_i = 1 , a_i ≥ 0`.  
The algorithm is a *multi‑start Adam optimiser* that works on a **smooth approximation** of the
max‑operator (soft‑max) for the first half of the iterations and then refines the result with a
*sub‑gradient* that focuses on the currently active (maximal) convolution entries.

Key ideas

* **FFT based convolutions** – `O(n log n)` per iteration.
* **Soft‑max smoothing** → differentiable surrogate of `max(conv)`.  
  `β` (inverse temperature) is increased from `1` up to `5000` to gradually sharpen the
  approximation.
* **Adam** – adaptive step sizes, proven to work well for non‑convex problems.
* **Two‑phase gradient** – soft‑max gradient first, active‑set gradient later.
* **Multiple random starts** – each start receives a short Adam run (≈ 150 iterations).  
  The best vector found over the whole budget is returned.
* **Simplex projection** – ensures feasibility after every update.
* **Stagnation jitter** – if no improvement is observed for a while the current best
  vector is perturbed with small multiplicative Gaussian noise.

The code follows the restriction that all helper functions are defined at the top level
(no closures, no lambdas).  

```python
"""
Multi‑start Adam optimiser with soft‑max smoothing → sub‑gradient refinement.
Searches for a probability vector that minimises
    J(a) = 2·n·max(convolution(a, a))

* FFT is used for fast convolution and for the weighted gradient.
* The first half of the iterations uses a soft‑max (smooth max) weight,
  the second half uses the exact active‑set weight.
* Adam adapts the step size; after every update the vector is projected onto the simplex.
* Several random initialisations (uniform, Dirichlet, geometric, sinusoidal, jittered best)
  are tried until the time budget expires.
* The best vector discovered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, n=fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, n=fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # the valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective value 2·n·max(conv(vec,vec))."""
    conv = _conv_full(vec, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return a probability weight vector proportional to exp(beta * conv).
    Numerical stability is achieved by subtracting the maximum.
    """
    max_c = conv.max()
    exp_shift = np.exp(beta * (conv - max_c))
    s = exp_shift.sum()
    if s == 0.0:
        return np.full_like(exp_shift, 1.0 / exp_shift.size)
    return exp_shift / s


def _active_weights(conv: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """
    Return a uniform probability distribution over the active set,
    i.e. indices where conv is (almost) maximal.
    """
    max_c = conv.max()
    active = (conv >= max_c - max(tol, max_c * tol)).astype(np.float64)
    total = active.sum()
    if total == 0.0:
        return np.full_like(active, 1.0 / active.size)
    return active / total


def adam_optimize(start_vec: np.ndarray,
                  n: int,
                  fft_len: int,
                  fft_len_corr: int,
                  conv_len: int,
                  max_iters: int,
                  beta_start: float,
                  beta_end: float,
                  lr_start: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation with a two‑phase gradient:
      – first half: soft‑max (smooth max)
      – second half: active‑set (sub‑gradient of max)
    Returns the best vector found and its objective value.
    """
    # Adam hyper‑parameters
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    m = np.zeros_like(start_vec)
    v = np.zeros_like(start_vec)

    # initialise
    cur = start_vec.copy()
    cur_J = compute_J(cur, n, fft_len, conv_len)
    best_vec = cur.copy()
    best_J = cur_J

    # pre‑compute iteration split
    split = max_iters // 2 if max_iters >= 2 else max_iters

    for it in range(1, max_iters + 1):
        # current convolution (needed for both the gradient and the objective)
        conv = _conv_full(cur, fft_len, conv_len)

        # ----- choose weight vector -----
        if it <= split:
            # smooth approximation – soft‑max
            beta = beta_start * (beta_end / beta_start) ** ((it - 1) / split)
            w = _softmax_weights(conv, beta)
        else:
            # exact sub‑gradient – active set
            w = _active_weights(conv, tol=1e-12)

        # ----- gradient w.r.t. the chosen weights -----
        grad = _weighted_gradient(cur, w, fft_len_corr, n)

        # ----- Adam update -----
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)

        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)

        lr = lr_start / np.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cur = cur - step
        cur = proj_simplex(cur)

        # ----- evaluate -----
        cur_J = compute_J(cur, n, fft_len, conv_len)
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cur.copy()

    return best_vec, best_J


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds and
    returns the best vector discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size (FFT size stays modest)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlations

    # --------------------------------------------------------------
    # 2. Initial candidate pool (all vectors are normalised to sum=1)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # sinusoidal modulation (small oscillation)
    eps = 0.5
    v = 1.0 + eps * np.sin(2.0 * np.pi * np.arange(n) / n)
    v = np.clip(v, 0.0, None)
    v /= v.sum()
    init_cands.append(v)

    # Dirichlet draws – various concentrations (including sparse)
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Keep the best candidate seen so far (fast J evaluation)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop – many short Adam runs
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a safety margin
    total_time = deadline - start_time

    # annealing / jitter parameters
    last_improve = start_time
    stagnation_limit = 30.0            # seconds without improvement → jitter
    max_iters_per_run = 150            # Adam iterations per random start
    beta_start = 1.0
    beta_end = 5000.0
    lr_start = 0.5

    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # Choose a random initialisation for a fresh Adam run
        # ----------------------------------------------------------
        choice = rng.integers(0, len(init_cands) + 2)
        if choice < len(init_cands):
            p0 = init_cands[choice].copy()
        elif choice == len(init_cands):
            # jittered version of the current global best
            if best_vec is not None:
                noise = np.exp(rng.normal(0.0, 0.04, size=n))
                p0 = best_vec * noise
                p0 = proj_simplex(p0)
            else:
                p0 = np.full(n, 1.0 / n)
        else:
            # completely fresh random Dirichlet draw
            alpha = rng.choice([0.05, 0.2, 0.5, 1.0])
            p0 = rng.dirichlet(np.ones(n) * alpha)

        # ----------------------------------------------------------
        # Run a short Adam optimisation from this start
        # ----------------------------------------------------------
        vec, J = adam_optimize(
            start_vec=p0,
            n=n,
            fft_len=fft_len,
            fft_len_corr=fft_len_corr,
            conv_len=conv_len,
            max_iters=max_iters_per_run,
            beta_start=beta_start,
            beta_end=beta_end,
            lr_start=lr_start,
        )

        # ----------------------------------------------------------
        # Keep global best
        # ----------------------------------------------------------
        if J < best_J - 1e-12:
            best_J = J
            best_vec = vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # Stagnation handling – occasional jitter of the best solution
        # ----------------------------------------------------------
        if now - last_improve > stagnation_limit:
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            best_vec = proj_simplex(best_vec * noise)
            best_J = compute_J(best_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter after stagnation – best J = {best_J:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best vector (clip and renormalise for safety)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
```
