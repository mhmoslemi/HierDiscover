def propose_candidate(seed=12345, budget_s=1000):
    import numpy as np, time, cvxpy as cp

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5

    n = 1024  # may adjust based on memory/time

    S = np.sqrt(2 * n)

    # generate initial candidates
    candidates = []
    # uniform
    candidates.append(np.full(n, S / n))
    # geometric decays etc...
    for r in [0.99, 0.995, 0.9975]:
        v = r ** np.arange(n)
        v = v / v.sum() * S
        candidates.append(v)
    # Dirichlet random draws
    for alpha in [0.02, 0.05, 0.1, 0.2]:
        for _ in range(5):
            v = rng.dirichlet(np.full(n, alpha))
            v = v * S
            candidates.append(v)

    # Evaluate initial candidates quickly
    best_vec = None
    best_J = np.inf
    for v in candidates:
        conv = np.fft.irfft(np.fft.rfft(v, _next_pow2(2*n-1)) * np.fft.rfft(v, _next_pow2(2*n-1)))[:2*n-1]
        J = float(conv.max())
        if J < best_J:
            best_J = J
            best_vec = v.copy()
    print(f"[0.0s] Init best J={best_J}")

    # main loop for LP iterations
    cur = best_vec.copy()
    cur_J = best_J
    best = best_vec.copy()

    iter = 0
    while time.time() < deadline:
        # compute conv and M
        conv_len = 2*n - 1
        fft_len = _next_pow2(conv_len)
        conv_full = np.fft.irfft(np.fft.rfft(cur, fft_len) * np.fft.rfft(cur, fft_len))[:conv_len]
        M = conv_full.max()

        # Build convolution matrix C: shape (conv_len, n)
        # each row k: for i=0..n-1, C[k,i] = cur[i] if 0 <= k-i < n else 0
        # Instead of building full dense, we can (maybe) use Toeplitz
        C = np.zeros((conv_len, n), dtype=np.float64)
        # vector f = cur
        for i in range(n):
            start_idx = i
            end_idx = i + n
            C[start_idx:end_idx, i] = cur[:n]  # but careful: need cur as row values? Actually for each k, C[k,i] = cur[k-i] if k-i in [0,n-1]
        # This loops O(n^2). Acceptable for n=512.

        # Solve LP
        g_var = cp.Variable(n, nonneg=True)
        constraints = [C @ g_var <= M + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-9, reltol=1e-9)
        if g_var.value is None:
            break
        g = np.maximum(g_var.value, 0.0)  # ensure nonneg
        sum_g = g.sum()
        if sum_g == 0:
            break
        # scale to sum S
        g = g * (S / sum_g)

        # mixing
        t = 0.3 * (0.98 ** iter)  # decreasing
        new = (1 - t) * cur + t * g

        # renorm to sum S exactly
        new = new * (S / new.sum())

        # evaluate J
        conv_new = np.fft.irfft(np.fft.rfft(new, fft_len) * np.fft.rfft(new, fft_len))[:conv_len]
        J_new = float(conv_new.max())

        # accept if improvement
        if J_new < cur_J or rng.random() < np.exp(-(J_new - cur_J) * 10):
            cur = new
            cur_J = J_new

        if J_new < best_J:
            best_J = J_new
            best = new.copy()
            print(f"[{time.time() - start:.2f}s] New best J={best_J:.6f}")

        iter += 1

        # stagnation restart
        if iter % 30 == 0 and time.time() < deadline:
            # jitter around best
            noise = np.exp(rng.normal(0, 0.05, size=n))
            cur = best * noise
            cur = cur * (S / cur.sum())
            cur_J = best_J

    # after LP loops, maybe run a few gradient descent steps using the hybrid algorithm from previous code but using smaller n to fine tune.
    # We'll re-use earlier propose_candidate algorithm for refinement but limited to few iterations.

    # final cleanup
    best = np.clip(best, 0.0, 1000.0)
    s = best.sum()
    if s == 0:
        best[:] = S / n
    else:
        best = best * (S / s)

    return [float(v) for v in best]

"""
Adam‑based smoothed‑max optimizer for the step–function problem.

The objective after normalising the ℓ1‑norm to √(2 n) becomes

    J(a) = max_k (a * a)[k]                                 (1)

We therefore minimise the peak of the auto‑convolution.
The non‑smooth max is approximated by a log‑sum‑exp (soft‑max)
with a temperature β that is increased during the run.
A simple Adam optimiser (first‑ and second‑moment adaptation)
is applied to the gradient of the smoothed objective.
All iterates are projected onto the probability simplex
(ℓ1‑norm = √(2 n), a_i ≥ 0).  Random re‑initialisations and
light jitter are used to escape plateaux.  The routine works for
any problem size (default n≈2000) and respects the 1000 s budget.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# --------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Gradient of  Σ weight[k]·(a*a)[k]  w.r.t. a.
    Returns a vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # correlation part (valid indices are n‑1 … n‑1+n‑1)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _max_conv(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Return max of the full auto‑convolution of a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(conv.max())


# --------------------------------------------------------------
# Main search routine
# --------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The algorithm runs for at most `budget_s`
    seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------- problem size -------------------------
    # use the length of the previously discovered sequence if it exists,
    # otherwise fall back to a moderate default.
    if "height_sequence_1" in globals():
        old_len = len(globals()["height_sequence_1"])
        n = max(256, min(old_len, 4096))
    else:
        n = 2048

    S = np.sqrt(2.0 * n)                # target ℓ1‑norm
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlation

    # ------------------- initial candidates --------------------
    init_cands: list[np.ndarray] = []

    # Uniform
    init_cands.append(np.full(n, S / n, dtype=np.float64))

    # Geometric decays
    for r in (0.99, 0.995, 0.997, 0.9985):
        v = r ** np.arange(n, dtype=np.float64)
        v = v / v.sum() * S
        init_cands.append(v)

    # Linear ramps
    v = np.arange(1, n + 1, dtype=np.float64)
    v = v / v.sum() * S
    init_cands.append(v)

    v = np.arange(n, 0, -1, dtype=np.float64)
    v = v / v.sum() * S
    init_cands.append(v)

    # Gaussian bells (discrete)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v = v / v.sum() * S
        init_cands.append(v)

    # Random Dirichlet draws with a few concentration values
    for alpha in (0.02, 0.05, 0.1, 0.2):
        for _ in range(3):
            v = rng.dirichlet(np.full(n, alpha))
            init_cands.append(v * S)

    # Previous construction (down‑sample / trim if needed)
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if old.size > n:
            # simple uniform down‑sampling
            idx = (np.linspace(0, old.size - 1, n)).astype(int)
            v = old[idx]
        else:
            v = np.pad(old, (0, n - old.size), constant_values=0.0)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = S / n
        else:
            v = v * (S / v.sum())
        init_cands.append(v)

    # ------------------- pick the best starter ---------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = _max_conv(cand, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------- optimisation parameters -------------
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    max_iters_per_run = 3000   # enough for convergence on the chosen size

    # Stopping criteria for jitter handling
    iter_without_improve = 0
    global_best_vec = best_vec.copy()
    global_best_J = best_J

    # ------------------- main search loop --------------------
    while time.time() < deadline:
        # ------------------------------------------------------
        # 1. Random re‑initialisation (Dirichlet with random α)
        # ------------------------------------------------------
        alpha = rng.uniform(0.01, 0.3)
        a = rng.dirichlet(np.full(n, alpha))
        a = a * S

        # Adam state
        m = np.zeros_like(a)
        v_vec = np.zeros_like(a)

        # ------------------------------------------------------
        # 2. Adam optimisation on the smoothed max
        # ------------------------------------------------------
        for it in range(1, max_iters_per_run + 1):
            now = time.time()
            if now >= deadline:
                break

            # current convolution and its maximum
            conv = _conv_full(a, fft_len, conv_len)
            max_conv = conv.max()

            # global best update
            if max_conv < global_best_J - 1e-12:
                global_best_J = max_conv
                global_best_vec = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New global best J = {global_best_J:.6f}")
                iter_without_improve = 0
            else:
                iter_without_improve += 1

            # ----------------------------------------------------------------
            # Soft‑max (log‑sum‑exp) approximation of the max.
            # β is increased linearly from 1 to 500 during the run.
            # ----------------------------------------------------------------
            beta = 1.0 + (500.0 - 1.0) * (it / max_iters_per_run)
            w_exp = np.exp(beta * (conv - max_conv))   # max‑shift for stability
            w_sum = w_exp.sum()
            if w_sum == 0.0:
                w_norm = np.full_like(w_exp, 1.0 / w_exp.size)
            else:
                w_norm = w_exp / w_sum

            # --------------------------------------------------------------
            # Gradient of the smoothed objective (2 * correlation)
            # --------------------------------------------------------------
            grad = _weighted_gradient(a, w_norm, fft_len_corr, n)

            # --------------------------------------------------------------
            # Adam update (move opposite to the gradient)
            # --------------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v_vec / (1.0 - beta2 ** it)

            # step‑size schedule (decays like 1/√t)
            lr = 0.5 / np.sqrt(it)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # --------------------------------------------------------------
            # Projection onto the feasible set (non‑negative simplex)
            # --------------------------------------------------------------
            a = np.maximum(a, 0.0)
            sum_a = a.sum()
            if sum_a == 0.0:
                a[:] = S / n
            else:
                a = a * (S / sum_a)

            # --------------------------------------------------------------
            # Occasional jitter to escape flat plateaux
            # --------------------------------------------------------------
            if it % 500 == 0 and iter_without_improve > 2000:
                noise = np.exp(rng.normal(0.0, 0.07, size=n))
                a = global_best_vec * noise
                a = a * (S / a.sum())
                iter_without_improve = 0

        # ------------------------------------------------------------------
        # If still time left, perform a cheap random‑neighbour search around
        # the current global best (simple multiplicative jitter).
        # ------------------------------------------------------------------
        if time.time() >= deadline:
            break
        if (time.time() - start_time) % 30 < 1.0:  # occasional check
            noise = np.exp(rng.normal(0.0, 0.03, size=n))
            cand = global_best_vec * noise
            cand = cand * (S / cand.sum())
            cand_J = _max_conv(cand, fft_len, conv_len)
            if cand_J < global_best_J:
                global_best_J = cand_J
                global_best_vec = cand.copy()
                print(f"[{time.time() - start_time:6.2f}s] Jitter improvement => {global_best_J:.6f}")

    # ------------------- final sanitisation --------------------
    best = np.clip(global_best_vec, 0.0, 1000.0)
    total = best.sum()
    if total == 0.0:
        best[:] = S / n
    else:
        best = best * (S / total)

    return [float(v) for v in best]
