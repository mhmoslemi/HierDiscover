def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 5.0

    # Determine n: use length from height_sequence_1 if exists else around 4000.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 4000
    n = max(256, min(n, 8192))

    # precompute FFT lengths:
    conv_len = 2*n - 1
    fft_len_conv = _next_pow2(conv_len)
    fft_len_corr = _next_pow2(n + conv_len - 1)

    # Initialize with random Dirichlet draws and maybe height_sequence_1
    init_candidates = []
    # uniform
    init_candidates.append(np.full(n, 1.0/n))
    # geometric
    for r in (0.9, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_candidates.append(v)
    # Dirichlet draws
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_candidates.append(rng.dirichlet(np.full(n, alpha)))
    # previous best
    if 'height_sequence_1' in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            v /= s
            init_candidates.append(v)

    # select best in terms of actual J (max conv)
    best_vec = None
    best_J = np.inf
    for cand in init_candidates:
        # compute J via compute_J (fast using FFT)
        J = compute_J(cand, n, fft_len_conv, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    # Now iterative smooth gradient descent
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_vec_global = cur_vec.copy()
    best_J_global = cur_J
    # holds iteration counter
    it = 0
    momentum = np.zeros_like(cur_vec)
    beta = 0.9

    # schedule for eps
    eps_start = 5.0
    eps_end = 30.0
    # schedule for step size
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        # update of eps linearly over iterations
        eps = eps_start + (eps_end - eps_start) * min(1.0, elapsed / (deadline - start))
        # compute convolution and smooth J and gradient
        # convolution a*a:
        A_fft = np.fft.rfft(cur_vec, fft_len_conv)
        conv = np.fft.irfft(A_fft * A_fft, n=fft_len_conv)[:conv_len]
        # compute smooth objective J_s = 2*n * logsumexp(eps*conv)/eps
        # Use stable method:
        max_eps_conv = np.max(eps * conv)
        # Actually compute logsumexp:
        lse = max_eps_conv + np.log(np.sum(np.exp(eps * conv - max_eps_conv)))
        J_smooth = 2.0 * n * lse / eps
        # compute softmax weights:
        w = np.exp(eps * conv - max_eps_conv)
        w /= w.sum()
        # compute gradient:
        w_rev = w[::-1]
        G_fft = np.fft.rfft(cur_vec, fft_len_corr) * np.fft.rfft(w_rev, fft_len_corr)
        corr = np.fft.irfft(G_fft, n=fft_len_corr)
        start_idx = conv_len - 1
        grad = 2.0 * corr[start_idx:start_idx + n]
        # apply momentum:
        momentum = beta * momentum + (1 - beta) * grad
        # step size schedule
        step = 0.5 / (np.sqrt(it + 1))
        prop = cur_vec - step * momentum
        # project onto simplex
        prop = proj_simplex(prop)

        A_fft = np.fft.rfft(prop, fft_len_conv)

        accept = False

"""
Hybrid smooth‑max projected gradient descent.

The objective to minimise is   J(a) = 2·n·max(conv(a,a)),
which is invariant under scaling; we therefore keep the vector on the
probability simplex (∑a = 1, a_i ≥ 0).

Main ideas
-----------

* **Smooth approximation of the max** – we use a soft‑max with an
  exponential weight  w_k ∝ exp(‑ε·(M−c_k)/M)  where
  c_k = (a∗a)[k] and M = max(c).  When ε is small the weight is almost
  uniform, for large ε it concentrates on the active constraints.
  ε is increased slowly during the run.

* **Analytic gradient** – for a weight vector w the gradient of the
  smoothed objective is
      ∇ = 2·corr(w_rev , a) ,
  where `corr` is the cross‑correlation (computed by a single FFT).

* **Projected (momentum) gradient step** – the raw gradient is combined
  with a momentum term (β≈0.9) and a step‑size α_t∝1/√t.  After the step
  the vector is projected back onto the simplex.

* **Escaping stagnation** – if no improvement is seen for 20 s a short
  jitter (log‑normal multiplicative noise) centred on the current best
  solution is applied.

* **Time‑budget aware** – the loop runs until the supplied budget (≈ 1000 s)
  is exhausted, keeping the best feasible vector seen so far.

The returned vector is clipped to the allowed range [0, 1000] and
re‑normalised; it can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or v is zero); return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Fast evaluation of  J = 2·n·max(conv(a,a))   for a already on the simplex.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The function respects the 1000 s
    time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ---------- 1. Problem size ----------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    # FFT lengths for the two convolutions we need:
    fft_len_conv = _next_pow2(conv_len)                 # a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)         # a ⋆ w

    # ---------- 2. Initialise a diverse pool ----------
    init_cands = []

    # Uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – several concentrations
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Previous best, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0:
            v /= s
            init_cands.append(v)

    # ---------- 3. Pick the best initial candidate ----------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand, n, fft_len_conv, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ---------- 4. Optimisation loop ----------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    best_vec_global = cur_vec.copy()
    best_J_global = cur_J
    last_improve = start_time

    # algorithmic parameters
    beta = 0.9                     # momentum decay
    momentum = np.zeros_like(cur_vec)

    # smoothing schedule for the soft‑max (ε grows from ε0 to εmax)
    eps_start = 5.0
    eps_end = 250.0

    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total_time = deadline - start_time

        # ----------- 4.1  Temperature / smoothing schedules -----------
        # exponential cooling for the Metropolis acceptance rule
        T = max(1e-8, cur_J * (1.0 - elapsed / total_time))

        # ε evolves linearly (could also be exponential)
        eps = eps_start + (eps_end - eps_start) * (elapsed / total_time)

        # ----------- 4.2  Compute convolution and soft‑max weights ----------
        A_fft = np.fft.rfft(cur_vec, fft_len_corr)   # length L ≥ n+conv_len‑1
        conv_full = np.fft.irfft(A_fft * A_fft, n=fft_len_corr)[:conv_len]
        M = conv_full.max()

        # soft‑max weights that concentrate on the active set
        # w_k = exp(‑ε·(M‑c_k)/M),  0 ≤ w ≤ 1, sum w = 1
        diff = M - conv_full
        if M == 0:
            w = np.full_like(diff, 1.0 / diff.size)
        else:
            w = np.exp(-eps * diff / M)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

        # ----------- 4.3  Gradient of the smoothed objective ----------
        # correlation between a and the weight vector w
        w_rev = w[::-1]
        W_fft = np.fft.rfft(w_rev, fft_len_corr)
        corr = np.fft.irfft(A_fft * W_fft, n=fft_len_corr)
        start_idx = conv_len - 1                     # slice that yields length n
        grad = 2.0 * corr[start_idx:start_idx + n]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step size (decays like 1/sqrt(t))
        step = 0.8 / np.sqrt(iteration + 1.0)

        prop = cur_vec - step * momentum
        prop = proj_simplex(prop)

        # ----------- 4.4  Evaluate proposal ----------
        A_fft_prop = np.fft.rfft(prop, fft_len_conv)
        conv_prop = np.fft.irfft(A_fft_prop * A_fft_prop, n=fft_len_conv)[:conv_len]
        J_prop = float(2.0 * n * conv_prop.max())

        # ----------- 4.5  Metropolis acceptance ----------
        accept = False
        if J_prop < cur_J:
            accept = True
        else:
            # probability of accepting a worse move
            prob = np.exp(-(J_prop - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = prop
            cur_J = J_prop
            if cur_J < best_J_global - 1e-12:
                best_J_global = cur_J
                best_vec_global = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J_global:.6f}")

        # ----------- 4.6  Stagnation handling ----------
        if now - last_improve > 20.0:
            # Small jitter centred on the best solution so far
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = proj_simplex(best_vec_global * noise)
            # recompute its objective
            A_fft = np.fft.rfft(cur_vec, fft_len_conv)
            conv_cur = np.fft.irfft(A_fft * A_fft, n=fft_len_conv)[:conv_len]
            cur_J = float(2.0 * n * conv_cur.max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – best J = {best_J_global:.6f}")

        iteration += 1

    # ---------- 5. Final sanitisation ----------
    best_vec_global = np.clip(best_vec_global, 0.0, 1000.0)
    total = best_vec_global.sum()
    if total == 0.0:
        best_vec_global[:] = 1.0 / n
    else:
        best_vec_global /= total

    # Optional sanity check (not required for the framework)
    # print("Final J =", compute_J(best_vec_global, n, fft_len_conv, conv_len))

    return [float(v) for v in best_vec_global]
