"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Choose a problem size *n* (default 8192, or the length of an existing
   solution if it is already available).  A probability vector
   (non‑negative entries summing to 1) represents the step‑function;
   scale invariance of the objective guarantees that this normalisation
   does not affect the score.

2. Build a diverse portfolio of initial vectors (uniform, geometric,
   ramps, Gaussians, Dirichlet draws, sparse masks, and the previous best
   solution if present).  Pick the one with the smallest true objective
   as the starting point.

3. Optimise the smoothed ``max`` objective
   f(a)= (1/β)·log Σ exp(β·(a*a)) using projected Adam on the log‑space
   representation (θ_i = log a_i).  The gradient of f is obtained from a
   weighted correlation of a with the soft‑max weights on the convolution.
   After each update we exponentiate, project onto the probability simplex,
   and recompute θ = log a.

4. The temperature β starts small (β≈1) and is increased multiplicatively
   each iteration up to β_max≈2000, turning the smoothed objective into
   the true max.  The Adam learning‑rate decays like 1/√t.

5. If no improvement is observed for 30 s the current iterate is jittered
   (multiplicative log‑normal noise) and the Adam moments are reset – a
   cheap way to escape plateaus.

6. The loop runs until the time budget expires (leaving a 5 s safety
   margin).  The best vector seen is returned after a final simplex
   projection and clipping to the allowed range.

The method combines smooth‑max gradients, adaptive step‑sizes (Adam) and
periodic stochastic restarts, and typically beats the previous
exponentiated‑gradient implementation within the same time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to one).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Stable soft‑max weights for a vector of convolutions.
    Returns a vector w with sum(w) == 1.
    """
    shift = conv.max()
    # exponentials are never larger than 1 because of the shift
    exp_vals = np.exp(beta * (conv - shift))
    return exp_vals / exp_vals.sum()

# Main optimisation routine – required entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before
    the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size
    # --------------------------------------------------------------
    # default: a reasonably large power of two (still cheap for FFT)
    n = 8192
    # if a previous construction exists and its length is sensible,
    # reuse it to keep a similar scale.
    if "height_sequence_1" in globals():
        try:
            n0 = int(len(globals()["height_sequence_1"]))
            if 256 <= n0 <= 16384:
                n = n0

    conv_len = 2 * n - 1

    init_cands = []

        v /= v.sum()

        init_cands.append(mask)

    # --------------------------------------------------------------

        J = float(2.0 * n * conv.max())
        if J < best_J:

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    eps = 1e-12
    theta = np.log(cur_vec + eps)          # log‑space parameters

    v_hat = np.zeros_like(theta)           # second moment (named v_hat to avoid clash)
    beta1, beta2 = 0.9, 0.999

            theta = np.log(cur_vec + eps)

        # ---- increase smoothing parameter gradually ----
        if beta < beta_max:

        # ---- soft‑max weights for the current convolution ----
        w = _softmax_weights(conv, beta)           # shape (conv_len,)

        # ---- gradient of the smoothed max (2·(w⋆a)) ----
        W_hat = np.fft.rfft(w, fft_len)

        v_corr = v_hat / (1.0 - beta2 ** iteration)

        lr_t = base_lr / (np.sqrt(iteration) + 1.0)

    # 5. Return the best solution (final safety clipping & normalisation)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    if total <= 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
