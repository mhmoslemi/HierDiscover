    conv = np.fft.irfft(X * Y, n=fft_len)
    return conv[:len(x) + len(y) - 1]

"""
Hybrid smoothed‑gradient optimizer for the step‑function objective.

Core ideas
----------

* Work on the probability simplex (∑a_i = 1, a_i ≥ 0).  The objective
  J(a)=2·n·max(a∗a) is invariant under scaling, so we keep the vector
  normalised after each update.

* Approximate the non‑smooth max by a soft‑max (log‑sum‑exp) with a
  decreasing temperature β.  For a given temperature the gradient of
  the smoothed objective can be written analytically as a convolution,
  which is evaluated efficiently with real FFTs.

* Perform an additive gradient step (with normalisation via Euclidean
  projection onto the simplex) and occasionally inject random mass‑
  transfer moves to help escape shallow plateaus.

* Run many short independent restarts (random Dirichlet draws,
  jittered copies of the current best, and the previously discovered
  construction) until the supplied time budget expires.  The globally
  best sequence seen is returned.

The method is deliberately different from the previous exponentiated‑
gradient + active‑set routine and tends to locate better minima on the
same time budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of real vectors x and y using a real FFT."""
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    conv_len = len(x) + len(y) - 1
    return conv[:conv_len]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w | w_i ≥ 0, Σ w_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # if everything is non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            # fresh Dirichlet draw (moderate concentration)
            cur = rng.dirichlet(np.ones(n) * 0.6).astype(np.float64)

        # ----- 4.2 inner gradient descent for this run -----
        max_iters = 5000
        last_improve = time.monotonic()
        for it in range(max_iters):
            now = time.monotonic()
            if now >= deadline:
                break

            # convolution a * a
            conv = _conv_fft(cur, cur, fft_len_a)
            max_conv = conv.max()
            cur_J = 2.0 * n * max_conv

            # update global best if improved
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_seq = cur.copy()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

            # ----- compute smoothed gradient (soft‑max) -----
            # linear temperature schedule inside a run
            beta = max(beta_end,
                       beta_start * (1.0 - it / max_iters))

            # softmax over convolution entries (numerically safe)
            shift = conv - max_conv                    # ≤ 0
            w = np.exp(shift / beta)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.shape[0])
            else:
                w /= w_sum                              # now ∑ w = 1

            # gradient = 2 * (w ⊛ a_rev)  (⊛ denotes convolution)
            a_rev = cur[::-1]
            grad_full = _conv_fft(w, a_rev, fft_len_w)   # length 3n‑2
            grad = 2.0 * grad_full[n - 1:n - 1 + n]      # extract n‑vector

            # ----- gradient step with simplex projection -----
            gnorm = np.linalg.norm(grad)
            if gnorm > 0.0:
                eta = step0 / np.sqrt(it + 1.0)          # decreasing step
                cur = cur - eta * grad / gnorm
                cur = _project_simplex(cur)

            # ----- occasional random mass‑transfer move (exploration) -----
            if rng.random() < 0.01:
                # pick donor proportional to current mass
                donor = rng.choice(n, p=cur)
                # pick receiver favouring low‑mass locations
                inv = cur.max() - cur + 1e-12
                receiver = rng.choice(n, p=inv / inv.sum())
                delta = min(cur[donor], rng.random() * 0.05 * cur[donor])
                cur[donor] -= delta
                cur[receiver] += delta
                # re‑normalise (sum is unchanged but guard against rounding)
                cur = cur / cur.sum()

            # ----- stagnation handling inside a run -----
            if now - last_improve > 15.0:
                # apply a fresh jitter around the best found so far
                jitter = np.exp(rng.normal(0.0, 0.2, size=n))
                cur = best_seq * jitter
                cur = cur / cur.sum()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] Restart (jitter) – best J = {best_J:.6f}")

        # (end of inner loop)

    # ------------------------------------------------------------------
    # 5. Return the best sequence (clip to permissible range, renormalise).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / n
    else:
        best_seq /= total

    return [float(v) for v in best_seq]
