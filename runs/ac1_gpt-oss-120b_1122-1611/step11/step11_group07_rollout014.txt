def smooth_max_gradient(a, fft_len, conv_len, alpha):
    # compute conv using _conv_full
    conv = _conv_full(a, fft_len, conv_len)   # length conv_len
    maxc = conv.max()
    # compute softmax weights
    # shift for numerical stability
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum   # probabilities p_k
    # compute gradient via correlation
    # compute correlation length
    corr_len = conv_len + a.size - 1   # 3n-2
    L = 1 << ((corr_len - 1).bit_length())
    # FFTs
    FA = np.fft.rfft(a, L)
    FW = np.fft.rfft(w, L)
    corr = np.fft.irfft(FW * np.conj(FA), n=L)   # linear correlation (circular with padding)
    # slice needed part
    start = a.size - 1
    grad = 2.0 * corr[start:start + a.size]
    return conv, maxc, grad

    scored.sort(key=lambda x: x[0])
    best_overall_vec = scored[0][1].copy()
    best_overall_J = scored[0][0]

    print(f"[0.00s] Initial best J = {best_overall_J:.6f}")

    # Phase 1: short refinement on top K candidates

"""
Hybrid smooth‑max gradient descent + random exchange optimiser.

* Works on the probability simplex (∑a = 1, a_i ≥ 0) – the evaluation
  J(a) = 2·n·max(conv(a,a)) is invariant under scaling, so we keep the
  vector normalised.
* Instead of the hard max we minimise a smooth approximation
      f_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))
  whose gradient can be written in closed form and evaluated
  efficiently with an FFT based correlation.
* α is increased slowly (1 → 400) so the optimiser first sees a
  smoothed landscape and then focuses on the true maximum.
* After every 100 gradient steps a cheap “exchange” move (transfer a
  random fraction of mass between two coordinates) is performed to
  escape plateaus.
* The algorithm is repeatedly (re)started from a small set of diverse
  seed vectors (uniform, U‑shaped, Gaussian, Dirichlet draws and the
  previously best sequence if available).  The best vector seen so far
  is always kept and perturbed with a log‑normal jitter when the search
  stagnates.
* All operations are O(n log n) thanks to FFTs, allowing several
  thousand iterations inside the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – pure functions, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm from:
    Wang, Weiran, and Miguel Á. Carreira‑Perpiñán. “Projection onto
    the probability simplex: An efficient algorithm with a simple proof,
    and an application.” (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal (or v is zero); return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Exact objective J = 2·n·max(conv(a,a)) for a vector a already on the simplex.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def smooth_max_gradient(
    a: np.ndarray,
    conv_len: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Returns (conv, max_conv, grad) where

        conv      = full convolution a*a,
        max_conv  = max(conv),
        grad      = ∂  ( (1/α)·log Σ exp(α·conv_k) ) / ∂a   (≈ sub‑gradient of the max).

    The gradient is computed via a correlation
        grad_j = 2 Σ_k p_k·a_{k‑j},
    where p_k are the soft‑max weights over the convolution entries.
    All operations are performed with FFTs.
    """
    # --- 1) convolution and soft‑max weights ---------------------------------
    conv = _conv_full(a, conv_fft_len, conv_len)          # length = 2*n-1
    max_c = conv.max()
    # stable soft‑max of the convolution entries
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum          # probabilities – the smooth max

    # --- 2) correlation p ⋆ a  (linear, not circular) -----------------------
    # linear correlation length = conv_len + n - 1 (= 3n-2)
    corr_len = conv_len + a.size - 1
    L = corr_fft_len                     # already a power of two ≥ corr_len
    FA = np.fft.rfft(a, L)
    FP = np.fft.rfft(p, L)
    # correlation via FFT:  (p ⋆ a) = irfft( FP * conj(FA) )
    corr = np.fft.irfft(FP * np.conj(FA), n=L)
    # keep the linear part only (first corr_len entries)
    corr = corr[:corr_len]

    # gradient = 2 * (p ⋆ a) evaluated at lags 0 … n‑1
    start = a.size - 1                # offset where lag 0 appears
    grad = 2.0 * corr[start : start + a.size]

    return conv, max_c, grad


def refine_candidate(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    conv_fft_len: int,
    corr_fft_len: int,
    rng: np.random.Generator,
    time_budget: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑descent refinement of a single starting vector.
    Uses a gradually increasing smoothing parameter α and occasional
    exchange moves.  Returns the best vector found and its exact J.
    """
    # ensure we are on the simplex
    a = proj_simplex(a0.astype(np.float64, copy=False))
    start = time.monotonic()
    best = a.copy()
    conv, max_c, grad = smooth_max_gradient(
        a, conv_len, conv_fft_len, corr_fft_len, alpha=1.0
    )
    best_J = float(2.0 * n * max_c)

    # hyper‑parameters
    step0 = 0.3                     # initial step size scale
    alpha = 1.0
    alpha_max = 400.0
    it = 0

    while time.monotonic() - start < time_budget:
        # increase smoothing slowly
        alpha = min(alpha_max, alpha * 1.002)

        # gradient step
        conv, max_c, grad = smooth_max_gradient(
            a, conv_len, conv_fft_len, corr_fft_len, alpha
        )
        step = step0 / np.sqrt(it + 1.0)
        a_proposed = a - step * grad
        a_proposed = proj_simplex(a_proposed)

        # evaluate the *true* objective (hard max) for acceptance
        conv_p, max_c_p, _ = smooth_max_gradient(
            a_proposed, conv_len, conv_fft_len, corr_fft_len, alpha
        )
        J_p = float(2.0 * n * max_c_p)

        # Metropolis style acceptance (temperature decays with iterations)
        accept = False
        if J_p <= best_J:
            accept = True
        else:
            # very low temperature – we still allow occasional uphill jumps
            temp = max(1e-4, 0.01 / np.sqrt(it + 1))
            prob = np.exp(-(J_p - best_J) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_proposed
            if J_p < best_J - 1e-12:
                best_J = J_p
                best = a.copy()
                # small console feedback
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s]   refine improve J={best_J:.6f}")

        # --- occasional exchange move (large‑scale re‑balancing) ------------
        if it % 100 == 0 and it > 0:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(a[i], rng.random() * 0.05)       # move up to 5 % of mass
            a[i] -= delta
            a[j] += delta
            a = proj_simplex(a)

        it += 1

    return best, best_J


def propose_candidate(
    seed: int = 4242,
    budget_s: float = 1000.0,
    **unused_kwargs,
) -> list[float]:
    """
    Main entry point – returns a list of non‑negative heights that (hopefully)
    minimise ``evaluate_sequence`` within the 1000 s budget.
    The routine builds a diverse pool of seed vectors, runs a quick
    gradient refinement on the top few, then performs a longer refinement
    on the best solution found so far.  Random log‑normal jitter is applied
    whenever the search stagnates for a while.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. problem size (use previous solution if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    conv_fft_len = _next_pow2(conv_len)                     # for a*a
    corr_fft_len = _next_pow2(conv_len + n - 1)              # for correlation p⋆a

    # --------------------------------------------------------------
    # 2. build a diverse set of initial candidates (all on the simplex)
    # --------------------------------------------------------------
    init_cands: list[np.ndarray] = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (very close to uniform)
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (up / down)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells of various widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 20.0, n / 15.0, n / 10.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # U‑shaped distributions (large mass at the ends)
    dist = np.abs(x - centre) + 1e-9
    for p in (0.5, 1.0, 2.0):
        v = dist ** p
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – several concentrations
    for alpha in (0.2, 0.5, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best solution (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. quick evaluation of seeds – keep the best few
    # --------------------------------------------------------------
    scored: list[tuple[float, np.ndarray]] = []
    for cand in init_cands:
        J = compute_J(cand, n, conv_fft_len, conv_len)
        scored.append((J, cand))
    scored.sort(key=lambda t: t[0])

    best_vec = scored[0][1].copy()
    best_J = scored[0][0]
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Phase‑1: short refinements on the top 3 seeds
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep 5 s safety margin
    phase1_budget = min(30.0, (deadline - start_time) * 0.15)

    for rank in range(min(3, len(scored))):
        if time.monotonic() >= deadline:
            break
        cand = scored[rank][1]
        # split the Phase‑1 budget evenly among the considered seeds
        cand_best, cand_J = refine_candidate(
            cand,
            n,
            conv_len,
            conv_fft_len,
            corr_fft_len,
            rng,
            time_budget=phase1_budget / 3,
        )
        if cand_J < best_J:
            best_J, best_vec = cand_J, cand_best.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Phase‑1 improvement J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Phase‑2: long refinement on the current best vector,
    #    with occasional jitter when stagnating
    # --------------------------------------------------------------
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        now = time.monotonic()
        # allocate a modest time slice for each refinement round
        slice_budget = min(20.0, deadline - now)
        if slice_budget <= 0.0:
            break

        cur_best, cur_J = refine_candidate(
            best_vec,
            n,
            conv_len,
            conv_fft_len,
            corr_fft_len,
            rng,
            time_budget=slice_budget,
        )

        if cur_J < best_J - 1e-12:
            best_J, best_vec = cur_J, cur_best.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")

        # stagnation detection – if no improvement for ~15 s, jitter
        if now - last_improve > 15.0:
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            best_vec = proj_simplex(best_vec * jitter)
            best_J = compute_J(best_vec, n, conv_fft_len, conv_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Jitter – reset J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Final sanitisation (clip to allowed range, renormalise)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
