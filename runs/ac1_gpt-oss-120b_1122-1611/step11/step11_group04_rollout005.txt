"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence`` using a smooth‑max gradient descent.

Algorithm
---------

* Work on the probability simplex (∑a_i = 1, a_i ≥ 0).  The objective
  ``evaluate_sequence`` is invariant under scaling, so we keep the
  vector normalized after each update.

* Approximate the maximum of the auto‑convolution by a
  log‑sum‑exp (smooth max) with a temperature ``alpha``.
  The gradient of the smooth max can be written in closed form as a
  convolution of the current vector with a weight vector obtained from
  the softmax of the convolution values.

* Perform a multiplicative (exponentiated‑gradient) update
  ``a_i ← a_i·exp( - η·grad_i )`` and renormalise to the simplex.
  The learning rate η decays as 1/√t and the smoothing parameter α
  slowly grows, so early iterations move the mass broadly and later
  iterations concentrate on the true peak.

* Occasionally inject random “jitter’’ (multiplicative Gaussian noise)
  when no improvement has been seen for a while; this helps escape
  plateaus.

* Keep track of the best vector seen so far and return it (clipped to
  the limits imposed by ``evaluate_sequence``).

The routine runs for at most ``budget_s`` seconds and returns the best
sequence it has found.
"""

import time
import numpy as np

# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int, out_len: int) -> np.ndarray:
    """Full linear convolution of *a* with itself using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:out_len]
    return conv


def _conv_full(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of *x* and *y* (real FFT, length = len(x)+len(y)-1)."""
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    full = np.fft.irfft(X * Y, n=fft_len)
    return full[: len(x) + len(y) - 1]


def _smooth_max_gradient(a: np.ndarray,
                         conv_a: np.ndarray,
                         alpha: float,
                         grad_fft_len: int,
                         conv_len: int) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max  L(a)=log‑sum‑exp(alpha·conv(a,a))
    with respect to *a* (up to a constant factor alpha, which is omitted).

    Parameters
    ----------
    a         : current probability vector (sum == 1)
    conv_a    : full self‑convolution of *a* (length 2·n‑1)
    alpha     : smoothing parameter (α>0)
    grad_fft_len : FFT length for the gradient convolution (≥ 3·n‑2)
    conv_len  : 2·n‑1 (length of ``conv_a``)

    Returns
    -------
    grad : ndarray, shape (n,)
        Approximate sub‑gradient of ``max(conv_a)``.
    """
    # soft‑max weights (numerically stable)
    max_v = conv_a.max()
    # exp(alpha*(conv - max)) ≤ 1, no overflow
    w = np.exp(alpha * (conv_a - max_v))
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / conv_len
    else:
        w /= w_sum

    # reversed weight vector for the correlation
    w_rev = w[::-1]

    # convolution a * w_rev   → length 3·n‑2
    # use the pre‑allocated FFT length 'grad_fft_len'
    conv_rev = _conv_full(a, w_rev, grad_fft_len)

    # the needed slice is indices [n-1, 2·n-2] (length n), reversed.
    n = a.shape[0]
    sub = conv_rev[n - 1 : 2 * n - 1]          # length n
    grad = 2.0 * sub[::-1]                    # factor 2 from ∂conv/∂a
    return grad


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes
    ``evaluate_sequence``.  The function respects the given time budget
    and returns the best vector (as a list of floats) found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1                       # length of a*a
    fft_len = _next_pow2(conv_len)             # for self‑convolution
    grad_fft_len = _next_pow2(3 * n - 2)       # for a * w_rev

    # --------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Half‑sine shape (non‑negative)
    v = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    v = np.maximum(v, 0.0)
    v /= v.sum()
    init_cands.append(v)

    # Tent (triangular) shape
    v = np.minimum(np.arange(n, dtype=np.float64) + 1,
                   n - np.arange(n, dtype=np.float64))
    v = np.maximum(v, 0.0)
    v /= v.sum()
    init_cands.append(v)

    # Dirichlet draws (moderate and strong concentration)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # reuse previous construction if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (using the fast J = 2·n·max_conv)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_self(cand, fft_len, conv_len)
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # tiny safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # hyper‑parameter schedules
    lr0 = 0.8                     # initial learning rate for exp‑gradient
    alpha = 5.0                  # initial smoothing
    alpha_max = 200.0
    jitter_sigma = 0.25          # stddev of multiplicative Gaussian jitter
    patience = 30.0              # seconds without improvement before jitter

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- compute current convolution & J -----
        conv = _conv_self(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        # ----- update global best -----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation handling (jitter) -----
        if now - last_improve > patience:
            # multiplicative Gaussian jitter centred at the best solution
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cur_vec = best_vec * noise
            cur_vec = np.clip(cur_vec, 0.0, None)
            cur_vec /= cur_vec.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ----- gradient of smooth max -----
        # increase smoothing parameter slowly (log‑linear schedule)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.001)

        # compute gradient (no extra FFT for A – reuse in _conv_self)
        grad = _smooth_max_gradient(cur_vec, conv, alpha,
                                    grad_fft_len, conv_len)

        # ----- propose a new point -----
        # multiplicative (exponentiated‑gradient) update
        step = lr0 / np.sqrt(iteration + 1.0)   # decaying step
        factor = np.exp(-step * grad)          # element‑wise
        proposal = cur_vec * factor
        # avoid numerical under‑flow
        proposal = np.maximum(proposal, 1e-30)
        proposal_sum = proposal.sum()
        if proposal_sum == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= proposal_sum

        # ----- evaluate proposal -----
        prop_conv = _conv_self(proposal, fft_len, conv_len)
        prop_J = 2.0 * n * prop_conv.max()

        # ----- acceptance (simple greedy + occasional uphill) -----
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # weak Metropolis acceptance with a linearly cooling temperature
            T_start = best_J * 0.10
            T_end = 1e-9
            T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best solution (final clipping & normalisation)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # The caller will feed the list to ``evaluate_sequence``; we keep
    # the vector as pure Python floats to avoid accidental NumPy scalars.
    return [float(v) for v in best_vec]
