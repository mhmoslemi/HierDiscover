"""
Sub‑gradient descent on the simplex with exponentiated updates.

We minimise the quantity
        J(a) = 2·n·max(conv(a,a)) / (∑a_i)²
for a non‑negative step function a (the length n is chosen from the previous
construction if available, otherwise 8192).  The algorithm works on the
probability simplex (∑a_i = 1) – the objective is invariant under scaling as
long as the values stay in the allowed range [0,1000] – and repeatedly
reduces the current maximum entry of the auto‑convolution by a sub‑gradient
step:

* Compute the full convolution a*a via a real FFT.
* Determine the index (or a small set of indices) where the convolution attains
  its maximum.
* The sub‑gradient of max(conv) at a is
        g_i = 2·∑_{k∈argmax} a_{k‑i}   (only where the index k‑i is inside 0…n‑1)
  which can be obtained in O(n·|argmax|) time.
* Perform an exponentiated‑gradient move   a ← a·exp(‑η·g)   and project back
  onto the simplex (Euclidean projection).  η decays as 1/√t.
* A simple Metropolis acceptance with a linearly cooling temperature
  allows occasional uphill moves to escape plateaux.
* If no improvement is seen for 30 s a multiplicative Gaussian jitter is
  applied to the current best solution to restart the walk.
* Several diverse initial points (uniform, Dirichlet, Gaussian bells,
  random binary supports and the previous best) are evaluated and the best
  is used as a start.

The routine respects the 1000 s budget, prints progress and finally returns
the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works only on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of a with itself using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    # keep only the first 2·len(a)‑1 entries (the linear convolution length)
    return conv[: 2 * a.size - 1]


def compute_J(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Return the objective value J = 2·n·max(conv) / sum_a²."""
    return float(2.0 * n * conv.max() / (sum_a * sum_a))


def grad_max_convolution(a: np.ndarray, conv: np.ndarray,
                         argmax_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For every index k in argmax_idx we add 2·a_{k‑i} to g_i (if the index exists).
    """
    n = a.size
    g = np.zeros_like(a)
    for k in argmax_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


def jitter_vector(a: np.ndarray, rng: np.random.Generator,
                  sigma: float = 0.08) -> np.ndarray:
    """Multiplicative Gaussian jitter, followed by projection onto the simplex."""
    noisy = a * np.exp(rng.normal(0.0, sigma, size=a.shape))
    return proj_simplex(noisy)

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best normalised sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))           # keep it reasonable
        except Exception:
            n = 4096
    else:
        n = 8192

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a diverse pool of starting points (all on the simplex)
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws – from spiky to smooth
    for alpha in [0.1, 0.3, 0.7, 1.5]:
        for _ in range(2):
            init_candidates.append(rng.dirichlet(np.full(n, alpha)))

    # Gaussian bells of several widths (normalised)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 6.0, n / 4.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_candidates.append(v)

    # Random binary supports of various densities (equal weight on support)
    sqrt_n = int(np.sqrt(n))
    for factor in [1.0, 1.3, 1.6, 2.0]:
        m = max(2, int(factor * sqrt_n))   # size of support
        for _ in range(2):
            idx = rng.choice(n, size=m, replace=False)
            v = np.zeros(n, dtype=np.float64)
            v[idx] = 1.0 / m
            init_candidates.append(v)

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v = v / s
        if v.size != n:

                v = v[:n]
            else:
                v = np.pad(v, (0, n - v.size), mode="constant")
            v = proj_simplex(v)
        init_candidates.append(v)

    # --------------------------------------------------------------
    # 3. Evaluate all initial candidates, keep the best
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_candidates:
        conv = conv_fft(cand, fft_len)
        J = compute_J(conv, n, cand.sum())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop – sub‑gradient exponentiated descent
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # Metropolis temperature schedule (linear cooling)
    T_start = best_J * 0.08      # start with about 8 % of the current J
    T_end = 1e-9

    cur_vec = best_vec.copy()
    cur_sum = cur_vec.sum()
    cur_conv = conv_fft(cur_vec, fft_len)
    cur_J = compute_J(cur_conv, n, cur_sum)

    last_improve = start_time
    iteration = 0

    # step‑size parameters for exponentiated gradient
    step0 = 0.6                 # initial step factor (will be decayed)

    # tolerance for considering an index “maximal” (relative)
    rel_tol = 1e-12

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear cooling of the Metropolis temperature
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (
            elapsed / (deadline - start_time)
        )

        # ----------------------------------------------------------
        # 4.1  Evaluate current state (conv, J)
        # ----------------------------------------------------------
        cur_conv = conv_fft(cur_vec, fft_len)
        max_conv = cur_conv.max()
        cur_J = compute_J(cur_conv, n, cur_vec.sum())

        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4.2  Stagnation handling – jitter after 30 s without improvement
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            cur_vec = jitter_vector(best_vec, rng, sigma=0.07)
            last_improve = now
            print(
                f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}"
            )
            # recompute conv for the jittered vector in the next loop
            continue

        # ----------------------------------------------------------
        # 4.3  Compute sub‑gradient of max(conv) at the current vector
        # ----------------------------------------------------------
        # Identify all indices that are (almost) maximal
        max_val = max_conv
        # use a relative tolerance to capture ties caused by floating‑point noise
        argmax_idx = np.where(cur_conv >= max_val * (1.0 - rel_tol))[0]

        # gradient of the *max* term (not the full J – the scaling factor 2·n/∑a² is constant)
        g = grad_max_convolution(cur_vec, cur_conv, argmax_idx)

        # ----------------------------------------------------------
        # 4.4  Exponentiated‑gradient proposal
        # ----------------------------------------------------------

        factor = np.exp(-step * g)                      # always positive
        proposal = cur_vec * factor
        proposal = proj_simplex(proposal)                # back to simplex

        # ----------------------------------------------------------
        # 4.5  Evaluate proposal
        # ----------------------------------------------------------
        prop_conv = conv_fft(proposal, fft_len)
        prop_J = compute_J(prop_conv, n, proposal.sum())

        # ----------------------------------------------------------
        # 4.6  Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if prop_J < cur_J:

        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

        # ----------------------------------------------------------
        # 4.7  Early stop if target reached
        # ----------------------------------------------------------
        if best_J < 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # --------------------------------------------------------------
    # 5. Return the best solution (clip to allowed range, re‑normalise)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    # Convert to plain Python list of floats as required by the harness
    return [float(v) for v in best_vec]
