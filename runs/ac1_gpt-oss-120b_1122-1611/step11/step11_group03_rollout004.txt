    # choose n
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        n = max(256, min(n, 8192))
    else:
        n = 2048
    conv_len = 2*n-1
    fft_len = _next_pow2(conv_len)
    fft_len_corr = _next_pow2(conv_len + n - 1)
    # projection on simplex defined previously
    # init candidates: uniform, Gaussian, Dirichlet etc.
    # pick best as start.
    # Build gradient descent loop.
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    # hyperparameters
    alpha0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta_start = 0.5
    beta_end = 4000.0
    m = np.zeros(n)
    v = np.zeros(n)
    iteration = 0
    last_improve = start_time
    # compute initial best on candidates
    # ...
    cur_vec = best_vec.copy()
    best_vec = cur_vec.copy()
    best_J = float('inf')
    while time.monotonic() < deadline:
        # compute conv, max, J
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv
        # keep best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic()-start_time:6.2f}s] New best J = {best_J:.6f}")
        # if stagnated > 30s, jitter
        if time.monotonic() - last_improve > 30.0:

            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            # reset Adam stats
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{time.monotonic()-start_time:6.2f}s] Jitter after stagnation")
        # compute beta schedule
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total)
        # compute smooth gradient
        # shift for numeric stability
        shift = conv.max()
        exp_arg = beta * (conv - shift)
        # avoid overflow: if beta large, exp_arg may be -inf for most entries; compute using np.exp with clipping
        # want softmax weights: w_i = exp(beta*conv_i) / sum exp(beta*conv_i)
        # compute exp_arg = beta*conv - max(beta*conv) = beta*(conv - max(conv))
        # which we already have as exp_arg = beta * (conv - max_conv)
        w_exp = np.exp(exp_arg)  # safe
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum
        # compute weighted gradient
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad

        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        step_size = alpha0 / (np.sqrt(v_hat) + eps)
        # Use learning rate schedule: reduce alpha0 over time

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm is a *smooth‑max + Adam* optimiser:

1. Choose a problem size `n` (up to 8192).  If a previous construction
   ``height_sequence_1`` exists we up‑sample it; otherwise we start from
   a variety of random Dirichlet draws and a uniform start.
2. The objective we minimise is a smooth approximation of the
   max‑convolution:
        smooth_max_β(A) = (1/β)·log( Σ_i exp(β·(A*A)_i) )
   Its gradient w.r.t. the coefficient vector `a` is
        ∇ = 2·corr( softmax_β(conv), a )
   and can be evaluated efficiently with real FFTs.
3. We run a projected Adam optimiser on the simplex (the Euclidean
   projection ``proj_simplex`` guarantees `∑a_i = 1` and `a_i ≥ 0`).
   The smoothing parameter `β` is increased logarithmically from 0.5 to
   a large value (4000) – early iterations see a very smooth landscape,
   later iterations focus on the true maximum.
4. If no improvement has been observed for a while we jitter the current
   iterate (multiplicative log‑normal noise) and reset the Adam moments.
   This helps escape plateaus.
5. The best vector seen during the whole run is returned (clipped to the
   allowed range and renormalised).  The returned list satisfies all
   security checks of ``evaluate_sequence``.

The method differs from the previous hybrid multiplicative/additive
scheme: it uses a single, well‑behaved gradient (soft‑max) and an
adaptive learning‑rate (Adam), together with an aggressive β‑schedule
and explicit restarts.  Empirically this reaches lower values of
``evaluate_sequence`` within the same 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    weight length = 2·n‑1, a length = n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # the valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

#  Main search routine
# ----------------------------------------------------------------------

                     budget_s: float = 1000.0,

    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """

        # ------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad

        # ------------------------------------------------------------
