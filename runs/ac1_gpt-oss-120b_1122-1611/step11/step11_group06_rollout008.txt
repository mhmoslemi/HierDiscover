"""
Hybrid stochastic‑gradient + simulated‑annealing optimizer.

* Generates a diverse set of candidate step‑functions (uniform, geometric,
  power‑law, beta‑shaped, sinusoid, quadratic, Tukey windows, random
  Dirichlet draws, and a resized previous construction if available).

* Keeps the best candidate w.r.t.  J = 2·n·max(conv(a,a))  (the value returned
  by ``evaluate_sequence`` when the vector is normalized to sum = 1).

* Refines the current best with a multiplicative (exponentiated‑gradient)
  update based on a sub‑gradient of the active convolution entries.
  A simple linear temperature schedule together with occasional random
  jitter yields a simulated‑annealing acceptance rule, allowing the search
  to escape shallow plateaus.

* The routine respects the supplied time budget (≈1000 s) and returns the
  best sequence found.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Basic utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum value of the full linear convolution vec * vec."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def objective_from_vec(vec: np.ndarray, n: int,
                       fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(vec,vec)) (vec assumed to sum to 1)."""
    return 2.0 * n * full_conv_max(vec, fft_len, conv_len)


def normalize_and_objective(vec: np.ndarray, n: int,
                           fft_len: int, conv_len: int):
    """Clamp to ≥0, renormalise to sum = 1 and compute the objective."""
    v = np.maximum(vec, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.inf, None
    v = v / s
    J = objective_from_vec(v, n, fft_len, conv_len)
    return J, v


def resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.astype(np.float64).copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq).astype(np.float64)


# ----------------------------------------------------------------------
# Shape factories (parameterised families of step functions)
# ----------------------------------------------------------------------


def geometric_vec(n: int, r: float) -> np.ndarray:
    """Monotonically decaying geometric sequence."""
    v = r ** np.arange(n, dtype=np.float64)
    return v


def sym_geometric_vec(n: int, r: float) -> np.ndarray:
    """Symmetric sum of a forward and a reversed geometric decay."""
    g = geometric_vec(n, r)
    return g + g[::-1]


def powerlaw_vec(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i = 1/(i+1)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return 1.0 / (i ** p)


def beta_vec(n: int, alpha: float, beta_: float) -> np.ndarray:
    """Discretised Beta(α,β) shape on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1) ** (beta_ - 1.0)
    return left * right


def sinusoid_vec(n: int) -> np.ndarray:
    """Half‑period sinusoid (positive values)."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    return np.sin(np.pi * t)


def quadratic_vec(n: int) -> np.ndarray:
    """Quadratic hill: (i+1)*(n-i)."""
    i = np.arange(n, dtype=np.float64)
    return (i + 1.0) * (n - i)


def tukey_window_vec(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window (copied from the reference code)."""
    if alpha <= 0.0:
        return np.ones(n, dtype=np.float64)
    if alpha >= 1.0:
        return np.hanning(n)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(t)
    first = t < alpha / 2.0
    w[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
    last = t > 1.0 - alpha / 2.0
    w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
    return w


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def active_set_and_gradient(vec: np.ndarray, n: int,
                            fft_len: int, conv_len: int):
    """
    Compute the set of active convolution indices (those attaining the
    maximal value) and the corresponding sub‑gradient of max(conv).
    Returns (active_indices, gradient_array).
    """
    # full convolution via FFT
    conv_vals = np.fft.irfft(np.fft.rfft(vec, n=fft_len) ** 2,
                             n=fft_len)[:conv_len]
    max_c = conv_vals.max()
    tol = max(1e-12, max_c * 1e-12)
    active = np.where(conv_vals >= max_c - tol)[0]

    grad = np.zeros_like(vec)
    # accumulate contributions of each active shift
    for p in active:
        start = p - (n - 1)
        if start < 0:
            start = 0
        end = p if p < n else n - 1
        idx = np.arange(start, end + 1, dtype=np.int64)
        grad[idx] += 2.0 * vec[p - idx]
    return active, grad, max_c

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine problem size (length of the vector)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Track the global best
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    # ------------------------------------------------------------------
    # 3.  Broad initial portfolio (parameter sweep + random draws)
    # ------------------------------------------------------------------
    def try_candidate(v_raw: np.ndarray) -> None:
        nonlocal best_vec, best_J
        J, v = normalize_and_objective(v_raw, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = v
            print(f"[init] New best J = {best_J:.6f}")

    # uniform
    try_candidate(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (both monotone and symmetric)
    for r in np.linspace(0.5, 0.999, 60):
        try_candidate(geometric_vec(n, r))
        try_candidate(geometric_vec(n, r)[::-1])
        try_candidate(sym_geometric_vec(n, r))

    # power‑law shapes
    for p in np.linspace(0.1, 5.0, 50):
        try_candidate(powerlaw_vec(n, p))

    # beta‑shaped distributions
    for _ in range(40):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        try_candidate(beta_vec(n, a, b))

    # classic window functions
    windows = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),
    }
    for w in windows.values():
        if w.sum() > 0.0:
            try_candidate(w)

    # Tukey windows (alpha from 0 to 1)
    for alpha in np.linspace(0.0, 1.0, 11):
        try_candidate(tukey_window_vec(n, alpha))

    # sinusoid and quadratic shapes
    try_candidate(sinusoid_vec(n))
    try_candidate(quadratic_vec(n))

    # random Dirichlet draws (different concentration parameters)
    for conc in [0.3, 0.6, 1.0, 2.0, 5.0]:
        for _ in range(12):
            try_candidate(rng.dirichlet(np.full(n, conc)))

    # exponential edge blends (mix uniform + symmetric exponential)
    for _ in range(80):
        alpha = 10 ** rng.uniform(-3.0, 1.0)          # 0.001 … 10
        mix = rng.random()
        exp_vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
        exp_vec += np.exp(-alpha * (n - 1 - np.arange(n, dtype=np.float64)))
        exp_vec /= exp_vec.sum()
        v = (1.0 - mix) * np.full(n, 1.0 / n) + mix * exp_vec
        try_candidate(v)

    # previous construction, if it exists (resized if needed)
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = resize_sequence(old, n)
        try_candidate(v)

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Local refinement – exponentiated‑gradient + SA
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # temperature schedule (linear cooling)
    T_start = max(best_J * 0.2, 1e-6)   # start a modest temperature
    T_end = 1e-12

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # linear cooling
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * (elapsed / (deadline - start_time)))

        # --------------------------------------------------------------
        # 4.1  Compute active set and sub‑gradient for the current point
        # --------------------------------------------------------------
        active, grad, max_c = active_set_and_gradient(cur_vec, n, fft_len, conv_len)

        # --------------------------------------------------------------
        # 4.2  Propose a new point (multiplicative update + jitter)
        # --------------------------------------------------------------
        # step size decays slowly – we want to stay exploratory
        step = 0.30 / (1.0 + iteration * 1e-4)    # ~0.3 initially, very slow decay
        # avoid overflow in exp
        factor = np.exp(np.clip(-step * grad, -50.0, 50.0))

        # add a small Gaussian perturbation on the log‑scale
        factor *= np.exp(rng.normal(0.0, 0.02, size=n))

        proposal = cur_vec * factor

        # occasional blend with a fresh Dirichlet point (helps escape plateaus)
        if rng.random() < 0.06:
            eps = rng.uniform(0.0, 0.18)
            rand_vec = rng.dirichlet(np.ones(n))
            proposal = (1.0 - eps) * proposal + eps * rand_vec

        proposal = np.maximum(proposal, 0.0)
        s = proposal.sum()
        if s == 0.0:
            proposal = np.full(n, 1.0 / n)
        else:
            proposal /= s

        # --------------------------------------------------------------
        # 4.3  Evaluate the proposal
        # --------------------------------------------------------------
        prop_J = objective_from_vec(proposal, n, fft_len, conv_len)

        # --------------------------------------------------------------
        # 4.4  Metropolis acceptance rule
        # --------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:

            cur_J = prop_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = time.monotonic()
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

        # --------------------------------------------------------------
        # 4.5  Stagnation handling – jitter around the global best
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            jitter = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_vec * jitter
            cur_vec /= cur_vec.sum()
            last_improve = now

    # ------------------------------------------------------------------
    # 5. Return the best solution (clip, renormalise and cast to list)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
