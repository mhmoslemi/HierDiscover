"""
Propose a step‑function (non‑negative vector) that minimizes the
`evaluate_sequence` objective.

Algorithm
---------
1. Choose a problem size `n` (use the length of the previously found
   construction if available, otherwise 4096).

2. Initialise a small portfolio of diverse candidates (uniform,
   tapered windows, geometric decays, the previous best, etc.) and keep
   the best one according to the true objective
        J(v) = 2 * n * max(conv(v, v))          (∑v = 1).

3. Starting from the current best vector, run a *smooth‑max* gradient
   descent on the probability simplex:

   * The true max is approximated by
         τ(v) = (1/γ)·log∑_k exp(γ·conv_k(v)),
     where `conv(v)=v*v` (full linear convolution) and `γ` is
     increased gradually (soft → hard max).

   * The gradient of τ is
         ∇τ = 2·(v ⋆ w)      with   w_k = softmax_γ(conv_k(v)).
     The cross‑correlation `v ⋆ w` is computed efficiently by FFT as a
     convolution of `w` with the reversed vector `v[::-1]`.

   * A multiplicative (exponentiated‑gradient) update
         v ← v * exp(‑η·∇τ)          (η decays as 1/√t)
     followed by a Euclidean projection onto the probability simplex
     keeps the iterates feasible.

4. A simple simulated‑annealing acceptance rule (Metropolis) together
   with occasional random jitter (Dirichlet mixes) prevents premature
   stagnation.

5. The loop runs until the time budget (≈1000 s) is exhausted; the best
   vector seen is returned (clipped to the range required by
   `evaluate_sequence` and converted to a plain Python list).

The method mixes a smooth approximation of the max‑convolution objective
with a fast FFT‑based gradient, enabling many thousands of cheap
iterations while respecting the simplex constraints.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection requires a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or all non‑positive)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window."""
    if alpha <= 0.0:
        return np.ones(n, dtype=np.float64)
    if alpha >= 1.0:
        return np.hanning(n)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(t)
    first = t < alpha / 2.0
    w[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
    last = t > 1.0 - alpha / 2.0
    w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
    return w


def _full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum value of the full linear convolution vec * vec."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _objective(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(conv(vec,vec)) (∑vec = 1)."""
    return 2.0 * n * _full_conv_max(vec, fft_len, conv_len)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found within the
    time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = max(256, min(len(globals()["height_sequence_1"]), 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for self‑convolution
    fft_len_grad = _next_pow2(conv_len + n - 1)        # for cross‑correlation (≈3·n)

    # --------------------------------------------------------------
    # 2. Initial portfolio of candidates
    # --------------------------------------------------------------
    candidates: List[np.ndarray] = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous construction, if present
    if "height_sequence_1" in globals():
        v0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v0 = np.clip(v0, 0.0, None)
        if v0.sum() == 0.0:
            v0.fill(1.0 / n)
        else:
            v0 /= v0.sum()
        candidates.append(v0)

    # a few Tukey windows
    for alpha in np.linspace(0.0, 1.0, 7):
        w = tukey_window(n, float(alpha))
        if w.sum() > 0.0:
            candidates.append(w / w.sum())

    # geometric decays (both directions)
    for r in [0.99, 0.97, 0.94, 0.90]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)
        candidates.append(v[::-1])

    # random Dirichlet draws (various concentrations)
    for conc in [0.3, 0.6, 1.0, 2.0]:
        for _ in range(6):
            candidates.append(rng.dirichlet(np.full(n, conc)))

    # --------------------------------------------------------------
    # 3. Evaluate portfolio, keep the best
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    for v in candidates:
        J = _objective(v, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = v.copy()

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Smooth‑max gradient descent on the simplex
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # temperature schedule for Metropolis acceptance
    T_start = best_J * 0.1
    T_end = 1e-8

    # smooth‑max parameter schedule
    gamma = 5.0                     # start soft
    gamma_max = 800.0               # final (approximates hard max)
    gamma_mul = gamma_max ** (1.0 / (deadline - start_time))  # exponential increase

    iteration = 0
    last_improve = start_time
    stagnation_start = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # ---- 4.1 compute full convolution and soft‑max weights ----
        A = np.fft.rfft(cur_vec, n=fft_len)
        conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]

        max_conv = float(conv_full.max())
        cur_J = 2.0 * n * max_conv

        # record new global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            stagnation_start = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- 4.2 temperature (linear cooling) ----
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * elapsed / (deadline - start_time))

        # ---- 4.3 soft‑max over convolution ----
        # stability: shift by max before exponentiation
        shifted = gamma * (conv_full - max_conv)          # ≤ 0
        # avoid under‑flow for very negative values
        exp_shift = np.exp(shifted - np.max(shifted))
        w = exp_shift / exp_shift.sum()                  # soft‑max weights, sum to 1

        # ---- 4.4 gradient of the smooth objective ----
        # grad = 2 * (cur_vec ⋆ w)   (cross‑correlation)
        rev_vec = cur_vec[::-1]
        fw = np.fft.rfft(w, n=fft_len_grad)
        fr = np.fft.rfft(rev_vec, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fr, n=fft_len_grad)  # length = fft_len_grad
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]            # extract exactly n components

        # ---- 4.5 multiplicative (exponentiated‑gradient) update ----
        step = 0.5 / np.sqrt(iteration + 1.0)            # 1/√t schedule
        # protect against overflow/under‑flow in exp
        factor = np.exp(-step * grad)
        factor = np.clip(factor, 1e-12, 1e12)

        proposal = cur_vec * factor
        # re‑normalise to the simplex
        proposal = proj_simplex(proposal)

        # occasional random mixing to jump out of plateaus
        if rng.random() < 0.07:
            eps = rng.uniform(0.0, 0.12)
            rand_vec = rng.dirichlet(np.ones(n))
            proposal = proj_simplex((1.0 - eps) * proposal + eps * rand_vec)

        # ---- 4.6 evaluate proposal ----
        prop_J = _objective(proposal, n, fft_len, conv_len)

        # ---- 4.7 Metropolis acceptance ----
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ---- 4.8 stagnation handling ----
        if now - stagnation_start > 25.0:
            # apply a gentle random jitter around the current best
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            cur_vec = best_vec * jitter
            cur_vec = proj_simplex(cur_vec)
            stagnation_start = now
            # also increase the smooth‑max temperature a little (helps exploration)
            gamma = min(gamma * 1.2, gamma_max)

        # ---- 4.9 schedule updates ----
        gamma = min(gamma * gamma_mul, gamma_max)
        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best solution (clip, renormalise, cast to list)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s

    return [float(v) for v in best_vec]
