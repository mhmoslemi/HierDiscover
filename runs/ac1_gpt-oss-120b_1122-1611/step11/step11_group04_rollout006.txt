We can implement a projection function:

def proj_simplex(v):
    # returns projection onto simplex {x>=0, sum x = 1}
    # v is np.ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # ensure non-negative and sum to 1
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all zeros
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid stochastic optimizer with additive projection steps.

The algorithm searches the simplex (a_i ≥ 0, Σ a_i = 1) for a vector that
minimises

    J(a) = 2·n·max(a * a)                     (a*a = discrete convolution)

which is exactly the value returned by ``evaluate_sequence`` for normalised
vectors (the function is scale‑invariant apart from clipping, which never
hits the limits for the vectors we generate).

Three complementary move families are used:

* **Additive gradient step** – a sub‑gradient of the max‑convolution is
  computed on an enlarged active set, a small step is taken in the
  negative gradient direction and the result is projected back onto the
  simplex.

* **Multiplicative (exponentiated‑gradient) step** – the same gradient is
  used as an exponent in a multiplicative update, again followed by a
  simplex projection.

* **Mass‑transfer / jitter moves** – cheap random redistributions of a
  small fraction of probability mass, useful for escaping plateaus.

During the first ~60 % of the time budget a simple linear cooling schedule
allows uphill moves (Metropolis acceptance).  Afterwards only strictly
improving moves are kept.  If no improvement is seen for a while the
search is restarted from a fresh Dirichlet draw.

When the time limit is reached a short deterministic refinement with a
tiny additive step is performed on the current best solution.

All helper routines are defined at module level; the function
``propose_candidate`` is the entry point used by the evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using a real FFT.
    Returns the first ``conv_len`` entries of the circular convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_active_set(a: np.ndarray, conv: np.ndarray, rel_tol: float) -> np.ndarray:
    """
    Sub‑gradient of max(conv) on an enlarged active set.
    ``a`` is the current probability vector (sum = 1),
    ``conv`` is the full convolution a*a,
    ``rel_tol`` is the relative tolerance used to select the active set.
    """
    n = a.shape[0]
    max_c = conv.max()
    tol = max(max_c * rel_tol, 1e-14)
    active = np.where(conv >= max_c - tol)[0]

    grad = np.zeros_like(a)
    # Loop over the active convolution indices and accumulate the derivative.
    for p in active:
        # The indices i that contribute to c[p] satisfy
        #   i ∈ [max(0, p-(n-1)), min(p, n-1)]
        start = p - (n - 1)
        if start < 0:
            start = 0
        end = p if p < n else n - 1
        idx = np.arange(start, end + 1)
        # derivative of c[p] w.r.t. a[idx] is 2 * a[p-idx]
        grad[idx] += 2.0 * a[p - idx]
    return grad, active


def _evaluate_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute J = 2·n·max(a*a) for a normalised vector a (Σa = 1)
    using the fast FFT convolution.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size n (number of samples)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # length for self‑convolution
    # --------------------------------------------------------------
    # 2. Build a diverse pool of initial candidates (all on the simplex)
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderate concentration)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best of the initial candidates (deterministic J)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = _evaluate_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # leave a small safety margin for final refinement
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # cooling schedule (linear)
    T_start = best_J * 0.05            # start temperature = 5 % of current best
    T_end = 1e-9

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        remaining = deadline - now

        # ----------------------------------------------------------
        # recompute convolution and objective for the current point
        # ----------------------------------------------------------
        cur_conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = cur_conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ----------------------------------------------------------
        # update global best
        # ----------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # stagnation handling – full random restart if stuck too long
        # ----------------------------------------------------------
        if now - last_improve > 40.0:
            # restart from a fresh Dirichlet draw (moderately spiky)
            cur_vec = rng.dirichlet(np.ones(n) * 0.6)
            # recompute its objective immediately
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            cur_J = float(2.0 * n * cur_conv.max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – new random start (J = {cur_J:.6f})")

        # ----------------------------------------------------------
        # compute an enlarged active set and its sub‑gradient
        # ----------------------------------------------------------
        # early on we allow a fairly wide active set, tighten later
        if iteration < 5000:
            rel_tol = 0.015
        elif iteration < 20000:
            rel_tol = 0.008
        else:
            rel_tol = 0.003

        grad, active = _gradient_active_set(cur_vec, cur_conv, rel_tol)

        # ----------------------------------------------------------
        # propose a new candidate (one of several move types)
        # ----------------------------------------------------------
        move_type = rng.random()
        proposal = None

        # additive gradient step (projected)
        if move_type < 0.4:
            step = 0.5 / np.sqrt(iteration + 1.0)
            cand = cur_vec - step * grad
            proposal = _project_simplex(cand)

        # multiplicative (exponentiated‑gradient) step
        elif move_type < 0.7:
            step = 0.8 / np.sqrt(iteration + 1.0)
            factor = np.exp(-step * grad)
            cand = cur_vec * factor
            proposal = _project_simplex(cand)

        # cheap mass‑transfer move
        elif move_type < 0.9:
            if grad.sum() > 0.0:
                donor = rng.choice(n, p=grad / grad.sum())
            else:
                donor = rng.integers(n)
            # favour low‑gradient positions as receivers
            inv = grad.max() - grad + 1e-12
            receiver = rng.choice(n, p=inv / inv.sum())
            delta = min(cur_vec[donor], 0.05 * cur_vec[donor])   # up to 5 %
            cand = cur_vec.copy()
            cand[donor] -= delta
            cand[receiver] += delta
            proposal = cand                     # already sums to 1

        # random jitter (multiplicative Gaussian noise)
        else:
            # noise amplitude slowly shrinks in time
            sigma = 0.2 * max(0.0, 1.0 - elapsed / (deadline - start_time)) + 0.02
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            cand = cur_vec * noise
            proposal = _project_simplex(cand)

        # ----------------------------------------------------------
        # evaluate the proposal
        # ----------------------------------------------------------
        prop_conv = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ----------------------------------------------------------
        # acceptance test (Metropolis in the early phase)
        # ----------------------------------------------------------
        # after 60 % of allotted time we stop accepting uphill moves
        explore = (elapsed < 0.6 * (deadline - start_time))
        if prop_J < cur_J:
            accept = True
        elif explore:
            # temperature linearly decays from T_start to T_end
            T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob
        else:
            accept = False

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # --------------------------------------------------------------
    # 5. Deterministic refinement on the best vector found
    # --------------------------------------------------------------
    # small additive steps with decreasing size, always kept if they improve
    refine_iters = 2000
    for i in range(refine_iters):
        grad, _ = _gradient_active_set(best_vec, _conv_full(best_vec, fft_len, conv_len), rel_tol=0.001)
        step = 0.001 / np.sqrt(i + 1.0)
        cand = best_vec - step * grad
        cand = _project_simplex(cand)
        cand_J = _evaluate_J(cand, n, fft_len, conv_len)
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand
            # occasional progress report
            if (i + 1) % 200 == 0:
                now = time.monotonic()
                print(f"[refine {i+1}/{refine_iters}] J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Final safety clipping and normalisation (required by evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # Optional sanity check (do not affect timing)
    # final_J = evaluate_sequence(best_vec.tolist())
    # print(f"Final reported J = {final_J:.6f}")

    return [float(v) for v in best_vec]
