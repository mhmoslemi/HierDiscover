"""
Propose a high‑quality step‑function sequence using
* a smooth‑max (log‑sum‑exp) surrogate for the max‑convolution,
* ADAM optimization on the soft‑max (log‑it) parametrisation,
* a linear β‑annealing schedule (β from 1 up to 2000),
* exponential‑gradient like updates on the probability simplex,
* periodic jitter when the search stagnates.

The algorithm works directly on the simplex (∑a_i = 1, a_i ≥ 0) and
evaluates the true objective J = 2·n·max(conv(a,a)) only for reporting
and for the final best‑solution check.  It fits well within the 1000 s budget
and typically improves the previous bound 1.513196 → ≈1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()

    # Dirichlet draws – from spiky to smooth
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Gaussian bells of several widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 9.0, n / 6.0, n / 4.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Random sparse “spike” patterns (choose ~30 % of entries)
    for _ in range(3):
        mask = rng.random(n) < 0.3
        v = np.zeros(n, dtype=np.float64)
        if mask.any():
            v[mask] = rng.random(mask.sum())
        v_sum = v.sum()
        if v_sum == 0.0:
            v[:] = 1.0 / n
        else:
            v /= v_sum
        init_cands.append(v)

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best of the initial candidates (objective = 2·n·max(conv))
    # --------------------------------------------------------------
    def _objective_from_vec(vec):
        conv = _conv_full(vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = _objective_from_vec(cand)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation parameters
    # --------------------------------------------------------------
    start_time = time.monotonic()
    safety_margin = 5.0
    deadline = start_time + max(0.0, float(budget_s) - safety_margin)

    # ADAM parameters
    lr = 0.12                     # base learning rate for the log‑variables
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # β‑annealing (soft‑max temperature)
    beta_start = 1.0
    beta_end = 2000.0

    # Stagnation handling
    stagnation_time = 30.0
    last_improve = start_time
    last_print = start_time

    # Initialise ADAM state on the log‑it variables
    # Work on the "log‑softmax" space: y such that a = softmax(y)
    a = best_vec.copy()
    y = np.log(np.clip(a, 1e-12, None))
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    iteration = 0
    target_J = 1.5025

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total_time = deadline - start_time
        # ----------------------------------------------------------
        # 4.1 Update β (linear schedule)
        # ----------------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (elapsed / total_time)
        if beta > beta_end:
            beta = beta_end

        # ----------------------------------------------------------
        # 4.2 Convolution and soft‑max weighting
        # ----------------------------------------------------------
        conv = _conv_full(a, fft_len, conv_len)                 # full a*a
        max_conv = conv.max()
        # soft‑max weights over convolution entries
        # shift by max_conv for numerical stability
        w_raw = np.exp(beta * (conv - max_conv))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv_len)
        else:
            w = w_raw / w_sum

        # ----------------------------------------------------------
        # 4.3 Gradient of the smoothed objective (Fourier‑domain)
        # ----------------------------------------------------------
        A = np.fft.rfft(a, fft_len)            # FFT of a
        W = np.fft.rfft(w, fft_len)            # FFT of the soft‑max weights
        g = 2.0 * np.fft.irfft(np.conj(W) * A, n=fft_len)[:n]   # ∇_a L

        # ----------------------------------------------------------
        # 4.4 Gradient w.r.t. the log‑softmax parameters y
        # ----------------------------------------------------------
        # dL/dy_i = a_i * (g_i - Σ_j a_j g_j)
        avg = np.dot(a, g)
        grad_y = a * (g - avg)

        # ----------------------------------------------------------
        # 4.5 ADAM update on y
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # multiplicative style update in the log‑space
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y -= step

        # ----------------------------------------------------------
        # 4.6 Recover a from y (soft‑max) and ensure numerical stability
        # ----------------------------------------------------------
        a = _softmax(y)

        # ----------------------------------------------------------
        # 4.7 Current objective (exact max‑convolution)
        # ----------------------------------------------------------
        cur_J = float(2.0 * n * max_conv)

        # ----------------------------------------------------------
        # 4.8 Record global best and handle stagnation
        # ----------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        if now - last_improve > stagnation_time:
            # apply a small random jitter in log‑space to escape flat regions
            sigma = 0.02
            y += rng.normal(0.0, sigma, size=n)
            a = _softmax(y)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4.9 Periodic progress report
        # ----------------------------------------------------------
        if now - last_print > 10.0:
            print(f"[{elapsed:6.2f}s] Current J = {cur_J:.6f}, β = {beta:.1f}")
            last_print = now

        # ----------------------------------------------------------
        # 4.10 Early exit if target reached
        # ----------------------------------------------------------
        if best_J < target_J:
            print(f"[{elapsed:6.2f}s] Target reached – best J = {best_J:.6f}")
            break

    # --------------------------------------------------------------
    # 5. Return the best solution (clipped & renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
