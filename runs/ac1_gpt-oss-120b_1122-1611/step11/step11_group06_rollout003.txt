"""
Propose a step function with a low value of
evaluate_sequence() by alternating a linear‑programming (LP) refinement
and a short exponentiated‑gradient (EG) polishing phase.

*LP phase* – given a current shape f (∑f = 1, f ≥ 0) we solve

    min   t
    s.t.  (f ∗ g)_k ≤ t      for all k = 0 … 2n‑2
          ∑ g_i = 1
          g_i ≥ 0

which is a linear program in the variables g (the new shape) and the scalar
t.  The matrix that maps g → f∗g is a Toeplitz (convolution) matrix and is
built as a sparse CSR matrix.  Solving the LP yields a g that reduces the
peak of the cross‑convolution f∗g; we keep g only if its own autocorrelation
peak (computed with an FFT) also improves the objective.

*EG phase* – after a few LP steps we run a few hundred exponentiated‑gradient
updates (the same update that was used in the previous baseline).  This
provides a cheap local search that can sweep away small remaining plateaus.

The whole search respects the 1000 s budget, works on the probability
simplex (∑a_i = 1) and returns a list of non‑negative floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the linear convolution vec * vec."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def objective_from_vec(vec: np.ndarray, n: int,
                       fft_len: int, conv_len: int) -> float:
    """J = 2·n·max(conv(vec,vec))  (assumes sum(vec)=1)."""
    return 2.0 * n * full_conv_max(vec, fft_len, conv_len)


def tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window."""
    if alpha <= 0.0:
        return np.ones(n, dtype=np.float64)
    if alpha >= 1.0:
        return np.hanning(n)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(t)
    first = t < alpha / 2.0
    w[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
    last = t > 1.0 - alpha / 2.0
    w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
    return w


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n convolution matrix T such that
    T @ g  =  f * g   (full linear convolution, zero‑padded).
    The matrix is extremely sparse and is returned in CSR format.
    """
    n = f.shape[0]
    conv_len = 2 * n - 1
    rows = []
    cols = []
    data = []

    # each offset d corresponds to diagonal with constant value f[d]
    for d in range(n):
        # diagonal offset = -d  (col = row - d)
        row_start = d
        row_end = min(conv_len, n + d)          # exclusive
        length = row_end - row_start
        if length <= 0:
            continue
        rows.extend(range(row_start, row_end))
        cols.extend(range(row_start - d, row_end - d))
        data.extend([f[d]] * length)

    rows = np.asarray(rows, dtype=np.int32)
    cols = np.asarray(cols, dtype=np.int32)
    data = np.asarray(data, dtype=np.float64)
    T = sp.csr_matrix((data, (rows, cols)), shape=(conv_len, n))
    return T


def lp_step(current: np.ndarray) -> tuple[np.ndarray, float] | tuple[None, None]:
    """
    Solve the LP:
        min   t
        s.t.  (current * g)_k ≤ t      (all k)
              ∑ g_i = 1
              g_i ≥ 0
    Returns the optimal g (already sums to 1) and the associated bound t.
    If the solver fails, (None, None) is returned.
    """
    n = current.shape[0]
    conv_len = 2 * n - 1

    # Build the sparse Toeplitz matrix for the convolution with `current`.
    T = build_conv_matrix(current)                     # shape (conv_len, n)

    # Append the column for the variable t (coefficient –1 in each inequality).
    col_t = sp.csr_matrix((-np.ones(conv_len), (np.arange(conv_len), np.zeros(conv_len))),
                          shape=(conv_len, 1))
    A_ub = sp.hstack([T, col_t], format='csr')
    b_ub = np.zeros(conv_len)

    # Equality: sum(g) = 1  (t has zero coefficient)
    A_eq = sp.csr_matrix(np.hstack([np.ones(n), 0.0]).reshape(1, n + 1))
    b_eq = np.array([1.0])

    # Objective: minimise t
    c = np.concatenate([np.zeros(n), np.array([1.0])])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Solve with HiGHS (sparse‑aware).  No explicit time limit – a few seconds per call.
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:          # pragma: no cover
        return None, None

    if not res.success:
        return None, None

    g = res.x[:n]
    t = res.x[n]
    # Guard against numerical drift
    if g.sum() == 0.0:
        return None, None
    g = g / g.sum()
    return g, t


def eg_refine(start_vec: np.ndarray,
              n: int,
              fft_len: int,
              conv_len: int,
              rng: np.random.Generator,
              deadline: float) -> np.ndarray:
    """
    A short exponentiated‑gradient (multiplicative) refinement.
    The loop stops when the time budget expires.
    """
    cur = start_vec.copy()
    best = cur.copy()
    best_J = objective_from_vec(best, n, fft_len, conv_len)

    iteration = 0
    last_improve = time.time()

    while time.time() < deadline:
        # compute active set of the full convolution
        conv_vals = np.fft.irfft(
            np.fft.rfft(cur, n=fft_len) ** 2,
            n=fft_len)[:conv_len]
        max_c = conv_vals.max()
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv_vals >= max_c - tol)[0]

        # sub‑gradient of max(conv)
        grad = np.zeros_like(cur)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=np.int64)
            grad[idx] += 2.0 * cur[p - idx]

        # multiplicative (exponentiated) update
        step = 0.5 / np.sqrt(iteration + 1.0)
        factor = np.exp(-step * grad)
        proposal = cur * factor

        # occasional random Dirichlet blend
        if rng.random() < 0.08:
            eps = rng.uniform(0.0, 0.15)
            rand_vec = rng.dirichlet(np.ones(n))
            proposal = (1.0 - eps) * proposal + eps * rand_vec

        proposal = proj_simplex(proposal)

        prop_J = objective_from_vec(proposal, n, fft_len, conv_len)

        # simple Metropolis‑like acceptance
        if prop_J < best_J:
            best_J = prop_J
            best = proposal.copy()
            last_improve = time.time()
        if prop_J < objective_from_vec(cur, n, fft_len, conv_len):
            cur = proposal
        else:
            # occasional escape from stagnation
            if time.time() - last_improve > 20.0:
                noise = np.exp(rng.normal(0.0, 0.04, size=n))
                cur = best * noise
                cur = proj_simplex(cur)
                last_improve = time.time()

        iteration += 1
        if iteration >= 2000:      # safety stop for very long runs
            break

    return best

    # ------------------------------------------------------------------
    # 2. Build a modest portfolio of diverse starting points.
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (forward & reversed)
    for r in [0.905, 0.92, 0.94, 0.96, 0.98, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        candidates.append(v / v.sum())
        candidates.append(v[::-1] / v.sum())

    # power‑law shapes
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1, dtype=np.float64)
        v = 1.0 / (i ** p)
        candidates.append(v / v.sum())

    # classic window functions
    windows = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),
    }
    for w in windows.values():
        if w.sum() > 0.0:
            candidates.append(w / w.sum())

    # Tukey windows
    for alpha in np.linspace(0.0, 1.0, 11):
        w = tukey_window(n, alpha)
        if w.sum() > 0.0:
            candidates.append(w / w.sum())

    # random Dirichlet draws with various concentration parameters
    for conc in [0.3, 0.6, 1.0, 2.0]:
        for _ in range(6):
            candidates.append(rng.dirichlet(np.full(n, conc)))

    # the (possibly down‑sampled) historic best
    candidates.append(init_vec)

    # ------------------------------------------------------------------
    # 3. Pick the best among the portfolio.
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    def update_best(v: np.ndarray):
        nonlocal best_vec, best_J
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            return
        v = v / s
        J = objective_from_vec(v, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = v.copy()

    for cand in candidates:
        update_best(cand)

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. LP refinement loop.
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    iteration = 0
    last_improve = time.time()

    while time.time() < deadline:
        # Solve the LP for the current shape
        g, t_bound = lp_step(cur_vec)
        if g is None:
            # LP failed – exit the LP phase
            break

        # Evaluate the *true* objective of the candidate returned by the LP
        J_g = objective_from_vec(g, n, fft_len, conv_len)

        if J_g < best_J - 1e-12:
            best_J = J_g
            best_vec = g.copy()
            last_improve = time.time()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] LP improvement: J = {best_J:.6f}")

        # Accept the LP candidate if it improves the current objective
        if J_g < cur_J:
            cur_vec = g
            cur_J = J_g
        else:
            # If no improvement, try a small convex combination to keep moving
            alpha = 0.25
            blended = (1.0 - alpha) * cur_vec + alpha * g
            blended = blended / blended.sum()
            J_blend = objective_from_vec(blended, n, fft_len, conv_len)
            if J_blend < cur_J:
                cur_vec = blended
                cur_J = J_blend
            else:
                # Stagnation – inject a little random jitter
                if time.time() - last_improve > 30.0:
                    jitter = np.exp(rng.normal(0.0, 0.04, size=n))
                    cur_vec = cur_vec * jitter
                    cur_vec = proj_simplex(cur_vec)
                    last_improve = time.time()
        iteration += 1
        if iteration >= 30:          # keep LP phase short to reserve time for polishing
            break

    # ------------------------------------------------------------------
    # 5. Short exponentiated‑gradient polishing phase (uses remaining budget).
    # ------------------------------------------------------------------
    remaining_deadline = deadline
    if time.time() < remaining_deadline:
        best_vec = eg_refine(best_vec, n, fft_len, conv_len, rng, remaining_deadline)

    # ------------------------------------------------------------------
    # 6. Final sanitisation and return.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # Convert to plain Python list of floats as required by the evaluation harness
    return [float(v) for v in best_vec.tolist()]
