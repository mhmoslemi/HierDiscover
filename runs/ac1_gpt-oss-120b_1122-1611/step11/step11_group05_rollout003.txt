"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm works in two stages.

Stage 1 – fast Adam‑optimisation of a soft‑maxed approximation of the
∞‑norm of the autocorrelation.  A diverse portfolio of initial
probability vectors (uniform, Dirichlet draws, an interpolated copy of
the previous best construction) is projected onto the probability
simplex and refined.  The temperature‑like smoothing parameter β is
increased slowly so that the optimisation focuses on the true
maximum.

Stage 2 – long‑run simulated‑annealing on the original objective.
From the best vector found in stage 1 we perform random mass‑transfer
moves (pick i≠j, move a small δ from i to j) and accept them with a
Metropolis rule whose temperature decays linearly to near 0.  If the
search stagnates for >20 s a jitter restart (log‑normal multiplicative
noise) is performed.

Both stages respect the security checks in `evaluate_sequence`:
all entries are non‑negative, bounded, and the vector is renormalised
to sum 1 before returning.  The routine respects the 1000 s time
budget, prints intermediate best bounds and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to one.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights for a vector of convolutions."""
    shift = conv.max()
    exp_vals = np.exp(beta * (conv - shift))
    return exp_vals / exp_vals.sum()


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float,
                     fft_len_corr: int) -> np.ndarray:
    """
    Gradient of the smoothed max objective
        f(a) = (1/β)·log Σ exp(β·conv)
    w.r.t. a.  The gradient is 2·(w ⋆ a) where w are the soft‑max
    weights and ⋆ denotes linear cross‑correlation.
    """
    w = _softmax_weights(conv, beta)                 # shape (2n‑1,)
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    W_hat = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)   # linear correlation
    n = a.size
    start = n - 1                                      # index where shift = 0
    grad = 2.0 * corr[start:start + n]                 # shape (n,)
    return grad

# Main optimisation routine – required entry point
# ----------------------------------------------------------------------

                     budget_s: float = 1000.0,

    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (length of the step function)
    # --------------------------------------------------------------
    # If a previous construction exists, reuse its length as a guide,
    # otherwise pick a power‑of‑two size that is cheap for FFTs.
    if "height_sequence_1" in globals():

            n0 = int(len(globals()["height_sequence_1"]))

        v = np.maximum(v, 0.0)

        if s == 0.0:
