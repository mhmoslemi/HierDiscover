"""
Search for a non‑negative step function (list of heights) that minimizes

    J = 2·n·max(conv(a, a))   with   sum(a) = 1.

The method builds a rich initial portfolio (uniform, geometric, power‑law,
window functions, Dirichlet draws, previous best) and keeps the best
candidate.  It then refines this candidate with a stochastic gradient
descent on a smoothed version of the max‑convolution:

    f_ε(a) = ε * log‑sum‑exp( conv(a,a)/ε )

The gradient of f_ε is obtained via a single FFT‑based convolution,
the update follows the Adam scheme and the iterate is projected onto the
probability simplex.  A simple simulated‑annealing acceptance rule and
periodic jitter prevent premature convergence.  The routine stops
after the allotted time budget (≈1000 s) and returns the best sequence
found.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities – all functions are top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum value of the full linear convolution vec * vec."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def objective_from_vec(vec: np.ndarray, n: int,
                       fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(vec,vec)) (assumes sum(vec)=1)."""
    return 2.0 * n * full_conv_max(vec, fft_len, conv_len)


def softmax_weights(conv: np.ndarray, epsilon: float) -> np.ndarray:
    """Stable soft‑max of the convolution values with smoothing ε."""
    # shift by the maximum to avoid overflow
    max_c = conv.max()
    shifted = conv - max_c            # ≤ 0
    # exponentials are in [0, 1]; safe for very small ε
    exp_vals = np.exp(shifted / epsilon)
    total = exp_vals.sum()
    if total == 0.0:                  # numerical under‑flow – fall back to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


def gradient_from_weights(weight: np.ndarray,
                         a_rev: np.ndarray,
                         fft_len_grad: int,
                         n: int) -> np.ndarray:
    """
    Gradient of ε·log‑sum‑exp(conv/ε) with respect to a,
    given the soft‑max weights w (size 2·n‑1) and a_rev = a[::-1].

    Returns a vector of length n.
    """
    fw = np.fft.rfft(weight, n=fft_len_grad)
    fra = np.fft.rfft(a_rev, n=fft_len_grad)
    conv_full = np.fft.irfft(fw * fra, n=fft_len_grad)
    # slice the central n entries: indices n‑1 … n‑1+n‑1
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimizes ``evaluate_sequence``.  Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose the problem size (length of the vector).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)

    # for the gradient we need a longer FFT (conv_len + n - 1)
    conv_len_grad = conv_len + n - 1
    fft_len_grad = _next_pow2(conv_len_grad)

    # --------------------------------------------------------------
    # 2. Helper to evaluate a candidate and keep the global best.
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    def update_best(vec: np.ndarray) -> None:
        nonlocal best_vec, best_J
        # project onto simplex (ensures sum = 1, non‑negative)
        vec = proj_simplex(vec)
        J = objective_from_vec(vec, n, fft_len_conv, conv_len)
        if J < best_J:
            best_J = J
            best_vec = vec.copy()

    # --------------------------------------------------------------
    # 3. Broad initial portfolio (parameter sweep + random draws).
    # --------------------------------------------------------------
    # uniform
    update_best(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (both monotone and symmetric)
    for r in [0.905, 0.92, 0.94, 0.96, 0.98, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        update_best(v)
        update_best(v[::-1])

    # power‑law shapes
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1, dtype=np.float64)
        v = 1.0 / (i ** p)
        v /= v.sum()
        update_best(v)

    # exponential edge blends (mix uniform + symmetric exponential)
    for _ in range(50):
        alpha = 10 ** rng.uniform(-3.0, 1.0)        # 0.001 … 10
        mix = rng.random()
        exp_vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
        exp_vec += np.exp(-alpha * (n - 1 - np.arange(n, dtype=np.float64)))
        exp_vec /= exp_vec.sum()
        v = (1.0 - mix) * np.full(n, 1.0 / n) + mix * exp_vec
        v /= v.sum()
        update_best(v)

    # classic window functions
    windows = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),
    }
    for w in windows.values():
        if w.sum() > 0.0:
            update_best(w / w.sum())

    # Kaiser windows
    for beta in [0.5, 2.0, 5.0, 10.0, 15.0]:
        w = np.kaiser(n, beta)
        if w.sum() > 0.0:
            update_best(w / w.sum())

    # Tukey windows
    for alpha in np.linspace(0.0, 1.0, 11):
        if alpha <= 0.0:
            w = np.ones(n, dtype=np.float64)
        elif alpha >= 1.0:
            w = np.hanning(n)
        else:
            t = np.arange(n, dtype=np.float64) / (n - 1)
            w = np.ones_like(t)
            first = t < alpha / 2.0
            w[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
            last = t > 1.0 - alpha / 2.0
            w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
        update_best(w / w.sum())

    # random Dirichlet draws (different concentration parameters)
    for conc in [0.3, 0.6, 1.0, 2.0]:
        for _ in range(8):
            update_best(rng.dirichlet(np.full(n, conc)))

    # previous construction, if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= v.sum()
        update_best(v)

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Local refinement – Adam on a smoothed max‑convolution.
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05                     # base learning rate
    m = np.zeros(n, dtype=np.float64)
    v_hat = np.zeros(n, dtype=np.float64)

    iteration = 0
    last_improve = start_time

    # temperature for Metropolis acceptance (linear cooling)
    T_start = best_J * 0.1
    T_end = 1e-9

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear temperature schedule
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * elapsed / (deadline - start_time))

        # ----------------------------------------------------------
        # 4.1 Full auto‑convolution of the current point
        # ----------------------------------------------------------
        conv = np.fft.irfft(np.fft.rfft(cur_vec, n=fft_len_conv) ** 2,
                            n=fft_len_conv)[:conv_len]
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        # record a new global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4.2 Stagnation handling – jitter around the best solution
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # multiplicative Gaussian noise + re‑normalisation
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            # reset Adam moments to avoid stale information
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = now
            continue

        # ----------------------------------------------------------
        # 4.3 Compute smoothed gradient (soft‑max of the convolution)
        # ----------------------------------------------------------
        # smoothing parameter – decays gently as time progresses
        eps_factor = 0.5 * (1.0 - elapsed / (deadline - start_time)) + 0.05
        epsilon = max_conv * eps_factor + 1e-12   # avoid epsilon = 0

        weight = softmax_weights(conv, epsilon)

        # gradient via FFT (requires a reversed copy of the current vector)
        a_rev = cur_vec[::-1]
        grad = gradient_from_weights(weight, a_rev, fft_len_grad, n)

        # ----------------------------------------------------------
        # 4.4 Adam update on the simplex (multiplicative + projection)
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_corr = v_hat / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_corr) + eps_adam)

        proposal = cur_vec - step
        proposal = proj_simplex(proposal)

        # ----------------------------------------------------------
        # 4.5 Evaluate proposal and Metropolis acceptance
        # ----------------------------------------------------------
        prop_max = full_conv_max(proposal, fft_len_conv, conv_len)
        prop_J = 2.0 * n * prop_max

        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J
        # occasional random restart to diversify the search
        if rng.random() < 0.02:
            cur_vec = rng.dirichlet(np.ones(n))
            cur_vec = proj_simplex(cur_vec)

    # ------------------------------------------------------------------
    # 5. Return the best solution (clip, re‑normalise and cast to list)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
