"""
Gradient‐descent on the probability simplex with a smooth max approximation.

The objective is   J(a) = 2·n·max(conv(a,a))   (scale‑invariant because
∑a is fixed by the soft‑max parametrisation).
Instead of the hard max we minimise a smooth approximation
    smooth_max_β(b) = (1/β)·log∑_k exp(β·b_k) ,
whose gradient is a weighted sum of all shifted copies of a.
Using a continuation on β (slowly increasing it) gives a well‑behaved
gradient that eventually focuses on the true maximum.
Adam on the logits provides an adaptive step size,
while periodic Gaussian jitter on the logits together with a reset of the
Adam moments prevents the optimiser from getting stuck.
The best sequence seen (according to the true J) is kept and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT based convolutions, power‑of‑two utilities)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    The result has length ``conv_len = 2*len(a)-1``.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the normalized soft‑max weights w_k = exp(beta * (conv_k - max))
    which sum to 1 and give the gradient of smooth_max_β.
    """
    m = conv.max()
    # shift for numerical stability
    shifted = beta * (conv - m)
    # avoid overflow/underflow
    shifted = np.clip(shifted, -700, 0)
    exp_shifted = np.exp(shifted)
    sumexp = exp_shifted.sum()
    # w already sums to 1 because we divided by sumexp
    w = exp_shifted / sumexp
    return w


def _grad_smooth_max(a: np.ndarray, w: np.ndarray,
                     grad_fft_len: int) -> np.ndarray:
    """
    Gradient of smooth_max w.r.t. a.
    Formally  ∂/∂a_i  smooth_max = 2 * (w * rev(a))[i]  (convolution with
    reversed a).  The computation is performed via FFT for O(n log n) time.
    """
    rev_a = a[::-1]                      # reversed a
    # FFT of both vectors (same length)
    W = np.fft.rfft(w, grad_fft_len)
    A_rev = np.fft.rfft(rev_a, grad_fft_len)
    full = np.fft.irfft(W * A_rev, n=grad_fft_len)
    # The relevant slice starts at n-1 and has length n
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

# ----------------------------------------------------------------------

    self‑convolution is used for the gradient, with the smoothing parameter
    β increased gradually.  Adam with a 1/√t decay is used; every 30 s of
    stagnation a Gaussian jitter is added to the logits and Adam moments are
    reset.  The best solution seen (according to the true J) is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------

        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)            # for a*a
    grad_fft_len = _next_pow2(3 * n - 2)      # for w * rev(a)

    # --------------------------------------------------------------
    # 2. Build a diverse portfolio of initial probability vectors
    # --------------------------------------------------------------
    init_candidates = []

    # uniform distribution
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # V‑shaped (large at the edges, small in the centre)
    idx = np.arange(n, dtype=np.float64)
    v = 1.0 / ((idx + 1) * (n - idx))
    v = np.clip(v, 0, None)
    v /= v.sum()
    init_candidates.append(v)

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        g = r ** idx
        g /= g.sum()
        init_candidates.append(g)

    # linear ramps (decreasing / increasing)
    ramp = (n - idx) / n
    ramp /= ramp.sum()
    init_candidates.append(ramp)
    ramp = (idx + 1) / n
    ramp /= ramp.sum()
    init_candidates.append(ramp)

    # Gaussian bumps (different widths)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        g = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        g /= g.sum()
        init_candidates.append(g)

    # Dirichlet draws (moderately and strongly concentrated)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(4):
            init_candidates.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_candidates.append(v)

    # --------------------------------------------------------------
    # 3. Select the best seed according to the true (hard) objective J
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        conv = _conv_full(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop – Adam on logits + smooth‑max gradient
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    eps = 1e-14
    logits = np.log(np.maximum(best_a, eps))    # initialise logits
    m1 = np.zeros_like(logits)                  # Adam first moment
    v1 = np.zeros_like(logits)                  # Adam second moment

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.5

    # smoothing parameter β (start small, increase gradually)
    beta = 2.0
    beta_max = 200.0
    beta_inc = 1.002   # multiplicative increase per iteration

    iteration = 0
    last_improve = start_time

    # store global best seen so far
    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # ---- 4.1 Probabilities from logits (soft‑max) ----
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)          # shift for stability
        a = exp_logits / exp_logits.sum()              # lies on the simplex

        # ---- 4.2 Hard objective (for bookkeeping) ----
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ---- 4.3 Stagnation detection – jitter restart ----
        if now - last_improve > 30.0:
            jitter_sigma = 0.15
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m1.fill(0.0)
            v1.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

        # ---- 4.4 Smooth‑max weights (softmax over convolution) ----
        w = _smooth_max_weights(conv, beta)   # length 2n‑1, sums to 1

        # ---- 4.5 Gradient of smooth‑max w.r.t. a (via FFT) ----
        grad_a = _grad_smooth_max(a, w, grad_fft_len)

        # ---- 4.6 Chain rule to logits (soft‑max Jacobian) ----
        avg = np.dot(a, grad_a)              # ⟨a,∇_a⟩
        grad_z = a * (grad_a - avg)          # ∂J/∂z

        # ---- 4.7 Adam step on logits ----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)        # simple 1/√t decay

        m1 = beta1 * m1 + (1.0 - beta1) * grad_z
        v1 = beta2 * v1 + (1.0 - beta2) * (grad_z ** 2)

        m1_hat = m1 / (1.0 - beta1 ** iteration)
        v1_hat = v1 / (1.0 - beta2 ** iteration)

        logits -= lr * m1_hat / (np.sqrt(v1_hat) + adam_eps)
        logits = np.clip(logits, -500.0, 500.0)   # keep numbers in a sane range

        # ---- 4.8 Increase smoothing parameter β ----
        beta = min(beta_max, beta * beta_inc)

    # --------------------------------------------------------------
    # 5. Return the best sequence (clipped, renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total          # normalising is optional but keeps sum≈1

    print(f"Finished. Best J = {global_best_J:.6f}")
    return [float(v) for v in best_vec]
