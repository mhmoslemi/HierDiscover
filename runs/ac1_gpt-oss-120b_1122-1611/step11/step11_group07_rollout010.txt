def full_convolve(a, b, fft_len=None):
    na = a.shape[0]
    nb = b.shape[0]
    n = na + nb -1
    if fft_len is None:
        fft_len = _next_pow2(n)
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:n]
    return conv
```

"""
Hybrid soft‑max gradient + stochastic exchange optimiser.

* Works on the probability simplex (∑a = 1, a_i ≥ 0).  The true objective is
  J(a) = 2·n·max(conv(a,a)) (identical to the value returned by
  ``evaluate_sequence`` for a sum‑1 vector).

* A smoothed surrogate of the max – the log‑sum‑exp (soft‑max) – is used
  to obtain a dense gradient:
       F_α(a) = (1/α)·log Σ_k exp(α·conv_k)
  with α growing from a small value (≈5) to a large one (≈50) during the run.
  The gradient of F_α can be written as a pair of convolutions and is
  computed with a single real FFT per convolution.

* The update is projected gradient descent with Nesterov‑type momentum on the
  simplex (Euclidean projection).  Step size decays like 1/√t.

* To escape plateaus a small‑mass “exchange” move (transfer a random fraction
  from i to j) is performed with probability 0.25, and a jitter (multiplicative
  log‑normal noise) is applied if no improvement is seen for 15 s.

* The algorithm keeps the best vector seen so far and returns it after the
  1000 s budget (5 s safety margin).  All values are finally clipped to the
  allowed interval [0,1000] and re‑normalised before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):  # all zero or equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolve_real(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using a real FFT.
    Returns the full‑length result (len(a)+len(b)-1).
    """
    n = a.shape[0] + b.shape[0] - 1
    fft_len = _next_pow2(n)
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:n]
    return conv


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Convenient wrapper used by the original code – linear convolution a*a."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(conv(a,a)) for a vector already on the simplex."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # enough for a*a

    # ------------------------------------------------------------------
    # 2. Initial point – Dirichlet(1) (uniform on the simplex)
    # ------------------------------------------------------------------
    cur = rng.dirichlet(np.ones(n))         # sums to 1
    best = cur.copy()
    best_J = compute_J(best, n, fft_len, conv_len)

    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimisation parameters
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # soft‑max temperature (α) – start low, increase slowly
    alpha = 5.0
    alpha_max = 60.0
    alpha_inc = (alpha_max - alpha) / 40000.0   # linear increase

    # momentum for the gradient step
    momentum = np.zeros_like(cur)
    beta = 0.9

    # step‑size schedule (1/√t)
    step0 = 0.6

    # stagnation handling
    last_improve = start_time
    jitter_sigma = 0.03
    stagnation_seconds = 15.0

    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # -----------------------------------------------------------------
        # 3.1  Stagnation handling – jitter after a period without improvement
        # -----------------------------------------------------------------
        if now - last_improve > stagnation_seconds:
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cur = best * noise
            cur = proj_simplex(cur)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – best J = {best_J:.6f}")

        # -----------------------------------------------------------------
        # 3.2  Decide move type
        # -----------------------------------------------------------------
        use_gradient = rng.random() < 0.75  # majority gradient move

        if use_gradient:
            # ------- (a)  compute full convolution a*a (for J and for soft‑max) -------
            conv = _conv_full(cur, fft_len, conv_len)

            # ------- (b)  soft‑max weights over convolution entries -------
            shift = conv.max()                     # for numerical stability
            exp_arg = alpha * (conv - shift)
            exp_vals = np.exp(exp_arg)
            w = exp_vals / exp_vals.sum()          # shape (conv_len,)

            # ------- (c)  gradient of the soft‑max surrogate -------
            w_rev = w[::-1]

            # conv_a_w = cur * w   (full convolution)
            conv_a_w = full_convolve_real(cur, w)          # length n+conv_len-1
            conv_a_w_rev = full_convolve_real(cur, w_rev)  # same length

            # gradient = conv_a_w[:n] + conv_a_w_rev[conv_len-1 - np.arange(n)]
            idx_rev = conv_len - 1 - np.arange(n)
            grad = conv_a_w[:n] + conv_a_w_rev[idx_rev]

            # ------- (d)  momentum update -------
            momentum = beta * momentum + (1.0 - beta) * grad

            # ------- (e)  projected gradient step -------
            step = step0 / np.sqrt(iteration + 1.0)
            prop = cur - step * momentum
        else:
            # ------- (b)  exchange move – relocate a tiny amount of mass -------
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(cur[i], rng.random() * 0.02)   # up to 2 % of source mass
            prop = cur.copy()
            prop[i] -= delta
            prop[j] += delta

        # -----------------------------------------------------------------
        # 3.3  Projection onto simplex (ensures sum==1 and non‑negativity)
        # -----------------------------------------------------------------
        prop = proj_simplex(prop)

        # -----------------------------------------------------------------
        # 3.4  Evaluate true objective J (max‑convolution)
        # -----------------------------------------------------------------
        prop_J = compute_J(prop, n, fft_len, conv_len)

        # -----------------------------------------------------------------
        # 3.5  Acceptance rule (always accept improvement, occasional uphill)
        # -----------------------------------------------------------------
        accept = False
        cur_J = compute_J(cur, n, fft_len, conv_len)  # value for the current point
        if prop_J < cur_J:
            accept = True
        else:
            # a simple “temperature” that slowly cools – we reuse α as a proxy
            T = max(1e-8, (alpha_max - alpha) / alpha_max)
            prob = np.exp(-(prop_J - cur_J) / (max(T, 1e-12) * best_J))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = prop
            cur_J = prop_J

            # ------- record global best -------
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best = cur.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # -----------------------------------------------------------------
        # 3.6  Update α (slowly move towards the true max)
        # -----------------------------------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    total = best.sum()
    if total == 0.0:
        best[:] = 1.0 / n
    else:
        best /= total

    print(f"Finished after {time.monotonic() - start_time:.2f}s – best J = {best_J:.6f}")
    return [float(v) for v in best]
