"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Algorithm
---------
1. Initialise a diverse set of candidate vectors:
   * Analytic √‑difference shape (the classic start).
   * Power‑difference families a_i = (i+1)^-p - (i+2)^-p  for several p.
   * Exponential families a_i = (1‑e^{-λ})·e^{-λ i} for several λ.
   * Random Dirichlet samples.
   * The previous best construction if it exists in ``height_sequence_1``.
   All vectors are normalised to sum = 1 (the objective is scale‑invariant).

2. Pick the best of those candidates (fast evaluation via FFT).

3. Refine the current best with a projected mirror‑descent
   (exponentiated‑gradient) that works on the true L∞‑objective.
   * At each iteration the self‑convolution is computed.
   * All indices whose value is within 0.5 % of the current maximum are
     collected (max 12 indices). Their soft‑max weights give a sub‑gradient.
   * Momentum is applied to stabilise the update.
   * A linear schedule drives the step‑size η from 1.5 down to 0.001 and the
     soft‑max temperature β from 0.1 up to 5000.
   * Tiny random block perturbations and pairwise mass transfers keep the
     iterate from stagnating.

4. When the time budget is almost exhausted (≈20 s left) a short simulated‑
   annealing phase fine‑tunes the solution:
   * Random mass‑transfer moves between two positions are proposed.
   * Moves are accepted if they improve the score or with probability
     exp(−Δ/ T) where the temperature T cools exponentially.
   * The best vector seen throughout the whole run is retained.

5. The final vector is clipped to the limits imposed by ``evaluate_sequence``
   (0 ≤ value ≤ 1000) and returned as a plain list of floats.

The whole routine respects a 1000 s budget, prints progress stamps and
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Fast (exact) L∞‑objective for a probability vector.
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a*a) for a normalised (sum=1) non‑negative vector."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (used only when the active set is empty).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Grad of Σ_k w_k (a∗a)_k = 2·a ⋆ w (correlation)."""
    n = a.size
    L = w.size                # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Search for a non‑negative step‑function that minimises evaluate_sequence().
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Helper to build a normalised candidate
    # ------------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s <= 0.0:
            # fallback to uniform
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # ------------------------------------------------------------------
    # 3) Assemble a diverse pool of starting points
    # ------------------------------------------------------------------
    candidates = []

    # a) Analytic √‑difference start
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    candidates.append(_normalise(a_sqrt))

    # b) Previous construction if available
    if "height_sequence_1" in globals():
        try:

        beta = beta_start + (beta_end - beta_start) * frac

        active = np.where(conv >= cur_max * rel_tol)[0]

            w_sum = w_raw.sum()

            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            cur_sa = smoothed
            cur_sa_score = _score_of(cur_sa, n, fft_len)

    # ------------------------------------------------------------------
    # 7) Return the best sequence (respect clipping constraints)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
