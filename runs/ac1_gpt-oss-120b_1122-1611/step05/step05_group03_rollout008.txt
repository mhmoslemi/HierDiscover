def propose_candidate(seed=1234, budget_s=1000.0):
    import time, numpy as np
    import math
    from scipy.optimize import linprog
    from scipy.sparse import diags

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 0.1

    # load initial a
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
    else:
        n = 3000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce non-neg and standardize
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total <= 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.size
    conv = np.convolve(a, a)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    # mixing param T schedule
    # We'll use small t for mixing; adapt based on progress
    last_improve = start_time

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # recompute conv and max bound
        conv = np.convolve(a, a)
        M = conv.max()
        # LP to get g
        # Build A_ub
        A_ub = diags(a, offsets=np.arange(n), shape=(2*n-1, n), format='csr')
        b_ub = np.full(2*n-1, M)

        # Solve LP (maximize sum g)
        c = -np.ones(n)  # minimize -sum.
        # left time for solver:
        remaining = deadline - time.monotonic()
        # limit time for solver to a fraction of remaining
        solver_time = min(remaining * 0.5, 5.0)  # don't exceed 5 sec per solve
        # Options
        opt = {"presolve": True}
        if solver_time > 0:
            opt["time_limit"] = solver_time

        # call linprog
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                          method='highs', options=opt)
        except Exception as e:
            # fallback to simple perturbation if LP fails
            res = None

        if res is None or not res.success:
            # fallback: maybe random jitter
            # apply a small random block perturbation
            block_width = max(1, int(0.1 * n))
            start = rng.integers(0, n - block_width + 1)
            factor = math.exp(rng.normal(0, 0.1))
            a_pert = a.copy()
            a_pert[start:start+block_width] *= factor
            a_pert = np.clip(a_pert, 0, None)
            a_pert /= a_pert.sum()
            new_score = 2.0 * n * np.convolve(a_pert, a_pert).max()
            if new_score < cur_score:
                a = a_pert
                cur_score = new_score
                if new_score < best_score:
                    best = a.copy()
                    best_score = new_score
                    last_improve = time.monotonic()
                    print(f"[{last_improve - start_time:.2f}s] new best after perturb {best_score:.6f}")
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            # nothing useful
            continue

        # scale to sum 1
        g = g / sum_g

        # evaluate g alone
        conv_g = np.convolve(g, g)
        score_g = 2.0 * n * conv_g.max()
        # if better, accept g
        improved = False
        if score_g < cur_score:
            a = g
            conv = conv_g
            cur_score = score_g
            improved = True
            print(f"[{time.monotonic() - start_time:.2f}s] LP gave better score {cur_score:.6f}")

        # try mixing, only if not already improved or to see if extra improvement
        # define mix factors decreasing over time
        # compute fraction of elapsed
        elapsed = time.monotonic() - start_time
        frac_elapsed = elapsed / budget_s
        # early phases allow larger t
        mix_factors = [0.4, 0.2, 0.1]
        for t in mix_factors:
            a_mix = (1 - t) * a + t * g
            a_mix = np.clip(a_mix, 0.0, None)
            a_mix /= a_mix.sum()
            conv_mix = np.convolve(a_mix, a_mix)
            score_mix = 2.0 * n * conv_mix.max()
            if score_mix < cur_score - 1e-9:
                a = a_mix
                conv = conv_mix
                cur_score = score_mix
                improved = True
                print(f"[{time.monotonic() - start_time:.2f}s] mixed (t={t}) improved to {cur_score:.6f}")
                break

        if cur_score < best_score - 1e-9:
            best = a.copy()
            best_score = cur_score
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:.2f}s] new best overall {best_score:.6f}")

        # restart if stuck
        if time.monotonic() - last_improve > 30.0:
            # random restart
            a = np.abs(rng.normal(size=n))
            a /= a.sum()
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            print(f"[{time.monotonic() - start_time:.2f}s] restart random, score {cur_score:.6f}")

        # also occasional random block jitter (like earlier)
        if rng.random() < 0.005:
            a_pert = a.copy()
            # pick random block
            width = max(1, int(rng.random() * 0.2 * n))
            start = rng.integers(0, n - width + 1)
            factor = math.exp(rng.normal(0, 0.08))
            a_pert[start:start+width] *= factor
            a_pert = np.clip(a_pert, 0.0, None)
            a_pert /= a_pert.sum()
            conv_pert = np.convolve(a_pert, a_pert)
            score_pert = 2.0 * n * conv_pert.max()
            if score_pert < cur_score:
                a = a_pert
                conv = conv_pert
                cur_score = score_pert
                print(f"[{time.monotonic() - start_time:.2f}s] perturb improved to {cur_score:.6f}")
                if cur_score < best_score:
                    best = a.copy()
                    best_score = cur_score
                    last_improve = time.monotonic()
    # final best
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Propose a candidate step‑function by a hybrid of
  * linear‑programming “mass‑transfer” steps (maximise total mass
    while respecting the current convolution bound)
  * small stochastic mixing / jitter moves to escape plateaus
  * occasional random restarts with different initial shapes.

The LP step solves
      maximise   Σ g_i
      subject to 0 ≤ g_i ,   (a * g)[k] ≤ M   ∀k
where a is the current vector and M = max(a*a).  For a fixed M the
LP finds a vector that can carry more total mass without increasing the
peak of the mixed convolution.  Mixing the LP‑solution back into a
with a modest step size (t≈0.2…0.5) strictly reduces the self‑convolution
peak (or leaves it unchanged) and therefore improves the objective

    score = 2·n·max(a*a)  (the denominator Σa cancels out).

The routine runs as many LP‑updates as the 1000 s budget allows,
starting from the best known construction (global variable
``height_sequence_1``) and from a few random‑shape restarts.
All helpers are defined at top level to satisfy the “no closures”
rule.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1)."""
    return np.convolve(a, a)


def _lp_max_mass(a: np.ndarray, bound: float, time_limit: float):
    """
    Solve the linear programme
        maximise   Σ g_i
        subject to 0 ≤ g_i ,   (a * g)[k] ≤ bound  ∀k .
    Returns the optimisation result (scipy.optimize.OptimizeResult)
    or ``None`` if the solver fails.
    """
    n = a.size
    # Build the convolution matrix A_ub where each row k contains a_i
    # at column j whenever i + j = k.
    A_ub = diags(a, offsets=np.arange(n), shape=(2 * n - 1, n), format="csr")
    b_ub = np.full(2 * n - 1, bound, dtype=np.float64)

    # maximise Σ g_i  →  minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)

    opts = {"presolve": True}
    if time_limit > 0:
        opts["time_limit"] = time_limit

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options=opts,
        )
    except Exception:  # pragma: no cover – safety net
        return None

    return res if res.success else None


def _mix_and_score(a: np.ndarray, g: np.ndarray, t: float):
    """
    Return the mixture (1‑t)·a + t·g (renormalised to Σ=1) together
    with its score 2·n·max(conv).
    """
    mixed = (1.0 - t) * a + t * g
    mixed = np.clip(mixed, 0.0, None)
    total = mixed.sum()
    if total == 0.0:
        return None, np.inf
    mixed /= total
    conv = _self_convolution(mixed)
    score = 2.0 * mixed.size * conv.max()
    return mixed, score


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **_):
    """
    Search for a non‑negative step function (probability vector)
    that minimises ``evaluate_sequence``.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Determine problem size and an initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / a0.sum()
        n = a0.size
    else:
        # a modest length if no previous construction is available
        n = 3000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Helper to compute the (scale‑invariant) score of a vector
    # ------------------------------------------------------------------
    def _score(vec):
        conv = _self_convolution(vec)
        return 2.0 * vec.size * conv.max()

    # ------------------------------------------------------------------
    # 3) Store the overall best solution seen so far
    # ------------------------------------------------------------------
    best_vec = a0.copy()
    best_score = _score(best_vec)
    last_improve = start_time

    print(f"[{0.00:6.2f}s] start – length {n}, score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Main optimisation loop (multiple restarts + LP updates)
    # ------------------------------------------------------------------
    restart_count = 0
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 1.0:
            break

        # ----------------------------------------------------------------
        # a) Choose a starting shape for this restart
        # ----------------------------------------------------------------
        if restart_count == 0:
            # first restart uses the global best vector
            cur = best_vec.copy()
        else:
            # later restarts: pick a random prototype
            typ = rng.integers(0, 4)
            if typ == 0:  # uniform
                cur = np.full(n, 1.0 / n, dtype=np.float64)
            elif typ == 1:  # sine-shaped (positive half‑wave)
                x = np.arange(n, dtype=np.float64)
                cur = np.sin(np.pi * (x + 1) / (n + 1))
                cur = np.clip(cur, 0.0, None)
                cur /= cur.sum()
            elif typ == 2:  # quadratic bump
                x = np.arange(n, dtype=np.float64)
                cur = (x + 1) * (n - x)
                cur = np.clip(cur, 0.0, None)
                cur /= cur.sum()
            else:  # random Dirichlet
                cur = rng.random(n)
                cur = np.clip(cur, 0.0, None)
                cur /= cur.sum()

        cur_score = _score(cur)

        # ----------------------------------------------------------------
        # b) Local optimisation using LP + mixing
        # ----------------------------------------------------------------
        local_iters = 0
        while time.monotonic() < deadline:
            local_iters += 1
            # recompute convolution peak
            conv_cur = _self_convolution(cur)
            M = conv_cur.max()

            # time allocated to the LP (a fraction of the remaining budget)
            lp_time = min(3.0, (deadline - time.monotonic()) * 0.4)
            lp_res = _lp_max_mass(cur, M, lp_time)

            if lp_res is None:
                # LP failed – fall back to a tiny random jitter and continue
                if rng.random() < 0.02:
                    width = max(1, int(rng.random() * 0.15 * n))
                    start = rng.integers(0, n - width + 1)
                    factor = math.exp(rng.normal(0.0, 0.09))
                    jitter = cur.copy()
                    jitter[start:start + width] *= factor
                    jitter = np.clip(jitter, 0.0, None)
                    jitter /= jitter.sum()
                    jitter_score = _score(jitter)
                    if jitter_score < cur_score:
                        cur, cur_score = jitter, jitter_score
                        if cur_score < best_score:
                            best_vec, best_score = cur.copy(), cur_score
                            last_improve = time.monotonic()
                            print(f"[{last_improve - start_time:6.2f}s] "
                                  f"jitter improved to {best_score:.6f}")
                break  # exit inner LP loop – try a new restart

            g = lp_res.x
            sum_g = g.sum()
            if sum_g == 0.0:
                break
            g = g / sum_g  # renormalise to Σ=1

            # ------------------------------------------------------------
            # Try direct replacement (only if it actually improves)
            # ------------------------------------------------------------
            g_score = _score(g)
            improved = False
            if g_score < cur_score - 1e-12:
                cur, cur_score = g, g_score
                improved = True

            # ------------------------------------------------------------
            # Try a few mixing factors (larger early, smaller later)
            # ------------------------------------------------------------
            elapsed = time.monotonic() - start_time
            frac = elapsed / budget_s
            mix_factors = [0.4, 0.25, 0.1] if frac < 0.7 else [0.2, 0.1, 0.05]
            for t in mix_factors:
                mixed, mixed_score = _mix_and_score(cur, g, t)
                if mixed_score < cur_score - 1e-12:
                    cur, cur_score = mixed, mixed_score
                    improved = True
                    break

            # ------------------------------------------------------------
            # Record global best and possibly restart after stagnation
            # ------------------------------------------------------------
            if cur_score < best_score - 1e-12:
                best_vec, best_score = cur.copy(), cur_score
                last_improve = time.monotonic()
                print(f"[{last_improve - start_time:6.2f}s] "
                      f"new best {best_score:.6f}")

            # break if no improvement for a while
            if not improved:
                if time.monotonic() - last_improve > 25.0:
                    break
                # occasional tiny perturbation to shake out of a shallow plateau
                if rng.random() < 0.015:
                    width = max(1, int(rng.random() * 0.12 * n))
                    start = rng.integers(0, n - width + 1)
                    factor = math.exp(rng.normal(0.0, 0.07))
                    pert = cur.copy()
                    pert[start:start + width] *= factor
                    pert = np.clip(pert, 0.0, None)
                    pert /= pert.sum()
                    pert_score = _score(pert)
                    if pert_score < cur_score:
                        cur, cur_score = pert, pert_score
                        if cur_score < best_score:
                            best_vec, best_score = cur.copy(), cur_score
                            last_improve = time.monotonic()
                            print(f"[{last_improve - start_time:6.2f}s] "
                                  f"perturb improved to {best_score:.6f}")

        restart_count += 1

    # ------------------------------------------------------------------
    # 5) Return the best sequence found (clipped to evaluate limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec.tolist()]
