"""
Propose a step‑function by minimizing the worst self‑convolution value.

The algorithm works on the probability simplex (a_i ≥ 0, Σ a_i = 1) and
optimises a *smooth* surrogate of the max‑convolution:
    L(a) = log‑sum‑exp(α·(a*a)),
where “*” denotes the full linear convolution.
The gradient of L is a weighted cross‑correlation that can be computed
with two FFT‑based convolutions per iteration.  An Adam‑style
exponentiated‑gradient update (mirror‑descent on the log‑variables)
keeps the vector non‑negative and automatically satisfies the simplex
constraint.  The smoothing coefficient α is annealed from a small
value (α≈0.5) to a large one (α≈30) so that early iterations move
smoothly while later iterations focus on the true max‑value.
Occasional random block perturbations and soft restarts prevent
premature stagnation.  The best vector seen during the run is returned.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    if x <= 0:
        return 1
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution of ``a`` using FFT.
    Returns an array of length ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _cross_corr_fft(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation b_j = Σ_i a_i w_{i+j} for j=0..len(a)-1.
    Implemented as convolution of ``a`` with the reversed ``w``.
    Returns the slice of length ``len(a)`` (the central part of the
    linear convolution).
    """
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    conv = np.fft.irfft(fa * fw, n=fft_len)[: 3 * a.size - 2]
    start = a.size - 1
    return conv[start : start + a.size]


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):

    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector (unit sum)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():

                total = a.sum()
                if total > 0.0:
                    a = a / total
                else:
                    a = np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:

    else:
        a = np.full(256, 1.0 / 256, dtype=np.float64)

    n = a.size

    # FFT lengths for the two convolutions we need each iteration
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)               # for a*a
    grad_len = 3 * n - 2
    fft_len_grad = _next_pow2(grad_len)               # for cross‑correlation

    # ------------------------------------------------------------------
    # 2) Optimiser bookkeeping (Adam on the *log* of the variables)
    # ------------------------------------------------------------------

    beta2 = 0.999

    lr_end = 0.01           # final stepsize

    # 5) Return the best vector (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
