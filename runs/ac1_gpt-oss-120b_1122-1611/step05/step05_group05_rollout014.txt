"""
Gradient‑descent / sub‑gradient optimizer for the max‑convolution
objective used by ``evaluate_sequence``.  Starting from a uniform
probability simplex (or the provided ``height_sequence_1``) the algorithm
repeatedly:

* evaluates the convolution,
* finds the index where it attains its maximum,
* computes a sub‑gradient of that maximal entry (∇ max conv = 2·a shifted),
* takes a decaying step in the opposite direction,
* projects back onto the probability simplex to keep the sum = 1 and
  all entries non‑negative.

If progress stalls for ``restart_gap`` seconds the current best solution
is jittered with log‑normal noise and the search restarts, providing a
simple escape mechanism.  The best sequence seen before the time budget
(≈1000 s) expires is returned.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex {x≥0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_conv(a: np.ndarray):
    """
    Returns the sub‑gradient of the maximal convolution entry,
    the value of that entry and its index.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    max_val = conv[k]

    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    # gradient w.r.t a_i is 2·a_{k‑i} when the index is valid
    g[idx] = 2.0 * a[k - idx]
    return g, max_val, k


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Gradient‑descent optimizer for ``evaluate_sequence``.
    Returns a list of non‑negative heights that (approximately) minimizes the
    evaluation function within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Initialise the sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all():
                raise ValueError
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # use a length comparable to the previously reported best construction
        n = 3809
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform simplex vector
    else:
        a = np.clip(a0, 0.0, None)
        n = a.size
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s                                 # project onto simplex
    # ensure exact sum = 1
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = np.convolve(a, a)
    best = a.copy()
    best_score = 2.0 * n * np.max(conv)    # sum(a) == 1 after projection
    cur_score = best_score
    start_time = time.monotonic()
    last_improve = start_time

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3) Optimisation parameters
    # ------------------------------------------------------------------
    lr0 = 0.5                     # initial step size
    restart_gap = 30.0            # seconds without improvement → restart
    accept_prob_start = 0.08      # occasional uphill moves (annealing)
    accept_prob_end = 0.001

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- sub‑gradient step -----
        g, max_val, max_k = _subgradient_max_conv(a)

        # decaying learning rate
        lr = lr0 / np.sqrt(it)
        cand = a - lr * g
        cand = np.clip(cand, 0.0, None)
        cand = _project_simplex(cand)

        # evaluate candidate
        conv_cand = np.convolve(cand, cand)
        cand_score = 2.0 * n * np.max(conv_cand)

        # ----- acceptance -----
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")
        else:
            # simulated‑annealing style uphill acceptance
            now = time.monotonic()
            elapsed = now - start_time
            frac = min(1.0, elapsed / (budget_s - 0.1))
            accept_prob = accept_prob_start * (1.0 - frac) + accept_prob_end * frac
            if rng.random() < accept_prob:
                a = cand
                cur_score = cand_score

        # ----- restart if stuck -----
        if time.monotonic() - last_improve > restart_gap:
            # jitter the best solution and continue
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = best * noise
            a = _project_simplex(a)
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * np.max(conv)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence found (clip to allowed range)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
