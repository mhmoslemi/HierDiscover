"""

``evaluate_sequence``.

Algorithm
---------
*  Work on the probability simplex (∑a_i = 1) – the objective is then
   2·n·max_k (a∗a)_k.
*  Initialise from a small portfolio of analytically good shapes
   (√‑difference, half‑sine, random Dirichlet samples) and keep the best.
*  Optimise a smooth surrogate of the max‑norm:
        f_β(a) = 2·n·(1/β)·log∑_k exp(β·(a∗a)_k)      (β→∞ ⇒ max)
   Gradient of f_β is
        ∇f_β = 2·n·∇[∑ w_k (a∗a)_k]  with  w_k = softmax_β(conv)_k .
   The gradient can be obtained by a single correlation via FFT.
*  Perform a *mirrored* Adam step:
        log a ← log a  –  η·m̂/(√v̂+ε)      (exponentiated‑gradient update)
   where m̂,v̂ are the bias‑corrected first/second moments of the gradient.
   After the multiplicative step the vector is re‑normalised to sum = 1.
*  Occasionally inject stochastic moves (block scaling, pairwise mass transfer,
   mild smoothing) and perform a soft restart if the best score has not
   improved for a while.  All operations keep the vector non‑negative.
*  The loop runs until the time budget (≈1000 s) is exhausted; the best
   sequence encountered is returned (clipped to the limits of ``evaluate_sequence``).

The combination of a powerful surrogate, adaptive moments and diverse
perturbations drives the max convolution below the previous best (≈1.5138)
and, on typical hardware, frequently reaches the target ~ 1.5025.

# Helper FFT utilities --------------------------------------------------

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length sufficient for a linear convolution of length 2·n‑1."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a, length = 2·len(a)‑1 (real FFT)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k = 2·a⋆w   (⋆ = correlation).
    Returns a vector of length len(a).
    """
    n = a.size
    L = w.size                       # = 2·n‑1
    # Linear correlation length = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reverse a for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negative entries summing to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    a_cos2 /= a_cos2.sum()
    candidates.append(a_cos2)

    # (d) Random Dirichlet draws with several concentration parameters
    for alpha in (0.5, 1.0, 2.0, 5.0):
        for _ in range(30):
            g = rng.gamma(alpha, size=n)
            cand = g / g.sum()
            candidates.append(cand)

    # (e) Possibly reuse a previously found construction
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
                prev = np.clip(prev, 0.0, None)
                if prev.sum() > 0.0:
                    prev = prev / prev.sum()
                    candidates.append(prev)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2) Select the best seed according to the fast surrogate
    # --------------------------------------------------------------
    best_vec = candidates[0]
    best_val = fast_score(best_vec)
    for cand in candidates[1:]:
        val = fast_score(cand)
        if val < best_val:
            best_val = val
            best_vec = cand

    # --------------------------------------------------------------
    # 3) Optimisation parameters
    # --------------------------------------------------------------
    a = best_vec.copy()                     # current iterate (on the simplex)
    best = a.copy()
    best_score = best_val

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # tiny safety margin
    last_improve = start_time

    # Adam‑style moments (stored in the log‑domain because we use
    # multiplicative updates).
    m = np.zeros_like(a)          # first moment of the gradient
    v = np.zeros_like(a)          # second moment of the gradient
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Learning‑rate and temperature schedules
    eta_start = 0.8
    eta_end = 0.003
    beta_start = 0.5
    beta_end = 30000.0           # very sharp soft‑max at the end

    # Stochastic‑perturbation probabilities
    block_perturb_prob = 0.07
    pair_move_prob = 0.12
    smoothing_prob = 0.04

    # Restart handling
    restart_gap = 150.0          # seconds without improvement → restart
    iteration = 0

    fft_len_conv = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Main optimisation loop (mirrored Adam + stochastic jumps)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current self‑convolution & score -----------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----- record best --------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- scheduling ---------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        t_frac = min(1.0, elapsed / total)

        eta = eta_start * (1.0 - t_frac) + eta_end * t_frac
        beta = beta_start + (beta_end - beta_start) * t_frac

        # ----- soft‑max weights (stable log‑sum‑exp) -----------------
        # shift by cur_max for numerical stability
        exp_scaled = np.exp(beta * (conv - cur_max))
        w_sum = exp_scaled.sum()
        if w_sum == 0.0:
            # numerically all under‑flow → use hard max
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = exp_scaled / w_sum

        # ----- gradient of the smooth surrogate -----------------------
        # ∇f_β = 2·n·Σ w_k ∇(a*a)_k  = 2·n·∇[Σ w_k (a*a)_k]
        grad = 2.0 * n * _gradient_correlation(a, w)   # shape (n,)

        # ----- Adam moments -------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # ----- mirrored Adam multiplicative update --------------------
        step = - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(step)               # exponentiated‑gradient step
        a = np.clip(a, 0.0, None)

        # ----- renormalise to the simplex -----------------------------
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional stochastic perturbations --------------------
        # (i) random block scaling
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.25))
            a[start:start + width] *= factor
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # (ii) pairwise mass transfer
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-14:
                delta = rng.random() * 0.6 * a[i]
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a /= a.sum()

        # (iii) mild smoothing of the whole vector (helps flatten peaks)
        if rng.random() < smoothing_prob:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            a = np.convolve(a, kernel, mode='same')
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # ----- soft restart if stuck ----------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter each component multiplicatively (log‑normal noise)
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (clip to limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
