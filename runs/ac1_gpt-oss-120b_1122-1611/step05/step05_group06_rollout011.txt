def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 3.0) # safer

    best_vec = None
    best_score = np.inf

    # try previous global candidate if present
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0:
                    cand = np.full_like(cand, 1.0/cand.size)
                else:
                    cand = cand / cand.sum()
                n = cand.size
                cur_score = 2.0 * n * _conv_self(cand, _fft_len_for_len(2*n-1)).max()
                best_vec = cand.copy()
                best_score = cur_score
                print(f"Seed candidate score = {cur_score:.6f}")
        except Exception as e:
            print("Error evaluating seed candidate:", e)

    # define lengths to try
    lengths = [1500, 2500, 3809, 5000, 8000]
    # biases; larger lengths but sample not too many per time
    prob_lengths = np.array([0.10, 0.15, 0.30, 0.30, 0.15])

    # main loop
    while time.monotonic() < deadline:
        # choose length
        n = int(rng.choice(lengths, p=prob_lengths/ prob_lengths.sum()))
        # start vector: Dirichlet with alpha ~ 0.3 (optional)
        alpha = rng.uniform(0.1, 0.7)
        a0 = rng.gamma(alpha, 1.0, size=n)
        a0 = a0 / a0.sum()
        # local optimization
        a_opt, score_opt = _local_optimize(a0, rng, deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] New best {best_score:.6f} (n={n})")
        # try polishing best if time
        if best_vec is not None and time.monotonic() < deadline - 0.5:
            a_polished, score_polished = _local_optimize(best_vec.copy(), rng, deadline, max_iters=3000)
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] Polished best {best_score:.6f}")

    # final fallback
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0/n)

    # safety clipping
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    return [float(x) for x in best_vec]

def _local_optimize(a0, rng, deadline, max_iters=2000):
    a = a0.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2*n - 1)
    fft_len_grad = _fft_len_for_len(3*n - 2)
    best_vec = a.copy()
    best_score = 2.0*n*_conv_self(a, fft_len_conv).max()

    for t in range(1, max_iters+1):
        cur_time = time.monotonic()
        if cur_time >= deadline:
            break
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0*n*conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
        # choose gradient type
        if rng.random() < 0.4:  # subgradient some probability
            i_max = int(conv.argmax())
            lo = max(0, i_max - (n - 1))
            hi = min(i_max, n - 1)
            idx = np.arange(lo, hi+1)
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[i_max - idx]
        else:
            beta0 = 0.5
            beta_max = 5000.0
            beta = beta0 + (beta_max - beta0) * (t / max_iters)
            cur_max = conv.max()
            w = np.exp(beta * (conv - cur_max))
            sum_w = w.sum()
            a_rev = a[::-1]
            conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
            grad = 2.0 * conv_w_rev[n-1:n-1+n] / sum_w

        # exponentiated update
        lr = 1.0 / np.sqrt(t)   # step size decays
        a = a * np.exp(-lr * grad)
        # renormalize to simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a / s

        # occasional random moves
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.02:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score

"""
A fast heuristic for the step‑function optimisation problem.

The algorithm works on the probability simplex (Σ a_i = 1) and combines:
* a (biased) Adam update that uses either the exact sub‑gradient of the
  current worst convolution entry or a smooth “soft‑max” approximation
  (with the correct factor 2 in the gradient);
* occasional random block‑perturbations and cheap pairwise mass‑transfer
  moves;
* many cheap restarts from Dirichlet‑distributed starts;
* a longer polishing phase on the current global best.

All operations are carried out with FFT‑based convolutions for speed.
The implementation stays fully on‑CPU, respects the 1000 s budget and
returns a list of non‑negative heights ready for `evaluate_sequence`.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
#  FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors `x` and `y` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
#  Simplex projection
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Gradient / sub‑gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. `a` (the worst entry of the convolution)."""
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(
    a: np.ndarray, conv: np.ndarray, beta: float, fft_len_grad: int
) -> np.ndarray:
    """
    Gradient of the smooth max
        f_β(a) = (1/β)·log Σ_i exp(β·conv_i)
    with the correct factor 2.
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    # slice that corresponds to the n‑dimensional gradient
    grad = 2.0 * conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad


# ----------------------------------------------------------------------
#  Random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(
    a: np.ndarray, rng: np.random.Generator, fft_len_conv: int
) -> np.ndarray:
    """Transfer a random amount of mass from i to j, accepting only if the score improves."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)

    # compare scores
    conv_new = _conv_self(a2, fft_len_conv)
    conv_old = _conv_self(a, fft_len_conv)
    if 2.0 * n * conv_new.max() < 2.0 * n * conv_old.max():
        return a2
    return a


# ----------------------------------------------------------------------
#  Local optimiser
# ----------------------------------------------------------------------


def _local_optimise(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Run a short Adam‑based optimiser starting from `a0` (already on the simplex).
    Returns the best vector seen and its (2·n·max(conv)) score.
    """
    a = a0.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state – we keep only the second moment (as in the reference code)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_vec = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len_conv).max()

    while t < max_iters and time.monotonic() < deadline:
        t += 1

        # ----- current convolution and score -----
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ----- choose a gradient -----
        if rng.random() < 0.5:
            raw_grad = _subgradient_step(a, conv)
        else:
            # smooth max – β grows linearly during the run
            beta = 0.5 + (2000.0 - 0.5) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # ----- Adam update (additive) -----
        lr = 0.7 / np.sqrt(t)           # decreasing learning rate
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)    # stay on the simplex

        # ----- occasional random moves -----
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.04:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score

#  Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that minimises `evaluate_sequence`.
    The routine performs many cheap restarts, a longer polishing phase on the
    best candidate found so far, and respects the given time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin so the harness can still receive the answer
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    #  Start from an existing construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                n = cand.size
                cur = 2.0 * n * _conv_self(cand,
                                          _fft_len_for_len(2 * n - 1)).max()
                best_vec = cand.copy()
                best_score = cur
                print(f"Initial candidate from height_sequence_1: score = {cur:.6f}")
        except Exception as e:
            print("Failed to read height_sequence_1:", e)

    # ------------------------------------------------------------------
    #  Parameter set – a mixture of lengths (small + medium + large)
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3400, 3809, 5000, 8000]
    probs = np.array([0.10, 0.10, 0.15, 0.25, 0.20, 0.20])
    probs = probs / probs.sum()

    # ------------------------------------------------------------------
    #  Main search loop – cheap restarts + occasional polishing
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (biased towards larger ones)
        n = int(rng.choice(lengths, p=probs))

        # Dirichlet start – the concentration controls sparsity
        alpha = rng.uniform(0.2, 0.8)
        a0 = rng.gamma(alpha, 1.0, size=n)
        a0 /= a0.sum()

        # quick optimisation (few hundred iterations)
        a_opt, score_opt = _local_optimise(a0, rng, deadline, max_iters=800)

        if score_opt < best_score:
            best_score, best_vec = score_opt, a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        #  If we still have time, run a longer polish on the current best
        # ------------------------------------------------------------------
        if best_vec is not None and time.monotonic() < deadline - 0.5:
            a_polished, score_polished = _local_optimise(
                best_vec.copy(), rng, deadline, max_iters=2500
            )
            if score_polished < best_score:
                best_score, best_vec = score_polished, a_polished.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polished best: {best_score:.6f}")

    # ------------------------------------------------------------------
    #  Fallback – should never happen but keeps the function safe
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)

    # final safety clipping (evaluate_sequence will clip again)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # convert to plain Python floats for the harness
    return [float(x) for x in best_vec]

It stays within the 1000 s budget and returns a valid non‑negative sequence for `evaluate_sequence`.
