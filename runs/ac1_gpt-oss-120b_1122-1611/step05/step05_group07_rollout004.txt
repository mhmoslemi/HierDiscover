            # learning rate schedule (linear from lr_start to lr_end)
            lr = lr_start * (1 - frac) + lr_end * frac

            z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            # recompute a from z
            a = np.exp(z)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum

            # occasional jitter
            if rng.random() < 0.02:
                # small log-normal perturbation to a block
                width = rng.integers(1, max(2, n // 15))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.1))
                a[start:start+width] *= factor
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                z = np.log(np.clip(a, eps, None))

            continue  # next iteration

        # Phase 2: local stochastic refinement (simulated annealing)

        # compute temperature (linear decay)
        t_frac = (now - phase1_deadline) / (deadline - phase1_deadline)
        temperature = (0.05 * (1 - t_frac) + 1e-6 * t_frac)  # start moderate, go low

        # locate top peaks (maybe up to 3 indices)
        # get all indices sorted descending
        # but for speed just get max index
        k_max = int(conv.argmax())
        # compute contributions for this k_max
        contrib = np.zeros_like(a)
        # vectorized: i from 0..n-1, j = k_max - i in range?
        # Create mask
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        total_contrib = contrib.sum()
        # choose donor; probability proportional to contribution
        if total_contrib > 0 and rng.random() < 0.9:
            prob = contrib / total_contrib
            donor = rng.choice(n, p=prob)
        else:
            donor = rng.integers(0, n)

        # If donor mass tiny, skip
        if a[donor] < 1e-12:
            continue

        # choose recipient distinct
        recipient = rng.integers(0, n-1)
        if recipient >= donor:
            recipient += 1

        # choose delta up to a fraction (maybe up to 0.2 of donor)
        delta = rng.random() * 0.2 * a[donor]
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta
        # renorm is not needed because total mass unchanged
        conv_new = _self_conv(a_new, fft_len)
        new_score = 2.0 * n * conv_new.max()
        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
            a = a_new
            conv = conv_new
            cur_score = new_score

        # occasional block perturbation
        if rng.random() < 0.01:
            width = rng.integers(1, max(2, n // 15))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[start:start+width] *= factor
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            # update z for consistency
            z = np.log(np.clip(a, eps, None))

        # restart if stagnation
        if now - last_improve > 25.0:
            # reinitialize from best with noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.15, size=n))
            a *= noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            z = np.log(np.clip(a, eps, None))
            m[:] = 0.0
            v[:] = 0.0
            iter_count = 0
            last_improve = now
            print(f"[{now-start_time:6.2f}s] restart from best score {best_score:.6f}")

    # end while
    # final best scaling (clip to 1000)
    best = np.clip(best, 0.0, 1000.0)
    # Return as list
    return [float(x) for x in best]

"""
Propose a step‑function (non‑negative sequence) whose
self‑convolution has a small ℓ∞‑norm.
The algorithm mixes three ideas:

1. **Entropy‑mirror descent with Adam** – we keep a log‑parameter
   `z` (soft‑max representation of a probability vector) and minimise
   a smooth approximation of  max(conv(a,a)) .  The temperature β is
   annealed up to a large value so that the soft‑max becomes an exact
   max.  Adam gives an adaptive learning rate.

2. **Stochastic local refinement** – once the smooth phase has finished,
   a simulated‑annealing style mass‑transfer is applied.  The donor
   index is chosen proportionally to its contribution to the current
   peak of the convolution, a small amount of mass is moved to a random
   recipient and the move is accepted according to a Metropolis rule.
   Small block‑wise log‑normal perturbations are injected occasionally.

3. **Restart on stagnation** – if no improvement is seen for a while
   the current best solution is perturbed and the optimisation restarts.

The routine respects a 1000 s budget (5 s safety margin) and returns the
best sequence found as a plain Python list of floats.  All numbers are
non‑negative and limited to 1000, which matches the checks inside
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length `fft_len`
    (must be ≥ 2·len(a)‑1).  Returns the full convolution of length
    2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_from_softmax(p: np.ndarray, a_rev: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Given soft‑max weights `p` (same length as the self‑convolution) and
    the reversed probability vector `a_rev`, compute the gradient
    ∂/∂a  max(conv(a,a)) ≈ 2·(p * a_rev)  (convolution) and return the
    slice that corresponds to the original `a` indices.
    """
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(a_rev, n=fft_len_grad)
    full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * a_rev.size - 2]
    # slice indices n‑1 … n‑1+n‑1 contain the desired n‑vector
    n = a_rev.size
    return 2.0 * full[n - 1 : n - 1 + n]


def _score_from_a(a: np.ndarray, fft_len_self: int) -> float:
    """
    Compute the objective value used by `evaluate_sequence` when the
    vector `a` already sums to 1.
    """
    conv = _self_conv(a, fft_len_self)
    return 2.0 * a.size * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s`
    seconds (a 5 s safety margin is kept for the harness).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    a = None
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
    if a is None:
        # fall back to a Dirichlet draw; length ≈ 3800 (close to the best known)
        n = 3840
        a = rng.dirichlet(np.ones(n)).astype(np.float64)

    n = a.size
    # pre‑compute FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Parameters for the Adam‑based mirror descent (Phase 1)
    # ------------------------------------------------------------------
    eps = 1e-12                     # to avoid log(0)
    a = np.clip(a, eps, None)       # ensure positivity
    a /= a.sum()                    # normalise to 1
    z = np.log(a)                   # log‑softmax parameters

    m = np.zeros_like(z)            # Adam first moment
    v = np.zeros_like(z)            # Adam second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    iter_cnt = 0

    # schedules
    beta_start = 1.0                 # start temperature for soft‑max
    beta_target = 4000.0             # final temperature (≈ exact max)
    lr_start = 1.0                   # initial learning rate on log‑space
    lr_end = 0.001                   # final learning rate

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    cur_score = _score_from_a(a, fft_len_self)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.60 * total_time  # 60 % for mirror descent

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – Adam mirror descent on smooth max
        # --------------------------------------------------------------
        if now < phase1_end:
            # current temperature β (linear annealing)
            frac = (now - start_time) / (phase1_end - start_time)
            beta = beta_start + (beta_target - beta_start) * frac

            # self‑convolution and soft‑max weights
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()
            w = np.exp(beta * (conv - max_c))         # stable: max_c subtracted
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            # gradient of the smooth max w.r.t. a
            grad_a = _grad_from_softmax(p, a[::-1], fft_len_grad)

            # convert to gradient on the soft‑max parameters z
            avg = np.dot(a, grad_a)
            grad_z = a * (grad_a - avg)

            # Adam update (learning‑rate schedule)
            iter_cnt += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** iter_cnt)
            v_hat = v / (1.0 - beta2 ** iter_cnt)
            lr = lr_start * (1.0 - frac) + lr_end * frac
            z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # rebuild a from z and renormalise
            a = np.exp(z)
            a_sum = a.sum()
            if a_sum == 0.0:
                a[:] = 1.0 / n
                z[:] = np.log(a)
            else:
                a /= a_sum
                z = np.log(np.clip(a, eps, None))

            # occasional tiny block jitter to keep diversity
            if rng.random() < 0.02:
                width = rng.integers(1, max(2, n // 15))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.09))
                a[start:start + width] *= factor
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                z = np.log(np.clip(a, eps, None))

            # recompute current score (exact max, not the soft one)
            cur_score = _score_from_a(a, fft_len_self)

        # --------------------------------------------------------------
        # Phase 2 – stochastic local refinement (simulated annealing)
        # --------------------------------------------------------------
        else:
            # linear temperature cooling
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = 0.07 * (1.0 - t_frac) + 1e-6 * t_frac
            temperature = max(temperature, 1e-9)

            # current convolution and peak index
            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            k_max = int(conv.argmax())

            # ----- weighted mass‑transfer move (biased donor) -----
            if rng.random() < 0.85:
                # contribution of each i to the current peak
                i_vals = np.arange(n)
                j_vals = k_max - i_vals
                mask = (j_vals >= 0) & (j_vals < n)
                contrib = np.zeros_like(a)
                contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
                tot = contrib.sum()
                if tot > 0.0:
                    prob = contrib / tot
                    donor = rng.choice(n, p=prob)
                else:
                    donor = rng.integers(0, n)

                if a[donor] > 1e-12:
                    # pick a distinct recipient uniformly at random
                    recipient = rng.integers(0, n - 1)
                    if recipient >= donor:
                        recipient += 1
                    # transfer up to 25 % of the donor's mass
                    delta = rng.random() * 0.25 * a[donor]
                    a_new = a.copy()
                    a_new[donor] -= delta
                    a_new[recipient] += delta
                    new_score = _score_from_a(a_new, fft_len_self)
                    if new_score < cur_score or rng.random() < np.exp(
                        -(new_score - cur_score) / temperature
                    ):
                        a = a_new
                        cur_score = new_score

            # ----- occasional block‑level log‑normal perturbation -----
            else:
                width = rng.integers(1, max(2, n // 15))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.15))
                a[start:start + width] *= factor
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                cur_score = _score_from_a(a, fft_len_self)

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 25.0:
            # perturb current best and restart
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a *= noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            z = np.log(np.clip(a, eps, None))
            m.fill(0.0)
            v.fill(0.0)
            iter_cnt = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final polishing (a few cheap greedy moves)
    # ------------------------------------------------------------------
    # one short pass of guided transfers to squeeze a little more
    for _ in range(200):
        conv = _self_conv(best, fft_len_self)
        k_max = int(conv.argmax())
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[i_vals[mask]] * best[j_vals[mask]]
        if contrib.sum() == 0.0:
            break
        donor = rng.choice(n, p=contrib / contrib.sum())
        if best[donor] < 1e-12:
            continue
        recipient = rng.integers(0, n - 1)
        if recipient >= donor:
            recipient += 1
        delta = rng.random() * 0.15 * best[donor]
        cand = best.copy()
        cand[donor] -= delta
        cand[recipient] += delta
        cand_score = _score_from_a(cand, fft_len_self)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

    # ------------------------------------------------------------------
    # 6) Return result (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # optional sanity check (can be removed if speed is critical)
    try:

        external = evaluate_sequence(best.tolist())
        print(f"[final] external evaluate_sequence = {external:.6f}")
    except Exception:
        pass

    return [float(x) for x in best]
