    # Determine n
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n0 = base.size
        # if base length not big enough, we may enlarge
        desired_n = max(6000, n0)   # choose bigger size for extra degrees

        desired_n = 6000

    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]

def _gradient_from_weights(a, w):
    n = a.size
    Lw = w.size
    wr = w[::-1]  # reversed
    conv_len = next_pow2(n + Lw -1)
    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)
    corr = np.fft.irfft(Fa * Fwr, n=conv_len)
    # Take indices as described
    idx = Lw - 1 - np.arange(n)   # shape n
    grad = 2.0 * corr[idx]

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
* Phase 1 (≈55 % of the time) – smoothed‑max mirror descent.
  – The self‑convolution is computed with an FFT.
  – A weight vector is placed on the top‑fraction of the convolution values
    (instead of a full soft‑max).  The gradient of the weighted sum is obtained
    by one FFT‑based correlation.
  – An Adam‑style adaptive step is applied to an exponentiated‑gradient
    (multiplicative) update, followed by projection onto the probability simplex.
  – Occasionally a random contiguous block is perturbed multiplicatively.
    If no improvement is seen for >60 s a global jitter reset is performed.

* Phase 2 (≈45 % of the time) – local refinement.
  – Starting from the best vector of Phase 1 we maintain its exact
    self‑convolution and perform many O(n) pairwise mass‑transfer moves
    (simulated‑annealing style).  Improvements are always accepted,
    uphill moves are accepted with a temperature that decays linearly.
  – From time to time a random block perturbation is injected.

All numbers are kept non‑negative, projected onto the simplex (sum = 1) and
clipped to the limits of ``evaluate_sequence`` before returning.  The routine
runs for at most the supplied time budget (default 1000 s) and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negatives due to round‑off.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Uses a single FFT to compute the correlation a ⋆ w_rev.
    """
    n = a.size
    Lw = w.size
    wr = w[::-1]                     # reversed for correlation
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    corr_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    # Indices that give the required correlation values.
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * corr_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _pairwise_transfer_exact(
    a: np.ndarray,
    conv: np.ndarray,
    rng: np.random.Generator,
    max_delta_frac: float = 0.3,
) -> tuple[np.ndarray, np.ndarray, float | None]:
    """
    Exact O(n) update that moves a random amount of mass from index i to j.
    Returns (new_a, new_conv, new_score) if the move strictly improves the
    objective, otherwise (a, conv, None).
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    new_conv = conv.copy()
    # donor contribution removed
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    # receiver contribution added
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a
    # diagonal quadratic corrections
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)

    # objective (sum of a is 1 → denominator = 1)
    new_score = 2.0 * n * new_conv.max()
    cur_score = 2.0 * n * conv.max()
    if new_score < cur_score - 1e-15:
        return a2, new_conv, new_score
    else:
        return a, conv, None

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best coefficient list found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size and initial vector
    # ------------------------------------------------------------------
    # Target length – a larger n gives a lower theoretical bound.
    desired_n = 6000
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            desired_n = max(desired_n, base.size)
            a0 = base
        else:
            a0 = None
    else:
        a0 = None

    n = int(desired_n)

    if a0 is not None:
        # Pad / truncate to the chosen length.
        if a0.size > n:
            a = a0[:n].copy()
        else:
            pad = np.zeros(n - a0.size, dtype=np.float64)
            a = np.concatenate([a0, pad])
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0:
            a = a / total
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # Sinusoidal start – never zero, smooth shape.
        x = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (x + 0.5) / n)
        a = np.clip(a, 0.0, None)
        a = a / a.sum()

    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_conv = _self_conv_fft(best, fft_len_self)
    best_score = 2.0 * n * best_conv.max()          # sum(best)=1 → denominator = 1
    last_improve = time.monotonic()

    start_time = time.monotonic()
    # safety margin of a few seconds so we never overshoot the budget
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    phase_split = 0.55                      # fraction of total time for Phase 1

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # Learning‑rate schedule (linear in time)
    eta_start = 0.5
    eta_end = 0.02

    # Fraction of the convolution kept in the weight vector (top‑percent)
    top_frac = 0.02

    block_perturb_prob = 0.02               # probability per iteration

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # --------------------------------------------------------------
        # Phase decision
        # --------------------------------------------------------------
        if frac < phase_split:
            # ------------------- Phase 1: smoothed‑max descent ---------------
            conv = _self_conv_fft(a, fft_len_self)
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv               # sum(a)=1

            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                best_conv = conv.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] Phase1 new best {best_score:.6f}")

            # Build a weight vector that concentrates on the top‑fraction
            thresh = max_conv * (1.0 - top_frac)
            w = np.maximum(conv - thresh, 0.0)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

            # Gradient of the weighted sum (correlation)
            grad = _gradient_from_weights(a, w)

            # Adam‑style adaptive step (scalar eta, per‑coordinate scaling)
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            # Linear schedule for η
            eta = eta_start * (1.0 - frac) + eta_end * frac
            # Scaled gradient (prevent division by zero)
            scaled = m_hat / (np.sqrt(v_hat) + 1e-8)

            # Exponentiated‑gradient update
            a = a * np.exp(-eta * scaled)
            a = np.clip(a, 1e-12, None)      # keep tiny positive mass
            a = _project_onto_simplex(a)

            # Occasional random block perturbation
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # Stagnation escape – global jitter if nothing improved for a minute
            if now - last_improve > 60.0:
                noise = np.exp(rng.normal(0.0, 0.3, size=n))
                a = a * noise
                a = _project_onto_simplex(a)
                m = np.zeros_like(a)
                v = np.zeros_like(a)
                t_iter = 0
                last_improve = now
                print(f"[{elapsed:6.2f}s] jitter reset – best {best_score:.6f}")

        else:
            # ------------------- Phase 2: local refinement -----------------
            # Initialise local‑search structures lazily.
            if 'ref_a' not in locals():
                ref_a = best.copy()
                ref_conv = _self_conv_fft(ref_a, fft_len_self)
                ref_score = 2.0 * n * ref_conv.max()
                # Temperature for simulated‑annealing style acceptance
                temp_start = 0.05 * ref_score           # 5 % of current score
                temp_end = 1e-5 * ref_score
                ref_start = now
                ref_deadline = deadline

            # Linear temperature schedule
            ref_elapsed = now - ref_start
            ref_total = ref_deadline - ref_start
            temp_frac = max(0.0, 1.0 - ref_elapsed / ref_total)
            T = temp_start * temp_frac + temp_end * (1.0 - temp_frac)

            # Pairwise mass‑transfer attempts
            pairwise_prob = 0.9
            if rng.random() < pairwise_prob:
                new_a, new_conv, new_score = _pairwise_transfer_exact(
                    ref_a, ref_conv, rng, max_delta_frac=0.2)

                if new_score is not None:
                    # deterministic improvement
                    ref_a, ref_conv, ref_score = new_a, new_conv, new_score
                    if new_score < best_score:
                        best_score = new_score
                        best = new_a.copy()
                        last_improve = now
                        print(f"[{elapsed:6.2f}s] refinement improved to {best_score:.6f}")
                else:
                    # uphill move – Metropolis acceptance
                    cur_score = 2.0 * n * ref_conv.max()
                    if cur_score > ref_score:
                        # should not happen (because sum is 1), but keep consistency
                        ref_score = cur_score
                    delta = cur_score - ref_score
                    if delta > 0 and rng.random() < math.exp(-delta / max(T, 1e-12)):
                        # accept uphill state (the move has already been applied)
                        ref_a, ref_conv, ref_score = new_a, new_conv, cur_score

            # Occasionally apply a full block perturbation in the refinement phase
            if rng.random() < 0.02:
                ref_a = _random_block_perturb(ref_a, rng)
                ref_conv = _self_conv_fft(ref_a, fft_len_self)
                ref_score = 2.0 * n * ref_conv.max()
                if ref_score < best_score:
                    best_score = ref_score
                    best = ref_a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] block perturb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
