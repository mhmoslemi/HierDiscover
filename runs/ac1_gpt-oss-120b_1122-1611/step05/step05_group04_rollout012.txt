"""
Propose a candidate step function minimizing ``evaluate_sequence``.

The algorithm consists of two stages:

1. **Adam on a smooth‑max approximation**  
   The true objective contains ``max(conv)`` which is non‑differentiable.
   We replace it by a soft‑max (log‑sum‑exp) with a temperature β that
   grows from a small to a very large value.  The variables are kept on the
   probability simplex by working in log‑space (logits y) and applying the
   soft‑max mapping ``a = softmax(y)``.  Adam (adaptive moment estimation)
   is used for the updates – this converges faster than the simple
   exponentiated‑gradient used previously.

2. **Simulated‑annealing refinement with exact O(n) convolution updates**  
   Starting from the best point found by Adam we perform a large number of
   mass‑transfer moves ``a_i ← a_i – δ, a_j ← a_j + δ``.  The self‑convolution
   is updated in linear time (see ``_pairwise_move``).  A temperature schedule
   allows uphill moves early on.  In addition we try a *targeted* move that
   attacks the current maximal convolution lag, and we occasionally apply
   a multiplicative block perturbation.

Both phases respect the 1000 s budget.  The best sequence found is returned
as a plain Python list of floats (already clipped to the limits imposed by
``evaluate_sequence``)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # eliminate tiny negatives
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Derivation: grad_j = 2·Σ_i a_i·w_{i+j}.
    Implemented as a correlation a ⋆ w using a single FFT.
    """
    n = a.size
    Lw = w.size                     # = 2*n‑1
    wr = w[::-1]                    # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)   # length for convolution a ⋆ wr

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)         # mapping to obtain correlation result
    grad = 2.0 * conv_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)

    return _project_onto_simplex(out)


def _pairwise_move(a: np.ndarray,
                   conv: np.ndarray,
                   donor: int,
                   receiver: int,
                   delta: float):
    """
    Exact O(n) update for moving a quantity ``delta`` from index ``donor``
    to index ``receiver``.  Returns the new coefficient vector,
    the updated convolution and the new objective value.
    """
    n = a.size
    new_conv = conv.copy()

    # remove donor contribution
    new_conv[donor:donor + n] -= 2.0 * delta * a
    # add receiver contribution
    new_conv[receiver:receiver + n] += 2.0 * delta * a

    # diagonal self‑terms
    new_conv[2 * donor] += delta * delta
    new_conv[2 * receiver] += delta * delta

    # cross‑term correction (ordered pair counted twice)
    if donor != receiver:
        new_conv[donor + receiver] -= 2.0 * delta * delta

    a2 = a.copy()
    a2[donor] -= delta
    a2[receiver] += delta
    a2 = np.clip(a2, 0.0, None)

    new_score = 2.0 * n * new_conv.max()
    return a2, new_conv, new_score

                # viable i such that j = k-i lies in [0, n-1]
                i_low = max(0, k - (n - 1))
                i_high = min(k, n - 1)
                i_vals = np.arange(i_low, i_high + 1)
                j_vals = k - i_vals
                prod = cur_a[i_vals] * cur_a[j_vals]
                idx = int(np.argmax(prod))
                i = int(i_vals[idx])
                j = int(j_vals[idx])

                # choose the larger of the two as donor
                donor = i if cur_a[i] >= cur_a[j] else j
                receiver = rng.integers(0, n)
                if receiver == donor:
                    receiver = (receiver + 1) % n
                max_delta = min(cur_a[donor], max_delta_frac * cur_a[donor])
                if max_delta > 0:
                    delta = rng.random() * max_delta
                    a_new, conv_new, score_new = _pairwise_move(
                        cur_a, cur_conv, donor, receiver, delta)

                    # Metropolis acceptance
                    if score_new < cur_score or rng.random() < math.exp(-(score_new - cur_score) / max(T, 1e-12)):
                        cur_a, cur_conv, cur_score = a_new, conv_new, score_new
                        if cur_score < best_score:
                            best_score = cur_score
                            best = cur_a.copy()
                            last_improve = now
                            print(f"[{elapsed:6.2f}s] refined (targeted) best = {best_score:.6f}")

            # ----------------------------------------------------------
            # Random pairwise transfer (standard SA move)
            # ----------------------------------------------------------
            if rng.random() < pairwise_prob:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    max_delta = min(cur_a[i], max_delta_frac * cur_a[i])
                    if max_delta > 0:
                        delta = rng.random() * max_delta
                        a_new, conv_new, score_new = _pairwise_move(
                            cur_a, cur_conv, i, j, delta)

                        if score_new < cur_score or rng.random() < math.exp(-(score_new - cur_score) / max(T, 1e-12)):
                            cur_a, cur_conv, cur_score = a_new, conv_new, score_new
                            if cur_score < best_score:
                                best_score = cur_score
                                best = cur_a.copy()
                                last_improve = now
                                print(f"[{elapsed:6.2f}s] refined (random) best = {best_score:.6f}")

            # occasional block perturbation (full recompute of convolution)
            if rng.random() < 0.01:
                cur_a = _random_block_perturb(cur_a, rng)
                cur_conv = _self_conv_fft(cur_a, fft_len_self)
                cur_score = 2.0 * n * cur_conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best = cur_a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] block perturb improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:                     # safeguard against degenerate case
        best = np.full_like(best, 1.0 / best.size)
    return [float(x) for x in best]
