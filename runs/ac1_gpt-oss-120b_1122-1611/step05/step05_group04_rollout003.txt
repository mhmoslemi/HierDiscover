"""
Hybrid Linear‑Programming + Simple Local Search for the step‑function problem.

Algorithm
---------
1. Initialise the candidate from the global variable `height_sequence_1`
   (if it exists) otherwise use a uniform distribution.  All vectors are
   normalised to sum = 1 because the evaluation metric is homogeneous.

2. Repeatedly solve a small linear programme

        maximise   Σ_j b_j
        subject to   b_j ≥ 0
                     (f * b)_k ≤ M   for all k

   where `f` is the current candidate, `M = max(f*f)`.  The programme
   finds a direction `b` that can increase the total mass while keeping
   the convolution with `f` below the current worst value `M`.  If the
   optimum mass Σb is larger than 1 we move a fraction `t` towards the
   normalised direction `g = b/Σb`.  The step size `t` is reduced by
   halving until the objective (the evaluation score) actually improves.

   The LP uses CVXPY with the ECOS solver; the convolution constraint is
   expressed with `cvxpy.conv`, which is linear in the variable `b`.

3. After the LP phase (or when the time budget is almost exhausted) a
   cheap stochastic refinement is performed: random contiguous blocks are
   multiplicatively perturbed (log‑normal factor) and the result is
   projected back onto the probability simplex.  Improvements are kept.

4. The best vector found is returned as a plain list of floats.  All
   values are clipped to the interval `[0, 1000]` which is the domain
   accepted by `evaluate_sequence`.

The routine respects the 1000 s budget, uses at most two CPU threads
(FFT calls are single‑threaded) and avoids any file or network I/O.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries (non‑negative, noise‑clipped).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the evaluation score for a (already normalised to sum = 1)
    without calling `evaluate_sequence`.  The formula reduces to
    2·n·max(conv) because the denominator is 1.
    """
    conv = _self_conv_fft(a, fft_len)
    return float(2.0 * a.size * conv.max())

        if not improved:
            # If no improvement is possible with any t, stop the LP phase.
            print("[LP] convex combination did not improve – stopping LP")
            break

    # ------------------------------------------------------------------
    # 4) Phase B – cheap stochastic refinement (remaining time)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # random block perturbation
        a_candidate = _random_block_perturb(a, rng)
        cand_score = _score_from_a(a_candidate, fft_len)
        if cand_score < best_score - 1e-12:
            a = a_candidate
            best = a_candidate.copy()
            best_score = cand_score
            print(f"[refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best sequence (clipped to the domain of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
