    v = np.zeros_like(w)
    beta1 = 0.9

    eps = 1e-8
    t_iter = 0

    best = a0.copy()
    best_score = np.inf (or compute)
    # compute best_score
    fft_len = _fft_len_for_n(n)

    start_time = time.monotonic()

        # compute a = softmax w

"""
Adam‑based mirror‑descent on the simplex.

* Represent the step‑function as a soft‑max of real parameters w.
* The objective (with sum a = 1) is 2·n·max_k (a∗a)_k.
* Approximate the true max by a soft‑max over the top convolution entries,
  obtaining a sub‑gradient.
* Update w with Adam (adaptive first‑ and second‑moment estimates) and a
  linear learning‑rate schedule.
* The algorithm works in two phases:
      – a short coarse phase on a modest size (≈2000) to find a good
        shape quickly,
      – an upscale phase (≈6000) where the shape is interpolated and
        refined further.
* Random jitter in log‑space and a simple stagnation‑restart keep the
  search from getting stuck.
* The best sequence seen (clipped to the limits of evaluate_sequence)
  is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT, length = 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence() within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size – two phases (coarse → fine)
    # ------------------------------------------------------------------
    n_coarse = int(kwargs.get("n_coarse", 2000))
    n_fine   = int(kwargs.get("n_fine",   6000))

    # ------------------------------------------------------------------
    # 2) Starting shape (sqrt‑difference) – normalised to sum = 1
    # ------------------------------------------------------------------
    def _initial_a(length: int) -> np.ndarray:
        i = np.arange(length, dtype=np.float64)
        a = np.sqrt(i + 1.0) - np.sqrt(i)
        a /= a.sum()
        return a

    # Use a previously discovered construction if it exists and is better.
    a_initial = _initial_a(n_coarse)
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                    # quick score comparison (same normalisation)
                    fft_len_c = _fft_len_for_n(cand.size)
                    score_c = 2.0 * cand.size * np.max(_conv_self(cand, fft_len_c))
                    fft_len_s = _fft_len_for_n(a_initial.size)
                    score_s = 2.0 * a_initial.size * np.max(_conv_self(a_initial, fft_len_s))
                    if score_c < score_s:
                        a_initial = cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Helper to evaluate the objective (fast, equivalent to evaluate_sequence)
    # ------------------------------------------------------------------
    def _score(a: np.ndarray, fft_len: int) -> float:
        """Return 2·n·max(a∗a) assuming sum(a)=1."""
        return 2.0 * a.size * np.max(_conv_self(a, fft_len))

    # ------------------------------------------------------------------
    # 4) Adam optimiser operating on logits w (soft‑max -> a)
    # ------------------------------------------------------------------
    def _run_phase(a_start: np.ndarray, n_target: int, phase_deadline: float,
                   best_overall, best_score_overall):
        """Optimise a on a single phase and return the best (a, score)."""
        # Resize / interpolate start if needed
        if a_start.size != n_target:
            x_old = np.linspace(0, 1, a_start.size)
            x_new = np.linspace(0, 1, n_target)
            a = np.interp(x_new, x_old, a_start)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        else:
            a = a_start.copy()

        # logits for soft‑max
        w = np.log(a + 1e-30)

        # Adam moments
        m = np.zeros_like(w)
        v = np.zeros_like(w)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # Scheduling parameters
        lr_start = 0.3
        lr_end   = 0.004
        beta_start = 0.5      # temperature for soft‑max over conv entries
        beta_end   = 3000.0

        # Stagnation restart handling
        last_improve = time.monotonic()
        restart_gap = 180.0   # seconds without improvement before a restart

        fft_len = _fft_len_for_n(n_target)

        # initialise best for this phase
        cur_score = _score(a, fft_len)
        best_a = a.copy()
        best_score = cur_score

        it = 0
        while time.monotonic() < phase_deadline:
            it += 1
            # ------------------------------------------------------------------
            #  a = softmax(w)
            # ------------------------------------------------------------------
            max_w = np.max(w)
            exp_w = np.exp(w - max_w)
            a = exp_w / np.sum(exp_w)

            # ------------------------------------------------------------------
            #  convolution & current score
            # ------------------------------------------------------------------
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n_target * cur_max

            # ------------------------------------------------------------------
            #  record improvement
            # ------------------------------------------------------------------
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] phase {n_target} new best = {best_score:.6f}")

            # ------------------------------------------------------------------
            #  active‑set selection (indices close to the maximum)
            # ------------------------------------------------------------------
            # relative tolerance – keep entries within 0.5 % of the current maximum
            tol = cur_max * 0.995
            active = np.where(conv >= tol)[0]
            if active.size > 14:
                # keep only the largest 14 entries (empirically enough)
                active = np.argsort(conv)[-14:]

            # ------------------------------------------------------------------
            #  temperature schedule
            # ------------------------------------------------------------------
            now = time.monotonic()
            elapsed = now - start_time
            total = deadline - start_time
            frac = elapsed / total if total > 0 else 0.0
            beta = beta_start + (beta_end - beta_start) * frac

            # ------------------------------------------------------------------
            #  soft weights over the active set (smooth approximation of the max)
            # ------------------------------------------------------------------
            w_raw = np.exp(beta * (conv[active] - cur_max))
            w_sum = w_raw.sum()
            if w_sum > 0.0:
                weights = w_raw / w_sum
            else:
                weights = np.full_like(w_raw, 1.0 / w_raw.size)

            # ------------------------------------------------------------------
            #  gradient of the surrogate (weighted sum of (a∗a)_k)
            # ------------------------------------------------------------------
            grad_a = np.zeros_like(a)
            n = n_target
            for idx, wt in zip(active, weights):
                lo = max(0, idx - (n - 1))
                hi = min(n - 1, idx)
                i_range = np.arange(lo, hi + 1, dtype=np.int64)
                # contribution: 2·a[idx‑i] for each i
                grad_a[i_range] += wt * 2.0 * a[idx - i_range]

            # ------------------------------------------------------------------
            #  convert to gradient w.r.t logits w (chain rule through softmax)
            # ------------------------------------------------------------------
            dot = np.dot(a, grad_a)               # scalar ⟨a,∇_a⟩
            grad_w = a * (grad_a - dot)           # ∇_w = J^T ∇_a

            # ------------------------------------------------------------------
            #  Adam update for w
            # ------------------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_w
            v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
            m_hat = m / (1.0 - beta1**it)
            v_hat = v / (1.0 - beta2**it)

            lr = lr_start * (1.0 - frac) + lr_end * frac
            w -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ------------------------------------------------------------------
            #  occasional stochastic jitter (log‑space) to improve exploration
            # ------------------------------------------------------------------
            if rng.random() < 0.04:
                w += rng.normal(0.0, 0.25, size=n)

            # ------------------------------------------------------------------
            #  stagnation restart
            # ------------------------------------------------------------------
            if now - last_improve > restart_gap:
                # re‑initialise logits around the best seen so far
                w = np.log(best_a + 1e-30)
                w += rng.normal(0.0, 0.12, size=n)   # jitter
                m.fill(0.0)
                v.fill(0.0)
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] phase {n_target} restart – best {best_score:.6f}")

        # end of phase loop
        # compare phase best to overall best
        if best_score < best_score_overall:
            best_score_overall = best_score
            best_overall = best_a.copy()
        return best_overall, best_score_overall

    # ------------------------------------------------------------------
    # 5) Run the two‑phase optimisation within the overall budget
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    # Phase 1 – coarse resolution
    phase1_end = start_time + 0.25 * (deadline - start_time)   # 25 % of budget
    best_seq, best_val = a_initial.copy(), _score(a_initial, _fft_len_for_n(a_initial.size))
    best_seq, best_val = _run_phase(best_seq, n_coarse, phase1_end, best_seq, best_val)

    # Phase 2 – fine resolution (continue from the best coarse result)
    best_seq, best_val = _run_phase(best_seq, n_fine, deadline, best_seq, best_val)

    # ------------------------------------------------------------------
    # 6) Final wrap‑up: clip to limits enforced by evaluate_sequence and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
