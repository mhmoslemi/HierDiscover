"""
Propose a step‑function by fast projected gradient descent on a smooth
approximation of the “peak‑of‑convolution’’ objective.

The evaluation metric for a non‑negative vector a (∑a = 1) is
    score = 2·n·max_k (a*a)[k] .
Directly minimising the max is hard, so we replace the max by a
log‑sum‑exp surrogate:
    Φₚ(a) = (1/p)·log(∑ₖ exp(p·(a*a)[k]))  (p≫1)

∇Φₚ(a) can be written as a weighted cross‑correlation and is computed
with a single FFT per iteration (≈ O(n log n)).  We take a
multiplicative‑gradient step
    a ← a·exp(−η·∇Φₚ(a))
and project the result back onto the probability simplex.
The temperature p is slowly increased while the stepsize η is
adaptively reduced when the move is not improving.  Random block
perturbations and occasional restarts keep the search from stagnating.

The routine runs for the supplied time budget (≈ 1000 s) and returns the
best vector found.  All values are non‑negative, sum to 1 and are
clipped to the limits of ``evaluate_sequence`` before returning.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Convolution using a real FFT (fast for lengths up to a few 10⁴)."""
    n = len(u) + len(v) - 1
    L = 1 << ((n - 1).bit_length())          # next power of two
    fu = np.fft.rfft(u, n=L)
    fv = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(fu * fv, n=L)[:n]
    return conv


def _logsumexp_weights(conv: np.ndarray, p: float) -> np.ndarray:
    """
    Return a weight vector wₖ ∝ exp(p·convₖ) that is numerically stable.
    The largest entry of conv gets weight 1, others are ≤1.
    """
    max_val = conv.max()
    # shift to avoid overflow/underflow
    exponent = p * (conv - max_val)
    # clip exponent to keep np.exp in a safe range
    exponent = np.clip(exponent, -80.0, 0.0)
    w = np.exp(exponent)
    return w

            jitter = np.exp(rng.normal(0.0, 0.22, size=n))
            a = a * jitter
            a = _project_simplex(a)
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            eta = 0.15
            p = 5.0
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] RESTART – score reset to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence (clipped to evaluate_sequence limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best.tolist()]
