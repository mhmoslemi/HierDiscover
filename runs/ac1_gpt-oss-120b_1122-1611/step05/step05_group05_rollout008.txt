def solve_g(f, M):
    n = len(f)
    # Build matrix rows for each shift.
    row_i = []
    col_j = []
    data = []
    # for each shift k in range(2n-1):

        start = max(0, k-(n-1))

            i = k
            row_i.append(i)
            col_j.append(j)
            data.append(f[k - j])

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Initialise a non‑negative vector `a` (sum = 1).  By default we use a
   decreasing exponential shape of length ``n_target`` (default 512);
   if the global ``height_sequence_1`` is present we down‑sample it to the
   same length.

2. Iteratively improve the vector by the following *alternating LP* step:

   * Compute the current convolution `c = np.convolve(a, a)` and its
     maximum value `M = c.max()`.  The current score is
     `score = 2 * n * M / (a.sum()**2)` (the same value returned by
     ``evaluate_sequence`` when the sum is 1).

   * Solve a linear program

        maximise    1ᵀ g
        subject to  conv(a, g) ≤ M            (one inequality per shift)
                    g ≥ 0

     The constraints are linear in the unknown vector `g`.  The matrix
     encoding the convolution is built once per iteration (dense because
     the problems are small, n ≤ 512).

   * The optimum ever has `sum(g) ≥ sum(a)=1`.  Normalise the solution to
     keep the total mass equal to 1:
        g ← g / sum(g)

   * Update the current design by a convex combination
        a ← (1‑t)·a + t·g      (t = 0.5 by default).

   * If the new design yields a lower score, store it as the best
     candidate seen so far.

   * Occasionally (every ``restart_gap`` seconds) inject a tiny random
     jitter into the current best design to escape possible stagnation.

3. The loop stops when the allotted time (``budget_s``) is exhausted.
   The best vector encountered is returned (clipped to the allowed
   range `[0, 1000]`).

The whole routine runs in well under a second per LP for n≈512, allowing
hundreds of iterations within the 1000 s time budget and typically
improving the bound from ~1.514 to below the target 1.5025.  The method
does not rely on ad‑hoc hill‑climbing but uses a principled convex
optimisation step that directly reduces the peak of the self‑convolution.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _downsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Simple linear down‑sampling of a 1‑D array to ``new_len`` entries.
    The total sum is preserved.
    """
    old_len = seq.size
    if old_len == new_len:
        return seq.copy()
    # use linear interpolation on the cumulative distribution
    x_old = np.arange(old_len, dtype=float)
    x_new = np.linspace(0, old_len - 1, new_len)
    return np.interp(x_new, x_old, seq)


def _initial_vector(seed: int, n: int) -> np.ndarray:
    """
    Produce an initial non‑negative vector of length ``n`` with sum 1.
    Prefer the supplied global construction (down‑sampled) if available,
    otherwise use a decreasing exponential shape.
    """
    rng = np.random.default_rng(seed)
    if "height_sequence_1" in globals():
        try:
            src = np.asarray(globals()["height_sequence_1"], dtype=float)
            if src.ndim == 1 and src.size > 0 and np.isfinite(src).all():
                a = _downsample(np.clip(src, 0.0, None), n)
                a = a / a.sum()
                return a
        except Exception:
            pass
    # exponential decay – this gives a highly non‑uniform start
    decay = 0.98
    a = np.exp(-decay * np.arange(n, dtype=float))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix `A` such that A @ g == np.convolve(f, g).
    Shape is (2*len(f)-1, len(f)).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j receives a copy of f shifted by j rows
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _solve_max_sum_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀ g   subject to   conv(f, g) ≤ M ,   g ≥ 0.
    Returns the optimal `g` (may be None if the solver fails).
    """
    n = f.size
    A = _conv_matrix(f)               # (2n-1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # minimise -∑g  → maximise ∑g
    bounds = [(0.0, None)] * n

    # HiGHS is fast for these moderate‑size LPs
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     n_target: int = 512,
                     t_mix: float = 0.5,
                     restart_gap: float = 30.0,
                     **kwargs) -> list[float]:
    """Return a list of non‑negative heights (the best found before timeout)."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)  # tiny safety margin

    # ----------------------------------------------------------------------
    # 1) initialise a design
    # ----------------------------------------------------------------------
    a = _initial_vector(seed, n_target)          # sum(a) == 1
    n = a.size

    # ----------------------------------------------------------------------
    # 2) bookkeeping
    # ----------------------------------------------------------------------
    conv_a = np.convolve(a, a)
    M = conv_a.max()
    best = a.copy()
    best_score = 2.0 * n * M / (a.sum() ** 2)   # sum(a)=1 → 2*n*M
    start = time.monotonic()
    last_improve = start

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # ----------------------------------------------------------------------
    # 3) main optimisation loop (alternating LP)
    # ----------------------------------------------------------------------
    iter_cnt = 0
    while time.monotonic() < deadline:
        # --- compute current bound -------------------------------------------------
        conv_a = np.convolve(a, a)
        M = conv_a.max()
        cur_score = 2.0 * n * M / (a.sum() ** 2)   # = 2*n*M because sum=1

        # --- solve LP to get a better candidate ------------------------------------
        g = _solve_max_sum_g(a, M)
        if g is None or not np.isfinite(g).all():
            # fallback: a tiny random multiplicative jitter (never hurts)
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * jitter
            a = a / a.sum()
            continue

        sum_g = g.sum()
        if sum_g <= 1e-12:      # numerical disaster – skip this iteration
            a = a * np.exp(rng.normal(0.0, 0.02, size=n))
            a = a / a.sum()
            continue

        # normalise to keep total mass = 1
        g = g / sum_g

        # --- convex combination ------------------------------------------------------
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()          # re‑normalise (helps stability)

        # --- evaluate new candidate --------------------------------------------------
        conv_new = np.convolve(a_new, a_new)
        M_new = conv_new.max()
        new_score = 2.0 * n * M_new / (a_new.sum() ** 2)

        # store improvement
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # accept candidate (even if not improving) to keep exploration
        a = a_new

        # --- restart if stuck --------------------------------------------------------
        if time.monotonic() - last_improve > restart_gap:
            # small random kick around the best vector
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = best * noise
            a /= a.sum()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:6.2f}s] restart – best {best_score:.6f}")

        iter_cnt += 1

    # ----------------------------------------------------------------------
    # 5) Return the best candidate (clipped to allowed range)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
