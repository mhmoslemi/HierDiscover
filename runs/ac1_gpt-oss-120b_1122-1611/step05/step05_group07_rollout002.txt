"""
propose_candidate
=================

A two‑phase stochastic optimisation that searches for a non‑negative step
function (probability vector) minimising

    score = 2·n·max(a*a)          (sum(a) is normalised to 1)

The algorithm works inside the allotted time budget (≈ 1000 s) and does
not rely on any external files.

*Phase 1 – Adam‑based mirror descent*  
The vector is represented by unconstrained logits `x`.  
`a = softmax(x)` guarantees `a_i ≥ 0` and `∑a_i = 1`.  
At each iteration we compute the self‑convolution `c = a*a` (FFT) and
a smooth approximation of the max‑norm by a soft‑max with temperature
β that increases from ≈ 0.5 to ≈ 5000.  The gradient of
`max_β(c) = log‑sum‑exp(βc)/β` w.r.t. `a` is

    g_a = 2·(p  *  rev(a))

where `p` is the soft‑max weight vector and `*` denotes linear
convolution.  Using the Jacobian of the soft‑max we obtain the gradient
in logit space and update `x` with the Adam optimiser (β₁=0.9, β₂=0.999).

Occasionally a small multiplicative block perturbation (log‑normal)
is applied to escape shallow minima.

*Phase 2 – targeted mass‑transfer refinement*  
After the descent phase we take the best vector found so far and perform
a short simulated‑annealing style local search.  For the current peak
index `k_max = argmax(a*a)` we compute the contribution
`a_i·a_{k_max-i}` of each coordinate, pick a donor index proportionally
to that contribution and move a random fraction (≤ 20 %) of its mass to
a random recipient.  Moves that lower the true score are always kept,
otherwise they are accepted with a tiny Metropolis probability.
A few hundred such moves often improve the peak a little further.

The outer loop repeatedly launches independent trials with random
lengths (≈ 800 – 6000) and different random seeds; the best sequence
encountered over the whole budget is returned.

All numbers are clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a power‑of‑two FFT length.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of a and b using a power‑of‑two FFT length.
    Returns an array of length len(a)+len(b)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)[: a.size + b.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    safety_margin = 5.0
    deadline = time.monotonic() + max(0.0, float(budget_s) - safety_margin)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Global best bookkeeping
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 2) Optional warm start from a previously discovered construction
    # ------------------------------------------------------------------
    warm_start = None
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0.0:
                warm_start = cand / s

    # ------------------------------------------------------------------
    # 3) Main outer loop – launch independent trials until time runs out
    # ------------------------------------------------------------------
    trial = 0
    while time.monotonic() < deadline:
        trial += 1

        # --------------------------------------------------------------
        # 3.2) Initialise logits (log‑softmax parameters)
        # --------------------------------------------------------------

            eps = 1e-12

        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_lr = 0.20

        # 3.3) Phase 1 – Adam‑based mirror descent on the simplex
        # --------------------------------------------------------------
        max_iter = 1800                      # safe upper bound per trial
        beta_start, beta_end = 0.5, 5000.0   # temperature schedule

            # ---------- self‑convolution ----------
            conv = _self_conv(a, fft_len_self)        # length 2n‑1
            max_c = conv.max()

            beta = beta_start + (beta_end - beta_start) * (it / max_iter)
            w = np.exp(beta * (conv - max_c))          # non‑positive exponents

            if w_sum == 0.0:

            else:
                p = w / w_sum

            rev_a = a[::-1]
            fp = np.fft.rfft(p, n=fft_len_grad)
            fra = np.fft.rfft(rev_a, n=fft_len_grad)

            c_scalar = np.dot(grad_a, a)          # chain rule term

            m = beta1 * m + (1.0 - beta1) * g_logit

            if rng.random() < 0.02:

        # --------------------------------------------------------------
        # 3.4) Phase 2 – local stochastic refinement around the best seen
        # --------------------------------------------------------------
        if best_seq is None:
            continue

        # work on a copy of the best vector of the *current* length
        a_loc = best_seq.copy()
        # pad / truncate to match current n if lengths differ
        if a_loc.size != n:
            if a_loc.size > n:
                a_loc = a_loc[:n]
            else:
                pad = np.full(n - a_loc.size, 0.0)
                a_loc = np.concatenate([a_loc, pad])
            a_loc /= a_loc.sum()  # re‑normalise

        conv_loc = _self_conv(a_loc, fft_len_self)
        cur_score = 2.0 * n * conv_loc.max()

        local_moves = 400
        for mv in range(local_moves):

            # contribution of each index to the current peak
            contrib = np.zeros(n)
            for i in range(n):

                if 0 <= j < n:
                    contrib[i] = a_loc[i] * a_loc[j]
            total = contrib.sum()
            if total > 0.0:
                prob = contrib / total
                i = rng.choice(np.arange(n), p=prob)
            else:
                i = rng.integers(0, n)

            if a_loc[i] < 1e-12:
                continue

            # pick a distinct recipient
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move a modest fraction of mass
            delta = rng.random() * 0.20 * a_loc[i]
            a_new = a_loc.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_score = 2.0 * n * conv_new.max()

        # --------------------------------------------------------------
        # (the outer loop will start another independent trial)

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of a modest length
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
