def refine_sequence(a, start_time, deadline, rng):
    a = np.array(a, dtype=np.float64)
    n = a.size
    # Normalise to sum 1
    a = np.clip(a,0,None)
    s = a.sum()
    if s==0: a = np.full(n,1.0/n)
    else: a/=s
    # Precompute FFT lengths
    fft_len_self = _fft_len_for_n(n)
    fft_len_grad = 1 << (( (3*n-2) -1).bit_length())  # maybe next power of two, using _fft_len_for_3n?

    beta = 1.0
    eta = 0.5
    best_local_seq = a.copy()
    best_local_score = evaluate_sequence(a.tolist())

    while time.monotonic() < deadline:
        # compute self-convolution
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa*fa, n=fft_len_self)[:2*n-1]
        # compute weights
        maxb = np.max(conv)
        # shift for numerical stability
        exp_arg = beta * (conv - maxb)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w /= w_sum

        # gradient via convolution with reversed w
        w_rev = w[::-1]
        ga = np.fft.rfft(a, n=fft_len_grad)
        gw = np.fft.rfft(w_rev, n=fft_len_grad)
        conv_g = np.fft.irfft(ga*gw, n=fft_len_grad)[:3*n-2]
        # extract gradient entries
        # g_i = 2 * conv_g[ (2*n-2) - i ]
        idx = (2*n-2) - np.arange(n)
        grad = 2.0 * conv_g[idx]

        # exponentiated gradient update:
        a = a * np.exp(-eta * grad)

        # jitter occasionally
        if rng.random() < 0.005:
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            a *= jitter

        # renorm
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0:
            a[:] = 1.0 / n
        else:
            a /= s

        # adapt schedules
        beta = min(beta * 1.001, 30.0)  # slowly increase
        eta *= 0.9995

        # evaluate periodically
        if rng.random() < 0.02:
            sc = evaluate_sequence(a.tolist())
            if np.isfinite(sc) and sc < best_local_score:
                best_local_score = sc
                best_local_seq = a.copy()

                print(f"[{elapsed:6.2f}s] refined local score = {sc:.6f}")

        # early exit if deadline near.
        if time.monotonic() >= deadline - 0.1:
            break

    return best_local_seq, best_local_score

---------

# Helper FFT utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear (full) convolution of real vectors a and b using FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)
    # keep only the useful part (full linear convolution)
    return conv[: a.size + b.size - 1]


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution of a (a*a) using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Core refinement routine
# ----------------------------------------------------------------------
def _refine_sequence(
    init_seq,
    start_time: float,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Smoothed exponentiated‑gradient refinement.
    Returns (best_seq, best_score) found before deadline.
    """
    a = np.asarray(init_seq, dtype=np.float64).copy()
    n = a.size

    # Normalise to a probability vector (sum = 1)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s

    # FFT length that can accommodate the largest convolution we will need (3n‑2)
    fft_len = _next_pow2(3 * n - 2)

    # Schedules
    eta = 0.5                 # initial step size
    beta = 0.8                # start with a fairly smooth surrogate
    beta_max = 30.0
    eta_decay = 0.9995
    beta_mul = 1.001

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # evaluate a few times quickly to get a feel for the baseline time
    eval_interval_iters = 20
    it = 0
    last_eval = time.monotonic()

    while time.monotonic() < deadline:
        # ----- self‑convolution (b = a*a) -----
        b = _self_conv_fft(a, fft_len)                     # length 2n‑1

        # ----- soft‑max weights -----
        b_max = np.max(b)
        # numerical stable soft‑max
        w = np.exp(beta * (b - b_max))
        w_sum = w.sum()
        if w_sum == 0.0:   # should never happen, but guard against underflow
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        # g_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]                                     # length 2n‑1
        corr = _conv_full_fft(a, w_rev, fft_len)            # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                    # map to positions
        grad = 2.0 * corr[idx]

        # ----- exponentiated‑gradient step (mirror descent on simplex) -----
        # Clip the exponent to avoid overflow/underflow
        step_exp = -eta * grad
        step_exp = np.clip(step_exp, -50.0, 50.0)
        a = a * np.exp(step_exp)

        # occasional jitter to escape flat regions
        if rng.random() < 0.005:
            jitter = np.exp(rng.normal(0.0, 0.06, size=n))
            a *= jitter

        # renormalise to the simplex and enforce the hard caps of evaluate_sequence
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ----- schedule updates -----
        beta = min(beta_max, beta * beta_mul)
        eta *= eta_decay

        # ----- periodic evaluation -----
        it += 1
        if it % eval_interval_iters == 0:
            # cheap check: if enough time has passed, evaluate
            now = time.monotonic()
            if now - last_eval >= 0.18:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] refined best = {sc:.6f}")
                last_eval = now

        # safety break
        if time.monotonic() >= deadline - 0.1:
            break

    return best_seq.tolist(), best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget (default 1000 s) and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0) keep any previously discovered construction as a baseline
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(map(float, globals()["height_sequence_1"]))
            sc = evaluate_sequence(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] baseline score = {sc:.6f} (len {len(cand)})")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1) systematic sweep over arcsine‑type families (α and power p)
    # --------------------------------------------------------------
    # we concentrate on a length close to the current best if it exists,
    # otherwise we try a few representative lengths.
    lengths = [500, 1024, 2048, 3809, 4096]
    if best_seq is not None:
        lengths = [len(best_seq)] + lengths

    alphas = np.linspace(0.30, 0.70, 21)          # fine grid
    powers = np.linspace(0.6, 1.4, 17)           # extra shape freedom

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        for alpha in alphas:
            # pre‑compute the denominator part once per (n,α)
            denom = ((i + 0.5) * (n - 0.5 - i))
            # denominator can be zero only for pathological n; safe‑guard
            denom = np.where(denom <= 0.0, 1e-12, denom)
            base = denom ** (-alpha)                # arcsine‑type raw shape
            for p in powers:
                raw = base ** p
                raw = np.clip(raw, 0.0, 1000.0)
                raw_sum = raw.sum()
                if raw_sum == 0.0:
                    continue
                raw /= raw_sum                        # normalise to sum = 1
                seq = raw.tolist()
                sc = evaluate_sequence(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:6.2f}s] new best = {sc:.6f} (n={n}, α={alpha:.3f}, p={p:.3f})")
                # early exit if we're already under the target
                if best_score <= 1.5025:
                    break
            if best_score <= 1.5025:
                break
        if best_score <= 1.5025:
            break

    # --------------------------------------------------------------
    # 2) mixture with the uniform distribution (λ·arc + (1‑λ)·uniform)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        best_arr = np.asarray(best_seq, dtype=np.float64)
        n = best_arr.size
        uniform = np.full(n, 1.0 / n)
        for lam in np.linspace(0.0, 1.0, 21):
            mixed = lam * best_arr + (1.0 - lam) * uniform
            mixed = np.clip(mixed, 0.0, 1000.0)
            mixed /= mixed.sum()
            sc = evaluate_sequence(mixed.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = mixed.tolist()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] mixed λ={lam:.2f} -> {sc:.6f}")

    # --------------------------------------------------------------
    # 3) Random Dirichlet starts (optional, limited by remaining time)
    # --------------------------------------------------------------
    while time.monotonic() < deadline - 2.0:   # keep a small buffer
        n = 3809
        # Dirichlet with concentration 0.3 gives a spikier start, 1.0 more uniform
        conc = rng.uniform(0.3, 1.5)
        a0 = rng.dirichlet(np.full(n, conc))
        sc = evaluate_sequence(a0.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a0.tolist()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] random Dirichlet (c={conc:.2f}) -> {sc:.6f}")

    # --------------------------------------------------------------
    # 4) Refine the current best candidate with the smoothed EG loop
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline - 1.0:
        # give the refinement the remainder of the budget
        refined_seq, refined_score = _refine_sequence(
            best_seq, start, deadline, rng
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] post‑refinement best = {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5) Fallback – uniform vector if nothing succeeded (should never happen)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Clip once more to obey the hard limits of ``evaluate_sequence``.
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} (len {len(best_seq)})")
    return best_seq
