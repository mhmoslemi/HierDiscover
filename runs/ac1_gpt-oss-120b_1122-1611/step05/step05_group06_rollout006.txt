```
        # Choose mode randomly: adam or eg

            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] New best (mode {mode}) score={best_score:.6f}, n={n}")

        # Polishing of global best if time left
        if best_vec is not None and time.monotonic() < deadline - 0.2:
            # refine with same mode but more iterations
```

assistantfinal**Solution Overview**

The goal is to minimise  

```
score(a) = 2·n·max(conv(a,a)) / (∑a)²
```

where `conv(a,a)` is the linear self‑convolution of a non‑negative vector `a`.  
Because the score is *scale invariant* we can restrict ourselves to the probability
simplex  

```
Σ a_i = 1 ,   a_i ≥ 0 .
```

The algorithm consists of three stages

1. **Random initialisations** – many short runs on a variety of lengths.
2. **Local optimisation** – a fast Adam based gradient descent that
   alternates a hard sub‑gradient of the current worst convolution entry
   with a smooth “log‑sum‑exp” (soft‑max) gradient.  
   After each gradient step the iterate is projected onto the simplex.
3. **Fine‑tuning** – the current global best vector is polished for a long
   time with a more aggressive schedule (larger number of iterations,
   larger `β` for the soft‑max and a smaller learning‑rate).

All helper functions are top‑level, there are no closures or lambdas and the
code respects the 1000 s time budget.

```python
"""
Fast alternating‑gradient optimiser for the self‑convolution L∞ problem.

* works on the probability simplex (∑a = 1, a ≥ 0);
* many cheap random restarts on several lengths;
* each restart performs a short Adam optimisation (sub‑gradient + soft‑max);
* the best vector found is polished with a long Adam run (more iterations,
  higher soft‑max β, slower learning‑rate decay);
* random block perturbations and pairwise mass transfers are used as
  stochastic “jumps’’ to escape shallow local minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities – linear convolution via a power‑of‑two FFT length
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (fast for real FFT)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors `x` and `y` using an FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a * a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – needed after each additive update
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x≥0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the worst convolution entry.
    The worst index i_max is the one attaining conv.max().
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(
    a: np.ndarray,
    conv: np.ndarray,
    beta: float,
    fft_len_grad: int,
) -> np.ndarray:
    """
    Gradient of the smooth max
        f_β(a) = (1/β)·log Σ_i exp(β·conv_i)
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    # convolution of the weight vector with the reversed a
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(
    a: np.ndarray,
    rng: np.random.Generator,
    fft_len_conv: int,
) -> np.ndarray:
    """Transfer a tiny amount of mass from i to j (accept only if it improves)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)

    # cheap evaluation – reuse the already allocated FFT length
    conv_old = _conv_self(a, fft_len_conv).max()
    conv_new = _conv_self(a2, fft_len_conv).max()
    if 2.0 * n * conv_new < 2.0 * n * conv_old:
        return a2
    return a


# ----------------------------------------------------------------------
# Objective (fast, scale‑invariant version of evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    n = a.size
    fft_len = _fft_len_for_len(2 * n - 1)
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Core optimiser – Adam variant
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int,
    subgrad_prob: float = 0.7,
    beta_start: float = 0.5,
    beta_end: float = 5000.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser used both for short restarts and for final polishing.
    Parameters:
        a0               – initial probability vector (∑a = 1)
        max_iters        – number of gradient steps (subject to deadline)
        subgrad_prob     – probability of using the hard sub‑gradient
        beta_start / end – soft‑max temperature schedule
    Returns the best vector seen and its objective value.
    """
    n = a0.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    a = a0.copy()
    best_vec = a.copy()
    best_score = _objective(a)

    # Adam moments (only the second moment is really needed)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ----- gradient choice -------------------------------------------------
        if rng.random() < subgrad_prob:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = beta_start + (beta_end - beta_start) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # ----- Adam additive update --------------------------------------------
        lr = 0.7 / np.sqrt(t)                     # slowly decreasing step size
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # ----- occasional stochastic jumps --------------------------------------
        if rng.random() < 0.04:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.05:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    evaluate_sequence() within the allotted time.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin – the harness stops us as soon as we return
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Start from a previously found construction, if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                sc = _objective(cand)
                best_vec, best_score = cand.copy(), sc
                print(f"Initial candidate (from globals) score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore – a mixture of small, medium and large vectors
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3809, 5000, 8000, 12000, 16000]

    # Probability distribution that favours larger lengths a little
    probs = np.array([0.10, 0.12, 0.18, 0.20, 0.20, 0.10, 0.10])
    probs = probs / probs.sum()

    # ------------------------------------------------------------------
    # Main search loop – many cheap restarts
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length
        n = int(rng.choice(lengths, p=probs))
        a0 = rng.random(n)
        a0 = a0 / a0.sum()

        # short Adam run (few hundred iterations) – fast exploration
        a_opt, sc_opt = _adam_optimize(
            a0,
            rng,
            deadline,
            max_iters=800,          # 800 is fast enough for a single restart
            subgrad_prob=0.7,
            beta_start=0.5,
            beta_end=2000.0,
        )

        if sc_opt < best_score:
            best_score = sc_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})"
            )

        # ------------------------------------------------------------------
        # If we still have time, polish the *current* global best.
        # The polishing run is much longer and uses a higher β for a tighter
        # approximation of the true max.
        # ------------------------------------------------------------------
        if best_vec is not None and time.monotonic() < deadline - 0.5:
            a_pol, sc_pol = _adam_optimize(
                best_vec.copy(),
                rng,
                deadline,
                max_iters=4000,
                subgrad_prob=0.6,
                beta_start=0.5,
                beta_end=5000.0,
            )
            if sc_pol < best_score:
                best_score = sc_pol
                best_vec = a_pol.copy()
                elapsed = time.monotonic() - start_time
                print(
                    f"[{elapsed:6.2f}s] Polished global best: {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 3000
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final safety clipping – evaluate_sequence does this as well,
    # but we keep the vector well‑behaved.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Return plain Python floats for the harness
    return [float(x) for x in best_vec]
```
