def simplex_projection(v):
    # v shape (n,)
    # ensure non-negative? we can handle any.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (fast and numerically stable).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a*a (linear convolution) of a with itself, length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the objective (identical to evaluate_sequence for
# non‑negative, non‑zero sums).
# ----------------------------------------------------------------------
def _fast_score(seq: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv) for a sequence with sum(seq)=1."""
    n = seq.size
    conv = _conv_self(seq, fft_len)
    max_b = float(conv.max())
    # sum(seq) is 1 by construction, but keep the formula for safety.
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Parameterised family a_i = ( (i+off+1)^p - (i+off)^p ), normalised.
# ----------------------------------------------------------------------
def _build_powerlaw(p: float, off: float, n: int) -> np.ndarray:
    """Generate a non‑negative vector of length n from parameters p, off."""
    i = np.arange(n, dtype=np.float64)
    # avoid negative bases when off=0 and p<0 (p is never negative here)
    a = np.power(i + off + 1.0, p) - np.power(i + off, p)
    # Numerical safety: tiny negatives can appear due to rounding.
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        # fall back to a uniform vector (should never happen)
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size ------------------------------------------------
    # --------------------------------------------------------------
    # The length can be overridden via kwargs; default is 3000.
    n = int(kwargs.get("n", 3000))
    if n < 2:
        n = 2

    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 2) Very simple analytic start (sqrt‑difference) ----------------
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(i + 1.0) - np.sqrt(i)
    a_sqrt /= a_sqrt.sum()
    best = a_sqrt.copy()
    best_score = _fast_score(best, fft_len)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # spare second for tidy‑up

    # --------------------------------------------------------------
    # 3) Randomised search in the (p, offset) family ----------------
    # --------------------------------------------------------------
    # We keep the best candidate from this cheap search and use it as
    # the starting point for a full sub‑gradient optimisation.
    max_trials = 2000
    for trial in range(max_trials):
        if time.monotonic() > deadline:
            break
        # p in (0.25, 0.80) – the “√‑difference’’ corresponds to p≈0.5
        p = rng.uniform(0.25, 0.80)
        # offset ≥0 – a small offset smooths the singularity at i=0
        off = rng.exponential(scale=1.5)   # mean ≈1.5, allows zero too
        cand = _build_powerlaw(p, off, n)
        cand_score = _fast_score(cand, fft_len)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] random‑search improvement: score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # 4) Projected sub‑gradient descent on the true max‑convolution ---
    # --------------------------------------------------------------
    a = best.copy()            # current iterate
    grad_prev = np.zeros_like(a)
    momentum = 0.85            # classic heavy‑ball momentum
    # learning‑rate schedule: start relatively large, finish tiny
    eta_start = 0.6
    eta_end = 0.001

    while time.monotonic() < deadline:
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] sub‑grad new best: score={best_score:.6f}")

        # ----------------------------------------------------------
        # sub‑gradient of the (exact) max‑convolution
        # ----------------------------------------------------------
        j = int(conv.argmax())               # an index attaining the max
        lo = max(0, j - (n - 1))
        hi = min(n - 1, j)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[j - idx]         #  ∂(a*a)_j / ∂a_i

        # momentum blending
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----------------------------------------------------------
        # step‑size (linear decay) and Euclidean update + projection
        # ----------------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        a_new = a - eta * grad
        a = _simplex_projection(a_new)

        # ----------------------------------------------------------
        # occasional stochastic jitter (helps to escape shallow minima)
        # ----------------------------------------------------------
        if rng.random() < 0.015:
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * jitter
            a = _simplex_projection(a)

        # ----------------------------------------------------------
        # occasional block perturbation (like in the reference code)
        # ----------------------------------------------------------
        if rng.random() < 0.01:
            width = rng.integers(1, max(2, n // 15))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.22))
            a[start:start + width] *= factor
            a = _simplex_projection(a)

    # ------------------------------------------------------------------
    # 5) Return the best sequence (respect the clipping performed by
    #    evaluate_sequence).  The sequence already sums to one and is
    #    non‑negative, but we still apply the hard limits to be safe.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
