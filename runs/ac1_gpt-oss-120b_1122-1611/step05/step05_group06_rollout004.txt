def _exp_grad_step(a, grad, lr):
    # Exponential update: a_i := a_i * exp(-lr * grad_i)
    # For numeric stability, we can subtract max grad to avoid overflow.
    exp_factor = np.exp(-lr * (grad - grad.max()))
    a_new = a * exp_factor
    # Normalize to simplex
    sum_a = a_new.sum()
    if sum_a == 0:
        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new /= sum_a
    return a_new

"""
search for a non‑negative step‑function (vector) that minimizes
evaluate_sequence().  The optimiser mixes two first‑order methods:

* Adam (additive update) on the probability simplex – the classic
  approach from the reference implementation.
* Exponentiated‑gradient (mirror descent w.r.t. KL) – naturally stays
  on the simplex and often produces a sharper descent when the loss
  is the (soft‑max) approximation of the max‑convolution.

Both methods receive either a hard sub‑gradient (gradient of the worst
convolution entry) or a smooth‑max gradient (log‑sum‑exp of the
convolution).  Random block perturbations, pairwise mass‑transfer moves
and occasional swaps are injected to escape local minima.  The
algorithm runs many short restarts on various lengths, then spends the
remaining time polishing the best candidate with a longer run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers – linear convolution using real FFT
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors `x` and `y` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient / sub‑gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a (hard gradient of the worst entry).
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Random perturbations
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a modest amount of mass from i to j and keep the simplex."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    # accept only if improvement (cheap check)
    conv2 = _conv_self_fft(a2, fft_len_conv)
    if conv2.max() < _conv_self_fft(a, fft_len_conv).max():
        return a2
    return a


def _swap_move(a: np.ndarray,
               rng: np.random.Generator,
               fft_len_conv: int) -> np.ndarray:
    """Swap two random positions – occasionally keep the swap."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a
    a2 = a.copy()
    a2[i], a2[j] = a2[j], a2[i]
    # accept if it does not increase the score, otherwise keep original
    conv2 = _conv_self_fft(a2, fft_len_conv)
    if conv2.max() <= _conv_self_fft(a, fft_len_conv).max():
        return a2
    # small chance to accept a worse move (simulated annealing)
    if rng.random() < 0.02:
        return a2
    return a


# ----------------------------------------------------------------------
# Core optimiser (combines Adam and exponentiated‑gradient)
# ----------------------------------------------------------------------


def _optimize_mixed(a0: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float,
                    max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Optimise a vector on the simplex using a mixture of Adam (additive)
    and exponentiated‑gradient (multiplicative) updates.
    Returns the best vector found and its raw score 2·n·max(conv).
    """
    n = a0.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state
    m = np.zeros_like(a0)
    v = np.zeros_like(a0)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # EMA for exponentiated‑gradient
    grad_ema = np.zeros_like(a0)

    a = a0.copy()
    best_vec = a.copy()
    best_score = float('inf')

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # --------------------------------------------------------------
        # convolution, current score, bookkeeping
        # --------------------------------------------------------------
        conv = _conv_self_fft(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # choose a gradient source
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = 0.5 + (3000.0 - 0.5) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # decide which update rule to apply
        # --------------------------------------------------------------
        if rng.random() < 0.5:
            # ---------- Adam (additive) ----------
            lr = 0.7 / np.sqrt(t)
            m = beta1 * m + (1.0 - beta1) * raw_grad
            v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
        else:
            # ---------- Exponentiated‑gradient (multiplicative) ----------
            # maintain a moving average of the gradient for stability
            grad_ema = 0.9 * grad_ema + 0.1 * raw_grad
            lr = 0.9 / np.sqrt(t)          # decreasing learning rate
            # multiplicative update; shift to avoid overflow
            exp_factor = np.exp(-lr * (grad_ema - grad_ema.max()))
            a = a * exp_factor

        # --------------------------------------------------------------
        # projection back onto the simplex
        # --------------------------------------------------------------
        a = _project_onto_simplex(a)

        # --------------------------------------------------------------
        # occasional random moves to escape plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.04:
            a = _pairwise_move(a, rng, fft_len_conv)

        if rng.random() < 0.02:
            a = _swap_move(a, rng, fft_len_conv)

    return best_vec, best_score
