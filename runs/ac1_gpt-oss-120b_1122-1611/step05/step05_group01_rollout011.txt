    def refine(seq, time_left):
        a = np.array(seq, dtype=np.float64)
        n = a.size

        fft_len_self = _fft_len_for_n(n)
        # FFT length for cross correlation

        iter = 0
        while time.monotonic() < deadline:
            # compute conv
            conv = _conv_self(a, fft_len_self)
            max_conv = conv.max()

"""
Propose a candidate step‑function that minimizes the evaluation metric.

Strategy
--------
1.  Scan a modest set of lengths (powers of two and the known 3809) and
    a fine grid of the arcsine exponent α (the continuous‑arcsine
    distribution corresponds to α=0.5).  The best of these simple
    candidates provides a solid starting point.

2.  Refine the chosen sequence with a *soft‑max mirror‑descent* on the
    simplex:

    *   Each iteration computes the full self‑convolution `c = a*a`.
    *   A weight vector `w` is formed from a soft‑max of `c`
        (`w_k ∝ exp(β·c_k)`).  With a small β the update reacts to many
        large convolution entries; β is slowly increased so that the
        algorithm finally concentrates on the true maximum.
    *   The gradient of the weighted objective is
        `∂/∂a_i  Σ_k w_k (a*a)_k = 2·Σ_j a_j w_{i+j}`.
        This is a cross‑correlation and is evaluated efficiently by FFT.
    *   An exponentiated‑gradient (mirror descent on the probability
        simplex) updates `a ← a * exp(-η·grad)`, followed by
        normalisation (`∑a = 1`).  Random jitter is injected occasionally
        to escape plateaus.

3.  The best sequence seen during the whole run is returned.  All
    entries are finally clipped to the range required by `evaluate_sequence`
    (0 ≤ x ≤ 1000).

The whole procedure respects the 1000 s budget, prints progress
information and works without any external I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_self(n: int) -> int:
    """FFT length needed for a linear self‑convolution of length n."""
    # Convolution length = 2·n‑1
    return _next_pow2(2 * n - 1)


def _fft_len_cross(n: int) -> int:
    """FFT length needed for the cross‑correlation a⋆w where
    len(a)=n and len(w)=2·n‑1 (result length = 3·n‑2)."""
    return _next_pow2(3 * n - 2)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _cross_corr(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (a ⋆ w)[i] = Σ_j a_j·w_{i+j}.
    Returns the first `len(a)` values (i = 0 … n‑1)."""
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    corr = np.fft.irfft(np.conj(Fa) * Fw, n=fft_len)
    return corr[: a.size]          # shape (n,)


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a probability vector proportional to exp(beta·conv)."""
    # subtract the max for numeric stability
    m = conv.max()
    expw = np.exp(beta * (conv - m))
    s = expw.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return expw / s


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # stop a little early to avoid the process being killed
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq: list[float] | None = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0)  Use the provided baseline if it exists
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(map(float, globals()["height_sequence_1"]))
            sc = evaluate_sequence(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                now = time.monotonic() - start_time
                print(f"[{now:6.2f}s] baseline score = {sc:.6f} "
                      f"(length {len(cand)})")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1)  Deterministic arcsine‑family sweep
    # --------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 3809, 4096, 5000]
    alphas = np.linspace(0.30, 0.70, 21)      # 0.30, 0.32, …, 0.70
    for n in lengths:
        if time.monotonic() > deadline:
            break
        i = np.arange(n, dtype=np.float64)
        for alpha in alphas:
            if time.monotonic() > deadline:
                break
            raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
            raw = np.clip(raw, 0.0, 1000.0)          # safety – never triggered here
            s = raw.sum()
            if s == 0.0:
                continue
            raw /= s                                 # normalise to sum = 1
            seq = raw.tolist()
            sc = evaluate_sequence(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq
                now = time.monotonic() - start_time
                print(f"[{now:6.2f}s] new best (grid) = {sc:.6f} "
                      f"(n={n}, α={alpha:.3f})")

    # --------------------------------------------------------------
    # 2)  Refine the best candidate with soft‑max mirror descent
    # --------------------------------------------------------------
    def _refine(a: np.ndarray):
        """In‑place mirror‑descent refinement of the probability vector a."""
        nonlocal best_seq, best_score
        n = a.size
        fft_len_self = _fft_len_self(n)
        fft_len_cross = _fft_len_cross(n)

        eta = 0.5          # step‑size for the exponentiated update
        beta = 2.0         # start with a fairly smooth max‑approximation
        last_eval = time.monotonic()
        iter_cnt = 0

        while time.monotonic() < deadline:
            # --- self‑convolution and weight construction -----------------
            conv = _conv_self(a, fft_len_self)          # length 2·n‑1
            w = _softmax_weights(conv, beta)            # same length

            # --- gradient of Σ_k w_k (a*a)_k (cross‑correlation) -----
            grad = 2.0 * _cross_corr(a, w, fft_len_cross)   # shape (n,)

            # --- exponentiated‑gradient (mirror descent on simplex) -----
            a *= np.exp(-eta * grad)

            # occasional jitter to escape flat plateaus
            if rng.random() < 0.02:
                jitter = np.exp(rng.normal(0.0, 0.07, size=n))
                a *= jitter

            # ensure positivity and re‑normalise to a probability vector
            a = np.clip(a, 1e-12, None)
            a /= a.sum()

            # --- evaluate periodically ------------------------------------
            now = time.monotonic()
            if now - last_eval > 0.12:          # roughly every ~0.1 s
                cand = a.tolist()
                sc = evaluate_sequence(cand)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = cand
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] refined best = {sc:.6f}")
                last_eval = now

            # --- schedule updates -----------------------------------------
            eta *= 0.9997               # slight decay
            beta = min(beta * 1.001, 30.0)   # slowly sharpen the max‑focus

            iter_cnt += 1
            # safety net
            if now - start_time > budget_s - 0.6:
                break

    if best_seq is not None:
        a0 = np.array(best_seq, dtype=np.float64)
        _refine(a0)
    else:
        # fallback: a tiny uniform vector
        n = 256
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = a0.tolist()
        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 3)  Final sanitisation
    # --------------------------------------------------------------
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
