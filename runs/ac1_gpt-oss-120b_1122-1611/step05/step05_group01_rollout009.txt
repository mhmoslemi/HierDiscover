def _valid_crosscorr(w, a):
    # a: length n, w: length 2n-1.
    n = a.shape[0]
    # compute fft length
    fft_len = 1 << ((len(w) + len(a) - 2).bit_length())
    # compute convolution via FFT
    a_rev = a[::-1]
    f_w = np.fft.rfft(w, n=fft_len)
    f_a_rev = np.fft.rfft(a_rev, n=fft_len)
    prod = f_w * f_a_rev
    conv_full = np.fft.irfft(prod, n=fft_len)[:len(w)+len(a)-1]
    # valid part
    start = n-1
    return conv_full[start:start+n]

mirror‑descent on the self‑convolution.  Starting from an arcsine‑type
distribution for several lengths we run an exponentiated‑gradient update
where the gradient is taken with respect to a temperature‑smoothed
approximation of the max‑convolution.  The temperature is annealed so
that early iterations flatten the whole convolution and later
iterations focus on the remaining peaks.  The best sequence found within
the time budget is returned.

def _fft_len_for_at_least(size: int) -> int:
    """Return the smallest power‑of‑two >= size."""
    if size <= 0:
        return 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real vector using an FFT of length fft_len.
    Returns an array of length 2·len(a)-1."""
    # real FFT is enough (inputs are real)
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _cross_corr_valid_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the valid cross‑correlation r[k] = Σ_i w[i+k]·a[i]
    for k = 0,…,len(a)-1 using an FFT of length fft_len.
    This is equivalent to np.correlate(w, a, mode='valid') but works in O(N log N)."""
    # convolution of w with reversed a
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    prod = fw * fa_rev
    conv_full = np.fft.irfft(prod, n=fft_len)[: len(w) + len(a) - 1]
    start = len(a) - 1
    return conv_full[start:start + len(a)]


def _softmax(x: np.ndarray, temperature: float) -> np.ndarray:
    """Numerically stable soft‑max of x/temperature."""
    if temperature <= 0.0:
        # degenerate case – behave like hard max (one-hot at the max)
        idx = np.argmax(x)
        sm = np.zeros_like(x)
        sm[idx] = 1.0
        return sm
    # shift by max for stability
    shift = np.max(x)
    ex = np.exp((x - shift) / temperature)
    return ex / np.sum(ex)
