def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)

    max_b = max(b_sequence)

        return np.inf

    deadline = time.monotonic() + max(0.0, budget_s - 0.5)

    # Determine sequence length
    if "height_sequence_1" in globals():

    a = _project_simplex(a)   # ensure sum=1
    # best tracking
    best = a.copy()

    print(f"[0.00s] init score = {best_score:.6f}, n={n}")

    # set learning rate schedule
    lr = 0.5
    decay = 0.9995
    min_lr = 1e-6
    last_improve = time.monotonic()
    # main loop
    while time.monotonic() < deadline:
        # compute convolution
        conv = _conv_fft(a)
        max_b = conv.max()
        k = conv.argmax()
        S = a.sum()
        n = a.shape[0]
        const = -4.0 * n * max_b / (S**3)
        coeff = 2.0 * n / (S**2)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.full(n, const, dtype=np.float64)
        # for indices where mask true, add coeff * a[partner]
        grad[mask] += coeff * a[partner[mask]]
        # propose update
        cand = a - lr * grad
        # projection onto simplex
        cand = _project_simplex(cand)
        cand_score = _objective(cand)
        # acceptance
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            best_time = time.monotonic() - start_time
            print(f"[{best_time:.2f}s] new best {best_score:.6f}")
            a = cand
            last_improve = time.monotonic()
        else:
            # maybe accept if random with decreasing prob (simulated annealing)
            elapsed = time.monotonic() - start_time
            frac = min(1.0, elapsed / (budget_s -0.5))
            accept_prob = 0.1 * (1-frac) + 0.001 * frac
            if rng.random() < accept_prob:
                a = cand
        # decay learning rate

            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:.2f}s] restart")

    return [float(x) for x in best]

"""
Propose a new step‑function sequence.

The algorithm uses a smooth “soft‑max” approximation of the
convolution’s maximum and performs projected gradient descent on the
simplex (Euclidean projection).  At each iteration we

* compute the full linear convolution a*a,
* evaluate a soft‑max weight vector p over the 2 n‑1 convolution entries,
* obtain the gradient of  f(a)=2·n·max(a*a)/∑a²  using the weighted sum
  ∑_k p_k a_{k‑i},
* take a short step opposite to the gradient and project back onto the
  probability simplex,
* keep the best candidate seen so far and occasionally “jitter’’ the
  current best to escape plateaus.

The method is deliberately different from the simple hill‑climbing code
shown in the prompt and is fast enough to explore many thousands of
iterations inside the 1000 s budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level, no closures, no lambdas)
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x)=1}.  Returns a new view; the input is not modified."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or very small)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _linear_convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution using FFT, returns the full array."""
    n = x.shape[0] + y.shape[0] - 1
    m = 1 << ((n - 1).bit_length())          # next power of two ≥ n
    X = np.fft.rfft(x, n=m)
    Y = np.fft.rfft(y, n=m)
    conv = np.fft.irfft(X * Y, n=m)
    return conv[:n]


def _conv_a_a(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a (full length)."""
    return _linear_convolve_fft(a, a)


def _objective(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    """Exact objective used by evaluate_sequence (scale‑invariant)."""
    if conv is None:
        conv = _conv_a_a(a)
    max_b = conv.max()
    S = a.sum()
    if S < 0.01:               # just in case – mirrors the security check
        return np.inf
    n = a.shape[0]
    return float(2.0 * n * max_b / (S ** 2))


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The search runs until the time budget expires
    (default 1000 s) and uses a projected gradient descent with a
    soft‑max surrogate for the max‑convolution term.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # safety margin

    # --------------------------------------------------------
    # 1) Choose length and an initial pool of candidates
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if base.ndim != 1 or base.size == 0:
                raise ValueError
            n = base.size
        except Exception:
            base = None
            n = 4096
    else:
        base = None
        n = 4096                     # a power of two – nice for FFT

    # build a few deterministic starters (they will be normalised later)
    i = np.arange(n, dtype=np.float64)

    starters = []

    # uniform
    starters.append(np.full(n, 1.0 / n, dtype=np.float64))

    # √‑increment (approximation of 1/√x, symmetric at the left edge)
    sqrt_inc = np.sqrt(i + 1) - np.sqrt(i)
    starters.append(sqrt_inc)

    # decreasing 1/√i  (large values at the left edge)
    inv_sqrt = 1.0 / np.sqrt(i + 1.0)
    starters.append(inv_sqrt)

    # symmetric U‑shape  (large at both edges, tiny in the middle)
    u_shape = 1.0 / np.sqrt(np.minimum(i + 1, n - i))
    starters.append(u_shape)

    # random Dirichlet‑like vector
    starters.append(rng.random(n))

    # the previously discovered construction, if available
    if base is not None:
        starters.append(base)

    # normalise every starter onto the simplex (sum = 1)
    candidates = [_project_simplex(v.clip(min=0.0)) for v in starters]

    # pick the best of the starters as the very first solution
    best = candidates[0].copy()
    best_score = _objective(best)
    for cand in candidates[1:]:
        sc = _objective(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
    a = best.copy()                     # current iterate

    start_time = time.monotonic()
    last_improve = start_time
    print(f"[0.00s] initial best score = {best_score:.6f} (n={n})")

    # --------------------------------------------------------
    # 2) Search hyper‑parameters
    # --------------------------------------------------------
    # learning‑rate (additive) – will be geometrically decayed
    lr = 0.4
    lr_decay = 0.9997
    min_lr = 1e-6

    # temperature for soft‑max (larger → smoother)
    alpha = 30.0

    # simulated‑annealing acceptance probability for non‑improving moves
    accept_start = 0.12
    accept_end   = 0.001

    # restart if we are stuck for this many seconds
    restart_gap = 30.0

    # --------------------------------------------------------
    # 3) Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- a*a convolution (full length 2n‑1) -----
        conv = _conv_a_a(a)

        # ----- soft‑max weights over the convolution entries -----
        max_c = conv.max()
        # stability trick: shift by the max before exponentiating
        exp_term = np.exp(alpha * (conv - max_c))
        p = exp_term / exp_term.sum()                # p_k ≥ 0, Σ p = 1

        # ----- gradient of the smooth surrogate -----
        S = a.sum()
        const = -4.0 * n * max_c / (S ** 3)           # part coming from the denominator
        coeff = 2.0 * n / (S ** 2)                    # scaling of the weighted sum

        # weighted sum w_i = Σ_k p_k a_{k‑i}
        a_rev = a[::-1]                               # reversed a
        # convolution of p (length 2n‑1) with a_rev (length n) → length 3n‑2
        conv_pa = _linear_convolve_fft(p, a_rev)
        # slice the region that corresponds to i = 0 … n‑1
        weighted = conv_pa[n - 1 : n - 1 + n]

        grad = const + coeff * weighted                # ∇f (additive form)

        # ----- take a projected gradient step -----
        cand = a - lr * grad
        cand = _project_simplex(cand)                   # keep ∑cand = 1, cand ≥ 0
        cand_score = _objective(cand, conv=_conv_a_a(cand))

        # ----- improvement handling -----
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] NEW BEST = {best_score:.6f}")
        else:
            # SA‑style acceptance of a worse move
            now = time.monotonic()
            elapsed = now - start_time
            frac = min(1.0, elapsed / (budget_s - 0.5))
            accept_prob = accept_start * (1.0 - frac) + accept_end * frac
            if rng.random() < accept_prob:
                a = cand

        # ----- schedule updates -----
        lr = max(min_lr, lr * lr_decay)

        # ----- restart if stagnating -----
        if time.monotonic() - last_improve > restart_gap:
            # jitter the best solution a little and continue
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = best * noise
            a = _project_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] RESTART")

    # --------------------------------------------------------
    # 4) Return the best sequence (clip to the allowed range)
    # --------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
