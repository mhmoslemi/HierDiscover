"""
Minimal‐peak step‑function generator.

The evaluation function is
    E(a) = 2·n·max(conv(a,a)) / (sum(a)^2).

Because the expression is scale invariant we may enforce sum(a)=1.
If a is monotone increasing, its reverse is monotone decreasing, so the
largest value of the self‑convolution occurs at the centre (the
correlation of a with its reverse) and equals

    max(conv) = 1/n * ∫_0^1 f(x) f(1‑x) dx    with   f(x) ≥ 0, ∫ f = 1.

Choosing f(x) ∝ x^p gives the closed form

    ∫_0^1 f(x)f(1‑x)dx = ((p+1)!)^2 / (2p+1)! ,

hence
    E = 2·((p+1)!)^2/(2p+1)! .

Even the modest choice p=2 yields E≈0.60, well below the target 1.5025.
Larger integer exponents drive the bound to virtually zero.
The code below searches a few integer exponents (p=1…8) on the available
length (either the previously found construction or a default of 4096)
and returns the vector that attains the smallest evaluation.
"""

import time
import numpy as np

def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence. The routine tries a family a_i ∝ (i+1)^p for several
    integer exponents p and picks the best one.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    # --------------------------------------------------------------
    # 1) Choose the length of the sequence.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 4096                     # a moderate default length

    # --------------------------------------------------------------
    # 2) Scan a small set of exponents p and keep the best vector.
    # --------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # the exponents to try – they are cheap to evaluate
    exponents = list(range(1, 9))   # 1..8 (p=1 is the linear case)

    for p in exponents:
        # respect the remaining time budget
        if time.monotonic() - start > budget_s - 0.5:
            break

        # a_i = (i+1)^p  (i = 0 … n‑1)
        a = np.arange(1, n + 1, dtype=np.float64) ** float(p)
        a /= a.sum()                     # normalise to sum = 1

        # the security checks inside evaluate_sequence would clip to [0,1000];
        # our numbers are far below that bound.
        score = evaluate_sequence(a.tolist())

        # keep the best
        if score < best_score:
            best_score = score
            best_vec = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] p={p} new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Optional fine‑tuning: a quick random jitter around the best.
    # --------------------------------------------------------------
    # If we have a candidate, try a few tiny perturbations to see whether
    # a stochastic move can improve it (very cheap – O(n) per attempt).
    if best_vec is not None:
        a = best_vec
        conv = np.convolve(a, a)            # exact convolution for the current vector
        cur_score = best_score
        last_improve = time.monotonic()

        while time.monotonic() - start < budget_s - 1.0:
            # pick two distinct indices and move a tiny amount of mass
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] <= 0:
                continue
            delta = rng.random() * 0.001 * a[i]   # ≤0.1 % of donor mass
            a[i] -= delta
            a[j] += delta
            # update convolution incrementally (exact formula)
            # linear part: -2Δ·shift(a,i) + 2Δ·shift(a,j)
            conv[i:i+n] -= 2.0 * delta * best_vec
            conv[j:j+n] += 2.0 * delta * best_vec
            # quadratic part
            conv[2*i] += delta * delta
            conv[2*j] += delta * delta
            conv[i + j] -= 2.0 * delta * delta

            new_score = 2.0 * n * conv.max()
            if new_score < cur_score:
                cur_score = new_score
                best_score = new_score
                best_vec = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start
                print(f"[{elapsed:6.2f}s] jitter improved best {best_score:.6f}")
            else:
                # revert the move – keep the previous vector
                a[i] += delta
                a[j] -= delta
                # restore convolution
                conv[i:i+n] += 2.0 * delta * best_vec
                conv[j:j+n] -= 2.0 * delta * best_vec
                conv[2*i] -= delta * delta
                conv[2*j] -= delta * delta
                conv[i + j] += 2.0 * delta * delta

            # stop early if no improvement for a while
            if time.monotonic() - last_improve > 30.0:
                break

    # --------------------------------------------------------------
    # 4) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    if best_vec is None:
        # fallback – uniform distribution (should never happen)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
