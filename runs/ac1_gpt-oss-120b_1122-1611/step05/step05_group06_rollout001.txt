"""
Exponential‑gradient / soft‑max search for a low‑maximum self‑convolution.

The algorithm
----------------
* Works on the probability simplex (∑a = 1, a_i ≥ 0).
* Several initialisation families are tried (random Dirichlet, sine‑power,
  interpolation of the previously best known sequence).
* For each start a local optimiser runs a few thousand iterations.
  – With probability 0.5 we use the exact sub‑gradient of the worst
    convolution entry.
  – Otherwise we use a smooth “log‑sum‑exp’’ (soft‑max) gradient with a
    β that grows exponentially during a run.
  – The update is an *exponentiated‑gradient* (multiplicative) step,
    followed by normalisation (the natural projection onto the simplex).
* Occasionally a random block perturbation or a cheap pairwise mass‑transfer
  move is applied to escape plateaus.
* The best vector found over all restarts is repeatedly polished with a
  longer run.
* The whole search respects the supplied time budget (default 1000 s).

The score that is minimised is
        2·n·max(conv(a,a))          (for ∑a = 1)
which is exactly the value returned by ``evaluate_sequence`` after the
clipping steps.  The code keeps its own fast version of the score to
avoid the overhead of the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. ``a``.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))          # safe, max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    # extract the n entries that correspond to derivative wrt a
    grad_part = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Simple random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a random fraction of mass from i to j (accept only if improving)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    # limit the transferred amount to a modest fraction
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    # compute scores (cheap: just max of convolution)
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a


# ----------------------------------------------------------------------
# Score helper (fast, avoids evaluate_sequence overhead)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return the value 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Local optimisation routine
# ----------------------------------------------------------------------


def _local_optimise(initial: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float,
                    max_iters: int = 1500) -> tuple[np.ndarray, float]:
    """
    Refine ``initial`` (already on the simplex) using exponentiated‑gradient
    steps, occasional soft‑max gradients and random moves.
    Returns the best vector seen and its score.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # keep the best encountered during the run
    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    # exponential schedule for the soft‑max temperature
    beta = 0.5
    beta_mul = 1.08  # multiplicative increase per iteration

    for t in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # Choose a gradient type
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            # exact sub‑gradient of the worst entry (fast, cheap)
            raw_grad = _subgradient_step(a, conv)
        else:
            # smooth max – β grows exponentially
            beta = min(1e5, beta * beta_mul)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # Exponentiated‑gradient update (multiplicative)
        # --------------------------------------------------------------
        # learning rate decreasing like 1/sqrt(t)
        lr = 0.7 / np.sqrt(t)
        # Clip the gradient to keep exp() numerically stable
        grad_clip = np.clip(raw_grad, -20.0, 20.0)
        a = a * np.exp(-lr * grad_clip)
        # re‑normalise to stay on the simplex
        a_sum = a.sum()
        if a_sum == 0.0:
            # numerical disaster – fall back to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # --------------------------------------------------------------
        # Occasional random exploratory moves
        # --------------------------------------------------------------
        if rng.random() < 0.025:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.025:
            a = _pairwise_move(a, rng, fft_len_conv)

    # final evaluation of the best candidate
    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score

def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponentials."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Shape a_i ∝ sin^p( π·(i+0.5)/n ), with p sampled uniformly in [0.5, 4].
    """
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)  # guard against floating‑point round‑off
    a = np.power(vals, p)
    # if all zeros (numerical), fall back to uniform
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linearly interpolate an existing distribution to a different length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= a.sum()
    return a

            else:
                a0 = _init_random_dirichlet(rng, n)

        # local optimisation (short but intensive)
        # we grant each restart a modest number of iterations,
        # proportionally larger for smaller n to stay within the budget.
        iters = int(2000 * (5000.0 / max(n, 500)))  # about 2000 for n≈5000
        a_opt, score_opt = _local_optimise(a0, rng, deadline, max_iters=iters)

        # record improvement
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:5.2f}s] New global best: {best_score:.6f} (n={n})")

        # --------------------------------------------------------------
        # Occasionally devote a bit more time to polishing the current best
        # --------------------------------------------------------------
        if best_vec is not None and time.time() < deadline - 1.0:
            # polishing run – more iterations, slower β growth
            a_polished, score_polished = _local_optimise(
                best_vec.copy(), rng, deadline, max_iters=3000)
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:5.2f}s] Polished best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
