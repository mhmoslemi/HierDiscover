"""
Enhanced search for the step‑function that minimises ``evaluate_sequence``.
The algorithm is a two‑stage hybrid:

1. **Frank‑Wolfe (LP) phase** – repeatedly solves a linear programme that
   maximises the total mass of a candidate vector ``b`` under the current
   convolution bound ``M = max(f∗f)``.  The solution is blended into the
   current distribution ``f`` with a diminishing step size.  This drives the
   peak of the self‑convolution down while preserving the simplex constraint.

2. **Fine‑tuning phase** – a short Adam / exponentiated‑gradient descent
   (very similar to the previous code) refines the best vector found.
   Random block perturbations and pairwise mass‑transfer moves are also used.

The LP is expressed with ``cvxpy`` using the built‑in linear convolution
operator, which keeps the constraints linear in the unknown vector.  If
``cvxpy`` is unavailable the code falls back to the pure gradient‑descent
method.

The whole search respects the supplied time budget (default 1000 s) and
returns the best vector discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‐based convolution helpers (fast and numerically stable)
# ----------------------------------------------------------------------
def _fft_len_for_len(length: int) -> int:
    """Smallest power of two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors via real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]

def _max_conv(a: np.ndarray) -> float:
    """Return max entry of a∗a."""
    n = a.size
    fft_len = _fft_len_for_len(2 * n - 1)
    return _conv_self(a, fft_len).max()

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Gradient‑based local optimiser (same spirit as the reference code)
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the worst convolution entry."""
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad

def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """∇ of smooth max f_β = (1/β)·log Σ exp(β·conv)."""
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)

def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a small random amount of mass from i to j (accept if improves)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    conv = _conv_self(a2, _fft_len_for_len(2 * n - 1))
    if 2.0 * n * conv.max() < 2.0 * n * _conv_self(a, _fft_len_for_len(2 * n - 1)).max():
        return a2
    return a

def _local_optimise(a: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Short Adam‑style optimisation of a (already on the simplex)."""
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state (second moment only)
    v = np.zeros_like(a)
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_vec = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len_conv).max()

    max_iters = 1200                     # a bit longer than the reference
    while t < max_iters and time.monotonic() < deadline:
        t += 1
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # gradient choice
        if rng.random() < 0.55:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = 0.5 + (2000.0 - 0.5) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # Adam‑like additive update (only second moment)
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        v_hat = v / (1.0 - beta2 ** t)
        lr = 0.7 / np.sqrt(t)
        step = lr * raw_grad / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # occasional stochastic moves
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.04:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score

# ----------------------------------------------------------------------
# Frank‑Wolfe (LP) sub‑routine using cvxpy
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:                     # pragma: no cover
    _HAS_CVX = False

def _lp_step(f: np.ndarray, M: float, solver: str = "ECOS") -> np.ndarray | None:
    """
    Solve     max Σ b_j
    s.t.      f * b ≤ M   (linear convolution, f fixed)
              b ≥ 0
    Returns b (may be None if the solver failed).
    """
    if not _HAS_CVX:
        return None

    n = f.size
    b = cp.Variable(n, nonneg=True)
    # linear convolution of constant f with variable b
    conv_expr = cp.conv(cp.Constant(f), b)
    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=solver, warm_start=True, max_iters=20000, abstol=1e-9, reltol=1e-9)
    except Exception:                # pragma: no cover
        return None

    if b.value is None:
        return None
    b_val = np.asarray(b.value).reshape(-1)
    # Numerical cleanup
    b_val = np.maximum(b_val, 0.0)
    return b_val

def _frank_wolfe_optimize(f_init: np.ndarray,
                          max_steps: int,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a Frank‑Wolfe–type optimisation using the LP described above.
    Returns the best vector encountered and its score.
    """
    n = f_init.size
    f = f_init.copy()
    best_vec = f.copy()
    best_score = 2.0 * n * _conv_self(f, _fft_len_for_len(2 * n - 1)).max()

    step_size = 0.5          # initial blending factor
    for it in range(max_steps):
        if time.monotonic() >= deadline:
            break

        M = _max_conv(f)
        # Solve the LP – if unavailable, stop the FW phase
        b = _lp_step(f, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # Scale to unit sum (the simplex) – this may violate the M‑bound,
        # hence we blend cautiously.
        g = b / sum_b

        # Try a few decreasing step sizes until the bound does not increase.
        accepted = False
        t = step_size
        for _ in range(5):
            f_new = (1.0 - t) * f + t * g
            f_new = _project_onto_simplex(f_new)   # keep on simplex
            M_new = _max_conv(f_new)
            if M_new <= M * (1.0 + 1e-9):          # non‑increasing peak
                accepted = True
                break
            t *= 0.5
        if not accepted:
            # unable to improve – shrink step size permanently and continue
            step_size *= 0.5
            continue

        # accept the move
        f = f_new
        cur_score = 2.0 * n * M_new
        if cur_score < best_score:
            best_score = cur_score
            best_vec = f.copy()
            elapsed = time.monotonic()
            print(f"[{elapsed:.2f}s] FW step {it+1}, score = {best_score:.6f}")

        # slowly decay the blending factor
        step_size = max(0.05, step_size * 0.95)

    return best_vec, best_score

    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)   # small safety margin

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, evaluate it first
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cur_score = 2.0 * cand.size * _conv_self(cand, _fft_len_for_len(2 * cand.size - 1)).max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_vec = cand.copy()
                print(f"Initial candidate (from globals) score = {cur_score:.6f}")
        except Exception:                # pragma: no cover
            pass

    # ------------------------------------------------------------------
    # Choose a single length that historically gave a good score.
    # Using the same length as the previously reported best (3809) keeps
    # the search focussed and leaves plenty of time for the optimisation.
    # ------------------------------------------------------------------
    n = 3809
    fft_len_conv = _fft_len_for_len(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise with a uniform distribution (the simplex is a natural start)
    # ------------------------------------------------------------------
    f = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Frank‑Wolfe (+LP) phase – attempts to drive the peak down dramatically.
    # ------------------------------------------------------------------
    if _HAS_CVX:
        # at most 12 LP steps; each step is relatively cheap for n≈3800.
        f, fw_score = _frank_wolfe_optimize(f, max_steps=12,
                                            deadline=deadline,
                                            rng=rng)
        if fw_score < best_score:
            best_score = fw_score
            best_vec = f.copy()
            print(f"[{time.monotonic():.2f}s] After FW phase score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Gradient‑descent fine‑tuning phase (more iterations now that we are close).
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 0.5:
        f_fine, fine_score = _local_optimise(best_vec.copy() if best_vec is not None else f,
                                             rng, deadline)
        if fine_score < best_score:
            best_score = fine_score
            best_vec = f_fine.copy()
            print(f"[{time.monotonic():.2f}s] After fine‑tune score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded (should never happen) fall back to uniform.
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)

    # Final safety clipping (evaluate_sequence does this as well)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python list
    return [float(x) for x in best_vec]
