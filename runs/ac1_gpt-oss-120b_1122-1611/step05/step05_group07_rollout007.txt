"""
Hybrid optimisation for the self‑convolution bound.

The algorithm works in three stages on a probability vector `a`
(normalised to sum 1, all entries ≥0).

Stage 1 – soft‑max mirror descent.
    * A smooth approximation of the max‑norm is built by a soft‑max
      with a temperature β that is linearly increased up to a very
      large value.
    * The (entropic) gradient is obtained by a second FFT convolution.
    * An exponentiated‑gradient (EG) update with a linearly decreasing
      step‑size η is applied, followed by a projection onto the simplex.
    * Small block‑wise multiplicative jitter is added occasionally to
      keep diversity.

Stage 2 – hard‑max EG descent.
    * The exact sub‑gradient of `max(a*a)` is `g_i = 2·a_{k−i}` where
      `k = argmax(a*a)`.  No extra FFT is needed.
    * The same EG update as in stage 1 is performed with a finer η‑schedule.

Stage 3 – stochastic local refinement.
    * The current peak index `k` is located.
    * A donor index is sampled proportionally to its contribution
      `a_i·a_{k−i}` to that peak.
    * A random recipient receives a small fraction (≤ 30 %) of the donor.
    * Moves are accepted with a Metropolis rule using a linear cooling
      schedule; occasional block perturbations are also performed.
    * If no improvement is observed for a while, a soft restart from the
      best solution plus multiplicative noise is triggered.

All phases share the same bookkeeping: the best vector seen so far,
its score (computed as `2·n·max(conv)`, which equals the true
`evaluate_sequence` value when the vector is normalised) and a timestamp
of the last improvement.  At the end a trailing/leading zero tail is
trimmed (if any) and the best vector is returned as a plain Python list.

The routine respects a 5‑second safety margin and stops no later than the
provided time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT.  Returns length 2·len(a)-1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# ----------------------------------------------------------------------

    (with a 5‑second safety margin).
    """

    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initialise on the simplex
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():

        a = np.clip(cand, 0.0, None)

        a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
    else:
        # start from a Dirichlet draw of moderate size
        n0 = 3800
        a = rng.dirichlet(np.ones(n0)).astype(np.float64)

    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 2) Score helper (exact for normalised vectors)
    # --------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        conv = _self_conv(vec, fft_len_self)
        return float(2.0 * n * conv.max())

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 3) Phase division (percent of available time)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.55 * total_time   # soft‑max mirror descent
    phase2_end = start_time + 0.85 * total_time   # hard‑max EG descent
    # phase‑3 occupies the remaining time

    # --------------------------------------------------------------
    # 4) Parameters
    # --------------------------------------------------------------
    # Phase 1 (soft‑max)
    eta_start_1 = 8.0
    eta_end_1 = 0.02
    beta_start_1 = 0.5
    beta_end_1 = 15000.0          # larger final temperature → almost hard‑max
    momentum = 0.90
    grad_prev = np.zeros_like(a)
    jitter_prob_1 = 0.02

    # Phase 2 (hard‑max EG)
    eta_start_2 = 1.0
    eta_end_2 = 1e-5
    jitter_prob_2 = 0.015

    # Phase 3 (stochastic refinement)
    temp_start = 0.08
    temp_end = 1e-5
    restart_interval = 45.0       # seconds without improvement before a soft restart

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        # ----------------------------------------------------------
        # Phase 1 – soft‑max mirror descent
        # ----------------------------------------------------------
        if now < phase1_end:
            # linear schedules
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start_1 * (1.0 - frac) + eta_end_1 * frac
            beta = beta_start_1 + (beta_end_1 - beta_start_1) * frac

            # self‑convolution
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()

            # soft‑max distribution over convolution entries
            w = np.exp(beta * (conv - max_c))          # stable: subtract max
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            # gradient = 2·(p * rev_a)  (FFT for speed)
            rev_a = a[::-1]
            # length of convolution p*rev_a is len(p)+len(rev_a)-1 = 3n-2
            fft_len_grad = _next_pow2(len(p) + n - 1)
            fp = np.fft.rfft(p, n=fft_len_grad)
            fra = np.fft.rfft(rev_a, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient update
            factor = np.exp(-eta * grad)
            a *= factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / n)

            # occasional block jitter
            if rng.random() < jitter_prob_1:
                a = _random_block_perturb(a, rng)

            # evaluate exact score
            cur_score = _score(a)

        # ----------------------------------------------------------
        # Phase 2 – hard‑max EG descent (exact sub‑gradient)
        # ----------------------------------------------------------
        elif now < phase2_end:
            # linear schedule for η
            frac = (now - phase1_end) / (phase2_end - phase1_end)
            eta = eta_start_2 * (1.0 - frac) + eta_end_2 * frac

            # current convolution and peak
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()
            k = int(conv.argmax())

            # exact sub‑gradient of max‑norm:
            # g_i = 2·a_{k−i} for i where the index is valid, else 0.
            g = np.zeros_like(a)
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            rev_idx = k - idx
            g[idx] = 2.0 * a[rev_idx]

            # EG update
            factor = np.exp(-eta * g)
            a *= factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / n)

            # jitter (slightly less frequent)
            if rng.random() < jitter_prob_2:
                a = _random_block_perturb(a, rng)

            cur_score = _score(a)

        # ----------------------------------------------------------
        # Phase 3 – stochastic local refinement (simulated annealing)
        # ----------------------------------------------------------
        else:
            # cooling schedule for the Metropolis temperature
            t_frac = (now - phase2_end) / (deadline - phase2_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            conv = _self_conv(a, fft_len_self)
            cur_score = _score(a)
            k = int(conv.argmax())

            # ----- weighted donor selection (mass‑transfer) -----
            # contribution of each index to the current peak
            contrib = np.zeros_like(a)
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            if low <= high:
                idx = np.arange(low, high + 1, dtype=int)
                opp = k - idx
                contrib[idx] = a[idx] * a[opp]

            total_contrib = contrib.sum()
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            # occasional global block perturbation
            if rng.random() < 0.007:
                a = _random_block_perturb(a, rng)
                continue

            # recipient (any other index)
            recipient = rng.integers(0, n - 1)
            if recipient >= donor:
                recipient += 1

            # transfer a fraction of the donor's mass
            if a[donor] > 1e-14:
                max_delta = 0.30 * a[donor]
                delta = rng.random() * max_delta

                a_new = a.copy()
                a_new[donor] -= delta
                a_new[recipient] += delta

                # compute new score
                conv_new = _self_conv(a_new, fft_len_self)
                new_score = float(2.0 * n * conv_new.max())

                # Metropolis accept/reject
                if new_score < cur_score or rng.random() < np.exp(
                    -(new_score - cur_score) / temperature
                ):
                    a = a_new
                    cur_score = new_score

        # ----------------------------------------------------------
        # Keep the best solution seen so far
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Soft restart after stagnation
        # ----------------------------------------------------------
        if now - last_improve > restart_interval:
            # return to the current best and add moderate noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final trimming of zero tails (if any) – this reduces n without
    #    increasing the max‑convolution value.
    # ------------------------------------------------------------------
    eps_trim = 1e-13
    nz = np.where(best > eps_trim)[0]
    if nz.size > 0:
        trimmed = best[nz[0] : nz[-1] + 1]
    else:
        trimmed = best.copy()

    # recompute the exact score of the trimmed vector (for safety)
    final_score = _score(trimmed)
    if final_score < best_score:
        best = trimmed
        best_score = final_score

    # ------------------------------------------------------------------
    # 7) Return the best vector (clipped to the limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
