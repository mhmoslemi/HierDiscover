"""
Differential‑Evolution (DE) optimiser for the step‑function bound.

The objective is the exact value returned by the supplied
``evaluate_sequence`` – lower is better.  The DE algorithm works on a
population of candidate vectors, creates *mutant* vectors by mixing three
random individuals, applies a binomial crossover and always keeps the
better of the trial and the parent.  The population is re‑initialised
periodically if it stagnates for a while.  This strategy is orthogonal
to the simple block‑perturbation hill‑climber used previously and
typically discovers improvements on the benchmark
(≈ 1.514) within the 1000 s budget.

All helper functions are top‑level and contain no closures, complying
with the execution constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _clip_and_check(seq: np.ndarray) -> np.ndarray:
    """
    Clip a candidate to the limits required by ``evaluate_sequence``:
    * non‑negative
    * at most 1000 per entry.
    The function returns a copy that can safely be passed to the
    evaluator.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    # ``evaluate_sequence`` treats empty or near‑zero sums as inf,
    # guarantee a tiny positive total.
    if seq.sum() < 1e-8:
        seq[0] = 1e-4
    return seq


def _initial_population(base: np.ndarray,
                        popsize: int,
                        rng: np.random.Generator) -> list[np.ndarray]:
    """
    Build the initial DE population.
    * ``base``   – a high‑quality seed (e.g. the global ``height_sequence_1``
                  if present); it is inserted as the first individual.
    * ``popsize``– number of individuals (≥ 4).
    * ``rng``    – random generator.
    The remaining individuals are sampled from a Dirichlet distribution
    (concentration 0.5 → sparse shapes) and finally clipped.
    """
    n = base.size
    population = [base.copy()]                       # keep the seed
    # Dirichlet draws give a non‑negative vector that automatically sums
    # to a positive number; we simply scale it to the same total as ``base``.
    base_sum = base.sum()
    for _ in range(popsize - 1):
        # concentration <1 favours extreme, possibly better, shapes
        weights = rng.dirichlet([0.5] * n)
        cand = weights * base_sum
        population.append(_clip_and_check(cand))
    return population


def _de_score(candidate: np.ndarray) -> float:
    """
    Wrapper around the provided ``evaluate_sequence`` that expects a
    Python list.  The conversion is cheap compared with the convolution.
    """
    # The global ``evaluate_sequence`` is assumed to be already imported.
    return evaluate_sequence(candidate.tolist())  # type: ignore[name-defined]


def _de_step(population: list[np.ndarray],
             scores: np.ndarray,
             F: float,
             CR: float,
             rng: np.random.Generator) -> tuple[list[np.ndarray], np.ndarray, bool]:
    """
    Perform one DE generation over the whole population.
    Returns the new population, new scores and a flag indicating whether
    at least one individual improved.
    """
    popsize = len(population)
    n = population[0].size
    improved = False

    for i in range(popsize):
        # --- mutation: select three distinct donors different from i ----------
        idxs = rng.choice([j for j in range(popsize) if j != i], size=3, replace=False)
        a, b, c = [population[j] for j in idxs]

        mutant = a + F * (b - c)

        # --- crossover (binomial) -------------------------------------------
        cross_mask = rng.random(n) < CR
        # ensure at least one component from the mutant
        if not cross_mask.any():
            cross_mask[rng.integers(0, n)] = True
        trial = np.where(cross_mask, mutant, population[i])

        # --- enforce problem constraints --------------------------------------
        trial = _clip_and_check(trial)

        # --- selection ---------------------------------------------------------
        trial_score = _de_score(trial)

        if trial_score < scores[i]:
            population[i] = trial
            scores[i] = trial_score
            improved = True

    return population, scores, improved


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Find a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence`` within the given time budget.

    The optimiser uses a Differential‑Evolution strategy with
    occasional restarts.  If a global variable ``height_sequence_1`` is
    present it is used as a high‑quality seed; otherwise a uniform
    sequence of length 1024 (scaled to the same L1‑norm as the seed)
    starts the search.

    The function returns a plain Python list ready for the external
    evaluation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ------------------------------------------------------------------
    # 1) Determine problem dimension and create a sensible initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
                raise ValueError
        except Exception:
            base = None
    else:
        base = None

    if base is None:
        n = 1024
        # initialise with the “standard” normalisation used in the reference paper:
        # sum_j a_j = sqrt(2 n)
        base_sum = np.sqrt(2.0 * n)
        base = np.full(n, base_sum / n, dtype=np.float64)
    else:
        n = base.size
        base = _clip_and_check(base)

    # ------------------------------------------------------------------
    # 2) DE parameters (chosen for robustness on a ~1 k‑dim problem)
    # ------------------------------------------------------------------
    popsize = max(20, 4 * n // 256)          # at least 20 individuals
    F = 0.8                                   # differential weight
    CR = 0.9                                   # crossover probability
    restart_gap = 45.0                        # seconds without improvement -> restart

    # ------------------------------------------------------------------
    # 3) Initialise population and evaluate
    # ------------------------------------------------------------------
    population = _initial_population(base, popsize, rng)
    scores = np.array([_de_score(ind) for ind in population])
    best_idx = int(scores.argmin())
    best = population[best_idx].copy()
    best_score = float(scores[best_idx])

    start_time = time.monotonic()
    last_improve = start_time

    print(f"[0.00s] initial best score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 4) Main DE loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        population, scores, any_improved = _de_step(population, scores, F, CR, rng)

        # keep track of the overall best individual
        cur_best_idx = int(scores.argmin())
        cur_best_score = float(scores[cur_best_idx])
        if cur_best_score < best_score:
            best_score = cur_best_score
            best = population[cur_best_idx].copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------------
        # Restart logic – if stagnant for a while, re‑initialise the pop.
        # ----------------------------------------------------------------
        if time.monotonic() - last_improve > restart_gap:
            # keep the current best as the seed for the new population
            population = _initial_population(best, popsize, rng)
            scores = np.array([_de_score(ind) for ind in population])
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

        # ----------------------------------------------------------------
        # Adaptive tweak of F (optional simple scheme)
        # ----------------------------------------------------------------
        # slowly decay the mutation factor to focus the search later on.
        # This does not hurt early exploration but helps fine‑tuning.
        if time.monotonic() - start_time > budget_s * 0.6:
            F = max(0.3, F * 0.995)

    # ------------------------------------------------------------------
    # 5) Return the best candidate (final clipping for safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
