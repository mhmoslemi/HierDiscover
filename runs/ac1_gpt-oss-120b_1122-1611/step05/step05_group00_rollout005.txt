    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length n (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

    """Linear convolution a*a using an FFT of length `fft_len`.  Returns an
    array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional random block perturbation (adds diversity) -----
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            delta = rng.normal(0.0, 0.3)   # moderate log‑scale shift
            y[start:start + width] += delta

        # ----- soft restart if we have been stagnant too long -----
        if now - last_improve > restart_gap:
            # jitter the best found solution and continue from there
            y = np.log(best_a + 1e-24) + rng.normal(0.0, 0.12, size=n)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clipping respects evaluate_sequence)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
