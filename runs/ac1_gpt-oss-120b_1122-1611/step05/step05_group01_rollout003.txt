"""
propose_candidate
=================

Search for a non‑negative step function that minimises the supplied
``evaluate_sequence`` score.

The algorithm proceeds as follows:

1. **Initial families** – For each chosen length `n` we generate a few
   symmetric “arcsine‑type’’ candidates
   ```
   a_i ∝ 1 / ((i+½)(n‑½‑i))**α
   ```
   with exponents α≈0.5. The best of these candidates (as measured by
   ``evaluate_sequence``) is taken as a starting point.

2. **Smooth max objective** – The true objective is
   `2·n·max(b)/sum(a)²` with `b = a∗a`.  Instead of the non‑smooth max we
   optimise the smooth approximation
   ```
   Mₗ(a) = (1/λ)·log Σ_k exp(λ·(a∗a)_k)    (λ>0)
   ```
   whose gradient is easy to evaluate via FFTs.  As the optimisation
   proceeds λ is slowly increased, turning the smooth approximation into
   the true max.

3. **Adam on the simplex** – With the smooth objective we compute a
   (sub‑)gradient, then perform an Adam step.  After each step we
   * enforce non‑negativity,
   * optionally symmetrise the vector (`a[i]=a[n‑1‑i]`),
   * renormalise to keep `sum(a)=1` (the score is scale‑invariant).

4. **Time‑bounded search** – The outer loop iterates over several
   lengths (`1024, 2048, 4096, 8192`).  The inner Adam loop runs until the
   global time budget (≈ 1000 s) is exhausted, periodically evaluating
   the candidate with the true ``evaluate_sequence`` function and keeping
   the best one found.

The implementation uses only NumPy and the provided ``evaluate_sequence``
function, respects the “no‑IO’’ rule, and contains no closures.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real vector via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _correlate_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear correlation (w ⋆ a) via FFT.  Returns full vector of length
    len(w)+len(a)-1."""
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a, n=fft_len)
    corr = np.fft.irfft(fw * np.conj(fa), n=fft_len)
    return corr[: w.size + a.size - 1]


def _grad_and_objective(a: np.ndarray,
                        conv: np.ndarray,
                        lam: float,
                        fft_len_corr: int):
    """
    Returns (gradient, objective) for the smooth‑max objective.
    *a* – current vector (non‑negative, sum=1)
    *conv* – self‑convolution of *a* (length 2·n‑1)
    *lam* – smoothing parameter λ
    *fft_len_corr* – FFT length for the correlation step
    """
    n = a.size
    # ---- smooth max (log‑sum‑exp) ----
    lam_conv = lam * conv
    maxc = np.max(lam_conv)               # stability shift
    w = np.exp(lam_conv - maxc)           # unnormalised softmax weights
    w_sum = np.sum(w)
    w /= w_sum                           # softmax weights, sum to 1

    # ---- correlation w ⋆ a ----
    corr_full = _correlate_fft(w, a, fft_len_corr)   # length 3·n‑2
    # Gradient of Σ w_k·(a∗a)_k w.r.t. a_i is 2·(w⋆a)[n‑1‑i]
    grad_corr = 2.0 * corr_full[n - 1 :: -1]   # size n, reversed slice

    sum_a = np.sum(a)                 # = 1 after normalisation, kept for clarity
    # log‑sum‑exp value (smooth max)
    log_sum = maxc + np.log(w_sum)    # = log Σ exp(λ·conv)
    M_soft = log_sum / lam

    # ---- objective (exact same scaling as evaluate_sequence) ----
    obj = 2.0 * n * M_soft / (sum_a ** 2)

    # ---- gradient of the objective (exact) ----
    factor = 4.0 * n / (sum_a ** 2)
    grad = factor * grad_corr - (4.0 * n * M_soft / (sum_a ** 3)) * a
    return grad, obj

                best_score = sc
                best_seq = cand
                print(f"[{time.monotonic() - start:.2f}s] baseline score = "
                      f"{sc:.6f} (len {len(cand)})")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1)  Search over a few lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]            # reasonable sizes for the budget
    alphas = [0.45, 0.5, 0.55]                    # exponent family around arcsine

    for n in lengths:
        if time.monotonic() > deadline:
            break

        # ---- pick the best deterministic arcsine‑family start for this n ----
        best_start = None
        best_start_score = float("inf")
        i = np.arange(n, dtype=np.float64)

        for alpha in alphas:
            raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
            raw = np.clip(raw, 0.0, 1000.0)
            raw /= raw.sum()                     # normalise (sum = 1)
            seq = raw.tolist()
            sc = evaluate_sequence(seq)
            if np.isfinite(sc) and sc < best_start_score:
                best_start_score = sc
                best_start = raw.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] arcsine seed n={n}, α={alpha} → "
                      f"{sc:.6f}")

        # If for some reason none succeeded, fall back to uniform
        if best_start is None:
            best_start = np.full(n, 1.0 / n)

        # Add a tiny random jitter to break perfect symmetry before optimisation
        jitter = np.exp(rng.normal(0.0, 0.02, size=n))
        a = best_start * jitter
        a = np.maximum(a, 0.0)
        a /= a.sum()                     # keep sum = 1

        # --------------------------------------------------------------
        # 2)  Adam optimisation on the smooth‑max objective
        # --------------------------------------------------------------
        # FFT lengths for convolution and correlation
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n - 2)

        # Adam hyper‑parameters
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.05                         # base learning‑rate
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # λ schedule (soft‑max smoothness)
        lam = 5.0
        lam_max = 60.0

        t = 1
        while time.monotonic() < deadline:
            # self‑convolution (full linear)
            conv = _conv_self_fft(a, fft_len_conv)

            # gradient and current (smooth) objective value
            grad, obj = _grad_and_objective(a, conv, lam, fft_len_corr)

            # Adam update (minimise obj)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step                    # descend

            # enforce domain constraints
            a = np.maximum(a, 0.0)
            # symmetrise (helps keep the convolution flat)
            a = (a + a[::-1]) * 0.5
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # grow λ slowly – we start with a smooth landscape and finish
            # with an almost exact max.
            lam = min(lam * 1.002, lam_max)

            # occasional true evaluation (every ~0.2 s on average)
            if t % 50 == 0:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:6.2f}s] new best score = {sc:.6f} "
                          f"(n={n})")

            # basic termination guard
            t += 1
            if t > 200_000:               # safeguard against endless loops
                break

    # ------------------------------------------------------------------
    # 3)  Fallback if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Clip to the limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(len {len(best_seq)})")
    return best_seq
