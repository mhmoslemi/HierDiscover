def _targeted_move(a, conv, rng, max_delta_frac=0.3):
    k = np.argmax(conv)
    # compute contributions
    # compute valid indices
    # use vectorized: i array 0..n-1; j = k - i; valid = (j>=0)&(j<n)
    # contributions = a[i] * a[j] where valid else 0.
    a_comp = np.zeros_like(a)
    j = k - np.arange(n)
    valid = (j >= 0) & (j < n)
    a_comp[valid] = a[valid] * a[j[valid]]
    # Weighted donor selection
    total_contrib = a_comp.sum()
    if total_contrib <= 0:
        i = rng.integers(0, n)
    else:
        # Choose i with probability proportional to a_comp
        probs = a_comp / total_contrib
        # use rng.choice with probabilities
        i = rng.choice(np.arange(n), p=probs)
    # Choose recipient j: maybe low contribution:
    # Choose j from indices where contribution is minimal.
    # compute complement contributions for index i (maybe different?). For simplicity random j with low a_comp.
    # Let's compute probability inverse.
    inv_contrib = (total_contrib - a_comp) + 1e-12  # avoid zero
    probs_j = inv_contrib / inv_contrib.sum()
    j = rng.choice(np.arange(n), p=probs_j)
    # Ensure i != j
    if i == j:
        # pick random other
        candidates = np.delete(np.arange(n), i)
        j = rng.choice(candidates)
    # compute delta
    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0:
        return a, conv, None
    delta = rng.random() * max_delta
    # apply transfer
    a2, conv2 = _transfer_move(a, conv, i, j, delta)
    # compute new max
    new_score = 2.0 * n * conv2.max()
    cur_score = 2.0 * n * conv.max()
    if new_score < cur_score - 1e-15:
        return a2, conv2, new_score
    else:

"""
Propose a step‑function (non‑negative list) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – start from the best known construction
   (`height_sequence_1` if present) and normalise it to a probability
   vector (sum = 1).  If the variable does not exist a uniform vector of
   length 4000 is used.

2. **Exact self‑convolution** – computed once by FFT and thereafter
   updated in O(n) when a transfer of mass between two coordinates is
   performed (the update formula is derived from the definition of a
   linear convolution).

3. **Targeted simulated‑annealing moves** – at each iteration the
   index `k*` where the current self‑convolution attains its maximum is
   identified.  A *donor* coordinate `i` is chosen with probability
   proportional to its contribution `a_i·a_{k*-i}` to this peak, while a
   *receiver* coordinate `j` is chosen with probability proportional to
   the *lack* of contribution (`total‑contribution – contribution`).
   A small amount `δ` of mass is transferred from `i` to `j`.  The new
   convolution and the resulting objective value are obtained with the
   O(n) update, so millions of proposals can be examined per second.

4. **Metropolis acceptance** – improving moves are always accepted.
   Deteriorating moves are accepted with probability
   `exp(-(Δscore)/T)`, where the temperature `T` follows a linear cooling
   schedule from `0.5·current_score` down to `1e‑6·current_score`.

5. **Periodic re‑normalisation** – after each transfer the vector is
   projected onto the probability simplex.  Accumulated numerical drift
   is corrected every few thousand iterations by recomputing the full
   convolution with FFT.

6. **Diversification** – with a tiny probability a random contiguous
   block is multiplicatively perturbed (log‑normal factor) and the
   convolution is recomputed from scratch.  This helps escape plateaus.

The whole search runs for the allotted 1000 s (a 0.5 s safety margin)
and returns the best sequence found.

The method keeps all numbers non‑negative, respects the limits of
``evaluate_sequence`` and makes heavy use of NumPy’s vectorised
operations for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negatives caused by floating‑point noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho_arr) == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _transfer_move(a: np.ndarray, conv: np.ndarray,
                   i: int, j: int, delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Perform an exact O(n) update of the convolution when an amount
    ``delta`` of mass is moved from index ``i`` to index ``j``.
    Returns the updated coefficient vector and the updated convolution.
    """
    n = a.size
    # Update the convolution (linear part)
    new_conv = conv.copy()
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a

    # Exact quadratic corrections
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    # Update the coefficient vector
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)
    return a2, new_conv


def _targeted_pairwise_move(a: np.ndarray, conv: np.ndarray,
                            rng: np.random.Generator,
                            max_delta_frac: float = 0.4) -> tuple[np.ndarray,
                                                                 np.ndarray,
                                                                 float]:
    """
    Generate a *targeted* pairwise transfer that is biased to reduce the
    current maximum entry of the self‑convolution.
    Returns (new_a, new_conv, new_score).  ``new_score`` is the objective
    value after the move (2·n·max(conv)).
    """
    n = a.size
    # index of the current peak
    k = int(np.argmax(conv))

    # Contributions of each coordinate to the peak
    idx = np.arange(n)
    partner = k - idx
    valid = (partner >= 0) & (partner < n)
    contrib = np.zeros_like(a)
    contrib[valid] = a[idx[valid]] * a[partner[valid]]

    total_contrib = contrib.sum()
    # ----- choose donor (large contribution) -----
    if total_contrib > 0 and rng.random() < 0.9:
        # weighted draw without using rng.choice (faster)
        cum = np.cumsum(contrib)
        r = rng.random() * total_contrib
        i = int(np.searchsorted(cum, r, side='right'))
    else:
        i = rng.integers(0, n)

    # ----- choose receiver (small contribution) -----
    inv_weights = (total_contrib - contrib) + 1e-12
    cum_inv = np.cumsum(inv_weights)
    r2 = rng.random() * cum_inv[-1]
    j = int(np.searchsorted(cum_inv, r2, side='right'))

    if i == j:
        # pick a different index uniformly
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

    # ----- amount of mass to transfer -----
    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        # nothing to move – return the unchanged state
        return a, conv, 2.0 * n * conv.max()
    delta = rng.random() * max_delta

    a2, conv2 = _transfer_move(a, conv, i, j, delta)
    new_score = 2.0 * n * conv2.max()
    return a2, conv2, new_score

    Returns the best coefficient list found.
    """

    if "height_sequence_1" in globals():

            s = a.sum()

        else:
            n = 4000

    temp_end = cur_score * 1e-6     # near‑zero final temperature
