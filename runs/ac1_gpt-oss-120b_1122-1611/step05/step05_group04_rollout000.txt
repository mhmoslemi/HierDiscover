"""
Propose a step‑function (non‑negative list) that minimises
evaluate_sequence().

Algorithm
---------

1. **Initialisation** – use the provided `height_sequence_1` if it exists,
   otherwise start from the uniform distribution on *n* points
   (default n = 4000).  All vectors are normalised to sum = 1 because the
   objective is homogeneous.

2. **Phase A – Sub‑gradient descent on the simplex**  
   * At each iteration we compute the self‑convolution `c = a*a` (FFT),
     locate the maximal entries and form the exact sub‑gradient of the
     `max`‑norm (weights are equally split among the active indices).  
   * An exponentiated‑gradient (mirror‑descent) update
     `a ← a·exp(−η·grad)` is performed, followed by Euclidean projection
     onto the probability simplex.  
   * Step size η is linearly annealed from 1.0 to 0.01.
   * Very cheap random block perturbations are injected occasionally to
     escape plateaus.

   Multiple short runs are performed (each started from the current best
   sequence plus a small jitter) until roughly 70 % of the time budget is
   exhausted.

3. **Phase B – Simulated‑annealing refinement**  
   * From the best vector found in Phase A we run a Metropolis‑style SA,
     repeatedly moving a random amount of mass from a donor index *i* to a
     receiver index *j*.  The convolution is recomputed with FFT after each
     move (still cheap for n≈4000).  
   * Moves that improve the score are always accepted; worsening moves are
     accepted with probability `exp(−Δ/T)`, where the temperature T is
     linearly cooled from 5 % of the current score down to 1e‑6 of it.  
   * Sporadic block perturbations are also attempted.

4. **Result** – the best sequence seen so far (clipped to the limits of
   `evaluate_sequence`) is returned as a plain Python list of floats.

The whole routine respects the 1000 s budget, uses only numpy (no external
solvers), and works without closures or lambda expressions.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negative values caused by floating‑point noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Derivation: grad_j = 2·Σ_i a_i·w_{i+j}.
    Implemented as a correlation a ⋆ w using a single FFT.
    """
    n = a.size
    Lw = w.size                     # = 2*n‑1
    wr = w[::-1]                    # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)   # length for convolution a ⋆ wr

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)         # mapping to obtain correlation result
    grad = 2.0 * conv_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to counter floating‑point drift.
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.25))   # slightly larger spread than before

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)

    return _project_onto_simplex(out)


def _pairwise_move(a: np.ndarray,
                   conv: np.ndarray,
                   rng: np.random.Generator,
                   max_delta_frac: float = 0.3,
                   fft_len: int = None) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Propose a random transfer of mass from index i to index j.
    Returns (new_a, new_conv, new_score) where new_score = 2*n*max(new_conv).
    The move is always performed (the caller decides whether to accept it).
    """
    n = a.size
    # pick distinct i and j
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1

    # amount to move – a fraction of the donor's mass
    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        # nothing can be moved – keep everything unchanged
        score = 2.0 * n * conv.max()
        return a, conv, score

    delta = rng.random() * max_delta

    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)   # maintain exact sum = 1

    # recompute convolution (FFT – still cheap for n≈4000)
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    conv2 = _self_conv_fft(a2, fft_len)
    new_score = 2.0 * n * conv2.max()
    return a2, conv2, new_score

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimizes
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best coefficient list found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size and initial vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a0 = np.clip(cand0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = 4000
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    best = a0.copy()
    conv_best = _self_conv_fft(best, fft_len_self)
    best_score = 2.0 * n * conv_best.max()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # a safety margin
    last_improve = start_time

    # --------------------------------------------------------------
    # 3) Hyper‑parameters
    # --------------------------------------------------------------
    # Phase split: 70 % time for sub‑gradient descent, 30 % for SA.
    phase_split = 0.70
    phase_split_time = start_time + phase_split * (deadline - start_time)

    # Sub‑gradient descent schedule
    eta_start = 1.0
    eta_end = 0.01
    block_perturb_prob = 0.02        # probability per iteration

    # Simulated‑annealing schedule
    # temperature will be a fraction of the current score
    temp_start_frac = 0.05
    temp_end_frac = 1e-6
    max_delta_frac = 0.4             # donor may give up to 40 % of its mass

    # --------------------------------------------------------------
    # 4) Phase A – Sub‑gradient descent (multiple short runs)
    # --------------------------------------------------------------
    while time.monotonic() < phase_split_time:
        # start from the current best with a small jitter
        a = best.copy()
        jitter = np.exp(rng.normal(0.0, 0.07, size=n))
        a = a * jitter
        a = np.clip(a, 0.0, None)
        a = _project_onto_simplex(a)

        # number of inner iterations – bounded both by time and by a hard cap
        max_inner = 2500
        for it in range(max_inner):
            now = time.monotonic()
            if now >= phase_split_time:
                break

            # current convolution and its maximum
            conv = _self_conv_fft(a, fft_len_self)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # identify active indices (within a tiny tolerance)
            eps = 1e-12
            active = (conv >= cur_max - eps)
            count_active = np.count_nonzero(active)
            if count_active == 0:
                w = np.zeros_like(conv)
            else:
                w = active.astype(float) / float(count_active)

            # sub‑gradient of the max‑norm
            grad = _gradient_from_weights(a, w)

            # linear schedule for the learning rate
            elapsed = now - start_time
            total = phase_split_time - start_time
            frac = elapsed / total if total > 0 else 1.0
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # exponentiated‑gradient (mirror descent) update
            a = a * np.exp(-eta * grad)

            # keep numerics stable
            a = np.clip(a, 1e-15, None)
            a = _project_onto_simplex(a)

            # occasional random block perturbation
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # update global best if we improved
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best (grad) = {best_score:.6f}")

        # after each run we optionally do a larger reset if we have been stuck
        if now - last_improve > 90.0:
            # Apply a stronger jitter to escape a plateau
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            best = best * noise
            best = _project_onto_simplex(best)
            conv_best = _self_conv_fft(best, fft_len_self)
            best_score = 2.0 * n * conv_best.max()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] jitter reset – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Phase B – Simulated‑annealing refinement
    # --------------------------------------------------------------
    # Start from the best vector found so far.
    a = best.copy()
    conv = _self_conv_fft(a, fft_len_self)
    cur_score = 2.0 * n * conv.max()

    # temperature schedule (linear cooling)
    temp_start = cur_score * temp_start_frac
    temp_end = cur_score * temp_end_frac
    sa_start = time.monotonic()
    sa_deadline = deadline

    while time.monotonic() < sa_deadline:
        now = time.monotonic()
        elapsed = now - sa_start
        total = sa_deadline - sa_start
        frac = elapsed / total if total > 0 else 1.0
        # Linear interpolation from start to end temperature
        T = temp_start * (1.0 - frac) + temp_end * frac

        # propose a pairwise mass transfer
        a_new, conv_new, new_score = _pairwise_move(
            a, conv, rng,
            max_delta_frac=max_delta_frac,
            fft_len=fft_len_self)

        # Metropolis acceptance
        if new_score <= cur_score:
            accept = True
        else:
            # probability exp(−Δ/T)
            delta = new_score - cur_score
            accept = rng.random() < math.exp(-delta / max(T, 1e-14))

        if accept:
            a, conv, cur_score = a_new, conv_new, new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best (SA) = {best_score:.6f}")

        # occasional block perturbation during SA
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)
            conv = _self_conv_fft(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best (perturb) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return best sequence (clipped to limits of evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
