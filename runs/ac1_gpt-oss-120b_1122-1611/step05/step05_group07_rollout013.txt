Algorithm
---------
* Initialise a probability vector `a` (sum = 1).  If a previous best

  performed.  The step size decays linearly.

  minimise the true objective `max(c)`.  The sub‑gradient at the
  current peak index `k` is computed analytically and a projected
  additive step is taken with a decreasing step size.

* **Phase 3 – stochastic refinement** (≈ 30 % of the time)  
  perform the same pairwise mass‑transfer moves as in the reference
  code, with a slowly cooling Metropolis temperature.

* The best vector seen so far (lowest value of `evaluate_sequence`) is
  tracked and returned at the end.  Restarts are triggered after 20 s

All operations are vectorised and the self‑convolution is computed by
FFT for speed.

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length >= 2·len(a)-1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # implementation of the algorithm from
    # https://arxiv.org/abs/1309.1541 (Duchi et al.)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    return w


def _grad_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a*a) at the peak index k."""
    n = a.size
    g = np.zeros_like(a)
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    i = np.arange(low, high + 1, dtype=np.int64)
    j = k - i
    g[i] = a[j]
    # double the contribution for i!=j
    mask = i != j
    g[i[mask]] *= 2.0
    return g


def _grad_l2(a: np.ndarray, conv: np.ndarray, avg: float, fft_len_grad: int) -> np.ndarray:
    """Gradient of sum_k (conv_k - avg)^2 w.r.t. a."""
    e = conv - avg
    rev_a = a[::-1]
    fe = np.fft.rfft(e, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fe * fra, n=fft_len_grad)[: 3 * a.size - 1]
    # extract the slice that corresponds to indices 0..n-1
    g = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return g


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

    phase2_end = start_time + 0.7 * total_time      # max‑descent

        # Phase 1 – L2 flattening
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedule for step size
            frac = (now - start_time) / (phase1_end - start_time)
            eta = 0.8 * (1.0 - frac) + 0.02 * frac   # from 0.8 down to 0.02

            conv = _self_conv(a, fft_len_self)
            avg = 1.0 / (2 * n - 1)                 # because sum(a)=1 ⇒ Σconv =1
            grad = _grad_l2(a, conv, avg, fft_len_grad)

            a_new = a - eta * grad
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s > 0.0:
                a_new /= s
            else:
                a_new = np.full_like(a, 1.0 / n)

            a = a_new
            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()

        # Phase 2 – sub‑gradient descent on max(conv)

            cur_score = 2.0 * n * conv.max()
            k = int(conv.argmax())

            grad = _grad_max(a, k)

            # cap the step so that we never go negative
            pos = grad > 1e-12
            if np.any(pos):
                max_eta = np.min(a[pos] / grad[pos])
                eta = min(eta, 0.9 * max_eta)
            else:
                eta = 0.0

            a_new = a - eta * grad
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s > 0.0:
                a_new /= s
            else:
                a_new = np.full_like(a, 1.0 / n)

            # accept if it does not increase the objective; otherwise keep old a
            new_score = 2.0 * n * _self_conv(a_new, fft_len_self).max()
            if new_score <= cur_score:
                a = a_new
                cur_score = new_score
            # otherwise we simply stay with the current a (no Metropolis in this phase)

        # --------------------------------------------------------------
        # Phase 3 – stochastic local refinement (simulated annealing)
        # --------------------------------------------------------------
        else:
            # linear temperature cooling
            t_frac = (now - phase2_end) / (deadline - phase2_end)
            temperature = 0.06 * (1.0 - t_frac) + 1e-6 * t_frac

            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            k = int(conv.argmax())

            # compute contribution of each index to the current peak
            contrib = np.zeros_like(a)
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            i = np.arange(low, high + 1, dtype=np.int64)
            j = k - i
            contrib[i] = a[i] * a[j] * (2.0 - (i == j).astype(float))

            total = contrib.sum()
            if total > 0:
                prob = contrib / total
                i = rng.choice(np.arange(n), p=prob)
            else:
                i = rng.integers(0, n)

            if a[i] > 1e-12:

                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_delta = 0.25 * a[i]                     # move at most 25 %

                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = np.clip(a_new, 0.0, None)
                a_new /= a_new.sum()

                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                    cur_score = new_score

            if rng.random() < 0.03:

        # keep the best solution seen so far
        # --------------------------------------------------------------

            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # restart after long stagnation
        # --------------------------------------------------------------

            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise

            a /= a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # 5) return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

3. **Phase 2 – max‑descent** (≈ 30 % of the time)

4. **Phase 3 – stochastic refinement** (≈ 30 % of the time)

5. Throughout the run the best vector (lowest value of `evaluate_sequence`) is

All heavy work (self‑convolution) is performed by FFT, and every update is
followed by a projection onto the probability simplex.
The function returns the best sequence found before the prescribed time budget
expires.

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    (Duchi et al., 2008 – https://arxiv.org/abs/1309.1541)
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0.0] = 0.0
    return w


def _grad_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a*a) at the peak index `k`."""
    n = a.size
    g = np.zeros_like(a)
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    i = np.arange(low, high + 1, dtype=np.int64)
    j = k - i
    g[i] = a[j]
    mask = i != j
    g[i[mask]] *= 2.0
    return g


def _grad_l2(a: np.ndarray, conv: np.ndarray, avg: float,
             fft_len_grad: int) -> np.ndarray:
    """Gradient of Σ (conv‑avg)² w.r.t. `a` (via FFT)."""
    e = conv - avg
    rev_a = a[::-1]
    fe = np.fft.rfft(e, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fe * fra, n=fft_len_grad)[: 3 * a.size - 1]
    g = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return g


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out
