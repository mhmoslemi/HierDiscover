"""
Propose an improved step‑function using a fast Adam optimiser on a
soft‑max surrogate of the L∞ convolution objective.

Key ideas
---------
*   Work on the probability simplex (∑a_i = 1) – the evaluate_sequence
    objective is homogeneous, so scaling does not matter.
*   Approximate  max_k (a∗a)_k  by a smooth log‑sum‑exp with an annealed
    temperature β.  The gradient of the surrogate is a simple weighted
    correlation which can be evaluated with two FFTs per iteration.
*   Optimise the surrogate with Adam (adaptive moments) and after every
    update project the vector back onto the simplex.
*   The optimiser runs many short “restarts’’ (random Dirichlet starts,
    jittered copies of the current best, or the previously discovered
    construction).  Each restart receives at most a few dozen seconds,
    so the whole search respects the 1000 s budget.
*   The best vector found (clipped to the limits imposed by evaluate_sequence)
    is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of Σ_k w_k (a∗a)_k  = 2·a ⋆ w  (correlation).
# w has length 2·n‑1.
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Return ∂/∂a Σ_k w_k (a∗a)_k ."""
    n = a.size
    L = w.size                      # = 2·n‑1
    # needed length for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x >= 0, sum(x) = 1}."""
    if v.ndim != 1:
        raise ValueError("only 1‑D arrays are supported")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimiser for a single start – Adam on the soft‑max surrogate.
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    time_budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Run Adam for at most ``time_budget`` seconds, starting from a0.
    Returns the best vector seen (on the simplex) and its exact score.
    """
    a = a0.copy()
    best = a.copy()
    # compute initial exact score
    conv = _conv_self(a, fft_len)
    best_score = 2.0 * n * np.max(conv)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Schedules (linear in iteration count)
    lr_start, lr_end = 0.12, 0.001
    beta_start, beta_end = 1.0, 4000.0
    max_iters = 200_000  # hard ceiling – the loop also respects the time limit
    start_time = time.monotonic()

    for t in range(1, max_iters + 1):
        if time.monotonic() - start_time > time_budget:
            break

        # ---- convolution & exact score ------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()

        # ---- schedule -----------------------------------------------------------------
        frac = t / max_iters
        lr = lr_start * (1.0 - frac) + lr_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ---- soft‑max weights ---------------------------------------------------------
        # use the full soft‑max (no active‑set pruning) – numerically stable
        w_raw = np.exp(beta * (conv - cur_max))  # max(c) → 0, others ≤ 0
        w = w_raw / w_raw.sum()

        # ---- gradient ---------------------------------------------------------------
        grad = _gradient_correlation(a, w)            # ∇_a Σ w_k (a∗a)_k
        # Adam update (we minimise ⇒ move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- simplex projection -------------------------------------------------------
        a = _project_simplex(a)

        # ---- occasional jitter to escape shallow plateaus ----------------------------
        if rng.random() < 0.003:                      # ~0.3 % chance per iteration
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = _project_simplex(a * jitter)

    return best, best_score

            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            start_vec = _project_simplex(start_vec * jitter)
        elif rng.random() < 0.2 and "height_sequence_1" in globals():
            # reuse the historic construction if available
            start_vec = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            start_vec = np.clip(start_vec, 0.0, None)
            s = start_vec.sum()
            if s > 0:
                start_vec = start_vec / s
                start_vec = _project_simplex(start_vec)
            else:
                start_vec = a_sqrt.copy()
        else:
            # random Dirichlet start – concentration between 0.3 and 2.0
            alpha = rng.uniform(0.3, 2.0)
            start_vec = rng.dirichlet(alpha * np.ones(n))

        # run the Adam optimiser from this start
        cand, cand_score = _adam_optimize(
            start_vec, n, fft_len, inner_budget, rng
        )

        # update global best
        if cand_score < best_overall_score - 1e-12:
            best_overall_score = cand_score
            best_overall = cand.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new global best = {best_overall_score:.6f}")

        # soft restart if we have been idle for a while
        if time.monotonic() - last_improve > 90.0:
            # wipe current best with a small jitter to escape possible flat plateau
            jitter = np.exp(rng.normal(0.0, 0.20, size=n))
            best_overall = _project_simplex(best_overall * jitter)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] soft restart – keep best score {best_overall_score:.6f}")

    # --------------------------------------------------------------
    # final post‑processing – enforce the evaluate_sequence bounds
    # --------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)

    return [float(v) for v in best_overall]
