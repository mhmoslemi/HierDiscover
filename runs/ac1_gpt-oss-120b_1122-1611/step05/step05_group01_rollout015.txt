"""
propose_candidate
=================

Hill‑climbing optimizer for the “flat self‑convolution” problem.

* Start from the best known sequence (`height_sequence_1` if it exists,
  otherwise a discrete arcsine distribution).
* Repeatedly move a small amount of probability mass from a point that
  contributes most to the current maximal convolution value to a point
  that contributes the least.
* The move is accepted only if the evaluation score improves.
* The step size (fraction of the source entry to move) is adapted
  dynamically; occasional random jitter helps escape plateaus.
* All work stops before the allotted budget (default 1000 s) and the best
  sequence found is returned.

The algorithm works directly on the convolution array, updating it in
O(n) per iteration (no FFT after the initial convolution), which allows
hundreds of thousands of iterations within the time limit and often
pushes the score below the target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be top level
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _make_arcsine(n: int, alpha: float = 0.5) -> np.ndarray:
    """Return a normalized discrete arcsine‑type vector of length ``n``.
    The shape is  a_i ∝ 1/((i+½)(n‑½‑i))**alpha ."""
    i = np.arange(n, dtype=np.float64)
    raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
    raw /= raw.sum()                 # normalise to sum = 1
    return raw


def _apply_move(a: np.ndarray,
                b: np.ndarray,
                i: int,
                j: int,
                delta: float,
                diff_buf: np.ndarray) -> np.ndarray:
    """
    Apply the elementary move “transfer δ from a[i] to a[j]”.
    Updates ``b`` in place (full linear self‑convolution) and returns the
    updated convolution array.  ``diff_buf`` is a pre‑allocated buffer of
    length ``len(b)`` that is cleared and reused each call.
    """
    n = a.shape[0]
    L = b.shape[0]

    # clear buffer
    diff_buf.fill(0.0)

    # contribution of the source index (subtraction)
    diff_buf[i:i + n] -= 2.0 * delta * a
    # contribution of the destination index (addition)
    diff_buf[j:j + n] += 2.0 * delta * a

    # in‑place update of the convolution
    b += diff_buf
    # update the probability vector
    a[i] -= delta
    a[j] += delta
    return b


def _select_indices(a: np.ndarray,
                    k_max: int,
                    rng: np.random.Generator) -> tuple[int, int, float]:
    """
    Choose a source index `i` (large contribution to the current max) and
    a destination index `j` (minimal contribution).  Returns (i, j, max_contrib).
    """
    n = a.shape[0]
    partner = k_max - np.arange(n)               # partner[i] = k_max - i
    mask = (partner >= 0) & (partner < n)

    contrib = np.zeros(n, dtype=np.float64)
    contrib[mask] = a[partner[mask]]

    # source – the index with the largest contribution
    i = int(np.argmax(contrib))

    # destination – any index with the smallest contribution (prefer zero)
    min_val = contrib.min()
    candidates = np.where(contrib == min_val)[0]

    # make sure i is not selected as destination
    if candidates.size == 1 and candidates[0] == i:
        # fallback: pick a random index different from i
        j = (i + 1) % n
    else:
        # remove i if it appears (unlikely when min_val == 0)
        candidates = candidates[candidates != i]
        if candidates.size == 0:          # extremely unlikely
            j = (i + 1) % n
        else:
            j = int(rng.choice(candidates))

    return i, j, contrib[i]


def _recompute_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Recompute the full self‑convolution from scratch (used after jitter)."""
    return _conv_self(a, fft_len)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # a small safety margin – stop a little before the deadline
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0)  Load the existing construction (if any) as a baseline
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(map(float, globals()["height_sequence_1"]))
            sc = evaluate_sequence(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] baseline (global) score = {sc:.6f} "
                      f"(length {len(cand)})")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1)  Initialise a working sequence (arcsine if no global baseline)
    # ------------------------------------------------------------------
    if best_seq is None:
        # use a modest length to keep each iteration cheap
        n0 = 3809
        a = _make_arcsine(n0, alpha=0.5)
        seq = a.tolist()
        sc = evaluate_sequence(seq)
        best_score = sc
        best_seq = seq
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] initial arcsine score = {sc:.6f} (n={n0})")
    else:
        a = np.array(best_seq, dtype=np.float64)

    n = a.shape[0]
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2)  Compute the initial self‑convolution (full linear)
    # ------------------------------------------------------------------
    b = _conv_self(a, fft_len)          # length 2·n‑1
    cur_score = 2.0 * n * np.max(b)    # sum(a) == 1 → equivalent to evaluate_sequence

    # keep the overall best (may be better than the starting one)
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.tolist()
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] updated best = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Hill‑climbing loop
    # ------------------------------------------------------------------
    # parameters controlling the move magnitude
    step_frac = 0.5                 # initial fraction of source mass to move
    min_step = 1e-8                # avoid vanishing updates
    decay_when_reject = 0.90       # shrink step after a rejected move
    grow_when_accept = 1.02        # gently enlarge step after a success
    max_step = 0.9                 # clamp upper bound

    # pre‑allocate a buffer for the convolution update (avoids repeated allocations)
    diff_buf = np.empty(2 * n - 1, dtype=np.float64)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # locate the current worst convolution index
        k_max = int(np.argmax(b))

        # pick source and destination indices
        i_src, j_dst, contrib_i = _select_indices(a, k_max, rng)

        # if the source contributes almost nothing, we are stuck – break
        if contrib_i <= 0.0:
            break

        # compute a tentative move size
        delta = step_frac * a[i_src]
        if delta < min_step:
            delta = min_step
        if delta > a[i_src]:       # safety, should never happen
            delta = a[i_src]

        # apply the move (in place) and obtain the new convolution
        new_b = _apply_move(a, b, i_src, j_dst, delta, diff_buf)

        # evaluate the new objective (sum(a) stays 1)
        new_score = 2.0 * n * np.max(new_b)

        if new_score < cur_score:                     # *** ACCEPT ***
            cur_score = new_score
            step_frac = min(max_step, step_frac * grow_when_accept)
            # record the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_seq = a.tolist()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best = {new_score:.6f} "
                      f"(iteration {iteration})")
        else:                                          # *** REJECT ***
            # rollback the move by undoing the delta
            # (the inverse operation is the same with swapped indices)
            _apply_move(a, b, j_dst, i_src, delta, diff_buf)   # restores a and b
            step_frac = max(min_step, step_frac * decay_when_reject)

        # occasional random jitter to escape flat plateaus
        if rng.random() < 0.001:
            jitter = np.exp(rng.normal(0.0, 0.05, size=n))
            a *= jitter
            a = np.clip(a, 0.0, None)
            total = a.sum()
            if total == 0.0:
                a.fill(1.0 / n)
            else:
                a /= total
            # recompute the whole convolution after a jitter
            b = _recompute_conv(a, fft_len)
            cur_score = 2.0 * n * np.max(b)
            # after a big perturbation we reset the step size
            step_frac = 0.5

        # every few seconds we renormalise to guard against floating‑point drift
        if iteration % 5000 == 0:
            total = a.sum()
            if abs(total - 1.0) > 1e-12:
                a /= total
                b = _recompute_conv(a, fft_len)
                cur_score = 2.0 * n * np.max(b)

    # ------------------------------------------------------------------
    # 4)  Fallback – if nothing succeeded (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Clip to the limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
