def propose_candidate(seed=..., budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 1.0   # safety

    # Determine n: use length of known height_sequence_1 if exists, else default 3809? Might choose 4000.
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # Clip and normalize
        init = np.clip(init, 0, None)
        if init.sum() == 0:
            init = np.full_like(init, 1.0 / init.size)
        else:
            init = init / init.sum()
        n = init.size
        a = init.copy()
    else:
        n = 3809
        a = np.full(n, 1.0 / n)

    # Precompute FFT length
    fft_len = _fft_len_for_len(2 * n - 1)

    # Helper compute
    def cost(a):
        conv = _conv_self(a, fft_len)
        return 2.0 * n * conv.max()

    best_a = a.copy()
    best_score = cost(a)

    # Simulated annealing:
    # parameters
    max_iter = 10**7   # large, but we break on deadline
    t = 0
    # Starting temperature based on initial score difference: maybe 0.05 * best_score
    temp_start = best_score * 0.1
    temp_end = best_score * 0.0005
    # linear or exponential schedule based on fraction of budget elapsed
    while time.monotonic() < deadline:
        t += 1
        # compute temperature as exponential cooling
        fraction = (time.monotonic() - start) / (deadline - start)
        temp = temp_start * (temp_end / temp_start) ** fraction   # exponential decay

        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] == 0.0:
            continue
        # choose delta up to fraction of a[i] e.g., up to 0.2*a[i]
        max_delta = min(a[i], 0.2 * a[i])
        delta = rng.random() * max_delta
        # apply move
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # compute cost
        conv = _conv_self(a2, fft_len)
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # acceptance
        if new_score < best_score:
            # update best globally also
            best_score = new_score
            best_a = a2.copy()
            print(f"[{time.monotonic()-start:6.2f}s] New best SA: {best_score:.6f}")

"""
Simulated‑annealing + short Adam polishing.

The search works on the probability simplex (∑a=1).  Starting from the
previous best construction (if any) we run a long
simulated‑annealing loop that repeatedly transfers a small amount of mass
between two coordinates or perturbs a random block.  The acceptance
probability follows an exponential schedule with a high initial temperature
so the algorithm can escape the basin that gradient descent got stuck in.
Occasionally the current best vector is slightly perturbed to explore a new
neighbourhood.

When the time budget is almost exhausted a short Adam/soft‑max optimisation
(`_local_optimise`) is run to polish the best candidate.

The returned list is a non‑negative step‑function; its scale does not affect
`evaluate_sequence` (the objective is homogeneous), only the relative
shape matters.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random block perturbation (used by both SA and the polishing step)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


# ----------------------------------------------------------------------
# Gradient utilities for the polishing stage (Adam/soft‑max)
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. `a` (hard worst‑entry)."""
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(
    a: np.ndarray,
    conv: np.ndarray,
    beta: float,
    fft_len_grad: int,
) -> np.ndarray:
    """Gradient of the smooth max f_β(a) = (1/β)·log Σexp(β·conv)."""
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)  # length = 3n‑2
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Short Adam‑based polishing (identical to the earlier code but with a
# larger iteration budget)
# ----------------------------------------------------------------------


def _local_optimise(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Intensive Adam optimisation starting from a simplex‑feasible `a`."""
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state (only second moment is used)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_vec = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len_conv).max()

    max_iters = 3000  # more iterations than the original code
    while t < max_iters and time.monotonic() < deadline:
        t += 1

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # choose gradient (hard sub‑gradient 60 % of the time)
        if rng.random() < 0.6:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = 0.5 + (2000.0 - 0.5) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # Adam update (additive)
        lr = 0.7 / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # occasional random moves
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.04:
            # cheap pairwise move
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                max_delta = min(a[i], 0.2 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a2 = a.copy()
                    a2[i] -= delta
                    a2[j] += delta
                    a2 = _project_onto_simplex(a2)
                    conv2 = _conv_self(a2, fft_len_conv)
                    if 2.0 * n * conv2.max() < cur_score:
                        a = a2

    return best_vec, best_score

    total_time = deadline - start
    iter_counter = 0

    while time.monotonic() < deadline:
        iter_counter += 1
        # exponential cooling based on elapsed fraction
        elapsed = time.monotonic() - start
        frac = elapsed / total_time
        temperature = temp_start * (temp_final / temp_start) ** frac

        # ----- propose a move -------------------------------------------------
        move_type = rng.choice(["pair", "block"], p=[0.9, 0.1])
        if move_type == "pair":
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            max_delta = min(a[i], 0.2 * a[i])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
        else:  # block perturbation (larger, less frequent)
            a2 = _random_block_perturb(a, rng)

        # ----- evaluate the new candidate ------------------------------------
        conv2 = _conv_self(a2, fft_len)
        score2 = 2.0 * n * conv2.max()

        # ----- Metropolis acceptance -----------------------------------------
        accept = False
        if score2 < cur_score:
            accept = True
        else:
            prob = np.exp(-(score2 - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a2
            cur_score = score2

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{elapsed:6.2f}s] New best SA: {best_score:.6f}")

        # occasional “restart‑like” jump from the current best to keep diversity
        if iter_counter % 20000 == 0:
            a = best_a.copy()
            a = _random_block_perturb(a, rng)
            cur_score = 2.0 * n * _conv_self(a, fft_len).max()

    print(f"SA finished after {iter_counter} iterations, best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with Adam/soft‑max (if there is still time left)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 0.5:
        polished, polished_score = _local_optimise(best_a.copy(), rng, deadline)
        if polished_score < best_score:
            best_score = polished_score
            best_a = polished.copy()
            print(f"Polished final score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety clipping (evaluate_sequence does the same)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)

    # Return plain Python floats for the harness
    return [float(x) for x in best_a]
