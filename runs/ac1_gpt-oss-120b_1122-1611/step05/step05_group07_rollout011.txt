def _beta_shape(rng, n):
    alpha = rng.uniform(0.0, 10.0)  # maybe uniform 0-10

def _gaussian_shape(rng, n):
    mu_frac = rng.random()
    sigma_frac = rng.uniform(0.02, 0.3)
    xs = np.arange(n)
    mu = mu_frac * (n - 1)
    sigma = sigma_frac * (n - 1)
    a = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    # optionally normalize
    return a / a.sum()

Exponential shape:

def _exponential_shape(rng, n):
    lam = rng.uniform(0.001, 5.0)
    direction = rng.choice([0,1])  # 0 left->right decreasing, 1 reversed
    xs = np.arange(n)
    if direction == 0:
        a = np.exp(-lam * xs)
    else:
        a = np.exp(-lam * (n - 1 - xs))
    return a / a.sum()

def _sine_power_shape(rng, n):
    p = rng.uniform(0.5, 6.0)

    return a / a.sum()

"""
Propose a step‑function that improves the current bound.
--------------------------------------------------------------------
The algorithm consists of two phases:

1️⃣  **Massive parametric random search** (≈ 80 % of the time budget).
    A large number of simple analytical families (beta‑distribution,
    Gaussian, exponential, sine‑power, Hann‑window, Dirichlet draws and
    mixtures) are sampled.  Each candidate is normalised to sum = 1,
    its self‑convolution max is obtained by an FFT and the objective
    `2·n·max(conv)/(∑a)²` (identical to `evaluate_sequence`) is
    computed.  The best vector seen so far is kept.

2️⃣  **Fast mirror‑descent refinement** (≈ the remaining 20 %).
    Starting from the best vector we repeatedly
    - compute the index `k_max` where the self‑convolution attains its
      maximum,
    - evaluate the exact sub‑gradient `g_i = 2·a_{k_max‑i}` (zero outside
      the support),
    - perform an exponentiated‑gradient (entropy mirror) step
      `a_i ← a_i·exp(-η·g_i)`, normalising back to the simplex,
    - occasionally apply a small log‑normal jitter and restart if no
      improvement is observed for 30 s.

Both phases are time‑controlled; the routine stops 5 s before the
overall budget (to avoid a hard timeout) and returns the best sequence
found.  The returned list satisfies all security checks of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# cache for FFT lengths (reuse across many calls)
_fft_len_cache = {}


def _fft_len(n: int) -> int:
    """FFT length (power of two) for a self‑convolution of length n."""
    if n not in _fft_len_cache:
        _fft_len_cache[n] = _next_pow2(2 * n - 1)
    return _fft_len_cache[n]


def _self_conv_full(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (full length)."""
    n = a.size
    L = _fft_len(n)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence``.
    a is allowed to contain any non‑negative numbers; sum(a) must be >0.
    """
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.size
    L = _fft_len(n)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = conv.max()
    # evaluate_sequence uses 2·n·max_b/(sum_a**2)
    return float(2.0 * n * max_b / (s * s))

# Candidate families (return a vector that already sums to 1)
# ----------------------------------------------------------------------


def _beta_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Beta‑distribution shape: x**α·(1‑x)**β."""
    alpha = rng.uniform(0.0, 10.0)
    beta = rng.uniform(0.0, 10.0)
    x = np.linspace(0.0, 1.0, n)
    a = (x ** alpha) * ((1.0 - x) ** beta)
    # optional flip to obtain a right‑skewed version
    if rng.random() < 0.5:
        a = a[::-1]
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _gaussian_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Gaussian envelope with random centre and width."""
    mu_frac = rng.random()
    sigma_frac = rng.uniform(0.02, 0.35)
    mu = mu_frac * (n - 1)
    sigma = max(sigma_frac * (n - 1), 1e-6)
    xs = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((xs - mu) / sigma) ** 2)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _exponential_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Monotone exponential decay (or its mirror)."""
    lam = rng.uniform(0.001, 5.0)
    direction = rng.choice([0, 1])  # 0 → left‑to‑right, 1 → right‑to‑left
    xs = np.arange(n, dtype=np.float64)
    if direction == 0:
        a = np.exp(-lam * xs)
    else:
        a = np.exp(-lam * (n - 1 - xs))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _sine_power_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Half‑sine raised to a random power."""
    p = rng.uniform(0.5, 6.0)
    xs = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * xs / n) ** p
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _hann_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Standard Hann window (cosine‑bell)."""
    xs = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * xs / (n - 1)))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _dirichlet_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Standard Dirichlet draw (α_i = 1)."""
    a = rng.dirichlet(np.ones(n)).astype(np.float64)
    a /= a.sum()
    return a


def _dirichlet_conc_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Dirichlet with a common concentration parameter."""
    conc = rng.uniform(0.1, 5.0)
    a = rng.gamma(conc, 1.0, size=n).astype(np.float64)
    a /= a.sum()
    return a


def _mix_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Mixture of two random families."""
    families = [_beta_shape, _gaussian_shape, _exponential_shape,
                _sine_power_shape, _hann_shape, _dirichlet_shape,
                _dirichlet_conc_shape]
    f1, f2 = rng.choice(families, size=2, replace=True)
    a1 = f1(rng, n)
    a2 = f2(rng, n)
    w = rng.random()
    a = w * a1 + (1.0 - w) * a2
    a /= a.sum()
    return a


def _jitter_about(rng: np.random.Generator, a: np.ndarray) -> np.ndarray:
    """Multiplicative log‑normal jitter around an existing vector."""
    noise = np.exp(rng.normal(0.0, 0.07, size=a.shape))
    out = a * noise
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine the base length (use previous construction if available)
    # ------------------------------------------------------------------
    base_seq = globals().get("height_sequence_1", None)
    if isinstance(base_seq, (list, tuple, np.ndarray)):
        base_arr = np.asarray(base_seq, dtype=np.float64)
        if base_arr.ndim == 1 and base_arr.size > 0 and np.isfinite(base_arr).all():
            base_arr = np.clip(base_arr, 0.0, None)
            s = base_arr.sum()
            if s > 0.0:
                base_arr = base_arr / s
            else:
                base_arr = np.full_like(base_arr, 1.0 / base_arr.size)
            n = base_arr.size
            best = base_arr.copy()
            best_score = _score(best)
        else:
            # fallback to a uniform start
            n = 3840
            best = np.full(n, 1.0 / n, dtype=np.float64)
            best_score = _score(best)
    else:
        n = 3840
        best = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score(best)

    # ------------------------------------------------------------------
    # 2) Prepare candidate generators
    # ------------------------------------------------------------------
    families = [_beta_shape, _gaussian_shape, _exponential_shape,
                _sine_power_shape, _hann_shape, _dirichlet_shape,
                _dirichlet_conc_shape, _mix_shape]

    # ------------------------------------------------------------------
    # 3) Phase 1 – massive random search (≈80 % of the budget)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    phase1_deadline = start_time + 0.80 * (deadline - start_time)

    iteration = 0
    target_score = 1.5025  # stopping target

    while time.monotonic() < phase1_deadline:
        iteration += 1

        # with 10 % probability explore locally around the current best
        if rng.random() < 0.10:
            cand = _jitter_about(rng, best)
        else:
            maker = rng.choice(families)
            cand = maker(rng, n)

        if rng.random() < 0.03:
            cand = _jitter_about(rng, cand)

        cur = _score(cand)

        if cur < best_score - 1e-12:
            best = cand.copy()
            best_score = cur
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (random) score = {best_score:.6f}")
            if best_score <= target_score:
                break

    # ------------------------------------------------------------------
    # 4) Phase 2 – targeted mirror‑descent refinement (≈20 % of the budget)
    # ------------------------------------------------------------------
    a = best.copy()                     # working copy
    n = a.size
    last_improve = time.monotonic()
    iter_refine = 0

    while time.monotonic() < deadline:
        iter_refine += 1

        # ----- compute current convolution and the peak index -----
        conv = _self_conv_full(a)
        max_c = conv.max()
        k_max = int(conv.argmax())
        cur_score = 2.0 * n * max_c          # sum(a) == 1 after every step

        # ----- exact sub‑gradient of the max (one‑hot peak) -----
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1)
            partner = k_max - i_range
            grad[i_range] = 2.0 * a[partner]

        # ----- schedule for the step size η (linear decay) -----
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        eta = 1.5 * (1.0 - frac) + 0.001 * frac   # from ~1.5 → 0.001

        # ----- exponentiated‑gradient (entropy mirror) update -----
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ----- optional jitter (keeps the search from stalling) -----
        if rng.random() < 0.02:
            a = _jitter_about(rng, a)

        # ----- evaluate after the update -----
        conv_new = _self_conv_full(a)
        max_c_new = conv_new.max()
        new_score = 2.0 * n * max_c_new

        if new_score < best_score - 1e-12:
            best = a.copy()
            best_score = new_score
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] refined best score = {best_score:.6f}")
            if best_score <= target_score:
                break

        # ----- soft restart if we stagnate for >30 s -----
        if now - last_improve > 30.0:
            a = best.copy()
            a = _jitter_about(rng, a)      # moderate perturbation
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart during refinement – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
