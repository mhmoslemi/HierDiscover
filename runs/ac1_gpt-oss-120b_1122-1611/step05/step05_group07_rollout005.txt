"""
Propose a step‑function that beats the previous bound (≈1.514166).

Algorithm
---------
1. **Initialisation** – try a handful of hand‑crafted shapes
   (uniform, exponential, linear, two‑spike, etc.) together with a few
   Dirichlet samples.  Keep the best one (according to the exact
   evaluation) as the starting vector `a` (normalised to sum = 1).

2. **Phase 1 – deterministic mirror descent** (≈15 % of the budget)  
   Repeatedly move the distribution away from the indices that
   contribute most to the current peak of the self‑convolution.
   The exact sub‑gradient of `max(a*a)` at the peak `k_max` is
   `g_i = 2·a_{k_max‑i}` (with zero outside the support).  An
   exponentiated‑gradient update (`a_i ← a_i·exp(‑η·g_i)`) is applied
   with a linear schedule for the step size `η`.  After each update the
   full convolution is recomputed (FFT) to obtain a fresh peak.

3. **Phase 2 – fast local stochastic refinement** (≈85 % of the budget)  
   *Pick a donor*:  indices are sampled proportional to their
   contribution `a_i·a_{k_max‑i}` to the peak.  
   *Pick a recipient*:  a random index (biased towards small values).  
   *Mass transfer*:  move a fraction `δ ≤ 0.5·a_i` from the donor to the
   recipient.  The convolution is updated **incrementally** in
   O(n) time using the exact formula

       Δc = 2·δ·(shift_j – shift_i) + δ²·e_{2j}+δ²·e_{2i}–2·δ²·e_{i+j}

   where `shift_x` is the vector `a` placed at offset `x`.  The new
   score `2·n·max(c)` is accepted if it improves, otherwise with a
   Metropolis rule based on a linearly cooling temperature.

   Every 500 successful moves the full convolution is recomputed to
   eliminate accumulated floating‑point drift.  Occasionally (≈3 %)

4. **Restart** – if no improvement is observed for 30 s, restart from

The routine respects the 1000 s time budget (with a 5 s safety margin)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new a and c."""
    n = a.size
    # new a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # incremental update of convolution
    c_new = c.copy()
    # shift of donor i (subtract) and recipient j (add)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ part
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A small portfolio of diverse starting vectors."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) exponential decays (several bases)
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # 3) linear decreasing
    v = (np.arange(n, 0, -1, dtype=np.float64))
    v /= v.sum()
    candidates.append(v)

    # 4) two‑spike at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    candidates.append(v)

    # 5) piecewise constant (first half heavy, second half zero)
    half = n // 2
    v = np.zeros(n, dtype=np.float64)

    candidates.append(v)

    # 6) a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        candidates.append(v.astype(np.float64))

    return candidates
