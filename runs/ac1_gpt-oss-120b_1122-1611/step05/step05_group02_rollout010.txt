def compute_grad(a, p):
    # a: shape (n,)
    # p: shape (2n-1,)
    a_rev = a[::-1]
    # conv = np.convolve(p, a_rev) length = (2n-1)+n-1 = 3n-2
    # we need indices from n-1 to 2n-2 (inclusive) (since n-1 offset)
    # conv segment length n
    total_len = p.size + a_rev.size - 1
    # compute conv with FFT for speed when n large
    length_fft = 1 << ((total_len - 1).bit_length())
    fp = np.fft.rfft(p, n=length_fft)
    fa_rev = np.fft.rfft(a_rev, n=length_fft)
    prod = fp * fa_rev
    conv_full = np.fft.irfft(prod, n=length_fft)[:total_len]
    start = a.size - 1
    grad = 2.0 * conv_full[start:start + a.size]
    return grad

    # Determine n (maybe 4096)

        else:

        init_a = rng.random(n)

    y = np.log(np.clip(init_a, eps, None))

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = init_a.copy()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)
    last_improve = start_time
    iter_count = 0

    # schedule parameters
    lr_start = 0.2
    lr_end = 0.001
    tau_start = 0.5
    tau_end = 20.0

    while time.monotonic() < deadline:

        y_max = np.max(y)
        exp_y = np.exp(y - y_max)

        max_conv = np.max(conv)

"""
Smooth‑max gradient descent on the probability simplex.

We optimise a probability vector a (a_i ≥ 0, Σ a_i = 1) to minimise
    2·n·max_k (a*a)[k]    (the same objective that ``evaluate_sequence`` uses).

Instead of the non‑smooth max we minimise a smooth approximation:
    Lτ(a) = (1/τ)·log( Σ_k exp( τ·c_k ) )
where c_k = (a*a)[k] / max_k c_k ∈ [0,1] and τ grows from a small
value to a larger one.  The gradient of Lτ can be written in closed form
and is obtained efficiently with a second FFT‑based convolution.
Updates are performed on the log‑variables y (a = softmax(y)) using
an Adam optimiser.  Periodic restarts, occasional random jitter and a
final short random‑perturbation phase help escape local minima.

The routine respects the 1000 s budget, keeps the best vector seen,
and returns it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities (same as in the reference implementation)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a real vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_weighted_fft(p: np.ndarray, a_rev: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution of weight vector ``p`` with the reversed probability vector ``a_rev``."""
    fp = np.fft.rfft(p, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    prod = fp * fa
    conv_full = np.fft.irfft(prod, n=fft_len)[: p.size + a_rev.size - 1]
    return conv_full


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Initialise a probability vector (sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a0 = np.clip(cand, 0.0, None)
                s = a0.sum()
                a0 = a0 / s if s > 0.0 else np.full(cand.shape, 1.0 / cand.size)
                n = a0.size
            else:
                raise ValueError
        except Exception:
            n = 4096
            a0 = rng.random(n)
            a0 = a0 / a0.sum()
    else:
        n = 4096
        a0 = rng.random(n)
        a0 = a0 / a0.sum()

    # Fixed FFT lengths (pre‑computed for speed)
    fft_len_self = _fft_len_for_n(n)                     # for a*a
    fft_len_grad = 1 << ((3 * n - 2 - 1).bit_length())   # for p * a_rev (size ≈3n)

    # Optimisation variables: work in the log‑domain (softmax parametrisation)
    eps = 1e-12
    y = np.log(np.clip(a0, eps, None))

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin
    last_improve = start_time
    iteration = 0

    # Hyper‑parameters (linear schedules)
    lr_start = 0.20      # initial Adam step size
    lr_end = 0.001       # final step size
    tau_start = 0.5      # smooth‑max temperature (low → smoother)
    tau_end = 20.0       # final temperature (sharper, approaches true max)
    restart_gap = 30.0   # seconds without improvement → restart

    while time.monotonic() < deadline:
        # ----- recover probability vector from y -----
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)          # stabilise exponentials
        a = exp_y / exp_y.sum()            # a_i ≥ 0, Σ a_i = 1

        # ----- self‑convolution and current score -----
        conv = _conv_self_fft(a, fft_len_self)      # length 2n‑1
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f} "
                  f"(max conv={max_conv:.8f})")

        # ----- schedule for learning‑rate and temperature -----
        now = time.monotonic()
        elapsed_total = now - start_time
        total = deadline - start_time
        lr = lr_start + (lr_end - lr_start) * (elapsed_total / total)
        tau = tau_start + (tau_end - tau_start) * (elapsed_total / total)

        # ----- smooth‑max weights (soft‑max over normalized conv) -----
        # Normalise so that the maximum entry becomes 1.0
        conv_norm = conv / max_conv
        # w_k = exp( τ·(c_k – 1) ), i.e. max weight = 1, others ≤ 1
        w = np.exp(tau * (conv_norm - 1.0))
        w_sum = w.sum()
        if w_sum == 0.0:                     # safety fallback (should not happen)
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                         # probabilities p_k

        # ----- gradient of the smooth‑max objective w.r.t. a -----
        a_rev = a[::-1]                       # reversed probability vector
        conv_full = _conv_weighted_fft(w, a_rev, fft_len_grad)
        start = a.size - 1                    # offset where the needed segment begins
        grad_a = 2.0 * conv_full[start:start + a.size]   # shape (n,)

        # ----- chain rule to obtain gradient w.r.t. y (softmax re‑param) -----
        g_scalar = np.dot(grad_a, a)          # a·grad_a
        grad_y = a * (grad_a - g_scalar)      # ∂L/∂y

        # ----- Adam update on y -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional random jitter to escape plateaus -----
        if rng.random() < 0.01:
            y += rng.normal(0.0, 0.1, size=n)

        # ----- soft restart if stagnation is detected -----
        if now - last_improve > restart_gap:
            # Re‑initialise around the current best solution
            if best_a is not None:
                y = np.log(np.clip(best_a, eps, None))
                y += rng.normal(0.0, 0.2, size=n)
                m.fill(0.0)
                v.fill(0.0)
            else:
                y = np.log(np.clip(a0, eps, None)) + rng.normal(0.0, 0.2, size=n)
                m.fill(0.0)
                v.fill(0.0)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Light local search around the best candidate (time‑permitting)
    # ------------------------------------------------------------------
    if best_a is not None:
        # Use the given evaluate_sequence for a true score (protect against rounding)
        try:
            current_score = evaluate_sequence(best_a.tolist())
        except Exception:
            current_score = best_score

        # Allocate a small remaining slice of the budget for perturbations
        extra_deadline = start_time + budget_s
        while time.monotonic() < extra_deadline:
            # small multiplicative block perturbation
            width = rng.integers(1, max(2, n // 15))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            cand = best_a.copy()
            cand[start_idx:start_idx + width] *= factor
            cand = np.clip(cand, 0.0, None)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                continue
            cand /= cand_sum
            score = evaluate_sequence(cand.tolist())
            if score < current_score:
                best_a = cand
                current_score = score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] local improvement → score {current_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best solution (respect clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – return a uniform vector
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a.tolist()]
