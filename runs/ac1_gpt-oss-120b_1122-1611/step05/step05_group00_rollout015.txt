"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.  The routine works in three stages:

1. **Broad shape search** – a cheap grid over several parametric families
   (sqrt‑difference with offset, power‑differences, symmetric versions,
   exponentials).  The best shape on the simplex is kept as a warm‑start.

2. **Fast mirror‑descent** – an exponentiated‑gradient (EG) on the simplex
   that drives the ℓ∞‑norm of the self‑convolution down.  The EG uses an
   active‑set soft‑max (temperature β) and momentum, with occasional
   random block perturbations and soft restarts.

3. **Fine‑grained simulated annealing** – a cheap incremental update of the
   self‑convolution (O(n) per move) is used to explore tiny mass transfers
   between two coordinates.  A temperature schedule lets the search accept
   occasional uphill moves, while every few accepted moves the convolution
   is recomputed from scratch (FFT) to avoid drift.

The whole optimisation respects a 1000 s budget and returns the best
sequence discovered (clipped to the limits of ``evaluate_sequence``)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT length, convolution, gradient of soft‑max)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length to hold a full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Grad of Σ_k w_k (a∗a)_k = 2·a ⋆ w (used only as a fallback)."""
    n = a.size
    L = w.size                # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass transfer
# ----------------------------------------------------------------------
def _apply_move(conv: np.ndarray, a: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a fresh convolution array after moving ``delta`` mass from
    index ``i`` to ``j`` (sum of ``a`` stays constant).  The update follows
    the exact algebraic relation:

        a' = a  - delta·e_i + delta·e_j
        (a')*(a') = a*a
                    - 2·delta·shift_i(a)
                    + 2·delta·shift_j(a)
                    + delta^2·δ_{2i} + delta^2·δ_{2j}
                    - 2·delta^2·δ_{i+j}

    where ``shift_i(a)`` is ``a`` placed at positions ``i … i+n-1``.
    """
    n = a.shape[0]
    new_conv = conv.copy()

    # subtract contribution of the source index i
    new_conv[i:i + n] -= 2.0 * delta * a
    # add contribution of the destination index j
    new_conv[j:j + n] += 2.0 * delta * a
    # quadratic terms of the move
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv

            best = a.copy()
            best_score = sc
            print(f"[grid] power‑diff q={q:.3f} → {sc:.6f}")

    # (c) symmetric sqrt‑difference (mass at both edges)
    for delta in np.linspace(0.0, 50.0, 11):
        idx = np.minimum(np.arange(n, dtype=np.float64),
                         n - 1 - np.arange(n, dtype=np.float64))
        a = (np.sqrt(idx + 1.0 + delta) - np.sqrt(idx + delta))
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc
            print(f"[grid] sym‑sqrt δ={delta:.1f} → {sc:.6f}")

    # (d) simple exponential decay
    for alpha in np.linspace(5e-4, 1e-2, 12):
        a = np.exp(-alpha * np.arange(n, dtype=np.float64))
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_score:
            best = a.copy()
            best_score = sc
            print(f"[grid] exp‑decay α={alpha:.5f} → {sc:.6f}")

    # ensure we have a starting point
    a = best.copy()
    cur_score = best_score
    print(f"[init] chosen start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase‑1 : exponentiated‑gradient (mirror descent) on the simplex
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # tiny safety margin

    # schedule parameters (tuned a bit more aggressively)
    eta_start = 0.8
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 2500.0
    block_perturb_prob = 0.04
    pair_move_prob = 0.09
    restart_gap = 125.0
    momentum = 0.85

    grad_prev = np.zeros_like(a)

    # allocate roughly the first 40 % of the budget to this phase
    phase1_deadline = start_time + 0.4 * (deadline - start_time)

    while time.monotonic() < phase1_deadline:
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] EG new best = {best_score:.6f}")

        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear schedules
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # active set (values within 0.5 % of the current maximum)
        tol = cur_max * 0.995
        active = np.where(conv >= tol)[0]
        if active.size > 12:
            active = np.argsort(conv)[-12:]

        if active.size > 0:
            w_raw = np.exp(beta * (conv[active] - cur_max))
            w_sum = w_raw.sum()
            w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)

            grad = np.zeros_like(a)
            for idx, weight in zip(active, w):
                lo = max(0, idx - (n - 1))
                hi = min(n - 1, idx)
                i_range = np.arange(lo, hi + 1, dtype=np.int64)
                grad[i_range] += weight * 2.0 * a[idx - i_range]
        else:
            # Very unlikely – fall back to a full soft‑max gradient
            w = np.exp(beta * (conv - cur_max))
            w = w / w.sum()
            grad = _gradient_correlation(a, w)

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient step (mirror descent on simplex)
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # occasional random perturbations
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            blk_factor = np.exp(rng.normal(0.0, 0.25))
            a[start:start + width] *= blk_factor
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-14:
                delta = rng.random() * a[i] * 0.5
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

        # soft restart if no improvement for a while
        if now - (best_score_time if (best_score_time := start_time) else start_time) > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            grad_prev = np.zeros_like(a)
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Phase‑2 : Simulated annealing with incremental convolution updates
    # ------------------------------------------------------------------
    # initialise SA state with the best vector found so far
    cur = best.copy()
    conv_cur = _conv_self(cur, fft_len)
    cur_score = 2.0 * n * conv_cur.max()
    accepted = 0

    # temperature schedule (geometric decay)
    T_start = 2e-4
    T_end = 5e-7

    while time.monotonic() < deadline:
        # compute temperature based on overall progress
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * (1.0 - frac) + T_end * frac

        # propose a random mass transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(cur[i], 0.025)          # limit step size
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer

        # incremental convolution update
        conv_new = _apply_move(conv_cur, cur, i, j, delta)
        new_score = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-15)):
            # accept move
            cur[i] -= delta
            cur[j] += delta
            conv_cur = conv_new
            cur_score = new_score
            accepted += 1

            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] SA new best = {best_score:.6f}")

            # periodically recompute the convolution from scratch to avoid drift
            if accepted % 500 == 0:
                conv_cur = _conv_self(cur, fft_len)

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
