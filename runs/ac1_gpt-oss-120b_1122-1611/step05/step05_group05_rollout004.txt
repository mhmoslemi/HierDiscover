"""
Propose a step‑function (non‑negative sequence) that reduces the evaluation
metric

    score = 2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the linear convolution of the sequence with itself.
The algorithm consists of:

* initialise a feasible vector (the simplex ∑a_i = 1, a_i ≥ 0);
* repeatedly compute the convolution (FFT for speed) and locate the peak;
* take a sub‑gradient of the peak (g_i = 2·a_{p−i} when the index is valid);
* perform an exponentiated‑gradient step (a_i ← a_i·exp(−η·g_i));
* re‑normalise onto the simplex.
* occasional random “jitter’’ and shuffling of the vector are added to
  escape plateaus, and a simple simulated‑annealing acceptance rule
  allows occasional worsening moves.

The method keeps the best candidate seen before the time budget expires
and returns it as a plain Python list.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a 1‑D array onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        # all entries are equal – return the uniform simplex point
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using the FFT.  Returns a real
    ``np.ndarray`` of length 2*len(a)-1.
    """
    n = a.shape[0]
    # size of FFT – next power of two ≥ 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, n=m)
    prod = ft * ft
    conv = np.fft.irfft(prod, n=m)[: 2 * n - 1]
    return conv

            tmp[start:start + width] *= factor
            a = _project_simplex(tmp)

        # ----- restart if we are stuck for a while -----
        if time.monotonic() - last_improve > restart_gap:
            # random reshuffle of the best vector + small Gaussian noise
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * noise
            # optionally permute to explore different index alignments
            a = a[rng.permutation(n)]
            a = _project_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best candidate (clip to allowed range)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
