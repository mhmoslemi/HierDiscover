def simplex_projection(v):
    # solves min ||x - v||_2 s.t. x >=0, sum x = 1.
    # If sum of positive entries =0 -> return uniform?
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v))+1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):

    # Using algorithm from Duchi et al. (2008)
    if np.sum(v) == 1 and np.all(v >= 0):

"""
Propose a candidate step‑function using two complementary phases.

Phase 1 – smooth fast optimisation.
    • Treat the evaluation 2·n·max(a∗a) as 2·n·max(conv) with ∑a = 1.
    • Approximate the max by a smooth log‑sum‑exp (parameter β).
    • Compute the gradient of the smoothed objective using FFTs
      (the gradient is a convolution of the soft‑max weights with the
      reversed coefficient vector).
    • Perform a projected gradient descent with a linearly decreasing
      learning rate and an increasing β.  After each step the vector is
      projected onto the probability simplex (non‑negative, sum = 1).

Phase 2 – fine‑grained greedy mass transfers.
    • Starting from the best vector found in phase 1, run a fast
      greedy local search that repeatedly moves a small amount of mass
      from a “donor’’ (large partner contribution to the current peak)
      to a “receiver’’ (small partner).  The convolution is updated in
      O(n) using the exact formula for a rank‑one mass transfer.
    • Random block perturbations and occasional restarts are used to
      escape plateaus.

Both phases respect the security checks of ``evaluate_sequence``.
The routine runs for the allotted time (default 1000 s) and returns the
best sequence discovered.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]          # largest index satisfying the condition
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny rounding errors
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    i: int,
                    j: int,
                    delta: float) -> np.ndarray:
    """
    Return a new convolution after moving ``delta`` mass from index i to j.
    The update follows the exact formula derived from (a+Δ)*(a+Δ).
    """
    n = a.size
    new_conv = conv.copy()
    # linear part: -2Δ·a shifted by i  +  2Δ·a shifted by j
    new_conv[i: i + n] -= 2.0 * delta * a
    new_conv[j: j + n] += 2.0 * delta * a
    # quadratic part (only three entries change)
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          max_width_frac: float = 0.2,
                          sigma: float = 0.12) -> np.ndarray:
    """
    Apply a multiplicative log‑normal jitter to a random contiguous block
    and renormalise onto the probability simplex.
    """
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

        # progress fractions
        elapsed = time.monotonic() - start_time
        total = phase1_deadline - start_time
        frac = elapsed / total

        # schedule for smoothing and step size
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # ----- convolution a*a (full) via FFT -----
        fft_a = np.fft.rfft(a, size_conv)
        conv_full = np.fft.irfft(fft_a * fft_a, size_conv)[:2 * n - 1]

        # ----- smooth approximation to max (log‑sum‑exp) -----
        max_c = conv_full.max()
        shifted = conv_full - max_c               # avoid overflow
        exp_vals = np.exp(beta * shifted)
        sum_exp = exp_vals.sum()
        # probabilities proportional to exp(beta*conv)
        p = exp_vals / sum_exp

        # ----- gradient of the smoothed loss -----
        a_rev = a[::-1]
        fft_p = np.fft.rfft(p, size_grad)
        fft_ar = np.fft.rfft(a_rev, size_grad)
        conv_pa = np.fft.irfft(fft_p * fft_ar, size_grad)
        grad = 2.0 * conv_pa[n - 1: n - 1 + n]    # central slice

        # ----- momentum and projected gradient step -----
        v = momentum * v + (1.0 - momentum) * grad
        a_candidate = a - lr * v
        a = _simplex_projection(a_candidate)

        # ----- evaluate true objective (max‑convolution) -----
        fft_a = np.fft.rfft(a, size_conv)
        conv_full = np.fft.irfft(fft_a * fft_a, size_conv)[:2 * n - 1]
        cur_score = 2.0 * n * conv_full.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] phase‑1 new best {best_score:.6f}")

        # occasional jitter to keep diversity
        if rng.random() < 0.001:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _simplex_projection(a * noise)
            v[:] = 0.0

    # --------------------------------------------------------------
    # 3) Phase 2 – greedy targeted mass transfers (local refinement)
    # --------------------------------------------------------------
    a = best.copy()
    conv = np.convolve(a, a)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        # ---- locate the current max of the convolution ----
        k_max = int(np.argmax(conv))

        # ---- compute partner values a[k_max-i] where defined ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        partner = np.zeros_like(a)
        partner[valid] = a[k_max - idx[valid]]

        # ---- donor: index with large product a[i] * partner[i] ----
        donor_mask = a > 1e-12
        donor_score = partner * donor_mask.astype(float)
        if donor_score.max() == 0.0:
            # fallback random move
            i = rng.integers(0, n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_delta = 0.3 * a[i]
            delta = rng.random() * max_delta
            new_conv = _apply_transfer(a, conv, i, j, delta)
            new_score = 2.0 * n * new_conv.max()
            if new_score < cur_score:
                a[i] -= delta
                a[j] += delta
                conv = new_conv
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] rand‑move improved best {best_score:.6f}")
            continue

        donor = int(np.argmax(donor_score))

        # ---- receiver: index where partner is minimal ----
        receiver_candidates = np.where(partner == partner.min())[0]
        if donor in receiver_candidates:
            if len(receiver_candidates) > 1:
                receiver_candidates = receiver_candidates[receiver_candidates != donor]
            else:
                # pick the next smallest partner
                sorted_idx = np.argsort(partner)
                receiver = sorted_idx[1] if sorted_idx[0] == donor else sorted_idx[0]
                # execute move
                max_delta = 0.3 * a[donor]
                delta = rng.random() * max_delta
                new_conv = _apply_transfer(a, conv, donor, receiver, delta)
                new_score = 2.0 * n * new_conv.max()
                if new_score < cur_score:
                    a[donor] -= delta
                    a[receiver] += delta
                    conv = new_conv
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best = a.copy()
                        last_improve = time.monotonic()
                        elapsed = last_improve - start_time
                        print(f"[{elapsed:6.2f}s] special‑move improved best {best_score:.6f}")
                continue

        receiver = int(rng.choice(receiver_candidates))

        # ---- step size (fraction of donor mass) ----
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac_time = elapsed / total
        max_frac = 0.3 * (1.0 - frac_time) + 0.02 * frac_time
        max_delta = a[donor] * max_frac
        if max_delta < 1e-12:
            continue
        delta = rng.random() * max_delta

        # ---- apply transfer and accept if improvement ----
        new_conv = _apply_transfer(a, conv, donor, receiver, delta)
        new_score = 2.0 * n * new_conv.max()
        if new_score < cur_score:
            a[donor] -= delta
            a[receiver] += delta
            conv = new_conv
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] new best {best_score:.6f}")

        # ---- occasional block perturbation to escape plateaus ----
        if rng.random() < 0.015:
            a_pert = _random_block_perturb(a, rng)
            conv_pert = np.convolve(a_pert, a_pert)
            pert_score = 2.0 * n * conv_pert.max()
            if pert_score < cur_score:
                a = a_pert
                conv = conv_pert
                cur_score = pert_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] block‑perturb improved best {best_score:.6f}")

        # ---- restart if no progress for a while ----
        if time.monotonic() - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = _simplex_projection(a)
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – score reset to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence (clipped to evaluate_sequence limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
