def _pairwise_move(a, rng):
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a
    if a[i] <= 0:
        return a
    # Determine delta: up to some fraction of a[i]
    max_delta = min(a[i], 0.1 * a[i])  # limit to 10% of source mass
    if max_delta <= 0:

    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)

Algorithm
---------

4.  Phase 3 – final simulated‑annealing fine‑tuning on the best vector.
5.  Return the best vector found (clipped to the limits required by

The code respects a user‑supplied time budget (default 1000 s) and

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Next power of two ≥ `length` (sufficient for real FFT)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.size
    if fft_len is None:
        fft_len = _fft_len_for_len(2 * n - 1)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * n - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. `a`.
    The worst convolution index is i_max; for each coordinate j that
    participates we have ∂/∂a_j = 2·a_{i_max‑j}.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth‑max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extraneous “+a” term).
    """
    n = a.size
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    fft_len = _fft_len_for_len(3 * n - 2)
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    conv_w_rev = np.fft.irfft(fw * fa, n=fft_len)
    # slice that aligns with a (indices n‑1 … n‑1+n‑1)
    grad = conv_w_rev[n - 1 : n - 1 + n] / sum_w
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small amount of mass from i to j (mass‑conserving)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] <= 0.0:
        return a
    max_delta = min(a[i], 0.1 * a[i])        # move at most 10 % of source mass
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    return _project_onto_simplex(a2)


def _score_of(a: np.ndarray) -> float:
    """Return the objective 2·n·max(a*a) (sum of a is 1 after projection)."""
    n = a.size
    conv = _conv_self(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def _local_optimize(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 2000,
    beta_max: float = 5000.0,
    lr0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Intensive optimisation of a single vector `a` (already on the simplex).
    Uses Adam with a mixture of exact sub‑gradient and smooth‑max gradient,
    plus occasional random perturbations.
    """
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_vec = a.copy()
    best_score = _score_of(a)

    t = 0
    while t < max_iters and time.monotonic() < deadline:
        t += 1

        # current convolution and score
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # pick gradient type (smooth‑max early, sub‑gradient later)
        if rng.random() < 0.5:
            # smooth‑max with annealing β
            beta = 0.5 + (beta_max - 0.5) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta)
        else:
            # exact sub‑gradient of the current worst entry
            raw_grad = _subgradient_step(a, conv)

        # Adam update (additive)
        lr = lr0 / np.sqrt(t)                # simple decreasing learning rate
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # occasional stochastic moves
        if rng.random() < 0.04:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.05:
            a = _pairwise_move(a, rng)

    return best_vec, best_score


def _upsample_to_length(a: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of `a` onto a longer grid and re‑project onto the simplex."""
    old_len = a.size
    if new_len == old_len:
        return a.copy()
    old_pos = np.linspace(0, new_len - 1, old_len)
    new_pos = np.arange(new_len)
    interpolated = np.interp(new_pos, old_pos, a)
    interpolated = np.maximum(interpolated, 0.0)
    return _project_onto_simplex(interpolated)


def _simulated_annealing(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_moves: int = 5000,
    T0: float = 1e-2,
    T_end: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Monte‑Carlo annealing using the same move set as
    the gradient phase.  Accepts worsening moves with probability exp(-Δ/T).
    """
    n = a.size
    cur = a.copy()
    cur_score = _score_of(cur)

    best_vec = cur.copy()
    best_score = cur_score

    for i in range(max_moves):
        if time.monotonic() >= deadline:
            break

        # exponential temperature schedule
        T = T0 * (T_end / T0) ** (i / max_moves)

        # propose a move
        if rng.random() < 0.6:
            cand = _pairwise_move(cur, rng)
        else:
            cand = _random_block_perturb(cur, rng)

        cand_score = _score_of(cand)
        delta = cand_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            cur = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_vec = cur.copy()

    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    evaluate_sequence() before the allotted time expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # Global best container (vector on the simplex)
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If the notebook already supplied a candidate, evaluate it first
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                cand = _project_onto_simplex(cand)
                cur = _score_of(cand)
                if cur < best_score:
                    best_score = cur
                    best_vec = cand.copy()
                print(f"[init] candidate from globals – score={cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Phase 1 – cheap restarts on medium lengths
    # ------------------------------------------------------------------
    small_lengths = [2000, 2500, 3000, 3500, 3809, 4200, 4700, 5000]
    phase1_deadline = start_time + (deadline - start_time) * 0.55

    while time.monotonic() < phase1_deadline:
        n = int(rng.choice(small_lengths))
        a0 = rng.random(n)
        a0 /= a0.sum()
        a_opt, score_opt = _local_optimize(
            a0, rng, phase1_deadline, max_iters=800, beta_max=1500.0, lr0=0.6
        )
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best (phase 1) n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – enlarge and run a longer optimisation on large lengths
    # ------------------------------------------------------------------
    large_lengths = [8000, 12000, 15000, 18000, 20000]
    phase2_deadline = start_time + (deadline - start_time) * 0.90

    if best_vec is None:                     # fallback – uniform of size 3000
        best_vec = np.full(3000, 1.0 / 3000)

    for L in large_lengths:
        if time.monotonic() >= phase2_deadline:
            break
        up_vec = _upsample_to_length(best_vec, L)
        a_opt, score_opt = _local_optimize(
            up_vec,
            rng,
            phase2_deadline,
            max_iters=2500,
            beta_max=5000.0,
            lr0=0.5,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best (phase 2) L={L} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if best_vec is not None and time.monotonic() < deadline - 2.0:
        remaining_time = deadline - time.monotonic()
        # Rough estimate: each move costs ~0.001 s for n≈20000
        max_moves = int(min(20000, remaining_time / 0.001))
        a_opt, score_opt = _simulated_annealing(
            best_vec,
            rng,
            deadline,
            max_moves=max_moves,
            T0=5e-3,
            T_end=5e-7,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best (phase 3) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety handling
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(2000, 1.0 / 2000)

    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the external evaluator
    return [float(x) for x in best_vec]
