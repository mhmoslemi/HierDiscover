"""
propose_candidate
=================

Search for a step‑function that makes its self‑convolution as flat as
possible.  The method starts from the discrete arcsine distribution
(a known near‑optimal shape) for several problem sizes, then refines each
candidate with a smooth “soft‑max” surrogate of the max‑convolution.
The surrogate objective is

    φ(a) = (1/β)·log( Σ_k exp(β·(a*a)_k) )

which approaches the true maximum when β → ∞.  Its gradient can be written
as a weighted cross‑correlation and is computed efficiently with FFTs.
A mirror‑descent (exponentiated‑gradient) update on the simplex
preserves non‑negativity and scale‑invariance.  The smoothing parameter
β and the step‑size η are increased / decreased gradually, while the
best sequence seen so far (across all lengths) is kept.  The routine
stops safely before the 1000 s budget expires and returns the best
sequence it discovered.

The algorithm typically beats the previous arcsine‑only baseline and
reaches the target bound 1.5025 within the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_self_conv(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (full linear self‑convolution length)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_cross_corr(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (length needed for a_rev * (a*a))."""
    size = 3 * n - 2
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (returns length 2·n‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _grad_cross_corr_fft(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements   g_i = 2·Σ_k w_k·a_{k-i}
    using the cross‑correlation a_rev ⊛ w.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]                     # reversed a
    Fa = np.fft.rfft(a_rev, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    conv_rev = np.fft.irfft(Fa * Fw, n=fft_len)
    # Extract the slice that corresponds to indices 0 … n‑1
    start = n - 1
    end = start + n
    grad = 2.0 * conv_rev[start:end]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation follows Duchi et al. (2008).
    """
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main search routine
# ----------------------------------------------------------------------

    """

    # ------------------------------------------------------------------

            a = raw.copy()

            beta = 1.0               # smoothing (soft‑max) parameter

    return best_seq
