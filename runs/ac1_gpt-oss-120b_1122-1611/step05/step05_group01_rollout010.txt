def _project_simplex(v):
    # returns projection onto simplex {x>=0, sum x = 1}
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate
=================

Implements a multi‑stage search for a non‑negative step‑function that minimizes
``evaluate_sequence``.  The algorithm proceeds as follows:

1. **Baseline / arcsine sweep** – try several lengths (up to 8192) and a grid of
   exponents α in the family  a_i ∝ 1/((i+½)(n‑½‑i))^α .  The best of these
   candidates is kept.

2. **Smooth max‑gradient descent** – starting from the best candidate, minimise
   a differentiable surrogate of the max‑convolution:
   ``smooth_max_ε(c) = ε·log(∑_k exp(c_k/ε))``.
   Its gradient can be written as a correlation of the current step‑function
   with a soft‑max weight vector, which we compute efficiently by FFT.
   After each gradient step we project back onto the probability simplex
   (non‑negative entries that sum to one).

3. **Occasional random jitter** and a tiny amount of exponentiated‑gradient
   refinement are used to escape plateaus.

The whole routine stops a few tenths of a second before the allotted time
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – must be defined at the top level (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a real vector using the FFT.
    Returns the full convolution (length 2·len(a)‑1).
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using the FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    size = len(x) + len(y) - 1
    fft_len = _next_pow2(size)
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[:size]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0, ∑ w = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho where the threshold is positive
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # stop a little before the deadline to avoid edge‑case termination
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0)  Keep the previous construction (if any) as a baseline
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(map(float, globals()["height_sequence_1"]))
            sc = evaluate_sequence(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"[{time.monotonic() - start_time:6.2f}s] baseline score = {sc:.6f} "
                      f"(length {len(cand)})")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1)  Deterministic arcsine‑family sweep (moderate lengths)
    # --------------------------------------------------------------
    # Lengths – up to 8192 give a reasonable trade‑off between quality
    # and the cost of the exact ``evaluate_sequence`` call.
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    # Explore a finer grid of exponents around the theoretical value 0.5
    alphas = np.linspace(0.40, 0.60, 21)   # step of 0.01

    for n in lengths:
        if time.monotonic() > deadline:
            break
        i = np.arange(n, dtype=np.float64)
        for alpha in alphas:
            if time.monotonic() > deadline:
                break
            raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
            # Normalise to unit sum – the score is scale‑invariant.
            raw /= raw.sum()
            seq = raw.tolist()
            sc = evaluate_sequence(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (arcsine) score = {sc:.6f} "
                      f"(n={n}, α={alpha:.3f})")

    # --------------------------------------------------------------
    # 2)  Smooth‑max gradient descent on the best candidate found so far
    # --------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=np.float64)
        n = a.size
        # pre‑compute FFT length for the self‑convolution (used every iteration)
        fft_len_self = _next_pow2(2 * n - 1)

        # hyper‑parameters for the descent
        eta = 0.3                     # initial step size
        eps_decay = 0.995             # slowly tighten the soft‑max approximation
        eps_factor = 0.5              # initial factor for epsilon relative to the peak‑gap

        # gradient descent loop
        while time.monotonic() < deadline:
            # ---------- compute self‐convolution ----------
            conv = _conv_self(a, fft_len_self)            # length 2·n‑1
            max_val = float(conv.max())
            avg_val = float(conv.mean())

            # ---------- smooth max parameters ----------
            gap = max_val - avg_val                       # positive unless already flat
            eps = max(gap * eps_factor, 1e-12)            # keep eps > 0

            # soft‑max weights (numerically stable)
            shifted = (conv - max_val) / eps              # ≤ 0, max element becomes 0
            w = np.exp(shifted)
            w /= w.sum()                                  # w now forms a probability vector

            # ---------- gradient of the smooth max ----------
            # grad_i = 2 * Σ_k w_k * a_{k‑i}
            a_rev = a[::-1]
            corr_full = _conv_general(w, a_rev)           # length 3·n‑2
            grad = 2.0 * corr_full[n - 1: n - 1 + n]      # slice of length n

            # ---------- update with projected gradient ----------
            a_new = a - eta * grad
            a = _project_simplex(a_new)                    # enforce a≥0, Σa = 1

            # ---------- occasional jitter to escape plateaus ----------
            if rng.random() < 0.005:
                jitter = np.exp(rng.normal(0.0, 0.04, size=n))
                a = _project_simplex(a * jitter)

            # ---------- decay parameters ----------
            eta *= 0.9995
            eps_factor *= eps_decay

            # ---------- periodic evaluation ----------
            if rng.random() < 0.03:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] smooth‑descent best = {sc:.6f}")

            # safety break
            if time.monotonic() - start_time > budget_s - 0.5:
                break

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if best_seq is None:

        best_score = evaluate_sequence(best_seq)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
