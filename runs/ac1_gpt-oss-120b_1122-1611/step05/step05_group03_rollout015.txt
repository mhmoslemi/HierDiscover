"""
Simple parametric search for a low‑autocorrelation step function.

The evaluation 2·n·max(conv)/sum² is invariant under scaling, therefore we
work with sequences that sum to one.  For a monotone power law
    a_i ∝ (i+1)^p   (0 ≤ i < n)
the convolution can be computed via a single FFT.  The score as a function
of the exponent p is smooth and unimodal in the range where the optimum
lies (≈0.2–0.6 for the current data).  We therefore perform a coarse grid
search followed by a fine ternary‑like refinement around the best exponent.
The best sequence found is returned.

The routine respects the 1000 s budget (the whole search finishes in a
few seconds even for n≈4000) and works without any file or network IO.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _conv_score(a: np.ndarray) -> float:
    """
    Compute the value returned by ``evaluate_sequence`` for a non‑negative
    sequence that already sums to one.  The function directly evaluates

        2·n·max(conv)

    where ``conv`` is the linear self‑convolution of ``a``.
    """
    n = a.size
    # ``np.convolve`` switches to an FFT implementation for large arrays,
    # which is fast enough for our problem sizes.
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def _make_powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """
    Return a probability vector of length ``n`` with components proportional
    to ``(i+1)**p``.  The result is guaranteed to be non‑negative and sum to 1.
    """
    # Using ``np.arange`` avoids Python loops and gives a vectorised power.
    w = np.arange(1, n + 1, dtype=np.float64) ** p
    total = w.sum()
    # Guard against degenerate p that would produce a zero‑sum (should not happen).
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / total


def _refine_exponent(n: int, p_center: float, radius: float,
                    steps: int = 9) -> (float, np.ndarray, float):
    """
    Local refinement around ``p_center``.  ``radius`` defines the half‑width of
    the search interval and ``steps`` the number of points sampled (odd, so that
    the centre is evaluated).  Returns the best exponent, its sequence and
    the associated score.
    """
    best_score = np.inf
    best_seq = None
    best_p = p_center
    # Build a tiny symmetric grid around ``p_center``.
    for i in range(steps):
        p = p_center - radius + 2 * radius * i / (steps - 1)
        seq = _make_powerlaw_sequence(n, p)
        score = _conv_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_p = p
    return best_p, best_seq, best_score
