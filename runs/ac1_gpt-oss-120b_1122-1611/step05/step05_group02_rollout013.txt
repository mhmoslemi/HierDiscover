def build_A_matrix(a, shifts):
    # a: numpy array length n
    # shifts: list of integers shift indices (0 to 2n-2)
    # returns A matrix shape (len(shifts), n) as dense np.ndarray
    n = len(a)
    rows = []
    for k in shifts:
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # For j = i_range, coefficient a_{k - i}
        # Actually for each j (index in b), coefficient = a_{k-j} if k-j in [0,n-1].
        # Equivalent to for j from i_start to i_end: row[j] = a[k - j]
        j_range = np.arange(i_start, i_end+1)
        row[j_range] = a[k - j_range]
        rows.append(row)
    A = np.vstack(rows)   # shape (len(shifts), n)
    return A
```

"""
Hybrid optimizer for the worst self‑convolution problem.

The algorithm works on a probability vector a (non‑negative, sum = 1) and
minimises  M = max_k (a*a)[k]  which directly minimises the evaluation
function 2·n·M.  It combines three ideas:

1.  A soft‑max (tempered) sub‑gradient descent on the convolution max.
    The gradient is a weighted sum of the gradients of the K largest
    convolution entries, which gives a smoother descent direction than
    the pure max‑gradient used previously.

2.  Periodic linear‑programming refinement on the *active* constraints.
    For the K shifts with largest convolution values we solve a small
    LP that maximises Σ b_i under   a·b ≤ M   (component‑wise).  The
    resulting vector b is mixed into a with a small step size and thus
    pushes a towards a flatter convolution.

3.  Simple stochastic moves (pairwise mass transfers, block
    perturbations) and restarts to escape plateaus.

All heavy work (convolution) is performed with FFT, the LP uses only
the K active shifts so the matrix is tiny.  The routine stops after the
budget (default 1000 s) and returns the best vector found.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
        else:

    n = a.size
    fft_len = _fft_len_for_n(n)

    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    max_conv = float(conv.max())
    cur_score = 2.0 * n * max_conv
    best = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
