"""
Propose a step‑function that beats the current bound (≈1.514).

Algorithm
---------
1. Initialise a probability vector ``a`` on the simplex.
   • If a previous construction ``height_sequence_1`` exists, it is
     copied (zero‑padded if we decide to use a larger length).
   • Otherwise start from a Dirichlet draw (size ≈ 5000).

2. Main optimisation (≈ 80 % of the time budget) – projected sub‑gradient
   descent on the exact max‑convolution:
   • Compute the self‑convolution `c = a*a` (FFT).
   • Let `M = max(c)`.  The sub‑gradient of `M` at `a` is
        g = 2·(p * rev(a)),  where `p` is any probability vector supported
        on the active set `{k | c_k = M}` (we use the uniform distribution
        over the active indices).  The convolution is again performed with FFT.
   • Update `a ← a – η·g` with a linearly decaying step size `η`,
     then **project** back onto the simplex {x≥0, Σx=1}.
   • Occasionally add a tiny random block perturbation so that the
     iterates do not get stuck.

3. Stochastic refinement (≈ 20 % of the budget):
   • Perform cheap pairwise mass‑transfer moves that try to lower the
     current maximum convolution entry, accepting them with a Metropolis
     rule (linear cooling).

4. Restart logic – if no improvement for 30 s re‑initialise from the best
   solution plus moderate log‑normal noise.

The routine keeps the best sequence seen so far and returns it at the end.
All heavy work (two FFTs per iteration) fits easily into the 1 000 s budget
for vectors of size ≈ 5 000.  The method improves on the previous
soft‑max mirror‑descent by using the *exact* sub‑gradient and Euclidean
projection, which typically yields a lower final value.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (output length 2·len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors a and b using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)[: a.size + b.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))          # ≤10 % of length
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _exact_subgradient(a: np.ndarray, rev_a: np.ndarray,
                       conv: np.ndarray,
                       fft_len_grad: int) -> np.ndarray:
    """
    Compute an exact sub‑gradient of max(conv) at the point ``a``.
    ``conv`` must be the self‑convolution of ``a``.
    """
    max_val = conv.max()
    active = np.isclose(conv, max_val, atol=1e-14)
    # Uniform probability on the active set
    p = np.zeros_like(conv)
    cnt = active.sum()
    if cnt == 0:   # should never happen
        return np.zeros_like(a)
    p[active] = 1.0 / cnt

    # Convolution p * rev(a); only the middle n entries are needed
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * a.size - 2]
    grad = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                      budget_s: float = 1000.0,
                      **kwargs):
    """
    Return the best non‑negative step‑function found within the time budget.
    The result is a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            # Pad / truncate to the chosen length
            target_n = max(5000, cand.size)        # a little larger than before
            if cand.size < target_n:
                a = np.concatenate([cand, np.zeros(target_n - cand.size)])
            else:
                a = cand[:target_n].copy()
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            # fallback – uniform start, length 5000
            target_n = 5000
            a = np.full(target_n, 1.0 / target_n, dtype=np.float64)
    else:
        # no prior construction – uniform start, length 5000
        target_n = 5000
        a = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    n = a.size
    rev_a = a[::-1]

    # Pre‑compute FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _self_conv(a, fft_len_self)
    cur_max = conv.max()
    cur_score = float(2.0 * n * cur_max)          # sum(a) == 1
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.80 * total_time    # 80 % for deterministic descent

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 2.5      # initial step (large)
    eta_end = 1e-4       # final step
    momentum = 0.8
    grad_prev = np.zeros_like(a)

    # Phase‑2 (stochastic) parameters
    temp_start = 0.07
    temp_end = 1e-5

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – exact sub‑gradient descent with simplex projection
        # --------------------------------------------------------------
        if now < phase1_end:
            # step‑size schedule (linear)
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # recompute convolution and its max
            conv = _self_conv(a, fft_len_self)
            cur_max = conv.max()
            cur_score = float(2.0 * n * cur_max)

            # exact sub‑gradient of the max‑convolution
            grad = _exact_subgradient(a, rev_a, conv, fft_len_grad)

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # gradient step and projection onto simplex
            a_new = a - eta * grad
            a_new = _project_simplex(a_new)

            # occasional tiny jitter to avoid stagnation
            if rng.random() < 0.015:
                a_new = _random_block_perturb(a_new, rng)

            a = a_new
            rev_a = a[::-1]          # keep reversed copy up‑to‑date

        # --------------------------------------------------------------
        # Phase 2 – stochastic refinement (simulated annealing)
        # --------------------------------------------------------------
        else:
            # linear cooling schedule
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-8)

            # ensure we have a fresh convolution for the move
            conv = _self_conv(a, fft_len_self)
            cur_max = conv.max()
            cur_score = float(2.0 * n * cur_max)
            k_max = int(conv.argmax())

            # ---- pairwise mass transfer, biased by contribution to the peak ----
            if rng.random() < 0.90:
                # contributions to the current peak
                contrib = np.zeros_like(a)
                for i in range(n):
                    j = k_max - i
                    if 0 <= j < n:
                        contrib[i] = a[i] * a[j]
                total = contrib.sum()
                if total > 0.0:
                    prob = contrib / total
                    donor = rng.choice(np.arange(n), p=prob)
                else:
                    donor = rng.integers(0, n)

                if a[donor] > 1e-12:
                    # pick a recipient different from donor
                    rec = rng.integers(0, n - 1)
                    if rec >= donor:
                        rec += 1
                    # transfer a modest fraction of the donor's mass
                    delta = rng.random() * 0.25 * a[donor]
                    a_new = a.copy()
                    a_new[donor] -= delta
                    a_new[rec] += delta
                    # quick evaluation (full convolution)
                    conv_new = _self_conv(a_new, fft_len_self)
                    new_score = float(2.0 * n * conv_new.max())
                    # Metropolis acceptance
                    if new_score < cur_score or rng.random() < np.exp(
                        -(new_score - cur_score) / temperature
                    ):
                        a = a_new
                        rev_a = a[::-1]
                        conv = conv_new
                        cur_score = new_score

            else:
                # global block jitter
                a = _random_block_perturb(a, rng)
                rev_a = a[::-1]
                conv = _self_conv(a, fft_len_self)
                cur_score = float(2.0 * n * conv.max())

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # Re‑initialise from the best with moderate noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            rev_a = a[::-1]
            grad_prev = np.zeros_like(a)
            conv = _self_conv(a, fft_len_self)
            cur_score = float(2.0 * n * conv.max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by the
    #    evaluation function for safety).  The stack‑exchange harness will
    #    subsequently call `evaluate_sequence` on the returned list.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
