def make_seq(p, alpha, beta, n):
    i = np.arange(n, dtype=np.float64)
    half = n // 2
    # compute distance from nearest edge: d = np.minimum(i, n-1-i)
    d = np.minimum(i, n-1 - i)
    seq = (d + alpha) ** (-p) * np.exp(-beta * d / n)
    # ensure nonneg
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= total
    return seq
```

"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Algorithm
---------
1. Choose a problem size `n` (default ~3800) and pre‑compute the FFT
   length for fast linear convolutions.
2. Keep the best candidate seen so far (initialised from the provided
   `height_sequence_1` if any, otherwise from the classical √‑difference
   shape).
3. **Randomised parametric search** – draw thousands of simple
   parametric families (power‑law, half‑sine, Gaussian window).  For each
   generated vector the objective  `2·n·max(a*a)` is evaluated using an
   FFT‑based convolution.  The best vector is retained.
4. **Mirror‑descent fine tuning** – starting from the best vector, run a
   projected exponentiated‑gradient (Adam‑style) optimiser on the true
   L∞‑objective.  At every iteration the active set of convolution indices
   whose value is within 0.1 % of the current maximum is used to build a
   sub‑gradient.  The iterates stay on the probability simplex
   (∑aᵢ=1) and are periodically perturbed by random block‑multipliers
   to escape shallow minima.
5. Stop when the time budget (≈1000 s) is exhausted and return the best
   sequence found.  All entries are clipped to the limits imposed by
   `evaluate_sequence` (0 ≤ aᵢ ≤ 1000).

The combination of a cheap global search over a low‑dimensional
parameter space and a fast local mirror‑descent often finds solutions
substantially better than the previous score of 1.513856, pushing the
objective below the target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – smallest FFT length that can host a linear convolution.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two length sufficient for a*a (size 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Helper 2 – linear convolution via FFT (real‑valued input).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a, result length = 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3 – compute the objective used by evaluate_sequence().
# ----------------------------------------------------------------------
def _score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a)  (a is assumed to sum to 1)."""
    n = a.size
    conv = _conv_self(a, fft_len)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Parametric families.
# ----------------------------------------------------------------------
def _shape_power(p: float, alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone power‑law with optional exponential tail, normalised."""
    i = np.arange(n, dtype=np.float64)
    # avoid division by zero
    alpha = max(alpha, 1e-12)
    seq = (i + alpha) ** (-p)
    if beta != 0.0:
        seq = seq * np.exp(-beta * i / n)
    seq = np.clip(seq, 0.0, None)
    s = seq.sum()
    if s == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= s
    return seq


def _shape_sine(power: float, n: int) -> np.ndarray:
    """Half‑sine window raised to a power, normalised."""
    x = np.arange(n, dtype=np.float64) + 0.5
    seq = np.sin(np.pi * x / (n + 1.0))
    seq = np.clip(seq, 0.0, None) ** power
    s = seq.sum()
    if s == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= s
    return seq


def _shape_gaussian(sigma: float, n: int) -> np.ndarray:
    """Centered Gaussian window, normalised."""
    center = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    seq = np.clip(seq, 0.0, None)
    s = seq.sum()
    if s == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= s
    return seq


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine combines a coarse random search over
    a few parametric families with a fast mirror‑descent refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 2) Initialise best candidate (use a supplied construction if any)
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # try to use a previously discovered construction
    try:
        if "height_sequence_1" in globals():
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                    val = _score(cand, fft_len)
                    best_seq = cand.copy()
                    best_val = val
                    print(f"[0.00s] start from height_sequence_1 score {best_val:.6f}")
    except Exception:
        pass

    # fallback: the classic √‑difference shape
    i = np.arange(n, dtype=np.float64)
    sqrt_diff = np.sqrt(i + 1.0) - np.sqrt(i)
    sqrt_diff /= sqrt_diff.sum()
    val_sqrt = _score(sqrt_diff, fft_len)
    if val_sqrt < best_val:
        best_seq = sqrt_diff.copy()
        best_val = val_sqrt
        print(f"[0.00s] baseline √‑difference score {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Coarse random search over simple families
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin
    families = ("power", "sine", "gaussian")
    it = 0
    while time.monotonic() < deadline:
        # pick a family uniformly at random
        fam = rng.choice(families)

        if fam == "power":
            # log‑uniform for p, exponential for alpha & beta
            p = np.exp(rng.uniform(np.log(0.25), np.log(2.5)))   # roughly 0.25–2.5
            alpha = rng.exponential(scale=15.0)                # >0
            beta = rng.exponential(scale=2.0)                 # tail strength
            seq = _shape_power(p, alpha, beta, n)

        elif fam == "sine":
            power = rng.uniform(0.5, 2.0)                      # shape exponent
            seq = _shape_sine(power, n)

        else:  # gaussian
            # sigma in [0.05·n , 0.5·n]
            sigma = rng.uniform(0.05 * n, 0.5 * n)
            seq = _shape_gaussian(sigma, n)

        val = _score(seq, fft_len)

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            now = time.monotonic() - start_time
            print(f"[{now:6.2f}s] new best from {fam:<8}  score {best_val:.6f}")

        it += 1
        # occasional cheap time‑check (every 5000 iterations)
        if it % 5000 == 0 and time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # 4) Fine‑grained mirror‑descent (exponentiated‑gradient + Adam)
    # --------------------------------------------------------------
    # Initialise Adam moments
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    while time.monotonic() < deadline:
        step += 1
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()

        # active set – indices within 0.1 % of the current maximum (max 12 entries)
        tol = cur_max * 0.999
        active = np.where(conv >= tol)[0]
        if active.size > 12:
            active = np.argsort(conv)[-12:]

        # build sub‑gradient of the (soft) max
        grad = np.zeros_like(a)
        if active.size:
            w = np.full(active.shape, 1.0 / active.size, dtype=np.float64)  # uniform weight
            for idx, weight in zip(active, w):
                lo = max(0, idx - (n - 1))
                hi = min(n - 1, idx)
                rng_i = np.arange(lo, hi + 1, dtype=np.int64)
                grad[rng_i] += weight * 2.0 * a[idx - rng_i]
        else:
            # fallback (should never happen)
            grad = 2.0 * a

        # Adam updates (on the log‑domain)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # learning‑rate schedule (slow decay)
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        eta = 0.8 * (1.0 - elapsed / total) + 0.03   # from ≈0.8 down to ≈0.03

        # exponentiated‑gradient step
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional stochastic block perturbation (helps escape plateaus)
        if rng.random() < 0.012:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.25))
            a[start:start + width] *= factor
            a = a / a.sum()

        # evaluate current score
        cur_score = 2.0 * n * np.max(_conv_self(a, fft_len))
        if cur_score < best_val - 1e-12:
            best_val = cur_score
            best_seq = a.copy()
            now = time.monotonic() - start_time
            print(f"[{now:6.2f}s] improvement via descent score {best_val:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (respect evaluate_sequence clipping)
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
