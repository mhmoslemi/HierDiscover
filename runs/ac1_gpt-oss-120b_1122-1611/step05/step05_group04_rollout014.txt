"""candidate proposer – hybrid LP + smooth‑max + SA

The search proceeds in three stages:

1. **Shape seeding** – generate a few dozen simple parametric families
   (uniform, Beta‑shapes, sine windows, random mixtures).  The best
   candidate becomes the start point.

2. **LP refinement** – with the current vector `a` fix its convolution
   bound `M = max(a*a)`.  Solve a small linear programme

        minimise   M
        s.t.       a ⊛ g ≤ M   (only the K largest convolution entries)
                   Σ g = 1 , g ≥ 0

   The LP produces a new direction `g`.  We mix `a ← (1‑t)·a + t·g`
   (projected onto the simplex) and keep the improvement.  This step
   is cheap because only the K≈30 most critical constraints are used.

3. **Continuous optimisation** – a smooth‑max (soft‑max) mirror‑descent
   on the probability simplex (exponentiated‑gradient with momentum) for
   roughly the first 55 % of the remaining budget, followed by a classic
   simulated‑annealing local search (pairwise mass transfers) for the
   rest.  Both phases respect non‑negativity, keep the vector on the
   simplex and constantly record the best objective value.

The objective returned by ``evaluate_sequence`` (with the sum normalised
to 1) is `2·n·max(conv)`.  The algorithm keeps the best sequence found
within the 1000 s budget and returns it as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT.  Returns the first (2*n‑1) entries."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)                     # clip tiny negatives
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Derivation: grad_j = 2·Σ_i a_i·w_{i+j}.
    Implemented as a correlation a ⋆ w using a single FFT.
    """
    n = a.size
    Lw = w.size                     # = 2*n‑1
    wr = w[::-1]                    # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)   # length for correlation a ⋆ wr

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)         # mapping to obtain correlation result
    grad = 2.0 * conv_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


# ----------------------------------------------------------------------
# Linear‑programming refinement (uses only the K largest convolution entries)
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray,
               top_k: int,
               rng: np.random.Generator,
               fft_len: int) -> tuple[np.ndarray, float] | None:
    """
    Solve a tiny LP that minimises the maximal convolution value of a⊛g
    over the `top_k` currently largest convolution indices.
    Returns the new direction `g` and the attained bound `M_opt`.
    If the LP fails, returns None.
    """
    n = a.size
    conv = _self_conv_fft(a, fft_len)
    # indices of the K largest entries (ties are irrelevant)
    k_idx = np.argpartition(conv, -top_k)[-top_k:]

    # Build sparse matrix for the inequality a ⊛ g ≤ M
    rows, cols, data = [], [], []
    for row, k in enumerate(k_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(row)
            cols.append(j)          # column for g_j
            data.append(a[i])       # coefficient a_i
        # coefficient for the auxiliary variable M (last column)
        rows.append(row)
        cols.append(n)               # M is at position n
        data.append(-1.0)

    A_ub = csr_matrix((data, (rows, cols)), shape=(top_k, n + 1))
    b_ub = np.zeros(top_k)

    # Equality: sum(g) = 1
    A_eq = np.hstack((np.ones((1, n)), np.zeros((1, 1))))  # dense is fine
    b_eq = np.array([1.0])

    # Bounds: g_j ≥ 0, M ≥ 0
    bounds = [(0, None)] * n + [(0, None)]

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Solve – use HiGHS (fast, works with sparse A_ub)
    res = linprog(c,
                  A_ub=A_ub,
                  b_ub=b_ub,
                  A_eq=A_eq,
                  b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 8})
    if not res.success:
        return None
    g = np.clip(res.x[:n], 0.0, None)
    g = _project_onto_simplex(g)           # guard against numeric drift
    M_opt = float(res.x[-1])
    return g, M_opt


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step function that minimises the evaluation metric."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size & initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a0 = np.clip(cand0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = 4000
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(2 * n - 1)          # length for all FFT‑based convolutions

    # ------------------------------------------------------------------
    # Helper for scoring
    # ------------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        conv = _self_conv_fft(vec, fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 1) Seed generation – cheap parametric families
    # ------------------------------------------------------------------
    best = a0.copy()
    best_score = _score(best)
    print(f"[0.00s] initial seed score = {best_score:.6f}")

    # uniform (already a0 but re‑evaluate in case a0 differs)
    uni = np.full(n, 1.0 / n)
    sc = _score(uni)
    if sc < best_score:
        best, best_score = uni, sc
        print(f"[0.00s] uniform improved to {best_score:.6f}")

    # Beta‑type shapes: a_i ∝ (i+½)^{α‑1}·(n‑½‑i)^{α‑1}
    alphas = np.linspace(0.2, 2.0, 19)
    idx = np.arange(n, dtype=np.float64) + 0.5
    rev = (n - 0.5) - idx

    for α in alphas:
        raw = (idx ** (α - 1)) * (rev ** (α - 1))
        raw = np.maximum(raw, 0.0)
        vec = raw / raw.sum()
        sc = _score(vec)
        if sc < best_score:
            best, best_score = vec, sc
            print(f"[0.00s] beta α={α:.2f} improved to {best_score:.6f}")

    # random beta shapes
    for _ in range(30):
        α = rng.uniform(0.2, 2.0)
        raw = (idx ** (α - 1)) * (rev ** (α - 1))
        raw = np.maximum(raw, 0.0)
        vec = raw / raw.sum()
        sc = _score(vec)
        if sc < best_score:
            best, best_score = vec, sc
            print(f"[0.00s] random β={α:.3f} improved to {best_score:.6f}")

    # sine window (positive, symmetric)
    sin_vec = np.sin(np.pi * idx / n)
    sin_vec = sin_vec / sin_vec.sum()
    sc = _score(sin_vec)
    if sc < best_score:
        best, best_score = sin_vec, sc
        print(f"[0.00s] sine window improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Main optimisation loop – split in two phases
    # ------------------------------------------------------------------
    cur_a = best.copy()
    cur_conv = _self_conv_fft(cur_a, fft_len)
    cur_score = 2.0 * n * cur_conv.max()
    last_improve = time.monotonic()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    phase_split = 0.55          # fraction of remaining time spent on gradient phase
    grad_prev = np.zeros_like(cur_a)

    # hyper‑parameters for the smooth‑max phase
    eta_start, eta_end = 1.5, 0.02
    beta_start, beta_end = 0.3, 3000.0
    momentum = 0.85
    block_perturb_prob = 0.025

    # SA hyper‑parameters (second phase)
    temp_start = best_score * 0.03
    temp_end = best_score * 1e-5
    lp_interval = 120.0        # seconds between LP refinements (if stagnating)
    next_lp_time = start_time + lp_interval

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # --------------------------------------------------------------
        # Phase decision
        # --------------------------------------------------------------
        if frac < phase_split:
            # ------------------- Smooth‑max Mirror Descent ---------------
            conv = cur_conv
            cur_max = conv.max()

            # keep global best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = cur_a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] gradient best = {best_score:.6f}")

            # schedules (linear interpolation)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # soft‑max weights (stable)
            exp_terms = np.exp(beta * (conv - cur_max))
            w = exp_terms / exp_terms.sum()

            # gradient (with momentum)
            grad = _gradient_from_weights(cur_a, w)
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient update
            cur_a = cur_a * np.exp(-eta * grad)
            cur_a = np.clip(cur_a, 1e-15, None)
            cur_a = _project_onto_simplex(cur_a)

            # recompute convolution
            cur_conv = _self_conv_fft(cur_a, fft_len)
            cur_score = 2.0 * n * cur_conv.max()

            # occasional random block perturbation
            if rng.random() < block_perturb_prob:
                cur_a = _random_block_perturb(cur_a, rng)
                cur_conv = _self_conv_fft(cur_a, fft_len)
                cur_score = 2.0 * n * cur_conv.max()

            # stagnation handling – try LP refinement if no improvement for a while
            if now >= next_lp_time:
                lp_res = _lp_refine(cur_a, top_k=30, rng=rng, fft_len=fft_len)
                next_lp_time = now + lp_interval
                if lp_res is not None:
                    g, M_opt = lp_res
                    # mix current vector towards the LP direction
                    t = 0.35
                    new_a = (1.0 - t) * cur_a + t * g
                    new_a = _project_onto_simplex(new_a)
                    new_conv = _self_conv_fft(new_a, fft_len)
                    new_score = 2.0 * n * new_conv.max()
                    if new_score < cur_score:
                        cur_a, cur_conv, cur_score = new_a, new_conv, new_score
                        if cur_score < best_score:
                            best_score = cur_score
                            best = cur_a.copy()
                            last_improve = now
                            print(f"[{elapsed:6.2f}s] LP‑mix improved to {best_score:.6f}")

        else:
            # ------------------- Simulated‑Annealing Refinement ----------
            # Linear temperature schedule
            t_frac = (now - start_time - phase_split * total) / ((1.0 - phase_split) * total)
            T = temp_start * (1.0 - t_frac) + temp_end * t_frac

            # propose a random pairwise mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                # move a random fraction of the donor mass
                max_delta = cur_a[i]
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a2 = cur_a.copy()
                    a2[i] -= delta
                    a2[j] += delta
                    a2[i] = max(a2[i], 0.0)          # protect against tiny negatives
                    conv2 = _self_conv_fft(a2, fft_len)
                    score2 = 2.0 * n * conv2.max()
                    if score2 < cur_score or rng.random() < math.exp(-(score2 - cur_score) / max(T, 1e-12)):
                        cur_a, cur_conv, cur_score = a2, conv2, score2
                        if cur_score < best_score:
                            best_score = cur_score
                            best = cur_a.copy()
                            last_improve = now
                            print(f"[{elapsed:6.2f}s] SA improved to {best_score:.6f}")

            # occasional block perturbation in SA phase
            if rng.random() < 0.015:
                cur_a = _random_block_perturb(cur_a, rng)
                cur_conv = _self_conv_fft(cur_a, fft_len)
                cur_score = 2.0 * n * cur_conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best = cur_a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] SA block perturb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best sequence (clip to allowed range of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
