```
def step_reduce_peak(a, conv, rng):
    n = a.size
    # choose k as one of top peaks
    # find top indexes with conv near max
    max_val = conv.max()
    # find indices where conv >= max_val - tol
    tol = 0.01 * max_val  # 1% tolerance
    top_idx = np.where(conv >= max_val - tol)[0]
    if top_idx.size > 1:
        k = rng.choice(top_idx)
    else:
        k = int(np.argmax(conv))
    # compute partner array
    idx = np.arange(n)
    valid = (k - idx >= 0) & (k - idx < n)
    partner = np.zeros_like(a)
    partner[valid] = a[k - idx[valid]]
    donor_mask = a > 1e-12
    donorscore = partner * donor_mask
    if donorscore.max() <= 0:
        return a, conv, False
    donor = int(np.argmax(donorscore))
    # receiver candidates: minimal partner
    min_partner = partner.min()
    receiver_candidates = np.where(partner == min_partner)[0]
    # exclude donor
    if donor in receiver_candidates:
        receiver_candidates = receiver_candidates[receiver_candidates != donor]
    if receiver_candidates.size == 0:
        # choose random other index
        receiver = (donor + rng.integers(1, n)) % n
    else:
        receiver = int(rng.choice(receiver_candidates))
    # delta: as earlier but maybe adapt to product diff
    # compute difference in contribution
    diff = partner[donor] - partner[receiver]
    if diff <= 0:
        # no improvement possible, return
        return a, conv, False
    # choose delta <= a[donor] * 0.3 * (diff / partner[donor]) maybe
    max_frac = 0.3
    max_delta = a[donor] * max_frac
    delta = rng.random() * max_delta
    # apply transfer
    new_conv = _apply_transfer(a, conv, donor, receiver, delta)
    new_max = new_conv.max()
    # accept if new max reduced
    if new_max < conv.max():
        # update a and conv
        a[donor] -= delta
        a[receiver] += delta
        return a, new_conv, True
    else:
        return a, conv, False
```

It combines three ideas that improve on the previous greedy code:

1. **Diverse random starts** – a mixture of uniform, sparse Dirichlet and a copy of the best known solution (`height_sequence_1`).  
2. **Multi‑peak targeted mass transfers** – instead of always moving mass for a single maximal convolution entry, the algorithm picks one of the top‑few peaks (so it can flatten several “hot spots’’ at once).  
3. **Escaping local minima** – occasional global jitter, block‑wise log‑normal perturbations, symmetric averaging and short‑term restarts (when no improvement for a while) act like a lightweight simulated‑annealing scheme.

The routine runs until the allocated wall‑clock budget (≈ 1000 s) is exhausted and returns the best sequence it has found.

```python
"""
propose_candidate – stochastic multi‑restart search for a low‑max convolution step function.

The algorithm works on the simplex (non‑negative entries, sum = 1) and tries
to minimise 2·n·max(conv(a,a)).  It uses several independent restarts with
different initialisations (uniform, sparse Dirichlet, and optionally the
previous best solution) and iteratively performs *targeted mass transfers*:
for a convolution index that belongs to the current top‑k peaks we move a
fraction of mass from a “donor’’ with a large partner value to a “receiver’’
with a small partner value.  The transfer is accepted only if it reduces the
global maximum of the convolution.  Random block‑perturbations, symmetric
averaging and occasional jitter provide escape mechanisms.  The best vector
encountered over the whole budget is returned (clipped to the limits that
`evaluate_sequence` enforces).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    return np.convolve(a, a)  # NumPy uses an efficient routine for large n


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    i: int,
                    j: int,
                    delta: float) -> np.ndarray:
    """
    Return the convolution after moving `delta` mass from index i to index j.
    The update follows the exact algebraic identity (a+Δ)*(a+Δ) = a*a
    + 2·a·Δ + Δ·Δ, where Δ has a +δ at j and –δ at i.
    """
    n = a.size
    new_conv = conv.copy()
    # linear contribution: -2·δ·shift(a,i) + 2·δ·shift(a,j)
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a
    # quadratic contribution of the transferred mass
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          max_width_frac: float = 0.2,
                          sigma: float = 0.12) -> np.ndarray:
    """
    Jitter a random contiguous block of `a` multiplicatively with a log‑normal
    factor, then renormalise onto the simplex.
    """
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the value of evaluate_sequence for a vector already normalised to sum=1."""
    return float(2.0 * n * conv.max())


def _pick_peak_index(conv: np.ndarray, rng: np.random.Generator,
                     top_fraction: float = 0.02) -> int:
    """
    Return an index belonging to the set of the largest convolution entries.
    `top_fraction` determines how many of the highest entries are considered.
    """
    n = conv.size
    # number of candidates – at least 1, at most 20
    k = max(1, min(20, int(top_fraction * n)))
    # argpartition gives the k‑largest values efficiently
    idx = np.argpartition(conv, -k)[-k:]
    return int(rng.choice(idx))


def _targeted_transfer(a: np.ndarray,
                       conv: np.ndarray,
                       rng: np.random.Generator) -> tuple[np.ndarray, np.ndarray, bool]:
    """
    Perform one mass‑transfer step aimed at reducing the maximum of the
    convolution.  The function chooses a peak index from the top k entries,
    selects a donor with a large partner value and a receiver with a small
    partner value, tries a random delta and accepts the move only if the
    global maximum improves.
    Returns the possibly updated `a`, the new convolution and a flag that
    signals whether an improvement occurred.
    """
    n = a.size

    # ---- pick a peak index (k) among the top‑few peaks ----
    k = _pick_peak_index(conv, rng, top_fraction=0.02)

    # partner[i] = a_{k-i} when it exists, else 0
    idx = np.arange(n)
    valid = (k - idx >= 0) & (k - idx < n)
    partner = np.zeros_like(a)
    partner[valid] = a[k - idx[valid]]

    # ---- donor: the index with the largest product a_i * partner_i ----
    donor_mask = a > 1e-14
    donor_score = partner * donor_mask.astype(float)
    if donor_score.max() <= 0.0:
        # no useful donor – abort this step
        return a, conv, False
    donor = int(np.argmax(donor_score))

    # ---- receiver: any index with a very small partner value ----
    min_part = partner.min()
    receiver_candidates = np.where(partner == min_part)[0]
    # make sure donor is not chosen
    if donor in receiver_candidates:
        receiver_candidates = receiver_candidates[receiver_candidates != donor]
    if receiver_candidates.size == 0:
        # fallback – pick a random different index
        receiver = (donor + rng.integers(1, n)) % n
    else:
        receiver = int(rng.choice(receiver_candidates))

    # ---- determine a step size ------------------------------------------------
    # the larger the difference partner[donor]‑partner[receiver] the larger the gain
    diff = partner[donor] - partner[receiver]
    if diff <= 0.0:
        return a, conv, False
    # limit the transferred fraction (larger when we are early in the run)
    # we use a simple linear schedule based on the current global max
    max_frac = 0.25
    max_delta = a[donor] * max_frac
    if max_delta < 1e-14:
        return a, conv, False
    delta = rng.random() * max_delta

    # ---- evaluate the candidate move -----------------------------------------
    new_conv = _apply_transfer(a, conv, donor, receiver, delta)
    new_score = _score_from_conv(new_conv, n)
    cur_score = _score_from_conv(conv, n)

    if new_score < cur_score:
        # accept the move
        a[donor] -= delta
        a[receiver] += delta
        return a, new_conv, True
    return a, conv, False


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **_kwargs):
    """
    Search the space of non‑negative step functions for the vector that
    minimises `evaluate_sequence`.  The routine respects the given time
    budget (≈ 1000 s) and returns the best sequence it discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # leave a small safety margin for the surrounding harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_a = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # Helper to (re)initialise a candidate vector
    # ------------------------------------------------------------------
    def _initialise(n: int) -> np.ndarray:
        """Return a random probability vector of length n."""
        # 1/3 of the time start from the previously found solution (if any)
        if "height_sequence_1" in globals() and rng.random() < 0.33:
            arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if arr.size != n:
                # resize by simple linear interpolation
                x_old = np.linspace(0, 1, arr.size)
                x_new = np.linspace(0, 1, n)
                arr = np.interp(x_new, x_old, arr)
            arr = np.clip(arr, 0.0, None)
            s = arr.sum()
            if s == 0.0:
                arr[:] = 1.0 / n
            else:
                arr /= s
            return arr

        # 2/3 of the time start from a sparse Dirichlet (Gamma) distribution
        # the concentration parameter is randomly sampled to vary sparsity.
        alpha = rng.uniform(0.05, 0.6)   # smaller → sparser
        vec = rng.gamma(alpha, 1.0, size=n)
        vec = np.clip(vec, 0.0, None)
        s = vec.sum()
        if s == 0.0:
            vec[:] = 1.0 / n
        else:
            vec /= s
        return vec

    # ------------------------------------------------------------------
    # Decide a set of problem sizes – larger n gives more freedom.
    # ------------------------------------------------------------------
    candidate_ns = [2048, 4096, 8192]

    # each size gets a few independent restarts
    for n in candidate_ns:
        # quick exit if the global time budget is exhausted
        if time.monotonic() >= deadline:
            break

        # estimate how many restarts we can afford for this n
        # (roughly 2–3 restarts for the smaller n, 1–2 for the biggest)
        restarts = 3 if n <= 4096 else 2

        for restart_idx in range(restarts):
            if time.monotonic() >= deadline:
                break

            a = _initialise(n)
            conv = _self_convolution(a)
            cur_score = _score_from_conv(conv, n)

            # keep a copy of the best state seen inside the current restart
            local_best_a = a.copy()
            local_best_score = cur_score
            last_improve = time.monotonic()

            # ------------------------------------------------------------------
            # Main inner optimisation loop
            # ------------------------------------------------------------------
            while time.monotonic() < deadline:
                # 1) try a targeted transfer that reduces the current peak(s)
                a, conv, improved = _targeted_transfer(a, conv, rng)
                if improved:
                    cur_score = _score_from_conv(conv, n)
                    if cur_score < local_best_score:
                        local_best_score = cur_score
                        local_best_a = a.copy()
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] n={n} restart={restart_idx} "
                              f"new global best {best_score:.6f}")

                    last_improve = time.monotonic()
                    continue  # successful move – go to the next iteration

                # 2) occasionally perform a cheap random block perturbation
                if rng.random() < 0.012:
                    a_pert = _random_block_perturb(a, rng)
                    conv_pert = _self_convolution(a_pert)
                    pert_score = _score_from_conv(conv_pert, n)
                    if pert_score < cur_score:
                        a, conv, cur_score = a_pert, conv_pert, pert_score
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                            elapsed = time.monotonic() - start_time
                            print(f"[{elapsed:6.2f}s] n={n} restart={restart_idx} "
                                  f"block‑perturb improved best {best_score:.6f}")
                        last_improve = time.monotonic()
                        continue

                # 3) small symmetric averaging (forces a≈reverse(a) and helps flatten)
                if rng.random() < 0.006:
                    a = (a + a[::-1]) * 0.5
                    a /= a.sum()
                    conv = _self_convolution(a)
                    cur_score = _score_from_conv(conv, n)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] n={n} restart={restart_idx} "
                              f"symmetric averaging improved best {best_score:.6f}")
                    last_improve = time.monotonic()
                    continue

                # 4) if we have been stuck for a while, perform a short restart
                if time.monotonic() - last_improve > 22.0:
                    # jitter the best solution seen in this restart
                    a = local_best_a.copy()
                    noise = np.exp(rng.normal(0.0, 0.17, size=n))
                    a = a * noise
                    a = np.clip(a, 0.0, None)
                    a /= a.sum()
                    conv = _self_convolution(a)
                    cur_score = _score_from_conv(conv, n)
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] n={n} restart={restart_idx} "
                          f"restart – score reset to {cur_score:.6f}")

            # after the inner loop we retain the best vector discovered in this
            # restart (it may already be the global best)
            if local_best_score < best_score:
                best_score = local_best_score
                best_a = local_best_a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] n={n} restart={restart_idx} "
                      f"restart‑final best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # If everything failed (should never happen) fall back to a uniform vector
    # ----------------------------------------------------------------------
    if best_a is None:
        n = 2048
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits that evaluate_sequence enforces and return as list
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a.tolist()]
```
