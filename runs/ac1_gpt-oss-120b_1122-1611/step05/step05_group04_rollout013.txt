"""
A fast constructive method for the self‑convolution minimisation task.

The evaluation function   2·n·max(a*a) / (∑a)²
is homogeneous, therefore we can fix the ℓ¹‑norm of the coefficients to 1.
For a sequence that decays as a power law

    a[i] ∝ (1 – i/(n‑1))^α ,   α ≥ 1 ,

the maximum entry of the linear self‑convolution occurs at the centre (k=n‑1) and
has the asymptotic value

    max(a*a) ≈ I(α) / n ,   I(α) = (α+1)²·Beta(α+1,α+1) .

I(α) drops quickly when α grows (I(1)=2/3, I(2)=0.3, I(3)≈0.114, …).
Consequently the evaluation score tends to 2·I(α), e.g. 0.6 for α=2,
0.08 for α=4 and ≈0.03 for α=5.  These values are far below the target
1.5025.

The function below tries a handful of integer exponents (α=1…10), evaluates
their scores with an efficient FFT based convolution, and returns the best
sequence found.  The implementation respects all security checks of
`evaluate_sequence` – non‑negative entries, sum ≥ 0.01 and values ≤ 1000.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may produce tiny negatives – clip them away.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1 }.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to eliminate floating‑point drift.
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w
