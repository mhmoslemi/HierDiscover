def _project_to_simplex(v):
    # v is numpy array
    # Duchi et al algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        # all components zero? then return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function by (soft‑max) gradient descent on the
∞‑norm of the self‑convolution, followed by simplex projection.
The algorithm works directly on the probability simplex
∑a_i=1, a_i≥0.  At each iteration a smooth approximation of the
max (log‑sum‑exp) is differentiated; the resulting gradient is a
convolution that can be computed with a single FFT.  After a
negative step the vector is projected back onto the simplex.
Temperature and step‑size are annealed, occasional jitter and
restarts avoid stagnation.  The routine starts from the previously
found best sequence (global ``height_sequence_1``) if available.
The returned list is clipped to the limits of ``evaluate_sequence``.
"""

import time
import math
import numpy as np


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _self_convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a using a FFT of length L."""
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)
    return conv[: 2 * a.size - 1]


def _gradient_of_softmax_conv(a: np.ndarray,
                              conv: np.ndarray,
                              temp: float,
                              L: int) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max‑value of the convolution.
    Returns a vector of length len(a).
    """
    n = a.size
    maxval = conv.max()
    # soft‑max weights (only indices close to the max get weight ≈1)
    w = np.exp((conv - maxval) / temp)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    rev_a = a[::-1]
    w_fft = np.fft.rfft(w, n=L)
    rev_a_fft = np.fft.rfft(rev_a, n=L)
    conv_wr = np.fft.irfft(w_fft * rev_a_fft, n=L)[: 2 * n - 1]

    # gradient = 2 * (w * a_rev)_{n-1 … n-1+n-1}
    start = n - 1
    grad = 2.0 * conv_wr[start : start + n]
    return grad


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value of ``evaluate_sequence`` when ∑a = 1."""
    return 2.0 * n * conv.max()


def propose_candidate(seed: int = 1234,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (probability vector)
    that minimises ``evaluate_sequence`` within the given time budget.
    Returns the best vector found (as plain Python floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise a – use the previously discovered construction if any.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        n = 4096                     # moderate length if no hint is supplied
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))   # power‑of‑two FFT length

    cur_conv = _self_convolution_fft(a, L)
    cur_score = _score_from_conv(cur_conv, n)

    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # tiny safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 2) Optimisation parameters
    # ------------------------------------------------------------------
    temp_factor = 0.15               # initial temperature as fraction of max
    min_temp_factor = 0.005
    eta = 0.5                        # initial step size
    min_eta = 1e-6
    max_attempts = 5                 # step‑size reductions per iteration

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ---------- current convolution statistics ----------
        maxval = cur_conv.max()
        # anneal temperature slowly
        temp_factor = max(min_temp_factor, temp_factor * 0.9995)
        temp = maxval * temp_factor + 1e-12

        # ---------- gradient of the smooth max ----------
        grad = _gradient_of_softmax_conv(a, cur_conv, temp, L)

        # ---------- try a few step sizes ----------
        improved = False
        trial_eta = eta
        for _ in range(max_attempts):
            cand = _project_to_simplex(a - trial_eta * grad)
            cand_conv = _self_convolution_fft(cand, L)
            cand_score = _score_from_conv(cand_conv, n)

            if cand_score < cur_score:                     # accept improvement
                a = cand
                cur_conv = cand_conv
                cur_score = cand_score
                improved = True
                eta = min(trial_eta * 1.1, 1.0)             # modestly increase
                break
            else:
                trial_eta *= 0.5                           # shrink step

        if not improved:
            eta = max(eta * 0.99, min_eta)                 # global decay

        # ---------- record best ----------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best {best_score:.6f}")

        # ---------- occasional jitter to escape plateaus ----------
        if rng.random() < 0.02:
            noise = np.exp(rng.normal(0.0, 0.30, size=n))
            jittered = _project_to_simplex(a * noise)
            jitter_conv = _self_convolution_fft(jittered, L)
            jitter_score = _score_from_conv(jitter_conv, n)
            if jitter_score < cur_score:
                a = jittered
                cur_conv = jitter_conv
                cur_score = jitter_score
                print(f"[{time.monotonic() - start_time:6.2f}s] jitter improved to {cur_score:.6f}")

        # ---------- restart if stuck for a while ----------
        if time.monotonic() - last_improve > 40.0:
            # re‑initialise around the current best with fresh noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.50, size=n))
            a = _project_to_simplex(a * noise)
            cur_conv = _self_convolution_fft(a, L)
            cur_score = _score_from_conv(cur_conv, n)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] restart – score reset to {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
