"""
Propose a step‑function by (1) exhaustive sweep of a symmetric
exponential (“U‑shaped”) family a_i ∝ r**i + r**(n‑1‑i)   (0<r<1) and
(2) a short projected‑gradient refinement.  The exponential family places
most mass near the ends of the interval, which strongly reduces the
worst self‑convolution value.  After the sweep the best vector is used as
a warm‑start for a mirror‑descent style projected‑gradient descent on the
probability simplex (sum‑to‑one, a_i≥0).  The algorithm respects the
1000 s budget, prints intermediate improvements and finally returns the
best coefficient list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper routines (FFT convolution, sub‑gradient, block perturb, etc.)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two that can hold the full linear convolution of two
    vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the max‑convolution term at index ``k_max``.
    For a given shift k, the gradient is 2·a_{k‑i} (zero outside the support)."""
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k_max - i_range
        grad[i_range] = 2.0 * a[j]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 5))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s > 0.0:
        out /= s
    else:
        out[:] = 1.0 / n
    return out


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1} (Duchi et al., ICML 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(a*a) under the convention Σ a_i = 1."""
    conv = _conv_self_fft(a, fft_len)
    return 2.0 * a.size * float(conv.max())


# ----------------------------------------------------------------------
#  Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose the length of the vector.
    # ------------------------------------------------------------------
    default_n = 256
    n = default_n
    if "height_sequence_1" in globals():

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Exhaustive sweep of the symmetric exponential family.
    #    a_i(r) = r**i + r**(n-1-i) ,   0 < r < 1
    # ------------------------------------------------------------------
    best = None

    score_uniform = _fast_score(uniform, fft_len)

    best_score = score_uniform

    # sweep r on a dense grid (more points near 1 give finer resolution)

    a = best.copy()

    cur_score = best_score

        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))
        max_conv = float(conv[k_max])
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] refinement best = {best_score:.6f}")

        # ----- schedule ------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        temp = temp_start * (1.0 - elapsed / total) + temp_end * (elapsed / total)

        # ----- gradient of the active (maximum) shift -----------------
        grad = _gradient_max(a, k_max)

        # ----- projected gradient step (mirror descent) ---------------
        a_candidate = a - eta * grad
        a_candidate = _simplex_projection(a_candidate)

        # ----- occasional random block perturbation -------------------
        if rng.random() < perturb_prob:
            a_candidate = _random_block_perturb(a_candidate, rng)

        # ----- evaluate candidate and accept with Metropolis rule -----
        conv_cand = _conv_self_fft(a_candidate, fft_len)
        cand_score = 2.0 * n * float(conv_cand.max())

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            if rng.random() < np.exp((cur_score - cand_score) / max(temp, 1e-15)):
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score

        # ----- soft restart if we have been stuck for a while ---------
        if now - last_improve > restart_gap:
            a = best.copy()
            # small jitter to escape flat regions
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a = np.clip(a * jitter, 0.0, None)
            a = _simplex_projection(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best vector (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
