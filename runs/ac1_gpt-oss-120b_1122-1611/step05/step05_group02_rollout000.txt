"""
Propose a step‑function by minimizing the worst self‑convolution value.

Algorithm
---------
* Work on the probability simplex (a_i ≥ 0, Σ a_i = 1).  The score is
  invariant to uniform scaling, therefore normalising to unit sum is safe.
* At each iteration compute the full self‑convolution via an FFT.
* Collect all “active’’ shifts k whose convolution value is within a tiny
  relative tolerance of the current maximum (these are the binding constraints).
* The sub‑gradient of the max‑convolution at a set of active shifts is the sum of
  the individual gradients  ∂(a*a)_k / ∂a_i = 2·a_{k‑i}.
* Use an Adam‑style adaptive stepsize on the *log‑variables* (mirror descent
  with KL‑geometry).  The update is multiplicative, guarantees non‑negativity
  and stays on the simplex after normalisation.
* Occasionally perform large random perturbations, pairwise mass transfers,
  and soft‑restarts when no improvement has been seen for a while.
* After a prolonged plateau the current vector is up‑sampled (linear
  interpolation) to a larger length – a finer discretisation often yields a
  flatter autocorrelation and thus a smaller maximum.
* The best vector encountered during the whole run is returned.

The routine respects the 1000‑second budget, prints progress messages and
returns a list of floats that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures) as required

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that accommodates a linear
    convolution of two vectors of length ``n`` (needs 2·n‑1 points)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    # FFT of real signal – rfft is slightly faster and returns only the
    # non‑redundant frequencies.
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_for_active_shifts(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """Sum of sub‑gradients ∂(a*a)_k / ∂a for all active shift indices k.
    For a single shift k the gradient is 2·a_{k‑i} on the overlapping region.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active_shifts:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs from i_start … i_end inclusive
        # corresponding j = k - i is reversed slice of a
        # Compute slice a[j] = a[k-i_start : k-i_end-1 : -1]
        # To avoid python slicing pitfalls we use explicit indexing
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        grad[i_range] += 2.0 * a[j_range]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_mass_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random amount of probability mass from i to j."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a
    # Restrict transfer to at most 30 % of the source mass (empirically stable)
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger size, followed by renormalisation."""
    old_n = a.size
    if new_n <= old_n:
        return a
    old_x = np.arange(old_n)
    new_x = np.linspace(0, old_n - 1, new_n)
    upsampled = np.interp(new_x, old_x, a)
    upsampled = np.clip(upsampled, 0.0, None)
    upsampled /= upsampled.sum()
    return upsampled

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector (unit sum)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            # fallback to a modest uniform start
            n0 = 256
            a = np.full(n0, 1.0 / n0, dtype=np.float64)
    else:
        n0 = 256
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    # Ensure we always start with a proper probability vector
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    n = a.size
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping & hyper‑parameters
    # ------------------------------------------------------------------
    best = a.copy()
    # Initial score (the exact evaluation formula will be used later)
    conv = _conv_self_fft(a, fft_len)
    cur_score = 2.0 * n * float(conv.max())
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    # Adam parameters (operating on log‑variables)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)     # first moment estimate
    v = np.zeros_like(a)     # second moment estimate
    t_iter = 0                # iteration counter for bias correction

    # Learning‑rate schedule (log‑space)
    eta_start = 0.5
    eta_end = 0.001

    # Temperature schedule for occasional uphill moves
    temp_start = 0.3
    temp_end = 0.001

    # Stagnation handling
    last_improve = start_time
    restart_gap = 30.0        # seconds without improvement → soft restart
    upsample_gap = 180.0      # seconds without improvement → try larger n

    # Probabilities for random perturbations
    block_perturb_prob = 0.02
    pair_move_prob = 0.07

    # Relative tolerance for defining the *active* set of shifts
    active_rel_tol = 1e-5

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        #   a) Evaluate current state (full convolution & score)
        # --------------------------------------------------------------
        conv = _conv_self_fft(a, fft_len)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        #   b) Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        #   c) Schedules
        # --------------------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # Linear interpolation of schedules
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        temp = temp_start * (1.0 - elapsed / total) + temp_end * (elapsed / total)

        # --------------------------------------------------------------
        #   d) Determine active shifts (those essentially attaining the max)
        # --------------------------------------------------------------
        # Use a small relative tolerance – this gives a richer sub‑gradient
        # when the convolution peak is flat.
        tol = active_rel_tol * max_conv
        active = np.where(conv >= max_conv - tol)[0]
        # If for any reason no active element is found (shouldn't happen), fall back.
        if active.size == 0:
            active = np.array([int(np.argmax(conv))], dtype=np.int64)

        # --------------------------------------------------------------
        #   e) Compute (sub‑)gradient of the max‑convolution term
        # --------------------------------------------------------------
        grad = _gradient_for_active_shifts(a, active)

        # --------------------------------------------------------------
        #   f) Adam on log‑variables (mirror descent with KL geometry)
        # --------------------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # Multiplicative update: log a <-- log a - η * m̂ / (√v̂ + ε)
        update = eta * (m_hat / (np.sqrt(v_hat) + eps_adam))
        # Work in log‑space to guarantee positivity
        log_a = np.log(a + 1e-32) - update
        a = np.exp(log_a)
        # Re‑normalise to the simplex
        a_sum = a.sum()
        if a_sum == 0.0:
            # This can only happen if numerical underflow wiped everything;
            # fall back to uniform distribution.
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # --------------------------------------------------------------
        #   g) Occasional random large perturbations
        # --------------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < pair_move_prob:
            a_candidate = _pairwise_mass_transfer(a, rng)
            # Evaluate candidate – cheap because we already have the FFT length
            conv_cand = _conv_self_fft(a_candidate, fft_len)
            cand_score = 2.0 * n * float(conv_cand.max())
            # Metropolis acceptance
            if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / max(temp, 1e-12)):
                a = a_candidate
                # The next loop iteration will recompute conv & score

        # --------------------------------------------------------------
        #   h) Soft restart if we have been stuck for a while
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            # Return to the best seen so far and add a little jitter
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            # Reset Adam state
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] soft restart – best score {best_score:.6f}")

        # --------------------------------------------------------------
        #   i) Upsample when a plateau lasts long enough (finer discretisation)
        # --------------------------------------------------------------
        if now - last_improve > upsample_gap:
            # Double the dimension but cap at ~5000 to keep FFT cheap.
            new_n = min(2 * n, 5000)
            if new_n > n:
                a = _upsample(a, new_n)
                n = a.size
                fft_len = _fft_len_for_n(n)
                # Resize Adam buffers
                m = np.zeros_like(a)
                v = np.zeros_like(a)
                # Reset iteration counter (learning‑rate schedule continues unchanged)
                t_iter = 0
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] up‑sampled to n={n}")
                # Restart from the current best after up‑sampling
                best = a.copy()
                best_score = cur_score
                last_improve = now

    # ------------------------------------------------------------------
    # 5) Return best (clipped to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
