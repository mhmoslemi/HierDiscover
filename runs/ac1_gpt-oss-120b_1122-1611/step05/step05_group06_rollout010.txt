def _isotonic_decreasing(v):
    # Return nonincreasing vector as L2 projection (PAVA)
    n = len(v)
    # use PAVA for decreasing, essentially run increasing on negative reversed.
    # Approach: use algorithm from statsmodels.
    out = v.copy()
    i = 0
    w = []  # list of blocks (sum, weight)
    for x in out:
        w.append([x, 1])
        i = len(w) - 1
        while i > 0 and w[i-1][0] < w[i][0]:
            # merge blocks
            total = w[i-1][0] * w[i-1][1] + w[i][0] * w[i][1]
            weight = w[i-1][1] + w[i][1]
            avg = total / weight
            w[i-1] = [avg, weight]
            w.pop()
            i -= 1
    # now expand
    result = np.empty_like(v)
    idx = 0
    for avg, weight in w:
        result[idx: idx+weight] = avg
        idx += weight
    return result

def _isotonic_inc(v):
    n = len(v)
    solution = v.copy()
    weight = np.ones(n)
    i = 0
    while i < n - 1:
        if solution[i] <= solution[i+1]:

            tot = solution[i] * weight[i] + solution[i+1] * weight[i+1]
            wtot = weight[i] + weight[i+1]
            avg = tot / wtot
            solution[i] = solution[i+1] = avg
            weight[i] = weight[i+1] = wtot
            # backtrack

"""
Propose a step‑function (non‑negative vector) that minimises evaluate_sequence().

Algorithm Overview
------------------
* Search over several sequence lengths (2 k – 8 k).  
* Initialise candidates with a few simple shapes (uniform, exponential, sinusoid,
  random Dirichlet) and with pure random draws.
* Refine each candidate by a projected Adam optimiser.
  – The objective is 2·n·max(conv(a,a)); the sum of a is kept =1.
  – Gradient is either the exact sub‑gradient of the worst convolution entry
    or the smooth‑max (log‑sum‑exp) gradient with a linearly increasing β.
  – After each Adam step we (i) project onto the probability simplex,
    (ii) enforce a *non‑increasing* shape by isotonic regression,
    (iii) renormalise (projection preserves sum, scaling a monotone vector
         keeps it monotone).
* Occasionally apply cheap random moves (block multiplicative perturb,
  pairwise mass transfer) to escape plateaus.
* Keep the global best vector; periodically polish it with a longer optimisation
  run.
* The whole search respects the supplied time budget (default 1000 s) and
  returns the best vector found.

The monotone projection (PAVA) is a cheap way to bias the search towards
shapes that minimise the centre of the autocorrelation (large values at the
ends are paired with small values when the sequence is reversed).  Empirically
this drives the maximal convolution value below the previous best of ~1.514.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution via real FFT (full length of the result)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _isotonic_decreasing(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the cone of non‑increasing vectors.
    Implements the Pool Adjacent Violators Algorithm (PAVA).
    """
    n = v.shape[0]
    # Each block stores (sum, weight)
    sums = v.tolist()
    weights = [1] * n
    i = 0
    while i < len(sums) - 1:
        avg_i = sums[i] / weights[i]
        avg_next = sums[i + 1] / weights[i + 1]
        if avg_i >= avg_next - 1e-12:               # already non‑increasing
            i += 1
        else:                                        # merge blocks i and i+1
            total = sums[i] + sums[i + 1]
            wtot = weights[i] + weights[i + 1]
            sums[i] = total
            weights[i] = wtot
            del sums[i + 1]
            del weights[i + 1]
            i = max(i - 1, 0)
    # Expand merged blocks
    out = np.empty(n, dtype=np.float64)
    pos = 0
    for s, w in zip(sums, weights):
        out[pos:pos + w] = s / w
        pos += w
    return out


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    For the worst index i_max we have ∂b_i/∂a_j = 2·a_{i_max‑j} when the index is valid.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max
        f_β(a) = (1/β)·log Σ_i exp(β·conv_i) .
    Returns ∂f_β/∂a (no extraneous +a term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    # convolution of w with reversed a
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    # central n entries correspond to the gradient
    start = a.size - 1
    grad_part = conv_w_rev[start:start + a.size] / sum_w
    return grad_part


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = _project_onto_simplex(out)
    # enforce monotonicity after the perturb
    out = _isotonic_decreasing(out)
    out /= out.sum()
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """
    Transfer a modest amount of mass from a random index i to a random j.
    The move is kept only if it improves the current score.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    a2 = _isotonic_decreasing(a2)
    a2 /= a2.sum()
    conv = _conv_self(a2, fft_len_conv)
    cur_conv = _conv_self(a, fft_len_conv)
    if conv.max() < cur_conv.max():
        return a2
    return a


# ----------------------------------------------------------------------
# Optimisation core
# ----------------------------------------------------------------------


def _local_optimize(a0: np.ndarray,
                    rng: np.random.Generator,
                    max_iters: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑based optimiser with monotonic projection.
    Returns the best vector (sum = 1) encountered and its score
    (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state – only first‑moment is needed for simplicity
    m = np.zeros_like(a)
    beta1 = 0.9
    t = 0

    best_vec = a.copy()
    conv = _conv_self(a, fft_len_conv)
    best_score = 2.0 * n * conv.max()

    # linear schedule for β in the smooth‑max (starts small, finishes large)
    beta_start = 1.0
    beta_end = 1e5

    while t < max_iters and time.monotonic() < deadline:
        t += 1

        # --------------------------------------------------------------
        # 1) evaluate current score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # 2) compute gradient
        # --------------------------------------------------------------
        if rng.random() < 0.35:
            # hard sub‑gradient of the current worst entry
            raw_grad = _subgradient_step(a, conv)
        else:
            # smooth‑max gradient with gradually increasing β
            beta = beta_start + (beta_end - beta_start) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # 3) Adam update (additive) – step‑size decays like 1/√t
        # --------------------------------------------------------------
        lr = 0.25 / np.sqrt(t)               # modest initial step‑size
        m = beta1 * m + (1.0 - beta1) * raw_grad
        m_hat = m / (1.0 - beta1 ** t)
        step = lr * m_hat

        a = a - step

        # --------------------------------------------------------------
        # 4) feasibility projections (simplex + monotone)
        # --------------------------------------------------------------
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a)

        # enforce a non‑increasing shape
        a = _isotonic_decreasing(a)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        # --------------------------------------------------------------
        # 5) occasional random moves to inject diversity
        # --------------------------------------------------------------
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.04:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score

def _init_uniform_random(rng: np.random.Generator, n: int) -> np.ndarray:
    """Plain Dirichlet(1) draw."""
    a = rng.random(n)
    return a / a.sum()


def _init_exponential(rng: np.random.Generator, n: int) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-λ·i) with random λ."""
    lam = rng.uniform(0.0, 0.005) + 1e-6
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a = _isotonic_decreasing(a)                # ensure monotone decreasing
    a /= a.sum()
    return a


def _init_sinusoid(rng: np.random.Generator, n: int) -> np.ndarray:
    """Half‑sine shape (non‑negative, monotone after projection)."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.sin(np.pi * x) ** 2
    a = _isotonic_decreasing(a)
    a = _project_onto_simplex(a)
    return a


def _init_triangular(rng: np.random.Generator, n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the middle)."""
    half = (n + 1) // 2
    left = np.linspace(1, half, half, endpoint=True)
    right = left[:-1][::-1] if n % 2 else left[::-1]
    a = np.concatenate([left, right])
    a = a.astype(np.float64)
    a = _isotonic_decreasing(a)               # makes it decreasing from the left
    a = _project_onto_simplex(a)
    return a


def _candidate_initialisers(rng: np.random.Generator, n: int):
    """Yield a few deterministic initialisers plus a pure random one."""
    yield _init_uniform_random(rng, n)
    yield _init_exponential(rng, n)
    yield _init_sinusoid(rng, n)
    yield _init_triangular(rng, n)
    # a few pure random draws for diversity
    for _ in range(2):
        yield _init_uniform_random(rng, n)

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    evaluate_sequence() before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Keep a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, evaluate it first (baseline)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                conv = _conv_self(cand, _fft_len_for_len(2 * cand.size - 1))
                cur = 2.0 * cand.size * conv.max()
                if cur < best_score:
                    best_score = cur
                    best_vec = cand.copy()
                print(f"[baseline] score = {cur:.6f} (n={cand.size})")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore – mix of small / medium / large vectors
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 3809, 5000, 8000]

    # ------------------------------------------------------------------
    # Main search loop – many cheap restarts, occasional long polish
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Pick a length (biased towards the larger ones)
        probs = np.array([0.15, 0.20, 0.30, 0.20, 0.15])
        n = int(rng.choice(lengths, p=probs))

        # --------------------------------------------------------------
        # Choose an initialiser (deterministic shapes + random draws)
        # --------------------------------------------------------------
        init_gen = _candidate_initialisers(rng, n)
        # pick the next initialiser; if exhausted, fall back to random
        try:
            a0 = next(init_gen)
        except StopIteration:
            a0 = _init_uniform_random(rng, n)

        # --------------------------------------------------------------
        # Short local optimisation
        # --------------------------------------------------------------
        max_iters = 1200 if n <= 3000 else 800   # a few hundred iterations
        a_opt, score_opt = _local_optimize(a0, rng, max_iters, deadline)

        # --------------------------------------------------------------
        # Record global improvement
        # --------------------------------------------------------------
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})")

        # --------------------------------------------------------------
        # If time remains, try a longer polish on the current best
        # --------------------------------------------------------------
        if (best_vec is not None) and (time.monotonic() < deadline - 0.5):
            # longer run, more iterations, higher β ceiling
            polished, score_polished = _local_optimize(
                best_vec.copy(),
                rng,
                max_iters=2500,
                deadline=deadline)
            if score_polished < best_score:
                best_score = score_polished
                best_vec = polished.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polished global best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: uniform distribution if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also clips, but we do it anyway)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
