def _solve_lp(a, M):
    import cvxpy as cp
    n = a.shape[0]
    b = cp.Variable(n, nonneg=True) # nonneg constraints redundant
    # conv = cp.conv(b, a) # this returns length 2n-1.
    conv = cp.conv(b, a)
    constraints = [conv <= M]
    # b >= 0 implicitly due to nonneg variable?
    # Actually cp.Variable(..., nonneg=True) ensures b >=0.

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-8, reltol=1e-6, feastol=1e-6)
    except cp.SolverError:
        return None
    if b.value is None:
        return None
    b_sol = np.array(b.value).flatten()
    # Clip negative tiny values to 0
    b_sol[b_sol < 0] = 0.0
    return b_sol

import numpy as np
import cvxpy as cp

def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """gradient of sum_k w_k (a*a)_k = 2 * correlation of a and w."""
    n = a.size
    L = w.size
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

def _solve_lp(a: np.ndarray, M: float, warm_start_b=None):
    """
    Solve max sum(b) s.t. b >=0 and (a*b) <= M.
    Returns b (numpy array) or None if solve fails.
    """
    n = a.shape[0]
    b = cp.Variable(n, nonneg=True)
    # linear convolution of variable b with constant a
    conv = cp.conv(b, a)  # length = 2*n-1
    constraints = [conv <= M]  # elementwise
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    if warm_start_b is not None:
        b.value = warm_start_b
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-6, feastol=1e-7)
    except Exception:
        return None
    if b.value is None:
        return None
    b_sol = np.asarray(b.value).flatten()
    b_sol[b_sol < 0] = 0.0
    return b_sol

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid mirror‑descent + occasional linear‑programming refinement
    for minimising evaluate_sequence().
    The algorithm works on the probability simplex (∑a_i = 1) and
    strives to flatten the self‑convolution.  Gradient steps use a
    soft‑max surrogate of the L∞‑norm and an exponentiated‑gradient
    update; every few seconds a small LP (maximise sum(b) subject to
    (a*b)≤current_max) is solved to make a strong global jump.
    """
    rng = np.random.default_rng(seed)

    # ---- problem size -------------------------------------------------
    n = int(kwargs.get("n", 2500))
    if n < 2:
        n = 2

    # ---- initial vector (sqrt‑difference) -----------------------------
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a = a / a.sum()

    # ---- bookkeeping --------------------------------------------------
    fft_len = _fft_len_for_n(n)
    conv = _conv_self(a, fft_len)
    best = a.copy()
    best_score = 2.0 * n * np.max(conv)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 2.0)   # safety margin

    # ---- hyper‑parameters ---------------------------------------------
    # learning‑rate schedule (exponentiated‑gradient)
    eta_start = 1.0
    eta_end = 0.015
    # temperature schedule for soft‑max surrogate
    beta_start = 0.2
    beta_end = 4000.0
    # momentum for gradient smoothing
    momentum = 0.88
    # LP refinement interval (seconds)
    lp_interval = 30.0
    last_lp = start_time
    b_warm = None  # warm‑start variable for LP

    # random perturbation probabilities
    block_perturb_prob = 0.025
    pair_move_prob = 0.07
    restart_gap = 180.0
    last_improve = start_time
    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:
        # ---- current convolution and score ---------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- keep best -----------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- LP refinement --------------------------------------------
        now = time.monotonic()
        if now - last_lp >= lp_interval:
            # solve LP
            b_sol = _solve_lp(a, cur_max, warm_start_b=b_warm)
            if b_sol is not None:
                s = b_sol.sum()
                if s > 1.0 + 1e-6:
                    # adopt the LP solution (full step)
                    a = b_sol / s
                    b_warm = b_sol           # warm‑start next LP
                    # recompute conv for this new a and continue
                    last_lp = now
                    continue
            last_lp = now   # even if LP fails we wait until next interval

        # ---- schedule -------------------------------------------------
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- compute soft‑max weights over the full convolution ----------
        # to avoid numerical overflow we shift by cur_max
        w_raw = np.exp(beta * (conv - cur_max))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ---- gradient of the smooth surrogate -------------------------
        grad = _gradient_correlation(a, w)

        # ---- momentum -------------------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- exponentiated‑gradient (mirror) update -------------------
        factor = np.exp(-eta * grad)
        a = a * factor
        # protect against numerical underflow
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ---- occasional random block perturbation --------------------
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            factor_blk = np.exp(rng.normal(0.0, 0.3))
            a[start:start + width] *= factor_blk
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # ---- occasional pairwise mass transfer ------------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-12:
                delta = rng.random() * a[i] * 0.6
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

        # ---- soft restart if stagnating -------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Return the best sequence (clip to limits of evaluate_sequence)
    # -----------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

                    b_warm = b_sol            # warm‑start next LP
                    last_lp = now
                    # skip the gradient step this round (re‑evaluate immediately)
                    continue
            last_lp = now

        # ----- schedule for η and β ---------------------------------------
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ----- soft‑max weights over the full convolution -------------------
        # shift by cur_max for numerical stability
        w_raw = np.exp(beta * (conv - cur_max))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ----- gradient of the smooth surrogate -----------------------------
        grad = _gradient_correlation(a, w)

        # ----- momentum smoothing -----------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror‑descent) update ---------------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- occasional random block perturbation ------------------------
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            factor_blk = np.exp(rng.normal(0.0, 0.3))
            a[start:start + width] *= factor_blk
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # ----- occasional pairwise mass transfer ---------------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-12:
                delta = rng.random() * a[i] * 0.6
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

        # ----- soft restart if we have been stagnant -----------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
