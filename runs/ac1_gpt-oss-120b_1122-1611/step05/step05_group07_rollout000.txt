def _proj_simplex(v):
   """Project vector v onto the probability simplex {x >=0, sum x = 1}."""
   # Assuming v is np.ndarray
   n = v.shape[0]
   # sort descending
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
   if rho.size == 0:
       # all <=0 case: return uniform
       return np.full_like(v, 1.0 / n)
   rho = rho[-1]
   theta = cssv[rho] / (rho + 1)
   w = np.maximum(v - theta, 0)
   return w

assistantfinal**Solution Overview**

The evaluation function is  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\]

where `a*a` denotes the linear self‑convolution.  
Because the objective is homogeneous, we may (and do) keep the vector on the
probability simplex  \(\sum a_i =1\).  
The problem is therefore

*minimise*   \(\displaystyle \max_k (a*a)_k\)  
*subject to* \(a_i\ge 0,\; \sum_i a_i =1\).

The objective is convex (pointwise maximum of quadratic forms).  
We use a two‑phase optimisation:

1. **Smooth mirror‑descent** – an exponentiated‑gradient step on a soft‑max
   approximation of the maximum.  The temperature \( \beta \) is increased
   up to \(5\!\times\!10^4\) so that the algorithm focuses on the true peak,
   while a linearly decaying learning‑rate together with momentum gives fast
   global progress.

2. **Projected sub‑gradient refinement** – once the temperature is large we
   switch to a genuine sub‑gradient of the (nonsmooth) max‑norm.
   The sub‑gradient for a set of active peak indices is obtained by a second
   convolution (exact, no approximation).  A Euclidean projection onto the
   simplex enforces the constraints.  The step size follows the classic
   \(c/\sqrt{t}\) schedule.

Both phases repeatedly keep the best vector seen so far.  If no improvement
occurs for 25 s the algorithm restarts from the current best with a modest
multiplicative noise, which helps escape local minima.

All operations are \(O(n\log n)\) thanks to FFT‑based convolutions;
for the typical size \(n\approx 3800\) the whole search finishes comfortably
within the 1000 s budget.

```python
"""
Propose a step‑function (non‑negative heights) that improves the current
upper bound for `evaluate_sequence`.  The algorithm consists of:

1. Initialise a probability vector (sum = 1) – either from the previously
   found construction or a fresh Dirichlet draw.
2. Phase 1 – smooth mirror‑descent with an annealed soft‑max (β up to 5e4)
   and exponentially‑decaying step‑size.  The gradient is obtained by a
   second FFT‑convolution.
3. Phase 2 – projected sub‑gradient descent on the true max‑norm.  The
   sub‑gradient for the active peak indices is computed exactly via
   convolution, and the iterate is projected onto the simplex.
4. Periodic jitter and a “restart after stagnation” mechanism keep the
   search diverse.
5. The best vector encountered before the time budget expires is returned.

All helper functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using a pre‑allocated FFT length."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` under the supplied time budget.
    Returns the best sequence (list of floats) found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            n = 3840
            a = rng.dirichlet(np.ones(n)).astype(np.float64)
    else:
        n = 3840
        a = rng.dirichlet(np.ones(n)).astype(np.float64)

    n = a.size
    # FFT length for self‑convolution (power of two, ≥ 2·n‑1)
    fft_len_self = _next_pow2(2 * n - 1)
    # FFT length needed for the gradient convolutions (≥ 3·n‑2)
    fft_len_grad = _next_pow2(3 * n - 2)

    # 2) Book‑keeping of the best solution
    # ------------------------------------------------------------------
    conv = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.65 * total_time   # 65 % of the budget

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Phase 1 (smooth mirror descent)
    eta_start = 8.0          # initial step‑size on the simplex
    eta_end = 0.01           # final step‑size
    beta_start = 0.5
    beta_end = 5.0e4         # final temperature (sharp soft‑max)
    momentum = 0.9
    grad_prev = np.zeros_like(a)

    # Phase 2 (projected sub‑gradient)
    step0 = 0.5              # factor for 1/√t step‑size
    active_eps = 1e-12      # tolerance for “active” peak indices

    it_sub = 0               # iteration counter for the sub‑gradient phase

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – smooth mirror‑descent (soft‑max of the convolution)
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedules for η and β
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # self‑convolution
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()

            # soft‑max weights (stable implementation)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:

            fp = np.fft.rfft(p, n=fft_len_grad)

            grad = 2.0 * grad_full[n - 1: n - 1 + n]

            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            a *= factor

            if rng.random() < 0.02:

            # recompute exact score (no soft‑max)
            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Phase 2 – projected sub‑gradient descent on the true max‑norm
        # --------------------------------------------------------------
        else:
            # compute exact convolution and current maximum
            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            max_c = conv.max()

            # active set of indices attaining (or nearly attaining) the maximum
            active = np.where(conv >= max_c - active_eps)[0]

            # build a probability vector that puts equal mass on the active indices
            p = np.zeros_like(conv)
            if active.size > 0:

            rev_a = a[::-1]

            it_sub += 1
            step = step0 / np.sqrt(it_sub)

            # Euclidean projected gradient step
            a_tmp = a - step * grad
            a = _proj_simplex(a_tmp)

            # small block‑perturbations keep the walk from stagnating
            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)

        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:

        # --------------------------------------------------------------

            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _proj_simplex(a)
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
