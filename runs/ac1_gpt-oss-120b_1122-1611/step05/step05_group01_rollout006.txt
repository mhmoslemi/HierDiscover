def refine_seq(a, n, start_time, deadline):
    # compute FFT length
    fft_len = _fft_len_for_n(n)
    # set initial parameters
    eta = 0.5
    beta = 1.0
    # while time left:
    while time.monotonic() < deadline:
        # compute conv_self using FFT
        conv = _conv_self_fft(a, fft_len)
        max_conv = conv.max()
        # update beta gradually
        if beta < 200: beta *= 1.001
        # compute softmax weights
        # w = exp(beta * (conv - max_conv)) normalized
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to uniform
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / w_sum
        # compute gradient via convolution of a and reversed w
        rev_w = w[::-1]
        # compute convolution via FFT: a (len=n) * rev_w (len=2n-1)
        # FFT length must accommodate n + (2n-1) - 1 = 3n-2
        fft_len2 = 1 << ((3*n - 3).bit_length())  # small overhead
        # Compute FFT of a padded and rev_w padded to fft_len2
        Fa = np.fft.rfft(a, n=fft_len2)
        Fr = np.fft.rfft(rev_w, n=fft_len2)
        conv_full = np.fft.irfft(Fa * Fr, n=fft_len2)[: 3*n - 2]
        L = 2*n - 2
        grad = 2.0 * conv_full[L - np.arange(n)]
        # Mirror descent update
        a = a * np.exp(-eta * grad)
        # reproject to simplex (renormalize)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a / a_sum
        # Evaluate periodically
        if (np.random.rand() < 0.01):
            seq = a.tolist()
            sc = evaluate_sequence(seq)
            if np.isfinite(sc) and sc < best_score_global:
                best_score_global = sc
                best_seq_global = seq
                print(f"[{time.monotonic() - start_time:.2f}s] improved to {sc:.6f} len={n}")
        # decay eta a little
        eta *= 0.99995

        # break if time nearly exhausted

Return best_seq_global.

def _refine_sequence(a_init, n, start, deadline, rng, current_best_score):

    best_local_score = evaluate_sequence(best_local_seq)  # call once
    # parameters
    eta = 0.5  # step size for additive gradient (maybe gradually decrease)
    step = eta
    step_decay = 0.9995
    # iterations while time left
    iter_count = 0
    while time.monotonic() < deadline:
        # compute convolution self via FFT
        fft_len_self = _next_power_of_two(2*n - 1)
        Fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(Fa * Fa, n=fft_len_self)[:2*n-1]

        diff = conv - mu

"""
Propose a candidate step‑function sequence.

The algorithm builds on the known “discrete arcsine’’ sequence but then
refines it with a two‑phase mirror‑descent / projected‑gradient scheme:

* Phase 1 – variance minimisation  
  Minimise the L2‑variance of the self‑convolution (making it as flat as
  possible).  The gradient of  ∑(conv_k‑μ)² w.r.t. a can be written as a
  convolution, which we evaluate efficiently by FFT.

* Phase 2 – max‑convolution reduction  
  After the variance has been reduced, a sub‑gradient for the maximum
  convolution value is added (the classic “max‑index’’ gradient).  The
  weight of this term grows during optimisation, allowing a final fine‑
  tuning of the peak.

All updates keep the vector on the probability simplex by a simple
projection (clipping negative entries and renormalising).  The routine
runs for the whole time budget (default 1000 s), trying several lengths
and a few exponent‑variants of the arcsine family, and stores the best
sequence found.

The implementation uses only NumPy and the supplied ``evaluate_sequence``
function; no file or network I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D real arrays using FFT."""
    size = a.size + b.size - 1
    fft_len = _next_pow2(size)
    Fa = np.fft.rfft(a, n=fft_len)
    Fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Fa * Fb, n=fft_len)[:size]
    return conv


def _self_conv_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    return _conv_fft(a, a)


def _grad_variance(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of  L2 variance  ∑(conv_k‑μ)²  w.r.t. a.
    μ = 1/(2n‑1)  (since sum(conv)=1 when sum(a)=1).
    Returns a vector of length n.
    """
    n = a.size
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                      # shape (2n‑1,)
    rev_diff = diff[::-1]                 # reversed
    full_conv = _conv_fft(a, rev_diff)    # length 3n‑2
    L = 2 * n - 2
    grad = 4.0 * full_conv[L - np.arange(n)]
    return grad


def _grad_max_index(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution term.
    For the index k_max where conv attains its maximum,
    ∂max/∂a_i = 2·a_{k_max‑i}  (if the partner index lies in [0,n)).
    """
    n = a.size
    k_max = int(np.argmax(conv))
    idx = np.arange(n, dtype=np.int64)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partner[mask]]
    return grad


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the simplex
    {x | x_i ≥ 0, ∑x_i = 1}.
    """
    # ensure non‑negative
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _refine_sequence(a0: np.ndarray,
                     start: float,
                     deadline: float,
                     rng: np.random.Generator,
                     global_best_score: float) -> tuple[np.ndarray, float]:
    """
    Perform the two‑phase descent on a single starting vector.

    Returns the best vector found (as a NumPy array) together with its
    evaluation score.  The function stops when the global deadline is
    reached.
    """
    a = a0.copy()
    n = a.size

    # algorithmic hyper‑parameters
    step = 0.5                     # additive step size (decays)
    step_decay = 0.9995
    gamma = 0.1                    # weight of the max‑gradient part
    gamma_inc = 1.001

    best_local_seq = a.copy()
    best_local_score = evaluate_sequence(a.tolist())

    iter_cnt = 0
    while time.monotonic() < deadline:
        # 1) compute self‑convolution
        conv = _self_conv_fft(a)

        # 2) variance gradient
        grad_var = _grad_variance(a, conv)

        # 3) max‑index gradient (sub‑gradient)
        grad_max = _grad_max_index(a, conv)

        # 4) combined update (additive, then project)
        grad_total = grad_var + gamma * grad_max
        a = a - step * grad_total
        a = _project_to_simplex(a)

        # occasional stochastic jitter to avoid stagnation
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.06, size=n))
            a = a * jitter
            a = _project_to_simplex(a)

        # periodic evaluation (roughly every ~0.2 s)
        if iter_cnt % 30 == 0:
            sc = evaluate_sequence(a.tolist())
            if np.isfinite(sc) and sc < best_local_score:
                best_local_score = sc
                best_local_seq = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s]   local improvement: {sc:.6f} (n={n})")

        # decay schedules
        step *= step_decay
        gamma *= gamma_inc
        iter_cnt += 1

        # safety: stop early if we already beat the global best by a noticeable margin
        if best_local_score < global_best_score * 0.995:
            # we are already better than the current global best by >0.5 %
            # no need to waste more time on this particular length
            break

    # final check
    final_score = evaluate_sequence(a.tolist())
    if np.isfinite(final_score) and final_score < best_local_score:
        best_local_score = final_score
        best_local_seq = a.copy()

    return best_local_seq, best_local_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.

    The routine tries several lengths and arcsine‑exponent variants,
    refines each with the gradient‑descent procedure above and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # leave a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0)  Use the previous construction if it is available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(map(float, globals()["height_sequence_1"]))
            sc = evaluate_sequence(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"[{time.monotonic() - start_time:6.2f}s] baseline score = {sc:.6f} (len {len(cand)})")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1)  Scan a handful of lengths and arcsine exponents
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 3072, 4096, 5120, 6144]
    alphas = [0.45, 0.5, 0.55]          # exponent in the denominator

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)

        for alpha in alphas:
            if time.monotonic() >= deadline:
                break
            raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
            raw = np.clip(raw, 0.0, 1000.0)
            raw /= raw.sum()                     # normalise to sum = 1
            # add a tiny random perturbation to break symmetry
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a0 = raw * jitter
            a0 = _project_to_simplex(a0)

            # ------------------------------------------------------------------
            # 2)  Refine this candidate
            # ------------------------------------------------------------------
            a_refined, sc_refined = _refine_sequence(
                a0, start_time, deadline, rng, best_score
            )
            if np.isfinite(sc_refined) and sc_refined < best_score:
                best_score = sc_refined
                best_seq = a_refined.tolist()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {sc_refined:.6f} (n={n}, α={alpha})")

        # also try a uniform start for this length (cheap baseline)
        if time.monotonic() >= deadline:
            break
        a_uniform = np.full(n, 1.0 / n)
        a_refined, sc_refined = _refine_sequence(
            a_uniform, start_time, deadline, rng, best_score
        )
        if np.isfinite(sc_refined) and sc_refined < best_score:
            best_score = sc_refined
            best_seq = a_refined.tolist()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST (uniform start) = {sc_refined:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3)  Fallback in case nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] score = {best_score:.6f}")

    # Clip to the limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    total_elapsed = time.monotonic() - start_time
    print(f"[{total_elapsed:6.2f}s] final best score = {best_score:.6f} (length {len(best_seq)})")
    return best_seq
