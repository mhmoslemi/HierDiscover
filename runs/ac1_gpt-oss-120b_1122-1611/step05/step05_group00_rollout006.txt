"""
Propose a step‑function (non‑negative heights) that minimizes
evaluate_sequence().

Algorithm
---------
*   Initialise with the analytically‑derived “√‑difference’’ shape
    a_i = √(i+1)−√i, normalised to sum = 1.
*   Optimise the log‑probabilities p_i = log a_i using the Adam optimiser
    on a smooth approximation of the objective:
        L(p) = Σ_k w_k·(a∗a)_k ,   w_k ∝ exp(β·((a∗a)_k−max(a∗a)))
    where β grows from a small value to a large value so the surrogate
    converges to the true max.
*   The gradient ∂L/∂a = 2·a⋆w (correlation) is obtained efficiently via
    FFT; the gradient w.r.t. the logits p is
        ∇_p L = a·(∇_a L − ⟨a,∇_a L⟩).
*   Adam’s adaptive step‑size together with a linear learning‑rate decay
    yields fast, stable convergence.
*   Every few minutes, if no improvement has been seen, the optimiser is
    “restarted’’ by jittering the current best logits – this keeps the
    search from stagnating.
*   Random small perturbations of the logits are applied with low probability
    to explore the neighbourhood.
*   The best vector found during the allotted time (default ≈ 1000 s) is
    returned, clipped to the limits imposed by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold a linear convolution
# of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT.
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a, returned length = 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of Σ_k w_k·(a∗a)_k with respect to a (correlation a⋆w).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Grad of Σ_k w_k·(a∗a)_k = 2·a⋆w (correlation)."""
    n = a.size
    L = w.size                # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
