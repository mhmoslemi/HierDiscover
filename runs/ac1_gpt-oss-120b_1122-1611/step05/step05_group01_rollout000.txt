"""
Algorithm Overview
------------------
1. **Parametric sweep** – Generate candidates from a generalized
   arcsine family  
   a_i ∝ 1 / ((i+ε)*(n‑1‑i+ε))**α   (ε ≥ 0, α≈0.5) for many values of
   length *n*, exponent *α* and endpoint shift *ε*.  All candidates are
   normalised to unit sum (the score is scale‑invariant) and the best
   one is kept.

2. **Smooth‑max gradient descent** – Starting from the best candidate,
   minimise a differentiable surrogate of the true objective  
   `max(conv(a,a))`.  The surrogate is the soft‑max  
   `ϕ_β(b) = Σ w_k b_k` with weights `w_k ∝ exp(β·b_k)`.  The gradient
   of this surrogate is
        ∇ϕ_β = 2·corr(a, w)  (correlation = convolution with a reversed)
   and is obtained efficiently by FFT‑based convolutions.
   An exponentiated‑gradient update (`a ← a·exp(-η∇ϕ_β)`) keeps the
   iterates non‑negative; after each step the vector is re‑normalised to
   unit sum.

   *β* is increased slowly (sharpening the soft‑max) while the step size
   η decays, and occasional random jitter helps escape plateaus.

3. **Random restarts** – If time permits, a few Dirichlet‑initialised
   sequences of the same length are refined with the same descent
   procedure; the overall best sequence is returned.

All helper functions are defined at module level (no closures), the code
uses only `numpy` and standard library modules, respects the
`evaluate_sequence` interface, and prints timing / improvement messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors via FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    n = x.size + y.size - 1
    L = _next_pow2(n)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    Z = X * Y
    conv = np.fft.irfft(Z, n=L)[:n]
    return conv


def _self_conv(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using FFT."""
    return _conv_fft(a, a)


def _score_from_a(a: np.ndarray) -> float:
    """
    Compute the exact value returned by ``evaluate_sequence`` for a
    non‑negative vector a (assumed to have sum > 0).  The vector is not
    clipped; the caller must guarantee 0 ≤ a_i ≤ 1000.
    """
    n = a.size
    b = _self_conv(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _smooth_grad(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of the max self‑convolution.
    Returns a vector of the same size as a.
    """
    n = a.size
    # 1. compute self‑convolution (full length 2n‑1)
    b = _self_conv(a)
    # 2. soft‑max weights (numerically stable)
    b_max = b.max()
    w = np.exp(beta * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # 3. correlation a ↔ w = 2·corr(a, w)
    a_rev = a[::-1]
    corr_full = _conv_fft(a_rev, w)          # length 3n‑2
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]  # slice of length n
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    # Stop a bit before the hard deadline to leave room for finalisation.
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0)  Use the previous construction (if present) as a baseline.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(map(float, globals()["height_sequence_1"]))
            sc = _score_from_a(np.array(cand, dtype=np.float64))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"[{time.monotonic() - start:6.2f}s] baseline = {sc:.6f}"
                      f" (len={len(cand)})")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1)  Parametric sweep over the generalized arcsine family.
    # --------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 3072, 4096, 6144, 8192]
    alphas = [0.40, 0.45, 0.48, 0.50, 0.52, 0.55, 0.60]
    epsilons = [0.2, 0.5, 0.8, 1.0]           # endpoint shift

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        for alpha in alphas:
            if time.monotonic() >= deadline:
                break
            for eps in epsilons:
                if time.monotonic() >= deadline:
                    break
                # raw shape; protect against division by zero
                denom = (i + eps) * (n - eps - i)
                # avoid negative or zero denominators
                denom = np.where(denom <= 0, 1.0, denom)
                raw = 1.0 / (denom ** alpha)
                # Clip to the limits enforced by ``evaluate_sequence``.
                raw = np.clip(raw, 0.0, 1000.0)
                total = raw.sum()
                if total == 0.0:

                raw /= total                     # normalise to unit sum
                sc = _score_from_a(raw)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = raw.tolist()
                    print(f"[{time.monotonic() - start:6.2f}s] sweep "

    def _run_descent(a0: np.ndarray, max_iters: int = 10_000) -> None:
        """Refine a starting vector a0 in place, updating global best."""
        nonlocal best_seq, best_score
        a = a0.copy()
        n = a.size
        # FFT lengths (pre‑computed for speed)
        fft_len_self = _next_pow2(2 * n - 1)
        fft_len_grad = _next_pow2(3 * n - 2)

        # Descent hyper‑parameters
        eta = 0.6               # initial step size for exponentiated update
        beta = 6.0              # initial soft‑max sharpness
        beta_max = 200.0
        beta_inc = 1.03         # multiplicative increase per 100 iters
        it = 0

        while time.monotonic() < deadline and it < max_iters:
            # 1) self‑convolution and max value
            b = _self_conv(a)                     # length 2n‑1
            max_b = b.max()

            # 2) compute smooth gradient
            #    (use separate FFT length for convolution with w)
            w_raw = np.exp(beta * (b - max_b))
            w_sum = w_raw.sum()
            if w_sum == 0.0:

            else:

            # correlation a ↔ w   (via convolution with a reversed)
            a_rev = a[::-1]

            corr_full = _conv_fft(a_rev, w)        # length 3n‑2
            grad = 2.0 * corr_full[n - 1 : n - 1 + n]

            # 3) exponentiated‑gradient update (keeps positivity)
            a = a * np.exp(-eta * grad)

                a *= jitter

            # 5) project onto the probability simplex (scale‑invariant)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

            # 6) evaluate (every ~0.2 s on average)

                    best_score = sc
                    best_seq = a.tolist()

            # 7) schedule updates
            if it % 100 == 0 and beta < beta_max:
                beta = min(beta_max, beta * beta_inc)
            eta *= 0.9997          # gentle decay
            it += 1

    # --------------------------------------------------------------
    # 2)  Refine the best candidate found so far.
    # --------------------------------------------------------------
    if best_seq is not None:
        a0 = np.array(best_seq, dtype=np.float64)
        _run_descent(a0, max_iters=30_000)

    # --------------------------------------------------------------
    # 3)  Random restarts (if time permits).
    # --------------------------------------------------------------
    if best_seq is not None:
        n_best = len(best_seq)
        restarts = 5
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break
            # Dirichlet with concentration 0.8 → moderately sparse start
            a_rand = rng.dirichlet(np.full(n_best, 0.8))
            _run_descent(a_rand, max_iters=8_000)

    # --------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = [1.0 / n] * n
        best_score = _score_from_a(np.array(best_seq, dtype=np.float64))

    # 5)  Final sanity clipping (required by ``evaluate_sequence``).
    # --------------------------------------------------------------
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
