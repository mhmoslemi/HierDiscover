def project_simplex(v):
    """Projects vector v onto simplex {x>=0, sum x = 1}."""
    # Implementation from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def conv_full_fft(a, b, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)
    # Return full conv length len(a)+len(b)-1
    return conv[: a.size + b.size - 1]

"""
Hybrid simulated‑annealing / mirror‑descent optimizer for the
self‑convolution max‑norm problem.

The algorithm keeps a current probability vector (non‑negative,
unit‑sum) and repeatedly proposes a new candidate by one of the
following moves:

* exponentiated‑gradient step on the active (maximum) convolution index,
* gradient descent on a smooth Lp‑approximation of the max (p grows
  from 4 to 12),
* a random pairwise mass transfer,
* a multiplicative log‑normal block perturbation.

Each candidate is accepted if it improves the objective
``2 * n * max(conv) / (sum(a)**2``, otherwise with a temperature‑dependent
probability (simulated annealing).  The best vector found is stored,
and after a period without improvement the algorithm restarts from a
noisy copy of the best solution.  At the end the vector is trimmed of
negligible leading/trailing entries and returned as a plain Python
list.

The implementation uses FFT‑based convolutions for speed and a
projection onto the probability simplex after gradient steps.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a full linear convolution
    of two length‑``n`` vectors (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_full(len_a: int, len_b: int) -> int:
    """Smallest power‑of‑two length for the full linear convolution of
    lengths ``len_a`` and ``len_b``."""
    size = len_a + len_b - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of *a* and *b* using FFT. Length of the
    result is ``len(a) + len(b) - 1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)
    return conv[: a.size + b.size - 1]


def _gradient_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the max‑convolution term at shift ``k_max``.
    For a given shift k, the gradient is 2·a_{k‑i} (zero outside the support)."""
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k_max - i_range
        grad[i_range] = 2.0 * a[j]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the O(n log n) algorithm
    of Duchi et al. (2008)."""
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _trim_sequence(seq: np.ndarray, eps: float = 1e-14) -> np.ndarray:
    """Remove leading and trailing entries that are ≤ eps (treated as zero)."""
    if seq.size == 0:
        return seq
    mask = seq > eps
    if not np.any(mask):
        # all entries negligible – return a single tiny element
        return np.array([0.0])
    start = np.argmax(mask)
    end = seq.size - np.argmax(mask[::-1]) - 1
    return seq[start : end + 1]


def _score_from_vector(a: np.ndarray) -> float:
    """Exact score used by ``evaluate_sequence`` (but computed with FFT)."""
    n = a.size
    if n == 0:
        return np.inf
    conv = _conv_self_fft(a, _fft_len_for_n(n))
    max_conv = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (sum_a ** 2)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.  The algorithm
    combines exponentiated‑gradient, Lp‑smooth gradient descent and
    simulated‑annealing style random moves.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector (unit sum)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            cand0 = np.clip(cand0, 0.0, None)
            if cand0.sum() == 0.0:
                cand0 = np.full_like(cand0, 1.0 / cand0.size)
            else:
                cand0 = cand0 / cand0.sum()
            cand0 = _trim_sequence(cand0, eps=1e-14)
            a = cand0.copy()
        except Exception:
            # fallback to a modest uniform vector
            n0 = 1024
            a = np.full(n0, 1.0 / n0, dtype=np.float64)
    else:
        n0 = 1024
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size
    fft_len_self = _fft_len_for_n(n)
    fft_len_full = _fft_len_for_full(n, 2 * n - 1)  # for Lp‑gradient

    # ------------------------------------------------------------------
    # 2) Book‑keeping of best solution
    # ------------------------------------------------------------------
    cur = a.copy()
    cur_score = _score_from_vector(cur)
    best = cur.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = last_improve

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # simulated‑annealing temperature schedule
    T_start = 0.02
    T_end = 1e-7

    # Exponentiated‑gradient step‑size schedule
    eta_eg_start = 1.0
    eta_eg_end = 0.001

    # Lp‑gradient step‑size schedule
    eta_lp_start = 0.10
    eta_lp_end = 0.001

    # move probabilities
    prob_eg = 0.30
    prob_lp = 0.30
    prob_pair = 0.20
    prob_block = 0.20

    restart_gap = 30.0  # seconds without improvement → restart
    min_delta_frac = 0.3  # max fraction of a[i] that can be moved in a pair step

    # p‑schedule for Lp‑approximation
    p_start = 4
    p_end = 12

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # temperature and step‑size schedules (linear)
        T = T_start * (1.0 - elapsed / total) + T_end * (elapsed / total)
        eta_eg = eta_eg_start * (1.0 - elapsed / total) + eta_eg_end * (elapsed / total)
        eta_lp = eta_lp_start * (1.0 - elapsed / total) + eta_lp_end * (elapsed / total)
        p = int(p_start + (p_end - p_start) * (elapsed / total))

        # --------------------------------------------------------------
        # Choose a move type
        # --------------------------------------------------------------
        r = rng.random()
        if r < prob_eg:
            # ---- Exponentiated‑gradient on the active index ----
            conv_cur = _conv_self_fft(cur, fft_len_self)
            k_max = int(np.argmax(conv_cur))
            grad = _gradient_max(cur, k_max)
            factor = np.exp(-eta_eg * grad)
            cand = cur * factor
            cand = np.clip(cand, 0.0, None)
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand /= cand_sum
            else:
                cand = np.full_like(cand, 1.0 / cand.size)

        elif r < prob_eg + prob_lp:
            # ---- Lp‑smooth gradient descent (smooth max) ----
            conv_cur = _conv_self_fft(cur, fft_len_self)
            # weight w = conv^{p‑1} (keep zeros as zeros)
            w = np.power(conv_cur, p - 1, dtype=np.float64)
            w_rev = w[::-1]
            # full convolution of a with w_rev
            grad_full = _conv_full_fft(cur, w_rev, fft_len_full)
            # indices that correspond to Σ_j a_j w_{i+j}
            idx = 2 * n - 2 - np.arange(n)
            grad = 2.0 * p * grad_full[idx]
            # gradient descent step (move opposite to gradient)
            cand = cur - eta_lp * grad
            cand = _project_simplex(cand)  # back to the simplex

        elif r < prob_eg + prob_lp + prob_pair:
            # ---- Random pairwise mass transfer ----
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(cur[i], min_delta_frac * cur[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    cand = cur.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = _project_simplex(cand)
                else:
                    cand = cur.copy()
            else:
                cand = cur.copy()

        else:
            # ---- Random contiguous block multiplicative perturbation ----
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.15))
            cand = cur.copy()
            cand[start : start + width] *= factor
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)

        # --------------------------------------------------------------
        # Evaluate candidate and accept / reject
        # --------------------------------------------------------------
        cand_score = _score_from_vector(cand)

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # simulated‑annealing acceptance
            if rng.random() < np.exp((cur_score - cand_score) / max(T, 1e-12)):
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score

            if cand_score < best_score - 1e-12:
                best = cand.copy()
                best_score = cand_score
                last_improve = now
                elapsed_disp = now - start_time
                print(f"[{elapsed_disp:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart if stuck too long
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            # jitter the best solution and continue from there
            cur = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            cur = cur * noise
            cur = _project_simplex(cur)
            cur_score = _score_from_vector(cur)
            last_improve = now
            elapsed_disp = now - start_time
            print(f"[{elapsed_disp:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final post‑processing – trim negligible leading/trailing mass
    # ------------------------------------------------------------------
    best_trimmed = _trim_sequence(best, eps=1e-14)
    best_trimmed = np.clip(best_trimmed, 0.0, 1000.0)

    # Return as a plain Python list of floats
    return [float(v) for v in best_trimmed]
