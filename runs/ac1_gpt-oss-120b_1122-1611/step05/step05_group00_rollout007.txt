"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimizes

    score = 2·n·max_k (a ∗ a)_k

where `∗` is the linear convolution and `∑a_i = 1`.
The search proceeds in three stages:

1. **Analytic families** – generate candidates from two simple
   one‑parameter families:
      * a_i = (i+1)^r – i^r   (r∈[0.5,0.95])
      * a_i = (i+1)^(-p) – (i+2)^(-p)   (p∈[0.1,2.0])
   (Both are decreasing, non‑negative and sum to 1 after normalisation.)
   The best of these is kept as the initial point.

2. **Exponentiated‑gradient refinement** – minimise the true L∞‑objective
   by a mirror‑descent (exponentiated‑gradient) update.  The gradient of
   the soft‑maximum `log‑sum‑exp(β·conv)` is computed efficiently via a
   single FFT‑based convolution (correlation).  The temperature `β` is
   raised from a small value to a large one so that the surrogate converges
   to the true max.  The learning rate `η` decays linearly.

3. **Exploration & restarts** – occasional block perturbations,
   pairwise mass transfers and soft restarts (jittering the current best)
   keep the search from stagnating.

The routine respects the given time budget (≈1000 s) and returns the
best sequence found, already clipped to the limits imposed by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two size that accommodates a linear
    convolution of length `2·n‑1`."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable softmax with temperature `beta`."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute gradient 2·a ⋆ w  (correlation), where
    `w` has length 2·n‑1 and `a` length n.
    This uses a single FFT‑based convolution.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1                       # length of full convolution
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    # gradient = 2 * corr_full[n-1 : n-1+n]
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value for a probability vector `a`."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Candidate generators for the analytic families
# ----------------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray:
    """Shape a_i = (i+1)^r - i^r  (decreasing for 0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_shape_p(p: float, n: int) -> np.ndarray:
    """Shape a_i = (i+1)^(-p) - (i+2)^(-p) (p>0)."""
    # Create f_k = (k+1)^(-p) for k = 0..n
    k = np.arange(1, n + 2, dtype=np.float64)          # 1 .. n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]                                 # length n
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    cand = cand / cand.sum()
                    sc = _score(cand, _fft_len_for_n(cand.size))
                    if sc < best_score:
                        best_a, best_score = cand, sc
                        n = cand.size
                        fft_len = _fft_len_for_n(n)
                        print(f"[init] using height_sequence_1 score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Main optimisation – exponentiated gradient on a soft‑max surrogate
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin

    # schedules (linear decay/increase)
    eta_start, eta_end = 1.0, 0.005
    beta_start, beta_end = 0.2, 5000.0

    # hyper‑parameters for exploration
    block_prob = 0.05
    pair_prob = 0.10
    restart_gap = 180.0          # seconds without improvement → restart

    cur_a = best_a.copy()
    cur_score = best_score
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # Linear schedules
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # Convolution and exact score (for bookkeeping)
        conv = _conv_self(cur_a, fft_len)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        # Update best solution
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # Soft‑max weights over the full convolution (smooth approximation of max)
        w = _softmax(conv, beta)

        # Gradient of the smooth surrogate (2·a⋆w)
        grad = _gradient_via_correlation(cur_a, w)

        # Exponentiated‑gradient (mirror descent) update
        factor = np.exp(-eta * grad)
        cur_a = cur_a * factor
        cur_a = np.clip(cur_a, 0.0, None)
        s = cur_a.sum()
        if s == 0.0:
            cur_a = np.full_like(cur_a, 1.0 / n)
        else:
            cur_a = cur_a / s

        # ------------------------------------------------------------------
        # Random exploration moves
        # ------------------------------------------------------------------
        if rng.random() < block_prob:                 # block perturbation
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.25))
            cur_a[start:start + width] *= mult
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = cur_a / cur_a.sum()

        if rng.random() < pair_prob:                  # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > 1e-14:
                delta = rng.random() * cur_a[i] * 0.5
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = np.clip(cur_a, 0.0, None)
                cur_a = cur_a / cur_a.sum()

        # ------------------------------------------------------------------
        # Soft restart when stagnation is detected
        # ------------------------------------------------------------------
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            cur_a = best_a * jitter
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = cur_a / cur_a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (already normalised, safe for evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
