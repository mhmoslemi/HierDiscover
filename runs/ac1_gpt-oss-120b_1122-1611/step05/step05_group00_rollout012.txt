"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.  The method consists of two stages:

1. **Coarse parametric sweep** – a two‑parameter family
   a_i ∝ (i+α)^(-γ) (α>0, 0<γ<1) is sampled on the probability simplex.
   The best vector from this sweep is used as a warm start.

2. **Refinement with Adam** – the true L∞‑objective
   f(a)=2·n·max_k (a∗a)_k is approximated by a soft‑max surrogate
   using a temperature β that is increased from a small to a very large
   value during the run.  An Adam optimiser is applied to the
   log‑weights w (a_i = softmax(w)), which guarantees positivity and
   the simplex constraint.  Random jitter, pairwise mass transfers and
   soft restarts keep the search from stagnating.

The routine respects a time budget (default 1000 s), prints progress
information and finally returns the best sequence found (clipped to the
limits enforced by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a, returned length = 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolution entries:
#   Σ_k w_k (a∗a)_k  =  2·a ⋆ w   (correlation)
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Return ∇_a Σ_k w_k (a∗a)_k."""
    n = a.size
    L = w.size                # = 2·n‑1
    # Linear correlation length needed = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # a reversed for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    evaluate_sequence before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    n = max(2, n)

    # --------------------------------------------------------------
    # 2) Helper to compute the objective (fast, without the security
    #    wrapper of evaluate_sequence)
    # --------------------------------------------------------------
    fft_len_conv = _fft_len_for_n(n)

    def _score(a_vec):
        conv = _conv_self(a_vec, fft_len_conv)
        return 2.0 * n * float(conv.max())

    # --------------------------------------------------------------
    # 3) Initial candidates – parametric family a_i ∝ (i+α)^(-γ)
    # --------------------------------------------------------------
    best_a = None
    best_obj = np.inf

    # include the analytic sqrt‑difference start as a fallback
    i = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(i + 1.0) - np.sqrt(i)
    a_sqrt /= a_sqrt.sum()
    score_sqrt = _score(a_sqrt)
    best_a = a_sqrt.copy()
    best_obj = score_sqrt

    # also consider a previously saved construction if it exists
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand /= s
                else:
                    raise ValueError
                cand_score = _score(cand)
                if cand_score < best_obj:
                    best_obj = cand_score
                    best_a = cand.copy()
        except Exception:
            pass

    # parametric sweep (α,γ)
    alphas = np.logspace(-2, 2, 20)          # 0.01 … 100
    gammas = np.linspace(0.30, 0.99, 30)     # 0.30 … 0.99
    i_vec = np.arange(n, dtype=np.float64)

    start_time = time.monotonic()
    # spend at most ~8 % of the budget on the sweep
    sweep_deadline = start_time + 0.08 * budget_s

    for alpha in alphas:
        # stop early if time budget exceeded
        if time.monotonic() > sweep_deadline:
            break
        # pre‑compute i+α to avoid repeated work inside inner loop
        base = (i_vec + alpha)
        for gamma in gammas:
            a_raw = base ** (-gamma)
            s = a_raw.sum()
            if s == 0.0:
                continue
            a_vec = a_raw / s
            obj = _score(a_vec)
            if obj < best_obj:
                best_obj = obj
                best_a = a_vec.copy()
                # tiny progress print
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] sweep improvement: score={best_obj:.6f} (α={alpha:.3g},γ={gamma:.3f})")

    # --------------------------------------------------------------
    # 4) Adam optimisation on the log‑softmax parametrisation.
    # --------------------------------------------------------------
    # initialise log‑weights
    eps = 1e-15
    w = np.log(best_a + eps)            # shape (n,)
    m = np.zeros_like(w)                # first moment
    v = np.zeros_like(w)                # second moment
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0

    # hyper‑parameters for the schedule
    lr_start = 0.12
    lr_end = 0.003
    beta_start = 0.5          # soft‑max temperature start
    beta_end = 3000.0         # final temperature (very sharp)
    block_perturb_prob = 0.03
    pair_move_prob = 0.07
    restart_gap = 150.0       # seconds without improvement → restart

    best_seq = best_a.copy()
    best_score = best_obj
    last_improve = time.monotonic()

    # main optimisation loop
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin
    while time.monotonic() < deadline:
        # ---- compute convolution and current objective ----
        a = np.exp(w)
        a_sum = a.sum()
        if a_sum == 0.0:                 # numerical safety
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- record best ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule ----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total
        lr = lr_start * (1.0 - frac) + lr_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ---- soft‑max weights over convolution entries ----
        # improve numerical stability by subtracting max before exp
        exp_arg = beta * (conv - cur_max)
        # cap exponent to avoid overflow (numpy will clip to inf otherwise)
        exp_arg = np.clip(exp_arg, -700, 700)
        w_raw = np.exp(exp_arg)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            p = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            p = w_raw / w_sum

        # ---- gradient of the soft‑max surrogate ----
        grad_a = _gradient_correlation(a, p)          # shape (n,)

        # ---- convert to gradient w.r.t. log‑weights (softmax trick) ----
        avg = np.dot(a, grad_a)                       # scalar = ⟨a,∇f⟩
        grad_w = a * (grad_a - avg)                   # shape (n,)

        # ---- Adam update ----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional random block perturbation ----
        if rng.random() < block_perturb_prob:
            blk_width = rng.integers(1, max(2, n // 12))
            blk_start = rng.integers(0, n - blk_width + 1)
            factor = np.exp(rng.normal(0.0, 0.18))
            # apply directly on the underlying weights to keep geometry
            w[blk_start:blk_start + blk_width] += np.log(factor)

        # ---- occasional pairwise mass transfer ----
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # compute current probabilities
                a_i = a[i]
                a_j = a[j]
                if a_i > 1e-14:
                    delta = rng.random() * a_i * 0.4   # move up to 40 % of a_i
                    # modify log‑weights so that mass is transferred
                    # using a simple approximation:
                    w[i] += np.log(max(eps, a_i - delta) / max(eps, a_i))
                    w[j] += np.log((a_j + delta) / max(eps, a_j))

        # ---- soft restart if we have been stuck too long ----
        if now - last_improve > restart_gap:
            # jump back to the best so far and inject a small jitter
            a = best_seq.copy()
            w = np.log(a + eps)
            jitter = np.exp(rng.normal(0.0, 0.09, size=n))
            w += np.log(jitter)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # final safety clamp
    return [float(v) for v in best_seq]
