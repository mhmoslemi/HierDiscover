Propose an improved step‑function sequence.

Algorithm
---------
* **Initialisation** – start from the previous construction if it exists,
  otherwise draw a Dirichlet vector of length ≈ 3800 and normalise it.

* **Phase 1 – soft‑max mirror descent** (≈ 40 % of the time budget)  
  Repeatedly minimise a smooth approximation of  max(a∗a)  
  by exponentiated‑gradient updates.  The soft‑max temperature β is
  increased linearly, the step size η is decreased, and a tiny log‑normal
  block jitter is applied from time to time.

* **Phase 2 – deterministic hill‑climbing** (≈ 55 % of the budget)  
  The current convolution `c = a*a` is kept explicitly.  For the index
  `kmax = argmax(c)` we compute the partner values
  `p[i]=a[kmax‑i]` (zero outside the support) and the contributions
  `c_i = a[i]·p[i]`.  A donor index is sampled proportionally to `c_i`,
  a recipient is chosen among 20 random candidates that have the smallest
  partner value, and a small amount of mass δ is transferred from donor
  to recipient.

  The convolution is updated *exactly* in O(n) time using the identity

        (a+Δe_i+Δe_j)*(a+Δe_i+Δe_j)
        = a*a + 2Δ_i·a_{·‑i} + 2Δ_j·a_{·‑j}
          + Δ_i²·δ_{·,2i} + Δ_j²·δ_{·,2j} + 2Δ_iΔ_j·δ_{·,i+j}.

  If the move reduces the objective `2·n·max(c)`, it is kept;
  otherwise it is reverted.  Every 200 accepted moves a full FFT
  recomputation of the convolution is performed to avoid drift.

* **Restarts** – if no improvement occurs for 30 s the current state is
  re‑initialised from the best sequence seen so far with a moderate
  log‑normal perturbation.

* **Multiple lengths** – the routine tries a few lengths

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (fft_len must be ≥ 2·len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

            a *= np.exp(-eta * grad)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / n)

            # occasional jitter
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 2.4) Phase 2 – deterministic hill‑climbing on the true max‑norm
        # --------------------------------------------------------------
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] n={n} initial score = {best_score:.6f}")

        # bookkeeping for stagnation detection
        last_improve = time.monotonic()
        moves_since_full = 0

        # main hill‑climbing loop
        while time.monotonic() < deadline:
            # ----------------------------------------------------------
            # 2.4.1) Identify the current peak and its partner values
            # ----------------------------------------------------------
            k_max = int(conv.argmax())
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            idx = np.arange(low, high + 1)
            partner = np.empty(n, dtype=np.float64)
            partner.fill(0.0)
            partner[idx] = a[k_max - idx]

            contrib = a * partner
            total_contrib = contrib.sum()
            if total_contrib == 0.0:
                # nothing contributes to the peak – random perturbation
                a = _random_block_perturb(a, rng)
                conv = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * conv.max()
                continue

            # ----------------------------------------------------------
            # 2.4.2) Choose donor (biased by contribution) and recipient
            # ----------------------------------------------------------
            donor = rng.choice(n, p=contrib / total_contrib)

            # a small random set of candidates for the recipient
            cand = rng.choice(n, size=min(20, n), replace=False)
            cand = cand[cand != donor]
            if cand.size == 0:
                continue
            recipient = cand[np.argmin(partner[cand])]

            # ----------------------------------------------------------
            # 2.4.3) Propose a mass transfer
            # ----------------------------------------------------------
            max_delta = 0.20 * a[donor]          # up to 20 % of donor's mass
            if max_delta <= 1e-14:
                continue
            delta = rng.random() * max_delta
            if delta <= 0.0:
                continue

            # ----------------------------------------------------------
            # 2.4.4) Apply the move using the exact O(n) update formula
            # ----------------------------------------------------------
            # store a backup of the convolution (cheap copy, ~2n floats)
            conv_before = conv.copy()
            # keep donor/recipient values for a possible revert
            donor_old = a[donor]
            recip_old = a[recipient]

            # linear part: 2·Δ·a (use the *current* a – still the old one)
            conv[donor: donor + n] += 2.0 * (-delta) * a
            conv[recipient: recipient + n] += 2.0 * delta * a

            # quadratic corrections (see derivation in the header)
            conv[2 * donor] += delta * delta
            conv[2 * recipient] += delta * delta
            conv[donor + recipient] += -2.0 * delta * delta

            # finally update the vector a
            a[donor] -= delta
            a[recipient] += delta

            # ----------------------------------------------------------
            # 2.4.5) Evaluate the new objective
            # ----------------------------------------------------------
            new_max = conv.max()
            new_score = 2.0 * n * new_max

            if new_score < cur_score:
                # accepted move
                cur_score = new_score
                moves_since_full += 1
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_seq = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{time.monotonic() - start_time:6.2f}s] new best score = {best_score:.6f}")
            else:
                # reject – revert both a and the convolution
                conv = conv_before
                a[donor] = donor_old
                a[recipient] = recip_old

            # ----------------------------------------------------------
            # 2.4.6) Periodic full recomputation (to avoid drift)
            # ----------------------------------------------------------
            if moves_since_full >= 200:
                conv = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * conv.max()
                moves_since_full = 0

            # ----------------------------------------------------------
            # 2.4.7) Optional small‑temperature random jumps (helps escape)
            # ----------------------------------------------------------
            if rng.random() < 0.005:
                a = _random_block_perturb(a, rng)
                conv = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * conv.max()

            # ----------------------------------------------------------
            # 2.4.8) Restart on prolonged stagnation
            # ----------------------------------------------------------
            if time.monotonic() - last_improve > 30.0:
                # re‑initialise from the best seen so far with moderate noise
                a = best_seq.copy()
                a = _random_block_perturb(a, rng)
                a /= a.sum()
                conv = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * conv.max()
                last_improve = time.monotonic()
                print(f"[{time.monotonic() - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Return the best sequence (clipped to the limits of the evaluator)
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but fall back to a uniform vector
        best_seq = np.full(3840, 1.0 / 3840, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
