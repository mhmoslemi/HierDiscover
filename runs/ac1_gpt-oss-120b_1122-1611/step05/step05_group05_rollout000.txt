"""
Sub‑gradient descent on the L∞‑norm of the self‑convolution.
The objective to minimise is

    score = 2 * n * max_k (a * a)[k]          (sum(a) = 1)

which is exactly the quantity returned by ``evaluate_sequence`` up to the
fixed normalisation.  The function treats the step‑function as a
probability vector, computes the maximal convolution entry, and moves mass
away from the indices that contribute to that maximum.
A Euclidean projection onto the probability simplex keeps the vector
feasible after every update.  Small random jitter and periodic restarts
help to escape plateaux.  The routine respects the 1000 s time budget and
returns the best sequence found.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a 1‑D array onto the simplex {x ≥ 0, sum(x) = 1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: fast self‑convolution via FFT (real input → real output)
# ----------------------------------------------------------------------
def _autocorr_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the discrete linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    # length of FFT: next power of two >= 2*n-1
    N = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values; clip.
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Return a non‑negative list of heights that (approximately) minimises
    ``evaluate_sequence``.  The algorithm runs a sub‑gradient descent on
    the maximal entry of the self‑convolution, keeping the vector on the
    probability simplex.  Random jitter and timed restarts are used to
    avoid stagnation.  The best candidate encountered before the deadline
    is returned.
    """
    rng = np.random.default_rng(seed)
    # give ourselves a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # 1) Initialise the vector a
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # a modest size keeps each iteration cheap; 1024 works well.
        n = 1024
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size
        a = np.clip(a, 0.0, None)

    # Normalise to a probability vector (sum = 1)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    conv = _autocorr_fft(a)
    best_score = 2.0 * n * np.max(conv)          # because sum(a)=1
    best = a.copy()
    last_improve = start_time

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # 3) Hyper‑parameters
    # --------------------------------------------------------------
    step0 = 0.55          # initial step size for sub‑gradient updates
    restart_gap = 30.0    # seconds without improvement → perform a restart
    max_iter = 10_000_000  # just a huge safety ceiling
    it = 0

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline and it < max_iter:
        # ---- evaluate the current vector ----
        conv = _autocorr_fft(a)
        M = np.max(conv)
        cur_score = 2.0 * n * M

        # ---- keep the best seen so far ----
        if cur_score < best_score - 1e-12:  # a tiny tolerance for floating point
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- sub‑gradient of the max‑convolution entry ----
        peaks = np.flatnonzero(conv == M)
        if peaks.size == 0:
            break  # should never happen

        grad = np.zeros_like(a)
        # for each peak position k, add 2*a_{k-i} to grad_i (where defined)
        for k in peaks:
            start_i = max(0, k - (n - 1))
            end_i = min(k, n - 1)
            idx = np.arange(start_i, end_i + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # ---- step‑size schedule (diminishing) ----
        step = step0 / math.sqrt(it + 1)

        # ---- stochastic jitter (rare) ----
        if rng.random() < 0.015:  # 1.5 % of the iterations
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * jitter
            a = _project_simplex(a)

        # ---- gradient step and projection back onto simplex ----
        cand = a - step * grad
        cand = _project_simplex(cand)

        # ---- acceptance (always accept the projected point) ----
        a = cand
        it += 1

        # ---- periodic restart if stuck for a while ----
        if time.monotonic() - last_improve > restart_gap:
            # jitter the current best and continue from there
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * noise
            a = _project_simplex(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best candidate (clip to allowed range)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
