def proj_simplex(v):
    # ensure non-negative sum 1
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def compute_gradient(a, conv_self, alpha):
    # conv_self length 2n-1
    # compute weights
    max_val = np.max(alpha * conv_self)  # for scaling
    exp_vals = np.exp(alpha * conv_self - max_val)
    weight = exp_vals / exp_vals.sum()
    # reverse weight
    weight_rev = weight[::-1]
    fft_len = 1 << ((len(a) + len(weight_rev) - 1 - 1).bit_length())
    # use convolution via FFT
    # zero pad
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(weight_rev, n=fft_len)
    conv = np.fft.irfft(Fa * Fw, n=fft_len)
    # conv length = fft_len truncated to len(a)+len(weight_rev)-1
    conv = conv[:len(a)+len(weight_rev)-1]
    # take gradient entries:
    # gradient_i = 2 * conv[2*n-2 - i] for i = 0..n-1
    n = len(a)
    grad = 2.0 * conv[2*n-2 - np.arange(n)]
    return grad
```

"""
Propose a step‑function by (1) smoothing the max‑convolution with a
soft‑max (log‑sum‑exp) and performing projected gradient descent on the
simplex, (2) gradually turning the soft‑max into a hard max (weighting in
the true sub‑gradient), (3) adding occasional random block perturbations,
pairwise mass transfers and soft restarts.  The optimiser runs until the
time budget expires and returns the best vector seen.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions – all top‑level (no closures)
# ------------------------------------------------------------

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, sum x_i = 1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of two
    vectors of lengths *n* and *2·n‑1* (i.e. length *3·n‑2*)."""
    size = 3 * n - 2
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of *a* using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return a probability vector proportional to exp(alpha*conv).
    Numerically stable via log‑sum‑exp trick.
    """
    x = alpha * conv
    xmax = np.max(x)
    exp_shift = np.exp(x - xmax)
    probs = exp_shift / np.sum(exp_shift)
    return probs


def _gradient_soft(a: np.ndarray, weight_rev: np.ndarray,
                   fa: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the soft‑max loss.
    The formula reduces to 2*correlation(weight, a) evaluated at the
    appropriate offsets.
    """
    # FFT of reversed weight (already reversed)
    fw = np.fft.rfft(weight_rev, n=fft_len)

    # Linear convolution a * weight_rev (length 3n‑2)
    conv_rev = np.fft.irfft(fa * fw, n=fft_len)[: 3 * a.size - 2]

    n = a.size
    # slice [n‑1 : 2n‑1) (forward order) and reverse to obtain
    # gradient_i = 2 * conv_rev[2n‑2‑i]
    slice_part = conv_rev[n - 1 : 2 * n - 1]   # length n, forward order
    grad = 2.0 * slice_part[::-1]
    return grad


def _gradient_hard(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the *hard* max term.
    For shift k_max the derivative is 2·a_{k_max‑i} (zero outside support).
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k_max - i_range
        grad[i_range] = 2.0 * a[j]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_move(a: np.ndarray, rng: np.random.Generator,
                   temp: float, cur_score: float) -> np.ndarray:
    """Attempt a random mass transfer between two indices and accept it if
    it improves the true score or with a Metropolis probability."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta

    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    cand = np.clip(cand, 0.0, None)
    cand /= cand.sum()

    # evaluate true score (max‑convolution) – cheap with FFT
    conv = _conv_self_fft(cand, _fft_len_for_n(n))
    new_score = 2.0 * n * conv.max()

    if new_score < cur_score:
        return cand
    # uphill move with temperature
    if rng.random() < np.exp((cur_score - new_score) / max(temp, 1e-12)):
        return cand
    return a


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method combines a soft‑max approximation of the max‑convolution
    with projected gradient descent, occasional stochastic moves and
    soft restarts.  It returns the best vector found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector (unit sum)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            # fall back to uniform of length 2000
            n = 2000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)                # reuse throughout the run

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (tuned empirically)
    # ------------------------------------------------------------------
    eta_start = 1.0          # initial learning rate
    eta_end = 0.001          # final learning rate
    alpha_start = 5.0        # start with a very soft max
    alpha_end = 300.0        # end with a rather hard max
    momentum = 0.9
    block_perturb_prob = 0.02
    pair_move_prob = 0.07
    restart_gap = 40.0       # seconds without improvement → soft restart
    temp_start = 0.4
    temp_end = 0.001
    jitter_std = 0.12        # multiplicative log‑normal jitter at restart
    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution -----
        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))
        max_conv = float(conv[k_max])
        cur_score = 2.0 * n * max_conv

        # ----- keep best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule for parameters -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        t = elapsed / total                       # 0 … 1

        eta = eta_start * (1.0 - t) + eta_end * t
        alpha = alpha_start * (1.0 - t) + alpha_end * t
        temp = temp_start * (1.0 - t) + temp_end * t
        hard_weight = (alpha - alpha_start) / (alpha_end - alpha_start)  # 0→1

        # ----- FFT of current vector (re‑used) -----
        fa = np.fft.rfft(a, n=fft_len)

        # ----- soft‑max gradient -----
        weights = _softmax_weights(conv, alpha)          # length 2n‑1
        weights_rev = weights[::-1]
        grad_soft = _gradient_soft(a, weights_rev, fa, fft_len)

        # ----- hard max gradient (sub‑gradient at the active shift) -----
        grad_hard = _gradient_hard(a, k_max)

        # ----- blend the two gradients -----
        grad = (1.0 - hard_weight) * grad_soft + hard_weight * grad_hard

        # ----- momentum smoothing -----
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- gradient step -----
        a = a - eta * grad
        a = proj_simplex(a)                     # enforce simplex constraints

        # ----- occasional stochastic moves -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < pair_move_prob:
            a = _pairwise_move(a, rng, temp, cur_score)

        # ----- soft restart when stuck -----
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter multiplicatively (log‑normal)
            a *= np.exp(rng.normal(0.0, jitter_std, size=n))
            a = proj_simplex(a)
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
