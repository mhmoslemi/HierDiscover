def propose_candidate(seed=1234, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0,budget_s - 5.0)
    best_score = float('inf')
    best_seq = None
    best_n = None

    # phases: each of a list of lengths and time fraction
    phases = [
        (2000, 0.25),
        (4000, 0.30),
        (8000, 0.45)
    ]
    overall_start = start

    for n, frac in phases:
        # allocate time slice
        phase_deadline = start + budget_s * frac
        if phase_deadline > deadline:
            phase_deadline = deadline
        # compute fft length
        fft_len = _next_pow2(2*n - 1)
        # number of restarts
        restarts = 4
        for r in range(restarts):
            # random initial a
            a = rng.random(n)
            a = a / a.sum()
            # gradient descent loops
            # we will compute a few hundred (maybe 5000) iterations but break on time
            # maintain momentum
            grad_prev = np.zeros_like(a)
            # loops
            while time.monotonic() < phase_deadline:
                # compute conv and score
                conv = _self_conv_fft(a, fft_len)
                cur_max = conv.max()
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    best_n = n
                    print(f"[{time.monotonic()-overall_start:.2f}s] new global best score = {best_score:.6f} (n={n})")
                # schedule
                elapsed = time.monotonic() - start
                total = budget_s
                frac_overall = elapsed / total
                # set eta and beta based on frac_overall
                eta = eta_start * (1 - frac_overall) + eta_end * frac_overall
                beta = beta_start + (beta_end - beta_start) * frac_overall
                # compute softmax weights
                # subtract max to avoid overflow
                max_val = cur_max
                # using conv as array
                exps = np.exp(beta * (conv - max_val))
                sum_exps = exps.sum()
                if sum_exps == 0:
                    w = np.zeros_like(conv)
                    w[np.argmax(conv)] = 1.0
                else:
                    w = exps / sum_exps
                # compute gradient via correlation
                grad = _gradient_from_weights(a, w)
                # momentum
                grad = momentum * grad_prev + (1 - momentum) * grad
                grad_prev = grad
                # exponentiated gradient update
                factor = np.exp(-eta * grad)
                a = a * factor

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Search on several problem sizes** – we run independent searches for
   n = 2000, 4000 and 8000 (larger n gives more freedom to flatten the
   self‑convolution).

2. **Smooth‑max (soft‑max) mirror descent** on the probability simplex.
   * The objective is `max_k (a*a)_k`.  We approximate the max by a
     soft‑max with temperature β that is slowly increased.
   * The gradient of `∑_k w_k (a*a)_k` (where `w` are the soft‑max
     weights) is computed by a single FFT‑based correlation.
   * An exponentiated‑gradient (mirror‑descent) step is performed and the
     result is projected back onto the simplex.
   * Momentum and occasional multiplicative block perturbations help to
     escape shallow plateaus.

3. **Local refinement** – after the gradient phase we run a fast
   O(n) pairwise‑mass‑transfer heuristic that updates the self‑convolution
   exactly.  Any improving move is kept; uphill moves are ignored (the
   temperature is already very low at this stage).

4. **Random restarts** – for each n we try a few deterministic starts
   (ramp, quadratic, sinusoidal) and a few random Dirichlet starts.
   The best sequence over all phases is returned.

All heavy work (convolution, gradient) uses FFT, so a single iteration
for n = 8000 costs only a few milliseconds, allowing thousands of
iterations within the 1000 s budget.

The returned list respects the limits of ``evaluate_sequence`` (non‑negative,
≤ 1000) and is automatically normalised – the evaluation metric is
homogeneous, so the normalisation does not affect optimality.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – pure functions, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries (non‑negative, clipped).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negative values caused by floating‑point noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Implements grad_j = 2·Σ_i a_i·w_{i+j}.
    """
    n = a.size
    Lw = w.size                       # = 2*n‑1
    wr = w[::-1]                      # reversed for correlation
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    corr_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    # keep only the required part
    corr_full = corr_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)        # mapping to obtain correlation result
    grad = 2.0 * corr_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _pairwise_transfer_exact(a: np.ndarray, conv: np.ndarray,
                             rng: np.random.Generator,
                             max_delta_frac: float = 0.3):
    """
    Move a random amount of mass from a donor index i to a receiver index j.
    The convolution vector is updated in O(n) time.
    Returns (new_a, new_conv, new_score) if the move strictly improves the
    objective, otherwise the original tuple (a, conv, None).
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Update convolution
    new_conv = conv.copy()
    # donor contribution removed
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    # receiver contribution added
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a
    # diagonal (i,i) and (j,j) corrections
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    # cross term correction when i != j
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    # Update coefficient vector
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)

    new_score = 2.0 * n * new_conv.max()
    cur_score = 2.0 * n * conv.max()
    if new_score < cur_score - 1e-15:
        return a2, new_conv, new_score
    else:
        return a, conv, None

def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found (list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Global best
    best_score = math.inf
    best_seq = None
    best_n = None

    # ------------------------------------------------------------------
    # Problem sizes – we allocate more time to larger n
    # ------------------------------------------------------------------
    # (n, fraction of total budget)
    phases = [
        (2000, 0.25),
        (4000, 0.30),
        (8000, 0.45),
    ]

    # Hyper‑parameters for the smooth‑max descent
    eta_start = 1.0          # initial step size
    eta_end = 0.01           # final step size
    beta_start = 0.2         # start temperature (soft‑max smoothness)
    beta_end = 4e5           # final temperature – approximates true max
    momentum = 0.9
    block_perturb_prob = 0.025   # probability per iteration

    # Number of random restarts per phase (including a few deterministic starts)
    restarts_per_phase = 6

    # ------------------------------------------------------------------
    # Helper to create deterministic initial vectors
    # ------------------------------------------------------------------
    def _make_initial(n, kind):
        if kind == "ramp":
            v = np.arange(1, n + 1, dtype=np.float64)
        elif kind == "quadratic":
            v = np.square(np.arange(1, n + 1, dtype=np.float64))
        elif kind == "sin":
            x = np.linspace(0, math.pi, n, dtype=np.float64)
            v = np.sin(x) ** 2
        else:
            # random Dirichlet (gamma) – avoid exact zeros
            v = rng.random(n) + 1e-6
        v /= v.sum()
        return v

    # ------------------------------------------------------------------
    # Main search loop over phases
    # ------------------------------------------------------------------
    for phase_idx, (n, frac) in enumerate(phases):
        phase_deadline = start_time + budget_s * sum(p[1] for p in phases[:phase_idx + 1])
        phase_deadline = min(phase_deadline, deadline)

        if time.monotonic() >= phase_deadline:
            break

        fft_len = _next_pow2(2 * n - 1)

        # Pre‑compute deterministic initial kinds
        deterministic_kinds = ["ramp", "quadratic", "sin"]
        # Choose restarts (deterministic first, then random)
        for restart_idx in range(restarts_per_phase):
            if time.monotonic() >= phase_deadline:
                break

            if restart_idx < len(deterministic_kinds):
                a = _make_initial(n, deterministic_kinds[restart_idx])
            else:
                a = _make_initial(n, "random")

            # Momentum buffer
            grad_prev = np.zeros_like(a)

            # --------------------------------------------------------------
            # Gradient‑descent (smooth‑max) loop
            # --------------------------------------------------------------
            while time.monotonic() < phase_deadline:
                # Compute self‑convolution (FFT based)
                conv = _self_conv_fft(a, fft_len)
                cur_max = conv.max()
                cur_score = 2.0 * n * cur_max

                # Record global best
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    best_n = n
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] new global best = {best_score:.6f} (n={n})")

                # ------------------------------------------------------------------
                # Scheduling of step size and temperature (based on global time)
                # ------------------------------------------------------------------
                elapsed = time.monotonic() - start_time
                total = budget_s
                overall_frac = elapsed / total
                eta = eta_start * (1.0 - overall_frac) + eta_end * overall_frac
                beta = beta_start + (beta_end - beta_start) * overall_frac

                # ------------------------------------------------------------------
                # Soft‑max weights for the max‑approximation
                # ------------------------------------------------------------------
                # Subtract max to keep exponentials well‑scaled
                exp_terms = np.exp(beta * (conv - cur_max))
                sum_exp = exp_terms.sum()
                if sum_exp > 0.0:
                    w = exp_terms / sum_exp
                else:
                    w = np.zeros_like(conv)
                    w[np.argmax(conv)] = 1.0

                # ------------------------------------------------------------------
                # Gradient (correlation a ⋆ w) with momentum
                # ------------------------------------------------------------------
                grad = _gradient_from_weights(a, w)
                grad = momentum * grad_prev + (1.0 - momentum) * grad
                grad_prev = grad

                # ------------------------------------------------------------------
                # Exponentiated‑gradient (mirror descent) step
                # ------------------------------------------------------------------
                factor = np.exp(-eta * grad)
                a = a * factor
                a = np.clip(a, 1e-15, None)          # keep tiny positive mass
                a = _project_onto_simplex(a)

                # ------------------------------------------------------------------
                # Occasional random block perturbation (helps escape plateaus)
                # ------------------------------------------------------------------
                if rng.random() < block_perturb_prob:
                    a = _random_block_perturb(a, rng)

                # ------------------------------------------------------------------
                # Simple stagnation escape – if no improvement for a while inject jitter
                # ------------------------------------------------------------------
                if best_score < math.inf and (time.monotonic() - start_time) > 60.0:
                    # This cheap check is inexpensive; we only add jitter when the
                    # whole run has been idle for a while.
                    if time.monotonic() - start_time - (best_score / 2) > 120.0:
                        noise = np.exp(rng.normal(0.0, 0.2, size=n))
                        a = a * noise
                        a = _project_onto_simplex(a)
                        grad_prev = np.zeros_like(a)

            # --------------------------------------------------------------
            # Local refinement via exact pairwise mass transfers
            # --------------------------------------------------------------
            # We now take the current best for this restart and run a short SA‑style
            # local search while time permits.
            if best_seq is None:
                continue
            # Start from the best found so far (may belong to a different restart)
            a_ref = best_seq.copy()
            conv_ref = _self_conv_fft(a_ref, fft_len)
            ref_score = 2.0 * n * conv_ref.max()

            # Linear temperature schedule for the short refinement phase
            refine_start = time.monotonic()
            refine_deadline = phase_deadline
            while time.monotonic() < refine_deadline:
                a_ref, conv_ref, new_score = _pairwise_transfer_exact(
                    a_ref, conv_ref, rng, max_delta_frac=0.25)

                if new_score is not None and new_score < ref_score:
                    ref_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = a_ref.copy()
                        best_n = n
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] refinement improved to {best_score:.6f} (n={n})")

                # occasional block perturbation during refinement
                if rng.random() < 0.015:
                    a_ref = _random_block_perturb(a_ref, rng)
                    conv_ref = _self_conv_fft(a_ref, fft_len)
                    ref_score = 2.0 * n * conv_ref.max()
                    if ref_score < best_score:
                        best_score = ref_score
                        best_seq = a_ref.copy()
                        best_n = n
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] refinement block gave {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to limits enforced by evaluate_sequence and return a plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
