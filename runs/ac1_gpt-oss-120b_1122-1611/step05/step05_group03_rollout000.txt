"""
propose_candidate
-----------------

Search for a non‑negative step‑function (vector on the simplex) that
minimises ``evaluate_sequence``.  The algorithm combines three ideas:

1. **Projected sub‑gradient descent on the peak of the self‑convolution**.
   The index ``k_max`` where the autocorrelation attains its maximum is
   identified, its derivative with respect to each coefficient is
   ``g_i = 2 * a_{k_max-i}`` (zero when the partner is out of range).  The
   algorithm moves probability mass from indices with large derivative
   (large partner) to indices with a small derivative (small partner).

2. **Optimal donor‑receiver transfer**.
   For a chosen donor ``i`` and receiver ``j`` the reduction of the peak
   can be estimated analytically:
   ``Δmax ≈ 2·δ·(partner_i – partner_j)``.
   The step size ``δ`` is chosen (up to the donor’s mass) so that the
   current peak is driven down towards the second‑largest convolution
   value.  This yields a much larger improvement per iteration than a
   tiny random step.

3. **Exploration tricks** – occasional log‑normal block jitter,
   occasional random resets, and a decreasing step‑size schedule.

The routine works on the simplex (sum = 1), which makes the objective
scale‑invariant.  The best vector found before the time budget expires is
returned (clipped to the limits of ``evaluate_sequence``).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1)."""
    # Using FFT for speed on larger n
    n = a.size
    size = 1 << (2 * n - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    # Numerical noise can produce tiny negative values; clip to 0.
    np.maximum(conv, 0, out=conv)
    return conv


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    i: int,
                    j: int,
                    delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving ``delta`` mass from index i
    to index j (i != j).  The update uses the exact quadratic formula.
    """
    n = a.size
    new_conv = conv.copy()
    # linear part: -2Δ·a shifted by i  + 2Δ·a shifted by j
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a
    # quadratic part
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          max_width_frac: float = 0.18,
                          sigma: float = 0.12) -> np.ndarray:
    """
    Multiplicative log‑normal jitter on a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    minimises ``evaluate_sequence`` within the given time budget.
    The algorithm is a projected sub‑gradient descent with optimal
    donor‑receiver mass transfers and occasional random perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Initialise the coefficient vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # start from a uniform distribution of moderate size
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    conv = _self_convolution(a)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 4.0)  # safety margin
    last_improve = start_time

    # parameters that will be adapted during the run
    max_candidates = max(8, n // 200)   # number of donors/receivers examined
    min_frac = 0.02                     # minimal fraction of donor mass that can be moved
    start_frac = 0.4                    # initial maximal fraction of donor mass
    restart_gap = 25.0                  # seconds without improvement before a restart
    perturb_prob = 0.017                # probability per iteration of a block jitter

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac_time = elapsed / total

        # ----------------------------------------------------------
        # locate the current maximum of the convolution
        # ----------------------------------------------------------
        k_max = int(np.argmax(conv))
        cur_max = conv[k_max]

        # ----------------------------------------------------------
        # second‑largest value (used as a target for the peak)
        # ----------------------------------------------------------
        tmp = conv.copy()
        tmp[k_max] = -np.inf
        second_max = tmp.max()

        # ----------------------------------------------------------
        # partner values a_{k_max-i}
        # ----------------------------------------------------------
        partner = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # slice assignment for efficiency
        idx_range = np.arange(i_start, i_end + 1)
        partner[idx_range] = a[k_max - idx_range]

        # ----------------------------------------------------------
        # donor / receiver candidate selection
        # ----------------------------------------------------------
        # donors = indices with large partner (and non‑zero mass)
        donor_mask = (a > 1e-12) & (partner > 0)
        if not donor_mask.any():
            # fallback to random perturbation if all partners are zero
            if rng.random() < 0.5:
                a = _random_block_perturb(a, rng)
                conv = _self_convolution(a)
                cur_score = 2.0 * n * conv.max()
                continue
        donor_candidates = np.argpartition(-partner, np.arange(min(max_candidates, donor_mask.sum())))[
            :min(max_candidates, donor_mask.sum())
        ]
        # receivers = indices with small partner
        receiver_candidates = np.argpartition(partner, np.arange(min(max_candidates, n)))[:min(max_candidates, n)]

        # ----------------------------------------------------------
        # evaluate best donor‑receiver pair using a cheap linear estimate
        # ----------------------------------------------------------
        best_improve = 0.0
        best_pair = None
        best_delta = 0.0

        # the maximal allowed fraction of donor mass (shrinks with time)
        max_frac = start_frac * (1.0 - frac_time) + min_frac * frac_time

        for d in donor_candidates:
            pd = partner[d]
            if pd <= 0.0:
                continue
            for r in receiver_candidates:
                if d == r:
                    continue
                pr = partner[r]
                diff = pd - pr
                if diff <= 1e-12:
                    continue
                # theoretical delta that would bring the peak down to second_max
                needed = (cur_max - second_max) / (2.0 * diff)
                # clamp by donor's available mass and by the schedule
                max_delta = min(a[d] * max_frac, needed)
                if max_delta <= 1e-12:
                    continue
                # linear estimate of improvement in the peak
                est_new_max = cur_max - 2.0 * max_delta * diff
                improve = cur_max - est_new_max
                if improve > best_improve:
                    best_improve = improve
                    best_pair = (d, r)
                    best_delta = max_delta

        # ----------------------------------------------------------
        # If we found a promising move, evaluate it exactly
        # ----------------------------------------------------------
        if best_pair is not None:
            i, j = best_pair
            delta = best_delta
            new_conv = _apply_transfer(a, conv, i, j, delta)
            new_max = new_conv.max()
            new_score = 2.0 * n * new_max
            if new_score < cur_score - 1e-12:
                # accept the move
                a[i] -= delta
                a[j] += delta
                conv = new_conv
                cur_score = new_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve - start_time:6.2f}s] new best {best_score:.6f}")

        else:
            # ------------------------------------------------------
            # No good donor‑receiver pair found – fallback random move
            # ------------------------------------------------------
            i = rng.integers(0, n)
            if a[i] <= 1e-12:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_delta = a[i] * max_frac
            delta = rng.random() * max_delta
            new_conv = _apply_transfer(a, conv, i, j, delta)
            new_score = 2.0 * n * new_conv.max()
            if new_score < cur_score - 1e-12:
                a[i] -= delta
                a[j] += delta
                conv = new_conv
                cur_score = new_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve - start_time:6.2f}s] new best {best_score:.6f}")

        # ----------------------------------------------------------
        # Occasionally apply a random block perturbation to escape plateaus
        # ----------------------------------------------------------
        if rng.random() < perturb_prob:
            a_pert = _random_block_perturb(a, rng)
            conv_pert = _self_convolution(a_pert)
            pert_score = 2.0 * n * conv_pert.max()
            if pert_score < cur_score - 1e-12:
                a = a_pert
                conv = conv_pert
                cur_score = pert_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve - start_time:6.2f}s] perturb improved best {best_score:.6f}")

        # ----------------------------------------------------------
        # Restart if we have been stagnant for a while
        # ----------------------------------------------------------
        if time.monotonic() - last_improve > restart_gap:
            # jitter the best solution and start from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _self_convolution(a)
            cur_score = 2.0 * n * conv.max()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart – score reset to {cur_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluate_sequence limits)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
