"""
Hybrid exponentiated‑gradient + periodic linear‑programming improvement.

The search starts from the best known construction (``height_sequence_1``)
or from a uniform distribution.  A fast mirror‑descent loop (exp‑gradient
on the simplex) reduces the peak of the self‑convolution.  Every few
minutes the current vector is used as the fixed matrix ``f`` in a linear
program

    maximise   1ᵀ g
    subject to  f * g ≤ M   (point‑wise self‑convolution bound)
                g ≥ 0

where ``M`` is the present worst convolution value.
The LP yields a direction ``g`` that can safely be mixed into the current
vector.  A small convex combination (the mixing weight is annealed) is
performed and the loop continues.  Random block perturbations and a
soft‑restart mechanism help to escape plateaus.  The best vector seen
throughout the run is returned.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _gradient_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the active (maximum) convolution shift."""
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k_max - i_range
        grad[i_range] = 2.0 * a[j]
    return grad

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    # Width is chosen from 1 … max(2, n//10)
    width = rng.integers(1, max(2, n // 10) + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.2))   # slightly stronger than before
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """Build the (2·n‑1)×n matrix representing the linear map g ↦ f∗g."""
    n = f.size
    # each diagonal d ≥ 0 contains the constant value f[d] on n entries
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int64)      # 0,1,…,n‑1
    A = sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csc')
    return A

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.

    The routine mixes a fast exponentiated‑gradient descent on the simplex
    with periodic linear‑programming steps that generate a provably safe
    descent direction.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector (unit sum)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # fall back to a modest size uniform vector
        default_n = 512
        a = np.full(default_n, 1.0 / default_n, dtype=np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    cur_score = 2.0 * n * conv.max()                # sum(a)=1 ⇒ score = 2·n·max(conv)
    best = a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Gradient‑descent phase
    eta_start = 2.0
    eta_end = 0.001
    momentum = 0.85
    block_perturb_prob = 0.02
    pair_move_prob = 0.07
    restart_gap = 40.0      # seconds without improvement → restart
    temp_start = 0.3        # temperature for occasional uphill Metropolis moves
    temp_end = 0.001

    # Linear‑programming phase
    lp_interval = 250.0     # try an LP roughly every this many seconds
    next_lp_time = start_time + lp_interval
    lp_mix_weights = [0.2, 0.1, 0.05, 0.02, 0.01]   # tried in decreasing order

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # --------------------------------------------------------------
        # a) Gradient‑descent (mirror‑descent) update
        # --------------------------------------------------------------
        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))
        max_conv = float(conv[k_max])
        cur_score = 2.0 * n * max_conv

        # store best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now

        eta = eta_start * (1.0 - elapsed / budget_s) + eta_end * (elapsed / budget_s)
        temp = temp_start * (1.0 - elapsed / budget_s) + temp_end * (elapsed / budget_s)

        # gradient of the active shift
        grad = _gradient_max(a, k_max)

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)

                        a = a_candidate
                        # cur_score will be recomputed next loop

        # soft restart if stuck
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

        # --------------------------------------------------------------
        # b) Periodic linear‑programming step
        # --------------------------------------------------------------
        if now >= next_lp_time:
            # Build convolution matrix for current a
            try:
                A = _build_conv_matrix(a)
                M = max_conv                     # current worst value
                b_ub = np.full(2 * n - 1, M, dtype=np.float64)
                c = -np.ones(n, dtype=np.float64)   # maximise sum(g) == minimise -sum
                bounds = [(0.0, None)] * n
                # Solve the LP
                res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
                if res.success:
                    g_raw = res.x
                    sum_g = g_raw.sum()
                    if sum_g > 1e-12:
                        # Normalise to unit sum (probability vector)
                        g = g_raw / sum_g
                        # Try a few mixing weights and keep the best improvement
                        improved = False
                        best_local = a
                        best_local_score = cur_score
                        for t_mix in lp_mix_weights:
                            mixed = (1.0 - t_mix) * a + t_mix * g
                            mixed = np.clip(mixed, 0.0, None)
                            mixed /= mixed.sum()
                            conv_mix = _conv_self_fft(mixed, fft_len)
                            mix_score = 2.0 * n * conv_mix.max()
                            if mix_score < best_local_score:
                                best_local_score = mix_score
                                best_local = mixed
                                improved = True
                        if improved:
                            a = best_local
                            cur_score = best_local_score
                            # record possible improvement
                            if cur_score < best_score:
                                best_score = cur_score
                                best = a.copy()
                                last_improve = time.monotonic()
                                print(f"[{elapsed:6.2f}s] LP‑mix improved score = {best_score:.6f}")
                else:
                    print(f"[{elapsed:6.2f}s] LP solver failed: {res.message}")
            except Exception as exc:
                print(f"[{elapsed:6.2f}s] LP step exception: {exc}")

            # schedule next LP attempt
            next_lp_time = now + max(lp_interval * rng.uniform(0.8, 1.2), 30.0)

        # end of while loop
        # Allow loop to iterate quickly; we already spent constant time per iteration

    # ------------------------------------------------------------------
    # 5) Return the best vector (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

"""
Hybrid exponentiated‑gradient + periodic linear‑programming improvement.

The search starts from the best known construction (``height_sequence_1``) or,
if unavailable, from a uniform distribution.  A fast mirror‑descent loop
(exponentiated‑gradient on the probability simplex) continuously reduces the
peak of the self‑convolution.  Every few minutes a linear program is solved:

    maximise   1ᵀ g
    subject to  f * g ≤ M          (point‑wise bound with the current worst value)
                g ≥ 0

where ``f`` is the current vector and ``M`` its worst self‑convolution value.
The LP yields a safe descent direction ``g``; a small convex combination of
``f`` and the normalised ``g`` is accepted if it improves the objective.
Random block perturbations, pair‑wise mass transfers and a soft‑restart
mechanism help to escape plateaus.  The best vector observed is returned.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the active (maximum) convolution shift."""
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k_max - i_range
        grad[i_range] = 2.0 * a[j]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10) + 1)        # inclusive lower, exclusive upper
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.2))              # stronger perturbation than before
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """Build the (2·n‑1)×n matrix representing the linear map g ↦ f∗g.
    The matrix is lower‑triangular; each diagonal d (0 ≤ d < n) is constant
    equal to f[d] and has length n."""
    n = f.size
    # each diagonal d holds the constant value f[d] on n entries
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int64)   # 0 … n‑1
    A = sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector (unit sum)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        default_n = 512
        a = np.full(default_n, 1.0 / default_n, dtype=np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    cur_score = 2.0 * n * conv.max()          # sum(a)=1 ⇒ score = 2·n·max(conv)
    best = a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # small safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Gradient‑descent phase
    eta_start = 2.0
    eta_end = 0.001
    momentum = 0.85
    block_perturb_prob = 0.02
    pair_move_prob = 0.07
    restart_gap = 40.0          # seconds without improvement → restart
    temp_start = 0.30
    temp_end = 0.001

    # Linear‑programming phase
    lp_interval = 250.0         # try an LP roughly every this many seconds
    next_lp_time = start_time + lp_interval
    lp_mix_weights = [0.20, 0.10, 0.05, 0.02, 0.01]   # tried in decreasing order

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # a) Exponentiated‑gradient (mirror‑descent) update
        # --------------------------------------------------------------
        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))
        max_conv = float(conv[k_max])
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # schedules for step size and temperature
        eta = eta_start * (1.0 - elapsed / budget_s) + eta_end * (elapsed / budget_s)
        temp = temp_start * (1.0 - elapsed / budget_s) + temp_end * (elapsed / budget_s)

        # sub‑gradient of the active (maximum) shift
        grad = _gradient_max(a, k_max)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update (mirrored on the simplex)
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional random block perturbation
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # occasional pair‑wise mass transfer (Metropolis acceptance)
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate /= a_candidate.sum()
                    conv_cand = _conv_self_fft(a_candidate, fft_len)
                    cand_score = 2.0 * n * conv_cand.max()
                    if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / max(temp, 1e-12)):
                        a = a_candidate

        # soft restart if no progress for a while
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

        # --------------------------------------------------------------
        # b) Periodic linear‑programming step
        # --------------------------------------------------------------
        if now >= next_lp_time:
            # Avoid LP on extremely large vectors (memory explosion)
            if n <= 6000:
                try:
                    A = _build_conv_matrix(a)                # (2n‑1)×n sparse matrix
                    M = max_conv
                    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
                    c = -np.ones(n, dtype=np.float64)        # maximise sum(g)
                    bounds = [(0.0, None)] * n
                    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
                    if res.success:
                        g_raw = res.x
                        sum_g = g_raw.sum()
                        if sum_g > 1e-12:
                            g = g_raw / sum_g                     # normalise to unit sum
                            # try a few mixing weights
                            improved = False
                            best_local = a
                            best_local_score = cur_score
                            for t_mix in lp_mix_weights:
                                mixed = (1.0 - t_mix) * a + t_mix * g
                                mixed = np.clip(mixed, 0.0, None)
                                mixed /= mixed.sum()
                                conv_mix = _conv_self_fft(mixed, fft_len)
                                mix_score = 2.0 * n * conv_mix.max()
                                if mix_score < best_local_score:
                                    best_local_score = mix_score
                                    best_local = mixed
                                    improved = True
                            if improved:
                                a = best_local
                                cur_score = best_local_score
                                if cur_score < best_score:
                                    best_score = cur_score
                                    best = a.copy()
                                    last_improve = time.monotonic()
                                    print(f"[{elapsed:6.2f}s] LP‑mix improved score = {best_score:.6f}")
                    else:
                        print(f"[{elapsed:6.2f}s] LP solver failed: {res.message}")
                except Exception as exc:
                    print(f"[{elapsed:6.2f}s] LP step exception: {exc}")
            else:
                # Too large – skip this round
                pass

            # schedule next LP attempt (some jitter)
            next_lp_time = now + max(lp_interval * rng.uniform(0.8, 1.2), 30.0)

        # loop back; next iteration will recompute everything
        # (the cost of a single FFT for n≈4000 is ~0.1 ms, negligible)

    # ------------------------------------------------------------------
    # 5) Return the best vector (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
