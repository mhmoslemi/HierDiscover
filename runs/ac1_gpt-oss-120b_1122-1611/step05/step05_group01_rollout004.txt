    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_lengths(len1: int, len2: int) -> int:
    """Length of the FFT that can hold the full linear convolution of
    two real vectors of length ``len1`` and ``len2``."""
    return _next_pow2(len1 + len2 - 1)


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors using a pre‑allocated FFT length."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[:x.size + y.size - 1]


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a∗a (linear, no wrap‑around)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# --------------------------------------------------------------
# Simplex projection (used only for a safe fallback)
# --------------------------------------------------------------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Candidate generators
# --------------------------------------------------------------

def _initial_arcsine(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta‑arc initialisation (α≈0.5 gives the arcsine law)."""
    i = np.arange(n, dtype=np.float64)
    raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
    # normalise to a probability vector (sum = 1)
    total = raw.sum()
    if total <= 0:
        raw = np.full_like(raw, 1.0 / n)
    else:
        raw = raw / total
    return raw


# --------------------------------------------------------------
# Core optimisation routine for a single (n,α) pair
# --------------------------------------------------------------

def _optimise_sequence(a: np.ndarray,
                       start_time: float,
                       deadline: float,
                       rng: np.random.Generator,
                       best_score: float,
                       best_seq: list[float]) -> tuple[float, list[float]]:
    """
    Mirror‑descent minimise  max(a∗a)  starting from ``a``.
    Returns the best score seen in this run and the corresponding sequence.
    """
    n = a.size
    # FFT length that can hold both a∗a (2n‑1) and w∗a_rev (3n‑2)
    fft_len = _fft_len_for_lengths(2 * n - 1, n)

    # hyper‑parameters
    eta = 0.5                 # initial mirror‑descent step size
    beta = 0.5                # softness of the max‑approximation
    eval_every = 200         # evaluate the true score every this many iters
    jitter_prob = 0.01       # probability of a small random perturbation
    max_iters = 30000        # safe upper bound; the time check will stop earlier

    # main loop
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # ----- self‑convolution (a∗a) -----
        conv = _self_conv_fft(a, fft_len)
        max_conv = conv.max()

        # ----- soft‑max weights on convolution indices -----
        # shift by max ensures the largest entry gets weight 1 (exp(0))
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        # grad_j = 2 * Σ_i w_i * a_{i-j}
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev, fft_len)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]          # slice length n

        # ----- mirror‑descent update (exponentiated gradient) -----
        a = a * np.exp(-eta * grad)

        # occasional jitter to escape flat regions
        if rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            a *= jitter

        # renormalise to the simplex (sum = 1)
        total = a.sum()
        if total <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        # schedule adjustments
        eta = max(eta * 0.9999, 1e-6)   # slowly decay step size
        beta = min(beta * 1.001, 10.0)  # make the soft‑max sharper over time

        # ----- periodic evaluation of the true objective -----
        if (it % eval_every) == 0:
            # Because sum(a)=1, the evaluate_sequence score equals
            #   2·n·max(a∗a)   (the function also clips values to [0,1000],
            #   which is already satisfied).
            score = float(2.0 * n * max_conv)
            if score < best_score:
                best_score = score
                best_seq = a.tolist()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new global best = {score:.6f} "
                      f"(n={n})")
        it += 1

        # safety break in case we are extremely close to the deadline
        if time.monotonic() - start_time > (deadline - start_time) - 0.2:
            break

    return best_score, best_seq


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine explores several sequence lengths, uses a smooth L∞‑approximation,
    and applies mirror‑descent on the probability simplex.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # stop a little before the hard deadline to avoid the -inf penalty
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0)  Baseline – use the previous construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(map(float, globals()["height_sequence_1"]))
            sc = evaluate_sequence(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] baseline score = {sc:.6f} "
                      f"(length {len(cand)})")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1)  Search over a handful of lengths and α‑parameters
    # --------------------------------------------------------------
    # lengths we will try – powers of two are FFT‑friendly; we also keep
    # the previously best length (3809) so that the algorithm can improve it.
    lengths = [1024, 2048, 4096, 8192, 3809]
    alphas = [0.40, 0.45, 0.48, 0.50, 0.52, 0.55, 0.60]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for alpha in alphas:
            if time.monotonic() >= deadline:
                break

            # ----- initialise -----
            a = _initial_arcsine(n, alpha)

            # quick baseline score for this raw shape
            conv0 = _self_conv_fft(a, _fft_len_for_lengths(n, n))
            score0 = float(2.0 * n * conv0.max())
            if score0 < best_score:
                best_score = score0
                best_seq = a.tolist()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] raw ({n},α={alpha}) score = {score0:.6f}")

            # ----- local optimisation (mirror‑descent) -----
            best_score, best_seq = _optimise_sequence(
                a=a,
                start_time=start_time,
                deadline=deadline,
                rng=rng,
                best_score=best_score,
                best_seq=best_seq,
            )

    # --------------------------------------------------------------
    # 2)  If nothing succeeded, fall back to a tiny uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Clamp to the limits required by evaluate_sequence (0 ≤ x ≤ 1000)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
