def _subgradient_step(a, conv):
    n = a.shape[0]
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi+1)
    grad = np.zeros_like(a)
    # For each j=idx, compute a[i_max - j] + a[j]
    grad[idx] = a[i_max - idx] + a[idx]
    return grad
```

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().

Algorithm
---------

* Work on the probability simplex (∑a = 1).  The score is 2·n·max(conv(a,a)).
* For a given length n we run a projected *exponentiated* gradient descent
  (mirror descent on the simplex).  The descent direction is a mixture of:
  – the exact sub‑gradient of the current worst convolution entry,
  – the gradient of a smooth approximation  (log‑sum‑exp) of the max,
    with a temperature β that grows linearly during the run.
* The step is multiplicative:
        a ← a · exp(−η·grad),   a ← a / ∑a
  (no need for an explicit projection).
* Occasionally inject cheap random moves – block‑wise log‑normal
  perturbations and pairwise mass transfers – and keep the change only if
  it improves the score.
* Many short independent restarts with a variety of lengths (up to 20000)
  are performed inside the time budget (≈1000 s).  The current global best
  is repeatedly polished with a longer run.
* If a previous construction `height_sequence_1` exists it is used as an
  initial candidate.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _fft_len_for_len(length: int) -> int:
    """Return the smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex projection (used only for occasional random moves)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    For the entry i_max that attains the maximum we have
        ∂conv_i_max / ∂a_j = a_{i_max‑j} + a_j   (if the index is valid)
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = a[i_max - idx] + a[idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no spurious extra term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Random perturbations (used sporadically)
# ----------------------------------------------------------------------
def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a random amount of mass from i to j (accepted only if it improves)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    conv_old = _conv_self(a, fft_len_conv).max()
    conv_new = _conv_self(a2, fft_len_conv).max()
    if 2.0 * n * conv_new < 2.0 * n * conv_old:
        return a2
    return a


# ----------------------------------------------------------------------
# Core optimiser – exponentiated gradient descent on the simplex
# ----------------------------------------------------------------------
def _local_optimize_exp(a0: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Intensive optimisation of vector a (already on the simplex) using
    exponentiated gradient descent with occasional random moves.
    Returns the best vector encountered and its score (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # best‑so‑far bookkeeping
    best_vec = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len_conv).max()

    for t in range(1, max_iter + 1):
        if time.monotonic() > deadline:
            break

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # Choose a gradient (hard sub‑gradient ≈70 % of the time)
        # --------------------------------------------------------------
        if rng.random() < 0.7:
            grad = _subgradient_step(a, conv)
        else:
            # smooth‑max gradient – β grows linearly
            beta = 0.5 + (2000.0 - 0.5) * (t / max_iter)
            grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # Exponentiated‑gradient (mirror) update
        # --------------------------------------------------------------
        lr = 0.5 / np.sqrt(t)          # decreasing step‑size
        a = a * np.exp(-lr * grad)    # multiplicative move
        # renormalise to stay on the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --------------------------------------------------------------
        # Occasional stochastic moves to escape shallow plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.05:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.05:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best vector found) that
    minimises evaluate_sequence() before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, evaluate it first
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                n0 = cand.size
                fft_len0 = _fft_len_for_len(2 * n0 - 1)
                cur = 2.0 * n0 * _conv_self(cand, fft_len0).max()
                if cur < best_score:
                    best_score = cur
                    best_vec = cand.copy()
                print(f"Initial global candidate score = {cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore – we allow up to 20 000 entries (still cheap for FFT)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 8000, 12000, 16000, 20000]
    probs = np.array([0.12, 0.13, 0.15, 0.15, 0.15, 0.12, 0.10, 0.08])
    probs = probs / probs.sum()

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        n = int(rng.choice(lengths, p=probs))

        # ----- random initial vector (mixture of a few patterns) -----
        if rng.random() < 0.6:
            a0 = rng.random(n)
        else:
            # smooth bump – a discretised half‑sine (forces small values at the edges)
            x = np.arange(n)
            a0 = np.sin(np.pi * (x + 0.5) / n) ** 2
            # add a touch of randomness to avoid symmetry lock‑in
            a0 += 0.01 * rng.random(n)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

        # ----- short intensive optimisation -----
        a_opt, score_opt = _local_optimize_exp(a0, rng, deadline, max_iter=2000)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})")

        # ----- occasional polishing of the current global best -----
        if best_vec is not None and time.monotonic() < deadline - 0.5:
            a_polish, score_polish = _local_optimize_exp(
                best_vec.copy(), rng, deadline, max_iter=4000)
            if score_polish < best_score:
                best_score = score_polish
                best_vec = a_polish.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polished global best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 4000
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final safety clipping – evaluate_sequence will also clip, but we do it here
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
