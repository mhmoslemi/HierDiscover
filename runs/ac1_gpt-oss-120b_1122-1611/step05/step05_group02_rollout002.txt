    best = np.clip(best, 0.0, 1000.0)

"""
Propose a step‑function (non‑negative heights) that minimizes the worst
self‑convolution value used by ``evaluate_sequence``.

The algorithm combines three ideas:

1. **Fast parametric initialization** – we scan a few simple families
   (U‑shaped power‑law and symmetric Beta densities) on a fixed length
   (default 4096).  The best of these vectors provides a solid starting
   point better than the uniform distribution.

2. **Top‑K sub‑gradient descent** – instead of following only the single
   maximal convolution entry, we form a gradient that reduces the
   largest few peaks (default K=3).  The update is an exponentiated‑gradient
   (mirror‑descent) step with a linearly decreasing stepsize and a small
   momentum term.

3. **Stochastic perturbations & soft restarts** – occasional log‑normal
   noise, random pairwise mass transfers and simulated‑annealing style
   acceptance help to escape local minima.  If no improvement occurs for
   about 30 s the current state is re‑jittered around the best solution
   found so far.

All calculations are performed with FFT‑based convolutions for speed.
The routine respects the 1000 s time budget and returns the best vector
found (clipped to the range required by ``evaluate_sequence``).
"""

import numpy as np

# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def _score_of(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) (the part of ``evaluate_sequence`` that
    depends on the shape).  ``a`` must already be normalised to sum = 1."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return 2.0 * n * float(conv.max())


def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    """Sub‑gradient of the max‑convolution term summed over the ``top_k``
    largest convolution entries."""
    n = a.shape[0]
    if top_k < conv.size:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
    else:
        top_idx = np.arange(conv.size)

    grad = np.zeros_like(a)
    for k_idx in top_idx:
        i_start = max(0, k_idx - (n - 1))
        i_end = min(n - 1, k_idx)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k_idx - i_range
            grad[i_range] += 2.0 * a[j]
    return grad


def _normalise(a: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    s = a.sum()
    if s > 0.0:
        return a / s
    else:
        # fallback to uniform
        n = a.shape[0]
        return np.full_like(a, 1.0 / n)


def _init_from_global() -> np.ndarray:
    """If a previous construction is available (global ``height_sequence_1``)
    use it (padded or trimmed) and normalise; otherwise return a uniform
    vector of the chosen length."""
    n_target = 4096
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                a = _normalise(a)
                if a.size < n_target:
                    a = np.concatenate([a, np.zeros(n_target - a.size)])
                elif a.size > n_target:
                    a = a[:n_target]
                a = _normalise(a)
                return a
        except Exception:
            pass
    # default: uniform
    return np.full(n_target, 1.0 / n_target, dtype=np.float64)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the allotted time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Fixed problem size – a power of two is convenient for the FFT.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 1) Start from a reasonable point.
    # ------------------------------------------------------------------
    a = _init_from_global()
    best = a.copy()
    best_score = _score_of(best)
    print(f"Initial score (uniform / previous) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Quick scan of simple parametric families (U‑shaped power law,
    #    symmetric Beta densities).  All candidates are normalised.
    # ------------------------------------------------------------------
    # Power‑law (U‑shape) family: w_i = (i+1)^(-p) + (n-i)^(-p)
    power_ps = np.concatenate(
        [np.linspace(0.05, 0.5, 10), np.linspace(0.5, 2.0, 10)]
    )
    idx = np.arange(n, dtype=np.float64)
    for p in power_ps:
        w = (np.power(idx + 1.0, -p) + np.power(n - idx, -p))
        cand = _normalise(w)
        sc = _score_of(cand)
        if sc < best_score:
            best_score, best = sc, cand.copy()
            print(f"Power‑law p={p:.3f} → new best {best_score:.6f}")

    # Symmetric Beta‑like family: pdf ∝ x^{α‑1}(1‑x)^{α‑1}
    alphas = np.concatenate(
        [np.linspace(0.1, 0.9, 9), np.linspace(1.1, 3.0, 10)]
    )
    x = (np.arange(n) + 0.5) / n  # mid‑point sampling
    eps = 1e-12
    for alpha in alphas:
        pdf = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, alpha - 1.0)
        cand = _normalise(pdf)
        sc = _score_of(cand)
        if sc < best_score:
            best_score, best = sc, cand.copy()
            print(f"Beta α={alpha:.3f} → new best {best_score:.6f}")

    # Use the best found so far as the current state
    a = best.copy()

    # ------------------------------------------------------------------
    # 3) Main optimisation loop – top‑K sub‑gradient + stochastic moves.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    # hyper‑parameters
    eta_start = 0.8          # initial stepsize for exponentiated gradient
    eta_end = 0.001          # final stepsize
    momentum = 0.85
    top_k = 3
    restart_gap = 30.0       # seconds without improvement → restart
    last_improve = start_time
    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:
        # ---- evaluate current state -------------------------------------------------
        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * float(conv.max())

        # ---- keep global best -------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule for stepsize --------------------------------------------------
        now = time.monotonic()
        done = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - done) + eta_end * done
        # simulated‑annealing temperature (small, only for uphill acceptance)
        temp = 0.4 * (1.0 - done) + 0.001 * done

        # ---- compute a summed top‑K sub‑gradient ------------------------------------
        grad = _gradient_topk(a, conv, top_k)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- generate a small set of candidate updates ------------------------------
        candidates = []

        # 1) exponentiated‑gradient step
        a_exp = a * np.exp(-eta * grad)
        candidates.append(_normalise(a_exp))

        # 2) occasional multiplicative log‑normal jitter
        if rng.random() < 0.06:
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            candidates.append(_normalise(a * noise))

        # 3) random pairwise mass transfer
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta_max = min(a[i], 0.3 * a[i])
                delta = rng.random() * delta_max
                a_pair = a.copy()
                a_pair[i] -= delta
                a_pair[j] += delta
                candidates.append(_normalise(a_pair))

        # ---- pick the best candidate (with occasional uphill acceptance) ------------
        best_local = a
        best_local_score = cur_score
        for cand in candidates:
            sc = _score_of(cand)
            if sc < best_local_score or rng.random() < np.exp((best_local_score - sc) / max(temp, 1e-12)):
                best_local, best_local_score = cand, sc

        a = best_local

        # ---- soft restart when stuck ------------------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = _normalise(a * noise)
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best vector (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
