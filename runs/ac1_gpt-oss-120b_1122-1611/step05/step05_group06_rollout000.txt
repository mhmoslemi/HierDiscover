"""
Hybrid optimization for the self‑convolution peak problem.

The algorithm combines several ideas that were not used together before:

*  Adam updates (additive) **and** exponentiated‑gradient updates (multiplicative).
*  Optional symmetrisation after each step (the optimum is known to be symmetric).
*  Light Gaussian smoothing (moving‑average) to keep the shape smooth.
*  A more aggressive β‑schedule for the log‑sum‑exp (smooth‑max) gradient.
*  Random block perturbations and pairwise mass‑transfer moves as in the reference,
   but with a slightly higher acceptance probability.
*  A two‑phase strategy: many cheap short runs on a variety of lengths,
  followed by a long polishing run on the current global best.
*  Deterministic “U‑shaped” seeds (inverse‑sqrt distribution) that are known to
  produce low convolution peaks, in addition to random Dirichlet starts.

The code respects the 1000 s budget, prints progress, and returns a list of
non‑negative heights (a probability vector) that should improve on the previous
best score (~1.514).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors `x` and `y` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t a (averaged over all ties)."""
    max_val = conv.max()
    ties = np.where(conv >= max_val - 1e-12)[0]
    n = a.size
    grad = np.zeros_like(a)
    for i_max in ties:
        lo = max(0, i_max - (n - 1))
        hi = min(i_max, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[i_max - idx]
    grad /= max(1, len(ties))
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    No extra “+a” term.
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    grad = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Multiplicative (exponentiated‑gradient) update, followed by simplex projection."""
    # exponentiate negative gradient (we want to descend)
    mult = np.exp(-eta * grad)
    a_new = a * mult
    a_new = _project_onto_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Cheap mass‑transfer move; accept only if it improves the exact score."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    conv = _conv_self(a2, fft_len_conv)
    if 2.0 * n * conv.max() < 2.0 * n * _conv_self(a, fft_len_conv).max():
        return a2
    return a


def _apply_smoothing(a: np.ndarray, width: int) -> np.ndarray:
    """Simple moving‑average smoothing (preserves mass, then simplex‑project)."""
    if width < 3:
        return a
    kernel = np.ones(width) / width
    smooth = np.convolve(a, kernel, mode='same')
    # due to border effects the sum is unchanged, but enforce non‑negativity
    smooth = np.clip(smooth, 0.0, None)
    return _project_onto_simplex(smooth)


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry a_i = a_{n-1-i}."""
    return 0.5 * (a + a[::-1])


# ----------------------------------------------------------------------
# Local optimiser (hybrid Adam / EG)
# ----------------------------------------------------------------------


def _local_optimise(a0: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float,
                    max_iters: int = 2000,
                    use_smoothing: bool = True,
                    smooth_width_factor: float = 0.02,
                    symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """
    Hybrid local optimisation.
    Returns the best vector (sum=1) found and its exact score 2·n·max(conv).
    """
    n = a0.size
    a = a0.copy()
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state (we keep first and second moments)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_vec = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len_conv).max()

    smooth_width = max(3, int(smooth_width_factor * n))

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ----- current convolution & score -----
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ----- gradient choice -----
        if rng.random() < 0.55:
            # hard sub‑gradient (average over ties)
            raw_grad = _subgradient_step(a, conv)
        else:
            # smooth max – β grows quadratically for a sharper focus
            beta = 0.5 + (3000.0) * (t / max_iters) ** 2
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # ----- update type -----
        if rng.random() < 0.45:
            # multiplicative exponentiated‑gradient update
            eta = 0.8 / np.sqrt(t)          # decaying step size
            a = _exp_update(a, raw_grad, eta)
        else:
            # additive Adam update
            lr = 0.9 / np.sqrt(t)
            m = beta1 * m + (1.0 - beta1) * raw_grad
            v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _project_onto_simplex(a)

        # ----- optional symmetry & smoothing -----
        if symmetrize:
            a = _symmetrize(a)
        if use_smoothing and (t % 50 == 0):
            a = _apply_smoothing(a, smooth_width)

        # ----- random cheap moves (accept‑if‑better) -----
        if rng.random() < 0.04:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.05:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
