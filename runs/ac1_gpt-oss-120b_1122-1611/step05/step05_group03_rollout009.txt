    """Projection onto probability simplex."""

which is obtained by a single cross‑correlation.  The optimisation
uses an exponential‑decay schedule for the soft‑max temperature β and a
decreasing step size η.  Occasional random jitter and restarts are
added to escape plateaus.  The routine stops after the allotted time and
returns the best vector discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution using FFT (real inputs). Returns the full‑length result."""
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    C = A * B
    conv = np.fft.irfft(C, n=fft_len)
    size = a.size + b.size - 1
    return conv[:size]


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a*a using FFT (full length 2·n‑1)."""
    return _fft_convolve(a, a, fft_len)


def _correlate_fft(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation a ⋆ w (full length n+len(w)‑1) via FFT."""
    # correlation a⋆w = a convolve reverse(w)
    w_rev = w[::-1]
    return _fft_convolve(a, w_rev, fft_len)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Simple Euclidean projection onto the probability simplex."""
    # Clip negative values, then renormalise – cheap and always feasible.
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        # fallback to uniform distribution
        v.fill(1.0 / v.size)
    return v

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget in seconds.
    kwargs : dict
        Ignored, present for API compatibility.

    Returns
    -------
    list[float]
        The best height sequence discovered (already clipped to evaluate_sequence limits).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Choose problem size and initialise a
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        total = a0.sum()
        if total > 0.0:
            a = a0 / total
        else:
            # uniform fallback
            n = a0.size
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # moderate size (≈4000) – enough degrees of freedom and fast FFTs
        n = 4096
        # Dirichlet with concentration <1 gives a slightly spiky start,
        # which is useful for escaping the uniform 2.0 bound.
        a = rng.dirichlet(alpha=np.full(n, 0.5))

    n = a.size

    # Pre‑compute FFT length that can accommodate both a*a and a⋆w.
    size_conv = 2 * n - 1
    size_corr = 3 * n - 2            # a (len n) conv w (len 2n‑1)
    fft_len = _next_pow2(max(size_conv, size_corr))

    # --------------------------------------------------------------
    # 2) Initial evaluation
    # --------------------------------------------------------------
    conv = _self_convolution_fft(a, fft_len)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # small safety margin

    # --------------------------------------------------------------
    # 3) Hyper‑parameter schedules
    # --------------------------------------------------------------
    beta_start = 0.5         # low temperature → smooth landscape
    beta_end = 200.0         # high temperature → focus on the true max
    eta_start = 0.20         # initial additive step size
    eta_end = 0.001          # final step size (fine tuning)

    iteration = 0

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        progress = elapsed / total

        # ---- 4.1) Annealing schedules -------------------------------------------------
        beta = beta_start * (beta_end / beta_start) ** progress   # exponential schedule
        eta = eta_start * (eta_end / eta_start) ** progress      # exponential decay

        # ---- 4.2) Compute convolution and soft‑max weights -----------------------------
        conv = _self_convolution_fft(a, fft_len)                 # length 2n‑1
        max_conv = conv.max()
        # soft‑max on the convolution, shifted to avoid overflow
        exp_arg = beta * (conv - max_conv)                       # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback – fall back to a uniform weight distribution
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- 4.3) Gradient of the soft‑max term ---------------------------------------
        # ∇_i = 2·∑_j a_j w_{i+j}
        corr = _correlate_fft(a, w, fft_len)                     # length 3n‑2
        offset = size_conv - 1                                   # = 2n‑2
        grad = 2.0 * corr[offset : offset + n]                   # same shape as a

        # ---- 4.4) Gradient step (additive) -------------------------------------------
        a_new = a - eta * grad
        a_new = _project_to_simplex(a_new)

        # ---- 4.5) Evaluate ------------------------------------------------------------
        conv_new = _self_convolution_fft(a_new, fft_len)
        new_score = 2.0 * n * conv_new.max()

        # ---- 4.6) Accept if improvement (or with tiny Metropolis chance) ------------
        # The Metropolis term helps to escape tiny local valleys when the
        # temperature β is still low.
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # simple probabilistic acceptance that vanishes as η → 0
            prob = math.exp(-(new_score - cur_score) / max(1e-6, eta))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            # update best if needed
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start_time:6.2f}s] new best {best_score:.6f}")

        # ---- 4.7) Occasional random jitter to break plateaus -------------------------
        if rng.random() < 0.012:
            jitter = np.exp(rng.normal(0.0, 0.13, size=n))
            a_pert = a * jitter
            a_pert = _project_to_simplex(a_pert)
            conv_pert = _self_convolution_fft(a_pert, fft_len)
            pert_score = 2.0 * n * conv_pert.max()
            if pert_score < cur_score:
                a = a_pert
                cur_score = pert_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve - start_time:6.2f}s] jitter improved best {best_score:.6f}")

        # ---- 4.8) Restart when stuck for a while --------------------------------------
        if now - last_improve > 30.0:
            # Restart from the current best, add moderate jitter.
            restart_noise = np.exp(rng.normal(0.0, 0.25, size=n))
            a = best * restart_noise
            a = _project_to_simplex(a)
            conv = _self_convolution_fft(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart – score reset to {cur_score:.6f}")

        # ---------------------------------------------------------
        # Optional early‑exit if we already beat the target.
        # ---------------------------------------------------------
        if best_score <= 1.5025:
            # We reached the required bound – stop early.
            print(f"[{now - start_time:6.2f}s] target reached (score {best_score:.6f})")
            break

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluate_sequence limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
