def _simplex_projection(v):
    """Projects vector v onto the probability simplex (nonnegative, sum=1)."""
    # from https://arxiv.org/pdf/1309.1541.pdf
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho = last index where condition holds
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

"""
Propose a step‑function with a lower bound than the previous 1.514.
The algorithm combines three ingredients:

1. **Variance minimisation** – minimise the L2‐distance of the self‑
   convolution to the flat target 1/(2n‑1).  The gradient of this
   quadratic loss can be obtained by a single FFT‑convolution, and the
   update is performed by Euclidean projected gradient descent on the
   probability simplex.

2. **Soft‑max (log‑sum‑exp) descent** – a smooth surrogate of the
   max‑norm.  The gradient is the usual “soft” sub‑gradient
   2·(p * rev(a)) where p is the soft‑max distribution over the
   convolution entries.  One of the two gradients (variance or soft‑max)
   is used at each iteration.

3. **Hard max sub‑gradient + stochastic mass‑transfer** – after the
   variance/soft‑max phase the algorithm switches to a direct
   sub‑gradient of the true max (the convolution entry that attains the
   current peak).  A simple mirror‑step on the simplex is applied,
   followed by occasional random mass‑transfer moves and block
   perturbations to escape plateaus.

The whole routine respects a 1000 s budget (5 s safety margin) and
periodically restarts from the best solution seen so far with a small
log‑normal perturbation.  The best sequence found is returned as a list
of floats ready for ``evaluate_sequence``.

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using a pre‑allocated FFT length."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the probability simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]               # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(cand, 0.0, None)
        s = a.sum()
        a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
    else:
        # default size ≈ 3800 (power of two convenient for FFT)
        n = 4096
        a = rng.dirichlet(np.ones(n)).astype(np.float64)

    n = a.size                                 # length of the vector
    fft_len_self = _next_pow2(2 * n - 1)        # for self‑convolution
    fft_len_grad = _next_pow2(3 * n - 2)        # for gradient convolutions
    mu = 1.0 / (2 * n - 1)                     # target flat convolution value

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * conv.max()           # = evaluate_sequence on a (sum=1)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.55 * total_time   # 55 % for variance / soft‑max

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # schedules for the first phase
    eta_start = 0.6
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 2000.0

    # schedules for the second (hard‑max) phase
    eta_max = 0.35
    eta_min = 0.015
    mass_transfer_prob = 0.4
    jitter_prob = 0.025

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ==============================================================
        # Phase 1 – flatten the convolution (variance + soft‑max)
        # ==============================================================
        if now < phase1_end:
            # linear schedules for step‑size and soft‑max temperature
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # decide which gradient to use (variance or soft‑max)
            if rng.random() < 0.5:
                # ---- variance gradient (L2 distance to flat target) ----
                conv = _self_conv(a, fft_len_self)
                diff = conv - mu
                rev_a = a[::-1]
                f_diff = np.fft.rfft(diff, n=fft_len_grad)
                f_rev = np.fft.rfft(rev_a, n=fft_len_grad)
                grad_full = np.fft.irfft(f_diff * f_rev, n=fft_len_grad)[: 3 * n - 2]
                grad = 2.0 * grad_full[n - 1 : n - 1 + n]
            else:
                # ---- smooth max (log‑sum‑exp) gradient ----
                conv = _self_conv(a, fft_len_self)
                max_c = conv.max()
                w = np.exp(beta * (conv - max_c))
                w_sum = w.sum()
                p = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)
                rev_a = a[::-1]
                fp = np.fft.rfft(p, n=fft_len_grad)
                fra = np.fft.rfft(rev_a, n=fft_len_grad)
                grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
                grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # Euclidean projected gradient descent on the simplex
            a = _simplex_projection(a - eta * grad)

            # occasional block‑wise jitter to keep diversity
            if rng.random() < jitter_prob:
                a = _random_block_perturb(a, rng)
                a = _simplex_projection(a)

        # ==============================================================
        # Phase 2 – direct max descent + stochastic refinement
        # ==============================================================

        else:
            # recompute convolution and current score
            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            k_max = int(conv.argmax())

            # ------ hard‑max sub‑gradient step ------
            indicator = np.zeros_like(conv)
            indicator[k_max] = 1.0
            rev_a = a[::-1]
            f_ind = np.fft.rfft(indicator, n=fft_len_grad)
            f_rev = np.fft.rfft(rev_a, n=fft_len_grad)
            grad_full = np.fft.irfft(f_ind * f_rev, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # annealed step size for the hard‑max phase
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            eta = eta_max * (1.0 - t_frac) + eta_min * t_frac
            a_candidate = _simplex_projection(a - eta * grad)

            # accept if the max improves (deterministic descent)
            conv_cand = _self_conv(a_candidate, fft_len_self)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                conv = conv_cand

            # ------ stochastic mass‑transfer moves (Metropolis) ------
            if rng.random() < mass_transfer_prob:
                # contributions of each index to the current peak
                contrib = np.zeros_like(a)
                for i in range(n):
                    j = k_max - i
                    if 0 <= j < n:
                        contrib[i] = a[i] * a[j]
                tot_contrib = contrib.sum()
                if tot_contrib > 0.0:
                    prob = contrib / tot_contrib
                    donor = rng.choice(np.arange(n), p=prob)
                else:
                    donor = rng.integers(0, n)

                if a[donor] > 1e-12:
                    recip = rng.integers(0, n - 1)
                    if recip >= donor:
                        recip += 1
                    # transfer up to 30 % of donor's mass
                    delta = rng.random() * 0.30 * a[donor]
                    a_new = a.copy()
                    a_new[donor] -= delta
                    a_new[recip] += delta
                    conv_new = _self_conv(a_new, fft_len_self)
                    new_score = 2.0 * n * conv_new.max()
                    # simple Metropolis acceptance (temperature linearly cooled)
                    temperature = 0.08 * (1.0 - t_frac) + 1e-5 * t_frac
                    if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
                        a = a_new
                        cur_score = new_score
                        conv = conv_new

            # ------ occasional global jitter ------
            if rng.random() < jitter_prob:
                a = _random_block_perturb(a, rng)
                a = _simplex_projection(a)

        # --------------------------------------------------------------
        # Track the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = _simplex_projection(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (ensuring evaluation‑function safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
