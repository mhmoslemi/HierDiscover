def _optimize_candidate(a0, n, fft_len, max_iters, rng):
    a = a0.copy()
    best_local = a.copy()
    best_local_score = 2.0 * n * _conv_self_fft(a, fft_len).max()
    # parameters
    eta_start = 1.0
    eta_end = 0.001
    alpha_start = 0.5
    alpha_end = 30.0
    beta1 = 0.9
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps = 1e-8
    for it in range(max_iters):
        # compute conv and max
        conv = _conv_self_fft(a, fft_len)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_local_score:
            best_local_score = score
            best_local = a.copy()
        # schedule
        frac = it / max_iters
        eta = eta_start * (1-frac) + eta_end * frac
        alpha = alpha_start * (1-frac) + alpha_end * frac
        # compute softmax weights
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0/conv.size)
        # compute gradient via correlation
        grad = 2.0 * np.correlate(a, w, mode='valid')
        # momentum / Adam
        m = beta1 * m + (1-beta1) * grad
        v = beta1 * v + (1-beta1) * (grad**2)  # using same beta for simplicity
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta1**(it+1))
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        # update multiplicatively
        a = a * np.exp(-step)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= s
        # occasional random perturb
        if rng.random() < 0.005:
            a = _random_block_perturb(a, rng)
    return best_local, best_local_score

"""
Propose a step‑function that minimises the worst self‑convolution value.
The method builds on the previous exponentiated‑gradient scheme but
replaces the hard max‑gradient by a *smooth* max based on a soft‑max
over the (scaled) convolution values.  The smoothness parameter α and
the learning‑rate η are annealed from a gentle start to an aggressive
finish.  A simple Adam‑style momentum adapts the step for each
coordinate.  Several diverse initialisations (uniform, random
Dirichlet draws, discretised Beta‑densities and sine‑shapes) are tried,
each is refined with the gradient descent, and the best vector found
within the time budget is returned.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can accommodate the full linear
    convolution of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution (a * a) using an FFT of length ``fft_len``.
    Returns an array of length ``2·len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _soft_max_gradient(a: np.ndarray,
                       conv: np.ndarray,
                       n: int,
                       alpha: float,
                       rng: np.random.Generator,
                       prob_hard: float = 0.1) -> np.ndarray:
    """Return a gradient estimate of the (scaled) max‑convolution.
    With probability ``prob_hard`` we use the exact sub‑gradient at the
    maximal index; otherwise we use a soft‑max weighted gradient.
    """
    # ----- occasional hard gradient (same as the previous code) -----
    if rng.random() < prob_hard:
        k_max = int(np.argmax(conv))
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k_max - i_range
            grad[i_range] = 2.0 * a[j]
        return grad

    # ----- smooth max gradient --------------------------------------
    # Scale the convolution so that its average becomes 1.
    # For a probability vector (sum=1) the average of conv is 1/(2n‑1).
    scale = 2 * n - 1
    conv_scaled = conv * scale
    max_scaled = conv_scaled.max()

    # Soft‑max weights (stable)
    exps = np.exp(alpha * (conv_scaled - max_scaled))
    w_sum = exps.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exps / w_sum

    # Gradient = 2 * correlation(a, w)  (valid mode returns length n)
    grad = 2.0 * np.correlate(a, w, mode='valid')
    return grad


def _optimize_candidate(a0: np.ndarray,
                        n: int,
                        fft_len: int,
                        max_iters: int,
                        rng: np.random.Generator,
                        time_deadline: float) -> tuple[np.ndarray, float]:
    """
    Run a limited‑time exponentiated‑gradient / Adam optimisation from a
    starting vector ``a0``.  Returns the best vector found and its
    corresponding score (2·n·max(conv)).
    """
    a = a0.copy()
    # Adam statistics
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # keep the best local solution
    conv = _conv_self_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()
    best_vec = a.copy()

    start_time = time.monotonic()
    for it in range(1, max_iters + 1):
        # ---- time check -------------------------------------------------
        if time.monotonic() >= time_deadline:
            break

        # ---- convolution ------------------------------------------------
        conv = _conv_self_fft(a, fft_len)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ---- keep the best ------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s]   local best = {best_score:.6f}")

        # ---- schedules --------------------------------------------------
        t_frac = it / max_iters
        # learning rate (η) – geometric decay
        eta_start, eta_end = 0.5, 0.001
        eta = eta_start * (1.0 - t_frac) + eta_end * t_frac
        # soft‑max sharpness (α) – linear increase
        alpha_start, alpha_end = 1.0, 30.0
        alpha = alpha_start * (1.0 - t_frac) + alpha_end * t_frac

        # ---- gradient (soft‑max or hard) -------------------------------
        grad = _soft_max_gradient(a, conv, n, alpha, rng, prob_hard=0.05)

        # ---- Adam momentum ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ---- multiplicative (exponentiated) update ----------------------
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)
        a = np.clip(a, 0.0, None)

        # ---- renormalise to unit sum ------------------------------------
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---- occasional random block perturbation ----------------------
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

    return best_vec, best_score

def propose_candidate(seed: int = 42,

    """

    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------

    if "height_sequence_1" in globals():

            else:

        except Exception:

            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:

        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    start_vectors = []

    # (a) the best known vector (if any)
    start_vectors.append(a0)

    # (b) plain uniform
    start_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # (c) a few random Dirichlet draws
    for _ in range(4):

        x /= x.sum()
        start_vectors.append(x)

    # (d) discretised Beta‑densities
    x_grid = (np.arange(n) + 0.5) / n
    for _ in range(5):
        alpha = rng.uniform(0.2, 5.0)

        y = np.power(x_grid, alpha - 1.0) * np.power(1.0 - x_grid, beta - 1.0)
        y[np.isnan(y)] = 0.0
        y = np.clip(y, 0.0, None)
        s = y.sum()
        if s == 0.0:

        y = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
        y = np.clip(y, 0.0, None)
        y /= y.sum()
        start_vectors.append(y)

    # (f) lightly jittered versions of the above (to escape local minima)
    jittered = []
    for vec in start_vectors:
        noise = np.exp(rng.normal(0.0, 0.1, size=n))
        y = vec * noise
        y = np.clip(y, 0.0, None)
        y /= y.sum()
        jittered.append(y)

    start_time = time.monotonic()

    # Helper to evaluate the objective directly (no call to evaluate_sequence)

            global_best_score = sc

    for idx, vec in enumerate(start_vectors):
        if time.monotonic() >= deadline:

    global_best_vec = np.clip(global_best_vec, 0.0, 1000.0)
    return [float(v) for v in global_best_vec]
