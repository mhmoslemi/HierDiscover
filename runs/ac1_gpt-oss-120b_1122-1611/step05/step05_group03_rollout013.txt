def propose_candidate(seed=1234, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.1
    n = 4096
    target_sum = math.sqrt(2*n)
    # Initialize a as uniform distribution
    a = np.full(n, target_sum / n)
    # compute conv and best
    conv = np.convolve(a,a)
    best = a.copy()
    best_score = conv.max()
    last_improve = start
    iteration = 0
    while time.monotonic() < deadline:
        # compute current max conv and M
        conv = np.convolve(a,a)
        M = conv.max()
        cur_score = M
        # solve LP
        # build A_ub matrix
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n)
        A_ub = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
        # solve LP
        c = -np.ones(n)  # maximize sum(b)
        res = opt.linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
        if not res.success:
            # fallback random perturb
            break
        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            break
        g = b * (target_sum / sum_b)
        # mixing coefficient: gradually decreasing
        frac = (iteration) / 200.0  # dec schedule maybe
        t = max(0.2 * (1 - frac), 0.02)  # ensure min
        a = (1-t)*a + t*g
        # clip
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, target_sum / n)
        else:
            a = a * (target_sum / a_sum)
        # evaluate
        conv = np.convolve(a, a)
        cur_score = conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:.2f}s] new best {best_score:.6f}")
        # occasional random jitter to escape local optimum
        if rng.random() < 0.01:
            # jitter: multiply each entry by random factor and renormalize
            factor = np.exp(rng.normal(0.0, 0.3, size=n))
            a = a * factor
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())
        iteration += 1
        # check stagnation
        if time.monotonic() - last_improve > 30.0:
            # restart with random distribution
            a = rng.random(n)
            a = a * (target_sum / a.sum())
            last_improve = time.monotonic()
    # return best
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Propose a candidate step‑function using a two‑phase optimisation:

Phase 1 – Linear‑programming boost.
    Starting from a normalised non‑negative vector a (∑a = √(2 n)),
    we repeatedly solve the LP

        maximise   1ᵀb
        subject to  a * b ≤ M   (component‑wise convolution)
                    b ≥ 0

    where M = max(a*a).  The LP finds a direction b that can increase the
    ℓ₁‑mass without raising the current peak M.  After scaling b to the
    normalisation we mix a←(1–t)a + t b (t slowly decreasing).  Each LP
    iteration reduces the evaluation score (which equals the peak after
    normalisation).

Phase 2 – Greedy mass‑transfer refinement.
    After the LP stage we fine‑tune the solution with the fast
    “targeted‑mass‑transfer” heuristic used in earlier work, but starting
    from the much improved LP result.  Random jitter and occasional restarts
    are kept to escape plateaus.

The whole routine respects the 1000 s budget, works for any length up to
a few‑thousand entries and returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1)."""
    return np.convolve(a, a)


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    i: int,
                    j: int,
                    delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` mass from index i to j.
    The update follows (a+Δ)*(a+Δ) = a*a + 2*a*Δ + Δ*Δ.
    """
    n = a.size
    new_conv = conv.copy()
    # linear part: -2Δ·shift(a,i) + 2Δ·shift(a,j)
    new_conv[i: i + n] -= 2.0 * delta * a
    new_conv[j: j + n] += 2.0 * delta * a
    # quadratic part
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a list of floats (the best sequence found before the time budget expires).
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ----------------------------------------------------------------------
    # 1) Initialise the vector a.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        n0 = a0.size
        # keep the length modest for the LP phase (≈2000–4000)
        n = min(2048, max(512, n0))
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.concatenate([a0, np.zeros(n - a0.size)])
        a = a0.copy()
    else:
        n = 2048
        a = np.full(n, 1.0, dtype=np.float64)

    # normalise to the standard sum √(2 n)
    target_sum = math.sqrt(2.0 * n)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)

    conv = _self_convolution(a)
    cur_score = conv.max()                     # after normalisation this equals the evaluate value
    best = a.copy()
    best_score = cur_score
    last_improve = start_time

    # ----------------------------------------------------------------------
    # 2) Phase 1 – Linear‑programming boost.
    # ----------------------------------------------------------------------
    max_lp_iter = 180
    iteration = 0
    while time.monotonic() < deadline and iteration < max_lp_iter:
        # recompute convolution and its peak
        conv = _self_convolution(a)
        M = conv.max()
        cur_score = M

        # Build the constraint matrix A where (A b)_k = (a*b)_k.
        # It consists of n diagonals, the i‑th diagonal (offset = i) is filled with a[i].
        diag_data = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        A_ub = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")

        # Linear programme: maximise 1ᵀb  <=> minimise -1ᵀb
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve with HiGHS (fast interior‑point/simplex)
        res = opt.linprog(
            c_obj,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # LP failed – fall back to the greedy phase
            print("LP failed, terminating LP phase.")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b == 0.0:
            # nothing useful can be added
            break

        # Scale b back to the required ℓ₁‑norm.
        g = b * (target_sum / sum_b)

        # Mixing coefficient – large at the start, shrinking to ~0.02.
        t = 0.22 - 0.20 * (iteration / max_lp_iter)   # from 0.22 down to 0.02
        a = (1.0 - t) * a + t * g

        # Re‑normalise (numerical drift)
        a = a * (target_sum / a.sum())

        # Evaluate the new peak
        conv = _self_convolution(a)
        cur_score = conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] LP phase new best {best_score:.6f}")

        # occasional random jitter to escape shallow plateaus
        if rng.random() < 0.01:
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())

        iteration += 1
        if best_score < 1.5025:
            # target already reached – stop early
            break

    # ----------------------------------------------------------------------
    # 3) Phase 2 – Greedy targeted mass transfers (fine‑tuning).
    # ----------------------------------------------------------------------
    # Only run this phase if time permits.
    while time.monotonic() < deadline:
        conv = _self_convolution(a)
        M = conv.max()
        cur_score = M

        # locate the index of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # partner values a[k_max‑i] for all i where defined
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        partner = np.zeros_like(a)
        partner[valid] = a[k_max - idx[valid]]

        # donor = index with large a[i]·partner[i]
        donor_mask = a > 1e-12
        donor_score = partner * donor_mask.astype(float)
        if donor_score.max() == 0.0:
            # fallback: random move
            i = rng.integers(0, n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_delta = 0.25 * a[i]
            delta = rng.random() * max_delta
            a[i] -= delta
            a[j] += delta
            conv = _self_convolution(a)
            cur_score = conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] random move new best {best_score:.6f}")
            continue

        donor = int(np.argmax(donor_score))

        # receiver = index with the smallest partner value (ideally zero)
        min_partner = partner.min()
        receiver_candidates = np.where(partner == min_partner)[0]
        # avoid picking the donor itself
        if donor in receiver_candidates:
            receiver_candidates = receiver_candidates[receiver_candidates != donor]
        if receiver_candidates.size == 0:
            # take any other index
            receiver = (donor + 1) % n
        else:
            receiver = int(rng.choice(receiver_candidates))

        # step size – larger early on, shrink later
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        max_frac = 0.25 * (1.0 - frac) + 0.03 * frac
        max_delta = a[donor] * max_frac
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta

        # apply the transfer using an incremental convolution update
        new_conv = _apply_transfer(a, conv, donor, receiver, delta)
        new_score = new_conv.max()
        if new_score < cur_score:
            a[donor] -= delta
            a[receiver] += delta
            conv = new_conv
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] greedy new best {best_score:.6f}")

        # occasional block‑wise multiplicative jitter
        if rng.random() < 0.015:
            block_width = max(1, int(0.12 * n))
            start_blk = rng.integers(0, n - block_width + 1)
            factor = math.exp(rng.normal(0.0, 0.12))
            a[start_blk:start_blk + block_width] *= factor
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())
            conv = _self_convolution(a)
            cur_score = conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] block‑jitter improved best {best_score:.6f}")

        # restart if stuck for a while
        if time.monotonic() - last_improve > 30.0:
            a = best.copy()
            # small random perturbation of the best solution

            cur_score = conv.max()

        if best_score < 1.5025:

    # 4) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
