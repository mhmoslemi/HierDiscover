"""
Propose a step‑function by minimizing the maximal self‑convolution value.
The algorithm works on the probability simplex (a_i ≥ 0, Σ a_i = 1) and
optimises a smooth approximation of the max‑convolution:
  L(a) = log‑sum‑exp(β·(a*a)) .
For β → ∞   L(a) ≈ β·max(a*a)  and the sub‑gradient of L(a) is
  ∇L_i = β·2·∑_k w_k·a_{k-i}  where w_k = soft‑max(β·(a*a))_k .
We perform exponentiated‑gradient (mirror‑descent) updates on the simplex,
with a linear schedule for the stepsize η and for β (annealing from a
soft to a sharp approximation).  Occasional random “block‑perturb’’ and
pair‑wise mass‑transfer moves plus timed restarts help to escape local
plateaus.  The best vector found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be defined at top level – no closures)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    # rfft works on real input, returns complex spectrum of size fft_len//2+1
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable soft‑max of β·x (returns a probability vector)."""
    y = x * beta
    m = np.max(y)
    exp_y = np.exp(y - m)
    s = np.sum(exp_y)
    if s == 0.0:
        # fallback to uniform distribution (should not happen)
        return np.full_like(x, 1.0 / x.size)
    return exp_y / s


def _gradient_weighted(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp approximation L(a)=log∑exp(β·conv)
    with respect to a (up to the factor η in the exponentiated‑gradient).
    """
    # w_k = softmax(β·conv_k)
    w = _softmax(conv, beta)               # shape (2n-1,)
    # ∇L_i = β·2·∑_k w_k·a_{k-i}  = β·2·(w ⋆ a)[i]  (cross‑correlation)
    # np.correlate with mode='valid' gives exactly ∑_j w_{i+j} a_j, i=0..n-1
    grad = 2.0 * np.correlate(w, a, mode='valid')
    grad *= beta
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    # block width at most n/8 (to keep perturbations local)
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector (unit sum)
    # ------------------------------------------------------------------
    # Try to reuse a previously discovered good solution if it exists.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim != 1 or cand.size == 0 or not np.isfinite(cand).all():
                raise ValueError
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
        except Exception:
            # fallback to a modest size uniform vector
            default_n = 2048
            a = np.full(default_n, 1.0 / default_n, dtype=np.float64)
    else:
        default_n = 2048
        a = np.full(default_n, 1.0 / default_n, dtype=np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping and performance tracking
    # ------------------------------------------------------------------
    # Compute the initial score using the external evaluator (once)
    # The evaluator is homogeneous, so we can safely keep Σ a_i = 1.
    best = a.copy()
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:   # safety – the harness guarantees it exists
        # fallback manual computation
        conv = _conv_self_fft(a, fft_len)
        best_score = 2.0 * n * conv.max()
    print(f"[0.00s] initial score = {best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # small safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (all linear schedules)
    # ------------------------------------------------------------------
    # stepsize η: start relatively aggressive, end gentle
    eta_start = 5.0
    eta_end = 0.02

    # soft‑max temperature β: start soft (β≈0.1), finish hard (β≈25)
    beta_start = 0.15
    beta_end = 22.0

    momentum = 0.85
    grad_prev = np.zeros_like(a)

    block_perturb_prob = 0.015
    pair_move_prob = 0.04
    restart_gap = 28.0          # seconds without improvement → restart

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ----- compute self‑convolution and current score -----
        conv = _conv_self_fft(a, fft_len)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        # ----- keep the best solution found so far -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule of η and β (linear annealing) -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start * (1.0 - elapsed / total) + beta_end * (elapsed / total)

        # ----- gradient of the smoothed max (log‑sum‑exp) -----
        grad = _gradient_weighted(a, conv, beta)

        # ----- momentum smoothing -----
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror descent) update -----
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pair‑wise mass transfer -----
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # move a random fraction of mass from i to j
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate /= a_candidate.sum()
                    # cheap evaluation using convolution
                    conv_cand = _conv_self_fft(a_candidate, fft_len)
                    cand_score = 2.0 * n * conv_cand.max()
                    # accept if better or with a tiny uphill probability
                    if cand_score < cur_score or rng.random() < np.exp(
                        (cur_score - cand_score) / max(1e-12, eta)
                    ):
                        a = a_candidate
                        # grad_prev reset to avoid stale momentum after a jump
                        grad_prev = np.zeros_like(a)

        # ----- soft restart when stuck for a while -----
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter all coordinates multiplicatively (log‑normal noise)
            jitter = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

        # ----- optional early termination if we already hit the target -----
        # The target bound is 1.5025; we can stop a tiny bit earlier.
        if best_score <= 1.5025:
            print(f"[{elapsed:6.2f}s] target reached (score {best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # 5) Return the best vector (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
