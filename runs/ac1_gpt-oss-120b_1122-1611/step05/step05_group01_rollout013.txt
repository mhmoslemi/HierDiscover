"""
Optimise a non‑negative step‑function for the “max‑self‑convolution” loss.

The algorithm proceeds in three stages:

1.  Deterministic families.
    For several lengths *n* and several exponent parameters *α* we build a
    discrete arcsine‑type shape  

        a_i ∝ 1 / ((i+½)(n‑½‑i))**α .

    The shape is normalised to sum = 1 (the loss is scale‑invariant) and its
    score is evaluated.  The best of those candidates is kept.

2.  Smoothing‑gradient refinement.
    Starting from each candidate we repeatedly minimise the loss
    *L(a)=2·n·max(conv(a,a)) / (∑a)²* using the following mirror‑descent
    step (no explicit constraints – positivity is obtained by the
    exponential update and renormalisation to the simplex).

        • compute the full self‑convolution c = a*a.
        • build a soft‑max weight vector w_k ∝ exp(c_k / τ)   (τ > 0).
          This is a smooth approximation of the max‑operator; as τ → 0 the
          weight concentrates on the peak of the convolution.
        • the gradient of   f(a)=∑_k w_k·c_k   w.r.t. a is

                ∇f_i = 2·∑_k w_k·a_{k‑i}
                     = 2·(a ⋆ w)_i       (⋆ = cross‑correlation).

          It is computed efficiently with a single FFT‑based convolution.
        • exponential (mirror) update   a_i ← a_i·exp(‑η·∇f_i)  and re‑normalise.

    η (step size) and τ (temperature) are slowly decayed, while occasional
    random jitter helps to escape plateaus.  The current candidate is scored
    with the provided ``evaluate_sequence`` every few iterations; the best
    overall sequence is retained.

3.  Fallback.
    If everything fails a tiny uniform vector is returned.

The whole search respects the 1000 s wall‑time budget.  All helper
functions are defined at the top level (no closures) and the final
sequence contains only finite numbers in the interval [0, 1000].
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_size(size: int) -> int:
    """Return the smallest power‑of‑two ≥ size (useful for FFT length)."""
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real vector a using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)‑1."""
    size = x.size + y.size - 1
    fft_len = _fft_len_for_size(size)
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    conv = np.fft.irfft(fz, n=fft_len)[:size]
    return conv


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """Gradient of the soft‑max smoothed max‑convolution.
    Parameters
    ----------
    a   – current sequence (len = n, sum = 1)
    conv – self‑convolution of ``a`` (len = 2·n‑1)
    tau  – temperature (>0)

    Returns
    -------
    grad – ndarray of shape (n,) containing ∇f_i.
    """
    # stable soft‑max on the convolution values
    shifted = (conv - conv.max()) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w  =  (a * rev(w)) at appropriate indices
    w_rev = w[::-1]
    full = _conv_general(a, w_rev)                # length 3·n‑2
    start = w.size - 1                             # = 2·n‑2
    grad = 2.0 * full[start - np.arange(a.size)]
    return grad

                break

            # ----- build the raw arcsine‑type shape --------------------
            raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
            raw = np.clip(raw, 0.0, 1000.0)
            raw /= raw.sum()                     # normalise to sum = 1
            seq = raw.tolist()
            sc = evaluate_sequence(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] raw candidate n={n:5d} α={alpha:.3f} "
                      f"score={sc:.6f}")

            # ----- start a short mirror‑descent refinement -------------
            a = raw.copy()                         # work on a NumPy array
            # hyper‑parameters for this run
            eta = 0.6                               # initial step size
            tau = 0.25                               # initial temperature
            it = 0
            # we limit the inner loop to at most ~2 000 iterations or the
            # remaining time budget, whichever comes first
            while time.monotonic() < deadline and it < 2000:
                # 1) self‑convolution
                conv = _conv_self(a, fft_len_self)          # length 2·n‑1

                # 2) gradient of a smoothed max‑convolution
                grad = _softmax_gradient(a, conv, tau)

                # 3) exponential (mirror) update and normalisation
                a = a * np.exp(-eta * grad)

                # occasional tiny jitter to avoid stagnation
                if rng.random() < 0.015:
                    jitter = np.exp(rng.normal(0.0, 0.07, size=n))
                    a *= jitter

                # project back onto the probability simplex
                a = np.clip(a, 0.0, None)
                total = a.sum()
                if total == 0.0:
                    a = np.full_like(a, 1.0 / n)
                    total = 1.0
                a /= total

                # periodic evaluation (about every 0.2 s on typical hardware)
                if (it & 31) == 0:   # every 32 iterations
                    sc = evaluate_sequence(a.tolist())
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a.tolist()
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] refined n={n:5d} α={alpha:.3f} "
                              f"it={it:4d} score={sc:.6f}")

                # decay schedules (very gentle)
                eta *= 0.9996
                tau = max(0.005, tau * 0.9999)

                it += 1

    # ------------------------------------------------------------------
    # 2)  Light random‑restart hill‑climbing (optional, cheap)
    # ------------------------------------------------------------------
    # If the deterministic sweep did not explore a particular length we
    # can try a few random perturbations around the current best.
    if best_seq is not None:
        a = np.array(best_seq, dtype=np.float64)
        n = a.size
        fft_len_self = _fft_len_for_n(n)
        # short stochastic climb
        eta = 0.4
        for step in range(800):
            if time.monotonic() > deadline:
                break
            # pick two indices and move a tiny amount from i→j
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = 1e-4 * a[i]    # proportional move
            if delta <= 0:
                continue
            b = a.copy()
            b[i] -= delta
            b[j] += delta
            # renormalise (numerical drift)
            b = np.clip(b, 0.0, None)
            s = b.sum()
            if s == 0.0:
                continue
            b /= s
            sc = evaluate_sequence(b.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = b.tolist()
                a = b
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] random tweak step={step:4d} score={sc:.6f}")
            else:
                # occasional acceptance of neutral moves to inject noise
                if rng.random() < 0.001:
                    a = b

    # ------------------------------------------------------------------
    # 3)  Fallback if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # final safety clipping (required by evaluate_sequence)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
