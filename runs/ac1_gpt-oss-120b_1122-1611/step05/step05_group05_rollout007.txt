    n = a.shape[0]

    return float(np.max(conv[:2*n-1]))

"""
Hybrid gradient‑descent / simulated‑annealing optimizer.

The algorithm works on the probability simplex (sum = 1, a_i ≥ 0) and
minimises the quantity

    score(a) = 2·n·max(a * a)

where “*” is the linear convolution.  Starting from the globally supplied
`height_sequence_1` (or a uniform distribution) we repeatedly perform one of
two moves:

* **Random block perturbation** – multiplicative log‑normal noise on a
  random contiguous chunk, followed by projection onto the simplex.
* **Gradient descent step** – identify the index `k_max` where the current
  convolution attains its maximum, compute the exact gradient of that
  maximum (∂max/∂a_i = 2·a_{k_max‑i}) and move a small step opposite the
  gradient, again projected onto the simplex.

The step size for gradient steps and the acceptance probability for
non‑improving moves follow a linear schedule that cools down during the
run.  If no improvement is seen for `restart_gap` seconds we jitter the
current best solution and continue.  The best sequence found before the
deadline is returned.

All helper functions are top‑level, there are no closures or lambdas, and
the routine respects the provided time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a 1‑D array onto the probability simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random multiplicative perturbation of a contiguous block
def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a log‑normal factor to a random block and re‑project."""
    n = a.size
    # block length between 1 and max(2, n//5)
    width = rng.integers(1, max(2, n // 5))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))   # log‑normal factor
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Fast max‑convolution using real FFT (linear convolution)
def _max_conv_fft(a: np.ndarray) -> float:
    """Return max(a * a) using zero‑padded FFT (linear convolution)."""
    n = a.shape[0]
    L = 2 * n                     # power‑of‑two not required, next‑power works fine
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    # Numerical noise may create tiny negative values – the max stays correct
    return float(np.max(conv))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (probability distribution)
    that (approximately) minimises ``evaluate_sequence`` within the given
    time budget.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # 1) Choose vector length and initial point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
                raise ValueError
        except Exception:
            init = None
    else:
        init = None

    if init is None:
        n = 4096                     # default size – fits comfortably in the budget
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = init.size
        a = np.clip(init, 0.0, None)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s                # ensure sum = 1

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _max_conv_fft(a)   # sum(a) == 1, so denominator = 1
    start_time = time.monotonic()
    last_improve = start_time

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3) Search parameters
    # ------------------------------------------------------------------
    accept_start = 0.12            # acceptance prob for worse moves (start)
    accept_end   = 0.001           # ... (end)
    restart_gap  = 30.0            # seconds without improvement → restart
    grad_prob_start = 0.4          # probability to perform a gradient step early
    grad_prob_end   = 0.8          # later, favour gradient descent
    eta_start = 0.10               # step size for gradient descent (initial)
    eta_end   = 0.001              # ... (final)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    cur_score = best_score
    cur_maxconv = _max_conv_fft(a)

    while time.monotonic() < start_time + max(0.0, float(budget_s) - 0.5):
        elapsed = time.monotonic() - start_time
        frac_time = min(1.0, elapsed / (budget_s - 0.5))

        # ------------------------------------------------------------------
        #   Decide which move to perform
        # ------------------------------------------------------------------
        prob_grad = grad_prob_start * (1.0 - frac_time) + grad_prob_end * frac_time
        if rng.random() < prob_grad:
            # ---- Gradient descent step ----
            k_max = int(np.argmax(_max_conv_fft(a) and np.convolve(a, a)))  # placeholder
            # The above line is only to get k_max; we recompute conv soon anyway.
            # Compute full convolution once (fast FFT)
            n_fft = 2 * n
            fa = np.fft.rfft(a, n=n_fft)
            conv_full = np.fft.irfft(fa * fa, n=n_fft)[: 2 * n - 1]
            k_max = int(np.argmax(conv_full))

            # gradient of the maximum entry with respect to a
            idx = k_max - np.arange(n)               # possible partner indices
            valid = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[valid] = 2.0 * a[idx[valid]]

            # scheduled step size
            eta = eta_start * (1.0 - frac_time) + eta_end * frac_time
            cand = a - eta * grad
        else:
            # ---- Random block perturbation (exploration) ----
            cand = _random_block_perturb(a, rng)

        # Re‑project onto the simplex (guarantees non‑negativity & sum = 1)
        cand = _project_simplex(cand)

        # Compute score for the candidate (fast FFT)
        cand_maxconv = _max_conv_fft(cand)
        cand_score = 2.0 * n * cand_maxconv

        # ------------------------------------------------------------------
        #   Acceptance logic (simulated annealing)
        # ------------------------------------------------------------------
        if cand_score < cur_score:
            # improvement – accept unconditionally
            a = cand
            cur_score = cand_score
            cur_maxconv = cand_maxconv
        else:
            # possibly accept a worse move
            accept_prob = accept_start * (1.0 - frac_time) + accept_end * frac_time
            if rng.random() < accept_prob:
                a = cand
                cur_score = cand_score
                cur_maxconv = cand_maxconv

        # ------------------------------------------------------------------
        #   Update global best
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:     # strict improvement
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        #   Restart if we are stuck too long
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > restart_gap:
            # jitter the current best to escape a plateau
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = best * noise
            a = _project_simplex(a)
            cur_score = 2.0 * n * _max_conv_fft(a)
            cur_maxconv = _max_conv_fft(a)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best candidate (clip to the allowed magnitude range)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
