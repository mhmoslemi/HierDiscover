"""
Propose a candidate step function by alternating a linear‑program “flatten‑the‑max”
step with simple stochastic refinements.

Algorithm
---------
1.  Initialise a non‑negative vector `a` (sum = 1).  If a previous construction
    `height_sequence_1` exists we reuse it, otherwise we start from a random
    Dirichlet draw of length *n* (default 4000).

2.  Loop until the time budget expires
    * Compute the self‑convolution `c = a * a` (FFT, O(n log n)).
    * Let `M = max(c)`.  The current score is `score = 2·n·M` because the
      denominator `sum(a)^2` equals 1 after projection onto the simplex.
    * Keep the best vector seen so far.
    * Build a tiny linear programme:
          maximise   Σ b_j
          subject to   (a * b)_k ≤ M   for a few “active’’ indices k
                       b_j ≥ 0
      Only the indices where `c` is close to its maximum are kept (at most
      12 constraints).  This LP finds a new distribution `b` that spreads the
      mass without increasing the current peak `M`.
    * Mix the old and new vectors:
          a ← (1‑τ)·a + τ·b   with τ slowly decreasing from 0.2 to 0.02.
      The mixture is projected back onto the probability simplex so the sum
      stays equal to one.
    * Every few seconds we inject a multiplicative block perturbation
      (log‑normal factor on a random contiguous block) and, if no improvement
      occurred for a long while, we apply a stronger global jitter.
    * After the LP phase we finish with a short simulated‑annealing style
      pairwise‑mass‑transfer refinement (exact O(n) update) to polish the
      solution.

The LP‑step is the key novelty compared with the previous smooth‑max approach;
it directly enforces the current peak bound on the most critical convolution
entries and therefore drives the peak down more aggressively.

The function returns the best sequence found (clipped to the limits required
by `evaluate_sequence`).  The whole routine respects the 1000 s wall‑time
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / FFT helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)      # kill tiny negatives
    return conv_full


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σx_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _pairwise_transfer_exact(a: np.ndarray, conv: np.ndarray,
                             rng: np.random.Generator,
                             max_delta_frac: float = 0.3):
    """
    Exact O(n) update for moving a random amount of mass from index i to j.
    Returns (new_a, new_conv, new_score) if the move reduces the objective,
    otherwise (a, conv, None).
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Update convolution (ordered pair contributions)
    new_conv = conv.copy()
    # remove donor contribution
    new_conv[i:i + n] -= 2.0 * delta * a
    # add receiver contribution
    new_conv[j:j + n] += 2.0 * delta * a
    # diagonal corrections
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    # cross‑term second order
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)

    new_score = 2.0 * a.size * new_conv.max()
    cur_score = 2.0 * a.size * conv.max()
    if new_score < cur_score - 1e-15:
        return a2, new_conv, new_score
    else:
        return a, conv, None


# ----------------------------------------------------------------------
# Linear program solving (active‑set of constraints)
# ----------------------------------------------------------------------


def _solve_lp_against(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Solve  max Σ b_j   subject to   (f * b)_k ≤ M   for k ∈ active_idx,
    b_j ≥ 0.
    Returns a feasible b (projected onto the simplex; if the LP fails,
    a uniform vector is returned).
    """
    import cvxpy as cp

    n = f.size
    b = cp.Variable(n, nonneg=True)

    constraints = []
    # Build each active‑set constraint: Σ_j f_{k‑j} b_j ≤ M
    for k in active_idx:
        coeff = np.zeros(n, dtype=np.float64)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            coeff[idx] = f[k - idx]          # f_{k‑j}
        constraints.append(coeff @ b <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:                     # fall back to a safe solver
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            prob.status = 'infeasible'

    if b.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        # LP failed – return a uniform distribution
        b_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        b_vec = np.asarray(b.value, dtype=np.float64)
        b_vec = np.clip(b_vec, 0.0, None)

    s = b_vec.sum()
    if s > 0:
        b_vec /= s
    else:
        b_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return b_vec

    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0 and np.isfinite(a0).all():
            n = a0.size
            a = np.clip(a0, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = 4000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(2 * n - 1)          # length for self‑convolution FFT

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    best_conv = _self_conv_fft(best, fft_len)
    best_score = 2.0 * n * best_conv.max()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin
    last_improve_iter = 0
    iteration = 0

    # parameters for the alternating LP / mixing schedule
    tau_start = 0.20
    tau_end = 0.02
    max_active = 12          # at most this many LP constraints
    rel_tol = 1e-4           # tolerance for “active’’ indices

    # --------------------------------------------------------------
    # 3) Main optimisation loop (LP + mixing)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time

        # current convolution and score
        conv = _self_conv_fft(a, fft_len)
        M = conv.max()
        cur_score = 2.0 * n * M                     # sum(a) = 1 by construction

        # keep the best vector seen so far
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve_iter = iteration
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        #   3a) build a tiny active set of constraints (indices near M)
        # ----------------------------------------------------------
        # pick up to `max_active` indices with largest convolution values
        if conv.size <= max_active:
            active = np.arange(conv.size)
        else:
            # argpartition gives the top‑max_active values (unsorted)
            topk = np.argpartition(conv, -max_active)[-max_active:]
            # keep only those sufficiently close to the maximum
            active = topk[conv[topk] >= M * (1.0 - rel_tol)]
            if active.size == 0:                     # fallback – always include the argmax
                active = np.array([int(np.argmax(conv))])

        # ----------------------------------------------------------
        #   3b) linear‑program step: find a partner vector `b`
        # ----------------------------------------------------------
        b = _solve_lp_against(a, M, active)

        # ----------------------------------------------------------
        #   3c) convex combination and projection
        # ----------------------------------------------------------
        frac = (now - start_time) / (deadline - start_time)
        tau = tau_start * (1.0 - frac) + tau_end * frac   # anneal mixing weight
        a = (1.0 - tau) * a + tau * b
        a = _project_onto_simplex(a)

        # ----------------------------------------------------------
        #   3d) occasional global jitter to escape plateaus
        # ----------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        if iteration - last_improve_iter > 2000:
            # long stagnation → apply a stronger multiplicative jitter
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            a *= noise
            a = _project_onto_simplex(a)
            last_improve_iter = iteration   # reset counter

    # --------------------------------------------------------------
    # 4) Fine‑grained simulated‑annealing refinement (pairwise moves)
    # --------------------------------------------------------------
    # run a short SA phase for the last few seconds of the budget
    refinement_deadline = deadline + 1.0    # a little extra time for final polishing
    a_ref = best.copy()
    conv_ref = _self_conv_fft(a_ref, fft_len)
    score_ref = 2.0 * n * conv_ref.max()
    sa_start = time.monotonic()
    sa_temp_start = max(0.02 * score_ref, 1e-3)
    sa_temp_end = 1e-7 * score_ref

    while time.monotonic() < refinement_deadline:
        now = time.monotonic()
        frac = (now - sa_start) / (refinement_deadline - sa_start)
        T = sa_temp_start * (1.0 - frac) + sa_temp_end * frac

        a_ref, conv_ref, new_score = _pairwise_transfer_exact(
            a_ref, conv_ref, rng, max_delta_frac=0.3)

        if new_score is not None and new_score < score_ref:
            score_ref = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_ref.copy()
                print(f"[{now - start_time:6.2f}s] SA improvement to {best_score:.6f}")
        else:
            # Metropolis uphill acceptance
            cur_score = 2.0 * n * conv_ref.max()
            if cur_score > score_ref:
                delta = cur_score - score_ref
                if rng.random() < math.exp(-delta / max(T, 1e-12)):
                    score_ref = cur_score

    # --------------------------------------------------------------
    # 5) Return best sequence (clipped to limits of evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
