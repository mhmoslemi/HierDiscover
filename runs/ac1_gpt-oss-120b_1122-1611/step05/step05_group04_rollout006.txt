"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum a)^2

where conv denotes the linear self‑convolution.  The algorithm consists of

1. **Fast parametric warm‑start** – we scan a one‑parameter family
   `a_i ∝ ((i+1)*(n-i))^{-p}` (U‑shaped, heavier at the ends) and convex
   mixtures with the uniform distribution.  The best vector from this cheap
   grid search becomes the starting point.

2. **Targeted simulated‑annealing refinement** – at every step we look at the
   current maximal convolution index `k*`.  The sub‑gradient of the
   `L_∞`‑norm tells us that the entries `a_i` for which the product
   `a_i * a_{k*-i}` is large are the ones that contribute most to the peak.
   We therefore choose a donor index `i` with probability proportional to that
   product and move a random fraction of its mass to a uniformly chosen
   receiver `j`.  The convolution is updated in O(n) time using the exact
   formula for a pairwise transfer, so we can perform many moves inside the
   1000 s budget.

   Moves are accepted according to a simple Metropolis rule with a linear
   temperature schedule; a small jitter (multiplicative log‑normal block
   perturbation) is applied periodically and when the search stagnates for
   >60 s.

3. **Book‑keeping** – the best vector found so far is stored; every improvement
   is printed.  The final vector is clipped to the limits required by
   `evaluate_sequence` and returned as a plain list of floats.

The whole search runs on a single (or two) CPU cores, uses only NumPy and
standard library modules, and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negative values caused by floating‑point noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to counter floating‑point drift.
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)

    return _project_onto_simplex(out)


def _apply_pairwise_move(a: np.ndarray,
                         conv: np.ndarray,
                         i: int,
                         j: int,
                         delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (new_a, new_conv) after moving `delta` mass from index i to j.
    The update of the self‑convolution is performed in O(n) time.
    """
    n = a.size
    new_conv = conv.copy()

    # remove contribution of donor i
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a

    # add contribution of receiver j
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a

    # diagonal / cross‑term corrections
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    # update the coefficient vector
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)        # safeguard against negatives

    # Numerical drift is tiny – we keep the sum unchanged.
    return a2, new_conv


def _initial_candidate(n: int,
                       rng: np.random.Generator,
                       fft_len: int) -> tuple[np.ndarray, float]:
    """
    Scan a simple parametric family (U‑shaped power law) and mixtures with
    the uniform distribution to obtain a good starting vector.
    Returns (vector, score).
    """
    # grids for the exponent and the mixture weight
    p_grid = np.linspace(0.2, 2.0, 19)          # exponent controlling end‑bias
    lam_grid = np.linspace(0.0, 0.5, 11)       # mixture with uniform

    # pre‑compute the uniform vector and its score (the baseline 2.0)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    conv_uni = _self_conv_fft(uniform, fft_len)
    best_a = uniform.copy()
    best_score = 2.0 * n * conv_uni.max()       # = 2.0 exactly

    # enumerate candidates
    idx = np.arange(n, dtype=np.float64)        # 0 … n‑1
    i_plus = idx + 1.0
    n_minus_i = n - idx

    for p in p_grid:
        # U‑shaped power‑law: a_i ∝ ((i+1)*(n-i))^{-p}
        shape = (i_plus * n_minus_i) ** (-p)
        shape_sum = shape.sum()
        if shape_sum == 0:
            continue
        shape /= shape_sum                     # now a probability vector

        for lam in lam_grid:
            # convex mixture with the uniform vector
            a = lam * uniform + (1.0 - lam) * shape
            # a already sums to 1 (both components do)
            conv_a = _self_conv_fft(a, fft_len)
            score = 2.0 * n * conv_a.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def _targeted_pairwise_move(a: np.ndarray,
                            conv: np.ndarray,
                            k_star: int,
                            rng: np.random.Generator,
                            max_delta_frac: float) -> tuple[np.ndarray,
                                                            np.ndarray,
                                                            float,
                                                            bool]:
    """
    Perform a single targeted pairwise transfer.
    Returns (new_a, new_conv, new_score, accepted_flag).
    """
    n = a.size

    # --- choose donor according to sub‑gradient weights w_i = a_i * a_{k*-i} ---
    idx = np.arange(n, dtype=np.int64)
    partner = k_star - idx
    mask = (partner >= 0) & (partner < n)
    w = np.zeros_like(a)
    w[mask] = a[mask] * a[partner[mask]]
    total_w = w.sum()
    if total_w > 0.0:
        probs = w / total_w
        i = rng.choice(n, p=probs)
    else:
        i = rng.integers(0, n)

    # --- choose a receiver uniformly (different from i) ---
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1

    # --- amount to move ----------------------------------------------------
    if a[i] <= 0.0:
        return a, conv, 2.0 * n * conv.max(), False
    delta = rng.random() * max_delta_frac * a[i]   # 0 ≤ delta ≤ max_delta_frac·a[i]

    # --- apply the exact O(n) update ---------------------------------------
    a2, conv2 = _apply_pairwise_move(a, conv, i, j, delta)
    new_score = 2.0 * n * conv2.max()
    return a2, conv2, new_score, True


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best coefficient list found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine problem size and initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)
        else:
            n = 4000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len_self = _next_pow2(2 * n - 1)          # for a*a

    # ------------------------------------------------------------------
    # 2) Warm‑start via a cheap parametric search (may overwrite a)
    # ------------------------------------------------------------------
    a0, score0 = _initial_candidate(n, rng, fft_len_self)
    # compute the score of the provided global candidate, keep the best
    conv0 = _self_conv_fft(a0, fft_len_self)
    cur_score = 2.0 * n * conv0.max()
    if cur_score < score0:
        # should not happen, but keep the direct evaluation
        cur_score = cur_score
        a = a0.copy()
        conv = conv0.copy()
        best_score = cur_score
        best = a.copy()
    else:
        # use the parametric candidate
        a = a0.copy()
        conv = conv0.copy()
        best = a.copy()
        best_score = cur_score

    # ------------------------------------------------------------------
    # 3) Book‑keeping and SA schedule
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # temperature parameters (linear schedule)
    T_start = best_score * 0.12          # a modest fraction of the current value
    T_end = best_score * 1e-6

    # move‑size parameters
    max_delta_start = 0.5
    max_delta_end = 0.02

    # block perturb probability
    block_prob = 0.015
    stagnation_limit = 60.0   # seconds without improvement before a reset

    # ------------------------------------------------------------------
    # 4) Main simulated‑annealing loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # update schedules
        T = T_start * (1.0 - frac) + T_end * frac
        max_delta_frac = max_delta_start * (1.0 - frac) + max_delta_end * frac

        # locate current peak of the convolution
        k_star = int(np.argmax(conv))

        # propose a targeted pairwise transfer
        a_new, conv_new, new_score, moved = _targeted_pairwise_move(
            a, conv, k_star, rng, max_delta_frac)

        # acceptance test (Metropolis)
        if new_score < cur_score:
            accept = True
        else:
            # uphill move – accept with prob exp(-(Δ)/T)
            delta_obj = new_score - cur_score
            accept = rng.random() < math.exp(-delta_obj / max(T, 1e-12))

        if accept and moved:
            a, conv, cur_score = a_new, conv_new, new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:7.2f}s] new best score = {best_score:.6f}")

        # occasional global block perturbation (helps escape plateaus)
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)
            conv = _self_conv_fft(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:7.2f}s] block perturb improved to {best_score:.6f}")

        # stagnation handling: reset with a mild jitter if we have been idle
        if now - last_improve > stagnation_limit:
            jitter = np.exp(rng.normal(0.0, 0.25, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)   # keep sum = 1
            conv = _self_conv_fft(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{elapsed:7.2f}s] jitter reset – current score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (apply the evaluator's hard caps)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
