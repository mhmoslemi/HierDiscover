"""
Gradient‑based flattening of the self‑convolution.

The evaluation function is
    score = 2·n·max(conv(a,a)) / (sum(a)²)
with a normalized to sum(a)=1.  For a uniform vector the score is 2.
The goal is to make the distribution of the sum X+Y (X,Y i.i.d. with
probabilities a) as flat as possible, because a flatter autocorrelation
reduces its maximum.

The algorithm works as follows:

*   Initialise a as the supplied ``height_sequence_1`` (if present) or
    a uniform probability vector.
*   Repeatedly:
        –   compute the linear convolution a∗a by FFT,
        –   form the error δ = conv – C where C = 1/(2·n‑1) is the target
            constant level (the ideal flat convolution),
        –   compute the gradient of the squared error
                L = Σ (conv[t]–C)²
            which is  g = 4·(δ * a_reversed) (a simple convolution).
        –   update a ← a – η·g, clip to non‑negative and project back onto
            the probability simplex.
        –   keep the best vector seen (lowest score) and occasionally
            inject a tiny random block perturbation or restart if no
            improvement has occurred for a while.

The step size η decays slowly and the search is stopped when the
time budget (default 1000 s) is exhausted.  The best sequence found is
returned as a plain ``list[float]``.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 5))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out

    """

            a = np.clip(a, 0.0, None)

            if s == 0.0:

        except Exception:

    else:

    if a is None:

    else:

        cur_score = float(2.0 * n * conv.max())
        if cand_score < cur_score:

        else:

            if rng.random() < 0.001:
                a = _random_block_perturb(a, rng)

        # ----- decay learning rate -----
        lr *= decay

        # ----- restart if we are stuck -----
        now = time.monotonic()
        if now - last_improve > restart_gap:

            noise = np.exp(rng.normal(0.0, 0.10, size=n))

            print(f"[{elapsed:6.2f}s] iteration {iteration}, best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best candidate (clip to allowed magnitude)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
