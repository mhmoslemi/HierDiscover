"""
Propose a candidate step function that improves the current best score.

The algorithm consists of two stages:

1. **Hard‑max mirror‑descent** (≈55 % of the time budget)
   * At each iteration we compute the exact self‑convolution, locate the
     index `k_max` where it attains its maximum and use the sub‑gradient of
     that maximum (the correlation of `a` with the unit impulse at `k_max`).
   * An exponentiated‑gradient (mirror descent on the probability simplex)
     update `a ← a·exp(-η·grad)` is applied, followed by Euclidean projection
     onto the simplex.
   * The step‑size η is annealed from 2.0 to 0.01.
   * Occasionally a multiplicative log‑normal block perturbation is applied
     and a stronger jitter reset is triggered if no improvement occurs for
     45 s.

2. **Exact pair‑wise refinement (simulated‑annealing style)**
   * Starts from the best vector found in stage 1.
   * Repeatedly moves a random amount of mass from a donor index to a
     receiver index, updating the convolution in O(n) time (see
     `_pairwise_transfer_exact`).  The donor‑to‑receiver fraction may be as
     large as 90 % of the donor mass.
   * Moves are accepted if they strictly improve the objective; otherwise
     they are accepted with Metropolis probability based on a temperature that
     decays linearly from 0.07·score to 1e‑5·score.
   * Every 2 % of the refinement steps a random block perturbation is tried.
   * The global best vector (and its score) is kept throughout.

Both stages respect the 1000 s budget (a 5 s safety margin is kept).  All
coefficients stay non‑negative, are projected onto the simplex (so their sum is 1)
and finally clipped to the limits required by `evaluate_sequence`.  The returned
list of floats can be fed directly to the evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core linear‑algebra helpers (FFT based self‑convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a with itself using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Eliminate tiny negatives caused by numerical noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1 }.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Random block perturbations (log‑normal multiplicative noise)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a randomly chosen contiguous block by a log‑normal factor and
    re‑project onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


# ----------------------------------------------------------------------
# Exact O(n) pairwise mass transfer (used in the refinement stage)
# ----------------------------------------------------------------------


def _pairwise_transfer_exact(a: np.ndarray, conv: np.ndarray,
                             rng: np.random.Generator,
                             max_delta_frac: float = 0.5):
    """
    Move a small amount of mass from a donor index i to a receiver index j.
    The convolution vector `conv` (length 2*n-1) is updated in O(n) time.
    Returns (new_a, new_conv, new_score) if the move improves the objective,
    otherwise (a, conv, None).  The move amount is at most
    `max_delta_frac * a[i]`.
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # --- convolution update -------------------------------------------------
    new_conv = conv.copy()
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    # --- coefficient update -------------------------------------------------
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)   # protect against rounding drift

    new_score = 2.0 * n * new_conv.max()
    cur_score = 2.0 * n * conv.max()
    if new_score < cur_score - 1e-15:
        return a2, new_conv, new_score
    else:
        return a, conv, None


# ----------------------------------------------------------------------
# Gradient of the (hard) max‑convolution entry
# ----------------------------------------------------------------------


def _gradient_max(a: np.ndarray, conv: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Sub‑gradient of the L∞‑norm of the self‑convolution.
    Returns (grad, k_max) where k_max is the index attaining the maximum.
    grad_j = 2 * a_{k_max-j} for those j with a valid partner, otherwise 0.
    """
    n = a.size
    k = int(np.argmax(conv))                         # position of the peak
    grad = np.zeros_like(a)

    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)                               # inclusive
    idx = np.arange(i_start, i_end + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad, k


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
