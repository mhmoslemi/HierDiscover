def _eg_local_optimise(a0, rng, deadline):
    a = a0.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2*n -1)
    best_vec = a.copy()
    best_score = 2*n * _conv_self(a, fft_len_conv).max()
    t = 0
    max_iters = 2000
    while t < max_iters and time.monotonic() < deadline:
        t += 1

        conv = _conv_self(a, fft_len_conv)
        score = 2*n*conv.max()
        if score < best_score:
            best_score = score; best_vec = a.copy()
        # compute gradient either subgrad or softmax
        if rng.random() < 0.5:
            # subgrad
            grad = _subgradient_step(a, conv)
        else:
            beta = 0.5 + (5000.0 - 0.5)*(t/max_iters)
            grad = _softmax_gradient(a, conv, beta, _fft_len_for_len(3*n - 2))
        # exponentiated update
        eta = 0.5 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum
        # occasional random perturbations (block)
        if rng.random() < 0.1: a = _random_block_perturb(a, rng)
        # occasional pairwise (maybe now integrated)
        if rng.random() < 0.1: a = _pairwise_move(a, rng, fft_len_conv)
        # symmetrize
        a = (a + a[::-1]) / 2
        # ensure sum 1
        a = a / a.sum()
    return best_vec, best_score

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm (high‑level)
----------------------
1. Work on the probability simplex (∑a = 1).  The objective then is
   ``score = 2·n·max(conv(a, a))``.
2. For a given length ``n`` we run a **exponentiated‑gradient** optimizer
   that uses either
   * the exact sub‑gradient of the worst convolution entry, or
   * the smoothed gradient of a log‑sum‑exp (soft‑max) approximation.
   The update is multiplicative (a_i ← a_i·exp(‑η·g_i)) and the vector is
   re‑normalised after each step.
3. After every gradient step we
   * force exact symmetry (a ≔ (a + a[::-1]) / 2),
   * occasionally apply cheap random perturbations (block scaling,
     pairwise mass transfer) and a tiny local smoothing filter.
4. The optimiser is run many times with different lengths (from a few
   hundred up to ~15 000).  Restarts are cheap; the best candidate is
   kept globally.
5. When time is almost exhausted we give the current global best a
   longer “polish” run with more iterations and a higher soft‑max β.

The method stays fully on‑simplex, needs only FFTs for the
convolution, and fits comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities – linear (full) convolution via real FFT
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length`` (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a * a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex projection (used only for the random block perturbation)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of ``max(conv)`` w.r.t. ``a``."""
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(
    a: np.ndarray, conv: np.ndarray, beta: float, fft_len_grad: int
) -> np.ndarray:
    """Gradient of the smooth max  f_β(a) = (1/β)·log Σ exp(β·conv)."""
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    # slice that corresponds to the n components of the gradient
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Random local moves (used with a small probability)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(
    a: np.ndarray, rng: np.random.Generator, fft_len_conv: int
) -> np.ndarray:
    """Transfer a random amount of mass from i to j – accept only if improving."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    # quick check – accept only if the score improves
    conv_old = _conv_self(a, fft_len_conv).max()
    conv_new = _conv_self(a2, fft_len_conv).max()
    if 2.0 * n * conv_new < 2.0 * n * conv_old:
        return a2
    return a


# ----------------------------------------------------------------------
# Helper: up‑sample / down‑sample a vector while staying on the simplex
# ----------------------------------------------------------------------


def _resize_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``v`` to length ``new_n`` and re‑normalisation."""
    old_n = v.size
    if new_n == old_n:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    v_new = np.interp(x_new, x_old, v)
    v_new = np.clip(v_new, 0.0, None)
    if v_new.sum() == 0.0:
        v_new[:] = 1.0 / new_n
    else:
        v_new /= v_new.sum()
    return v_new


# ----------------------------------------------------------------------
# Core optimiser – exponentiated‑gradient on the simplex
# ----------------------------------------------------------------------


def _eg_local_optimise(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex.
    Returns the best vector seen and its score (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    best_vec = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len_conv).max()

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # ----------------------------------------------------------
        # current score
        # ----------------------------------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ----------------------------------------------------------
        # choose gradient (hard sub‑gradient or soft‑max)
        # ----------------------------------------------------------
        if rng.random() < 0.55:
            grad = _subgradient_step(a, conv)
        else:
            beta = 0.5 + (8000.0 - 0.5) * (t / max_iters)   # grow soft‑max temperature
            grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # ----------------------------------------------------------
        # exponentiated‑gradient update
        # ----------------------------------------------------------
        eta = 0.7 / np.sqrt(t)          # decreasing step size
        a = a * np.exp(-eta * grad)     # multiplicative update
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----------------------------------------------------------
        # enforce exact symmetry (helps flatten the autocorrelation)
        # ----------------------------------------------------------
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()

        # ----------------------------------------------------------
        # occasional random local moves
        # ----------------------------------------------------------
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.04:
            a = _pairwise_move(a, rng, fft_len_conv)

        # ----------------------------------------------------------
        # very occasional tiny smoothing (moving average)
        # ----------------------------------------------------------
        if rng.random() < 0.02:
            kernel = np.array([0.25, 0.5, 0.25])
            a = np.convolve(a, kernel, mode='same')
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_vec, best_score


# ----------------------------------------------------------------------
# Score helper (mirrors ``evaluate_sequence`` for speed)
# ----------------------------------------------------------------------


def _score_on_simplex(a: np.ndarray, fft_len_conv: int) -> float:
    """Score = 2·n·max(conv)  (∑a = 1 is assumed)."""
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * a.size * conv.max()

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    ``evaluate_sequence`` value.  The routine respects the time budget,
    runs many exponentiated‑gradient restarts on varying lengths and
    returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, evaluate it first
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cur = _score_on_simplex(cand, _fft_len_for_len(2 * cand.size - 1))
                if cur < best_score:
                    best_score = cur
                    best_vec = cand.copy()
                print(f"Initial incumbent score = {cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore – we go up to ~15000 (still cheap for FFT)
    # ------------------------------------------------------------------
    lengths = [800, 1500, 2500, 3809, 5000, 8000, 12000, 15000]
    probs   = np.array([0.08, 0.10, 0.12, 0.20, 0.18, 0.15, 0.10, 0.07])

    # ------------------------------------------------------------------
    # Main search loop – cheap restarts with occasional polishing
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (biased towards the larger ones)
        n = int(rng.choice(lengths, p=probs / probs.sum()))

        # initial vector: either an up‑sampled version of the current best
        # or a fresh random point on the simplex
        if best_vec is not None and best_vec.size != n:
            a0 = _resize_vector(best_vec, n)
        else:
            a0 = rng.random(n)
            a0 /= a0.sum()

        # number of EG steps – larger n gets a few more iterations
        max_iters = int(1500 * (n / 3000.0))
        max_iters = max(800, min(max_iters, 3000))

        a_opt, score_opt = _eg_local_optimise(a0, rng, deadline, max_iters=max_iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})")

        # ----------------------------------------------------------------
        # If time permits, polish the current global best more thoroughly
        # ----------------------------------------------------------------
        time_left = deadline - time.monotonic()
        if best_vec is not None and time_left > 1.0:
            # a longer polish run with a higher iteration budget
            polish_iters = int(2500 * (best_vec.size / 3000.0))
            polish_iters = max(2000, polish_iters)
            a_polished, score_polished = _eg_local_optimise(
                best_vec.copy(), rng, deadline, max_iters=polish_iters
            )
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polished global best: {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
