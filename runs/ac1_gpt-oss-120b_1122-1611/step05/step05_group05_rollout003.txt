def fft_convolve(x,y):
    size = len(x) + len(y) - 1
    N = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, N)
    fy = np.fft.rfft(y, N)
    conv = np.fft.irfft(fx*fy, N)[:size]
    return conv

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    size = x.size + y.size - 1

"""
Hybrid gradient‑descent / stochastic search for a non‑negative step‑function
that minimises the evaluation metric in ``evaluate_sequence``.

The algorithm works with the probability‑simplex (the vector is kept
non‑negative and normalised to sum = 1, which makes the objective
scale‑invariant).  In each iteration we either

* perform a smooth‑max gradient step (the max of the self‑convolution is
  approximated by a soft‑max; the gradient is obtained by a second FFT‑
  based convolution), or
* apply a random multiplicative perturbation of a contiguous block
  (the hill‑climbing move of the previous baseline).

A diminishing learning‑rate and an annealing schedule for the soft‑max
parameter guide the optimisation, while periodic restarts from the best
solution seen so far help escape local minima.  The best candidate found
before the time budget expires is returned as a list of floats.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Real convolution via the FFT (full mode).  Returns a real 1‑D array
    of length len(x) + len(y) - 1.
    """
    size = x.size + y.size - 1
    # next power of two for efficient FFT
    N = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, N)
    fy = np.fft.rfft(y, N)
    conv = np.fft.irfft(fx * fy, N)[:size]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    # block length between 1 and max(2, n//5)
    width = rng.integers(1, max(2, n // 5))
    start = rng.integers(0, n - width + 1)
    # multiplicative factor drawn from a log‑normal distribution
    factor = np.exp(rng.normal(0.0, 0.25))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the time budget (default
    1000 s).  It starts from the globally supplied ``height_sequence_1``
    if present; otherwise from a uniform distribution.  A hybrid of
    smooth‑max gradient descent and stochastic block perturbations is run
    until the time limit, keeping the best sequence encountered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)  # safety margin

    # ------------------------------------------------------------------
    # 1) Initial vector (normalised to sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = 2000
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / total

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    # exact objective (scale‑invariant, thanks to normalisation)
    conv_best = _fft_convolve(best, best)
    best_score = 2.0 * n * conv_best.max() / (best.sum() ** 2)

    start_time = time.monotonic()
    last_improve = start_time

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # learning‑rate schedule (from 0.2 down to 0.001)
    lr_start, lr_end = 0.20, 0.001
    # soft‑max temperature (beta) schedule (from 5 up to 800)
    beta_start, beta_end = 5.0, 800.0
    # probability of taking a random block perturb instead of a gradient step
    prob_perturb = 0.15
    # restart timeout (seconds without improvement)
    restart_timeout = 30.0
    # tiny acceptance probability for non‑improving moves (simulated annealing)
    accept_min, accept_max = 0.0005, 0.020

    iteration = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / (budget_s - 0.2))

        # --------------------------------------------------------------
        #   decide which move to try
        # --------------------------------------------------------------
        if rng.random() < prob_perturb:
            # --- stochastic block perturbation (large exploratory move) ---
            cand = _random_block_perturb(a, rng)
        else:
            # --- smooth‑max gradient descent step ----------------------
            # compute self‑convolution (full length 2n‑1)
            conv = _fft_convolve(a, a)
            maxc = conv.max()

            # soft‑max weighting of the convolution entries
            beta = beta_start + (beta_end - beta_start) * (frac ** 1.5)
            # keep exponent in a safe range to avoid under/overflow
            exp_arg = beta * (conv - maxc)
            exp_arg = np.clip(exp_arg, -700, 0)   # np.exp(-700) ≈ 5e-305
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smooth max (2 · conv(w, a_rev))
            a_rev = a[::-1]
            grad_full = _fft_convolve(w, a_rev)
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # learning‑rate annealing
            lr = lr_start * (1.0 - frac) + lr_end * frac
            cand = a - lr * grad
            cand = _project_simplex(cand)

        # --------------------------------------------------------------
        #   evaluate candidate (exact objective, no extra function calls)
        # --------------------------------------------------------------
        cand_sum = cand.sum()
        if cand_sum < 0.01:            # safety – evaluate_sequence would reject
            cand_sum = 0.01
        conv_cand = _fft_convolve(cand, cand)
        cur_score = 2.0 * n * conv_cand.max() / (cand_sum ** 2)

        # --------------------------------------------------------------
        #   record improvement
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cand.copy()
            a = cand.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")
            # early exit if we have already beaten the target
            if best_score <= 1.5025:
                break
        else:
            # simulated‑annealing style acceptance of worse moves
            accept_prob = accept_min * (1.0 - frac) + accept_max * frac
            if rng.random() < accept_prob:
                a = cand.copy()

        # --------------------------------------------------------------
        #   restart if we are stuck for a while
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > restart_timeout:
            # jitter the current best to escape a possible plateau
            noise = np.exp(rng.normal(0.0, 0.35, size=n))
            a = best * noise
            a = _project_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] restart – best score {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5) Return the best candidate (clip to allowed range)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
