def _update_conv(conv, a, i, j, delta):
    """Return a new conv after moving delta from i to j using current a."""
    # create copy
    new_conv = conv.copy()
    # shift contributions
    new_conv[j:j+len(a)] += 2*delta*a

    # clip

    return new_conv

"""
Propose a candidate step‑function that improves on the previous 1.514007 bound.

Algorithm
---------
* Initialise the sequence with the best known construction (the global
  ``height_sequence_1`` if present, otherwise a discrete arcsine shape) and
  normalise it to sum = 1 (the objective is scale‑invariant).
* Compute the full self‑convolution once (FFT based) and keep it together with
  the current score  ``2·n·max(conv)``.
* Perform a *greedy hill‑climbing* search:
    – Randomly pick a donor index ``i`` (where ``a[i] > 0``) and a receiver
      index ``j``.
    – Transfer a tiny amount ``δ`` from ``i`` to ``j`` (``δ`` is a fraction of
      ``a[i]`` controlled by a decaying step‑size).
    – Update the convolution analytically (the convolution of the
      modified vector can be obtained from the old one by adding the
      contribution of the shifted copies of ``a`` and the quadratic term
      ``δ²``).  This update is O(n) and avoids a full FFT at every trial.
    – Accept the move only if the new maximal convolution value decreases.
      Occasionally a small random “jitter’’ (multiplicative noise) is applied
      to escape plateaux.
    – Periodically recompute the convolution from scratch to eliminate
      accumulated rounding error.
* Keep track of the best sequence seen (both the cheap approximate score and,
  sparsely, the exact ``evaluate_sequence`` score).
* Stop when the allotted time (≈ 1000 s) is exhausted and return the best
  sequence found, clipped to the range required by ``evaluate_sequence``.

The method explores a vastly larger number of tiny local moves than the
previous exponentiated‑gradient routine, while still being cheap enough to run
for many tens of thousands of iterations within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be defined at the top level)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two length that can hold the full
    linear convolution of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector using FFT.
    Result has length 2·len(a)‑1."""
    n = a.size
    fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * n - 1]
    return conv


def _update_conv(conv: np.ndarray, a: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a new convolution vector that corresponds to moving ``delta``
    mass from index ``i`` to index ``j`` of the vector ``a``.
    ``conv`` must be the current self‑convolution of ``a`` (length 2·n‑1).
    The update follows the exact algebraic formula:
        conv' = conv + 2·δ·(shift_j(a) - shift_i(a))
                + δ²·e_{2j} + δ²·e_{2i} - 2·δ²·e_{i+j}
    where ``shift_s(a)`` is ``a`` shifted by ``s`` positions.
    """
    # make a fresh copy to avoid mutating the original
    new_conv = conv.copy()
    n = a.size

    # add the linear part 2·δ·(shift_j - shift_i)
    new_conv[j:j + n] += 2.0 * delta * a
    new_conv[i:i + n] -= 2.0 * delta * a

    # add the quadratic δ² contributions (three isolated entries)
    new_conv[2 * j] += delta * delta
    new_conv[2 * i] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta

    # enforce non‑negativity (tiny rounding errors can produce tiny negatives)
    np.clip(new_conv, 0.0, None, out=new_conv)
    return new_conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function with a low ``evaluate_sequence``
    score.  The routine runs for at most ``budget_s`` seconds and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # stop a little early to give a safety margin
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 0)  Load an initial candidate (prefer the previously found one)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a = np.array([float(x) for x in globals()["height_sequence_1"]], dtype=np.float64)
            # Normalise to sum = 1 (the objective does not depend on the scale)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / a.size)
            else:
                a /= s
        except Exception:
            a = None
    else:
        a = None

    # if no good global is available, fall back to a discrete arcsine shape
    if a is None:
        n = 4096
        i = np.arange(n, dtype=np.float64)
        raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** 0.5)   # α = 0.5
        raw = np.clip(raw, 0.0, 1000.0)
        a = raw / raw.sum()
    n = a.size

    # ------------------------------------------------------------------
    # 1)  Initial convolution and score
    # ------------------------------------------------------------------
    conv = _conv_self(a)
    cur_score = 2.0 * n * conv.max()          # exact objective for sum‑= 1
    best_a = a.copy()
    best_score = cur_score

    # Verify the exact score once (cheap compared to a long search)
    try:
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact) and exact < best_score:
            best_score = exact
            print(f"[{0.0:6.2f}s] exact baseline score = {exact:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2)  Greedy hill‑climbing with analytical convolution updates
    # ------------------------------------------------------------------
    # Parameters (they decay slowly so the search becomes finer)
    step_factor = 0.35          # initial fraction of donor mass that may be moved
    step_decay = 0.99996        # geometric decay per iteration
    # A tiny “temperature’’ to allow occasional uphill moves when the
    # landscape is flat – it decays very fast.
    temperature = cur_score * 0.001
    temp_decay = 0.99990

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 2a)  Propose a move: transfer a small δ from i → j
        # --------------------------------------------------------------
        i_idx = rng.integers(0, n)
        if a[i_idx] <= 1e-16:               # nothing to give, pick another
            continue
        j_idx = rng.integers(0, n)
        if i_idx == j_idx:
            continue

        # δ is at most a fraction of the donor mass; we also randomise it
        max_delta = a[i_idx]
        delta = step_factor * max_delta * rng.random()
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # 2b)  Compute the new convolution analytically
        # --------------------------------------------------------------
        conv_new = _update_conv(conv, a, i_idx, j_idx, delta)

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # 2c)  Acceptance test (mostly greedy)
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # occasional uphill acceptance (simulated annealing)
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            a[i_idx] -= delta
            a[j_idx] += delta
            conv = conv_new
            cur_score = new_score

            # keep the best ever seen (approximate score)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best approx score = {best_score:.6f}")

                # sporadic exact evaluation to guard against drift
                if rng.random() < 0.02:
                    try:
                        exact = evaluate_sequence(best_a.tolist())
                        if np.isfinite(exact) and exact < best_score:
                            best_score = exact
                            print(f"[{elapsed:6.2f}s] new best exact score = {exact:.6f}")
                    except Exception:
                        pass

        # --------------------------------------------------------------
        # 2d)  Decay hyper‑parameters
        # --------------------------------------------------------------
        step_factor *= step_decay
        temperature *= temp_decay

        # --------------------------------------------------------------
        # 2e)  Occasional jitter to escape flat plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.001:
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            a *= jitter
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()
            # accept jitter as a fresh start (no need for acceptance test)

        # --------------------------------------------------------------
        # 2f)  Periodic full recomputation to wash out numeric error
        # --------------------------------------------------------------
        if iteration % 5000 == 0:
            conv = _conv_self(a)
            cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # 2g)  Light progress reporting
        # --------------------------------------------------------------
        if iteration % 20000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] iter {iteration:,} – best approx = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3)  Return the best sequence discovered (clipped to the required range)
    # ------------------------------------------------------------------
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_a.tolist()]
    final_score = evaluate_sequence(best_seq)
    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] final exact score = {final_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
