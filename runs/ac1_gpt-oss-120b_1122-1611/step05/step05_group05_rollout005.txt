def gradient_softmax(a, alpha):
    b = np.convolve(a, a)  # length 2n-1
    # compute softmax weights over b scaled by alpha.
    # To avoid overflow, subtract max.
    maxb = np.max(b)
    exp_terms = np.exp(alpha * (b - maxb))
    w = exp_terms / np.sum(exp_terms)  # shape 2n-1
    w_rev = w[::-1]
    conv = np.convolve(a, w_rev)  # length 3n-2
    grad = 2.0 * conv[len(w)-1 - np.arange(len(a))]  # indexing
    return grad

"""Gradient‑based search for a low‑max‑convolution step‑function.

The objective (as used in ``evaluate_sequence``) is
    f(a) = 2·n·max(conv(a,a)) / (sum(a)²) .
Because f is homogeneous we may enforce sum(a)=1 and minimise
``max(conv(a,a))``.
A smooth approximation of the max is built with a soft‑max
(log‑sum‑exp) on the convolution values.  The gradient of this
approximation is computed analytically and an exponentiated‑gradient
update on the probability simplex is performed.  The temperature of the
soft‑max (α) is increased from a modest value to one proportional to the
sequence length, so the optimisation gradually focuses on the true
peak.  The step‑size is annealed, and if no improvement has been seen
for a while the current best solution is jittered and the search
restarts.  The best sequence found before the time budget expires is
returned.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, sum(x)=1}.  Implements the O(n) algorithm from
    Duchi et al. (2008)."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max weights of the vector ``b`` with temperature ``α``.
    The computation is numerically stable (max‑subtraction).
    """
    maxb = b.max()
    # exp(α·(b−maxb)) is always ≤ 1, preventing overflow.
    exp_term = np.exp(alpha * (b - maxb))
    w = exp_term / exp_term.sum()
    return w


def _gradient_softmax(a: np.ndarray, alpha: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the gradient of the smoothed max (log‑sum‑exp) of
    ``conv(a, a)`` with respect to ``a``.
    Returns (grad, b) where ``b`` = conv(a, a) (full length 2·n−1).
    """
    # convolution a with itself
    b = np.convolve(a, a)                     # length 2·n‑1
    w = _softmax_weights(b, alpha)            # same length as b
    # gradient = 2 * Σ_j a_j w_{i+j}   (i = 0..n‑1)
    w_rev = w[::-1]
    conv = np.convolve(a, w_rev)              # length 3·n‑2
    wlen = w.shape[0]
    n = a.shape[0]
    grad = 2.0 * conv[wlen - 1 - np.arange(n)]
    return grad, b

    distribution, then performs a projected exponentiated‑gradient
    descent with a soft‑max approximation of the max‑convolution.
    Periodic jitter‑restarts avoid being trapped in poor local minima.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    best = a.copy()
    b_best = np.convolve(best, best)
    best_score = 2.0 * n * b_best.max() / (best.sum() ** 2)

    start_time = time.monotonic()
    last_improve = start_time

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Learning‑rate (exponentiated‑gradient) anneals from max_lr → min_lr
    max_lr = 0.6
    min_lr = 0.02

    # Soft‑max “temperature’’ α = β·n, β anneals from β_start → β_end
    beta_start = 0.5
    beta_end = 4.0

    # Restart if we see no improvement for this many seconds
    restart_gap = 30.0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / (budget_s - 0.5))

        # annealed step‑size and soft‑max scale
        lr = max_lr * (1.0 - frac) + min_lr * frac
        beta = beta_start + (beta_end - beta_start) * frac
        alpha = beta * n                        # effective temperature

        # gradient of the smoothed max
        grad, _ = _gradient_softmax(a, alpha)

        # exponentiated‑gradient update on the simplex
        a_new = a * np.exp(-lr * grad)
        a_new = _project_simplex(a_new)

        # evaluate the true objective (no need to call evaluate_sequence)
        b_new = np.convolve(a_new, a_new)
        cur_score = 2.0 * n * b_new.max() / (a_new.sum() ** 2)

        # --------------------------------------------------------------
        # bookkeeping / acceptance
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            last_improve = time.monotonic()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # keep the new point as the current iterate – it is always feasible
        a = a_new

        # --------------------------------------------------------------
        # restart when stuck
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > restart_gap:
            # jitter the current best and re‑project
            sigma = 0.08
            jitter = np.exp(rng.normal(0.0, sigma, size=n))
            a = best * jitter
            a = _project_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] restart – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best candidate found
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
