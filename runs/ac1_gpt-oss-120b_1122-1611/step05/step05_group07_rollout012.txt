import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    # block width at most 10 % of the vector, at least 1
    max_width = max(2, n // 10)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out

import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns the first (2·len(a)‑1) entries (the linear convolution).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    max_width = max(2, n // 10)                 # ≤10 % of the vector, at least 2
    width = rng.integers(1, max_width + 1)      # block length
    start = rng.integers(0, n - width + 1)      # block start index
    factor = np.exp(rng.normal(0.0, 0.15))      # log‑normal scaling
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Simulated‑annealing search for a non‑negative step‑function that
    minimises ``evaluate_sequence``.  The algorithm works directly on the
    convolution vector: a mass‑transfer move changes only three entries of
    the convolution, allowing many cheap proposals.  Occasionally a
    block‑wise multiplicative perturbation is performed (full FFT

    vector is perturbed and the search restarts.

    if "height_sequence_1" in globals():
