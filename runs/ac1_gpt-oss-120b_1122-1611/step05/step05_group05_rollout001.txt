   rng = np.random.default_rng(seed)
   deadline = time.monotonic() + max(0.0, budget_s - 0.1) # small safety margin

   # Determine n: we can try to match size of previous sequence if available; otherwise use default like 4000.
   if 'height_sequence_1' in globals():
      base = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
      n = base.size
   else:
      n = 4000

   # Initialize population
   popsize = 30
   # Dirichlet parameter: possibly 0.5 for sparsity
   alpha = 0.5
   pop = rng.dirichlet([alpha]*n, size=popsize)  # shape (popsize, n) ensures sum=1.
   # compute scores:
   scores = np.empty(popsize)
   for i in range(popsize):
       scores[i] = _compute_score(pop[i], n)

   # Keep best score and best vector.
   best_idx = np.argmin(scores)
   best = pop[best_idx].copy()
   best_score = scores[best_idx]

   # DE parameters
   F = 0.7
   CR = 0.9

   gen = 0
   last_improve_time = time.monotonic()

   while time.monotonic() < deadline:
       # For each individual
       for i in range(popsize):
           # select three distinct individuals different from i
           idxs = rng.choice([j for j in range(popsize) if j != i], size=3, replace=False)
           x1,x2,x3 = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
           mutant = x1 + F * (x2 - x3)
           # Clip negative values
           mutant = np.clip(mutant, 0, None)
           # Normalize
           s = mutant.sum()
           if s == 0:
               mutant = np.full(n, 1.0/n)
           else:
               mutant = mutant / s
           # Crossover: binomial
           cross = rng.random(n) < CR
           if not np.any(cross):
               cross[rng.integers(0,n)] = True
           trial = np.where(cross, mutant, pop[i])
           # Ensure nonneg and normalize again (just in case)
           trial = np.clip(trial, 0, None)
           s = trial.sum()
           if s == 0:
               trial = np.full(n, 1.0/n)
           else:
               trial = trial / s
           # Evaluate trial
           trial_score = _compute_score(trial, n)
           if trial_score < scores[i]:
               pop[i] = trial
               scores[i] = trial_score
               if trial_score < best_score:
                   best_score = trial_score
                   best = trial.copy()
                   last_improve_time = time.monotonic()
                   elapsed = last_improve_time - start_time
                   print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")
           # maybe adapt F and CR schedule? We can gradually reduce F to refine.

def _compute_score(a: np.ndarray, n: int) -> float:
    # a is non-negative and sum>0.
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)
    # avoid near-zero S from numeric; but S is 1.
    return 2.0 * n * M / (S**2)

"""
Propose a good step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The algorithm combines two complementary phases:

1. **Global search** – a small Differential Evolution (DE) population
   evolves on the simplex (sum = 1, entries ≥ 0).  Individuals are created
   by mutation, clipping to non‑negative values and renormalising.
   The DE parameters (scale factor F and crossover probability CR) are
   annealed linearly with the remaining time.  If no improvement occurs
   for a while, half the population is re‑initialised.

2. **Local refinement** – a simulated‑annealing hill‑climber that
   repeatedly perturbs a random contiguous block by a log‑normal factor,
   projects back onto the simplex and accepts worsening moves with a
   temperature schedule.  This phase runs after the DE budget is spent
   (or when the deadline is near).

Both phases use a cheap hand‑written evaluation
`_score(a) = 2·n·max(conv(a,a)) / (sum(a)**2)` which is identical to
`evaluate_sequence` after the input is clamped.  The best vector found
before the time budget expires is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------

def _score(a: np.ndarray) -> float:
    """Return the evaluation metric for a non‑negative vector a (sum > 0)."""
    # a is assumed to be a 1‑D numpy array of floats, non‑negative.
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    s = np.sum(a)
    # s is kept > 0 by the optimisation code
    n = a.size
    return 2.0 * n * max_conv / (s * s)


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.  Used only as a fallback when clipping + renorm
    could produce a zero‑vector.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _normalise_positive(v: np.ndarray) -> np.ndarray:
    """Clip to ≥0, then renormalise to sum = 1 (fallback to simplex)."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s > 0.0:
        return v / s
    # safety net – create a uniform simplex point
    return np.full_like(v, 1.0 / v.size)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Mutate the contiguous block of `a` by a log‑normal factor,
    then project back onto the simplex.
    """
    n = a.size
    # block length: at least 1, at most max(2, n//5)
    width = rng.integers(1, max(2, n // 5) + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))   # typical perturbation size
    out = a.copy()
    out[start:start + width] *= factor
    return _normalise_positive(out)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget (default 1000 s).

    The function uses a two‑phase strategy:
      * a Differential Evolution (DE) global optimiser on the probability simplex,
      * followed by a hill‑climbing simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Give a small safety margin so that we can always finish cleanup work
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Determine problem size (length of the vector)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            base_seq = None
        else:
            base_seq = np.clip(base_seq, 0.0, None)
            s = base_seq.sum()
            if s > 0.0:
                base_seq = base_seq / s
    else:
        base_seq = None

    # Choose a length.  If we have a reference sequence we keep its length,
    # otherwise we take a power‑of‑two convenient for FFT‑based convolutions.
    if base_seq is not None:
        n = base_seq.size
    else:
        n = 4096                    # default length (can be tuned)

    # --------------------------------------------------------------
    # Initialise Differential Evolution population
    # --------------------------------------------------------------
    popsize = 30                    # reasonable trade‑off between diversity & speed
    # First individual is the known best (if any); the rest are random.
    population = np.empty((popsize, n), dtype=np.float64)
    if base_seq is not None:
        population[0] = base_seq
    else:
        population[0] = np.full(n, 1.0 / n)

    # Remaining individuals: Dirichlet draws (α < 1 => sparser candidates)
    alpha = 0.5
    if popsize > 1:
        pop_rest = rng.dirichlet([alpha] * n, size=popsize - 1)
        population[1:] = pop_rest

    # Evaluate the initial population
    scores = np.empty(popsize, dtype=np.float64)
    for i in range(popsize):
        scores[i] = _score(population[i])

    best_idx = int(np.argmin(scores))
    best = population[best_idx].copy()
    best_score = float(scores[best_idx])
    last_improve = start_time

    print(f"[0.00s] initial best score = {best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # DE hyper‑parameters (annealed)
    # --------------------------------------------------------------
    F_start, F_end = 0.7, 0.3          # scale factor (mutation)
    CR_start, CR_end = 0.9, 0.5       # crossover probability
    restart_gap = 30.0                # seconds without improvement → partial restart

    # --------------------------------------------------------------
    # Phase 1 – Differential Evolution (global search)
    # --------------------------------------------------------------
    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # Annealing factor based on elapsed time
        elapsed = now - start_time
        frac = min(1.0, elapsed / (budget_s - 0.5))
        F = F_start * (1.0 - frac) + F_end * frac
        CR = CR_start * (1.0 - frac) + CR_end * frac

        # One DE generation (popsize trial vectors)
        for i in range(popsize):
            # Choose three distinct donors different from i
            # Simple fast method: sample without replacement from the whole set,
            # then discard i if it appears.
            idxs = rng.choice(popsize - 1, size=3, replace=False)
            # Ensure we never pick i itself
            idxs = [(idx if idx < i else idx + 1) for idx in idxs]
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]

            # Mutation
            mutant = x1 + F * (x2 - x3)
            mutant = _normalise_positive(mutant)

            # Crossover (binomial)
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = _normalise_positive(trial)

            # Evaluate trial vector
            trial_score = _score(trial)

            # Selection
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
                    last_improve = now
                    elapsed_disp = now - start_time
                    print(f"[{elapsed_disp:6.2f}s] new best score = {best_score:.6f}")

        # Restart if we have been stuck for a while
        if now - last_improve > restart_gap:
            # Replace half of the population with fresh random Dirichlet points
            replace_n = popsize // 2
            new_inds = rng.dirichlet([alpha] * n, size=replace_n)
            population[-replace_n:] = new_inds
            for j in range(popsize - replace_n, popsize):
                scores[j] = _score(population[j])
            last_improve = now
            elapsed_disp = now - start_time
            print(f"[{elapsed_disp:6.2f}s] partial restart (kept best)")

        # Check if remaining time is small enough to switch to local search
        remaining = deadline - now
        if remaining < 200.0:           # reserve ~200 s for hill‑climbing
            break

    # --------------------------------------------------------------
    # Phase 2 – Simulated‑annealing hill‑climbing (local refinement)
    # --------------------------------------------------------------
    # Parameters for the hill‑climber
    accept_start = 0.12
    accept_end = 0.001
    climb_start = time.monotonic()

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # Propose a neighbour by perturbing a random block of the current best
        cand = _random_block_perturb(best, rng)
        cand_score = _score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            last_improve = now
            elapsed_disp = now - start_time
            print(f"[{elapsed_disp:6.2f}s] improved (local) score = {best_score:.6f}")
        else:
            # Simulated‑annealing acceptance
            elapsed = now - start_time
            frac = min(1.0, elapsed / (budget_s - 0.5))
            accept_prob = accept_start * (1.0 - frac) + accept_end * frac
            if rng.random() < accept_prob:
                best = cand.copy()   # move to a worse point with small prob

        # Optional very cheap restart if stuck for a while (helps escape)
        if now - last_improve > 60.0:
            # jitter the current best a little and continue
            jitter = np.exp(rng.normal(0.0, 0.10, size=n))
            best = _normalise_positive(best * jitter)
            last_improve = now
            elapsed_disp = now - start_time
            print(f"[{elapsed_disp:6.2f}s] local restart (jitter)")

    # --------------------------------------------------------------
    # Return the best discovered sequence (clamped to allowed range)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
