def proj_simplex(v):
    # Ensure nonnegatives sum to 1
    # Implementation: sort, find rho, theta
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
* Keep the vector on the probability simplex (∑a_i = 1).  For such a
  normalisation the objective is   score = 2·n·max_k (a∗a)_k.
* Optimise a *smooth* surrogate of the max using a soft‑max:
         f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k) .
  The gradient is 2·a correlated with the soft‑max weights.
* Use Adam (adaptive moment estimation) on that smooth objective and
  project after each step onto the simplex (non‑negative + unit sum).
* β is increased from a small value (≈0.5) to a large value (≈2500) so the
  optimisation first sees a smooth landscape and gradually concentrates
  on the true L∞‑objective.
* Occasionally inject multiplicative log‑normal noise and perform a
  random mass‑transfer to escape shallow local minima.
* If no improvement is seen for ``restart_gap`` seconds, restart from the
  best solution plus a small jitter.
* The routine respects the 1000 s budget and returns the best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length for a linear convolution of size n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for use with FFT."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return linear convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate.
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w  (correlation).
    ``a`` is length n, ``w`` length 2·n‑1.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # Length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reversed a for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, unit sum).
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # Find the largest rho satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical cleanup – force exact sum 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Length of the step sequence.  The default (3800) matches the
        previously successful construction.

    Returns
    -------
    list[float]
        The best sequence found (values clipped to the limits imposed by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    # --------------------------------------------------------------
    # 2) Initial vector – prefer a previously discovered construction.
    # --------------------------------------------------------------
    # Analytic “sqrt‑difference’’ start: a_i = √(i+1)‑√i (unit‑sum).
    i = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(i + 1.0) - np.sqrt(i)
    a_sqrt /= a_sqrt.sum()

    a = a_sqrt.copy()
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand /= s
                    # quick score estimate (same normalisation as a)
                    fft_len_cand = _fft_len_for_n(cand.size)
                    score_cand = 2.0 * cand.size * np.max(_conv_self(cand, fft_len_cand))
                    score_sqrt = 2.0 * n * np.max(_conv_self(a_sqrt, _fft_len_for_n(n)))
                    if score_cand < score_sqrt:
                        a = cand
                        n = cand.size          # adopt its length
        except Exception:
            pass

    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 3) Optimisation state
    # --------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * np.max(_conv_self(best, fft_len))

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin
    last_improve = start_time

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # Schedules
    lr_start = 0.25
    lr_end = 0.001
    beta_start = 0.5          # soft‑max temperature start
    beta_end = 2500.0         # temperature at the end (sharp)

    # Random‑perturbation parameters
    noise_prob = 0.032
    noise_sigma = 0.14
    swap_prob = 0.07
    restart_gap = 130.0       # seconds without improvement before a restart

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        ratio = min(1.0, elapsed / total)

        # ---- schedule for learning rate and soft‑max temperature ----
        lr = lr_start * (1.0 - ratio) + lr_end * ratio
        beta = beta_start + (beta_end - beta_start) * ratio

        # ---- compute self‑convolution and smooth‑max weights ----
        conv = _conv_self(a, fft_len)                     # length 2·n‑1
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- keep the best solution found so far ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- soft‑max weights (stable computation) ----
        diff = conv - cur_max                               # ≤ 0
        # exponent may underflow – that is fine; we guard against total zero
        w_raw = np.exp(beta * diff)
        w_sum = w_raw.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback – uniform weight on the maximal indices
            mask = conv == cur_max
            w_raw = np.where(mask, 1.0, 0.0)
            w_sum = w_raw.sum()
        w = w_raw / w_sum

        # ---- gradient of the smooth surrogate ----
        grad = _gradient_correlation(a, w)                 # shape (n,)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # ---- projection onto the simplex ----
        a = _proj_simplex(a)

        # ---- occasional random block perturbation (log‑normal) ----
        if rng.random() < noise_prob:
            factor = np.exp(rng.normal(0.0, noise_sigma, size=n))
            a = a * factor
            a = _proj_simplex(a)

        # ---- occasional pairwise mass transfer ----
        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-15:
                delta = rng.random() * a[i] * 0.5
                a[i] -= delta
                a[j] += delta
                a = _proj_simplex(a)

        # ---- soft restart if stuck for a while ----
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = _proj_simplex(a)
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (apply the clipping imposed by
    #    evaluate_sequence).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
