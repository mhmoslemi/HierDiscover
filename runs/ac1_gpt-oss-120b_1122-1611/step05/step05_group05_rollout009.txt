def projected_gradient_step(a, step):
    # compute conv
    conv = np.convolve(a, a)
    max_b = conv.max()
    # indices where max occurs
    idxs = np.where(conv == max_b)[0]
    g = np.zeros_like(a)
    n = a.size
    for i in idxs:
        start = max(0, i - (n - 1))
        end = min(i, n - 1)
        # j range start..end
        # compute contributions: for each k in start..end: add 2*a[i - k]
        # vectorized:
        k = np.arange(start, end+1)
        g[k] += 2 * a[i - k]
    # gradient of max_b = g
    # reduce max_b: move opposite direction
    a_new = a - step * g
    # ensure positivity
    a_new = np.clip(a_new, 0, None)
    # project onto simplex to keep sum = 1
    a_new = _project_simplex(a_new)
    return a_new
```

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a 1‑D array onto {x ≥ 0, sum(x) = 1}."""
    if v.ndim != 1:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Random block perturbation (used by the original baseline)
# ----------------------------------------------------------------------
def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = a.size
    width = rng.integers(1, max(2, n // 5))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))   # log‑normal multiplicative factor
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out

# ----------------------------------------------------------------------
# Smoothing move (moving average)
# ----------------------------------------------------------------------
def _smooth_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = a.size
    width = rng.integers(3, max(4, n // 10))
    kernel = np.full(width, 1.0 / width, dtype=np.float64)
    smoothed = np.convolve(a, kernel, mode="same")
    smoothed = np.clip(smoothed, 0.0, None)
    return _project_simplex(smoothed)

# ----------------------------------------------------------------------
# Score routine (exactly what `evaluate_sequence` computes, but cheaper)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.size
    conv = np.convolve(a, a)
    return float(2.0 * n * np.max(conv) / (np.sum(a) ** 2))

# ----------------------------------------------------------------------
# Sub‑gradient of the maximum convolution entry (vector of length n)
# ----------------------------------------------------------------------
def _max_conv_subgradient(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    For each index i where the convolution attains its maximum,
    add the gradient of b_i = (a * a)_i to the result.
    """
    n = a.size
    g = np.zeros_like(a)
    for i in idxs:
        start = max(0, i - (n - 1))
        end = min(i, n - 1)
        ks = np.arange(start, end + 1)
        g[ks] += 2.0 * a[i - ks]
    return g

# ----------------------------------------------------------------------
# Main proposal routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    # reserve a small safety margin so we never over‑run the budget
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # 1) Determine length n
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if tmp.ndim == 1 and tmp.size > 0 and np.isfinite(tmp).all():
                n = tmp.size
            else:
                raise ValueError
        except Exception:
            n = 4096          # fallback
    else:
        n = 4096

    # ------------------------------------------------------------------
    # 2) Helper to keep track of the best solution found so far
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    best_score = _score(best_seq)
    start_time = time.monotonic()
    print(f"[0.00s] initial uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Quick exploration of simple parametric families
    # ------------------------------------------------------------------
    # Power‑law family a_i ∝ ((i+1)/n)^γ
    for gamma in np.linspace(-0.8, 3.0, 41):
        x = (np.arange(1, n + 1) / n) ** gamma
        if x.sum() == 0:
            continue
        a = x / x.sum()
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] power γ={gamma:.3f} → {sc:.6f}")

    # Exponential family a_i ∝ exp(β·i/(n‑1))
    for beta in np.linspace(-4.0, 4.0, 41):
        x = np.exp(beta * np.arange(n) / (n - 1))
        a = x / x.sum()
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] exp β={beta:.3f} → {sc:.6f}")

    # Random Dirichlet draws (concentration α ∈ [0.1, 3])
    for _ in range(200):
        alpha = rng.uniform(0.1, 3.0)
        a = rng.dirichlet(np.full(n, alpha))
        sc = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] dirichlet α={alpha:.3f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Local optimisation – hybrid of gradient, random block,
    #    and smoothing moves
    # ------------------------------------------------------------------
    a_cur = best_seq.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    step = 0.20                     # step size for the multiplicative gradient step

    while time.monotonic() < deadline:
        # decide which move to apply
        move_type = rng.choice(["gradient", "block", "smooth"], p=[0.5, 0.35, 0.15])

        if move_type == "gradient":
            # compute convolution once
            conv = np.convolve(a_cur, a_cur)
            max_b = conv.max()
            idxs = np.where(conv == max_b)[0]   # indices attaining the max
            g = _max_conv_subgradient(a_cur, idxs)
            # multiplicative update in the direction that reduces max_b
            cand = a_cur * np.exp(-step * g)
            cand = _project_simplex(cand)
        elif move_type == "block":
            cand = _random_block_perturb(a_cur, rng)
        else:   # smoothing
            cand = _smooth_move(a_cur, rng)

        cand_score = _score(cand)

        # deterministic acceptance of improvements
        if cand_score < cur_score:
            a_cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] new best = {best_score:.6f} (move={move_type})")
            # slowly increase step after a successful gradient move
            if move_type == "gradient":
                step = min(2.0, step * 1.07)
        else:
            # simulated‑annealing acceptance for occasional uphill moves
            now = time.monotonic()
            elapsed = now - start_time
            frac = min(1.0, elapsed / (budget_s - 0.2))
            accept_prob = 0.12 * (1.0 - frac) + 0.001 * frac
            if rng.random() < accept_prob:
                a_cur, cur_score = cand, cand_score
            # on a failed gradient move shrink the step
            if move_type == "gradient":
                step = max(0.02, step * 0.70)

        # ------------------------------------------------------------------
        # Restart if stuck for a while
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > 30.0:
            # jitter the best solution and continue from there
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            a_cur = best_seq * jitter
            a_cur = _project_simplex(a_cur)
            cur_score = _score(a_cur)
            step = 0.20
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart – jittered, step reset")

    # ------------------------------------------------------------------
    # 5) Return the best sequence found (clip to allowed range)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:                              # safety – should never happen
        best_seq = np.full_like(best_seq, 1.0 / n)
    else:
        best_seq = best_seq / total               # renormalise just in case
    final_score = _score(best_seq)
    print(f"Finished. best score = {final_score:.6f} (length = {n})")
    return [float(x) for x in best_seq]
