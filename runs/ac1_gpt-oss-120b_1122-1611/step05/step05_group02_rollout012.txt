import cvxpy as cp
def solve_lp(f, M):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]   # f_{k-j}
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return np.zeros(n)
    else:
        return np.maximum(g.value, 0.0)

```python
        # Returns matrix A of shape (2n-1, n) where A[k,j] = f[k-j] if 0 <= k-j < n else 0
        N = len(f)
        rows = 2 * N - 1
        A = np.zeros((rows, N), dtype=np.float64)
        for k in range(rows):
            # j runs over max(0, k-(N-1)) .. min(N-1, k)
            j_start = max(0, k - (N - 1))
            j_end = min(N - 1, k)
            # for each j in that range, f_idx = k - j
            A[k, j_start:j_end+1] = f[k - np.arange(j_start, j_end+1)]
        return A

    # main loop
    while time.monotonic() < deadline:
        # compute current max conv
        conv = np.convolve(a, a)
        M_current = float(conv.max())

        # Build LP matrix
        A = build_conv_matrix(a)

        # Solve LP: maximize sum(g) subject to A @ g <= M_current, g >= 0
        g_var = cp.Variable(n, nonneg=True)
        constraints = [A @ g_var <= M_current]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9)
        except Exception as e:
            # fallback: break
            print("LP solver failed:", e)
            break
        if g_var.value is None:
            # infeasible or solver issue
            break
        g0 = np.maximum(g_var.value, 0.0)

        # Rescale g to have target sum
        sum_g0 = g0.sum()
        if sum_g0 > 0:
            g = g0 * (target_sum / sum_g0)
        else:
            g = np.full(n, target_sum / n)

        # blend
        a = (1 - t) * a + t * g

        # occasional random perturbations to escape
        if rng.random() < 0.03:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())

        # evaluate
        score = evaluate_sequence(list(a))
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] new best score {best_score:.6f}")

        # optionally break if improvement small
        # continue loop

    # after main LP loop, optionally refine with gradient descent method (similar to earlier code)
    # Use few steps of exponentiated gradient from best

    # gradient descent phase
    a = best.copy()
    n = a.size
    fft_len = _fft_len_for_n(n) # we can reuse earlier defined function; define at top

    # Add gradient descent loop similar to previous code but limited time
    grad_prev = np.zeros_like(a)
    eta_start = 1.0
    eta_end = 0.01
    momentum = 0.9

    while time.monotonic() < deadline:
        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))
        # compute gradient w.r.t max entry
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1)
            j = k_max - i_range
            grad[i_range] = 2.0 * a[j]

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        a = a / a.sum() * target_sum

        # occasional perturb to avoid stagnation
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng) * (target_sum / _random_block_perturb(a, rng).sum()) # Actually we must recompute; but we can just call function then renorm

        # evaluate
        score = evaluate_sequence(list(a))
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] refined best {best_score:.6f}")

        # break if no improvement for some time
        # ( no break here, loop continues until deadline )
        # we may also add small random restart

        if time.monotonic() >= deadline:
            break

    # clip and return best as list
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
```

The objective to minimise is  

\[
J(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `(a*a)` is the ordinary (linear) self‑convolution of the non‑negative vector `a`
and `n=len(a)`.  The quantity is *scale‑invariant* – only the shape of `a`
matters – therefore we normalise every candidate to the standard sum  

\[
\sum_i a_i = \sqrt{2n},
\qquad\text{so that }J(a)=\max_k (a*a)_k .
\]

A simple uniform vector gives `J=2`.  The current state‑of‑the‑art (exponentiated‑gradient
descent) reaches `J≈1.514`.  To improve further we combine two ideas:

1. **Linear‑programming “balancing’’ step** –  
   With the present vector `f` and its worst convolution value `M = max_k (f*f)_k`,
   we maximise `∑ g_i` under the linear constraints  

   \[
   (f*g)_k \le M\quad\forall k ,
   \qquad g_i\ge0 .
   \]

   Because `f` is fixed, the constraint is *linear* in the unknown `g`.  
   The obtained `g` puts mass where `f` interferes the least.  After rescaling `g`
   to the required sum we mix `f←(1-t)f + t g`.  This step never increases `M`
   and often reduces it.

2. **Fine‑grained exponentiated‑gradient descent** –  
   After a few LP‑updates we switch to the fast mirror‑descent routine
   (the one used previously) to polish the solution.

Both phases are run inside the allowed 1000 s budget; the best vector found
throughout the run is returned.

The implementation uses only `numpy` and `cvxpy` (ECOS solver).  All helper
functions are top‑level, there are no closures, and the code respects the
security checks performed by `evaluate_sequence`.  

```python
"""
Hybrid optimiser for the step‑function problem.

Phase 1 – a linear‑programming “balancing’’ step.
    Given the current vector f (normalised to sum sqrt(2 n)),
    compute M = max_k (f*f)_k and solve
        maximise   ∑ g_i
        subject to (f*g)_k ≤ M   for every convolution index k
                    g_i ≥ 0 .
    The solution g is rescaled to the same sum and mixed into f:
        f ← (1‑t)·f + t·g   (t≈0.15).

Phase 2 – exponentiated‑gradient (mirror‑descent) polishing.
    The same update as in the reference code, but started from the
    vector obtained in phase 1.

Both phases run until the time budget expires; the best vector encountered
(in terms of evaluate_sequence) is returned.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
#  Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block,
    followed by renormalisation to the required sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    # renormalise to the same L1‑norm (the caller supplies the target sum)
    return out


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A such that (A @ g) = f * g
    (ordinary linear convolution).  Entry A[k, j] = f_{k‑j} if 0 ≤ k‑j < n,
    otherwise 0.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # Fill the Toeplitz‑like matrix column‑wise – this is cheap for n≈256.
    for j in range(n):
        # column j corresponds to f shifted by +j positions
        start = j
        end = j + n
        A[start:end, j] = f[:]
    return A


# ----------------------------------------------------------------------
#  Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Choose the working dimension (larger n → more expressive,
    #    but LP becomes slower).  256 is a good compromise.
    # --------------------------------------------------------------
    n = 256

    # --------------------------------------------------------------
    # 2) Initialise the vector.
    #    If a previous construction is available, use it (cropped or padded);
    #    otherwise start from the uniform distribution normalised to
    #    sqrt(2 n) (the scaling used by the original evaluation).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                # adapt size to n
                if a.size > n:
                    a = a[:n]
                elif a.size < n:
                    a = np.concatenate([a, np.full(n - a.size, 0.0)])
                s = a.sum()
                a = a * (target_sum / s) if s > 0 else np.full(n, target_sum / n)
            else:
                raise ValueError
        except Exception:
            a = np.full(n, target_sum / n, dtype=np.float64)
    else:
        a = np.full(n, target_sum / n, dtype=np.float64)  # uniform, correctly scaled

    # --------------------------------------------------------------
    # 3) Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"[0.00s] initial score = {best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # --------------------------------------------------------------
    # 4) Phase 1 – LP balancing (run for at most ~30 % of the budget)
    # --------------------------------------------------------------
    phase1_time = (deadline - start_time) * 0.30
    mix_factor = 0.15        # convex‑combination weight t
    while time.monotonic() - start_time < phase1_time:
        # current worst self‑convolution value M
        conv_self = np.convolve(a, a)
        M_current = float(conv_self.max())

        # Build the linear operator (f * g) as a matrix.
        A = _build_conv_matrix(a)

        # Linear program: maximise sum(g) s.t. A@g ≤ M_current , g ≥ 0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M_current]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=5000, abstol=1e-9, reltol=1e-9)
        except Exception as exc:
            print("LP solve failed:", exc)
            break

        if g.value is None:
            # infeasible – should not happen, abort LP phase
            break

        g0 = np.maximum(g.value, 0.0)

        # Rescale to the required L1‑norm.
        s_g = g0.sum()
        if s_g > 0:
            g_rescaled = g0 * (target_sum / s_g)
        else:
            g_rescaled = np.full(n, target_sum / n)

        # Convex combination
        a = (1.0 - mix_factor) * a + mix_factor * g_rescaled

        # Occasional random jitter to avoid stagnation
        if rng.random() < 0.02:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())

        # Evaluate – keep the best ever seen
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] LP phase new best = {best_score:.6f}")

        # Simple restart if we are stuck for a while
        if time.monotonic() - last_improve > 25.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] LP restart")

    # --------------------------------------------------------------
    # 5) Phase 2 – exponentiated‑gradient polishing (remaining time)
    # --------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    # hyper‑parameters for the mirror‑descent
    eta_start = 1.0
    eta_end = 0.01
    momentum = 0.9
    grad_prev = np.zeros_like(a)
    block_perturb_prob = 0.02
    pair_move_prob = 0.04
    restart_gap = 30.0

    while time.monotonic() < deadline:
        # current convolution and the index of the maximal entry
        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))

        # gradient of the active (max) shift (sub‑gradient of max‑conv)
        grad = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k_max - i_range
            grad[i_range] = 2.0 * a[j]

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # schedule for η (linear cooling)
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # exponentiated‑gradient (mirror‑descent) update on the simplex
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        a = a * (target_sum / a.sum())

        # occasional random block perturbation
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)
            a = a * (target_sum / a.sum())

        # occasional pair‑wise mass transfer (small local search)
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    cand = a.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = np.clip(cand, 0.0, None)
                    cand = cand * (target_sum / cand.sum())
                    cand_score = evaluate_sequence(cand.tolist())
                    cur_score = evaluate_sequence(a.tolist())
                    if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(0.001, 1.0)):
                        a = cand

        # soft restart if no improvement for a while
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] gradient restart – best {best_score:.6f}")

        # keep the best vector seen so far
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] new best after polishing = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best candidate (respect the limits in evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
```
