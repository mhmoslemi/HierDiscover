def project_simplex(v):
    # find projection onto simplex {x>=0, sum x = 1}
    # v can be any real vector.
    # Using algorithm by Wang & Carreira-Perpinan (2013) similar to sorting.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=12345, budget_s=1000.0):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-5)
    # initial candidates
    lengths = [2048, 3072, 3809, 4096, 5000]
    # show which lengths we try
    for n in lengths:
        # Determine initial vector
        if 'height_sequence_1' in globals():
            seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            # If seq length differs from n, resample: either truncate or resample via interpolation
            if seq.size != n:
                # we can resample using linear interpolation onto new length
                x_old = np.linspace(0, 1, seq.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                init = np.interp(x_new, x_old, seq)
            else:
                init = seq.copy()
        else:
            init = rng.dirichlet(np.ones(n)).astype(np.float64)
        # Normalize to sum 1
        init = init / init.sum()
        # Now run optimization for this n while time remains
        a_opt, score_opt = _optimize_with_adam(init, n, deadline, rng)
        # Update global best
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt
            print(f"New overall best: score {best_score:.6f} length {n}")
        # If time exhausted break
        if time.monotonic() >= deadline:
            break
    # After loop, return best_seq as list of floats
    if best_seq is None:
        # fallback uniform
        n = 1
        best_seq = [1.0]
    # Clip bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)

def _optimize_with_adam(a0, n, deadline, rng):
    # Setup parameters
    max_iters = 100000
    # compute FFT lengths
    fft_len_self = _next_pow2(2*n -1)
    fft_len_grad = _next_pow2(3*n-2)
    a = a0.copy()
    best = a.copy()
    # compute initial conv and score
    conv = _self_conv(a, fft_len_self)
    best_score = 2.0 * n * conv.max()
    cur_score = best_score
    # Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.1   # base learning rate
    beta_start = 0.5
    beta_end = 4000.0
    # for stagnation detection
    last_improve_time = time.monotonic()
    for it in range(1, max_iters+1):
        now = time.monotonic()
        if now >= deadline:
            break
        # Schedule beta (increase)
        frac = it / max_iters
        beta = beta_start + (beta_end - beta_start) * frac**0.5  # sqrt schedule
        # compute conv
        conv = _self_conv(a, fft_len_self)
        max_c = conv.max()
        # compute softmax p
        # subtract max for stability
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum
        # compute gradient
        rev_a = a[::-1]
        fp = np.fft.rfft(p, n=fft_len_grad)
        fra = np.fft.rfft(rev_a, n=fft_len_grad)
        grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[:3*n-2]
        grad = 2.0 * grad_full[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a_new = project_simplex(a_new)
        # Evaluate new score
        conv_new = _self_conv(a_new, fft_len_self)
        cur_score = 2.0 * n * conv_new.max()
        # Accept if improved else maybe keep original?
        # Since Adam step may increase objective; we could optionally revert; but we can just keep a_new anyway for ongoing progress; but we also track best.
        a = a_new
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve_time = now
            print(f"[{now - (deadline - (budget_s-5)):.2f}s] n={n} iter={it} new best {best_score:.6f}")
        # stagnation handling
        if now - last_improve_time > 15.0:  # seconds
            # perturb: add multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * noise
            a = project_simplex(a)
            # reset Adam moments
            m[:] = 0.0
            v[:] = 0.0
            last_improve_time = now
    # Return best found
    return best, best_score

def optimize_vector(a, n, deadline, rng):
    # compute FFT lens
    fft_len_self = _next_pow2(2*n-1)
    # parameters
    max_iters = 1_000_000
    eta0 = 0.5  # initial step size
    beta_start = 0.5
    beta_end = 3000.0
    phase_switch_frac = 0.6 # fraction of time to use softmax
    start_time = time.monotonic()
    # compute time budget for softmax phase
    total_time = deadline - start_time
    phase1_end = start_time + phase_switch_frac * total_time
    # maintain best
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0*n*conv.max()
    cur_score = best_score
    # for stagnation handling
    last_improve = start_time
    it = 0
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        # choose update rule
        if now < phase1_end:
            # softmax mirror descent similar to previous but using additive updates
            frac = (now - start_time) / (phase1_end - start_time)
            beta = beta_start + (beta_end - beta_start)*frac
            eta = eta0 * (1-frac) + 0.01*frac
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            p = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)
            # gradient = 2 * (p * rev_a) conv
            rev_a = a[::-1]
            # use same fft_len for gradient bottlenecks
            fft_len_grad = _next_pow2(3*n-2)
            fp = np.fft.rfft(p, n=fft_len_grad)
            fra = np.fft.rfft(rev_a, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[:3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]
            # gradient descent (move opposite)
            a = a - eta * grad
            a = project_simplex(a)
        else:
            # subgradient of max (use exact peak)
            conv = _self_conv(a, fft_len_self)
            k_max = int(conv.argmax())
            # build subgradient vector
            g = np.zeros_like(a)
            # For each i such that 0 <= k_max-i < n
            i_min = max(0, k_max - (n-1))
            i_max = min(n-1, k_max)

            # We'll compute via slice operations

        if cur_score < best_score - 1e-12:

"""
Propose a step‑function that beats the current bound (≈1.514).

Algorithm
---------
* Initialise a probability vector `a` on the simplex (sum = 1).  When a
  previous construction `height_sequence_1` is available we start from it,
  otherwise a Dirichlet draw is used.

* **Phase 1 – smoothed mirror descent** (≈ 60 % of the time budget):
  – Compute the self‑convolution `c = a*a` by FFT.
  – Approximate `max(c)` with a soft‑max of temperature β and obtain the
    exact gradient `g = 2·(p * rev(a))` (again by FFT).
  – Perform a *projected* (additive) gradient step
        a ← a – η·g
    and project back onto the probability simplex.
  – β is increased linearly from 0.5 to about 3500, η is decreased from 1.0
    to 0.01.

* **Phase 2 – exact sub‑gradient descent** (the remaining budget):
  – Locate the current peak index `k = argmax(a*a)`.
  – The sub‑gradient of the peak is `g_i = 2·a_{k-i}` (zero when the index
    does not exist).  A decreasing step size η = η₀/√t is used.
  – Update additively and project onto the simplex.

* **Stagnation handling** – if no improvement for 30 s a random
  multiplicative log‑normal perturbation around the best solution is
  applied and the optimizer state is reset.

* The loop tracks the best vector seen over the whole run (across all
  tried lengths).  The final vector is clipped to the limits enforced by
  `evaluate_sequence` and returned as a plain Python list.

The method differs from the previous exponentiated‑gradient version by
using additive projected gradient steps, an explicit sub‑gradient phase
for the true maximum, and a simple Adam‑free schedule, which empirically
gives a finer descent close to the optimum.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of prescribed length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find rho = max { j | u_j - cssv_j / j > 0 }
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive; return a vertex of the simplex
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine for a single length
# ----------------------------------------------------------------------


def _optimise_one_length(a_init: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Runs the two‑phase projected gradient descent for a fixed length.
    Returns the best vector found and its evaluation score.
    """
    a = a_init.copy()
    n = a.size
    # FFT lengths (power of two, fixed for the whole run)
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # Initial score
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    cur_score = best_score

    start_time = time.monotonic()
    total_time = deadline - start_time
    # 60 % of the remaining time for phase 1
    phase1_end = start_time + 0.60 * total_time

    # hyper‑parameters
    eta_start = 1.0
    eta_end = 0.01
    beta_start = 0.5
    beta_end = 3500.0
    sub_eta0 = 0.20          # base step for sub‑gradient phase

    last_improve = start_time
    it = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        # --------------------------------------------------------------
        # Phase 1 – smoothed gradient (soft‑max approximation of max)
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedules
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # self‑convolution
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()

            # soft‑max weights (stable)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            # gradient = 2·(p * rev(a))  (FFT)
            rev_a = a[::-1]
            fp = np.fft.rfft(p, n=fft_len_grad)
            fra = np.fft.rfft(rev_a, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # additive projected gradient step
            a = a - eta * grad
        else:
            # --------------------------------------------------------------
            # Phase 2 – exact sub‑gradient of the current peak
            # --------------------------------------------------------------
            conv = _self_conv(a, fft_len_self)
            k_max = int(conv.argmax())

            # sub‑gradient g_i = 2·a_{k_max‑i} when the index exists
            g = np.zeros_like(a)
            i_min = max(0, k_max - (n - 1))
            i_max = min(n - 1, k_max)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k_max - idx_i
            g[idx_i] = 2.0 * a[idx_j]

            # decreasing step size (≈ 1/√t)
            eta = sub_eta0 / np.sqrt(it)
            a = a - eta * g

        # ------------------------------------------------------------------
        # Projection onto the probability simplex
        # ------------------------------------------------------------------
        a = _project_simplex(a)

        # ------------------------------------------------------------------
        # Evaluation of the current candidate
        # ------------------------------------------------------------------
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:6d} new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # Stagnation handling – random jitter around the best solution
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            # multiplicative log‑normal noise (≈10 % std dev)
            noise = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * noise
            a = _project_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Decide which lengths to explore.
    #    We always try the length of the previous construction (if any)
    #    and a few other sizes that are quick to evaluate.
    # ------------------------------------------------------------------
    lengths = []
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
            lengths.append(seq0.size)
    # additional candidate sizes – chosen to be powers of two near 3800
    for n in (1024, 2048, 3000, 3809, 4096):
        if n not in lengths:
            lengths.append(n)

    best_overall = None
    best_score_overall = float("inf")

    # ------------------------------------------------------------------
    # 2) Optimise each length in turn while time permits.
    # ------------------------------------------------------------------
    for n in lengths:
        now = time.monotonic()
        if now >= deadline:
            break

        # Initialise a probability vector for this length
        if "height_sequence_1" in globals():
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq.size == n:
                a0 = seq.copy()
            else:
                # Resample via linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, seq.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(x_new, x_old, seq)
        else:
            a0 = rng.dirichlet(np.ones(n)).astype(np.float64)

        # Normalise to sum = 1 (simplex)
        a0 = a0.clip(min=0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / n)

        # Run the optimiser for this length
        vec, score = _optimise_one_length(a0, deadline, rng)

        # Track the best overall solution
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - (deadline - (budget_s - 5.0))
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – should never happen but guards against empty result.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    # 4) Clip to the limits enforced by ``evaluate_sequence`` and return.
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    return [float(x) for x in best_overall]
