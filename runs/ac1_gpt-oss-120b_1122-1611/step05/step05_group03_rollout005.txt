def _project_simplex(v):
    # v is 1-D array
    # Algorithm from Duchi et al.
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u - (cssv - 1) / (np.arange(1, len(v)+1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimise a non‑negative step function (probability vector) to minimise
``evaluate_sequence``. The routine mixes two complementary phases:

1. **Smoothed gradient descent** – the max in the convolution is
   approximated by a log‑sum‑exp (soft‑max) with a temperature α.
   The gradient of this smooth surrogate is computed analytically
   (requires one extra convolution) and a projected‑gradient step on
   the simplex is performed.  α is increased during the run so that
   the optimisation gradually focuses on the true supremum.

2. **Greedy mass‑transfer refinement** – after the gradient phase the
   algorithm switches to targeted transfers of mass between a donor
   (large contribution to the current peak) and a receiver (small
   contribution).  Several candidate step sizes are tried and the
   best improving move is applied.  Random block perturbations and
   periodic restarts keep the search from stalling.

The routine respects the 1000 s time budget, returns the best vector
found (clipped to the limits of ``evaluate_sequence``) and uses the
previous best construction if the global variable ``height_sequence_1``
is present.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear (full) self‑convolution a*a."""
    return np.convolve(a, a)  # length 2·len(a)‑1


def _apply_transfer(
    a: np.ndarray,
    conv: np.ndarray,
    i: int,
    j: int,
    delta: float,
) -> np.ndarray:
    """
    Return a new convolution array after moving ``delta`` mass from index i
    to index j (i != j).  Uses the exact formula derived from
    (a+Δ)*(a+Δ) = a*a + 2·a·Δ + Δ·Δ.
    """
    n = a.size
    new_conv = conv.copy()
    # linear part: -2Δ·shift(a,i) + 2Δ·shift(a,j)
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a
    # quadratic part Δ²
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def _random_block_perturb(
    a: np.ndarray,
    rng: np.random.Generator,
    max_width_frac: float = 0.2,
    sigma: float = 0.12,
) -> np.ndarray:
    """
    Apply a multiplicative log‑normal jitter to a random contiguous block
    and renormalise onto the probability simplex.
    """
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_array = u - (cssv - 1) / (np.arange(1, len(v) + 1))
    rho = np.nonzero(rho_array > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights proportional to exp(α·conv).  The largest entry of
    ``conv`` is subtracted for numerical stability.
    """
    if alpha <= 0.0:
        # plain average – avoid overflow/underflow
        return np.full_like(conv, 1.0 / conv.size)
    max_c = np.max(conv)
    exp_term = np.exp(alpha * (conv - max_c))
    sum_exp = np.sum(exp_term)
    if sum_exp == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / sum_exp


def _gradient_a(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of ``max(conv)`` with respect to
    the coefficient vector ``a``.  Uses the identity
    ∂(a*a)_k/∂a_i = 2·a_{k‑i} (zero‑padded) and the soft‑max weights.
    """
    w = _softmax_weights(conv, alpha)
    # convolution of w with reversed a gives Σ_k w_k a_{k‑i}
    grad = 2.0 * np.convolve(w, a[::-1], mode="full")[len(a) - 1 : len(a) - 1 + len(a)]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  Returns the best sequence found
    within the allocated time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the caller reserves a few seconds margin).

    Returns
    -------
    list[float]
        The best coefficient list discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Initialise the coefficient vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            a = a / total
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # moderate uniform start – length 4096 is a good trade‑off
        n0 = 4096
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    conv = _self_convolution(a)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    last_improve = start_time
    iteration = 0

    # Phase split – first 40 % of the time for gradient descent,
    # remaining time for greedy refinement.
    total_time = deadline - start_time
    grad_phase_end = start_time + 0.4 * total_time

    # --------------------------------------------------------------
    # 3) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        iteration += 1

        # ----------------------------------------------------------
        # 3a) Gradient‑descent phase (soft‑max smoothing)
        # ----------------------------------------------------------
        if now < grad_phase_end:
            # recompute convolution for the current a
            conv = _self_convolution(a)
            cur_score = 2.0 * n * conv.max()

            # temperature schedule – start low, grow quadratically
            elapsed = now - start_time
            frac = elapsed / (grad_phase_end - start_time + 1e-12)
            alpha = 2.0 + 200.0 * (frac ** 2)   # α ∈ [2, ≈202]

            grad = _gradient_a(a, conv, alpha)

            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0.0:
                # nothing to do – break out of gradient phase early
                grad_phase_end = now
                continue

            # base step size that slowly shrinks
            base_step = 0.25
            step = base_step / (1.0 + iteration * 0.001)

            # unit direction
            direction = grad / grad_norm

            # try a few scaling factors and keep the best improvement
            best_local_score = cur_score
            best_candidate = a
            best_conv = conv

            for scale in (1.0, 0.5, 0.25, 0.1):
                a_tmp = a - step * scale * direction
                a_tmp = _project_simplex(a_tmp)

                conv_tmp = _self_convolution(a_tmp)
                score_tmp = 2.0 * n * conv_tmp.max()

                if score_tmp < best_local_score:
                    best_local_score = score_tmp
                    best_candidate = a_tmp
                    best_conv = conv_tmp

            if best_local_score < cur_score - 1e-12:
                a = best_candidate
                conv = best_conv
                cur_score = best_local_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve-start_time:6.2f}s] grad new best {best_score:.6f}")

            # occasional random block jitter to escape plateaus
            if rng.random() < 0.008:
                a = _random_block_perturb(a, rng)
                a = _project_simplex(a)
                conv = _self_convolution(a)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve-start_time:6.2f}s] grad jitter improved best {best_score:.6f}")

            continue   # skip the greedy part until the time budget moves on

        # ----------------------------------------------------------
        # 3b) Greedy targeted mass‑transfer refinement
        # ----------------------------------------------------------
        # (same idea as the previous state‑of‑the‑art but with a tiny
        # line‑search on the transfer amount)
        conv = _self_convolution(a)
        cur_score = 2.0 * n * conv.max()
        k_max = int(np.argmax(conv))

        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        partner = np.zeros_like(a)
        partner[valid] = a[k_max - idx[valid]]

        donor_mask = a > 1e-12
        donor_score = partner * donor_mask.astype(float)

        if donor_score.max() == 0.0:
            # fallback: random move if no useful donor exists
            i = rng.integers(0, n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_delta = 0.3 * a[i]
            delta = rng.random() * max_delta
            new_conv = _apply_transfer(a, conv, i, j, delta)
            new_score = 2.0 * n * new_conv.max()
            if new_score < cur_score:
                a[i] -= delta
                a[j] += delta
                conv = new_conv
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve-start_time:6.2f}s] greedy new best {best_score:.6f}")
            continue

        donor = int(np.argmax(donor_score))
        # receiver: an index with the smallest partner value
        min_partner = partner.min()
        receiver_candidates = np.where(partner == min_partner)[0]
        if donor in receiver_candidates:
            receiver_candidates = receiver_candidates[receiver_candidates != donor]
        if receiver_candidates.size == 0:
            receiver = (donor + 1) % n
        else:
            receiver = int(rng.choice(receiver_candidates))

        # maximum transferable fraction (shrinks over time implicitly via a[donor])
        max_frac = 0.3
        max_delta = a[donor] * max_frac
        if max_delta < 1e-13:
            continue

        # try a few deltas – the one giving the largest drop is kept
        best_local_score = cur_score
        best_delta = 0.0
        for frac in (1.0, 0.5, 0.25, 0.1):
            delta = max_delta * frac
            new_conv = _apply_transfer(a, conv, donor, receiver, delta)
            new_score = 2.0 * n * new_conv.max()
            if new_score < best_local_score:
                best_local_score = new_score
                best_delta = delta

        if best_delta > 0.0:
            new_conv = _apply_transfer(a, conv, donor, receiver, best_delta)
            a[donor] -= best_delta
            a[receiver] += best_delta
            conv = new_conv
            cur_score = best_local_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve-start_time:6.2f}s] greedy new best {best_score:.6f}")

        # occasional block perturbation to shake the search
        if rng.random() < 0.006:
            a = _random_block_perturb(a, rng)
            a = _project_simplex(a)
            conv = _self_convolution(a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve-start_time:6.2f}s] block‑perturb improved best {best_score:.6f}")

        # restart if we have been idle for a while
        if now - last_improve > 30.0:
            a = best.copy()
            # small random multiplicative noise before restarting
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = _project_simplex(a)
            conv = _self_convolution(a)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{now-start_time:6.2f}s] restart – score reset to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
