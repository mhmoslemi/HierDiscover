def simplex_projection(v):
    # v is np array.
    # Using algorithm from Duchi et al.

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, len(v) + 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Define function cross_corr_valid(w, a):
    # w length m = 2n-1, a length n.
    # Use FFT with length L = smallest power of two >= m + n - 1 = 3n -2.
    L = 1 << ((m+n-1 - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a[::-1], n=L)
    conv = np.fft.irfft(W * A_rev, n=L)[: m + n - 1]
    # cross-correlation result of length m + n -1
    # extract valid part: conv[ n-1 : n-1 + n] = conv[n-1:n-1+n]
    result = conv[n-1 : n-1 + n]
    return result

Implementation of projection onto simplex:

def project_simplex(v):
    # returns projected vector onto simplex {x>=0, sum=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all zeros? return uniform?
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
smooth_max_optimize.py
~~~~~~~~~~~~~~~~~~~~~~

A self‑contained proposal for the ``propose_candidate`` entry point.
The algorithm consists of three stages:

1. **Broad sweep of the arcsine family** –  
   for several lengths and a fine grid of the exponent α we construct
   ``a_i ∝ ((i+½)*(n‑½‑i))^(-α)`` (the continuous arcsine density).
   The best candidate from this sweep (or the previously stored
   ``height_sequence_1``) is taken as the starting point.

2. **Gradient descent on a smooth approximation of the max‑convolution** –  
   we minimise the smooth objective  

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

   (which approaches the true max when β → ∞).  Its gradient is

        ∇f_β(a) = 2·β·corr(w, a) ,

   where ``w`` is the soft‑max weights of the convolution and
   ``corr`` denotes the *valid* cross‑correlation.
   The update is a projected gradient step onto the probability simplex
   (non‑negative entries with unit sum).  β is slowly increased from 1
   up to a few hundred, while the step size η is scaled as 1/β.

3. **Fine‑grained exponentiated‑gradient polishing** –  
   after the smooth phase we finish with a few hundred iterations that
   act only on the index attaining the maximal convolution value,
   exactly as in the reference implementation.  This often yields a
   tiny extra drop in the final score.

All heavy work is performed with FFT‑based convolutions, which makes a
single iteration on a vector of a few thousand entries take well under a
millisecond.  The whole routine respects the 1000 s budget and prints
progress updates together with the best score found so far.

The returned sequence contains only non‑negative numbers in the interval
[0, 1000] as required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n(n: int) -> int:
    """
    Length of the FFT that can hold the full linear convolution of two
    vectors of length ``n`` (i.e. length 2·n‑1).
    """
    return _next_pow2(2 * n - 1)


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative real vector ``a`` using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _cross_corr_valid(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Valid cross‑correlation ``corr_i = Σ_j w[i+j]·a[j]`` (length = len(a)).
    Computed via FFT as ``conv(w, a_rev)`` and then slicing.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_len)
    A_rev = np.fft.rfft(a_rev, n=fft_len)
    conv_full = np.fft.irfft(W * A_rev, n=fft_len)
    # keep only the first (len(w)+len(a)-1) entries
    conv_full = conv_full[: w.size + a.size - 1]
    start = a.size - 1               # index n‑1
    return conv_full[start:start + a.size]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are (almost) zero
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # stop a little before the deadline to avoid a hard kill
    deadline = start + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0)  Baseline: either the previously stored sequence or a sweep
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.array(list(map(float, globals()["height_sequence_1"])),
                            dtype=np.float64)
            if cand.sum() < 1e-12:
                cand = np.full_like(cand, 1.0 / cand.size)
            cand = cand / cand.sum()
            n = cand.size
            conv = _self_conv(cand, _fft_len_for_n(n))
            score = 2.0 * n * np.max(conv)
            best_seq = cand.tolist()
            best_score = float(score)
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] baseline (global) score = {best_score:.6f}"
                  f" (n={n})")
        except Exception:
            pass

    # If we have no baseline yet, perform a quick sweep over the arcsine
    # family for a few lengths and a dense grid of exponents α.
    if best_seq is None:
        lengths = [1024, 2048, 4096, 5120]          # a few reasonable sizes
        alphas = np.linspace(0.35, 0.65, 31)        # fine grid around 0.5
        for n in lengths:
            if time.monotonic() > deadline:
                break
            i = np.arange(n, dtype=np.float64)
            for alpha in alphas:
                if time.monotonic() > deadline:
                    break
                raw = 1.0 / ((i + 0.5) * (n - 0.5 - i)) ** alpha
                raw = np.clip(raw, 0.0, 1000.0)
                raw /= raw.sum()
                conv = _self_conv(raw, _fft_len_for_n(n))
                score = 2.0 * n * np.max(conv)
                if score < best_score:
                    best_score = float(score)
                    best_seq = raw.tolist()
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:6.2f}s] sweep → n={n} α={alpha:.3f}"
                          f" score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1)  Smooth‑max gradient descent
    # ------------------------------------------------------------------
    a = np.array(best_seq, dtype=np.float64)     # start point (already normalised)
    n = a.size
    fft_len_self = _fft_len_for_n(n)
    fft_len_corr = _next_pow2(3 * n - 2)          # for the cross‑correlation

    beta = 1.0
    max_beta = 200.0
    beta_mul = 1.001          # geometric increase per iteration
    base_eta = 0.02           # base step size (scaled by 1/β later)

    it = 0
    # we evaluate only every few iterations to keep the output readable
    eval_every = max(1, n // 200)

    while time.monotonic() < deadline:
        # ---- convolution and soft‑max weights ---------------------------------
        conv = _self_conv(a, fft_len_self)         # self‑convolution (2·n‑1 entries)
        max_conv = np.max(conv)

        # soft‑max (numerically stable)
        shifted = conv - max_conv
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        w /= w_sum

        # ---- gradient of the smooth max (log‑sum‑exp) -------------------------
        grad = 2.0 * beta * _cross_corr_valid(w, a, fft_len_corr)

        # ---- projected gradient step -----------------------------------------
        eta = base_eta / beta
        a = a - eta * grad
        a = _project_simplex(a)

        # ---- occasional stochastic jitter (helps escape plateaus) ------------
        if rng.random() < 0.001:        # about one jitter per thousand iterations
            jitter = np.exp(rng.normal(0.0, 0.008, size=n))
            a *= jitter
            a = _project_simplex(a)

        # ---- update beta ------------------------------------------------------
        beta = min(beta * beta_mul, max_beta)

        # ---- keep the best observed score ------------------------------------
        if it % eval_every == 0:
            score = 2.0 * n * max_conv    # sum(a) == 1 by construction
            if score < best_score:
                best_score = float(score)
                best_seq = a.tolist()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] smooth‑max best = {best_score:.6f}"
                      f" (β={beta:.2f}, iter={it})")
        it += 1

    # ------------------------------------------------------------------
    # 2)  Final polishing with pure max‑gradient (exponentiated‑gradient)
    # ------------------------------------------------------------------
    # If a few seconds are still left we run a short classic refinement.
    while time.monotonic() < deadline - 0.1:
        conv = _self_conv(a, fft_len_self)
        max_conv = np.max(conv)
        k_max = int(np.argmax(conv))

        # gradient of the *true* max‑convolution (sub‑gradient at the active index)
        grad = np.zeros_like(a)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        grad[mask] = 2.0 * a[partner[mask]]

        # mirror descent on the simplex (exponentiated‑gradient)
        eta = 0.12                     # small step size for the fine phase
        a = a * np.exp(-eta * grad)
        a = _project_simplex(a)

        # keep track of improvement
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = float(score)
            best_seq = a.tolist()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] final polish best = {best_score:.6f}")

        # tiny decay of the step size – prevents overshooting
        eta *= 0.9999

    # ------------------------------------------------------------------
    # 3)  Safe fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = float('inf')

    # Clip to the limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f}"
          f" (length {len(best_seq)})")
    return best_seq
