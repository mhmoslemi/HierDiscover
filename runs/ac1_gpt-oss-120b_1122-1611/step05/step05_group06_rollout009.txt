def propose_candidate(seed=42,budget_s=1000.0,**kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    
    best_vec = None; best_score = float('inf')
    
    # include existing hi_seq
    if 'height_sequence_1' in globals():
        process candidate as before (normalize to sum=1). compute score via _compute_score.
    
    lengths = [1500,2000,3000,3809,5000,8000,12000]  # Various

    # main loop: while time left
    while time.monotonic() < deadline:
        n = int(rng.choice(lengths))
        a0 = rng.random(n)
        a0 /= a0.sum()
        a_opt, score_opt = _local_optimize(a0, rng, deadline, max_iters=800)  # short
        if score_opt < best_score:
            best_score = score_opt; best_vec = a_opt.copy()
            print(f"[{time.monotonic()-start:.2f}s] New global best {best_score:.6f} (n={n})")
    
        # occasional polishing of global best
        if best_vec is not None and time.monotonic() < deadline - 5.0:
            a_polished, score_polished = _polish_optimize(best_vec.copy(), rng, deadline)
            if score_polished < best_score:
                best_score = score_polished; best_vec = a_polished.copy()
                print(f"[{time.monotonic()-start:.2f}s] Polished to {best_score:.6f}")
    
    if best_vec is None:
        n=2000
        best_vec = np.full(n, 1.0/n)
    
    # Clip and return list

    return [float(x) for x in best_vec]

def _local_optimize(a0,rng,deadline,max_iters=1200):
    a = a0.copy()
    n = a.size
    fft_len_self = _fft_len_for_len(2*n - 1)
    fft_len_grad = _fft_len_for_len(3*n - 2)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_a = a.copy()
    conv = _conv_self(a, fft_len_self)
    best_score = 2.0*n*conv.max()
    for t in range(1, max_iters+1):
        if time.monotonic() >= deadline: break
        # compute current conv and score
        conv = _conv_self(a, fft_len_self)
        cur_score = 2.0*n*conv.max()
        if cur_score < best_score:
            best_score = cur_score; best_a = a.copy()
        # choose gradient
        if rng.random() < 0.5:
            grad = _subgradient_step(a, conv)
        else:
            beta = 1.0 + (2000.0-1.0)*(t/max_iters)
            grad = _softmax_gradient(a, conv, beta, fft_len_grad)
        # Adam
        lr = 0.5/np.sqrt(t)    # schedule
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)

    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimises `evaluate_sequence`.

Algorithm
---------
* Work on the probability simplex Σa = 1 (the objective is scale invariant).
* Perform many cheap random restarts with a fast Adam‑style optimiser.
* The optimiser mixes a hard sub‑gradient of the current worst convolution
  entry with a smooth “log‑sum‑exp” gradient (soft‑max) whose temperature
  β grows linearly during a run.
* Random block perturbations and pairwise mass‑transfer moves are injected
  regularly; a targeted pairwise move preferentially removes mass from
  indices that contribute most to the current worst offset.
* The best vector seen so far is further polished with a long optimisation
  run (several thousand iterations) and a final simulated‑annealing style
  hill‑climb that accepts occasional worsening moves.
* The whole search respects a 1000 s time budget; it stops a few seconds
  early to guarantee a return value.

The code is fully self‑contained, uses only NumPy and the standard library,
and respects the “no closures” rule.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors `x` and `y` via FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max(conv)` with respect to the vector `a`.
    For the worst offset `i_max` the gradient entry j is
    `2 * a[i_max-j]` whenever the pair (j,i_max-j) is inside the array.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(
    a: np.ndarray, conv: np.ndarray, beta: float, fft_len_grad: int
) -> np.ndarray:
    """
    Gradient of the smooth max `f_β(a) = (1/β)·log Σ_i exp(β·conv_i)`.
    This uses the convolution of the weight vector `w` with the reversed `a`.
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)  # length = 3n‑2
    # extract the slice that corresponds to the gradient w.r.t. a
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    # allow fairly wide blocks (up to ~10 % of the vector)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.25))
    out = a.copy()
    out[start : start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(
    a: np.ndarray, rng: np.random.Generator, fft_len_self: int
) -> np.ndarray:
    """Cheap move: transfer a modest amount of mass from i to j."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    # accept only if it improves the score
    conv2 = _conv_self(a2, fft_len_self)
    if conv2.max() < _conv_self(a, fft_len_self).max():
        return a2
    return a


def _targeted_pairwise_move(
    a: np.ndarray,
    conv: np.ndarray,
    rng: np.random.Generator,
    fft_len_self: int,
) -> np.ndarray:
    """
    Move mass from an index that contributes heavily to the worst
    convolution offset towards an index that contributes little.
    """
    n = a.size
    i_max = int(conv.argmax())
    # contributions of each index to the worst offset
    contrib = np.zeros(n)
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    contrib[idx] = a[idx] * a[i_max - idx]
    total = contrib.sum()
    if total == 0.0:
        return a

    # source index (high contribution)
    probs_src = contrib / total
    i = rng.choice(n, p=probs_src)

    # target index (low contribution)
    low = contrib.max() - contrib
    low += 1e-12  # avoid zeroes
    probs_tgt = low / low.sum()
    j = rng.choice(n, p=probs_tgt)

    if i == j or a[i] == 0.0:
        return a

    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)

    # accept only if the worst convolution actually decreases
    conv2 = _conv_self(a2, fft_len_self)
    if conv2.max() < conv.max():
        return a2
    return a


def _compute_score(a: np.ndarray, fft_len_self: int) -> float:
    """Return the raw objective `2·n·max(conv)` for a simplex‑normalised vector."""
    n = a.size
    conv = _conv_self(a, fft_len_self)
    return 2.0 * n * conv.max()


def _local_optimize(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 1200,
    use_targeted_move: bool = True,
    accept_worse: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Short, intensive optimisation of a single vector.
    Returns the best vector discovered and its score.
    """
    a = a0.copy()
    n = a.size
    fft_len_self = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state (only the second‑moment is really needed)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = _compute_score(a, fft_len_self)

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # --------------------------------------------------------------
        # 1) current convolution and score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # --------------------------------------------------------------
        # 2) choose a gradient (hard sub‑gradient or smooth soft‑max)
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            raw_grad = _subgradient_step(a, conv)
        else:
            # β grows from ≈1 to a few thousand – linear schedule works well
            beta = 1.0 + (2500.0 - 1.0) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # 3) Adam‑type additive update
        # --------------------------------------------------------------
        lr = 0.6 / np.sqrt(t)          # decreasing learning rate
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate)

        cand_score = _compute_score(a_candidate, fft_len_self)

        # --------------------------------------------------------------
        # 4) accept or reject the Adam step
        # --------------------------------------------------------------
        if cand_score <= cur_score or accept_worse:
            # a move that improves (or is allowed to worsen) is kept
            a = a_candidate
        # otherwise we keep the old a (implicit rejection)

        # --------------------------------------------------------------
        # 5) random perturbations
        # --------------------------------------------------------------
        if rng.random() < 0.025:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.03:
            a = _pairwise_move(a, rng, fft_len_self)

        if use_targeted_move and rng.random() < 0.015:
            a = _targeted_pairwise_move(a, conv, rng, fft_len_self)

    return best_a, best_score


def _simulated_annealing(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_moves: int = 6000,
) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing hill‑climb that tries random pairwise
    transfers; it accepts worsening moves with a temperature that decays
    exponentially.  Returns the best vector seen.
    """
    a = a0.copy()
    n = a.size
    fft_len_self = _fft_len_for_len(2 * n - 1)

    cur_score = _compute_score(a, fft_len_self)
    best_a = a.copy()
    best_score = cur_score

    # exponential temperature schedule
    T0 = 0.02
    Tf = 0.0002
    for move in range(1, max_moves + 1):
        if time.monotonic() >= deadline:
            break
        T = T0 * ((Tf / T0) ** (move / max_moves))

        # propose a small pairwise transfer
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        max_delta = min(a[i], 0.1 * a[i])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = _project_onto_simplex(a2)

        new_score = _compute_score(a2, fft_len_self)
        delta_score = new_score - cur_score

        if delta_score <= 0.0 or rng.random() < np.exp(-delta_score / T):
            a = a2
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine respects a 1000 s time‑budget,
    performs many random restarts followed by a long polishing phase,
    and finally returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin so that the harness can still evaluate
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) initialise from any previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cur = _compute_score(cand, _fft_len_for_len(2 * cand.size - 1))
                if cur < best_score:
                    best_score = cur
                    best_vec = cand.copy()
                print(f"Initial global candidate score = {cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) lengths to explore – a mix of small / medium / large vectors
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 3000, 3809, 5000, 8000, 12000]

    # ------------------------------------------------------------------
    # 3) Main search loop – many cheap restarts
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (biased towards larger ones)
        probs = np.array([0.12, 0.15, 0.18, 0.20, 0.18, 0.12, 0.05])
        probs = probs / probs.sum()
        n = int(rng.choice(lengths, p=probs))

        a0 = rng.random(n)
        a0 /= a0.sum()

        # short optimisation (few hundred iterations)
        a_opt, score_opt = _local_optimize(
            a0, rng, deadline, max_iters=800, use_targeted_move=True
        )
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        # 4) Occasionally polish the current global best with a long run
        # ------------------------------------------------------------------
        if best_vec is not None and time.monotonic() < deadline - 2.0:
            # longer Adam‑style optimisation
            a_polished, score_polished = _local_optimize(
                best_vec.copy(),
                rng,
                deadline,
                max_iters=3500,
                use_targeted_move=True,
                accept_worse=False,
            )
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polished global best (Adam): {best_score:.6f}")

            # final simulated‑annealing pass for fine adjustments
            if time.monotonic() < deadline - 1.0:
                a_sa, score_sa = _simulated_annealing(
                    best_vec.copy(), rng, deadline, max_moves=6000
                )
                if score_sa < best_score:
                    best_score = score_sa
                    best_vec = a_sa.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] Polished global best (SA): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 6) Final safety clipping (the evaluator clips as well)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
