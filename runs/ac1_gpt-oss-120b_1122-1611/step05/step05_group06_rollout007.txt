def _optimise_exponential(a, rng, deadline, max_iters=2000):
   n = a.size
   fft_len = _fft_len_for_len(2*n-1)
   best_vec = a.copy(); best_score = 2*n*_conv_self_fft(a, fft_len).max()
   for t in range(1, max_iters+1):
        if time.monotonic() >= deadline:
            break
        conv = _conv_self_fft(a, fft_len)
        cur_score = 2*n*conv.max()
        if cur_score < best_score:
            best_vec = a.copy()
            best_score = cur_score

        # worst index and subgradient
        i_max = int(conv.argmax())
        # compute gradient
        lo = max(0, i_max - (n - 1))
        hi = min(i_max, n - 1)
        idx = np.arange(lo, hi+1)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[i_max - idx]
        # step size
        eta = 0.5 / np.sqrt(t)
        # multiplicative update
        a = a * np.exp(-eta * grad)
        a = a / a.sum()

        # random perturbations
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.05:
            # pairwise move with acceptance/annealing
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                max_delta = min(a[i], 0.2 * a[i])
                delta = rng.random() * max_delta
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                # evaluate new score
                conv2 = _conv_self_fft(a2, fft_len)
                new_score = 2*n*conv2.max()
                # temperature decreasing
                T = max(0.001, 0.01 * (1 - t/max_iters))
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
                    a = a2
                    # no need to renorm as sum preserved

   return best_vec, best_score

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().

Approach
--------
* Work on the probability simplex ( Σ a = 1 ).
* Perform a first phase of cheap restarts using an Adam‑style additive
  update on the exact sub‑gradient (or a smooth log‑sum‑exp gradient).
* Keep the best candidate found.
* In a second phase run a longer fine‑tuning using an exponentiated‑gradient
  (mirror‑descent w.r.t. KL) which naturally stays on the simplex.
* Occasionally apply random block perturbations and pairwise mass‑transfer
  moves; the latter are accepted with a simple simulated‑annealing rule.
* All convolutions are performed by a real FFT for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors `x` and `y` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient / perturbation helpers
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv) w.r.t. `a` (worst convolution entry)."""
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(
    a: np.ndarray,
    conv: np.ndarray,
    beta: float,
    fft_len_grad: int,
) -> np.ndarray:
    """∂/∂a ( 1/β·log Σ_i e^{β·conv_i} )."""
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    # slice to obtain the gradient entries
    grad = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start : start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(
    a: np.ndarray,
    rng: np.random.Generator,
    fft_len_self: int,
    cur_score: float,
    temperature: float,
) -> np.ndarray:
    """
    Transfer a random amount of mass from i to j.
    Accepts if the new score is lower or with a Metropolis probability.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    # convolution of the proposal
    conv2 = _conv_self_fft(a2, fft_len_self)
    new_score = 2.0 * n * conv2.max()
    if new_score < cur_score:
        return a2
    # occasional uphill acceptance (simulated annealing)
    if rng.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
        return a2
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style additive updates on the simplex.
    Returns best vector seen and its 2·n·max(conv) score.
    """
    a = a0.copy()
    n = a.size
    fft_len_self = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_vec = a.copy()
    conv = _conv_self_fft(a, fft_len_self)
    best_score = 2.0 * n * conv.max()

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = _conv_self_fft(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------- gradient ---------------------------
        if rng.random() < 0.55:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = 0.5 + (2000.0 - 0.5) * (t / max_iters)   # linear beta schedule
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # ----------------------------- Adam step -----------------------
        lr = 0.7 / np.sqrt(t)                # decreasing learning rate
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # ---------------------- occasional random moves ---------------
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.04:
            # temperature decays linearly inside this short run
            temperature = 0.02 * (1.0 - t / max_iters)
            a = _pairwise_move(a, rng, fft_len_self, cur_score, temperature)

    return best_vec, best_score


def _exponential_optimize(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 10000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the simplex.
    Uses a sub‑gradient of the exact max (occasionally the smooth log‑sum‑exp).
    """
    a = a0.copy()
    n = a.size
    fft_len_self = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    best_vec = a.copy()
    conv = _conv_self_fft(a, fft_len_self)
    best_score = 2.0 * n * conv.max()

    eta0 = 0.5

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        conv = _conv_self_fft(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # gradient choice
        if rng.random() < 0.5:
            # exact sub‑gradient of max
            grad = _subgradient_step(a, conv)
        else:
            beta = 0.5 + (2000.0 - 0.5) * (t / max_iters)
            grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # exponentiated (multiplicative) update
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()    # stays on simplex

        # random perturbations
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # pairwise mass‑transfer with simulated‑annealing acceptance
        if rng.random() < 0.04:
            temperature = 0.015 * (1.0 - t / max_iters) + 1e-6
            a = _pairwise_move(a, rng, fft_len_self, cur_score, temperature)

    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence() within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the final conversion/return
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Initialise from a previous solution if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / s
                cur = 2.0 * cand.size * _conv_self_fft(
                    cand, _fft_len_for_len(2 * cand.size - 1)
                ).max()
                if cur < best_score:
                    best_score = cur
                    best_vec = cand.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] Loaded prior candidate "
                      f"score = {cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    lengths = [800, 1200, 2000, 3000, 3809, 5000, 7200]
    probs = np.array([0.1, 0.1, 0.2, 0.2, 0.2, 0.1, 0.1])
    probs = probs / probs.sum()

    # ------------------------------------------------------------------
    # Phase 1 – many cheap restarts with Adam optimisation
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 30.0:   # keep ~30 s for Phase 2
        n = int(rng.choice(lengths, p=probs))
        a0 = rng.random(n)
        a0 /= a0.sum()

        # number of inner iterations grows for smaller n (cheaper)
        max_iters = int(2000 * 2000 / n)
        max_iters = max(300, min(max_iters, 1200))

        a_opt, score_opt = _adam_optimize(a0, rng, deadline, max_iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best (phase 1): "
                  f"{best_score:.6f}  (n={n})")

    # ------------------------------------------------------------------
    # Phase 2 – long fine‑tune on the current best
    # ------------------------------------------------------------------
    if best_vec is None:
        # fallback: uniform distribution of moderate size
        n = 2000
        best_vec = np.full(n, 1.0 / n)

    # allocate remaining time to the fine‑tuning phase
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        # the number of iterations is chosen adaptively (more when n is small)
        max_iters = int(5e5 / best_vec.size)          # a few hundred‑thousand updates total
        max_iters = max(5000, min(max_iters, 20000))
        a_opt, score_opt = _exponential_optimize(best_vec, rng, deadline, max_iters)
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best (phase 2): {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping – evaluate_sequence will also clip,
    # but we do it here for completeness.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
