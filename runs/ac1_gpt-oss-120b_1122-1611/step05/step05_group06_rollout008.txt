"""
Improved stochastic optimisation for the step‑function problem.

The algorithm works on the probability simplex (Σa = 1) and
combines three ideas that were missing or not fully exploited
by the reference implementation:

1. **Hybrid updates** – each iteration picks either an additive
   (Adam‑style) update or a multiplicative (exponentiated‑gradient)
   update, whichever yields a decrease.  A simple back‑tracking line
   search guarantees monotone improvement.

2. **More aggressive temperature schedule** – the smooth‑max (soft‑max)
   surrogate is driven to a very large β (up to 5 000) so that the
   gradient approximates the true sub‑gradient more closely.

3. **Two‑phase search** – many cheap restarts (≈200 iterations) explore
   a wide range of lengths; the current global best is then refined
   for thousands of iterations with a slower learning‑rate decay.

All helper functions are top‑level, there are no closures or lambdas.
The routine respects the 1000 s time budget and returns a list of
non‑negative heights that minimise ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors `x` and `y` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a * a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient computations
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. `a` (hard max).
    For the worst convolution index i_max,
    ∂/∂a_j max(conv) = 2·a_{i_max‑j} (if the index is inside the vector).
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(
    a: np.ndarray, conv: np.ndarray, beta: float, fft_len_grad: int
) -> np.ndarray:
    """
    Gradient of the smooth max
        f_β(a) = (1/β)·log Σ_i exp(β·conv_i)
    Returns ∂f_β/∂a (the correct factor 2 is included).
    """
    cur_max = conv.max()
    # stabilised exponentials
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    # slice to obtain n‑length result
    grad_part = 2.0 * conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Random perturbations (used as occasional “jumps”)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    b = a.copy()
    b[start:start + width] *= factor
    return _project_onto_simplex(b)


def _pairwise_move(
    a: np.ndarray, rng: np.random.Generator, fft_len_conv: int
) -> np.ndarray:
    """Transfer a tiny amount of mass from i to j; keep the move only if it improves."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    # move at most 20 % of the source mass
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    b = _project_onto_simplex(b)

    conv_old = _conv_self(a, fft_len_conv).max()
    conv_new = _conv_self(b, fft_len_conv).max()
    if conv_new < conv_old:
        return b
    return a


# ----------------------------------------------------------------------
# Core optimiser (single restart)
# ----------------------------------------------------------------------


def _local_optimise(
    a0: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 2000,
) -> Tuple[np.ndarray, float]:
    """
    Perform a (potentially long) local optimisation of a simplex vector.
    Returns the best vector seen and its score 2·n·max(conv).
    """
    n = a0.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    a = a0.copy()
    best_vec = a.copy()
    conv = _conv_self(a, fft_len_conv)
    best_score = 2.0 * n * conv.max()

    # Adam state – only second moment (v) is required for stability
    v = np.zeros_like(a)
    beta2 = 0.999
    eps = 1e-8

    # schedule parameters
    base_lr = 30.0          # base learning rate for multiplicative updates
    base_lr_add = 1.0       # base learning rate for additive Adam steps
    beta = 0.5              # initial smooth‑max temperature

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # --------------------------------------------------------------
        # 1) current convolution and score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # 2) choose a gradient (hard sub‑gradient vs soft‑max)
        # --------------------------------------------------------------
        if rng.random() < 0.75:
            # hard sub‑gradient (exact)
            raw_grad = _subgradient_step(a, conv)
        else:
            # smooth max – increase β slowly (starts from 0.5 up to ~5 000)
            beta = min(5000.0, beta + (5000.0 - 0.5) / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # 3) decide on additive or multiplicative update
        # --------------------------------------------------------------
        if rng.random() < 0.5:
            # ----  additive Adam‑style step  -------------------------
            lr = base_lr_add / np.sqrt(t)          # decreasing step size
            v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
            step = lr * raw_grad / (np.sqrt(v) + eps)

            # simple back‑tracking: try a few smaller steps if no improvement
            accepted = False
            for _ in range(5):
                a_new = a - step
                a_new = _project_onto_simplex(a_new)
                conv_new = _conv_self(a_new, fft_len_conv)
                score_new = 2.0 * n * conv_new.max()
                if score_new < cur_score:
                    a = a_new
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                # keep current a (no progress this iteration)
                pass
        else:
            # ---- multiplicative (exponentiated‑gradient) step -------
            lr = base_lr / np.sqrt(t)
            # limit gradient magnitude to avoid overflow in the exponent
            grad_clipped = np.clip(raw_grad, -10.0, 10.0)
            a_new = a * np.exp(-lr * grad_clipped)
            a_new = a_new / a_new.sum()
            conv_new = _conv_self(a_new, fft_len_conv)
            score_new = 2.0 * n * conv_new.max()
            if score_new < cur_score:
                a = a_new
            # otherwise reject – keep the old a

        # --------------------------------------------------------------
        # 4) occasional random “jumps” to escape shallow minima
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.02:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Top‑level entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: float = 1000.0,
    **kwargs,
) -> List[float]:
    """
    Return a non‑negative height vector that (approximately) minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)  # tiny safety margin

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Initialise from any previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cur = 2.0 * cand.size * _conv_self(
                    cand, _fft_len_for_len(2 * cand.size - 1)
                ).max()
                if cur < best_score:
                    best_score, best_vec = cur, cand.copy()
                    print(f"[init] Candidate from globals score={cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to explore – mix of small, medium and large vectors
    # ------------------------------------------------------------------
    lengths = [1500, 2500, 3809, 5000, 7500, 10000]

    # ------------------------------------------------------------------
    # Phase 1 – many cheap restarts (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    while time.monotonic() < phase1_deadline:
        n = int(rng.choice(lengths))
        a0 = rng.random(n)
        a0 /= a0.sum()
        a_opt, score_opt = _local_optimise(a0, rng, phase1_deadline, max_iters=250)
        if score_opt < best_score:
            best_score, best_vec = score_opt, a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] New best after short run (n={n}): {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Phase 2 – intensive refinement of the current global best
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        if best_vec is None:
            # fallback: uniform distribution of a random length
            n = int(rng.choice(lengths))
            best_vec = np.full(n, 1.0 / n)

        # inject a mild log‑normal noise to escape plateaus
        noise_scale = 0.02
        noisy = best_vec * np.exp(rng.normal(0.0, noise_scale, size=best_vec.size))
        noisy = np.clip(noisy, 0.0, None)
        if noisy.sum() == 0.0:
            noisy = best_vec.copy()
        else:
            noisy /= noisy.sum()

        # longer run – up to a few thousand iterations (time dependent)
        remaining = deadline - time.monotonic()
        # allow more iterations when we still have plenty of time
        max_iters = int(min(8000, max(2000, remaining * 8)))  # approximate 0.125 it/s
        a_opt, score_opt = _local_optimise(noisy, rng, deadline, max_iters=max_iters)

        if score_opt < best_score:
            best_score, best_vec = score_opt, a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] Global improvement (n={best_vec.size}): {best_score:.6f}"
            )
            # after a good improvement we briefly re‑run a pure‑additive
            # refinement to polish the solution
            a_polished, score_polished = _local_optimise(
                best_vec.copy(), rng, deadline, max_iters=1500
            )
            if score_polished < best_score:
                best_score, best_vec = score_polished, a_polished.copy()
                elapsed = time.monotonic() - start_time
                print(
                    f"[{elapsed:6.2f}s] Polished after improvement: {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also clips, but we do it here)
    # ------------------------------------------------------------------
    if best_vec is None:
        # should never happen – fall back to a uniform vector of length 2000
        n = 2000
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
