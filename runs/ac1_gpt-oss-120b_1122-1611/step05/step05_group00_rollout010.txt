def _project_simplex(v):
    """Projects v onto the probability simplex (non-negative sum=1)."""
    v = np.asarray(v, dtype=np.float64)
    if v.sum() == 1 and np.all(v >=0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

*   A larger default length (≈8000) gives more flexibility.
*   Fast FFT‑based self‑convolution supplies the objective.
*   Each iteration computes a soft‑max over the near‑max convolution
    entries (active set) and builds a sub‑gradient of the L∞‑objective.
*   Two complementary updates are used:
        – multiplicative (exponentiated‑gradient) mirror descent,
        – additive projected‑gradient descent on the probability simplex.
    The update type is chosen randomly each step.
*   Momentum stabilises the gradient estimate.
*   A schedule slowly shrinks the learning‑rate η and raises the
    soft‑max temperature β, so the method initially explores and then
    focuses on the true max.
*   Light random perturbations (block scaling, pairwise mass transfer)
    keep the search from getting stuck.
*   If no improvement occurs for ``restart_gap`` seconds the algorithm
    restarts from the current best with a small log‑normal jitter.
*   Several analytic initialisations (sqrt‑difference, sinusoidal,
    triangular, Dirichlet) are tried and the one with the smallest
    initial score is taken as start.

The routine runs until the time budget expires (≈1000 s) and returns the
best sequence found, clipped to the range required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex:
#   {x | x_i >= 0, sum_i x_i = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    # Fast path for already feasible vectors
    if v.sum() == 1.0 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]                 # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise if needed
    s = w.sum()
    if s == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w = w / s
    return w


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    evaluate_sequence before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size ------------------------------------------------
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 8000))
    if n < 2:
        n = 2

    # --------------------------------------------------------------
    # 2) Build a collection of plausible initial vectors ------------
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # (a) sqrt‑difference (the classical analytic start)
    a_sqrt = np.sqrt(i + 1.0) - np.sqrt(i)
    a_sqrt = a_sqrt / a_sqrt.sum()

    # (b) smooth sinusoidal shape (non‑negative half‑wave)
    a_sin = np.sin(np.pi * (i + 0.5) / n)
    a_sin = np.clip(a_sin, 0.0, None)
    a_sin = a_sin / a_sin.sum()

    # (c) symmetric triangular shape
    a_tri = np.minimum(i + 1.0, n - i)
    a_tri = a_tri / a_tri.sum()

    # (d) random Dirichlet sample (full support)
    a_rand = rng.dirichlet(np.ones(n))

    # (e) possibly reuse a previously found construction
    init_candidates = [a_sqrt, a_sin, a_tri, a_rand]
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                    # adjust length if necessary
                    if cand.size != n:
                        if cand.size > n:
                            cand = cand[:n]
                        else:
                            cand = np.concatenate([cand, np.zeros(n - cand.size)])
                    init_candidates.append(cand)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3) Pick the best initial vector (fast evaluation) ------------
    # --------------------------------------------------------------
    fft_len_tmp = _fft_len_for_n(n)
    best_init = None
    best_init_score = np.inf
    for cand in init_candidates:
        # safety: ensure non‑negative and normalised
        cand = np.clip(cand, 0.0, None)
        sc = cand.sum()
        if sc == 0.0:
            continue
        cand = cand / sc
        cur_score = 2.0 * n * np.max(_conv_self(cand, fft_len_tmp))
        if cur_score < best_init_score:
            best_init_score = cur_score
            best_init = cand.copy()

    if best_init is None:
        # Fallback to sqrt‑difference if everything failed
        best_init = a_sqrt.copy()
        best_init_score = 2.0 * n * np.max(_conv_self(best_init, fft_len_tmp))

    # --------------------------------------------------------------
    # 4) Book‑keeping ------------------------------------------------
    # --------------------------------------------------------------
    a = best_init.copy()                     # current iterate (sum = 1)
    fft_len_conv = _fft_len_for_n(n)

    best = a.copy()
    best_score = best_init_score
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # tiny safety margin
    last_improve = start_time

    # --------------------------------------------------------------
    # 5) Hyper‑parameters -------------------------------------------
    # --------------------------------------------------------------
    eta_start = 1.0          # start learning‑rate (additive & multiplicative)
    eta_end = 0.001          # final learning‑rate
    beta_start = 0.5         # start temperature for the soft‑max on conv
    beta_end = 5000.0        # very sharp soft‑max (almost true max)
    block_perturb_prob = 0.025
    pair_move_prob = 0.08
    restart_gap = 150.0      # seconds without improvement before a restart
    momentum = 0.9

    grad_prev = np.zeros_like(a)

    # --------------------------------------------------------------
    # 6) Main optimisation loop ---------------------------------------
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- current convolution & score -------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----- keep the best solution -------------------------------
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- scheduling --------------------------------------------
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ----- active set (indices close to the maximum) ------------
        # use a very tight relative tolerance (0.05 % of the maximum)
        tol = cur_max * 0.9995
        active = np.where(conv >= tol)[0]
        if active.size == 0:
            # fallback – the argmax is definitely active
            active = np.array([int(np.argmax(conv))], dtype=np.int64)

        # keep at most 20 largest indices to bound work
        if active.size > 20:
            active = np.argsort(conv)[-20:]

        # ----- compute soft‑max weights over the active set ----------
        # (more weight for larger values)
        exp_vals = np.exp(beta * (conv[active] - cur_max))
        w_sum = exp_vals.sum()
        if w_sum > 0.0:
            w = exp_vals / w_sum
        else:
            w = np.full_like(exp_vals, 1.0 / exp_vals.size)

        # ----- gradient of the (soft) max ---------------------------
        grad = np.zeros_like(a)
        for idx, weight in zip(active, w):
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            # vector of indices i in [lo, hi]
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution: 2 * a_{k-i}
            grad[i_range] += weight * 2.0 * a[idx - i_range]

        # ----- momentum -----------------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- choose update type (multiplicative or additive) -------
        if rng.random() < 0.5:
            # * multiplicative (mirror‑descent) *
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s
        else:
            # * additive (projected gradient) *
            a_tmp = a - eta * grad
            a = _project_simplex(a_tmp)

        # ----- occasional random block perturbation ------------------
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor_blk = np.exp(rng.normal(0.0, 0.15))
            a[start:start + width] *= factor_blk
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # ----- occasional pairwise mass transfer --------------------
        if rng.random() < pair_move_prob:
            i_idx, j_idx = rng.integers(0, n, size=2)
            if i_idx != j_idx and a[i_idx] > 1e-15:
                delta = rng.random() * a[i_idx] * 0.5
                a[i_idx] -= delta
                a[j_idx] += delta
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

        # ----- soft restart if stagnated ----------------------------
        if now - last_improve > restart_gap:
            # jitter the current best to escape a flat region
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (respect the clipping performed by
    #    evaluate_sequence).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
```
