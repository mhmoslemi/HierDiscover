"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Warm‑up (≈55 % of the budget)**  
   * Mirror‑descent on the probability simplex using an
     exponentiated‑gradient (Adam‑style) update.
   * The L∞‑norm in the objective is approximated by a soft‑max
     with a temperature β that grows from a small value to a very large
     one, so the landscape becomes increasingly sharp.
   * After each gradient step the vector is projected onto the simplex.
   * Occasional multiplicative block perturbations and a “stagnation
     jitter” help escape plateaus.

2. **Simulated‑annealing refinement (≈45 % of the budget)**  
   * The current best vector is used as the seed.
   * Two types of moves are tried:
        – Exact pairwise mass transfer – O(n) update of the self‑convolution.
        – Multiplicative block perturbation.
   * Moves that lower the score are always accepted; uphill moves are
     accepted with Metropolis probability using a linear temperature
     schedule.
   * The convolution is kept up‑to‑date with an O(n) incremental update,
     occasional full FFT recomputation prevents drift.

3. **Return the best sequence** – clipped to the limits enforced by
   ``evaluate_sequence`` (0 … 1000).

The code respects the 1000 s budget, prints progress, and uses only
NumPy (no external solvers).  It works both when a previous solution
is available via the global ``height_sequence_1`` variable and when it
needs to start from scratch.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (expects x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # remove tiny negatives
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements  grad_j = 2·Σ_i a_i·w_{i+j}
    as a correlation a ⋆ w using a single FFT.
    """
    n = a.size
    Lw = w.size                       # = 2*n‑1
    wr = w[::-1]                      # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    corr_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    # keep only needed part (size n)
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * corr_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / len(v)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _pairwise_transfer_exact(a: np.ndarray,
                             conv: np.ndarray,
                             rng: np.random.Generator,
                             max_delta_frac: float = 0.3) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Exact O(n) update for moving a random amount of mass from a donor index i
    to a receiver index j.
    Returns (new_a, new_conv, new_score).  The move is always performed;
    the calling code decides whether to accept it.
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if i == j:
        return a, conv, 2.0 * n * conv.max()

    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        return a, conv, 2.0 * n * conv.max()
    delta = rng.random() * max_delta

    # Updated convolution (linear in delta)
    new_conv = conv.copy()
    # Remove donor contribution
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    # Add receiver contribution
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a
    # Diagonal corrections (+delta²)
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    # Cross‑term correction (−2·delta²) for i≠j
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    # Update a
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)

    new_score = 2.0 * n * new_conv.max()
    return a2, new_conv, new_score

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best coefficient list found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Problem size and initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = 4000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(2 * n - 1)          # length for self‑convolution FFT

    # ------------------------------------------------------------------
    # 1) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _self_conv_fft(best, fft_len)
    best_score = 2.0 * n * conv_best.max()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep 5 s margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 2) Hyper‑parameters
    # ------------------------------------------------------------------
    warmup_frac = 0.55                     # portion of time for gradient phase
    # Adam‑style exponentiated‑gradient parameters
    lr_start = 1.0
    lr_end = 0.02
    beta_start = 0.2
    beta_end = 8000.0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    m = np.zeros_like(a)                   # first moment
    v = np.zeros_like(a)                   # second moment
    t_iter = 0

    block_perturb_prob = 0.025
    jitter_interval = 90.0                 # seconds without improvement → jitter

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # --------------------------------------------------------------
        # Phase decision
        # --------------------------------------------------------------
        if frac < warmup_frac:
            # ------------------- Warm‑up: Adam‑mirrored gradient -----
            conv = _self_conv_fft(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] warm‑up new best = {best_score:.6f}")

            # schedules
            lr = lr_start * (1.0 - frac) + lr_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # soft‑max (stable) weights
            exp_terms = np.exp(beta * (conv - cur_max))
            sum_exp = exp_terms.sum()
            if sum_exp > 0:
                w = exp_terms / sum_exp
            else:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

            # gradient of the smoothed max
            grad = _gradient_from_weights(a, w)

            # Adam moments
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            # exponentiated‑gradient (mirror descent) step
            factor = np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps))
            # clip exponent to avoid under/overflow
            factor = np.exp(np.clip(np.log(factor), -15, 15))
            a = a * factor
            a = np.clip(a, 1e-15, None)          # keep a tiny positive mass
            a = _project_onto_simplex(a)

            # occasional block perturbation
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # stagnation jitter
            if now - last_improve > jitter_interval:
                noise = np.exp(rng.normal(0.0, 0.25, size=n))
                a = a * noise
                a = _project_onto_simplex(a)
                m.fill(0.0)
                v.fill(0.0)
                t_iter = 0
                last_improve = now
                print(f"[{elapsed:6.2f}s] jitter reset – best {best_score:.6f}")

        else:
            # ------------------- Simulated annealing refinement -----
            # Initialise refinement structures lazily
            if 'ref_a' not in locals():
                ref_a = best.copy()
                ref_conv = _self_conv_fft(ref_a, fft_len)
                cur_score = best_score
                # temperature schedule (linear)
                temp_start = best_score * 0.4
                temp_end = best_score * 1e-5
                ref_start_time = now
                ref_deadline = deadline
                last_ref_improve = now

            # Linear temperature
            elapsed_ref = now - ref_start_time
            total_ref = ref_deadline - ref_start_time
            tmp = max(0.0, 1.0 - elapsed_ref / total_ref)
            T = temp_start * tmp + temp_end * (1.0 - tmp)

            # Choose a move
            if rng.random() < 0.65:                                  # pairwise transfer
                new_a, new_conv, new_score = _pairwise_transfer_exact(
                    ref_a, ref_conv, rng, max_delta_frac=0.45)
                # Metropolis acceptance
                if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                    ref_a, ref_conv, cur_score = new_a, new_conv, new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best = ref_a.copy()
                        last_improve = now
                        print(f"[{elapsed:6.2f}s] SA improved to {best_score:.6f}")
            else:                                                    # block perturb
                cand = _random_block_perturb(ref_a, rng)
                cand_conv = _self_conv_fft(cand, fft_len)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
                    ref_a, ref_conv, cur_score = cand, cand_conv, cand_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best = ref_a.copy()
                        last_improve = now
                        print(f"[{elapsed:6.2f}s] SA block pert improved to {best_score:.6f}")

            # Periodic *full* recomputation to avoid numerical drift
            if (int(elapsed * 10) % 97) == 0:    # roughly every ~10 s
                ref_conv = _self_conv_fft(ref_a, fft_len)
                cur_score = 2.0 * n * ref_conv.max()

    # ------------------------------------------------------------------
    # 4) Return best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # final sanity – ensure sum>0.01 (required by evaluate_sequence)
    if best.sum() < 0.01:
        best = np.full_like(best, 1.0 / best.size)
    return [float(x) for x in best]
