"""
Propose a step‐function minimizing the evaluation metric.

Algorithm – three stages (≈1000 s total)

1.  Warm‑up: exponentiated‑gradient (mirror descent) on the simplex.
    The true L∞‑norm of the self‑convolution is approximated by a
    soft‑max whose temperature β grows from a modest value to a very
    large one.  Momentum stabilises the updates.  Occasional random
    block perturbations inject global noise and a “jitter reset’’ helps
    escape stagnation.

2.  Targeted simulated‑annealing (SA):
    * Keep the full self‑convolution vector and update it in O(n) time
      when moving an amount δ from a donor index i to a receiver j.
    * The donor is chosen as the index that contributes most to the
      current maximal convolution entry; the receiver is taken among the
      weakest contributors.  This focuses the search on the actual
      bottleneck.
    * Moves that strictly lower the maximal convolution are always
      accepted; uphill moves are accepted with Metropolis probability
      according to a linear temperature schedule.

3.  Occasional block perturbations and a final projection onto the
    simplex guarantee feasibility (non‑negative, sum = 1).  The best
    sequence seen during the whole run is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negative values caused by floating‑point noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Derivation: grad_j = 2·Σ_i a_i·w_{i+j}.
    Implemented as a correlation a ⋆ w using a single FFT.
    """
    n = a.size
    Lw = w.size                     # = 2*n‑1
    wr = w[::-1]                    # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)   # length for convolution a ⋆ wr

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)         # mapping to obtain correlation result
    grad = 2.0 * conv_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to counter floating‑point drift.
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)

    return _project_onto_simplex(out)


def _apply_mass_move(a: np.ndarray, conv: np.ndarray,
                     i: int, j: int, delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Exact O(n) update when moving `delta` mass from index `i` to `j`.
    Returns the updated coefficient vector and self‑convolution.
    """
    n = a.size
    new_conv = conv.copy()

    # donor contribution removed
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a

    # receiver contribution added
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a

    # diagonal quadratic corrections
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta

    # cross‑term second‑order correction (i≠j)
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    # update the coefficient vector
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)

    return a2, new_conv


def _targeted_move(a: np.ndarray, conv: np.ndarray,
                   rng: np.random.Generator,
                   max_delta_frac: float = 0.4) -> tuple[np.ndarray, np.ndarray, bool]:
    """
    Proposes a move that reduces the current maximal convolution entry.
    * Finds the index k_max where conv is largest.
    * Chooses a donor i that contributes most to this entry.
    * Chooses a receiver j that contributes least.
    * Moves a random fraction of mass from i to j.
    Returns (new_a, new_conv, accepted_flag).
    """
    n = a.size
    k_max = int(conv.argmax())

    # contributions of each a_i to the maximal entry
    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    contrib = np.zeros(n, dtype=a.dtype)
    contrib[mask] = a[mask] * a[idx[mask]]

    donor = int(np.argmax(contrib))

    # pick a receiver among the weakest contributors (bottom 15 %)
    low_thr = np.percentile(contrib, 15)
    low_candidates = np.where(contrib <= low_thr)[0]
    if low_candidates.size == 0:
        # fallback to a random index ≠ donor
        receiver = rng.integers(0, n)
        while receiver == donor:
            receiver = rng.integers(0, n)
    else:
        receiver = int(rng.choice(low_candidates))
        if receiver == donor:
            # very unlikely, pick another low candidate
            alt = low_candidates[low_candidates != donor]
            if alt.size == 0:
                receiver = (donor + 1) % n
            else:
                receiver = int(rng.choice(alt))

    max_delta = min(a[donor], max_delta_frac * a[donor])
    if max_delta <= 0.0:
        return a, conv, False

    delta = rng.random() * max_delta

    a2, conv2 = _apply_mass_move(a, conv, donor, receiver, delta)

    # strict improvement test
    if conv2.max() < conv.max() - 1e-15:
        return a2, conv2, True
    else:
        return a, conv, False

# Main optimisation routine

def propose_candidate(seed: int = 1234,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence`` within the allotted time budget.
    Returns the best coefficient list found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size and initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = 4000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # small random jitter to avoid exact symmetry
    a *= np.exp(rng.normal(0.0, 0.08, size=n))
    a = _project_onto_simplex(a)

    fft_len_self = _next_pow2(2 * n - 1)          # for a*a

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _self_conv_fft(a, fft_len_self)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best = a.copy()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    phase_split = 0.58                 # fraction of time for gradient phase
    # Gradient‑descent schedules
    eta_start = 1.0
    eta_end = 0.02
    beta_start = 0.2
    beta_end = 25000.0
    momentum = 0.93
    block_perturb_prob = 0.025

    # Simulated‑annealing parameters (phase 2)
    temp_start = best_score * 0.12     # initial temperature (in score units)
    temp_end = best_score * 1e-8
    max_delta_frac = 0.4               # maximal transferable fraction per move
    accept_prob_target = 0.25          # used only for occasional aggressiveness

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # --------------------------------------------------------------
        # Phase decision
        # --------------------------------------------------------------
        if frac < phase_split:
            # ------------------- Smooth‑max Gradient Descent ---------------
            conv = _self_conv_fft(a, fft_len_self)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best (grad) = {best_score:.6f}")

            # schedules (linear interpolation)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # soft‑max weights (stable)
            exp_terms = np.exp(beta * (conv - cur_max))
            sum_exp = exp_terms.sum()
            if sum_exp > 0:
                w = exp_terms / sum_exp
            else:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

            grad = _gradient_from_weights(a, w)
            # momentum
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient update (mirror descent on simplex)
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 1e-15, None)
            a = _project_onto_simplex(a)

            # occasional random block perturbation
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # stagnation escape – jitter reset
            if now - last_improve > 90.0:
                jitter = np.exp(rng.normal(0.0, 0.3, size=n))
                a = a * jitter
                a = _project_onto_simplex(a)
                grad_prev = np.zeros_like(a)
                last_improve = now
                print(f"[{elapsed:6.2f}s] jitter reset – best {best_score:.6f}")

        else:
            # ------------------- Targeted Simulated Annealing -----------------
            # Initialise SA structures on first entry to this phase
            if 'sa_state' not in locals():
                # use the best vector found so far as the starting point
                a = best.copy()
                conv = _self_conv_fft(a, fft_len_self)
                cur_max = conv.max()
                cur_score = 2.0 * n * cur_max
                best_score = cur_score
                best = a.copy()
                # temperature schedule (linear)
                sa_start = now
                sa_deadline = deadline
                sa_last_improve = now

            # linear temperature schedule
            sa_elapsed = now - sa_start
            sa_total = sa_deadline - sa_start
            temp = temp_start * (1.0 - sa_elapsed / sa_total) + temp_end * (sa_elapsed / sa_total)

            # ---- attempt a targeted move ----
            a2, conv2, improved = _targeted_move(a, conv, rng,
                                                 max_delta_frac=max_delta_frac)

            new_max = conv2.max()
            delta_score = 2.0 * n * (new_max - cur_max)   # change in evaluation score

            accept = False
            if improved:
                accept = True
            else:
                # uphill move – Metropolis test
                if delta_score > 0.0:
                    prob = math.exp(-delta_score / max(temp, 1e-14))
                    if rng.random() < prob:
                        accept = True
                else:
                    accept = True    # tie or very tiny improvement

            if accept:
                a = a2
                conv = conv2
                cur_max = new_max
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    sa_last_improve = now
                    print(f"[{elapsed:6.2f}s] SA improved → {best_score:.6f}")

            # occasional larger block perturbation (helps escape plateaus)
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                conv = _self_conv_fft(a, fft_len_self)
                cur_max = conv.max()
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    sa_last_improve = now
                    print(f"[{elapsed:6.2f}s] SA block perturb → {best_score:.6f}")

            # if no improvement for a while, inject a stronger jitter
            if now - sa_last_improve > 120.0:
                jitter = np.exp(rng.normal(0.0, 0.35, size=n))
                a = a * jitter
                a = _project_onto_simplex(a)
                conv = _self_conv_fft(a, fft_len_self)
                cur_max = conv.max()
                cur_score = 2.0 * n * cur_max
                sa_last_improve = now
                print(f"[{elapsed:6.2f}s] SA jitter reset – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
