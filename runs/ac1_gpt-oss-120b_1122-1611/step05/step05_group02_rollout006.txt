"""
Propose a step‑function by minimizing the worst self‑convolution value.

The algorithm combines three ideas:

1. **Smooth‑max mirror descent** – the non‑smooth objective
   max(conv) is approximated by a soft‑max with temperature β.
   Its gradient (computed analytically) is used in an exponentiated‑gradient
   (mirror‑descent) update while keeping the vector on the probability
   simplex.

2. **Stochastic local moves** – random pairwise mass transfers are
   accepted if they improve the true objective or with a Metropolis
   probability based on a cooling temperature schedule.

3. **Occasional block perturbations and soft restarts** – to escape
   plateaus the algorithm occasionally multiplies a random contiguous
   block by a log‑normal factor and, after a period without improvement,
   restarts from the current best with a small jitter.

All updates keep the sequence non‑negative and normalised (sum = 1).  The
best vector found before the time budget expires is returned (clipped to
[0, 1000] as required by `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = len(a) + len(b) - 1
    fft_len = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)[:n]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return the (β‑softmax) weights w_i = exp(β·conv_i)/Σexp."""
    # shift by max for numerical stability
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    return exp_vals / exp_vals.sum()


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the β‑softmax approximation of max(conv) with respect to a.
    For a given weight vector w (sum to 1) the gradient is
        grad_j = Σ_i w_i a_{i-j}   +   a_j Σ_{i=j}^{j+n-1} w_i .
    The first term is obtained via a convolution of a with w reversed,
    the second via a sliding‑window sum of w.
    """
    n = a.size
    # --- sliding window sum S_j = Σ_{i=j}^{j+n-1} w_i --------------------
    cumsum = np.concatenate(([0.0], np.cumsum(w)))
    S = cumsum[n:] - cumsum[:n]          # shape (n,)

    # --- term T1_j = Σ_i w_i a_{i-j} (zero outside support) -------------
    w_rev = w[::-1]
    conv_rev = _conv_fft(a, w_rev)      # length = 3n-2
    # take the slice [n-1 : 2n-1) and reverse it → length n
    T1 = conv_rev[n - 1 : 2 * n - 1][::-1]

    # --- final gradient -------------------------------------------------
    grad = T1 + a * S
    return grad


def _score_from_vector(a: np.ndarray, n: int, fft_len_self: int) -> float:
    """Exact objective value for a (sum‑normalised to 1)."""
    conv = _conv_self_fft(a, fft_len_self)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Initialise the probability vector (normalised sum = 1)
    # --------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full(cand.size, 1.0 / cand.size, dtype=np.float64)
            else:
                raise ValueError
        except Exception:
            a = np.full(default_n, 1.0 / default_n, dtype=np.float64)
    else:
        a = np.full(default_n, 1.0 / default_n, dtype=np.float64)

    n = a.size
    fft_len_self = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    cur_score = _score_from_vector(a, n, fft_len_self)
    best = a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # --------------------------------------------------------------
    # 3) Hyper‑parameters
    # --------------------------------------------------------------
    eta_start = 2.0          # initial step size for exponentiated gradient
    eta_end = 1e-4           # final step size
    beta_start = 1.0         # start temperature for soft‑max
    beta_end = 200.0         # final temperature (more focussed on the max)
    momentum = 0.95
    block_perturb_prob = 0.02
    pair_move_prob = 0.07
    restart_gap = 15.0       # seconds without improvement → restart
    temp_start = 0.5
    temp_end = 1e-4

    grad_prev = np.zeros_like(a)

    iteration = 0
    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # ----- evaluate current vector and store best -----------------
        conv = _conv_self_fft(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule ------------------------------------------------
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac
        temp = temp_start * (1.0 - frac) + temp_end * frac

        # ----- smooth‑max weights and gradient -----------------------
        w = _softmax_weights(conv, beta)          # length 2n‑1, sum = 1
        grad = _gradient_smooth_max(a, w)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror descent) update ----------
        factor = np.exp(-eta * grad)
        a = a * factor
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----- occasional random block perturbation --------------------
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 8))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.4))
            a[start:start + width] *= factor
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # ----- occasional pair‑wise mass transfer -----------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                # move a random fraction (up to 30 % of a[i]) to j
                delta = rng.random() * min(a[i], 0.3 * a[i])
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                # re‑normalise (sum is preserved but rounding errors may occur)
                a_candidate = np.clip(a_candidate, 0.0, None)
                a_candidate = a_candidate / a_candidate.sum()
                cand_score = _score_from_vector(a_candidate, n, fft_len_self)
                if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(temp, 1e-12)):
                    a = a_candidate
                    # cur_score will be updated at top of the next loop

        # ----- local random moves (Metropolis style) -------------------
        # try a few cheap random moves each iteration
        for _ in range(3):
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = rng.random() * min(a[i], 0.2 * a[i])
            a_trial = a.copy()
            a_trial[i] -= delta
            a_trial[j] += delta
            a_trial = np.clip(a_trial, 0.0, None)
            a_trial = a_trial / a_trial.sum()
            trial_score = _score_from_vector(a_trial, n, fft_len_self)
            if trial_score < cur_score or rng.random() < np.exp((cur_score - trial_score) / max(temp, 1e-12)):
                a = a_trial
                cur_score = trial_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{time.monotonic() - start_time:6.2f}s] new best (local move) = {best_score:.6f}")

        # ----- soft restart when stuck ---------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter all coordinates multiplicatively
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (respect evaluate_sequence clipping)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best.tolist()]
