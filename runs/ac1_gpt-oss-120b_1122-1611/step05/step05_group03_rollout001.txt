"""
propose_candidate
-----------------

A Frank‑Wolfe style optimizer for the non‑negative step‑function problem.

The current evaluation function is

    E(a) = 2·n·max_k (a * a)_k / (sum_i a_i)^2 ,

where `*` is the linear convolution.  Because the expression is invariant
under positive scaling we restrict ourselves to the probability simplex
(sum = 1).  The algorithm iteratively improves a candidate `a` by solving
the following linear programme (the “direction sub‑problem”) :

    maximise    Σ_j g_j
    subject to  (a * g)_k ≤ M      for all k = 0 … 2·n‑2
                g_j ≥ 0

where `M = max_k (a * a)_k`.  The LP finds the feasible vector with the
largest ℓ₁‑norm under the current convolution bound.  Its ℓ₁‑norm is
always larger than 1 (the current `a` itself is feasible), so we simply
normalise the LP solution to a probability vector `g`.

After obtaining `g` we perform a short line‑search over a small set of
mixing parameters `t ∈ (0,1]` and replace

    a ← (1‑t)·a + t·g

when the objective improves.  Occasionally a random jitter is added or a
restart is performed if no improvement is seen for a while.

The LP is built with a sparse Toeplitz matrix that encodes the
convolution constraints, and solved with SciPy’s HiGHS backend.  The
method works well for lengths up to a few thousand; we use a moderate
size (default ≈ 1500) which gives a good trade‑off between solution
quality and runtime.  The function respects the 1000 s time budget and
returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse


def _build_constraint_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the sparse matrix A (shape (2*n‑1, n)) that implements the
    linear constraints   (f * g)_k ≤ M .
    For a fixed `f`, the entry A[k, j] = f[k‑j]  if 0 ≤ k‑j < n,
    otherwise 0.  The pattern of non‑zero entries is the same for every
    iteration; only the data values (the copies of `f`) change.
    """
    n = f.size
    K = 2 * n - 1                       # length of the full convolution
    # Row indices: for each column j we have rows j … j+n‑1
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    # Column indices: each column index j repeated n times
    cols = np.repeat(np.arange(n), n)
    # Data: each column carries a copy of the current `f`
    data = np.tile(f, n)
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))


def _solve_direction_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP   maximise Σ g_j   subject to  (f * g)_k ≤ M , g ≥ 0 .
    Returns the raw solution `g_raw` (not normalised) or None on failure.
    """
    n = f.size
    K = 2 * n - 1

    # Build constraint matrix A (sparse)
    A = _build_constraint_matrix(f)

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Right‑hand side: M for every convolution entry
    b = np.full(K, M, dtype=np.float64)

    # Bounds: g_j ≥ 0
    bounds = [(0, None) for _ in range(n)]

    # Solve with HiGHS (fast and supports sparse matrices)
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        return None
    return res.x


def _line_search(a: np.ndarray, g: np.ndarray, cur_score: float,
                 start: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Try a small set of mixing coefficients `t` and return the best
    (a_new, score_new) pair that improves upon `cur_score`.  If none
    improves, the original pair is returned unchanged.
    """
    # Candidate step sizes – larger steps early, finer later.
    candidates = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    best_a = a
    best_score = cur_score
    for t in candidates:
        a_new = (1.0 - t) * a + t * g
        score_new = evaluate_sequence(a_new.tolist())
        if score_new < best_score:

            best_score = score_new
            # Small print for progress tracking
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] t={t:.3f} → score {best_score:.6f}")
    return best_a, best_score


def _random_jitter(a: np.ndarray, rng: np.random.Generator,
                   sigma: float = 0.25) -> np.ndarray:
    """
    Apply multiplicative log‑normal noise to every entry and re‑normalise.
    Used for restarts when the search stalls.
    """
    noisy = a * np.exp(rng.normal(0.0, sigma, size=a.shape))
    noisy = np.clip(noisy, 0.0, None)
    s = noisy.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.size)
    return noisy / s
