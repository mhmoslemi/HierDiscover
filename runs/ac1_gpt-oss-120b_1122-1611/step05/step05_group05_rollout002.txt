def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 0.5)

    # Initialize a
    if "height_sequence_1" in globals():
        try:
            a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a.ndim != 1 or a.size == 0:
                raise Exception
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = 1024
        a = np.full(n, 1.0 / n)
    else:
        n = a.size
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    # set best
    conv = np.convolve(a, a)
    best_score = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
    best = a.copy()
    last_improve = time.monotonic()
    start = time.monotonic()
    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    step_size = 0.05   # magnitude of gradient step (in L2 direction)
    block_perturb_prob = 0.4
    grad_step_prob = 0.5
    transfer_prob = 0.1
    restart_gap = 20.0

    # tolerance for active conv indices
    epsilon_active = 1e-8

    while time.monotonic() < deadline:
        rand = rng.random()
        if rand < block_perturb_prob:
            # random block perturbation as before
            cand = _random_block_perturb(a, rng)
        elif rand < block_perturb_prob + grad_step_prob:
            # gradient step
            conv = np.convolve(a, a)
            M = np.max(conv)
            # active indices within tolerance
            tol = max(epsilon_active, 1e-6 * M)
            active = np.where(conv >= M - tol)[0]
            grad = np.zeros_like(a)
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1)
                j = k - idx
                # vectorized addition
                grad[idx] += 2.0 * a[j]
            # Normalize gradient direction (avoid zero)
            norm = np.linalg.norm(grad)
            if norm > 0:
                direction = grad / norm
                cand = a - step_size * direction
                # enforce non-negative then project
                cand = np.clip(cand, 0.0, None)
                cand = _project_simplex(cand)

                    sym = (cand + cand[::-1]) / 2.0
                    sym = _project_simplex(sym)

                    sym_score = 2.0 * n * np.max(conv_sym) / (np.sum(sym) ** 2)

                    cand_score = 2.0 * n * np.max(conv_cand) / (np.sum(cand) ** 2)
                    if sym_score <= cand_score:
                        cand = sym
                # update step size adaptively
                # we will evaluate later using cand_score
            else:
                cand = a.copy()
        else:
            # transfer mass between two indices (pairwise move)
            # Determine indices involved in current max conv
            conv = np.convolve(a, a)
            M = np.max(conv)
            tol = max(epsilon_active, 1e-6 * M)
            active = np.where(conv >= M - tol)[0]
            involved = set()
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):

"""
Hybrid stochastic optimizer for the “step‑function” bound.
The algorithm mixes four simple moves:

* multiplicative block perturbations (as in the baseline code);
* a projected sub‑gradient step that pushes mass away from the indices
  that realise the current maximal autocorrelation;
* a swap / relocation move that reorders heavy and light heights;
* a tiny pair‑wise transfer of mass.

All candidates are projected onto the probability simplex
{ x ≥ 0 , Σ x = 1 } after the move, keeping the objective
invariant under scaling.  The best sequence seen so far is kept,
and a simple restart with multiplicative jitter of the best vector
is performed when no improvement occurs for a while.  The routine
obeys the supplied time budget (default 1000 s) and returns the
best list of heights it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 5))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def _score(a: np.ndarray) -> float:
    """
    Manual computation of the objective used by ``evaluate_sequence``.
    The vector a is assumed non‑negative and its sum is not zero.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _gradient_step(a: np.ndarray, rng: np.random.Generator, step_sz: float) -> np.ndarray:
    """
    One projected sub‑gradient step.
    The sub‑gradient is the sum of the gradients of all (almost) active
    convolution entries; the step is taken opposite to the gradient,
    then the result is projected back onto the simplex.
    Occasionally the candidate is symmetrised if it helps.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = np.max(conv)
    # indices that are practically active
    tol = max(1e-8, 1e-6 * M)
    active = np.where(conv >= M - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        j = k - idx
        grad[idx] += 2.0 * a[j]

    norm = np.linalg.norm(grad)
    if norm == 0.0:
        cand = a.copy()
    else:
        direction = grad / norm
        cand = a - step_sz * direction
        cand = np.clip(cand, 0.0, None)
        cand = _project_simplex(cand)

    # occasional symmetrisation (sometimes helps to flatten the autocorrelation)
    if rng.random() < 0.2:
        sym = (cand + cand[::-1]) / 2.0
        sym = _project_simplex(sym)
        if _score(sym) <= _score(cand):
            cand = sym
    return cand


def _swap_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reorder the vector by swapping a heavy and a light entry.
    The swap is accepted irrespective of the actual improvement – the
    outer loop decides whether to keep it.
    """
    n = a.shape[0]
    # define “heavy’’ and “light’’ by percentiles
    high_thr = np.percentile(a, 80)
    low_thr = np.percentile(a, 20)

    heavy = np.where(a > high_thr)[0]
    light = np.where(a < low_thr)[0]

    if heavy.size and light.size and rng.random() < 0.7:
        i = rng.choice(heavy)
        j = rng.choice(light)
    else:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

    cand = a.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def _transfer_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Transfer a small amount of mass from a relatively large entry to a
    relatively small one.  The step size is chosen adaptively.
    """
    n = a.shape[0]
    # pick donor among the top 15 % and recipient among the bottom 15 %
    threshold_hi = np.percentile(a, 85)
    threshold_lo = np.percentile(a, 15)

    donors = np.where(a >= threshold_hi)[0]
    recipients = np.where(a <= threshold_lo)[0]

    if donors.size and recipients.size:
        donor = rng.choice(donors)
        recipient = rng.choice(recipients)
    else:
        donor = np.argmax(a)
        recipient = np.argmin(a)

    # move at most 20 % of the donor’s mass, capped at 0.01
    delta = min(a[donor] * 0.2, 0.01)
    cand = a.copy()
    cand[donor] -= delta
    cand[recipient] += delta
    cand = _project_simplex(cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  A hybrid of block perturbations,
    projected sub‑gradient steps, swaps and tiny mass transfers is
    employed.  The best vector seen before the time budget expires is
    returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # 1)  Starting vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = 1024                 # a reasonable default size
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s               # normalise to sum = 1

    # ------------------------------------------------------------------
    # 2)  Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = _score(best)

    start_time = time.monotonic()
    last_improve = start_time

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3)  Parameters
    # ------------------------------------------------------------------
    step_size = 0.08                 # size of the gradient move
    restart_gap = 20.0               # seconds without improvement → restart
    # relative probabilities of the four move types
    prob_block = 0.35
    prob_grad  = 0.30
    prob_swap  = 0.20
    prob_xfer  = 0.15

    # ------------------------------------------------------------------
    # 4)  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        r = rng.random()
        if r < prob_block:
            cand = _random_block_perturb(a, rng)
        elif r < prob_block + prob_grad:
            cand = _gradient_step(a, rng, step_size)
        elif r < prob_block + prob_grad + prob_swap:
            cand = _swap_move(a, rng)
        else:
            cand = _transfer_move(a, rng)

        cur_score = _score(cand)

        # ------------------------------------------------------------------
        # 5)  Acceptance logic
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cand.copy()
            a = cand.copy()
            last_improve = time.monotonic()
            step_size = min(step_size * 1.15, 0.25)     # reward improvement
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")
        else:
            # simulated‑annealing style acceptance for worse moves
            now = time.monotonic()
            elapsed = now - start_time
            frac = min(1.0, elapsed / (budget_s - 0.5))
            # probability falls from ~0.12 to ~0.001
            accept_prob = 0.12 * (1.0 - frac) + 0.001 * frac
            if rng.random() < accept_prob:
                a = cand.copy()
                # slightly shrink the step size after an unsuccessful move
                step_size = max(step_size * 0.9, 0.001)

        # ------------------------------------------------------------------
        # 6)  Restart if we are stuck
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > restart_gap:
            # jitter the current best – multiplicative noise of modest amplitude
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = best * noise
            a = _project_simplex(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7)  Return the best vector seen
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
