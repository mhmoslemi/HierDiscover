def propose_candidate(seed=42, budget_s=1000.0):
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)

    best_seq = None
    best_score = float('inf')

    # init candidates
    # Check baseline global var
    # arcsine candidates: for n in [1024, 2048, 4096, 8192] maybe, for alpha in some range using fine steps.

    def normalized(seq):
        total = sum(seq)
        if total <= 0:
            return seq
        return [x/total for x in seq]

    def evaluate_and_update(seq):
        nonlocal best_seq, best_score
        sc = evaluate_sequence(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best score = {sc:.6f} (len {len(seq)})")
        return sc

    # 0 baseline
    if "height_sequence_1" in globals():

    alphas = np.linspace(0.3, 0.7, 9)   # 0.3 to 0.7 step 0.05
    for n in lengths:
        if time.monotonic() > deadline:
            break
        i = np.arange(n, dtype=np.float64)

"""
Propose a step‑function with a low evaluation score.

The algorithm works in three phases:

1. **Baseline construction** –  evaluate any existing global solution and
   generate a family of “discrete arcsine’’ candidates for several lengths
   and exponents  α .  The best of these is kept as a starting point.

2. **Soft‑max mirror descent** –  treat the objective
   ``2 n·max(conv)/sum²`` as scale‑invariant and minimise the
   smooth surrogate  ε·log∑exp(conv/ε) .
   Its gradient w.r.t. the coefficients a is
   ``grad = 2·(w * a_rev)`` where ``w`` is the soft‑max of the
   current self‑convolution and ``a_rev`` is a reversed copy of ``a``.
   An exponentiated‑gradient (mirror‑descent on the simplex) update
   ``a←a·exp(‑η·grad)`` is performed, followed by normalisation.
   The temperature ε and step size η are slowly reduced, and occasional
   random jitter is added to escape plateaus.

3. **Polishing** –  evaluate the current best sequence periodically.
   When the time budget (≈ 1000 s) elapses the best sequence found so far
   is returned.  All numbers are clipped to the interval [0, 1000] as
   required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a using FFT (real‑valued)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of x and y using FFT of length fft_len."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv

def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.

    Parameters
    ----------
    seed:
        Random seed for reproducibility.
    budget_s:

    Returns
    -------

    """

    # Helper to evaluate a candidate and keep the best one

    def _maybe_update(seq: list[float]) -> None:

            return
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq

                  f"(length {len(seq)})")

    # ------------------------------------------------------------------
    # 0)  Baseline – use any previously stored solution

    if "height_sequence_1" in globals():

        a = a * np.exp(-eta * grad)

        if rng.random() < 0.005:

        else:

    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
