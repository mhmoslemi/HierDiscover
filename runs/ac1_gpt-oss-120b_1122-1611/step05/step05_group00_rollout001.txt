def gradient_of_active(a, conv, active, beta):
    # compute weights w via softmax on conv[active] - cur_max
    cur_max = conv.max()
    if len(active) == 0:
        return np.zeros_like(a)
    w_raw = np.exp(beta * (conv[active] - cur_max))
    w = w_raw / w_raw.sum()
    grad = np.zeros_like(a)
    n = len(a)
    for idx, weight in zip(active, w):
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi+1)
        grad[i_range] += weight * 2.0 * a[idx - i_range]

"""
Optimisation of a non‑negative step‑function for the objective

    score = 2·n·max_k (a∗a)_k / (∑a_i)^2

The problem is homogeneous, so we keep the vector on the probability
simplex (∑a_i = 1).  The algorithm consists of three stages:

1. **Initialisation** – start from the analytically known
   “√‑difference’’ shape (a_i = √(i+1)‑√i) or from a previously stored
   construction (global ``height_sequence_1``).

2. **Exploratory simulated‑annealing (SA)** – many cheap random
   mass‑transfer or block‑scaling moves are proposed.  A temperature
   schedule (exponential decay) and a Metropolis criterion allow the
   search to escape local minima while favouring improvements.  The
   convolution‐max (the only expensive operation) is evaluated with a
   single FFT per move.

3. **Deterministic polishing** – after the SA budget is exhausted,
   a few mirror‑descent / exponentiated‑gradient steps are performed
   on the true L∞‑objective using an active‑set of near‑maximal
   convolution indices.  This refines the best candidate found by SA.

The routine respects the 1000 s time budget, prints progress stamps and
returns the best sequence (clipped to the limits imposed by
``evaluate_sequence``).  It works for any length `n` (default 3800) and
uses only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helper utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real‑to‑real)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) – the objective when the vector is normalised."""
    return 2.0 * n * float(conv.max())


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a normalised vector a."""
    conv = _conv_self(a, fft_len)
    return _score_from_conv(conv, n)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _maybe_use_global() -> np.ndarray | None:
    """If a previous construction exists in the global namespace, return it normalised."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Polishing with exponentiated‑gradient (mirror‑descent)
# ----------------------------------------------------------------------
def _polish(a: np.ndarray, n: int, fft_len: int,
            deadline: float, start_score: float) -> tuple[np.ndarray, float]:
    """
    Perform a limited number of mirror‑descent steps on the true L∞‑objective.
    Returns (best_vector, best_score) found during polishing.
    """
    best = a.copy()
    best_score = start_score
    grad_prev = np.zeros_like(a)

    # hyper‑parameters for polishing
    eta_start = 0.12
    eta_end = 0.006
    beta_start = 50.0
    beta_end = 2000.0
    momentum = 0.85
    max_active = 20          # cap on active indices

    while time.monotonic() < deadline:
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # keep best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - (deadline - (deadline - start_time))
            print(f"[{elapsed:6.2f}s] polish: new best = {best_score:.6f}")

        # schedule
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # active set (indices within 0.2 % of the current maximum)
        tol = cur_max * 0.998
        active = np.where(conv >= tol)[0]
        if active.size > max_active:
            active = np.argsort(conv)[-max_active:]

        # soft‑max weights on the active set
        if active.size == 0:
            # fall back to full soft‑max (should never happen)
            w = np.exp(beta * (conv - cur_max))
            w /= w.sum()
            grad = _gradient_correlation(a, w)  # defined later
        else:
            w_raw = np.exp(beta * (conv[active] - cur_max))
            w = w_raw / w_raw.sum()
            grad = np.zeros_like(a)
            # gradient of (a∗a)_k is 2·a shifted by k
            for idx, weight in zip(active, w):
                lo = max(0, idx - (n - 1))
                hi = min(n - 1, idx)
                rng = np.arange(lo, hi + 1, dtype=np.int64)
                grad[rng] += weight * 2.0 * a[idx - rng]

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update and projection onto the simplex
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # stop early if time is very close
        if time.monotonic() > deadline - 0.1:
            break

    return best, best_score


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k (a∗a)_k .
    Implements the correlation a ⋆ w (see the original code).
    """
    n = a.size
    L = w.size                     # should be 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget elapses.
    The routine combines a simulated‑annealing exploration phase with a
    deterministic polishing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Initial vector (analytic start or a previously found construction)
    # --------------------------------------------------------------
    a = _analytic_start(n)
    global height_sequence_1  # silence linter, may be undefined

    cand_global = _maybe_use_global()
    if cand_global is not None:
        # evaluate which start is better (fast internal score)
        score_global = _compute_score(cand_global, _fft_len_for_n(cand_global.size), cand_global.size)
        score_analytic = _compute_score(a, fft_len, n)
        if score_global < score_analytic:
            a = cand_global
            n = a.size
            fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    best_score = _compute_score(best, fft_len, n)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # leave a small safety margin

    # --------------------------------------------------------------
    # Simulated‑annealing parameters
    # --------------------------------------------------------------
    # Temperature schedule (exponential decay)
    T0 = best_score * 0.20               # start relatively “hot”
    Tmin = 1e-8
    # Block‑perturbation scale (log‑normal multiplier width)
    sigma_block = 0.35
    # Pair‑transfer scale (max fraction of source mass)
    max_pair_frac = 0.5

    cur_score = best_score
    cur_a = a.copy()
    # Keep the gradient of the current vector for quick reuse (not needed in SA)
    # ------------------------------------------------------------------
    # Main SA loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        # exponential temperature decay
        T = T0 * (Tmin / T0) ** (elapsed / total)

        # ----------------------------------------------------------
        # Propose a random move
        # ----------------------------------------------------------
        a_new = cur_a.copy()
        r = rng.random()
        if r < 0.5:
            # ----- pairwise mass transfer -----
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 1e-15:
                delta = rng.random() * a_new[i] * max_pair_frac
                a_new[i] -= delta
                a_new[j] += delta
        elif r < 0.8:
            # ----- block scaling -----
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, sigma_block))
            a_new[start:start + width] *= factor
        else:
            # ----- global log‑normal jitter -----
            jitter = np.exp(rng.normal(0.0, sigma_block / 2.0, size=n))
            a_new *= jitter

        # Normalise onto the simplex
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s

        # ----------------------------------------------------------
        # Evaluate the move (fast internal score)
        # ----------------------------------------------------------
        new_score = _compute_score(a_new, fft_len, n)

        # ----------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability proportional to exp(-(new‑old)/T)
            if T > 0.0:
                prob = np.exp((cur_score - new_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_a = a_new
            cur_score = new_score

            # keep global best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = cur_a.copy()
                now = time.monotonic()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # Optional early‑exit safeguard
        if time.monotonic() > deadline - 0.2:
            break

    # --------------------------------------------------------------
    # Polishing phase (deterministic mirror‑descent)
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        # allocate the rest of the budget to polishing
        polish_deadline = time.monotonic() + remaining
        best, best_score = _polish(best, n, fft_len, polish_deadline, best_score)

    # --------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
