"""
A simulated‑annealing / soft‑max gradient method for minimizing the
worst self‑convolution value of a non‑negative step function.
The algorithm works on the simplex (a_i ≥ 0, Σ a_i = 1) and uses a
smooth approximation of the max‑operator to obtain a dense gradient.
A combination of exponential‑gradient updates, occasional pairwise
mass‑transfer moves and soft restarts drives the search.
The best vector seen before the time budget expires is returned.
"""

import time
import numpy as np

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation (a ⋆ w) for non‑negative lags 0…len(a)‑1.
    The result i is Σ_j a[j]·w[i+j] (zero outside the support)."""
    n = a.size
    m = w.size
    N = 1 << ((n + m - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(A * np.conj(W), n=N)
    # correlation values for shifts 0 … n‑1 are the first n entries.
    return corr[:n]

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = np.clip(out, 0.0, None)
    s = out.sum()

    else:
        out /= s
    return out

    # ---------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
