"""
propose_candidate
=================

Generate a non‑negative step‑function whose self‑convolution is as flat as
possible.  The algorithm proceeds as follows:

1.  Initialise the vector with a *discrete arcsine* shape
    (a_i ∝ 1/√((i+½)(n‑½‑i))).  This distribution is known to have a perfectly
    flat continuous self‑convolution.

2.  Refine the vector with a *smooth‑max* mirror‑descent (exponentiated‑gradient)
    on the simplex.  At each iteration we compute the self‑convolution,
    turn the maximum into a soft‑max (temperature τ), obtain a smoothed gradient
    (a weighted cross‑correlation of the current vector with the soft‑max
    weights) and perform an exponential update
        a_i ← a_i·exp(−η·∂ℓ/∂a_i) .
    After the update we renormalise to the simplex (sum = 1).  The step size
    η and the temperature τ are slowly decreased, while occasional random
    jitter helps to escape plateaus.

3.  The whole refinement is run for the allotted time (default 1000 s) on
    several candidate lengths (powers of two around the previous best length).
    The best sequence seen (lowest value of ``evaluate_sequence``) is returned.

The method is completely self‑contained, uses only NumPy and the provided
``evaluate_sequence`` function, and respects all constraints (non‑negative
entries, clamping to [0, 1000] is performed automatically by the evaluator).
"""

import time
import math

# Helper utilities – **must be defined at top level**
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``).
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_lengths(l1: int, l2: int) -> int:
    """
    Smallest power‑of‑two length that can hold the full linear convolution
    of lengths ``l1`` and ``l2`` (i.e. length ``l1 + l2 − 1``).
    """
    size = l1 + l2 - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative real vector using the FFT.
    Returns an array of length ``2·len(a)‑1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_smooth(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the *smooth‑max* of the self‑convolution w.r.t. ``a``.
    The smooth‑max uses temperature ``tau`` and the gradient can be written
    as a weighted cross‑correlation:

        ∂ℓ/∂a_i = 2·∑_k w_k·a_{k-i},

    where ``w`` are soft‑max weights derived from ``conv``.
    The cross‑correlation is computed efficiently by an FFT convolution.
    """
    n = a.size
    # ---------- soft‑max weights ----------
    max_c = conv.max()
    # shift to avoid overflow/underflow
    exp_arg = (conv - max_c) / tau
    # Clip the exponent to a reasonable range (prevents NaNs)
    exp_arg = np.clip(exp_arg, -50.0, 0.0)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ---------- weighted cross‑correlation ----------
    # grad_i = 2 * sum_k w_k * a_{k-i}
    # This equals 2 * (a ⋆ w)_i, i.e. a cross‑correlated with w.
    # Compute via FFT: (a * rev(w)), then select the appropriate slice.
    rev_w = w[::-1]                               # length 2·n‑1
    N = _fft_len_for_lengths(a.size, rev_w.size)  # FFT length for a * rev_w
    fa = np.fft.rfft(a, n=N)
    fw = np.fft.rfft(rev_w, n=N)
    conv_rev = np.fft.irfft(fa * fw, n=N)
    # conv_rev length = n + (2n‑1)‑1 = 3n‑2
    conv_rev = conv_rev[: 3 * n - 2]
    # mapping: grad_i = 2 * conv_rev[2·n‑2‑i]
    idx = 2 * n - 2 - np.arange(n, dtype=np.int64)
    grad = 2.0 * conv_rev[idx]
    return grad


def _make_arcsine(n: int, alpha: float = 0.5) -> np.ndarray:
    """
    Return a normalised discrete arcsine‑type distribution of length ``n``.
    The density is proportional to

        1 / ((i+½)(n‑½‑i))**alpha

    (α = 0.5 gives the classical arcsine shape).
    The result is forced onto the simplex (sum = 1) and clipped to
    the range accepted by ``evaluate_sequence``.
    """
    i = np.arange(n, dtype=np.float64)
    raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
    raw = np.clip(raw, 0.0, 1000.0)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw[:] = 1.0 / n
    else:
        raw /= raw_sum
    return raw

                        elapsed = time.monotonic() - start_time
                        print(
                            f"[{elapsed:6.2f}s] new best = {sc:.6f} "
                            f"(n={n}, α={alpha:.2f})"
                        )
                iter_counter += 1

                # ----------------------------------------------------------
                # schedule updates for step size and temperature
                # ----------------------------------------------------------
                eta *= eta_decay
                tau = max(tau * tau_decay, min_tau)

    # ------------------------------------------------------------------
    # 2)  Light final polishing on the overall best sequence (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=np.float64)
        n = a.size
        fft_len_self = _fft_len_for_n(n)
        iter_counter = 0
        eta = 0.3
        eta_decay = 0.9999
        jitter_prob = 0.01
        jitter_sigma = 0.08
        eval_interval = 80
        # initialise temperature from the current convolution spread
        conv = _conv_self(a, fft_len_self)
        tau = max((conv.max() - conv.min()) * 0.15, 1e-9)
        tau_decay = 0.9997
        min_tau = 1e-12

        while time.monotonic() < deadline:
            conv = _conv_self(a, fft_len_self)
            grad = _grad_smooth(a, conv, tau)

            a = a * np.exp(-eta * grad)

            if rng.random() < jitter_prob:
                jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
                a *= jitter

            a = np.clip(a, 0.0, None)
            total = a.sum()
            if total <= 0.0:
                a[:] = 1.0 / n
            else:
                a /= total

            if (iter_counter % eval_interval) == 0:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq
                    elapsed = time.monotonic() - start_time
                    print(
                        f"[{elapsed:6.2f}s] polished best = {sc:.6f} (n={n})"
                    )

            eta *= eta_decay
            tau = max(tau * tau_decay, min_tau)
            iter_counter += 1
            if time.monotonic() - start_time > budget_s - 0.5:
                break

    # ------------------------------------------------------------------
    # 3)  Fallback – if nothing succeeded, return a tiny uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # final safety clamp (evaluate_sequence will also clamp, but we do it anyway)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(
        f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq
