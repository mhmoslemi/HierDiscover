import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _self_conv(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[:2*a.size - 1]
    return conv

def _random_block_perturb(a, rng):
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start+width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _upsample(arr, target_n):
    if arr.size == target_n:
        return arr
    x_old = np.linspace(0, 1, arr.size)
    x_new = np.linspace(0, 1, target_n)
    out = np.interp(x_new, x_old, arr)
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / target_n
    else:
        out /= s

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _upsample(arr: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of ``arr`` to length ``target_n`` (preserves positivity)."""
    if arr.size == target_n:
        return arr
    x_old = np.linspace(0, 1, arr.size)
    x_new = np.linspace(0, 1, target_n)
    out = np.interp(x_new, x_old, arr)
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / target_n
    else:
        out /= s
    return out
