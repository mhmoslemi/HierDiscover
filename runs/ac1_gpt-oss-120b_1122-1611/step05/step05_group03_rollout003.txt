def _fft_convolve(a, b):
    n = a.size
    m = b.size
    L = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(A * B, n=L)
    return conv[:n+m-1]

"""
Gradient‑descent with a smoothed max‑convolution objective.

The evaluation function equals   2·n·max(conv(a,a))/(sum(a)²).
If we keep the L¹‑norm fixed to  √(2 n)  the denominator is constant and
the task reduces to minimising the peak of the self‑convolution.
The code therefore:

1. Works on a fixed length n (default 4096, or the length of a supplied
   ``height_sequence_1``) and keeps the sum exactly √(2 n).
2. Uses a smooth approximation of the max:
        sβ(a) = (1/β)·log Σ exp(β·conv_k) .
   Its gradient can be written with a single extra convolution, which is
   evaluated efficiently by FFT.
3. Performs a back‑tracking line‑search on the (negative) gradient,
   projects back onto the simplex (non‑negative, fixed L¹‑norm) and keeps
   the best vector found.
4. Occasionally injects a random log‑normal block perturbation and restarts
   if no improvement occurs for a while.
5. Stops after the allotted time (≈ 1000 s) and returns the best
   sequence (clipped to the limits of ``evaluate_sequence``).

The method is deliberately different from the greedy “mass‑transfer”
search that produced 1.514150 and often finds a lower peak, getting
closer to the target 1.5025.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level only
# ----------------------------------------------------------------------


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full convolution of length len(x)+len(y)-1.
    """
    n = x.size
    m = y.size
    L = 1 << ((n + m - 1).bit_length())           # next power of two
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: n + m - 1]                      # strip padding


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given the current vector ``a`` and a weight vector ``w`` (same length as
    the self‑convolution of ``a``), returns the gradient of the smoothed
    max w.r.t. ``a``:
        ∇_a  Σ w_k·(a*a)_k = 2·(a * w_rev)[offset‑j] .
    The implementation uses FFT for speed.
    """
    n = a.size
    L = 1 << ((n + w.size - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    W_rev = np.fft.rfft(w[::-1], n=L)
    conv_rev = np.fft.irfft(A * W_rev, n=L)
    conv_rev = conv_rev[: n + w.size - 1]         # true linear length
    offset = w.size - 1                           # = 2*n-2
    # gradient_j = 2 * conv_rev[offset - j]
    return 2.0 * conv_rev[offset - np.arange(n)]


def _project_to_fixed_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projects a non‑negative vector onto the simplex with the given L¹‑norm.
    """
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        v[:] = target_sum / v.size
    else:
        v *= target_sum / s
    return v


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          max_width_frac: float = 0.2,
                          sigma: float = 0.12) -> np.ndarray:
    """
    Apply a multiplicative log‑normal jitter to a randomly chosen contiguous
    block of ``a`` and re‑normalise to the fixed L¹‑norm.
    """
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, sigma))
    b = a.copy()
    b[start:start + width] *= factor
    b = np.clip(b, 0.0, None)
    return b


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function (probability vector) that
    minimises ``evaluate_sequence``.
    The algorithm uses a smoothed max‑convolution gradient descent with
    occasional random perturbations and restarts.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Initialise the coefficient vector and target L1‑norm
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        n = a.size
    else:
        n = 4096                     # a reasonable default length
        a = np.full(n, 1.0, dtype=np.float64)

    target_sum = math.sqrt(2.0 * n)   # normalisation used in the paper
    a = _project_to_fixed_sum(a, target_sum)

    # break the perfect symmetry of a fully uniform start
    a *= np.exp(rng.normal(0.0, 0.10, size=n))
    a = _project_to_fixed_sum(a, target_sum)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)    # safety margin
    last_improve = start_time

    # --------------------------------------------------------------
    # 2) Hyper‑parameters
    # --------------------------------------------------------------
    beta = 1.0          # smoothing parameter for the soft‑max
    beta_max = 50.0
    beta_mult = 1.2    # how quickly we increase beta
    step_factor = 0.25 * target_sum   # controls absolute step size
    min_step_scale = 0.01
    max_no_improve = 60.0                # seconds before a forced restart

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ----- compute full self‑convolution (via FFT) -----
        conv = _fft_convolve(a, a)                # length 2*n-1
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (target_sum ** 2)   # = max_b because target_sum²=2n

        # ----- smoothed‑max weights (soft‑max) -----
        # subtract max_b to avoid overflow when beta is large
        exp_vals = np.exp(beta * (conv - max_b))
        w = exp_vals / exp_vals.sum()             # shape (2*n-1,)

        # ----- gradient of the smoothed max -----
        grad = _gradient_from_weights(a, w)        # shape (n,)

        # ----- line‑search on the negative gradient -----
        improved = False
        for scale in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            # normalise step size by the gradient magnitude
            alpha = scale * step_factor / (np.linalg.norm(grad) + 1e-12)
            a_candidate = a - alpha * grad
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = _project_to_fixed_sum(a_candidate, target_sum)

            score = float(evaluate_sequence(a_candidate.tolist()))
            if score < cur_score - 1e-12:          # accept only true improvements
                a = a_candidate
                cur_score = score
                improved = True
                if score < best_score - 1e-12:
                    best_score = score
                    best = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] new best {best_score:.6f}")
                break

        # ----- occasional random block perturbation (helps escape plateaus) -----
        if not improved and rng.random() < 0.015:
            a_pert = _random_block_perturb(a, rng, max_width_frac=0.15, sigma=0.20)
            a_pert = _project_to_fixed_sum(a_pert, target_sum)
            pert_score = float(evaluate_sequence(a_pert.tolist()))
            if pert_score < cur_score - 1e-12:
                a = a_pert
                cur_score = pert_score
                if pert_score < best_score - 1e-12:
                    best_score = pert_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] block‑perturb improved best {best_score:.6f}")

        # ----- increase smoothing (sharpen the approximation of max) -----
        if iteration % 120 == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

        # ----- restart if we have been stuck for a while -----
        if time.monotonic() - last_improve > max_no_improve:
            # jitter the best solution a little and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.30, size=n))
            a = a * noise
            a = _project_to_fixed_sum(a, target_sum)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – score reset to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
