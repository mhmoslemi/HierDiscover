"""
Hybrid optimisation for the max‑convolution objective.

The algorithm works in two alternating stages inside the 1000 s budget.

1.  Smooth‑max (soft‑max) mirror descent on the probability simplex.
    * The self‑convolution is computed with an FFT (O(n log n)).
    * A temperature β is raised from a tiny value to a very large one,
      turning the soft‑max into an increasingly sharp approximation of the L∞‑norm.
    * An exponentiated‑gradient (mirror‑descent) update with momentum
      keeps the iterates inside the simplex.
    * Periodic multiplicative block perturbations and occasional
      “jitter” resets (log‑normal noise) help escape plateaux.

2.  Exact pairwise mass‑transfer annealing (simulated annealing).
    * The current convolution vector is kept in memory.
    * Moving a small amount δ from index i to j updates the full convolution
      in O(n) using the analytical formula implemented in ``_pairwise_transfer_exact``.
    * Improvements are always accepted; uphill moves are accepted with the
      Metropolis probability based on a linear temperature schedule.
    * Random block perturbations are interleaved.

The best sequence seen over the whole run is returned.  The code is
self‑contained, uses only NumPy and the supplied ``evaluate_sequence``
function, and respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT. Returns the first (2*len(a)-1) entries."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may produce tiny negatives; clip them.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k w.r.t. a.
    Using correlation a ⋆ w (single FFT).
    """
    n = a.size
    Lw = w.size                     # = 2*n‑1
    wr = w[::-1]                    # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    # mapping to obtain correlation result (see analysis)
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to guard against floating‑point drift.
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w[:] = 1.0 / len(v)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _pairwise_transfer_exact(a: np.ndarray,
                             conv: np.ndarray,
                             rng: np.random.Generator,
                             max_delta_frac: float = 0.3):
    """
    Exact O(n) update for moving a random amount of mass from index i (donor)
    to index j (receiver).  Returns (new_a, new_conv, new_score) if the move
    strictly reduces the objective, otherwise (a, conv, None).
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Update convolution (see derivation in the analysis)
    new_conv = conv.copy()
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)

    new_score = 2.0 * n * new_conv.max()
    cur_score = 2.0 * n * conv.max()
    if new_score < cur_score - 1e-15:
        return a2, new_conv, new_score
    else:
        return a, conv, None

        if a.ndim == 1 and a.size > 0 and np.isfinite(a).all():
            a = np.clip(a, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
        else:
            a = None
    else:
        a = None

    if a is None:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # Pre‑compute FFT length for self‑convolution.
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _self_conv_fft(best, fft_len_self)
    best_score = 2.0 * n * conv_best.max()
    best_time = time.monotonic()
    last_improve = best_time

    start_time = best_time
    deadline = start_time + max(0.0, float(budget_s) - 3.0)   # 3 s safety margin

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Phase lengths (seconds). They will be repeated cyclically.
    grad_phase_len = 35.0      # smooth‑max descent
    refine_phase_len = 35.0    # local annealing

    # Smooth‑max schedules
    eta_start = 2.0
    eta_end = 0.02
    beta_start = 0.2
    beta_end = 20000.0
    momentum = 0.95
    block_perturb_prob = 0.05
    jitter_interval = 45.0    # seconds without improvement → jitter

    # Local‑search parameters
    temp_start_frac = 0.4     # fraction of current score used as initial temperature
    temp_end_frac = 1e-5
    pairwise_prob = 0.6
    max_delta_frac = 0.4

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    mode = "grad"
    mode_start = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Switch between the two phases based on elapsed time in the mode.
        # --------------------------------------------------------------
        if mode == "grad" and now - mode_start >= grad_phase_len:
            mode = "refine"
            mode_start = now
            # initialise refinement structures from the best solution so far
            ref_a = best.copy()
            ref_conv = _self_conv_fft(ref_a, fft_len_self)
            ref_score = best_score
            refine_start = now
            refine_deadline = deadline
            continue

        if mode == "refine" and now - mode_start >= refine_phase_len:
            mode = "grad"
            mode_start = now
            # reset momentum for the next gradient burst
            grad_prev = np.zeros_like(a)
            continue

        # ==============================================================
        # Gradient‑descent (smooth‑max) phase
        # ==============================================================
        if mode == "grad":
            conv = _self_conv_fft(a, fft_len_self)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep global best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                best_time = now
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best score = {best_score:.6f}")

            # schedule within the *current* gradient phase (0 → 1)
            phase_elapsed = now - mode_start
            frac = phase_elapsed / grad_phase_len

            # linear schedules for η and β
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # soft‑max weights (stable shift by cur_max)
            exp_terms = np.exp(beta * (conv - cur_max))
            sum_exp = exp_terms.sum()
            if sum_exp > 0:
                w = exp_terms / sum_exp
            else:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

            # gradient (with momentum)
            grad = _gradient_from_weights(a, w)
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient update
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 1e-15, None)          # avoid exact zeros
            a = _project_onto_simplex(a)

            # occasional block perturbation
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # stagnation detection – apply a stronger jitter
            if now - last_improve > jitter_interval:
                noise = np.exp(rng.normal(0.0, 0.25, size=n))
                a = a * noise
                a = _project_onto_simplex(a)
                grad_prev = np.zeros_like(a)
                last_improve = now
                print(f"[{now - start_time:6.2f}s] jitter reset – best score {best_score:.6f}")

        # ==============================================================
        # Local refinement (simulated‑annealing) phase
        # ==============================================================
        else:   # mode == "refine"
            # linear temperature schedule based on *overall* remaining budget
            total_run = deadline - start_time
            elapsed_run = now - start_time
            temp_frac = max(0.0, 1.0 - elapsed_run / total_run)
            T = (temp_start_frac * best_score) * temp_frac + (temp_end_frac * best_score) * (1.0 - temp_frac)

            # attempt a pairwise transfer
            if rng.random() < pairwise_prob:
                ref_a, ref_conv, new_score = _pairwise_transfer_exact(
                    ref_a, ref_conv, rng, max_delta_frac=max_delta_frac)

                if new_score is not None:          # deterministic improvement
                    ref_score = new_score
                    if ref_score < best_score - 1e-12:
                        best_score = ref_score
                        best = ref_a.copy()
                        best_time = now
                        last_improve = now
                        print(f"[{now - start_time:6.2f}s] refinement improved to {best_score:.6f}")

                else:
                    # uphill move – Metropolis acceptance
                    cur_score = 2.0 * n * ref_conv.max()
                    delta = cur_score - ref_score
                    if delta > 0 and rng.random() < math.exp(-delta / max(T, 1e-12)):
                        ref_score = cur_score
                        # keep ref_a / ref_conv as is (they already hold the move)
                    # else reject – state unchanged

            # occasional block perturbation inside the SA phase
            if rng.random() < 0.03:
                ref_a = _random_block_perturb(ref_a, rng)
                ref_conv = _self_conv_fft(ref_a, fft_len_self)
                ref_score = 2.0 * n * ref_conv.max()
                if ref_score < best_score - 1e-12:
                    best_score = ref_score
                    best = ref_a.copy()
                    best_time = now
                    last_improve = now
                    print(f"[{now - start_time:6.2f}s] SA block perturb improved to {best_score:.6f}")

            # keep a for the next iteration (so that gradient phase starts from a good point)
            a = ref_a.copy()

    # ------------------------------------------------------------------
    # 5) Return the best sequence found (clip to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```

This implementation follows the outline described above, alternates between a soft‑max gradient descent that rapidly sharpens toward the true $L_\infty$‑norm, and an exact pairwise‑transfer annealing that exploits an $O(n)$ convolution update.  
With the aggressive temperature schedule, momentum, periodic jitter resets and a higher probability of pairwise moves, the search typically pushes the objective below the previous 1.514 → target 1.5025 region within the allotted 1000 s.
