def project_simplex(v):
    # v is a 1-d array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of two real 1‑d vectors via FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    conv = np.fft.irfft(Z, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Gradient‑descent (soft‑max) utilities
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights that approximate the max of ``conv``."""
    # stability: subtract the max before exponentiation
    cmax = np.max(conv)
    exp_part = np.exp(beta * (conv - cmax))
    w = exp_part / np.sum(exp_part)
    return w


def _grad_from_weights(weights: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate w.r.t. ``a``.
    For each index i:   grad_i = 2 * Σ_k weights_k * a_{k-i}
    (terms outside the support are ignored).
    """
    # convolution of weights with the reversed a, then take the central n entries
    rev_a = a[::-1]
    full = _full_convolution(weights, rev_a)          # length 3n‑2
    n = a.size
    # the gradient lies in the slice [n‑1 : n‑1+n]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Hill‑climbing (targeted mass transfer) utilities – same as the previous state‑of‑the‑art
# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2*n‑1)."""
    return _full_convolution(a, a)


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    i: int,
                    j: int,
                    delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving ``delta`` mass from index i
    to index j (i != j).  Uses the exact update formula.
    """
    n = a.size
    new_conv = conv.copy()
    # linear part: -2Δ·a shifted from i  + 2Δ·a shifted from j
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a
    # quadratic part Δ²
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          max_width_frac: float = 0.2,
                          sigma: float = 0.12) -> np.ndarray:
    """
    Apply a multiplicative log‑normal jitter to a random contiguous block,
    then re‑normalise onto the simplex.
    """
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (probability vector) that
    minimises ``evaluate_sequence``.  The routine combines a smooth
    gradient descent phase with a final greedy mass‑transfer phase and
    returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Problem size – we keep the classic size ~4000 (fast enough)
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 1) Initialise a probability vector (sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a = a0 / s0
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        if a.size != n:
            # resize (pad with zeros or truncate) to the target length
            if a.size < n:
                a = np.concatenate([a, np.full(n - a.size, 0.0)])
            else:
                a = a[:n]
            a = _project_simplex(a)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # keep track of the best vector and its true score
    cur_conv = _self_convolution(a)
    cur_score = 2.0 * n * np.max(cur_conv)          # exact objective (sum = 1)
    best_a = a.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # 2) Adam optimiser state (for the soft‑max gradient descent)
    # --------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    adam_iter = 0

    # --------------------------------------------------------------
    # 3) Timings
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # leave a safety margin
    last_improve = start_time
    phase_switch = 0.85    # fraction of total time after which we switch to polishing

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        phase = "grad" if elapsed / total < phase_switch else "hill"

        #################################################################
        # ----- Phase 1 : soft‑max gradient descent --------------------
        #################################################################
        if phase == "grad":
            # ----- 4.1 compute convolution (full) via FFT -----
            conv = _self_convolution(a)                # length 2n‑1
            max_conv = np.max(conv)

            # ----- 4.2 soft‑max weights (beta grows with time) -----
            # start with a mild beta, finish with a sharp one
            frac = elapsed / (total * phase_switch)
            beta = 5.0 + 195.0 * frac          # range 5 → 200 approx

            weights = _softmax_weights(conv, beta)

            # ----- 4.3 gradient of the smoothed objective -----
            grad = _grad_from_weights(weights, a)   # shape (n,)

            # ----- 4.4 Adam step -----
            adam_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** adam_iter)
            v_hat = v / (1.0 - beta2 ** adam_iter)

            # base learning rate schedule – start larger, decay to ~0.01
            base_lr = 0.15
            lr = base_lr * (1.0 - frac) + 0.01 * frac

            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # take a step and re‑project onto the simplex
            a_new = _project_simplex(a - step)

            # ----- 4.5 evaluate the true objective (no extra clipping) -----
            conv_new = _self_convolution(a_new)
            score_new = 2.0 * n * np.max(conv_new)

            # accept only if we made progress (monotone descent)
            if score_new <= cur_score:
                a = a_new
                cur_score = score_new
                cur_conv = conv_new
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    last_improve = now
                    print(f"[{now - start_time:6.2f}s] grad‑phase new best {best_score:.6f}")

            # occasional jitter to escape very flat plateaus
            if rng.random() < 0.003:
                noise = np.exp(rng.normal(0.0, 0.12, size=n))
                a = _project_simplex(a * noise)
                # reset Adam moments to avoid bias from the sudden jump
                m.fill(0.0)
                v.fill(0.0)

        #################################################################
        # ----- Phase 2 : targeted mass‑transfer polishing -------------
        #################################################################
        else:
            # Re‑use the fast helpers from the previous state‑of‑the‑art code.
            # The loop below is a lightweight version that performs a single
            # transfer per iteration.
            # locate current max of the convolution (already have cur_conv)
            k_max = int(np.argmax(cur_conv))

            # partner values for each index with respect to k_max
            idx = np.arange(n)
            valid = (k_max - idx >= 0) & (k_max - idx < n)
            partner = np.zeros_like(a)
            partner[valid] = a[k_max - idx[valid]]

            # donor: index with a large product a[i]*partner[i]
            donor_mask = a > 1e-12
            donor_score = partner * donor_mask.astype(float)
            if donor_score.max() == 0.0:
                # fallback random move
                i = rng.integers(0, n)
                if a[i] <= 0.0:
                    continue
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_delta = 0.3 * a[i]
                delta = rng.random() * max_delta
                new_conv = _apply_transfer(a, cur_conv, i, j, delta)
                new_score = 2.0 * n * np.max(new_conv)
                if new_score < cur_score:
                    a[i] -= delta
                    a[j] += delta
                    cur_conv = new_conv
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                        last_improve = time.monotonic()
                        print(f"[{time.monotonic() - start_time:6.2f}s] rand‑move new best {best_score:.6f}")
                continue

            donor = int(np.argmax(donor_score))

            # receiver: index with the smallest partner value
            min_partner = partner.min()
            receiver_candidates = np.where(partner == min_partner)[0]
            if donor in receiver_candidates:
                receiver_candidates = receiver_candidates[receiver_candidates != donor]
            if receiver_candidates.size == 0:
                receiver = (donor + 1) % n
            else:
                receiver = int(rng.choice(receiver_candidates))

            # step size – larger early, smaller later
            frac = (now - start_time) / total
            max_frac = 0.35 * (1.0 - frac) + 0.02 * frac   # up to 35 % of donor’s mass
            max_delta = a[donor] * max_frac
            if max_delta > 1e-12:
                delta = rng.random() * max_delta
                new_conv = _apply_transfer(a, cur_conv, donor, receiver, delta)
                new_score = 2.0 * n * np.max(new_conv)
                if new_score < cur_score:
                    a[donor] -= delta
                    a[receiver] += delta
                    cur_conv = new_conv
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                        last_improve = now
                        print(f"[{now - start_time:6.2f}s] hill‑move new best {best_score:.6f}")

            # occasional random block perturbation
            if rng.random() < 0.015:
                a_pert = _random_block_perturb(a, rng)
                conv_pert = _self_convolution(a_pert)
                pert_score = 2.0 * n * np.max(conv_pert)
                if pert_score < cur_score:
                    a = a_pert
                    cur_conv = conv_pert
                    cur_score = pert_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                        last_improve = time.monotonic()
                        print(f"[{time.monotonic() - start_time:6.2f}s] block‑perturb new best {best_score:.6f}")

            # restart if stuck for a while
            if now - last_improve > 30.0:
                a = best_a.copy()
                noise = np.exp(rng.normal(0.0, 0.25, size=n))
                a = _project_simplex(a * noise)
                cur_conv = _self_convolution(a)
                cur_score = 2.0 * n * np.max(cur_conv)
                last_improve = now
                print(f"[{now - start_time:6.2f}s] restart – score reset to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best vector (clip to the limits accepted by evaluate_sequence)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
