"""
Propose a candidate step‑function by *mirror descent* on the
L∞‑norm of the self‑convolution.

The algorithm works on the probability simplex (non‑negative entries,
sum = 1).  At each iteration we:
  1. Compute the full self‑convolution via an FFT (O(n log n)).
  2. Locate the index k_max where the convolution attains its maximum.
  3. The sub‑gradient of the maximal entry with respect to a_i is
        g_i = 2·a_{k_max‑i}   (zero outside the overlap range).
  4. Perform a multiplicative (exponentiated) update
        a_i ← a_i·exp(‑η·g_i)   and renormalise onto the simplex.
     This automatically keeps the vector non‑negative and respects the
     sum‑constraint, while moving mass away from components that
     heavily participate in the peak.
  5. Occasionally apply a random block jitter to escape shallow
     plateaus and periodically restart from the best solution found
     (with a small stochastic perturbation).

The step size η is linearly annealed from a modest initial value to a
tiny final value, giving large moves early on and fine‑grained tuning
later.  The whole search respects the 1000 s budget and returns the
best vector discovered (clipped to the limits of ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum x = 1}.
    Implements the O(n log n) algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should never happen for non‑zero vector)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a real FFT.
    The length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def subgradient_max(conv: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) with respect to the vector a.
    The gradient has the form
        g_i = 2 * a_{k_max - i}   (zero when the index is out of range).
    """
    n = a.shape[0]
    k_max = int(np.argmax(conv))

    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)

    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k_max - idx]
    return g


def evaluate_score(a: np.ndarray, conv: np.ndarray = None) -> float:
    """
    Compute the objective 2·n·max(conv) / (sum a)².
    Since we keep a on the simplex (sum a = 1) this reduces to
        score = 2·n·max(conv).
    """
    n = a.shape[0]
    if conv is None:
        conv = convolve_self(a)
    max_conv = conv.max()
    # sum a == 1 by construction, but keep the denominator for safety
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a ** 2)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0):
    """
    Search for a non‑negative step function (probability vector) that
    minimises ``evaluate_sequence``.  Returns the best sequence found
    within the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine the length of the vector (reuse previous solution if
    #    available, otherwise a default moderate size).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        init = np.clip(init, 0.0, None)
        if init.sum() > 0.0:
            init = init / init.sum()
        else:
            init = np.full_like(init, 1.0 / init.size)
        n = init.size
    else:
        n = 4096                     # moderate length – fast FFTs
        init = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 2) Global bookkeeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin
    best_vec = init.copy()
    best_score = evaluate_score(best_vec)

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    base_eta = 0.18          # initial step size for the exponentiated update
    min_eta = 0.001          # final step size
    max_iters_per_restart = 20000
    stagnation_limit = 30.0   # seconds without improvement → restart
    restart_counter = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop – many short restarts
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- initialise a new candidate (jittered best every second restart) -----
        if best_vec is not None and restart_counter % 2 == 0:
            a = best_vec.copy()
            jitter = np.exp(rng.normal(0.0, 0.25, size=n))
            a = a * jitter
        else:
            a = rng.random(n)          # Dirichlet‑like start
        a = np.clip(a, 0.0, None)
        a = a / a.sum()

        # ----- per‑restart state -----
        last_improve = time.monotonic()
        cur_best_score = evaluate_score(a)

        for it in range(max_iters_per_restart):
            now = time.monotonic()
            if now >= deadline:
                break

            # ----- evaluate current vector -----
            conv = convolve_self(a)
            score = evaluate_score(a, conv)

            # ----- update global best -----
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new global best {best_score:.6f}")

            # ----- compute sub‑gradient at the peak of the convolution -----
            grad = subgradient_max(conv, a)

            # ----- step size annealing -----
            frac = (now - start_time) / (deadline - start_time)
            eta = base_eta * (1.0 - frac) + min_eta * frac

            # ----- exponentiated (mirror) update + simplex projection -----
            # clip the gradient to avoid overflow in the exponential
            grad_clipped = np.clip(grad, -10.0, 10.0)
            a_proposed = a * np.exp(-eta * grad_clipped)
            a_proposed_sum = a_proposed.sum()
            if a_proposed_sum == 0.0:
                a_proposed = np.full_like(a, 1.0 / n)
            else:
                a_proposed = a_proposed / a_proposed_sum

            # ----- evaluate proposed move -----
            conv_prop = convolve_self(a_proposed)
            score_prop = evaluate_score(a_proposed, conv_prop)

            # ----- accept if it improves, otherwise accept with a tiny
            #       simulated‑annealing probability to escape shallow traps -----
            if score_prop <= score:
                a = a_proposed
                conv = conv_prop
                if score_prop < cur_best_score:
                    cur_best_score = score_prop
                    last_improve = now
            else:
                # temperature‑like schedule for uphill moves
                temperature = 0.02 * (1.0 - frac) + 0.001
                prob_uphill = np.exp(-(score_prop - score) / temperature)
                if rng.random() < prob_uphill:
                    a = a_proposed
                    conv = conv_prop

            # ----- occasional random block perturbation (helps escape plateaus) -----
            if rng.random() < 0.005:
                width = max(1, int(0.05 * n))
                start_idx = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.2))
                a[start_idx:start_idx + width] *= factor
                a = a / a.sum()

            # ----- restart if stuck for a while -----
            if now - last_improve > stagnation_limit:
                break

        restart_counter += 1

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped to evaluate_sequence limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
