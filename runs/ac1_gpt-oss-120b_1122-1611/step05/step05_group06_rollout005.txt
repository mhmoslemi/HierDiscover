def _local_optimise(a, rng, deadline, max_iters=500):
    n = a.size
    fft_len_conv = _fft_len_for_len(2*n-1)
    fft_len_grad = _fft_len_for_len(3*n-2)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    best_vec = a.copy()
    best_score = 2.0*n*_conv_self(a, fft_len_conv).max()
    while t < max_iters and time.monotonic() < deadline:
        t+=1
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0*n*conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
        # gradient selection
        if rng.random() < 0.5:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = 0.5 + (5000.0 - 0.5) * (t / max_iters)   # adjust dynamic max beta maybe larger.
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)
        # Adam update
        lr = 0.5 / np.sqrt(t)
        m = beta1*m + (1-beta1)*raw_grad
        v = beta2*v + (1-beta2)*(raw_grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_onto_simplex(a)
        # occasional random perturbations
        if rng.random() < 0.05:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.05:
            a = _pairwise_move(a, rng, fft_len_conv)
    return best_vec, best_score

def _smoothify(a, passes=3, kernel_size=3):
    k = np.ones(kernel_size)/kernel_size
    for _ in range(passes):
        a = np.convolve(a, k, mode='same')
        a = _project_onto_simplex(a)

"""
Hybrid Adam‑based optimization for the self‑convolution L∞‑problem.

The algorithm works on the probability simplex (∑a_i = 1) and repeatedly
minimises the smoothed max   f_β(a) = (1/β)·log Σ_i exp(β·(a*a)_i)
using Adam with both first‑ and second‑moment estimates.
A random sub‑gradient (exact worst‑entry) is interleaved with the smooth
gradient, and occasional random block‑perturbations and cheap pairwise
mass‑transfer moves are applied to escape shallow local minima.
The search consists of a fast exploratory phase (many short restarts)
followed by a long refinement phase on the current global best.
All vectors are projected onto the simplex after every update.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Return the smallest power‑of‑two ≥ length (suitable for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors x and y via a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a * a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    For the worst index i_max we have (∂/∂a_j) conv_i_max = 2·a_{i_max‑j}
    when the index lies inside the support.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (without any spurious extra term).
    """
    cur_max = conv.max()
    # work with shifted exponents for numerical stability
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    # convolution of w with the reversed a gives the required weighted sum
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    # extract the n‑length gradient slice
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    max_width = max(2, n // 30)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    b = a.copy()
    b[start:start + width] *= factor
    return _project_onto_simplex(b)


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """
    Cheap mass‑transfer move: move a random small amount of probability from i to j.
    The move is kept only if it improves the current objective.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    conv_new = _conv_self(a2, fft_len_conv)
    cur_score = 2.0 * n * _conv_self(a, fft_len_conv).max()
    new_score = 2.0 * n * conv_new.max()
    return a2 if new_score < cur_score else a


# ----------------------------------------------------------------------
# Core local optimiser
# ----------------------------------------------------------------------


def _local_optimise(a: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float,
                    max_iters: int = 500) -> tuple[np.ndarray, float]:
    """
    Perform a projected Adam optimisation of a (already normalised to Σa=1).
    Returns the best vector seen and its score 2·n·max(conv).
    """
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state – we keep both first (m) and second (v) moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_vec = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len_conv).max()

    t = 0
    while t < max_iters and time.monotonic() < deadline:
        t += 1

        # ---- current convolution & score ---------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ---- gradient choice ---------------------------------------------
        if rng.random() < 0.55:
            # hard sub‑gradient (exact worst entry)
            raw_grad = _subgradient_step(a, conv)
        else:
            # smooth max – β grows from 0.5 to a large value
            beta = 0.5 + (4000.0 - 0.5) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # ---- Adam update -------------------------------------------------
        lr = 0.7 / np.sqrt(t)                # gently decreasing learning rate
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # ---- occasional stochastic moves ----------------------------------
        if rng.random() < 0.04:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.04:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Optional post‑processing (light smoothing)
# ----------------------------------------------------------------------


def _smoothify(a: np.ndarray, passes: int = 2, kernel_size: int = 3) -> np.ndarray:
    """Apply a few passes of a simple moving‑average filter and re‑project."""
    if kernel_size < 2:
        return a
    k = np.ones(kernel_size) / kernel_size
    for _ in range(passes):
        a = np.convolve(a, k, mode='same')
        a = _project_onto_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence().  The routine respects the time budget, performs a
    fast exploratory phase with many short restarts and a slower refinement
    phase on the current global best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Initialise from any previously discovered candidate
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cur = 2.0 * cand.size * _conv_self(cand,
                                                   _fft_len_for_len(2 * cand.size - 1)).max()
                if cur < best_score:
                    best_score, best_vec = cur, cand.copy()
                print(f"Initial global candidate score = {cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Exploration phase: many cheap restarts
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 3809, 5000, 8000, 10000]
    phase1_deadline = start_time + 0.55 * float(budget_s)   # ~55 % of the budget

    while time.monotonic() < phase1_deadline:
        n = int(rng.choice(lengths,
                           p=np.array([0.15, 0.20, 0.25, 0.20, 0.15, 0.05])))
        # random start on the simplex (Dirichlet‑like)
        a0 = rng.random(n)
        a0 /= a0.sum()

        a_opt, score_opt = _local_optimise(a0, rng,
                                           deadline=phase1_deadline,
                                           max_iters=800)

        if score_opt < best_score:
            best_score, best_vec = score_opt, a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best (short run) = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Refinement phase: longer runs on the current best
    # ------------------------------------------------------------------
    if best_vec is not None:
        # The remaining time is split into a few long optimisations and
        # occasional random perturbations followed by a short optimisation.
        while time.monotonic() < deadline - 2.0:
            # long polish
            a_polished, score_polished = _local_optimise(best_vec.copy(),
                                                         rng,
                                                         deadline=deadline,
                                                         max_iters=3500)
            if score_polished < best_score:
                best_score, best_vec = score_polished, a_polished.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polished best = {best_score:.6f}")

            # occasional stochastic restart from a perturbed version
            if rng.random() < 0.25:
                pert = _random_block_perturb(best_vec, rng)
                a2, s2 = _local_optimise(pert, rng,
                                          deadline=deadline,
                                          max_iters=2000)
                if s2 < best_score:
                    best_score, best_vec = s2, a2.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] Perturbation improved = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 4000
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Light smoothing (helps to remove tiny high‑frequency spikes)
    # ------------------------------------------------------------------
    best_vec = _smoothify(best_vec, passes=2, kernel_size=5)

    # Final safety clipping – the harness also clips, but we do it here.
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
