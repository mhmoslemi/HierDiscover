"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence() = 2·n·max(conv(a,a)) / (sum(a))².

Algorithm
---------
1. Initialise a on the probability simplex (sum = 1).  If a previous
   construction `height_sequence_1` is present we start from it,
   otherwise we use a uniform distribution of length 4096 (a power‑of‑2
   for fast FFTs).

2. Split the 1000 s budget into three phases

   • Phase 1 (≈25 % of time) – “smooth‑max’’ gradient descent.
     The max‑convolution is approximated by a soft‑max with a temperature
     β that grows from 0.1 to ~3000.  The gradient of the soft‑max is
     obtained by a single FFT‑correlation; an exponentiated‑gradient
     (mirror‑descent) update is performed and the vector is projected
     back onto the simplex.  Small random block perturbations and a
     jitter‑reset guard against stagnation.

   • Phase 2 (≈60 % of time) – sub‑gradient descent on the true L∞‑norm.
     The index t of the maximal convolution entry is located; a
     one‑hot weight at t yields the exact sub‑gradient ∇(a*a)_t.
     Again an exponentiated‑gradient step with a decaying step‑size η is
     applied and projected onto the simplex.  Occasional block
     perturbations and a random Dirichlet restart are used when no
     improvement occurs for a minute.

   • Phase 3 (remaining time) – local refinement.
     The current best vector is refined by many cheap exact pairwise
     mass‑transfer moves (see _pairwise_transfer_exact).  Improving moves
     are always kept; uphill moves are accepted with a Metropolis
     probability that follows a linear temperature schedule.
     Additional block perturbations are injected periodically.

3. Throughout the run the best vector seen so far (lowest score) is
   tracked and printed.  At termination the best vector is clipped to
   the limits imposed by `evaluate_sequence` and returned as a plain
   Python list of floats.

All helper routines are top‑level and avoid closures; the implementation
works with a single CPU (FFT is the dominant cost) and stays well within
the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negative values caused by floating‑point noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Derivation: grad_j = 2·Σ_i a_i·w_{i+j}.
    Implemented as a correlation a ⋆ w using a single FFT.
    """
    n = a.size
    Lw = w.size                     # = 2*n‑1
    wr = w[::-1]                    # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)   # length for convolution a ⋆ wr

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)         # mapping to obtain correlation result
    grad = 2.0 * conv_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)

    return _project_onto_simplex(out)


def _pairwise_transfer_exact(a: np.ndarray, conv: np.ndarray,
                             rng: np.random.Generator,
                             max_delta_frac: float = 0.3):
    """
    Exact O(n) update for moving a random amount of mass from a donor index i
    to a receiver index j.  Returns (new_a, new_conv, new_score) if the move
    reduces the objective, otherwise the original tuple (a, conv, None).
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if i == j:
        return a, conv, None

    # maximal transferable mass – keep a[i] ≥ 0 after the move
    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Exact convolution update (see analysis for derivation)
    new_conv = conv.copy()
    # donor contribution removed
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    # receiver contribution added
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a
    # diagonal corrections (+delta^2)
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    # cross‑term second‑order correction
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    # update the coefficient vector
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)   # protects against rounding drift

    new_score = 2.0 * n * new_conv.max()
    cur_score = 2.0 * n * conv.max()
    if new_score < cur_score - 1e-15:      # strict improvement
        return a2, new_conv, new_score
    else:
        return a, conv, None

def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):

    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and returns the best coefficient list found.

        if cand0.ndim == 1 and cand0.size >= 2000 and cand0.size <= 8000 and np.isfinite(cand0).all():

    else:
        n = 4096

    eps = 1e-12
    a = np.maximum(a, eps)
    a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _self_conv_fft(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    print(f"initial score = {best_score:.6f}")

    # keep a small safety margin (5 s) for the outer harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    phase1_frac = 0.25          # smooth‑max descent
    phase2_frac = 0.60          # exact max‑gradient descent
    # the remaining 0.15 of the budget is for local refinement

    # schedules
    eta_start = 1.0
    eta_end = 0.01
    beta_start = 0.2
    beta_end = 4000.0
    momentum = 0.9

    grad_prev = np.zeros_like(a)

    # flags for refinement phase
    refinement_initialised = False

    # ------------------------------------------------------------------

        elapsed = now - start_time
        total = deadline - start_time

        conv = _self_conv_fft(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------------

        # --------------------------------------------------------------------
        if frac < phase1_frac:
            # ------------------- Phase 1 : smooth‑max gradient descent -------
            # temperature schedule
            beta = beta_start + (beta_end - beta_start) * (frac / phase1_frac)

            # soft‑max weights (stable formulation)
            exp_terms = np.exp(beta * (conv - cur_max))
            sum_exp = exp_terms.sum()
            if sum_exp > 0:
                w = exp_terms / sum_exp
            else:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

            # gradient (via correlation)
            grad = _gradient_from_weights(a, w)
            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # step‑size schedule
            eta = eta_start * (1.0 - (frac / phase1_frac)) + eta_end * (frac / phase1_frac)

            # exponentiated‑gradient update
            factor = np.exp(-eta * grad)
            a = a * factor

                if new_score is not None:          # improvement

                    if ref_score < best_score:
                        best_score = ref_score
                        best = ref_a.copy()
                        last_improve = now
                        print(f"[{elapsed:6.2f}s] refinement improved to {best_score:.6f}")

        a = np.clip(a, 0.0, 1000.0)

    # 5) Return the best sequence found (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
