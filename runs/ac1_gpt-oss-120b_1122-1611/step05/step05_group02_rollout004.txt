"""
Optimisation of a non‑negative step function by minimising the maximum
self‑convolution value.

The algorithm works on the probability simplex (a_i ≥ 0, Σ a_i = 1).  
At each iteration it:

1. Computes the full self‑convolution a*a (FFT‑based).
2. Forms a smooth approximation of the max‑operator:
   w_k ∝ exp( ( (a*a)_k – max ) / τ )   (soft‑max with temperature τ).
3. The gradient of the smoothed objective is the (linear) cross‑correlation
   of the weight vector w with a, i.e.   ∇f_j = 2·∑_i a_i·w_{i+j}.
   The correlation is obtained with a second FFT.
4. An Adam step is taken and the resulting vector is projected back onto the
   simplex.
5. Very occasional random jitter / block perturbations and restarts
   (after a period without improvement) help to escape local minima.

A few random Dirichlet initialisations (different concentration parameters)
as well as the previously‑found best sequence are used as starting points.
The best vector found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a power‑of‑two FFT."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    C = A * A
    conv = np.fft.irfft(C, n=L)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max over the convolution entries with temperature τ."""
    if tau <= 0.0:
        # degenerate case – behave like a hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    # subtract max to avoid overflow
    m = conv.max()
    ex = np.exp((conv - m) / tau)
    w = ex / ex.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇f where f is the τ‑soft‑max of the self‑convolution.
    The exact expression is
        g_j = 2 * Σ_i a_i * w_{i+j}
    which is a (linear) cross‑correlation of a and w.
    """
    n = a.size
    m = w.size          # = 2·n‑1
    L = _next_pow2(n + m - 1)
    # FFT of a
    A = np.fft.rfft(a, n=L)
    # FFT of w
    W = np.fft.rfft(w, n=L)
    # Linear cross‑correlation via IFFT of A * conj(W)
    corr_full = np.fft.irfft(A * np.conj(W), n=L)
    offset = m - 1                      # position of lag 0
    grad = 2.0 * corr_full[offset: offset + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    # Fast projection (Michelot / Duchi et al.)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_dirichlet(rng: np.random.Generator, n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex using a Dirichlet(α,…,α)."""
    if alpha <= 0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    g = rng.gamma(alpha, 1.0, size=n)
    s = g.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return g / s


def _jitter(a: np.ndarray, rng: np.random.Generator, scale: float = 0.12) -> np.ndarray:
    """Multiplicative log‑normal jitter of all coordinates."""
    noise = np.exp(rng.normal(0.0, scale, size=a.shape))
    a2 = a * noise
    a2 = np.clip(a2, 0.0, None)
    a2 /= a2.sum()
    return a2


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Choose the length of the vector (use previous solution if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size == 0 or not np.isfinite(prev).all():
            prev = None
    else:
        prev = None

    # target length – a few thousand works well with the time budget
    target_n = 6000
    if prev is not None:
        target_n = max(target_n, prev.size)

    n = int(target_n)

    # --------------------------------------------------------------
    # 2) Initialise a few starting points on the simplex
    # --------------------------------------------------------------
    candidates = []

    # a) previous best, padded / truncated to the new size
    if prev is not None:
        if prev.size < n:
            pad = np.zeros(n - prev.size, dtype=np.float64)
            a0 = np.concatenate([prev, pad])
        else:
            a0 = prev[:n].copy()
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s
        candidates.append(a0)

    # b) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # c) a few Dirichlet draws with different concentrations
    for alpha in (0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet(rng, n, alpha))

    # d) a simple two‑cluster extreme (mass at both ends)
    two = np.zeros(n, dtype=np.float64)
    two[0] = 0.5
    two[-1] = 0.5
    candidates.append(two)

    # --------------------------------------------------------------
    # 3) Optimisation state for each candidate
    # --------------------------------------------------------------
    states = []
    for a in candidates:
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum
        states.append(
            {
                "a": a,
                "m": np.zeros_like(a),          # Adam first moment
                "v": np.zeros_like(a),          # Adam second moment
                "t": 0,                         # iteration counter
            }
        )

    # --------------------------------------------------------------
    # 4) Global bookkeeping
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # hyper‑parameters (schedule will be linear in elapsed time)
    eta_start = 0.20          # initial Adam step size
    eta_end = 0.001           # final step size
    tau_start = 0.5           # initial soft‑max temperature
    tau_end = 1e-4            # final temperature
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    jitter_prob = 0.006
    block_perturb_prob = 0.008
    restart_gap = 40.0        # seconds without improvement before restart

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear schedules
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        tau = tau_start * (1.0 - elapsed / total) + tau_end * (elapsed / total)

        # iterate over all states (round‑robin)
        for state in states:
            if time.monotonic() >= deadline:
                break
            a = state["a"]
            # ----- evaluate current objective (sum(a)=1) -----
            conv = _self_convolution_fft(a)
            max_conv = float(conv.max())
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start_time:6.2f}s] new best score = {best_score:.6f}")

            # ----- compute smoothed gradient -----
            w = _softmax_weights(conv, tau)
            grad = _gradient_from_weights(a, w)

            # ----- Adam update -----
            t = state["t"] + 1
            m = beta1 * state["m"] + (1.0 - beta1) * grad
            v = beta2 * state["v"] + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = eta * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step

            # ----- projection onto the simplex -----
            a_proj = _project_simplex(a_new)

            # ----- occasional random jitter (global) -----
            if rng.random() < jitter_prob:
                a_proj = _jitter(a_proj, rng, scale=0.12)

            # ----- occasional block perturbation -----
            if rng.random() < block_perturb_prob:
                width = rng.integers(1, max(2, n // 12))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.15))
                block = a_proj[start:start + width] * factor
                a_proj[start:start + width] = block
                a_proj = np.clip(a_proj, 0.0, None)
                a_proj /= a_proj.sum()

            # ----- store updated state -----
            state["a"] = a_proj
            state["m"] = m
            state["v"] = v
            state["t"] = t

        # ----- soft restart if we have been stuck for a while -----
        if time.monotonic() - last_improve > restart_gap:
            if best_seq is not None:
                for state in states:
                    a_restart = _jitter(best_seq, rng, scale=0.10)
                    state["a"] = a_restart
                    state["m"] = np.zeros_like(a_restart)
                    state["v"] = np.zeros_like(a_restart)
                    state["t"] = 0
                last_improve = time.monotonic()
                print(f"[{last_improve - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best vector found (respect clipping enforced by evaluate_sequence)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
