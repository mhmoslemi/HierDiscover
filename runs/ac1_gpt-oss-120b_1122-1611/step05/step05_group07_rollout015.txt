import cvxpy as cp

def lp_step(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # Convolution constraints f * g <= M
    # For each k from 0 to 2n-2:
    # sum_{i=0}^{n-1} f[i] * g[k - i] <= M, where g index must be between 0 and n-1.
    # Use linear expression.
    constraints = []
    for k in range(2*n-1):
        expr = 0
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # for i in range(i_min,i_max+1):
        #     expr += f[i] * g[k-i]
        # Build using cp.sum(cp.multiply(...))
        # loop.
        terms = [f[i] * g[k-i] for i in range(i_min,i_max+1)]
        if terms:
            constraints.append(cp.sum(cp.hstack(terms)) <= M + 1e-12)  # some tolerance
    # Optionally also M should be nonneg.
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    return g.value, prob.value
```

def build_convolution_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # For each i, f[i] contributes to rows k = i + j for all j where 0 <= j < n (g_j)

        # Equivalent: for each i, row indices from i to i+n-1 (size n)

import scipy.sparse as sp

def build_conv_matrix(a):
    n = len(a)
    # Create list of diagonals: each diagonal offset i (0..n-1) has entries a[i] (of length n)
    # Row indices from i to i+n-1 (size 2n-1)
    # In sp.diags, we can specify offsets and values, but each diagonal corresponds to a constant offset across full rows.
    # The diagonal offset for a[i] is i: This corresponds to matrix where row = col + i
    # So shape (2n-1, n). We can generate using sp.diags with a reversed? Let's try:
    diagonals = []
    offsets = []
    for i, coeff in enumerate(a):
        # create a vector of length n with all entries = coeff
        diag = np.full(n, coeff)
        diagonals.append(diag)
        offsets.append(i)
    A = sp.diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
    return A
```

def _lp_step(a, M):
    n = a.size
    # Build sparse constraint matrix A
    # Use sp.diags
    diags = [np.full(n, coeff) for coeff in a]
    offsets = list(range(n))
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    # Solve LP
    c = -np.ones(n)
    bounds = [(0, None)] * n
    ub = np.full(2*n-1, M)
    res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    # Clip negative due to numerical error
    b = np.clip(b, 0, None)
    # Normalize to sum 1 (if sum is very small, fallback uniform)
    s = b.sum()
    if s <= 0:
        return None
    b = b / s
    return b
```

    # mu and log_sigma and log_w are numpy arrays length K
    sigma = np.exp(log_sigma)

    return a

            if rng.random() < 0.01:

        if cur_score < best_score - 1e-12:

            best = a.copy()

            elapsed = now - start

            print(f"[{now - start:6.2f}s] restart after stagnation, best {best_score:.6f}")

    # return best
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Fast search for a step‑function minimizing the evaluation function.
The algorithm has two stages.

Stage 1 – low‑dimensional simulated‑annealing:
    A distribution is represented as a mixture of a few Gaussian bumps
    (means, log‑sigmas, log‑weights).  A simple SA loop explores this
    parameter space quickly, using the exact max‑convolution objective
    (computed by FFT).  The best mixture found is used as a warm start.

Stage 2 – mirror‑descent + stochastic refinement on the full vector:
    Starting from the mixture’s discretisation, we run an entropy mirror
    descent (exponentiated‑gradient) on the probability simplex with a
    soft‑max approximation of the max‑norm.  The temperature (β) and step‑size
    (η) are linear schedules that become aggressive near the end.
    Periodic jitter, mass‑transfer moves and restarts keep diversity.
    The best full‑length vector seen during the whole run is returned.

The code respects the 1000 s time budget (5 s safety margin) and prints
progress messages with the current bound.
"""

import time
import numpy as np
import numpy.fft as fft

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    fa = fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0):
    """
    Search for a non‑negative step sequence with a low evaluation value.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    n = 3840                     # length used by the previous state‑of‑the‑art
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Stage 1 – low‑dimensional simulated annealing
    # ------------------------------------------------------------------
    K = 8                        # number of Gaussian components
    stage1_end = time.monotonic() + 0.30 * (deadline - time.monotonic())

    # initial mixture parameters
    mu = np.linspace(0, n - 1, K)                     # means
    log_sigma = np.log(np.full(K, n / 8.0))           # log‑standard‑deviations
    log_w = np.log(np.full(K, 1.0 / K))               # log‑weights (unnormalised)

    # grid used to evaluate the mixture
    ix = np.arange(n)

    def _build_from_params(mu_, ls_, lw_):
        """Construct a probability vector from mixture parameters."""
        sigma = np.exp(ls_)
        w_raw = np.exp(lw_)
        w = w_raw / w_raw.sum()
        diff = (ix[None, :] - mu_[:, None]) / sigma[:, None]
        gauss = np.exp(-0.5 * diff * diff)            # shape (K, n)
        a = (w[:, None] * gauss).sum(axis=0)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
        return a

    a = _build_from_params(mu, log_sigma, log_w)
    cur_conv = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * cur_conv.max()
    best_a = a.copy()
    best_score = cur_score

    # annealing schedules
    T_start = 0.2
    T_end = 1e-5
    sigma_mu_start = max(2.0, n / 10.0)
    sigma_mu_end = 0.5
    sigma_log_sigma_start = 0.5
    sigma_log_sigma_end = 0.02
    sigma_log_w_start = 0.5
    sigma_log_w_end = 0.02

    while time.monotonic() < stage1_end:
        now = time.monotonic()
        frac = (now - (deadline - (deadline - now))) / (stage1_end - (deadline - (deadline - now)))
        # fraction of the SA stage that has elapsed
        frac = (now - (deadline - (deadline - now))) / (stage1_end - (deadline - (deadline - now)))
        # compute schedules
        temperature = T_start * (1.0 - frac) + T_end * frac
        sigma_mu = sigma_mu_start * (1.0 - frac) + sigma_mu_end * frac
        sigma_ls = sigma_log_sigma_start * (1.0 - frac) + sigma_log_sigma_end * frac
        sigma_lw = sigma_log_w_start * (1.0 - frac) + sigma_log_w_end * frac

        # propose a new set of parameters (Gaussian perturbation)
        mu_new = mu + rng.normal(0.0, sigma_mu, size=K)
        mu_new = np.clip(mu_new, 0.0, n - 1)

        log_sigma_new = log_sigma + rng.normal(0.0, sigma_ls, size=K)
        log_w_new = log_w + rng.normal(0.0, sigma_lw, size=K)

        a_new = _build_from_params(mu_new, log_sigma_new, log_w_new)
        conv_new = _self_conv(a_new, fft_len_self)
        new_score = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if temperature > 0.0:
                accept = rng.random() < np.exp(-(new_score - cur_score) / temperature)

        if accept:
            mu, log_sigma, log_w = mu_new, log_sigma_new, log_w_new
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                elapsed = now - (deadline - (deadline - now))
                print(f"[{elapsed:6.2f}s] SA new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Stage 2 – high‑dimensional mirror‑descent + stochastic refinement
    # ------------------------------------------------------------------
    a = best_a.copy()                         # start from the best mixture
    grad_prev = np.zeros_like(a)

    # schedule for the mirror‑descent sub‑phase (≈60 % of the remaining time)
    phase1_end = time.monotonic() + 0.60 * (deadline - time.monotonic())
    eta_start = 8.0
    eta_end = 0.01
    beta_start = 0.5
    beta_end = 5000.0
    momentum = 0.95

    # bookkeeping for the overall best solution
    conv = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase 1 – smooth mirror‑descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedules for η and β
            frac = (now - (phase1_end - 0.60 * (deadline - (phase1_end - 0.60 * (deadline - time.monotonic()))))) / (phase1_end - (phase1_end - 0.60 * (deadline - (phase1_end - 0.60 * (deadline - time.monotonic())))))
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()

            # Soft‑max over the convolution entries
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            p = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

            # Gradient of the smooth max (convolution with reversed a)
            rev_a = a[::-1]
            fp = fft.rfft(p, n=fft_len_grad)
            fra = fft.rfft(rev_a, n=fft_len_grad)
            grad_full = fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # Exponentiated‑gradient (entropy mirror) update
            factor = np.exp(-eta * grad)
            a *= factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

            # occasional jitter to stay out of plateaus
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # Phase 2 – stochastic local refinement (simulated annealing)
        # --------------------------------------------------------------
        else:
            # simple linear cooling for the temperature
            temp_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = 0.08 * (1.0 - temp_frac) + 1e-5 * temp_frac
            temperature = max(temperature, 1e-8)

            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            k_max = int(conv.argmax())

            # ----- mass‑transfer move biased by contribution to the peak -----
            if rng.random() < 0.85:
                contrib = np.zeros_like(a)
                for i in range(n):
                    j = k_max - i
                    if 0 <= j < n:
                        contrib[i] = a[i] * a[j]
                total = contrib.sum()
                if total > 0.0:
                    prob = contrib / total
                    donor = rng.choice(np.arange(n), p=prob)
                else:
                    donor = rng.integers(0, n)

                if a[donor] > 1e-12:
                    rec = rng.integers(0, n - 1)
                    if rec >= donor:
                        rec += 1
                    delta = rng.random() * 0.25 * a[donor]   # up to 25 % transfer
                    a_new = a.copy()
                    a_new[donor] -= delta
                    a_new[rec] += delta
                    conv_new = _self_conv(a_new, fft_len_self)
                    new_score = 2.0 * n * conv_new.max()
                    if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                        a = a_new
                        cur_score = new_score

            # ----- occasional global jitter -----
            else:
                a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - (deadline - (deadline - time.monotonic()))
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Restart after a long period of stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            # moderate multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - (deadline - (deadline - time.monotonic()))
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final fine‑grained local search (tiny block perturbations)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 2.0:   # keep a tiny safety margin
        a_try = _random_block_perturb(best, rng)
        conv_try = _self_conv(a_try, fft_len_self)
        score_try = 2.0 * n * conv_try.max()
        if score_try < best_score:
            best = a_try
            best_score = score_try
            elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
            print(f"[{elapsed:6.2f}s] tiny perturbation improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
