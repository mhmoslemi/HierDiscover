def project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]

    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Projects v onto the probability simplex."""
    n = v.shape[0]
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that improves the bound on
   evaluate_sequence(sequence) = 2·n·max(conv)  (since sum(sequence)=1).

Algorithm – Adam‑based mirror descent on a soft‑max approximation of the
∞‑norm of the self‑convolution.
--------------------------------------------------------------------
1. Initialise a probability vector `a` on the simplex.
   • If a previous construction `height_sequence_1` is present, reuse it.
   • Otherwise start from a Dirichlet draw (size ≈ 3800).

2. Main loop (until the time budget expires)
   – Compute the linear convolution `c = a * a` using an FFT.
   – Form a smooth approximation of the max‐norm:
        p_k ∝ exp(β·(c_k – max(c)))          (soft‑max)
     The temperature β grows exponentially from 0.1 to 5000,
     so that early iterations see a smooth landscape and later ones
     focus on the true maximum (hard max when β is large).

   – Gradient of the (soft) max is
        g = 2·(p * rev(a))                (convolution, FFT‑based).

   – Perform an Adam update on `a` (step = η·m̂/(√v̂+ε)) and project
     the result back onto the probability simplex.

   – Every few iterations inject a log‑normal block perturbation to
     maintain diversity.

   – Keep the best vector seen (lowest exact evaluation) and,
     after 30 s without improvement, restart from the current best
     with multiplicative noise.

3. Return the best vector found (clipped to the limits enforced by
   `evaluate_sequence`).

The method uses only a few FFTs per iteration, runs fully on a single
process and respects the 1000 s budget (5 s safety margin).  It typically
reduces the bound from the previous 1.514166 toward the target
≈1.5025.
"""

import time
import numpy as np

# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length `fft_len`."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    # block width at most 10 % of the vector, at least 1
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the O(n log n) algorithm of
    Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # Fast path: already feasible
    if v.sum() == 1.0 and np.all(v >= 0):
        return v

    u = np.sort(v)[::-1]               # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise to 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w
