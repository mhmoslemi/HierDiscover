"""
Propose a step‑function by flattening its self‑convolution.
-----------------------------------------------------------------------
The objective 2·n·max(conv(a,a)) is scale‑invariant, so we keep the
probability simplex a_i ≥ 0, Σ a_i = 1.  The algorithm repeatedly

* evaluates the full self‑convolution (FFT based),
* builds a soft‑max weighting of the convolution values (temperature α),
* computes a smoothed sub‑gradient  ∇ = 2·(w * a)  (convolution of the
  weights with the current vector),
* performs an exponentiated‑gradient (mirror–descent) step,
* renormalises to the simplex,
* occasionally applies random block perturbations or pair‑wise mass
  transfers, and
* restarts with jitter when no improvement has been seen for a while.

The soft‑max (controlled by α) lets the update react to several large
convolution entries simultaneously, helping the search escape the
plateau where only the single maximal entry would be used (the behaviour
of the previous baseline).  With a linear schedule for the steps‑size η
and the temperature α the method quickly refines a good solution and
continues to improve it until the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``).
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a real vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = x.size + y.size - 1
    fft_len = 1 << ((size - 1).bit_length())
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[:size]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    The implementation follows the description above.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Choose a starting vector (use previous best if available)
    # --------------------------------------------------------------
    default_n = 1024
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    max_conv = float(conv.max())
    cur_score = 2.0 * n * max_conv                # sum(a)=1 ⇒ score = 2·n·max(conv)
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    # keep a small safety margin so that we can always return a result
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    last_improve = start_time

    # --------------------------------------------------------------
    # 3) Hyper‑parameters
    # --------------------------------------------------------------
    eta_start = 1.0          # initial exponentiated‑gradient stepsize
    eta_end = 0.001          # final stepsize
    alpha_start = 1.0        # start of soft‑max temperature
    alpha_end = 120.0        # final temperature (more focused on true max)
    block_perturb_prob = 0.015
    pair_move_prob = 0.07
    restart_gap = 25.0       # seconds without improvement → restart
    temp_start = 0.5         # temperature for occasional uphill moves
    temp_end = 0.001

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- evaluate current state ---------------------------------
        conv = _conv_self_fft(a, fft_len)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        # ----- keep the best solution ---------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule: linear annealing for η, α and temperature ----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        alpha = alpha_start * (1.0 - elapsed / total) + alpha_end * (elapsed / total)
        temp = temp_start * (1.0 - elapsed / total) + temp_end * (elapsed / total)

        # ----- soft‑max weighting of the convolution entries ----------
        # w_k ∝ exp(α·(conv_k − max_conv)); the max entry gets weight 1
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = w
        else:
            w_norm = w / w_sum

        grad_full = _conv_full_fft(w_norm, a)          # length = 3·n‑2
        grad = 2.0 * grad_full[:n]

        # ----- exponentiated‑gradient (mirror descent) update ----------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation --------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pair‑wise mass transfer -----------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate /= a_candidate.sum()
                    conv_cand = _conv_self_fft(a_candidate, fft_len)
                    cand_score = 2.0 * n * conv_cand.max()
                    # accept if better or with a small uphill probability
                    if cand_score < cur_score or rng.random() < np.exp(
                        (cur_score - cand_score) / max(temp, 1e-12)
                    ):

        # ----- soft restart if we have been stuck ----------------------
        if now - last_improve > restart_gap:
            # start from the current best but add a small multiplicative jitter
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
