evaluate_sequence:

- Input list sequence. Checks for type, emptiness, each element type (int/float, not bool), not NaN/inf.

- Then does:

    sequence = [float(x) for x in sequence]

"""
Propose a step–function (non‑negative vector) whose evaluation
``evaluate_sequence`` is as small as possible.

The algorithm works in two phases:

1. **LP “flattening” phase** –  
   Starting from the current vector `a` (taken from the global
   ``height_sequence_1`` if it exists, otherwise uniform) we compute the
   current maximum of the self‑convolution `M = max(a*a)`.  
   With `M` fixed we solve a linear program

        maximise   Σ b_i
        subject to  a * b ≤ M      (componentwise)
                    b_i ≥ 0

   The matrix of the convolution `a*b` is built as a sparse Toeplitz
   matrix; the LP is solved by SciPy’s HiGHS solver.  The optimal
   solution `b` spreads the mass more uniformly while respecting the
   same peak `M`.  By mixing `a ← (1‑t)·a + t·b` (t slowly decreasing)
   we obtain a new vector with a smaller self‑convolution peak.
   A handful of such LP steps already reduces the objective dramatically.

2. **Greedy fine‑tuning phase** –  
   After the LP phase we fall back to a fast greedy mass‑transfer
   routine (similar to the one that produced the previous best
   1.514…) but starting from the much better vector obtained above.
   Random block perturbations and occasional restarts keep the search
   from stagnating.

Both phases are limited by the 1000 s budget; the routine prints
intermediate scores so the progress can be monitored.  The final
vector is clipped to the limits of ``evaluate_sequence`` before being
returned.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
#  Helper routines (independent, no closures)
# --------------------------------------------------------------

def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    # for moderate n a direct convolution is fast enough
    return np.convolve(a, a)


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    i: int,
                    j: int,
                    delta: float) -> np.ndarray:
    """Exact update of the convolution after moving `delta` mass from i to j."""
    n = a.size
    new_conv = conv.copy()
    # linear part: -2·Δ·a shifted from i, +2·Δ·a shifted from j
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a
    # quadratic contributions
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          max_width_frac: float = 0.2,
                          sigma: float = 0.12) -> np.ndarray:
    """Multiplicative log‑normal jitter on a random contiguous block."""
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _max_convolution(a: np.ndarray):
    """Return (max value , full convolution) using an FFT (O(n log n))."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two is faster for the FFT backend
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:size]
    return float(conv.max()), conv


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    # implementation of the algorithm from
    #   "Efficient Projections onto the l1-Ball for Learning in High Dimensions"
    #   (Duchi et al., 2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # falls back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
#  Linear‑programming “flattening” step
# --------------------------------------------------------------

def _lp_maximize_mass(a: np.ndarray, M: float):
    """
    Solve
        maximise   Σ b_i
        subject to  a * b ≤ M   (componentwise)
                    b_i ≥ 0
    Returns the optimal vector b (may sum to more than 1).
    """
    try:
        import scipy.sparse as sp
        import scipy.optimize as opt
    except Exception as e:                     # pragma: no cover
        raise RuntimeError("SciPy is required for the LP phase") from e

    n = a.shape[0]

    # Build the sparse Toeplitz convolution matrix:
    # row k, col j contains a_{k-j} if 0 ≤ k-j < n, else 0.
    # The matrix has n diagonals (offset = 0 … n‑1) each with a constant
    # value equal to the corresponding entry of a.
    data = np.broadcast_to(a[:, None], (n, n))          # shape (n, n)
    offsets = np.arange(n)                              # 0 … n‑1
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    ub = np.full(2 * n - 1, M, dtype=np.float64)       # RHS of inequality
    c = -np.ones(n, dtype=np.float64)                  # maximise Σb  → minimise -Σb
    bounds = [(0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                     method="highs", options={"presolve": True})
    if not res.success:                               # pragma: no cover
        raise RuntimeError(f"LP failed: {res.message}")

    return res.x

    bounded by ``budget_s`` seconds.

        # The mixture factor is slowly reduced to fine‑tune later.

    # --------------------------------------------------------------
    n = a.size
    conv = _self_convolution(a)
    cur_score = 2.0 * n * conv.max()
    last_improve = time.monotonic()

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        k_max = int(np.argmax(conv))

        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        partner = np.zeros_like(a)
        partner[valid] = a[k_max - idx[valid]]

        donor_mask = a > 1e-12
        donor_score = partner * donor_mask.astype(float)

        if donor_score.max() == 0.0:
            # fallback: random tiny transfer
            i = rng.integers(0, n)
            if a[i] <= 0.0:

            j = rng.integers(0, n - 1)

            delta = rng.random() * 0.3 * a[i]
            new_conv = _apply_transfer(a, conv, i, j, delta)
            new_score = 2.0 * n * new_conv.max()
            if new_score < cur_score:
                a[i] -= delta
                a[j] += delta
                conv = new_conv
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{elapsed():.2f}s] new best (fallback) {best_score:.6f}")
            continue

        donor = int(np.argmax(donor_score))

        receiver_candidates = np.where(partner == partner.min())[0]
        if donor in receiver_candidates:
            receiver_candidates = receiver_candidates[receiver_candidates != donor]
        if len(receiver_candidates) == 0:
            receiver = (donor + 1) % n
        else:
            receiver = int(rng.choice(receiver_candidates))

        # adaptive step size – larger early, smaller later
        frac = elapsed() / (deadline - start_time)

        max_delta = a[donor] * max_frac
        if max_delta < 1e-12:

        delta = rng.random() * max_delta

        new_score = 2.0 * n * new_conv.max()

            a[donor] -= delta
            a[receiver] += delta

            cur_score = new_score
            if cur_score < best_score:

                best = a.copy()

                print(f"[{elapsed():.2f}s] new best {best_score:.6f}")

        # occasional random block perturbation to jump out of plateaus
        if rng.random() < 0.015:
            a_pert = _random_block_perturb(a, rng)
            conv_pert = _self_convolution(a_pert)
            pert_score = 2.0 * n * conv_pert.max()
            if pert_score < cur_score:
                a = a_pert
                conv = conv_pert
                cur_score = pert_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{elapsed():.2f}s] block‑perturb improved best {best_score:.6f}")

        # restart if we have been idle for a while
        if time.monotonic() - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _self_convolution(a)
            cur_score = 2.0 * n * conv.max()
            last_improve = time.monotonic()
            print(f"[{elapsed():.2f}s] restart – score reset to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 3) Return the best vector found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
