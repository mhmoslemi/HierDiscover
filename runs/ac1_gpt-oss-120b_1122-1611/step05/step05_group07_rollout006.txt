"""
Hybrid sub‑gradient mirror descent + targeted local moves.

* Phase 1 (≈ 65 % of the budget) – exact sub‑gradient of the
  ‖a*a‖∞ term.
  The gradient for a single maximising shift k is g_i = 2·a_{k‑i}
  (zero outside the support).  An exponentiated‑gradient (entropy)
  update keeps the vector on the simplex; the step‑size η decays
  linearly from a large start to a tiny end.  An occasional
  log‑normal block jitter prevents premature convergence.

* Phase 2 – stochastic refinement.
  The current peak index k_max is located, the contribution of each
  entry to that peak is a_i·a_{k_max‑i}.  A donor is drawn proportional
  to its contribution and a recipient is chosen among indices that
  contribute little to the peak.  Moving a small amount δ of mass from
  donor to recipient updates the convolution linearly (O(n) slice
  updates).  The best move among a few candidates is taken; a global
  block perturbation is applied sporadically.

* Restarts – after 30 s without improvement the current best vector is
  perturbed by multiplicative log‑normal noise and the search resumes.

The routine works entirely on the simplex (∑a=1) so the evaluation
score is simply 2·n·max(a*a).  The algorithm starts from the previous
construction if available, otherwise from a Dirichlet sample of length
≈ 5 000.  All helper functions are top‑level, there are no closures,
and the function respects the 1000 s time budget (with a 5 s safety
margin).  The returned list is clipped to the limits enforced by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    The returned array has length 2·len(a)‑1 (the linear convolution).
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may produce tiny negative values; clamp.
    conv[conv < 0] = 0.0
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# ----------------------------------------------------------------------

                        continue
                    max_delta = max_delta_frac * a[donor]
                    if max_delta <= 0.0:
                        continue
                    delta = rng.random() * max_delta

                    # linear update of the convolution (no FFT)
                    conv_tmp = conv.copy()
                    conv_tmp[donor:donor + n] -= delta * a
                    conv_tmp[rec:rec + n] += delta * a
                    new_score = 2.0 * n * conv_tmp.max()

                    if new_score < best_local_score:
                        best_local_score = new_score
                        best_rec = rec
                        best_delta = delta

                if best_rec is not None:
                    # commit the best move
                    a[donor] -= best_delta
                    a[best_rec] += best_delta
                    # recompute the exact convolution to keep it accurate
                    conv = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * conv.max()
                else:
                    # occasional global jitter when no improving move is found
                    if rng.random() < 0.05:
                        a = _random_block_perturb(a, rng)
                        conv = _self_conv(a, fft_len_self)
                        cur_score = 2.0 * n * conv.max()
            else:
                # donor empty – fall back to a global jitter
                if rng.random() < 0.05:
                    a = _random_block_perturb(a, rng)
                    conv = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > temp_restart:
            # perturb the best vector and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            print(
                f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
