def propose_candidate(seed=..., budget_s=1000.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.1

    best_score = np.inf
    best_seq = None

    # Parameters: list of lengths to try
    lengths = [256, 512, 1024, 2048] // maybe limit to 2560
    for n in lengths:
        # number of outer runs for each length
        for run in range(3):
            a = rng.random(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            # EG parameters
            eta0 = 0.5
            # smoothing temperature schedule
            tau_mul = 0.5
            tau = max(np.convolve(a, a).max() * tau_mul, 1e-12)
            iter = 0
            while time.monotonic() < deadline:
                iter += 1

                conv = np.convolve(a, a)
                max_conv = conv.max()
                # compute softmax weights for smooth max
                shifted = (conv - max_conv) / tau
                w = np.exp(shifted)
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(conv, 1.0/conv.size)
                else:
                    w = w / w_sum

                # gradient of smooth max: grad_i = 2 * correlate(w, a)
                grad = 2.0 * np.correlate(w, a, mode='full')[n-1:n-1+n]

                # exponentiated update
                a = a * np.exp(-eta0 * grad)
                # renormalize to simplex
                a_sum = a.sum()
                if a_sum == 0:
                    a = np.full(n, 1.0/n)
                else:
                    a = a / a_sum

                # occasionally add noise
                if rng.random() < 0.01:
                    noise = rng.normal(0.0, 0.02, size=n)
                    a = a + noise
                    a = np.maximum(a, 0)
                    a = a / a.sum()

                # compute true evaluation
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[best] n={n}, run={run}, iter={iter}, score={score:.6f}")

                # adjust tau
                tau = max(tau * 0.999, 1e-12)

                # early stop for this run if no improvement for many iterations
                if iter % 2000 == 0:
                    # If best_score unchanged for some iterations, break
                    pass

                # also break if best_score below target 1.505 maybe
                if best_score <= 1.5025:
                    break
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    return best_seq.tolist()

def propose_candidate(seed=0, budget_s=1000.0):
    import time, numpy as np, scipy.optimize
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.1

    best_score = float('inf')
    best_seq = None

    # list of n values to try, try small first for speed
    n_list = [256, 512, 1024]
    for n in n_list:
        # number of independent runs per n
        runs = 3
        for run in range(runs):
            # initialize f randomly from Dirichlet
            f = rng.random(n)
            f = np.maximum(f, 0)
            if f.sum() == 0:
                f = np.full(n, 1.0/n)
            else:
                f = f / f.sum()
            # mixing weight schedule
            t = 0.3   # can adjust
            # iteration counter for stagnation detection
            last_improve = 0
            iter_counter = 0
            while time.monotonic() < deadline:
                iter_counter += 1
                # compute self-convolution
                conv_f = np.convolve(f, f)
                M = conv_f.max()
                # build A matrix
                rows = 2*n - 1
                A = np.zeros((rows, n), dtype=np.float64)
                # fill A
                for j in range(n):
                    A[j:j+n, j] = f
                # linear program: maximize sum(b) => minimize -1^T b
                c = -np.ones(n, dtype=np.float64)
                b_ub = np.full(rows, M, dtype=np.float64)
                # Solve LP
                res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                            bounds=[(0, None)]*n,
                                            method='highs',
                                            options={'presolve': True})
                if not res.success:
                    # fallback: keep current f
                    # optionally break
                    break
                b = res.x
                # normalize b to sum 1
                sum_b = b.sum()
                if sum_b == 0:
                    g = np.full(n, 1.0/n)
                else:
                    g = b / sum_b
                # mixing step: decrease t over iterations
                f_new = (1 - t) * f + t * g
                # renormalize to sum=1 (should already be)
                f_new = np.maximum(f_new, 0)
                f_new = f_new / f_new.sum()
                # evaluate
                score = evaluate_sequence(f_new.tolist())
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = f_new.copy()
                    last_improve = iter_counter
                    print(f"[improve] n={n}, run={run}, iter={iter_counter}, score={score:.6f}")
                # simple early stop
                if best_score <= 1.5025:
                    return best_seq.tolist()
                # update mixing parameter slowly

* The evaluation score is 2·n·max(conv(a,a)) / (∑a)^2.
  With the ℓ₁‑norm fixed to 1 the denominator is constant,
  so we minimise the maximal entry of the self‑convolution.
* For a fixed length n we iteratively improve a distribution `a`
  by the classic “LP‑mixing” scheme (see the problem statement):
      – given `a` we compute M = max_k (a∗a)_k,
      – solve the linear programme
            maximise    ∑ b_j
            subject to  a ∗ b  ≤  M   (entry‑wise),
                        b_j ≥ 0,
        which finds a distribution `b` that can carry more mass
        without increasing the current peak M,
      – mix the current vector with the optimiser:
            a ← (1‑t)·a + t·b̂   where b̂ = b / ∑b  (renormalised)
        and slowly decay the mixing coefficient `t`.
  This guarantees a non‑increasing sequence of maxima M and
  converges towards the theoretical lower bound 2 n/(2 n‑1).
* To avoid getting stuck in a poor basin we perform several
  independent restarts with random Dirichlet initialisations,
  and we progressively enlarge the vector length (256 → 512 → 1024),
  up‑sampling the best solution from the previous stage as a warm start.
* Time is limited (≈1000 s).  The outer loop stops as soon as the
  target score 1.5025 is reached or the budget expires.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the linear operator C such that (C @ b) = a ∗ b (linear convolution)
    for vectors of length n = len(a).  The result has shape (2n‑1, n).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes a shifted copy of `a`
    for j in range(n):
        C[j:j + n, j] = a
    return C


# ----------------------------------------------------------------------
def _lp_optimize(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise    ∑ b_j
        subject to  a ∗ b ≤ M   (entry‑wise)
                    b ≥ 0
    Returns the optimal b (not normalised).  If the LP fails, returns `a`.
    """
    n = a.shape[0]
    C = _build_conv_matrix(a)               # (2n‑1) × n
    # Objective: minimise -∑b   (linprog solves minimisation problems)
    c = -np.ones(n, dtype=np.float64)
    # Upper‑bounds for each convolution entry
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    # Bounds for each variable b_j ≥ 0
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast interior‑point/simplex hybrid)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    # Fallback – return the current vector unchanged
    return a.copy()


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation score.  The function respects the 1000 s time budget and
    returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    **kwargs
        Ignored (kept for compatibility).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # overall deadline (tiny safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # Length schedule – start small for cheap LP solves, then enlarge.
    length_schedule = [256, 512, 1024]

    # best solution across all runs
    best_seq = None
    best_score = float('inf')

    for n in length_schedule:
        # number of independent random restarts for this length
        restarts = 4 if n <= 512 else 3

        for run in range(restarts):
            # initialise a (if we have a previous best, upsample it)
            if best_seq is not None and len(best_seq) < n:
                a = _upsample(np.asarray(best_seq, dtype=np.float64), n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            else:
                # random Dirichlet point (exponential sampling)
                a = rng.exponential(scale=1.0, size=n)
                a = a / a.sum()

            # mixing coefficient (starts relatively aggressive, then decays)
            t = 0.6
            t_decay = 0.9995

            # stagnation handling
            last_improve_iter = 0
            iteration = 0

            while time.monotonic() < deadline:
                iteration += 1

                # current max of the self‑convolution
                conv_self = np.convolve(a, a)
                M = conv_self.max()

                # solve the LP to obtain a better direction
                b = _lp_optimize(a, M)

                # normalise the LP solution to unit ℓ₁‑norm
                b_sum = b.sum()
                if b_sum == 0.0 or not np.isfinite(b_sum):
                    # fallback to the current a
                    g = a.copy()
                else:
                    g = b / b_sum

                # convex combination (the “mixing” step)
                a_new = (1.0 - t) * a + t * g
                # project back onto the simplex (numerical safety)
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()

                # evaluate the true objective
                score = evaluate_sequence(a_new.tolist())

                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_new.copy()
                    last_improve_iter = iteration
                    print(f"[improve] n={n:4d} run={run} iter={iteration:,} "
                          f"t={t:.4f} score={best_score:.6f}")

                # early exit if we already beat the target
                if best_score <= 1.5025:
                    print("[target reached] returning early")
                    return best_seq.tolist()

                # update mixing weight
                t = max(t * t_decay, 0.01)

                # check for stagnation – if no improvement for a while,
                # restart this run with a fresh random start.
                if iteration - last_improve_iter > 2000:
                    # random re‑initialisation
                    a = rng.exponential(scale=1.0, size=n)
                    a = a / a.sum()
                    t = 0.6
                    last_improve_iter = iteration
                else:
                    a = a_new

                # safety check – if we have spent too much time, break
                if time.monotonic() >= deadline:
                    break

        # after finishing all restarts for this length, continue with the next size

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should not happen) return a unit mass.
    if best_seq is None:
        return [1.0]
    return best_seq.tolist()
