def block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start+width
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

"""
Propose a candidate step function that improves the self‑convolution peak.

The algorithm
---------------
* Work on a probability vector `w` (non‑negative, sum = 1).  The final
  sequence is obtained by scaling `w` to have ℓ¹‑norm √(2 n) – this is
  exactly the scaling used by the scoring function.
* Initialise `w` with the Catalan distribution or, if available,
  with the best sequence from a previous run (`height_sequence_1`).
* Minimise the smooth approximation of the objective

        fβ(w) = (1/β)·log Σₖ exp(β·(w*w)[k])

  whose gradient is a weighted sum of shifted copies of `w`.  The weight
  vector `p` is the (stable) soft‑max of the convolution values and
  becomes concentrated on the maximal entries when β grows.
* Use an Adam‑style update (first‑ and second‑moment estimates) and
  project the iterates back onto the probability simplex after each step.
  A small symmetrisation step (`w←(w+w[::-1])/2`) reduces the search
  space without harming optimality – the optimal solution is known to be
  symmetric.
* β is increased linearly from a small value (≈0.5) to a large value
  (≈500) during the run, turning the smoothed problem gradually into the
  original max‑convolution problem.
* If the optimisation stalls, a cheap random block‑perturbation is
  applied to the current best point (followed by a reset of the Adam
  moments) – this helps escape shallow plateaus.
* The best vector found is finally scaled to ℓ¹‑norm √(2 n) and returned
  as a plain list of floats.

The whole optimisation runs for the allotted time budget (≈1000 s) and
keeps the best sequence discovered so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – defined at the top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` terms of a_i = C(2i,i)/4**i.
    Recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) is numerically stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negative round‑off errors for safety.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}.
    Uses the O(n log n) algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Fall back to the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random perturbation used as a jump when the search stalls."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # minimal block width
    hi = max(2, n // 12)           # maximal block width
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # multiplicative factor
    cand[start:stop] *= factor
    if rng.random() < 0.35:        # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) whose
    `evaluate_sequence` score is as low as possible within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # ------------------------------
    # Choose the length n.
    # ------------------------------
    if 'height_sequence_1' in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(seq0.size)
            # Normalise to a probability vector (ℓ¹‑norm = 1)
            w = seq0 / seq0.sum()
        except Exception:
            n = 8000
            w = catalan_sequence(n).astype(np.float64) / np.sum(catalan_sequence(n))
    else:
        n = 8000
        w = catalan_sequence(n).astype(np.float64) / np.sum(catalan_sequence(n))

    # Ensure we really are on the simplex and force symmetry.
    w = w.astype(np.float64)
    w = simplex_projection(w)
    w = (w + w[::-1]) * 0.5

    # ------------------------------
    # Adam optimiser state.
    # ------------------------------
    m_vec = np.zeros_like(w)            # first moment
    v_vec = np.zeros_like(w)            # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05                      # base learning rate before decay

    # ------------------------------
    # Initial best score.
    # ------------------------------
    b = convolve_fft(w, w)
    best_score = 2.0 * n * b.max()      # exactly the value returned by evaluate_sequence
    best_w = w.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------
    # Temperature schedule for the soft‑max smoothing.
    # ------------------------------
    beta_start = 0.5
    beta_end = 500.0

    iteration = 0
    no_improve = 0
    last_status = start_time

    # Main optimisation loop.
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # (1) Convolution and its maximum.
        b = convolve_fft(w, w)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # (2) Temperature (β) – grows linearly with the iteration counter.
        frac = min(iteration / 80000.0, 1.0)          # ramp over ~80 k iterations
        beta = beta_start + (beta_end - beta_start) * frac

        # (3) Soft‑max probabilities over convolution entries.
        #    Using the max for numerical stability.
        exp_terms = np.exp(beta * (b - max_b))
        p = exp_terms / exp_terms.sum()              # length 2n‑1

        # (4) Gradient of the smoothed objective.
        #     g = 2·conv(w_rev, p)[n‑1 : n‑1+n]
        grad = 2.0 * convolve_fft(w[::-1], p)[n - 1:n - 1 + n]

        # (5) Adam update (with a simple decay of the step size).
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)
        t = iteration + 1
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)

        # Decay the learning rate like 1/√t to keep convergence stable.
        lr = base_lr / np.sqrt(t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step

        # (6) Projection onto the probability simplex and symmetrisation.
        w_new = simplex_projection(w_new)
        w_new = (w_new + w_new[::-1]) * 0.5
        w = w_new

        # (7) Keep the best point seen so far.
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:06d}, β={beta:.1f}, score={best_score:.6f}")
        else:
            no_improve += 1

        # (8) Random block perturbation if we have been stagnant too long.
        if no_improve > 6000:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand = (cand + cand[::-1]) * 0.5
            w = cand
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            bc = convolve_fft(w, w)
            cand_score = 2.0 * n * bc.max()
            if cand_score < best_score:
                best_score = cand_score
                best_w = w.copy()
                print(f"[perturb] iter={iteration:06d}, new best score={best_score:.6f}")
            no_improve = 0

        # (9) Periodic status report (every ~30 s).
        if time.monotonic() - last_status > 30.0:
            print(f"[status] iter={iteration:06d}, best_score={best_score:.6f}")
            last_status = time.monotonic()

        iteration += 1

    # ------------------------------------------------------------------
    # Scale the best probability vector to ℓ¹‑norm √(2 n) as required by the
    # scoring routine, and report the final value.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score (pre‑scale)={best_score:.6f}, "
          f"evaluate={final_score:.6f}, iterations={iteration}")

    # Convert to a plain Python list.
    return [float(x) for x in final_seq]
