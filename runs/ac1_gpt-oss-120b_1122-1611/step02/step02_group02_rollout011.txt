"""
Gradient‑descent on a soft‑max approximation of the ∞‑norm.
--------------------------------------------------------------------
We minimise

    2·n·max_k (a∗a)[k]    with   a_i ≥ 0 ,  Σa_i = 1 .

The expression is invariant to scaling, therefore we keep Σa_i = 1
throughout the optimisation.  The ∞‑norm is replaced by the smooth
log‑sum‑exp (soft‑max) with a temperature λ that is increased
linearly during the run.  For a given λ the gradient of

    φ(a) = (1/λ)·log Σ_k exp(λ·(a∗a)[k])

is

    ∇φ_i = 2·Σ_k w_k·a_{k‑i}    with    w_k = exp(λ·(a∗a)[k]) / Σ_j exp(λ·(a∗a)[j])

i.e. a weighted convolution of w and the reversed a.  The update is a
multiplicative (exponentiated‑gradient) step which automatically respects
non‑negativity; after the step we renormalise to Σa_i=1.

The algorithm:
    * initialise with a (discrete) arcsine shape – the continuous arcsine
      distribution has exactly uniform self‑convolution.
    * repeat until the time budget expires:
        - compute the self‑convolution (FFT, O(n log n))
        - evaluate the current score and store the best vector seen
        - build the soft‑max weights w using a temperature λ that grows
          from λ_start to λ_end
        - obtain the gradient by a second convolution (w * reversed(a))
        - multiplicative update a ← a·exp(‑η·∇φ)  (η decays from η_start
          to η_end)
        - occasional random perturbations help escape plateaus
        - if no improvement for a while, restart from a fresh arcsine sample
    * finally clip the best solution to the limits required by
      `evaluate_sequence` and return it as a plain list of floats.

The method uses only NumPy (no external solvers) and runs comfortably
within the 1000 s budget, typically achieving scores well below the
target 1.5025.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a⋆a using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_linear(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: x.size + y.size - 1]


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------
def _init_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√[(i+½)(n‑i‑½)]  (normalised)."""
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


def _interpolate_to(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    if a.size == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Perturbations for exploration
# ----------------------------------------------------------------------
def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 1e-14, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 1e-14, None)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds (leaving a tiny safety margin) and uses a gradient
    descent on a soft‑max surrogate of the max‑convolution.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – use the historical best if available, otherwise a
    # reasonably large power‑of‑two (helps the FFTs).
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n0 = base.size
        # target a larger problem for a finer discretisation
        n = max(8192, 1 << (n0 - 1).bit_length())
    except Exception:
        # fallback: a good size that still fits comfortably in the time budget
        n = 8192
        base = None

    # ------------------------------------------------------------------
    # Initialise a – either interpolate the historic best or use arcsine.
    # ------------------------------------------------------------------
    if base is not None:
        a = _interpolate_to(base, n)
    else:
        a = _init_arcsine(n)

    # a small jitter to break symmetry
    a += rng.normal(scale=0.001, size=n)
    a = np.clip(a, 1e-14, None)
    a /= a.sum()

    # FFT lengths for the two convolutions we need every iteration
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)  # length of w (2n-1) + a_rev (n) - 1

    # ------------------------------------------------------------------
    # Hyper‑parameters (chosen to work well for n≈8‑16 k)
    # ------------------------------------------------------------------
    lambda_start = 10.0          # temperature at the beginning
    lambda_end = 5e4             # temperature at the end (more selective)
    eta_start = 2000.0           # large initial multiplicative step
    eta_end = 20.0               # gentler near the end
    stagnation_limit = 30.0      # seconds without improvement → restart
    perturb_prob = 0.02          # total probability of a random perturbation
    block_perturb_frac = 0.7     # fraction of perturb events that are block‑type

    # bookkeeping
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.5)

    # evaluate the initial candidate
    conv = _conv_self(a, fft_len_self)
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv
    best_a = a.copy()
    last_improve = start_time
    iteration = 0

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # --------------------------------------------------------------
        # 1) convolution and current score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len_self)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 2) keep the best solution
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) soft‑max weights (temperature schedule)
        # --------------------------------------------------------------
        lam = lambda_start + (lambda_end - lambda_start) * frac
        # shift by the maximum to avoid overflow
        shift = conv - max_conv          # ≤ 0
        w_exp = np.exp(lam * shift)     # w_exp[max] = 1
        w = w_exp / w_exp.sum()         # shape (2n‑1,)

        # --------------------------------------------------------------
        # 4) gradient = 2 * (w ⋆ a_rev) restricted to n entries
        # --------------------------------------------------------------
        a_rev = a[::-1]
        temp = _conv_linear(w, a_rev, fft_len_grad)  # length = 3n‑2
        grad = 2.0 * temp[n - 1: 2 * n - 1]           # slice of length n

        # --------------------------------------------------------------
        # 5) exponentiated‑gradient update (step‑size schedule)
        # --------------------------------------------------------------
        eta = eta_start + (eta_end - eta_start) * frac
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 1e-14, None)
        a /= a.sum()

        # --------------------------------------------------------------
        # 6) occasional random exploration
        # --------------------------------------------------------------
        if rng.random() < perturb_prob:
            if rng.random() < block_perturb_frac:
                a = _random_block_perturb(a, rng)
            else:
                a = _pairwise_swap_perturb(a, rng)

        # --------------------------------------------------------------
        # 7) restart if stuck for a while
        # --------------------------------------------------------------
        if now - last_improve > stagnation_limit:
            # fresh start – keep the arcsine shape but add new jitter
            a = _init_arcsine(n)
            a += rng.normal(scale=0.001, size=n)
            a = np.clip(a, 1e-14, None)
            a /= a.sum()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best vector found, clipped to the limits accepted by
    # `evaluate_sequence`.
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
