"""
Candidate generator for the step‑function minimisation task.

Approach
--------
1. Initialise a non‑negative sequence `a` of length `n = 1024`.
   * If a previous best sequence `height_sequence_1` is available it is used,
     otherwise a random feasible point on the ℓ₁‑simplex is generated.
   * The sequence is normalised so that sum(a) = sqrt(2·n);
     with this normalisation the evaluation score equals the maximal
     entry of the self‑convolution, i.e. we directly minimise max(conv(a,a)).

2. **LP refinement** (the core of the algorithm).
   For the current `a` let `M = max(conv(a,a))`.  We solve the linear program

        maximise    Σ b_j
        subject to  (a * b)_k ≤ M   for all k = 0 … 2n‑2
                    b_j ≥ 0

   where `(a * b)_k = Σ_i a_i·b_{k‑i}` is linear in `b`.  The solution `b`
   packs as much ℓ₁‑mass as possible under the same convolution bound `M`.
   Because `Σ b` is typically larger than the target sum, scaling `b` down
   to the fixed ℓ₁‑norm reduces the bound.  We then mix the new vector
   into the current iterate:
        a ← (1‑t)·a + t·b_scaled    (t = 0.5)

   The mixing keeps the ℓ₁‑norm constant while monotonically decreasing `M`.
   The LP is built once per iteration using a sparse Toeplitz matrix
   (≈ 1 M non‑zeros for n = 1024) and solved with SciPy’s HiGHS solver.

3. **Sub‑gradient polishing**.
   After the LP stage (or if time runs out) a simple projected sub‑gradient
   descent on the true max‑convolution is performed for a few thousand steps:
        – locate the shift attaining the maximum,
        – compute the corresponding sub‑gradient,
        – take a diminishing step and re‑project onto the non‑negative ℓ₁‑simplex.

   Random multiplicative “shakes’’ are added occasionally to escape plateaux.

4. The best sequence seen during the whole run is returned.
   All numbers are non‑negative and well below the clipping limit (1000).

The whole routine respects the 1000 s time budget, stopping a few seconds
early to guarantee a return value.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a non‑negative vector onto the simplex {x ≥ 0, Σx = target_sum}.
    Equivalent to a scaling followed by a simple non‑negative clipping.
    """
    if v.sum() == 0:
        # uniform fill if all zeros
        return np.full_like(v, target_sum / v.size)
    # ensure non‑negativity
    v = np.maximum(v, 0.0)
    # rescale to the required ℓ₁‑norm
    return v * (target_sum / v.sum())


def _random_initial_a(n: int, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """
    Return a random feasible point on the simplex Σa = target_sum.
    Three strategies are mixed to provide diversity.
    """
    choice = rng.integers(0, 3)
    if choice == 0:                         # uniform
        a = np.full(n, target_sum / n)
    elif choice == 1:                       # Dirichlet, α = 0.5 (moderately sparse)
        a = rng.dirichlet([0.5] * n) * target_sum
    else:                                   # sparse equal‑weight vector
        k_min = max(10, n // 30)
        k_max = max(20, n // 5)
        k = rng.integers(k_min, k_max + 1)
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = target_sum / k
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function height sequence that minimises the
    provided `evaluate_sequence` objective.  The algorithm first performs
    LP‑based refinement and then a short projected sub‑gradient polishing.
    """
    rng = np.random.default_rng(seed)

    # Give ourselves a safety margin.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Problem size (the reference implementation used 1024).
    n = 1024
    target_sum = np.sqrt(2.0 * n)          # enforced ℓ₁‑norm

    # ------------------------------------------------------------------
    # Initialise the sequence.
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=float)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.concatenate([a, np.zeros(n - a.size)])
        a = np.maximum(a, 0.0)
    else:
        a = _random_initial_a(n, rng, target_sum)

    a = _project_to_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Tracking of best solution.
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute index arrays for the convolution matrix (size n² entries).
    i_idx = np.repeat(np.arange(n), n)           # i in (i, j)
    j_idx = np.tile(np.arange(n), n)            # j in (i, j)
    rows = i_idx + j_idx                         # convolution shift k = i + j
    cols = j_idx                                 # column = j (the variable index)

    # ------------------------------------------------------------------
    # LP refinement loop.
    mixing_t = 0.5          # convex mixing weight
    max_lp_iter = 50       # safeguard against spending the whole budget
    lp_iter = 0

    while time.monotonic() < deadline and lp_iter < max_lp_iter:
        lp_iter += 1

        # Current bound M = max(conv(a, a))
        conv_a = np.convolve(a, a)
        M = conv_a.max()

        # Build the sparse matrix representing the linear constraints a * b ≤ M.
        data = a[i_idx]                      # coefficient a_i for each (i, j)
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Objective: maximise Σ b_j  →  minimise -Σ b_j
        c = -np.ones(n, dtype=float)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        # Solve the LP.
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs')   # HiGHS is fast and handles sparsity

        if not res.success:
            print(f"[LP] iteration {lp_iter} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()

        # No improvement possible if the LP cannot increase the ℓ₁‑mass.
        if sum_b <= target_sum * (1.0 + 1e-12):
            print(f"[LP] iteration {lp_iter}: sum_b ({sum_b:.6f}) ≤ target_sum, stopping LP.")
            break

        # Scale b back to the fixed ℓ₁‑norm (this strictly reduces the bound).
        scale = target_sum / sum_b
        g = b * scale

        # Convex update.
        a = (1.0 - mixing_t) * a + mixing_t * g
        # Numerical drift correction.
        a = _project_to_simplex(a, target_sum)

        # Evaluate the true objective.
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[improve] LP iter={lp_iter}, score={best_score:.6f}")

        # Early exit if we already beat the target.
        if best_score <= 1.5025:
            print("[target reached] LP stage")
            break

    # ------------------------------------------------------------------
    # Sub‑gradient polishing (simple projected sub‑gradient descent).
    sub_iter = 0
    while time.monotonic() < deadline:
        sub_iter += 1

        conv_a = np.convolve(a, a)
        max_val = conv_a.max()
        max_idx = int(np.argmax(conv_a))

        # Sub‑gradient of the max‑convolution entry.
        grad = np.zeros_like(a)
        start = max(0, max_idx - (n - 1))
        end = min(n - 1, max_idx)
        idx = np.arange(start, end + 1, dtype=int)
        grad[idx] = 2.0 * a[max_idx - idx]

        # Diminishing step size.
        step = 0.5 / np.sqrt(sub_iter)

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[improve] subgrad iter={sub_iter}, score={best_score:.6f}")

        if best_score <= 1.5025:
            print("[target reached] sub‑gradient stage")
            break

        # Occasional random shake‑up to avoid stagnation.
        if rng.random() < 0.01:
            pos = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.12))
            a[pos] *= factor
            a = _project_to_simplex(a, target_sum)

    print(f"[final] best score={best_score:.6f} (LP iters={lp_iter}, subgrad iters={sub_iter})")
    return best_a.tolist()
