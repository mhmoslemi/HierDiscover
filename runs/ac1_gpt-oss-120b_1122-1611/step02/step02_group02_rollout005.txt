```
def _project_simplex(v):
    # Project v onto simplex {x >=0, sum x = 1}
    # v is a numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    """Project the vector v onto the probability simplex (non-negative entries summing to 1)."""
    # Works for any real vector v.
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected Adam optimizer for minimizing the max self‑convolution of a
non‑negative step‑function.

The score to minimise (with ∑a = 1) is   2·n·max(a*a).
We keep the vector on the probability simplex using the Euclidean
projection, and perform Adam updates on the (sub‑)gradient built from the
active convolution peaks.
The algorithm works on a reasonably large support (≈ 4000–8000) and
uses occasional stochastic perturbations plus occasional restarts when
no improvement is observed for a while.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two length that can hold a full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a (real, non‑negative) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)[k] w.r.t. a.
    The vector w satisfies w_i = a_{k-i} (0 if out of range).
    """
    n = a.size
    w = np.zeros_like(a)
    if k < n:
        # w[0:k+1] = a[k], a[k-1], ..., a[0]
        w[: k + 1] = a[k :: -1]
    else:
        # start index where i = k-(n-1)
        start = k - (n - 1)
        length = (2 * n - 1) - k
        w[start:] = a[::-1][:length]
    return w


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would be projected to zero → return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: if rounding made sum zero, fall back to uniform
    if w.sum() == 0.0:
        w = np.full_like(v, 1.0 / n)
    return w


def _init_arcsine(n: int) -> np.ndarray:
    """
    Initialise a discrete arcsine‑density shape:
        a_i ∝ 1 / sqrt((i+½)*(n‑i‑½)).
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds (leaving a small safety margin) and uses a
    projected Adam optimiser on the max self‑convolution.
    """
    rng = np.random.default_rng(seed)
    # small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n0 = base.size
        # target size – at least a few thousand, up‑sample if possible
        target_n = max(4000, n0 * 2)
        if target_n != n0:
            # linear interpolation of the existing sequence
            x_old = np.arange(n0, dtype=np.float64)
            x_new = np.linspace(0, n0 - 1, target_n)
            a = np.interp(x_new, x_old, base)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = _init_arcsine(target_n)
            else:
                a /= s
        else:
            a = base.copy()
            a = np.clip(a, 0.0, None)
            a /= a.sum()
    except Exception:
        # fallback – a moderate‑size arcsine shape
        target_n = 8000
        a = _init_arcsine(target_n)

    # small random jitter to break perfect symmetry
    a += rng.normal(scale=0.001, size=target_n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    n = target_n
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(a)          # first moment estimate
    v = np.zeros_like(a)          # second moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                    # initial learning rate (will be decayed)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    iteration = 0

    # evaluate initial score
    conv = _conv_self(a, fft_len)
    cur_score = 2.0 * n * np.max(conv)
    best_score = cur_score
    best_a = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    last_improve = start_time
    stagnation_limit = 70.0       # seconds without improvement → restart

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) current convolution and active indices
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        max_val = np.max(conv)

        # Choose active indices – all where conv is within 90 % of the max,
        # capped to at most `max_k` entries for speed.
        max_k = 15
        active_mask = conv >= 0.90 * max_val
        active_idx = np.nonzero(active_mask)[0]
        if active_idx.size > max_k:
            # keep the first `max_k` (they are naturally ordered)
            active_idx = active_idx[:max_k]

        # --------------------------------------------------------------
        # 2) sub‑gradient of the active constraints
        # --------------------------------------------------------------
        grad = np.zeros_like(a)
        for k in active_idx:
            grad += _partner_vector(a, int(k))

        # --------------------------------------------------------------
        # 3) Adam step (minimisation → subtract)
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        bias_correction1 = 1.0 - beta1 ** iteration
        bias_correction2 = 1.0 - beta2 ** iteration

        # Linear decay of the base learning rate
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = elapsed / total
        lr = lr0 * (1.0 - frac)   # simple linear decay

        step = lr * (m / bias_correction1) / (np.sqrt(v / bias_correction2) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        # 4) occasional stochastic exploration
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.012:
            a_candidate = _random_block_perturb(a_candidate, rng)
        elif r < 0.022:
            a_candidate = _pairwise_swap_perturb(a_candidate, rng)

        # --------------------------------------------------------------
        # 5) evaluate, accept if improved (otherwise keep current a)
        # --------------------------------------------------------------
        conv_new = _conv_self(a_candidate, fft_len)
        new_score = 2.0 * n * np.max(conv_new)

        if new_score <= cur_score:
            a = a_candidate
            cur_score = new_score
        # else: reject – keep the previous a (but moments keep their state)

        # --------------------------------------------------------------
        # 6) record best ever
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed_print = last_improve - start_time
            print(f"[{elapsed_print:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 7) restart if stagnated too long
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > stagnation_limit:
            # fresh start – random Dirichlet mixed with a tiny arcsine bias
            a = rng.gamma(shape=1.0, size=n)
            a /= a.sum()
            if rng.random() < 0.3:
                a = 0.6 * a + 0.4 * _init_arcsine(n)
                a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            elapsed_restart = last_improve - start_time
            print(f"[{elapsed_restart:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to satisfy evaluate_sequence limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
