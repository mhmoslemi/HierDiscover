"""
Propose a step‑function height sequence with a low evaluation score.

Algorithm
---------
1. Work on the probability simplex (non‑negative entries summing to 1).
   The evaluation of a probability vector `p` is
        score(p) = 2·n·max_k (p * p)_k
   because the original objective is scale‑invariant.

2. Use a projected Adam optimiser where the gradient is the *exact*
   sub‑gradient of the max‑convolution (i.e. the gradient of the
   convolution entry attaining the current maximum).  This removes the
   smoothing bias of the previous soft‑max approach.

3. Occasionally perform a multiplicative “shake‑up’’ (randomly boost a
   coordinate) to escape plateaux.

4. Perform frequent random restarts with diverse initialisations
   (uniform, Dirichlet, sparse supports).  The best vector seen over
   the whole budget is returned.

5. All heavy work (convolution and its maximum) is performed via the
   FFT (`np.fft.rfft`) which is O(n log n) and fast for n≈1000.

The implementation respects the 1000 s budget, prints progress,
and finally returns the best sequence found (as a plain list of
floats).  The returned vector is already feasible for
`evaluate_sequence` – it contains only non‑negative numbers and its
scale does not matter because the objective is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _random_initial_p(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a random feasible probability vector (sum = 1).
    Three strategies are mixed:
        * uniform
        * Dirichlet (α = 0.5) – moderately sparse
        * sparse equal‑weight support (size random)
    """
    choice = rng.integers(0, 3)
    if choice == 0:                     # uniform
        p = np.full(n, 1.0 / n)
    elif choice == 1:                   # Dirichlet, α=0.5
        p = rng.dirichlet([0.5] * n)
    else:                               # sparse support
        k = rng.integers(max(8, n // 50), max(30, n // 20))
        idx = rng.choice(n, size=k, replace=False)
        p = np.zeros(n)
        p[idx] = 1.0 / k
    # ensure exact feasibility
    p = _project_simplex(p)
    return p


# ----------------------------------------------------------------------
def _conv_max(p: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max convolution entry of a probability vector with itself.
    Returns (max_value, full_convolution_array).
    """
    n = p.shape[0]
    # FFT length = next power of two ≥ 2n
    fft_len = 1 << ((2 * n - 1).bit_length())
    fp = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(fp * fp, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    return max_val, conv


# ----------------------------------------------------------------------
def _subgradient_max_conv(p: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (p * p)_k w.r.t. `p`.
    For each index k attaining the maximum, adds the contribution
    grad_i += 2·p_{k‑i} (when the latter is defined).
    """
    n = p.shape[0]
    max_val = conv.max()
    # consider all positions attaining the maximum (within a tiny tolerance)
    max_idx = np.where(conv >= max_val - 1e-14)[0]
    grad = np.zeros(n, dtype=np.float64)
    for k in max_idx:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i = np.arange(start, end + 1, dtype=np.int64)
        grad[i] += 2.0 * p[k - i]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function height sequence with a low evaluation
    score (subject to the 1000 s time budget).  The algorithm runs a
    projected Adam optimiser on the exact max‑convolution sub‑gradient,
    with random restarts and occasional random shakes to improve
    exploration.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – keep the original size (1024) which works well
    n = 1024

    # ------------------------------------------------------------------
    # initialise from a previously‑found construction if available
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=float)
        if init.size != n:

            else:

        # turn into a probability vector (sum = 1) and project
        tot = init.sum()
        if tot <= 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p = init / tot
        p = _project_simplex(p)
    else:
        p = _random_initial_p(n, rng)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters (operating on the simplex)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_eta = 0.6          # a bit larger than the previous 0.2
    iteration = 0

    # bookkeeping
    best_p = p.copy()
    _, conv = _conv_max(p)
    best_score = 2.0 * n * conv.max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # time budget handling
    dead_line = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # main optimisation loop
    restart_interval = 25000      # iterations between restarts
    shake_prob = 0.09             # probability of a multiplicative shake
    while time.monotonic() < dead_line:
        iteration += 1

        # compute convolution and its sub‑gradient
        max_val, conv = _conv_max(p)
        grad = _subgradient_max_conv(p, conv)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        eta = base_eta / np.sqrt(iteration)
        p = p - eta * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        p = _project_simplex(p)

        # occasional random multiplicative shake‑up (helps escape plateaux)
        if rng.random() < shake_prob:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.22)))
            p[idx] *= factor
            p = _project_simplex(p)

        # evaluate the true objective (fast internal version)
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_p = p.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # periodic restart to avoid being trapped
        if iteration % restart_interval == 0:
            p = _random_initial_p(n, rng)
            m.fill(0.0)
            v.fill(0.0)
            # keep the best seen so far (do not discard it)

        # early exit if we have already crossed the target threshold
        if best_score <= 1.5025:
            print("[target reached] score below target")
            break

    # ------------------------------------------------------------------
    # final report
    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # The sequence may be returned as‑is (it already satisfies all checks);
    # scaling does not affect the objective, but we keep the probability
    # vector for consistency.
    return [float(x) for x in best_p]
