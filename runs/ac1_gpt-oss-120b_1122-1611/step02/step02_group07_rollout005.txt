"""
propose_candidate – two‑phase projected Adam optimiser

This optimiser searches for a non‑negative step‑function (a probability vector
on the simplex) that minimises

        2·n·max_k (a∗a)[k] / (∑a_i)²

i.e. the `evaluate_sequence` score.  It works in two phases:

1.  A smooth‑max (log‑sum‑exp) surrogate of the ℓ∞‑norm is optimised with
    Adam.  The temperature β is increased gradually up to a large value
    (default β_max≈500) so that the surrogate becomes a tight approximation
    of the true max.

2.  After the temperature has saturated the optimiser switches to the exact
    sub‑gradient of the max‑convolution (the gradient of the active
    constraints).  This gives a precise descent direction when the landscape
    is already almost flat.

Both phases keep the iterate on the probability simplex (non‑negative,
unit ℓ₁‑norm) via a fast projection.  Small multiplicative “shake‑ups” are
applied periodically to avoid stagnation, and a simple restart‐on‑no‑improve
strategy injects random noise if the score does not improve for a long
period.  The routine stops either when the time budget expires or when the
target bound (≈1.5025) is reached.

The algorithm is deliberately lightweight (two 1‑D convolutions per
iteration) and fits easily into the 1000 s budget for n≈1024.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, sum(x) = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function height vector that minimises ``evaluate_sequence``.
    The optimiser combines a smooth‑max Adam phase with an exact sub‑gradient
    phase, uses random perturbations and occasional restarts, and returns the
    best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time handling (keep a small safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # problem size – stay at 1024 (fast enough and matches previous runs)
    n = 1024

    # ------------------------------------------------------------------
    # warm start: use previously discovered good vector if available,
    # otherwise start from the uniform distribution.
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a0.size != n:
                raise ValueError
            a = a0.copy()
        except Exception:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # make sure we are on the simplex
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters
    base_lr = 0.6                 # initial learning‑rate magnitude
    beta_min = 0.1                # start temperature for smooth‑max
    beta_max_target = 500.0       # final temperature (very sharp)
    beta_ramp_iters = 300_000     # iterations over which β is increased

    # after the temperature has saturated we switch to the exact sub‑gradient
    switch_to_subgrad_iter = beta_ramp_iters + 100_000

    # best solution seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    it = 0
    last_improve_it = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) compute full autocorrelation
        conv = np.convolve(a, a)           # length 2n‑1
        max_c = conv.max()

        # --------------------------------------------------------------
        # 2) decide which gradient to use
        if it <= switch_to_subgrad_iter:
            # ----- smooth‑max (log‑sum‑exp) gradient -----
            beta = beta_min + (beta_max_target - beta_min) * \
                min(1.0, it / beta_ramp_iters)
            # stable exponentiation
            exp_shift = np.exp(beta * (conv - max_c))
            w = exp_shift / exp_shift.sum()           # soft‑max weights
            # gradient = 2 * (w * a) convolved (as in the original code)
            a_rev = a[::-1]
            grad_full = np.convolve(w, a_rev)         # length 3n‑2
            grad = 2.0 * grad_full[n - 1:n - 1 + n]   # central n entries
        else:
            # ----- exact sub‑gradient of max(conv) -----
            # indices where conv is (near) maximal
            tol = 1e-12
            ks = np.where(conv >= max_c - tol)[0]
            # build a sparse weight vector (indicator on active ks)
            w = np.zeros_like(conv)
            w[ks] = 1.0 / len(ks)                     # average over active set
            a_rev = a[::-1]
            grad_full = np.convolve(w, a_rev)
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # --------------------------------------------------------------
        # 3) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple diminishing learning‑rate schedule
        lr = base_lr / np.sqrt(it)

        a_next = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 4) occasional multiplicative shake‑up (≈5 % of iterations)
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal ≈ ±12 %
            a_next[idx] *= factor

        # --------------------------------------------------------------
        # 5) project back onto the simplex
        a_next = _project_simplex(a_next)

        # --------------------------------------------------------------
        # 6) evaluate
        score = float(evaluate_sequence(a_next.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_next.copy()
            best_score = score
            last_improve_it = it
            print(f"[improve] iter={it:,}  score={best_score:.6f}  beta={beta:.1f}")

        # --------------------------------------------------------------
        # 7) early exit if target reached
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping early")
            break

        # --------------------------------------------------------------
        # 8) restart on stagnation (no improvement for many iterations)
        if it - last_improve_it > 120_000:
            # inject a modest random perturbation around the current best
            noise = rng.normal(0.0, 0.02, size=n)
            a_next = best + noise
            a_next = np.maximum(a_next, 0.0)
            a_next = _project_simplex(a_next)
            # reset Adam's moments to avoid stale direction information
            m.fill(0.0)
            v.fill(0.0)
            last_improve_it = it
            print(f"[restart] iter={it:,}  no improvement → perturbation applied")

        # --------------------------------------------------------------
        # 9) bookkeeping
        a = a_next

        # occasional progress print
        if it % 20000 == 0:
            elapsed = time.monotonic() - (deadline - budget_s + 0.5)
            print(f"[status] iter={it:,}  best_score={best_score:.6f}  time={elapsed:.1f}s")

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    return [float(x) for x in best]
