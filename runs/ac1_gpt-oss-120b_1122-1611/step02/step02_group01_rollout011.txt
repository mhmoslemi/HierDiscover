"""
Propose a step‑function (non‑negative vector) that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a fast mirrored‑gradient / deterministic pair‑wise mass‑transfer
scheme.  The algorithm works directly on the full‑length vector (the
global ``height_sequence_1`` is used if present) and runs for the given
time budget.  It repeatedly

* evaluates the self‑convolution via an FFT;
* builds a sub‑gradient from the K largest convolution entries;
* applies an exponentiated‑gradient (mirror descent) update on the
  probability simplex;
* performs a *deterministic* pair‑move that always reduces the current
  maximal convolution entry;
* occasionally perturbs a random block and restarts when stalled.

The routine returns the best vector seen before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Next power‑of‑two length sufficient for a linear convolution of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real, non‑negative array ``a`` using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the convolution entry ``(a*a)[k]`` with respect to ``a``:
    ``w[i] = a[k-i]`` for indices where ``k-i`` lies inside the array,
    otherwise zero.
    """
    n = a.size
    w = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    # fill w[i_min : i_max+1] = a[k - (i_min ... i_max)]
    idx = np.arange(i_min, i_max + 1)
    w[idx] = a[k - idx]
    return w


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Contribution of the coefficient ``a[idx]`` to the full self‑convolution:
    a copy of ``a`` placed at offset ``idx`` in the length‑``2n‑1`` result.
    """
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx: idx + n] = a
    return v


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _deterministic_pair_move(a: np.ndarray, conv: np.ndarray, k_max: int,
                            max_frac: float = 0.30) -> tuple[np.ndarray, float, bool]:
    """
    Try to reduce the peak at ``k_max`` by moving mass from the index that
    contributes the most to that peak to the index that contributes the
    least.  Returns the new vector, new score, and a flag indicating whether
    the move was successful.
    """
    n = a.size
    # partner vector w_i = a[k_max - i]
    w = _partner_vector(a, k_max)
    contrib = a * w                     # contribution of each index to conv[k_max]
    src = int(np.argmax(contrib))       # source index (largest contribution)
    if a[src] <= 0.0:
        return a, None, False

    # candidate sink: smallest partner value (where shifting mass hurts the peak least)
    # we avoid the same index
    sink_candidates = np.where(np.arange(n) != src)[0]
    if sink_candidates.size == 0:
        return a, None, False
    # pick the index with minimal w (tie‑break by small a)
    sink = int(sink_candidates[np.argmin(w[sink_candidates])])

    # amount to move – limited to a fraction of the source mass
    max_delta = min(a[src], max_frac * a[src])
    if max_delta <= 0.0:
        return a, None, False
    # try a few deltas (full, half) – keep the one that improves most
    best_delta = 0.0
    best_score = None
    for frac in (1.0, 0.5, 0.25):
        delta = max_delta * frac
        # update convolution analytically
        new_conv = conv.copy()
        # remove contribution of src
        new_conv[src: src + n] -= delta * a
        # add contribution of sink
        new_conv[sink: sink + n] += delta * a
        new_max = float(new_conv.max())
        new_score = 2.0 * n * new_max          # sum(a) stays 1
        if best_score is None or new_score < best_score:
            best_score = new_score
            best_delta = delta

    if best_delta == 0.0:
        return a, None, False

    # Apply the best delta
    new_a = a.copy()
    new_a[src] -= best_delta
    new_a[sink] += best_delta
    # renormalise (numerical drift)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / n
    else:
        new_a /= s
    # recompute conv after the move (cheaply via FFT to avoid drift)
    fft_len = _fft_len_for_n(n)
    new_conv = _conv_self(new_a, fft_len)
    new_score = 2.0 * n * float(new_conv.max())
    return new_a, new_score, True

    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    best = a.copy()
    conv_best = _conv_self(best, fft_len)
    best_score = 2.0 * n * float(conv_best.max())

    start_time = time.monotonic()
    last_improve = start_time

    # hyper‑parameters – tuned for quick convergence
    eta_initial = 1.0
    eta_final = 0.01
    K_active = 30                 # number of top convolution entries used for sub‑gradient
    block_perturb_prob = 0.03
    restart_gap = 35.0            # seconds without improvement before a soft restart
    max_frac_transfer = 0.35      # max fraction of source mass moved in a deterministic pair move

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution and score -----
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----- update best if improved -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- multi‑peak mirror‑gradient step -----
        # pick the K largest convolution entries
        if K_active >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            kth = conv.size - K_active
            part = np.argpartition(conv, kth)
            active_idx = part[kth:]

        # aggregate sub‑gradient (sum of partner vectors)
        w = np.zeros_like(a)
        for k_idx in active_idx:
            w += _partner_vector(a, int(k_idx))

        # learning‑rate schedule (linear decay)
        elapsed_total = time.monotonic() - start_time
        progress = elapsed_total / (deadline - start_time)
        eta = eta_initial * (1.0 - progress) + eta_final * progress
        eta = max(eta, eta_final)

        # mirror descent on the simplex (exponentiated‑gradient)
        factor = np.exp(-eta * w)
        a = a * factor
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- deterministic pair‑wise mass transfer -----
        # reduce the dominant peak directly
        k_max_idx = int(np.argmax(conv))
        a_new, new_score, succeeded = _deterministic_pair_move(
            a, conv, k_max_idx, max_frac=max_frac_transfer
        )
        if succeeded and new_score < cur_score:
            a = a_new
            cur_score = new_score
            cur_max = float(_conv_self(a, fft_len).max())
            # note: best will be updated at the top of the next loop

        # ----- soft restart when progress stalls -----
        now = time.monotonic()
        if now - last_improve > restart_gap:
            # take the current best, add a modest global noise and continue
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            a /= a_sum if a_sum > 0 else 1.0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – re‑starting from best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to the bounds enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
