def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The algorithm proceeds in two phases:
    1. Random greedy construction of a large support S such that each unordered
       sum i+j appears at most K=2 times (a B_2[2] set).  The resulting
       uniform distribution on S already yields a low max‑convolution.
    2. Starting from the best distribution found, a few hundred thousand
       Adam steps are performed on a smooth‑max surrogate of the objective.
       After each step the vector is projected onto the probability simplex.
    The best sequence seen during the whole run is returned.
    """
    import time
    import numpy as np

    # ------------------------------------------------------------------
    def _project_simplex(v: np.ndarray) -> np.ndarray:
        n = v.shape[0]
        if n == 0:
            return v
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1.0
        ind = np.arange(n) + 1.0
        cond = u - cssv / ind > 0
        if not np.any(cond):
            return np.full_like(v, 1.0 / n)
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w

    # ------------------------------------------------------------------
    def _greedy_low_collision_set(n: int, K: int, rng: np.random.Generator) -> np.ndarray:
        """Return a set of indices S ⊆ {0,…,n‑1} such that each unordered sum
        i+j appears at most K times.  Simple greedy construction using a random order."""
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
        selected = []
        order = np.arange(n)
        rng.shuffle(order)
        for i in order:
            # check diagonal
            if sum_counts[i + i] + 1 > K:
                continue
            ok = True
            for j in selected:
                if sum_counts[i + j] + 1 > K:
                    ok = False
                    break
            if ok:
                # add i
                selected.append(i)
                sum_counts[i + i] += 1
                # update counts with previous elements
                for j in selected[:-1]:
                    sum_counts[i + j] += 1
        return np.array(selected, dtype=np.int32)

        if iteration % 200_000 == 0:

       unordered contributions, guaranteeing a low max‑convolution.
