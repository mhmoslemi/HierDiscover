"""
Propose a step‑function that minimizes the evaluation score.

The algorithm works in three stages:

1. **Exploratory constructions** – a collection of analytic windows
   (uniform, triangular, Hann, Hamming, Blackman, Kaiser, Gaussian,
   exponential) and many random Dirichlet / sparse vectors are generated,
   projected onto the probability simplex and scored with a fast
   FFT‑based convolution.  The best of these seeds is kept.

2. **Gradient refinement** – starting from the best seed we minimise the
   true objective `max(conv(a,a))` on the simplex.  The gradient of the
   (non‑smooth) maximum is taken as the sub‑gradient of the active
   convolution entry, and an Adam‑style update is performed followed by
   a projection onto the simplex.  The step size decays as 1/√t.
   Occasionally a multiplicative “shake‑up’’ is applied to escape plateaux.

3. **Finalisation** – the best sequence found (according to the original
   `evaluate_sequence` routine) is returned.  The whole search respects a
   user‑provided time budget (default 1000 s) and stops a few seconds
   early to guarantee a return value.

All helper functions are defined at top‑level, no closures are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft_max(a: np.ndarray):
    """
    Compute the linear convolution of a with itself using FFT,
    returning (full_convolution, max_value, argmax_index).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    C = np.fft.irfft(A * A, n=size)
    conv = C[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)                # numerical safety
    max_val = float(conv.max())
    argmax = int(conv.argmax())
    return conv, max_val, argmax


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a at the active index k.
    For the linear convolution, ∂c_k/∂a_j = a_{k-j} + a_j when the
    indices are inside the valid range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    rev = k - idx
    grad[idx] = a[rev] + a[idx]
    return grad


# ----------------------------------------------------------------------
def _window_candidates(n: int):
    """Return a list of analytically‑defined windows (all normalised)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # triangular (Bartlett)
    w = np.bartlett(n)
    cand.append(w / w.sum())

    # Hann
    w = np.hanning(n)
    cand.append(w / w.sum())

    # Hamming
    w = np.hamming(n)
    cand.append(w / w.sum())

    # Blackman
    w = np.blackman(n)
    cand.append(w / w.sum())

    # Kaiser with several betas
    for beta in (2, 4, 6, 8, 10, 12, 14):
        w = np.kaiser(n, beta)
        cand.append(w / w.sum())

    # Gaussian bells with different widths
    x = np.arange(n)
    mu = (n - 1) / 2.0
    for sigma_frac in (0.12, 0.15, 0.20, 0.25):
        sigma = sigma_frac * n
        w = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        cand.append(w / w.sum())

    # Exponential decays
    for lam in (0.001, 0.003, 0.005, 0.01):
        w = np.exp(-lam * x)
        cand.append(w / w.sum())

    return cand

    """
    Search for a non‑negative step‑function (list of floats) that yields a
    low value of ``evaluate_sequence``.
    The function respects the given time budget and returns the best

    """

    # ---- problem size ---------------------------------------------------
    if 'height_sequence_1' in globals():

        n = init_seq.size
    else:
        n = 1024                                   # default length

    # ---- generate a pool of seeds ---------------------------------------
    seeds = []

    # analytic windows

    # random Dirichlet vectors (moderately sparse)
    for _ in range(60):
