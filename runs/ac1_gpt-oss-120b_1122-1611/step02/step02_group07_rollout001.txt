def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,float(budget_s)-0.2)
    n=1024
    best_score = np.inf
    best_a = None
    # try up to N restarts
    restart = 0
    while time.monotonic() < deadline:
        # initialize a: warm start from height_sequence_1 if restart==0 else random Dirichlet
        if restart == 0 and "height_sequence_1" in globals():
            a = np.asarray(globals()["height_sequence_1"], dtype=np.float64).copy()
            a = _project_simplex(a)
        else:
            # Dirichlet sample to get random simplex point
            alpha = np.full(n, 1.0)  # uniform Dirichlet
            a = rng.dirichlet(alpha)
        # optionally add small random perturbation to avoid immediate plateau.
        # Do gradient descent for certain steps or until time near deadline.
        it = 0
        last_improve_it = 0
        while time.monotonic() < deadline:
            it += 1
            # compute conv and max indices
            conv = np.convolve(a, a)  # length 2n-1
            M = conv.max()
            # Determine active k's within tolerance relative to max
            eps = 1e-12
            K = np.where(conv >= M - eps)[0]
            # gradient via subgradient
            w = np.zeros_like(conv)
            w[K] = 1.0 / len(K)
            a_rev = a[::-1]
            grad_full = np.convolve(w, a_rev)  # length 3n-2
            grad = 2.0 * grad_full[n-1:n-1+n]
            # step size diminishing
            lr = 0.5 / np.sqrt(it)  # can tune
            a_new = a - lr * grad
            # random jitter occasionally
            if rng.random() < 0.1:
                # pick random index and add factor
                idx = rng.integers(0,n)
                factor = np.exp(rng.normal(0.0, 0.2))
                a_new[idx] *= factor
            # project to simplex
            a_new = _project_simplex(a_new)
            # evaluate
            score = float(evaluate_sequence(a_new.tolist()))
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                last_improve_it = it
                print(f"[improve restart={restart} it={it} score={score:.6f}")
                if best_score <= 1.5025:
                    print("[target] reached")
                    return best_a.tolist()
            # early stopping if plateau for many iterations
            if it - last_improve_it > 20000:
                break
            a = a_new
        restart += 1
        # break if many restarts maybe.
    return best_a.tolist() if best_a is not None else [1.0]  # fallback uniform?

"""
Propose a step‑function height vector that (approximately) minimizes

    2·n·max_k (a * a)[k] / (∑ a_i)²

subject to a_i ≥ 0.  The algorithm consists of three stages:

1. **Fast exploratory sampling** – evaluate a variety of simple
   parametric families (exponential, power‑law) and many random Dirichlet
   draws.  The best sequence found so far is kept as a warm‑start.

2. **Projected Adam on a true max‑convolution sub‑gradient** – from the
   current best candidate we perform an Adam optimisation where the
   gradient is the exact sub‑gradient of the max‑convolution (i.e. a
   one‑hot weight vector on the active lag(s)).  After each update the
   vector is projected onto the probability simplex.

3. **Stochastic perturbations & early stopping** – occasional log‑normal
   multiplicative jitter helps to escape flat plateaus; if no improvement
   is observed for a long stretch the refinement phase stops.

All computation respects the 1000‑second wall‑time budget and prints
progress information.  The final sequence is returned as a list of
floats (already projected onto the simplex, hence non‑negative and
∑a_i = 1, which makes the objective independent of scaling).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0 , Σ x_i = 1 }.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:            # all entries are non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max_k (a*a)[k] w.r.t. a.
    The sub‑gradient equals 2 * conv(w, a_rev) where w is a one‑hot
    vector located at the lag(s) attaining the maximal convolution value.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)               # length 2n‑1
    M = conv.max()
    # tolerance to collect all tying lags (numerical safety)
    eps = 1e-12
    active = np.where(np.abs(conv - M) <= eps)[0]
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    # gradient via convolution with reversed a
    grad_full = np.convolve(w, a[::-1])
    grad = 2.0 * grad_full[n - 1:n - 1 + n]   # central n entries
    return grad

    """
