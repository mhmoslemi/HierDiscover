def _self_convolution_fft(a, N_fft):
    A = np.fft.rfft(a, N_fft)
    conv = np.fft.irfft(A * A, N_fft)
    return conv[:2*a.shape[0] - 1]

    # w length 2n-1
    W = np.fft.rfft(w, N_fft)
    corr_full = np.fft.irfft(W * a_rev_fft, N_fft)

    iteration +=1

    max_conv = conv.max()
    # compute temperature
    tau = max(max_conv * tau_factor, 1e-15)
    shifted = (conv - max_conv) / tau
    w_soft = np.exp(shifted)
    w_sum = w_soft.sum()
    w_soft = w_soft / w_sum
    # compute a_rev_fft
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, N_fft)

    error = conv - target_val
    corr_err = _correlation_fft(error, a_rev_fft, N_fft)
    grad_err = 2.0 * corr_err[n-1:n-1+n]

    alpha_err = 0.3
    alpha_max = 0.7
    grad = alpha_max * grad_max + alpha_err * grad_err

    m = beta1 * m + (1 - beta1) * grad

def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 0.1)
    best_global = None
    best_global_score = float('inf')
    run_id = 0
    while time.monotonic() < deadline - 5.0:
        # decide n
        if run_id == 0 and 'height_sequence_1' in globals():
            try:
                known = np.asarray(height_sequence_1, dtype=np.float64)
                if known.ndim == 1 and known.size > 0:
                    n = known.size
                    a0 = known.copy()
                else:
                    raise Exception
            except Exception:
                n = rng.integers(512, 2049)
                a0 = rng.random(n)
        else:
            n = rng.integers(512, 2049)  # choose random length between 512 and 2048
            # pick init method
            method = rng.choice(['dirichlet','hann','gaussian','spike'])
            if method == 'dirichlet':
                a0 = rng.exponential(1.0, size=n)
            elif method == 'hann':
                idx = np.arange(n)
                a0 = 0.5 - 0.5 * np.cos(2 * np.pi * idx / (n - 1))
            elif method == 'gaussian':
                idx = np.arange(n)
                sigma = 0.3
                a0 = np.exp(-0.5 * ((idx - (n-1)/2) / (sigma*n))**2)
            elif method == 'spike':
                # few random spikes
                a0 = np.zeros(n)
                k = max(2, n // 20)  # number of spikes
                spikes = rng.choice(n, size=k, replace=False)
                a0[spikes] = rng.random(k)
            else:
                a0 = rng.random(n)
            # ensure non-negativity
            a0 = np.maximum(a0, 0.0)
        # project onto simplex sum=1
        if a0.sum() == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = _project_simplex(a0)
        # compute inner deadline
        remaining = deadline - time.monotonic()
        if remaining <= 5.0:
            break
        inner_deadline = time.monotonic() + min(remaining * 0.7, 30.0) # allocate up to 30s for each run
        # compute N_fft length globally (pre-calc inside optimize)
        best_local, best_local_score = _optimize_one(a0, n, inner_deadline, rng)
        if best_local_score < best_global_score:
            best_global_score = best_local_score
            best_global = best_local.copy()
            print(f"[run {run_id}] new global best score={best_global_score:.6f} n={n}")
        run_id += 1
    if best_global is None:
        # fallback uniform
        n = 1024
        best_global = np.full(n, 1.0/n)
        best_global_score = 2.0 * n * (1.0/n)  # =2
    print(f"[final] best global score={best_global_score:.6f}")
    return [float(x) for x in best_global]

"""
Project: minimize 2·n·max(conv(a,a)) / (sum(a)^2) for a non‑negative
step‑function a.  The algorithm is a projected Adam optimiser that
minimises a blend of a smooth–max surrogate of the convolution peak and
an L2‑error towards a perfectly flat convolution (the theoretical lower
bound 2·n/(2·n‑1)).  Multiple random restarts with varied initial
shapes (Dirichlet, Hann, Gaussian, sparse spikes) are used; the best
vector found before the time budget expires is returned.

All helper routines are defined at module level (no closures) and only
standard scientific libraries are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, N_fft: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT (real input).  Returns the
    first 2·len(a)‑1 entries.
    """
    A = np.fft.rfft(a, N_fft)
    conv = np.fft.irfft(A * A, N_fft)
    return conv[: 2 * a.shape[0] - 1]


def _correlation_fft(w: np.ndarray, a_rev_fft: np.ndarray, N_fft: int) -> np.ndarray:
    """
    Full linear correlation of w with the reversed vector a (i.e. conv(w, a_rev)).
    Returned length is exactly N_fft (zero‑padded).
    """
    W = np.fft.rfft(w, N_fft)
    corr_full = np.fft.irfft(W * a_rev_fft, N_fft)
    return corr_full


def _optimize_one(
    a_init: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    *,
    lr_base: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_factor_start: float = 0.5,
    tau_decay: float = 0.9995,
    weight_max: float = 0.7,
    weight_err: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation for a given start vector `a_init` (already projected
    onto the simplex).  Returns the best vector seen and its internal score
    (2·n·max(conv)).
    """
    N_fft = _next_pow2(3 * n - 2)
    target_val = 1.0 / (2 * n - 1)  # uniform convolution target

    a = a_init.copy()
    # bookkeeping
    best = a.copy()
    conv_best = _self_convolution_fft(best, N_fft)
    best_score = 2.0 * n * conv_best.max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    tau_factor = tau_factor_start
    lower_bound = 2.0 * n / (2 * n - 1)

    while time.monotonic() < deadline:
        iteration += 1

        # ----- self‑convolution & soft‑max weights -----
        conv = _self_convolution_fft(a, N_fft)
        max_conv = conv.max()
        tau = max(max_conv * tau_factor, 1e-15)

        shifted = (conv - max_conv) / tau
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradients (smooth max + L2 error toward flat conv) -----
        a_rev_fft = np.fft.rfft(a[::-1], N_fft)

        # gradient of smooth‑max term
        corr_max = _correlation_fft(w, a_rev_fft, N_fft)
        grad_max = 2.0 * corr_max[n - 1 : n - 1 + n]

        # gradient of L2 error term (conv - uniform)
        err = conv - target_val
        corr_err = _correlation_fft(err, a_rev_fft, N_fft)
        grad_err = 2.0 * corr_err[n - 1 : n - 1 + n]

        # blended gradient
        grad = weight_max * grad_max + weight_err * grad_err

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr_base / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat

        # ----- projection onto simplex -----
        a_candidate = _project_simplex(a_candidate)

        # ----- occasional random shake‑up -----
        if rng.random() < 0.005:
            noise = rng.normal(0.0, 0.025, size=n)
            a_candidate = a_candidate + noise
            a_candidate = _project_simplex(np.maximum(a_candidate, 0.0))

        # ----- evaluate (internal score) -----
        conv_cand = _self_convolution_fft(a_candidate, N_fft)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score:
            best_score = score_cand
            best = a_candidate.copy()
            if iteration % 1000 == 0:
                print(f"[improve] iter={iteration:,} score={best_score:.6f}")

        a = a_candidate

        # ----- anneal temperature -----
        tau_factor *= tau_decay
        if iteration % 5000 == 0:
            tau_factor = max(tau_factor, 1e-6)

        # ----- early stop -----
        if best_score <= lower_bound * 1.01:
            print("[early stop] near theoretical bound")
            break

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a step‑function height sequence with low evaluation score.
    The function respects the 1000 s time budget, uses several random
    restarts with different initialisations and a projected Adam optimiser
    that balances a smooth‑max surrogate of the convolution peak with a
    flattening L2 term.
    """
    rng = np.random.default_rng(seed)

    # overall time limit (tiny safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    best_global = None
    best_global_score = float("inf")
    run_id = 0

    while time.monotonic() < deadline - 5.0:
        # ----- choose length and initial vector -----
        if run_id == 0:
            # try to reuse the known best sequence if available
            try:
                known = np.asarray(height_sequence_1, dtype=np.float64)
                if known.ndim == 1 and known.size > 0:
                    n = int(known.size)
                    a0 = known.copy()
                else:

            except Exception:
                n = rng.integers(800, 1800)  # modest size for first run
                a0 = rng.random(n)
        else:
            # random length between 512 and 2048
            n = int(rng.integers(512, 2049))
            # pick a random initialisation strategy
            init_type = rng.choice(["dirichlet", "hann", "gaussian", "spike"])
            if init_type == "dirichlet":
                a0 = rng.exponential(1.0, size=n)
            elif init_type == "hann":
                idx = np.arange(n)
                a0 = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
            elif init_type == "gaussian":
                idx = np.arange(n)
                sigma = 0.3
                a0 = np.exp(-0.5 * ((idx - (n - 1) / 2) / (sigma * n)) ** 2)
            else:  # "spike"
                k = max(2, n // 20)
                a0 = np.zeros(n)
                spikes = rng.choice(n, size=k, replace=False)
                a0[spikes] = rng.random(k)

        # ----- bring onto the simplex (sum = 1) -----
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = _project_simplex(a0)

        # ----- allocate time for this restart -----
        remaining = deadline - time.monotonic()
        if remaining <= 5.0:
            break
        # give each restart at most 30 s, but never exceed the total budget
        inner_deadline = time.monotonic() + min(30.0, remaining * 0.7)

        # ----- run the optimiser -----
        best_local, best_local_score = _optimize_one(
            a_init=a0,
            n=n,
            deadline=inner_deadline,
            rng=rng,
        )

        # ----- update global best -----
        if best_local_score < best_global_score:
            best_global_score = best_local_score
            best_global = best_local.copy()
            print(
                f"[run {run_id}] new global best score={best_global_score:.6f} (n={n})"
            )

        run_id += 1

    # fallback if nothing was produced (should not happen)
    if best_global is None:
        n = 1024
        best_global = np.full(n, 1.0 / n)

    print(f"[final] best score={best_global_score:.6f}")
    return [float(x) for x in best_global]
