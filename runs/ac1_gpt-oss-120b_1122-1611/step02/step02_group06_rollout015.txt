"""
Propose a candidate step‑function sequence.

Strategy
--------
1. Work on the probability simplex (non‑negative entries that sum to 1).
   The evaluation score is invariant to scaling, therefore we can always
   normalise the vector to unit ℓ₁‑norm.

2. Initialise the search with a handful of analytically motivated shapes:
      * uniform
      * 1/√(i+1)               – decreasing “square‑root’’ law
      * 1/√((i+1)(n-i))        – arcsine‑type law
      * truncated geometric families a_i ∝ exp(‑α·i) (α swept on a log‑grid)
      * a few random Dirichlet draws

   The best of these seeds is taken as the starting point.

3. Optimise the *hard* maximum of the autocorrelation
      b = a * a (linear convolution)
   using a projected Adam optimiser.  The sub‑gradient of
   max(b) w.r.t. a is simply 2·a shifted by the index of the
   current maximal entry, which can be computed in O(n).

4. Occasionally jitter the current vector (multiplicative log‑normal
   factor on a random coordinate) to escape plateaus and perform
   cheap random restarts after many iterations.

5. Keep the best vector seen so far; stop early if the score is
   within 2 % of the theoretical lower bound 2 n/(2 n‑1).

The whole routine respects the 1000 s budget, runs completely
on‑CPU and returns a list ready for `evaluate_sequence`.

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _random_initial_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random feasible point on the simplex."""
    choice = rng.integers(0, 3)
    if choice == 0:                     # uniform
        a = np.full(n, 1.0 / n)
    elif choice == 1:                   # Dirichlet α=0.5 – moderately sparse
        a = rng.dirichlet([0.5] * n)
    else:                               # a sparse equal‑weight vector
        k = rng.integers(max(10, n // 30), max(20, n // 5))
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0 / k
    return _project_simplex(a)


# ----------------------------------------------------------------------
def _make_initial_candidates(n: int, rng: np.random.Generator):
    """Generate a small collection of heuristic initial vectors."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # 1/sqrt(i+1)
    w = 1.0 / np.sqrt(np.arange(1, n + 1))
    cand.append(w / w.sum())

    # 1/sqrt((i+1)(n-i))
    i = np.arange(1, n + 1)
    w = 1.0 / np.sqrt(i * (n + 1 - i))
    cand.append(w / w.sum())

    # geometric families – sweep α (the exponential decay rate)
    for alpha in np.logspace(-4, -1, num=10):
        r = np.exp(-alpha)
        w = r ** np.arange(n)
        cand.append(w / w.sum())

    # a few Dirichlet draws
    for _ in range(5):
        cand.append(rng.dirichlet(np.full(n, 0.5)))

    # reuse a previously found construction if it exists
    if 'height_sequence_1' in globals():
        prev = np.array(height_sequence_1, dtype=float)
        if prev.size != n:
            if prev.size > n:
                prev = prev[:n]
            else:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
        cand.append(_project_simplex(prev))

    # final sanity projection
    cand = [_project_simplex(c) for c in cand]
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    n = 1024                     # problem size (works well within the budget)

    # --------------------------------------------------------------
    # generate a handful of sensible starting points and keep the best
    best = None
    best_score = float('inf')
    for cand in _make_initial_candidates(n, rng):
        conv = np.convolve(cand, cand)
        score = 2.0 * n * conv.max()          # sum(cand) == 1 after projection
        if score < best_score:
            best_score = score
            best = cand.copy()
            print(f"[init] seed score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser on the *hard* max of the convolution
    a = best.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_eta = 0.3                # base learning rate
    iteration = 0

    # theoretical lower bound (attainable only in the limit)
    theoretical_lb = 2.0 * n / (2 * n - 1)

    while time.monotonic() < deadline:
        iteration += 1

        # ------------------------------------------------------
        # compute convolution and locate the maximal entry
        conv = np.convolve(a, a)
        max_b = conv.max()
        k_max = int(conv.argmax())

        # ------------------------------------------------------
        # sub‑gradient of max_b (hard max)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # ------------------------------------------------------
        # Adam update (projected onto the simplex afterwards)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        eta = base_eta / np.sqrt(iteration)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # enforce feasibility
        a = _project_simplex(a)

        # ------------------------------------------------------
        # evaluate the true objective (cheap because we already have max_b)
        conv = np.convolve(a, a)
        max_b = conv.max()
        score = 2.0 * n * max_b          # sum(a) == 1 after projection

        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
            print(f"[improve] iter={iteration:,} score={best_score:.6f}")

        # ------------------------------------------------------
        # occasional stochastic shake‑up to hop out of plateaus
        if rng.random() < 0.01:
            pos = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.18)))
            a[pos] *= factor
            a = _project_simplex(a)

        # ------------------------------------------------------
        # early stop if we are already close to the bound
        if best_score <= theoretical_lb * 1.02:
            print("[early stop] near‑optimal bound reached")
            break

        # ------------------------------------------------------
        # random restart after a very long run (keeps diversity)
        if iteration % 200000 == 0:
            a = _random_initial_a(n, rng)
            m.fill(0.0)
            v.fill(0.0)
            print(f"[restart] iter={iteration:,}")

    # --------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    return [float(x) for x in best]
