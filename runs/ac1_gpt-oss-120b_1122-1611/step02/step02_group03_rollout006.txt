"""
Propose a low‑score height sequence for the self‑convolution objective.

Algorithm
---------
1. **Initial candidates** – several hand‑crafted families (uniform,
   arcsine, triangular, exponential, symmetric Beta‑shapes) and a random
   Dirichlet vector are generated.  The best among them (including the
   previous global `height_sequence_1` if present) is kept as a start.

2. **LP‑based improvement** – for the current sequence *f* we compute
   `M = max(f*f)`.  A linear program
       maximise   Σ g_j
       subject to  (f * g)_k ≤ M  for all k,
                   g_j ≥ 0
   is solved (SciPy highs).  The optimal *g* concentrates mass where *f*
   is small while respecting the convolution bound.  We mix the two
   sequences with a decaying factor `t` and project back onto the
   probability simplex.  Repeating this step (≈10–15 times) lowers the
   maximal convolution value substantially.

3. **Fine‑tuning** – the result of the LP phase is refined with a
   smoothed‑max Adam optimiser (the same idea as the reference code
   but with a slower temperature schedule and occasional random shakes).

4. **Termination** – the whole search stops when the wall‑clock budget
   expires (default ≈ 1000 s) or when the target score 1.5025 is reached.
   The best sequence found so far is returned.

The implementation keeps everything on the probability simplex
(`Σ a_i = 1`, `a_i ≥ 0`) so the objective reduces to `2·n·max(conv)`.
The LP step is inexpensive for `n = 1024` (≈ 10⁶ non‑zeros) and
greatly helps to escape the plateau observed in plain gradient descent.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        (f * g)_k ≤ M
    can be written as A @ g ≤ M where A is the Toeplitz matrix built
    from `f`.  The row/column pattern of A does not depend on the values
    of `f`; only the data changes.  This helper returns the fixed indices
    (row, col) that are reused for every LP solve.
    """
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    for j in range(n):
        start = j * n
        rows[start:start + n] = np.arange(j, j + n, dtype=np.int32)
        cols[start:start + n] = j
    return rows, cols


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Search for a non‑negative step‑function height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence discovered before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # reserve a couple of seconds for clean‑up/printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # problem size – 1024 is a good trade‑off between expressiveness and speed
    n = 1024

    # ------------------------------------------------------------------
    # 1) build a collection of diverse starting patterns
    patterns = []

    # uniform
    patterns.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine (Beta(0.5,0.5))
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)

    # triangular (tent)
    half = n // 2
    tent = np.concatenate(
        [np.linspace(1, 2, half, endpoint=False),
         np.linspace(2, 1, n - half)]
    )
    tent /= tent.sum()
    patterns.append(tent)

    # exponential decay
    exp_decay = np.exp(-np.linspace(0.0, 5.0, n))
    exp_decay /= exp_decay.sum()
    patterns.append(exp_decay)

    # symmetric Beta‑shapes with various α (α=β)
    for alpha in [0.2, 0.3, 0.4, 0.6, 0.8]:
        x = (np.arange(n) + 0.5) / n
        beta_shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        beta_shape /= beta_shape.sum()
        patterns.append(beta_shape)

    # random Dirichlet
    patterns.append(rng.dirichlet(np.ones(n)).astype(np.float64))

    # previous global solution, if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                if prev.size < n:
                    pad = np.zeros(n, dtype=np.float64)
                    pad[:prev.size] = prev
                    prev = pad
                else:
                    prev = prev[:n]
            prev = _project_simplex(prev)
            patterns.append(prev)
        except Exception:
            pass  # ignore malformed global variable

    # ------------------------------------------------------------------
    # evaluate the initial pool
    best_seq = None
    best_score = np.inf
    for idx, pat in enumerate(patterns):
        if time.monotonic() >= deadline:
            break
        score = float(evaluate_sequence(pat.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = pat.copy()
            print(f"[init] pattern {idx:02d} – score={best_score:.6f}")

    # fallback – should never happen
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 2) LP‑based improvement loop
    # Pre‑compute row/col indices for the Toeplitz matrix (independent of f)
    row_inds, col_inds = _precompute_toeplitz_indices(n)

    f = best_seq.copy()
    lp_iter = 0
    max_lp_iters = 15                     # enough to see a descent
    while time.monotonic() < deadline and lp_iter < max_lp_iters:
        conv = np.convolve(f, f)
        M = conv.max()

        # Build the sparse constraint matrix A where A @ g = f * g
        data = np.tile(f, n)               # length n*n
        A = sp.csr_matrix((data, (row_inds, col_inds)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # solve max Σ g_j  ⇔  min –Σ g_j  subject to A g ≤ b , g ≥ 0
        try:
            res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method='highs',
                options={"presolve": True}
            )
        except Exception as exc:
            print("[lp] linprog exception:", exc)
            break

        if not res.success:
            print(f"[lp] linprog failed (status {res.status}) – terminating LP phase")
            break

        g_raw = res.x
        if g_raw.sum() == 0.0:
            print("[lp] trivial solution obtained – stop LP")
            break
        g = g_raw / g_raw.sum()           # normalise to a probability vector

        # mixing factor – start relatively aggressive and decay
        t = 0.2 * (0.85 ** lp_iter)
        f_new = (1.0 - t) * f + t * g
        f_new = _project_simplex(f_new)

        # evaluate
        new_score = float(evaluate_sequence(f_new.tolist()))
        print(f"[lp] iter {lp_iter:02d} – t={t:.4f} – score={new_score:.6f}")

        if new_score < best_score:
            best_score = new_score
            best_seq = f_new.copy()
            print(f"[lp‑improve] new best = {best_score:.6f}")

        # always continue from the mixed point (even if not immediately better)
        f = f_new
        lp_iter += 1

    # ------------------------------------------------------------------
    # 3) fine‑tune with smoothed‑max Adam optimisation
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    total_iters = 0

    while time.monotonic() < deadline:
        # compute full self‑convolution
        conv = np.convolve(a, a)                 # length 2n‑1

        # smooth approximation of the maximum (soft‑max)
        mu = 5.0 * (1.0 + total_iters / 80000.0)  # temperature grows slowly
        cmax = conv.max()
        exp_args = mu * (conv - cmax)            # stabilise
        w = np.exp(exp_args)
        w /= w.sum()                             # ∂softmax/∂conv

        # gradient of the smooth max: 2·corr(w, a)
        corr_full = np.correlate(w, a, mode='full')
        grad = 2.0 * corr_full[n - 1: n - 1 + n]

        # Adam update on the simplex
        lr = 0.2 / np.sqrt(total_iters + 1.0)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (total_iters + 1))
        v_hat = v / (1.0 - beta2 ** (total_iters + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex (ensures Σa = 1, a≥0)
        a = _project_simplex(a)

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.008:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.18))
            a[idx] *= factor
            a = _project_simplex(a)

        total_iters += 1

        # periodic true evaluation
        if total_iters % 20 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[adam] iter {total_iters:06d} – score={best_score:.6f}")

            # early exit if we beat the target
            if best_score <= 1.5025:
                print("[early exit] target reached")
                return [float(x) for x in best_seq]

        # avoid a tight overrun
        if time.monotonic() + 0.05 >= deadline:
            break

    # ------------------------------------------------------------------
    print(f"[final] total_iters={total_iters}, best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
