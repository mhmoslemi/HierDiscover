    # conv length = len(w) + len(a_rev) - 1 = (2n-1)+n-1 = 3n-2

    # slice

    grad = conv[start:start+len(a)]
    return grad

def project_onto_simplex(v):
    # onto {x >= 0, sum x = 1}
    # v maybe negative after gradient step.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    # find rho = last index where condition holds
    rho = ind[cond][-1]  # if none? handle
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

via a projected (soft‑max) gradient descent on the smooth approximation
of the max‑convolution.  The algorithm works entirely with NumPy FFTs,
projects onto the probability simplex after each update, and uses a few
simple exploration tricks (random block perturbations, occasional pair‑wise
mass transfers and soft restarts).  It runs for the allotted time budget
(~1000 s) and returns the best vector seen.

Key ideas
---------
* Work with vectors normalised to unit sum → the denominator of the
  score is constantly 1.
* Approximate max(conv) with a soft‑max (log‑sum‑exp) whose temperature
  β is increased linearly during the run.  The gradient of this smooth
  surrogate is a weighted sum of the “partner” vectors of all convolution
  indices and can be obtained by a single FFT‑based convolution.
* Perform a projected gradient step (simple gradient descent) and keep
  the iterate inside the simplex (non‑negative, sum = 1).
* Occasionally apply a random multiplicative block perturbation and a
  cheap pair‑wise mass transfer to help escape shallow minima.
* When no improvement has been observed for a while, restart from the
  current best after a modest random perturbation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_self_conv(n: int) -> int:
    """FFT length required for a full linear self‑convolution of length n."""
    return _next_pow2(2 * n - 1)


def _fft_len_for_grad(n: int) -> int:
    """
    Convolution length needed for the gradient:
    (2n-1) (weights) + n (reversed a) – 1 = 3n‑2.
    """
    return _next_pow2(3 * n - 2)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a (real, non‑negative) using an FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may cause tiny negatives – they are harmless.
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a probability vector proportional to exp(beta * conv).
    Subtracting the maximum avoids overflow.
    """
    c_max = conv.max()
    exps = np.exp(beta * (conv - c_max))
    total = exps.sum()
    if total == 0.0:
        # fallback to uniform (should never happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exps / total


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the gradient ∑_k w_k * partner_vector(a, k) efficiently
    using one FFT‑based convolution (see analysis).
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(fw * fa, n=fft_len)
    start = a.size - 1  # index of the central element
    return conv[start : start + a.size]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Return the contribution of the entry a[idx] to the full self‑convolution.
    The result is a vector of length 2*n‑1 where the original a appears
    starting at position idx.
    """
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx : idx + n] = a
    return v
