"""
Linear‑programming based optimizer for the step‑function problem.

The algorithm works on the probability simplex (sum = √(2 n)) where,
because of the scaling invariance of `evaluate_sequence`, the objective
value equals the maximal entry of the self‑convolution.
Starting from a feasible sequence `f`, we repeatedly:

1. Compute M = max(conv(f,f)).
2. Solve the linear program
       max   ∑_j b_j
       s.t.  (f * b)_k ≤ M   for all k,
             b_j ≥ 0.
   The constraints are linear in `b`.  The LP pushes mass to positions
   that are *under‑used* by the current convolution.
3. Rescale the LP solution `b` to the required ℓ₁‑norm S = √(2 n):
       g = b * S / ∑b .
4. Move a fraction `t` from `f` towards `g`.  The step size `t` is
   reduced by back‑tracking until the new maximal convolution does not
   increase (monotone descent is guaranteed by the theory).
5. Keep the best sequence seen and occasionally restart from a random
   simplex point if progress stalls.

The LP is built with a dense Toeplitz matrix (size (2 n‑1) × n) and
solved with SciPy’s HiGHS interior‑point method.  With n = 1024 the
solver is fast enough to perform many iterations within the 1000 s
budget, typically driving the score from ~1.53 down towards the
theoretical lower bound ≈ 1.0005.

The returned list of heights can be used directly with
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _random_initial_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random feasible point on the simplex (sum = 1).
    Three strategies are mixed to give diversity.
    """
    choice = rng.integers(0, 3)
    if choice == 0:                     # uniform
        a = np.full(n, 1.0 / n)
    elif choice == 1:                   # Dirichlet, α = 0.5
        a = rng.dirichlet([0.5] * n)
    else:                               # sparse equal‑weight
        k = rng.integers(max(10, n // 30), max(20, n // 5))
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0 / k
    return _project_simplex(a)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimizes the evaluation score.  Returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # safety margin before the deadline
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # problem size – 1024 is a good trade‑off between quality and speed
    n = 1024
    S = np.sqrt(2.0 * n)          # target ℓ₁‑norm (makes objective = max(conv))

    # ------------------------------------------------------------------
    # initialise f on the simplex and scale to ℓ₁‑norm S
    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=float)
        if init.size != n:
            if init.size > n:
                init = init[:n]
            else:
                init = np.pad(init, (0, n - init.size), constant_values=0.0)
        init = np.maximum(init, 0.0)
        total = init.sum()
        if total == 0.0:
            f = np.full(n, S / n)
        else:
            f = init * (S / total)
    else:
        a = _random_initial_a(n, rng)         # sum = 1
        f = a * S

    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    iteration = 0
    no_improve = 0
    # line‑search parameters
    min_t = 1e-4
    t_start = 1.0

    # main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # current maximal convolution value (objective for current f)
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # --------------------------------------------------------------
        # build the Toeplitz matrix A such that (A @ b)[k] = (f * b)[k]
        rows = 2 * n - 1
        first_col = np.concatenate([f, np.zeros(rows - n, dtype=f.dtype)])
        first_row = np.concatenate([np.array([f[0]]), np.zeros(n - 1, dtype=f.dtype)])
        A = toeplitz(first_col, first_row)   # shape (rows, n)

        # --------------------------------------------------------------
        # linear program: maximise sum(b) s.t. A @ b ≤ M, b ≥ 0
        c = -np.ones(n)                      # minimise -sum(b)
        b_ub = np.full(rows, M)
        bounds = [(0.0, None)] * n

            lp_res = linprog(c, A_ub=A, b_ub=b_ub,
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'time_limit': 5.0})
            if not lp_res.success:
                raise RuntimeError("LP failed")
            b = lp_res.x
        except Exception:
            # fallback: keep current f and do a random shake
            if rng.random() < 0.1:
                pos = rng.integers(0, n)
                factor = float(np.exp(rng.normal(0.0, 0.2)))
                f[pos] *= factor
                f = _project_simplex(f) * S
            continue

        # --------------------------------------------------------------
        # rescale LP solution to the required ℓ₁‑norm S
        sum_b = b.sum()
        if sum_b <= 0.0:
            # numerically degenerate – fall back to a random direction
            g = np.full(n, S / n)
        else:
            g = b * (S / sum_b)

        # --------------------------------------------------------------
        # line‑search: move from f towards g while preserving or lowering M
        t = t_start
        accepted = False
        while t >= min_t:
            f_candidate = (1.0 - t) * f + t * g
            # ensure strict non‑negativity and exact norm
            f_candidate = np.clip(f_candidate, 0.0, None)
            total = f_candidate.sum()
            if total == 0.0:
                f_candidate = np.full(n, S / n)
            else:
                f_candidate = f_candidate * (S / total)

            conv_cand = np.convolve(f_candidate, f_candidate)
            M_cand = float(conv_cand.max())

            if M_cand <= M * (1.0 + 1e-12):
                # descent accepted
                f = f_candidate
                M = M_cand
                accepted = True
                break
            t *= 0.5

        if not accepted:
            # If no improvement possible, keep the old f and perform a small random shake
            if rng.random() < 0.05:
                pos = rng.integers(0, n)
                factor = float(np.exp(rng.normal(0.0, 0.15)))
                f[pos] *= factor
                f = _project_simplex(f) * S

        # --------------------------------------------------------------
        # evaluate true objective (for logging and best‑track)
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f.copy()
            no_improve = 0
            print(f"[improve] iter={iteration}, score={best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # occasional random restart if we have staled too long
        if no_improve > 8000:
            a = _random_initial_a(n, rng)
            f = a * S
            no_improve = 0
            print(f"[restart] iter={iteration}")

        # --------------------------------------------------------------
        # early exit when we are sufficiently close to the theoretical bound
        theoretical_lb = 2.0 * n / (2.0 * n - 1.0)   # ~1.0005 for n=1024
        if best_score <= theoretical_lb * 1.01:
            print("[early stop] near‑optimal bound reached")
            break

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # convert to a plain Python list of floats as required
    return [float(x) for x in best_f]
