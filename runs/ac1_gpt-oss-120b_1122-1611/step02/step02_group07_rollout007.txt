def propose_candidate(seed=42, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, budget_s - 0.5)
    n = 1024
    # possible initial candidates
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # descending linear
    linear = np.arange(n, 0, -1, dtype=np.float64)
    candidates.append(linear / linear.sum())
    # ascending linear (reverse)
    candidates.append(linear[::-1] / linear.sum())
    # random Dirichlet
    dir1 = rng.gamma(shape=0.5, size=n)
    candidates.append(dir1 / dir1.sum())
    # maybe previous best
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if prev.size != n:
                raise ValueError
            candidates.append(prev / prev.sum())
        except Exception:
            pass
    # refine each candidate using gradient descent
    best_seq = None
    best_score = float('inf')
    for init_idx, a in enumerate(candidates):
        a = _project_simplex(a)
        # gradient descent parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        base_lr = 0.6
        beta_min = 2.0
        beta_max = 150.0
        beta_inc_iters = 200_000
        it_local = 0
        while time.monotonic() < deadline and it_local < 200_000:
            it_local += 1
            conv = np.convolve(a, a)
            beta = beta_min + (beta_max - beta_min) * min(1.0, it_local / beta_inc_iters)
            max_c = conv.max()
            exps = np.exp(beta * (conv - max_c))
            w = exps / exps.sum()
            # gradient: conv of w with reversed a, times 2
            grad = 2.0 * np.convolve(w, a[::-1])[n-1:n-1+n]
            # Adam
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** it_local)
            v_hat = v / (1 - beta2 ** it_local)
            lr = base_lr / np.sqrt(it_local)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # occasional larger perturbation
            if rng.random() < 0.07:
                idx = rng.integers(0, n)
                a_new[idx] *= np.exp(rng.normal(0, 0.2))
            a_new = _project_simplex(a_new)
            # evaluate occasionally
            if it_local % 1000 == 0:
                score = float(evaluate_sequence(a_new.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_seq = a_new.copy()
                    best_score = score
                    print(f"[seed{init_idx}] new best {best_score:.6f} iter={it_local}")
                if best_score <= 1.5025:
                    print("[target] reached")
                    return [float(x) for x in best_seq]
            a = a_new
        # final eval of this seed
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            print(f"[seed{init_idx}] final best {best_score:.6f}")
    # SA phase
    cur = best_seq.copy()
    cur_score = best_score
    T0 = 0.02
    # schedule T = T0 * (time_remaining / total_remaining)
    while time.monotonic() < deadline:
        # temperature schedule based on remaining fraction
        fraction = (deadline - time.monotonic()) / (deadline - (deadline - budget_s + 0.5))
        T = T0 * fraction
        # propose perturbation
        a_new = cur.copy()
        # pick random subset size k
        k = max(1, int(0.02 * n))
        idxs = rng.choice(n, size=k, replace=False)
        # add Gaussian noise
        a_new[idxs] += rng.normal(0.0, 0.02, size=k)
        # clip negative to zero
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        new_score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(new_score):
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                cur = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_seq = cur.copy()
                    best_score = cur_score
                    print(f"[SA] improve {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[target] reached in SA")
                    break
        # occasionally random restart from best with noise
        if rng.random() < 0.01:
            cur = best_seq + rng.normal(0.0, 0.01, size=n)
            cur = np.maximum(cur, 0.0)
            cur = _project_simplex(cur)
            cur_score = float(evaluate_sequence(cur.tolist()))
    print(f"[final] best score {best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Propose a low‑score step‑function using a two‑stage search:

1️⃣  A fast Adam optimizer on a smooth‐max approximation of the
    convolution (as in the reference code) but with a sharper schedule,
    more aggressive shake‑ups and several different warm‑starts (uniform,
    linear, random Dirichlet and the previously discovered best).

2️⃣  A simulated‑annealing refinement that repeatedly moves a tiny amount
    of mass from one index to another.  The acceptance probability is
    temperature‑controlled, allowing occasional uphill moves to escape
    shallow local minima.  Random “large‑perturbation’’ restarts from the
    current best keep the search from stagnating.

Both stages keep the vector on the probability simplex (∑aᵢ = 1,
aᵢ ≥ 0) – the objective is invariant to scaling, so fixing the ℓ₁‑norm
removes a degree of freedom and improves numerical stability.

The routine respects the time budget (default 1000 s) and returns the
best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:               # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray,
                         beta_min: float,
                         beta_max: float,
                         it: int,
                         beta_inc_iters: int) -> np.ndarray:
    """
    Return the gradient of the smooth‑max approximation
    of max_k (a*a)[k] with respect to a.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    beta = beta_min + (beta_max - beta_min) * min(1.0, it / beta_inc_iters)
    # stable log‑sum‑exp
    max_c = conv.max()
    exps = np.exp(beta * (conv - max_c))
    w = exps / exps.sum()                         # soft‑max weights (2n‑1,)
    # gradient = 2 * (w * a_rev) convolved
    grad_full = np.convolve(w, a[::-1])          # length 3n‑2
    grad = 2.0 * grad_full[n - 1:n - 1 + n]      # keep the middle n entries
    return grad, beta


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (non‑negative heights) that minimises
    ``evaluate_sequence``.  The search consists of a gradient‑descent
    phase followed by a simulated‑annealing phase; multiple warm‑starts
    are tried and the best vector encountered is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time handling
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # helper: create a simplex‑normalised vector from a raw array
    def _norm(v):
        return _project_simplex(v.astype(np.float64))

    # ------------------------------------------------------------------
    # build a list of warm‑start candidates
    warm_starts = []

    # 1) uniform
    warm_starts.append(np.full(n, 1.0 / n))

    # 2) linearly decreasing (large at the left, small at the right)
    linear = np.arange(n, 0, -1, dtype=np.float64)
    warm_starts.append(_norm(linear))

    # 3) linearly increasing (mirror of #2)
    warm_starts.append(_norm(linear[::-1]))

    # 4) random Dirichlet (sparseish)
    rand_dir = rng.gamma(shape=0.5, size=n)
    warm_starts.append(_norm(rand_dir))

    # 5) previous best if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.size == n:
                warm_starts.append(_norm(prev))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # keep track of the overall best solution
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser on a smooth‑max loss
    # --------------------------------------------------------------
    # shared Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.6
    beta_min, beta_max = 5.0, 150.0
    beta_inc_iters = 200_000          # after this β stays at β_max

    print("[phase‑1] start – gradient descent on smooth max")
    for start_id, a0 in enumerate(warm_starts):
        a = _norm(a0)
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0

        # evaluate the start
        cur_score = float(evaluate_sequence(a.tolist()))
        if cur_score < best_score:
            best_seq, best_score = a.copy(), cur_score
            print(f"[phase‑1][{start_id}] initial score = {best_score:.6f}")

        # run until time runs out or we reach a reasonable iteration cap
        while time.monotonic() < deadline:
            it += 1

            # --- gradient of smooth max ---------------------------------
            grad, beta = _smooth_max_gradient(a, beta_min, beta_max, it, beta_inc_iters)

            # --- Adam update -------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            a_next = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # --- occasional stronger shake‑up ---------------------------
            if rng.random() < 0.07:                     # ~7 % of iterations

            # --- cheap periodic evaluation (every 2000 iters) ----------

    # ------------------------------------------------------------------

    # --------------------------------------------------------------
    print("[phase‑2] start – simulated annealing refinement")
    # use the best sequence from phase‑1 as the starting point
    cur = best_seq.copy()

        T = T_end + (T_start - T_end) * frac

        a_new = cur.copy()

        # pick a distinct destination uniformly

        a_new[src] -= frac_move

        a_new = _project_simplex(a_new)

            print(f"[status] moves={move_counter:,}  best={best_score:.6f}  elapsed={elapsed:.1f}s")

    # ------------------------------------------------------------------

    return [float(x) for x in best_seq]
