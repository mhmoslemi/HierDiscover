def fft_convolve(a, b):
    m = 1 << ( (len(a)+len(b)-1).bit_length() )
    A = np.fft.rfft(a, m)
    B = np.fft.rfft(b, m)
    return np.fft.irfft(A*B, m)[:len(a)+len(b)-1]

    # Adam variables
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_score = np.inf

    # precompute conv length size

    total_iters = 200000  # estimate

    beta1=0.9; beta2=0.999; eps_adam=1e-8; lr0=0.1

    iteration = 0
    # gradient descent phase
    while time.monotonic() < deadline:
        iteration+=1

        # compute a as softmax of x
        # stable softmax: subtract max(x)
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)

        m = beta1 * m + (1 - beta1) * grad_x

        lr = lr0 / (np.sqrt(iteration) + 1.0)

"""

The algorithm has two phases:

1. **Linear‑programming mixing (inspired by the reference method).**  
   Starting from an initial non‑negative vector `a` (∑a = 1) we repeatedly
   solve the LP  

       maximise   1ᵀ g
       subject to g ≥ 0 , (a * g) ≤ M                (M = max(a*a))

   where the convolution `(a * g)` is linear in the new variable `g`.  
   The optimal `g` is mixed into `a` (a ← (1‑t)·a + t·g) and the process
   is repeated a few times.  This step quickly lowers the peak of the
   self‑convolution.

2. **Smooth‑max Adam optimisation.**  
   The objective `max_k (a*a)[k]` is approximated by the
   log‑sum‑exp surrogate  

       Sα(a) = (1/α)·log Σ_k exp(α·(a*a)[k]) .

   applied to `a` while projecting onto the probability simplex after
   each update.  The surrogate temperature `α` is increased
   gradually so that the optimisation finally targets the true maximum.

allows us to work with n = 2048 (≈ 2 × 10⁶ variables in the LP) while

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution via FFT (real transforms).  Returns the full
    length‑(len(a)+len(b)-1) array.
    """
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    c = np.fft.irfft(A * B, fft_len)
    return c[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
def _fft_corr_valid(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Valid‑mode correlation of w (length 2n‑1) with a (length n)
    using FFT.  Returns a vector of length n.
    """
    a_rev = a[::-1]
    conv_full = _fft_convolve(w, a_rev, fft_len)
    n = a.shape[0]
    start = n - 1
    return conv_full[start:start + n]

            prob = cp.Problem(cp.Maximize(cp.sum(g)),
                              [A @ g <= M])
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
            if g.value is None:
                print("[lp] solver failed – abort LP phase")
                break

            g_vec = np.maximum(g.value, 0.0)
            if g_vec.sum() == 0.0:
                # degenerate – stop LP phase
                break
            g_vec /= g_vec.sum()

            # Mixing factor – start aggressive, then decay
            t = 0.4 / outer_iter   # e.g. 0.4, 0.2, 0.133, …
            a = (1.0 - t) * a + t * g_vec
            a = _project_simplex(a)   # ensure feasibility

            # evaluate
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a.copy()
                print(f"[lp improve] outer={outer_iter}, score={best_score:.6f}")

            # early exit if we already beat the target by a decent margin
            if best_score <= 1.55:
                print("[lp] reached score good enough – skipping remaining LP steps")
                break

    # --------------------------------------------------------------
    # Phase 2 – smooth‑max Adam optimisation ----------------------------------------
    # Adam hyper‑parameters (operating directly on the probability vector)
    beta1, beta2, eps_adam = 0.90, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max (smooth‑max)
    alpha = 2.0                 # start smooth
    alpha_target = 200.0
    alpha_inc = (alpha_target - alpha) / 300_000   # linear ramp

    # learning‑rate schedule
    lr0 = 0.12

    iteration = 0
    fft_len_corr = _next_pow2(3 * n - 2)   # length needed for w * a_rev

    while time.monotonic() < deadline:
        iteration += 1

        # ---------- self‑convolution (FFT) ----------
        conv_self = _fft_convolve(a, a, _next_pow2(2 * n - 1))
        max_conv = conv_self.max()

        # ---------- smooth‑max weights ----------
        # numerical stabilisation: shift by max_conv
        exps = np.exp(alpha * (conv_self - max_conv))
        w = exps / exps.sum()                     # length 2n‑1, sums to 1

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _fft_corr_valid(w, a, fft_len_corr)   # length n

        # ---------- Adam update on the probability vector ----------
        m = beta1 * m + (1.0 - beta1) * grad_a
        v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- projection onto the simplex ----------
        a = _project_simplex(a_new)

        # ---------- occasional random shake ----------
        if rng.random() < 0.04:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.2)))   # log‑normal jitter
            a[idx] *= factor
            a = _project_simplex(a)

        # ---------- evaluate exact score from time to time ----------
        if iteration % 200 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a.copy()
                print(f"[improve] iter={iteration:,}, score={best_score:.6f}")

        # ---------- temperature update ----------
        if alpha < alpha_target:
            alpha = min(alpha_target, alpha + alpha_inc)

        # ---------- early stop if we are very close to the theoretical bound ----------
        theoretical_lb = 2.0 * n / (2 * n - 1)
        if best_score <= theoretical_lb * 1.02:
            print("[early stop] reached near‑optimal theoretical bound")
            break

        # safety check for the time budget
        if iteration % 10_000 == 0 and time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # Final sanity: make sure we have a feasible answer
    if best.sum() == 0.0 or not np.all(np.isfinite(best)):
        best = np.full(n, 1.0 / n, dtype=np.float64)

    final_score = float(evaluate_sequence(best.tolist()))
    print(f"[final] best score={final_score:.6f} after {iteration:,} iterations")
    return [float(x) for x in best]
