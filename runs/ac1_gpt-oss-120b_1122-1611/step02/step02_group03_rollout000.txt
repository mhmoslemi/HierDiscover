"""
Propose a low‑score height sequence for the self‑convolution objective.

Algorithm
---------
We treat the problem as a *min‑max* optimisation:

    minimise   M(a) = max_k (a * a)[k]          (a ≥ 0, Σ a = 1)

The value returned by `evaluate_sequence` for a probability vector
(`Σa = 1`) is simply `2·n·M(a)`.  The function `M(a)` is the
∞‑norm of a linear mapping `g ↦ a * g`; therefore for a fixed `a`
the feasible set `{g ≥ 0 : a * g ≤ M(a)}` is a polyhedron.
We can move `a` towards a vector `g` that has the largest possible
ℓ₁‑mass while respecting the same bound `M(a)`.  This is a classic
conditional‑gradient (Frank‑Wolfe) step for the min‑max problem.

Procedure
---------
1. Initialise `a` with a diverse set of patterns (uniform, arcsine,
   triangular, random Dirichlet, and the previously best sequence if
   available).  All vectors are projected onto the probability simplex
   (`ℓ₁ = 1`, non‑negative).

2. Repeatedly:
   * Compute the current peak `M = max(convolve(a, a))`.
   * Solve a linear programme

        maximise   Σ g_i
        subject to A(a)·g ≤ M   (element‑wise)
                    g ≥ 0

     where `A(a)` is the full convolution matrix mapping `g` to
     `a * g`.  The LP finds the feasible `g` with maximal mass.
   * Normalise `g` to a probability vector (`ĝ = g / Σg`).
   * Update the current iterate with a convex combination
     `a ← (1‑t)·a + t·ĝ` (t is a modest step size, e.g. 0.2) and project
     back onto the simplex to avoid numerical drift.
   * Keep the best vector seen so far (lowest `2·n·M`).

   The loop runs until the wall‑clock budget expires or the target
   score (`≤ 1.5025`) is reached.  Several random restarts are performed
   automatically.

3. Return the best sequence found (as a plain list of floats).

The LP‑driven step is much more global than a plain gradient update:
it simultaneously reduces all active convolution constraints, which
typically drives the peak lower faster than the smooth‑max Adam
approach used previously.

The implementation uses `scipy.optimize.linprog` with the HiGHS solver,
which comfortably handles the dense convolution matrix of size
≈ 2 000 × 1 000 for `n = 1024`.  All heavy work is performed in NumPy /
SciPy C‑code, so the 1000 s budget is more than sufficient for many
thousands of LP iterations.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that A @ g = np.convolve(a, g, mode='full').

    For a of length n, A has shape (2n‑1, n).  The construction uses a
    Toeplitz matrix with the first column equal to a padded with n‑1 zeros
    and the first row equal to a[0] followed by n‑1 zeros.
    """
    from scipy.linalg import toeplitz

    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    return toeplitz(col, row)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence discovered before the
    time budget expires.

    Parameters
    ----------
    seed : int, default 42
        Random seed for reproducibility.
    budget_s : float, default 1100
        Wall‑clock time (seconds) allocated for the search.  A small
        safety margin is kept for the final return.
    """
    # --------------------------------------------------------------
    # Imports that may be heavy – do them lazily.
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)

    # Reserve a couple of seconds for clean‑up/printing.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Problem size – 1024 is a good balance between expressiveness and
    # linear‑program size (≈ 2 000 × 1 000).
    n = 1024

    # --------------------------------------------------------------
    # Build a collection of diverse starting points.
    patterns = []

    # Uniform
    patterns.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Arcsine‑shaped (heavy at the ends, light in the centre)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)

    # Simple triangular (tent) shape
    half = n // 2
    tent = np.concatenate([np.linspace(1, 2, half, endpoint=False),
                           np.linspace(2, 1, n - half)])
    tent /= tent.sum()
    patterns.append(tent)

    # Random Dirichlet vector – stochastic diversity
    patterns.append(rng.dirichlet(np.ones(n)).astype(np.float64))

    # Previous best if present
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                if prev.size < n:
                    # pad with tiny uniform tail
                    pad = np.full(n, 1.0 / n, dtype=np.float64)
                    pad[:prev.size] = prev
                    prev = pad
                else:
                    prev = prev[:n]
            prev = _project_simplex(prev)
            patterns.append(prev)
        except Exception:
            pass  # ignore malformed global variable

    # --------------------------------------------------------------
    # Optimisation state that survives restarts.
    best_seq = None
    best_score = float('inf')
    total_iters = 0

    # Mix‑in factor (convex‑combination weight).  We keep it modest
    # so the sequence evolves smoothly.
    t = 0.2

    # Main outer loop – several random restarts until time runs out.
    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # Pick a start pattern (randomly) and make sure it lies on the simplex.
        a = rng.choice(patterns).copy()
        a = _project_simplex(a)               # enforce Σa = 1, a ≥ 0

        # ------------------------------------------------------------------
        # Inner Frank‑Wolfe iterations.
        inner_limit = 300               # usually enough per restart
        for _ in range(inner_limit):
            if time.monotonic() >= deadline:
                break

            total_iters += 1

            # --------------------------------------------------------------
            # 1. Current peak of the self‑convolution.
            conv_aa = np.convolve(a, a)               # length 2n‑1
            M = conv_aa.max()
            # The true evaluation score (Σa = 1) is 2·n·M.
            score = 2.0 * n * M

            # --------------------------------------------------------------
            # 2. Record the best sequence seen so far.
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[improve] iter={total_iters}, score={best_score:.6f}")

                # Early exit if we have already beaten the target.
                if best_score <= 1.5025:
                    print("[early exit] target reached")
                    return [float(x) for x in best_seq]

            # --------------------------------------------------------------
            # 3. Build the convolution matrix A(a) such that A·g = a * g.
            A = _convolution_matrix(a)                # (2n‑1, n)

            # 4. Linear programme: maximise Σ g_i  subject to  A·g ≤ M , g ≥ 0.
            c = -np.ones(n, dtype=np.float64)         # minimise –Σg  → maximise Σg
            b_ub = np.full(A.shape[0], M, dtype=np.float64)

            # Use HiGHS (fast, robust) – bounds enforce g ≥ 0.
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})
            if not res.success:
                # If LP failed (numerical glitch), fall back to a tiny random shake.
                idx = rng.integers(0, n)
                a[idx] *= np.exp(rng.normal(0.0, 0.2))
                a = _project_simplex(a)
                continue

            g = res.x
            sum_g = g.sum()
            if sum_g <= 1e-12:
                # No feasible direction found – break inner loop.
                break

            # --------------------------------------------------------------
            # 5. Normalise g to the simplex and take a convex step.
            g_norm = g / sum_g
            a = (1.0 - t) * a + t * g_norm
            a = _project_simplex(a)   # guard against numerical drift

            # Occasionally inject a random perturbation to escape plateaus.
            if rng.random() < 0.005:
                j = rng.integers(0, n)
                a[j] *= np.exp(rng.normal(0.0, 0.15))
                a = _project_simplex(a)

        # End of inner loop – automatically restart with a fresh pattern.
        # (If we have already hit the target, the outer loop will terminate.)

    # ----------------------------------------------------------------------
    # If nothing ever succeeded (unlikely), fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(2.0 * n * np.convolve(best_seq, best_seq).max())

    print(f"[final] total_iters={total_iters}, best_score={best_score:.6f}")
    # Return a plain Python list as required by the evaluation wrapper.
    return [float(x) for x in best_seq]
