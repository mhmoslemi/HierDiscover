"""
propose_candidate
-----------------
A smooth‑max / Adam based optimizer for the convex problem  

    minimise   max_k (w*w)[k]          (w ≥ 0,  Σ w = 1)

The non‑smooth max is replaced by a softmax with a temperature
parameter α that is slowly increased.  The (smoothed) gradient is the
correlation of the softmax weights with w, which can be obtained via an
FFT‑based convolution.  Adam with a projected step onto the probability
simplex guides the search.  Periodic random block perturbations and an
optional symmetry enforcement help to escape local minima.  The best
candidate found within the time budget is rescaled to the conventional
√(2 n) L¹‑norm before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers scaled by 4⁻ⁱ:
    a_i = C(2i,i) / 4ⁱ   (recurrence a_{i+1}=a_i*(2i+1)/(2i+2))."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (non‑circular) using real FFT.
    Small negative round‑off errors are clipped to 0."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # guard against -0 caused by rounding
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump – multiply a random contiguous block by a
    log‑normal factor, optionally symmetrise."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that aims
    to minimise ``evaluate_sequence``.  The algorithm uses a smooth‑max
    proxy for the L∞ norm and Adam with simplex projection.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length – reuse a previously discovered construction if
    # available, otherwise fall back to a few thousand entries.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 6000
    else:
        n = 6000

    # --------------------------------------------------------------
    # Initialise w (probability vector).  Prefer the known construction,
    # otherwise start from the Catalan sequence.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init_arr = np.asarray(height_sequence_1, dtype=np.float64)
            if init_arr.size == n:
                w = np.maximum(init_arr, 0.0)
            else:
                raise ValueError
        except Exception:
            w = catalan_sequence(n)
    else:
        w = catalan_sequence(n)

    w = w / np.sum(w)                     # enforce Σw = 1

    # --------------------------------------------------------------
    # Book‑keeping of the best candidate seen so far.
    # --------------------------------------------------------------
    b = convolve_fft(w, w)
    best_score = float(2 * n * np.max(b))    # same as evaluate_sequence after scaling
    best_w = w.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------------
    m = np.zeros(n, dtype=np.float64)   # first moment
    v = np.zeros(n, dtype=np.float64)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.15                          # base learning rate
    alpha = 10.0                        # softmax temperature (larger → sharper)

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop – terminates a few seconds before the
    # given budget to give a safety margin.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ---- 1. current self‑convolution ---------------------------------
        conv = convolve_fft(w, w)                 # length 2n‑1

        # ---- 2. soft‑max weights over the convolution --------------------
        maxc = np.max(conv)
        # stabilise exponential by subtracting maxc
        exp_vals = np.exp(alpha * (conv - maxc))
        p = exp_vals / np.sum(exp_vals)          # sum_i p_i = 1

        # ---- 3. gradient of the smoothed objective -----------------------
        # ∂/∂w  Σ_i p_i (w*w)[i] = 2 * corr(p, w)
        w_rev = w[::-1]
        grad_full = convolve_fft(p, w_rev)       # length 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]   # size n

        # ---- 4. Adam update --------------------------------------------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate schedule (decays like 1/√t)
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w = simplex_projection(w_new)            # keep on simplex

        # ---- 5. optional symmetry enforcement ---------------------------
        if iteration % 800 == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

        # ---- 6. occasional evaluation of the true objective ------------
        if iteration % 400 == 0:
            conv_now = convolve_fft(w, w)
            score = 2.0 * n * np.max(conv_now)
            if score < best_score - 1e-9:

                best_w = w.copy()
                no_improve = 0

            else:

        if iteration % 1200 == 0 and iteration > 0:

            cand = cand / np.sum(cand)
            conv_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(conv_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

    # conventional √(2 n) L¹‑norm for readability.
    # --------------------------------------------------------------

    # Return plain Python list as required.
    return [float(x) for x in final_seq]
