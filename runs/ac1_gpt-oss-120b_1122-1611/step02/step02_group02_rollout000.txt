"""
Gradient‑descent on a smooth L∞ approximation.

We minimise 2·n·max(a*a)/(∑a)² for a non‑negative vector a.
Because the objective is scale‑invariant we keep ∑a = 1.
To smooth the non‑differentiable max we use a soft‑max (log‑sum‑exp)
approximation with a temperature α that grows during the run.
The gradient of the soft‑max is a weighted sum of the “partner” vectors,
which we compute efficiently by a second FFT‑based convolution.
The update is an exponentiated‑gradient (mirror descent on the simplex)
with momentum, occasional stochastic perturbations and a stagnation‑restart.
After the main phase a short local‑search (random swaps / block scaling)
finishes the optimisation.

The algorithm works on the full sequence (or the provided `height_sequence_1`);
symmetry is enforced by averaging with the reversed vector each iteration.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_self_conv(n: int) -> int:
    """FFT length needed for a linear self‑convolution of length n."""
    return _next_pow2(2 * n - 1)


def _fft_len_for_conv(la: int, lb: int) -> int:
    """FFT length needed for linear convolution of lengths la and lb."""
    return _next_pow2(la + lb - 1)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a (real, non‑negative) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k·(a*a)[k] efficiently.
    This equals the convolution of a with the reversed weight vector.
    """
    n = a.size
    # reverse the weight vector (size 2n‑1)
    w_rev = weights[::-1]
    fft_len = _fft_len_for_conv(n, w_rev.size)
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    prod = fa * fw
    full = np.fft.irfft(prod, n=fft_len)
    # slice the region corresponding to indices 0 … n‑1
    start = n - 1
    return full[start : start + n]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _init_arcsine(n: int) -> np.ndarray:
    """
    Initialise with a discrete arcsine‑like density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    This shape gives a very flat self‑convolution in the interior.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds (leaving a tiny safety margin).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Determine problem size and initialise a candidate vector
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # fallback size – a power of two works nicely with FFT
        n = 4096

    # Start from a smooth arcsine shape with a small random jitter
    a = _init_arcsine(n)
    a += rng.normal(scale=0.001, size=n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    # Pre‑computed FFT length for self‑convolution
    self_fft_len = _fft_len_for_self_conv(n)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    best = a.copy()
    # fast evaluation (since sum=1): 2·n·max(conv)
    conv = _conv_self(a, self_fft_len)
    best_score = 2.0 * n * np.max(conv)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # optimisation hyper‑parameters
    alpha_start, alpha_end = 0.5, 14.0          # temperature for soft‑max
    lr_start, lr_end = 0.8, 0.02                # exponentiated‑gradient step size
    beta = 0.9                                  # momentum factor
    momentum = np.zeros_like(a)
    stagnation_limit = 45.0                     # seconds without improvement → restart
    last_improve = start_time

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- 1. self‑convolution and current score -------------------
        conv = _conv_self(a, self_fft_len)
        cur_score = 2.0 * n * np.max(conv)

        # ----- 2. record improvement ----------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            t = last_improve - start_time
            print(f"[{t:6.2f}s] new best score = {best_score:.6f}")

        # ----- 3. schedule for temperature and step size ---------------
        elapsed_frac = (time.monotonic() - start_time) / (deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * elapsed_frac
        lr = lr_start + (lr_end - lr_start) * elapsed_frac

        # ----- 4. soft‑max weights over convolution entries ------------
        conv_max = np.max(conv)
        # work in the shifted domain to avoid overflow
        scaled = alpha * (conv - conv_max)
        exp_scaled = np.exp(scaled)
        weights = exp_scaled / exp_scaled.sum()            # shape = 2n‑1, sum = 1

        # ----- 5. gradient of the smooth objective --------------------
        grad = _partner_gradient(a, weights)                # size n
        # momentum smoothing
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- 6. exponentiated‑gradient update (mirror descent) ------
        a = a * np.exp(-lr * momentum)
        a = np.clip(a, 1e-14, None)                        # avoid exact zeros
        a /= a.sum()

        # ----- 7. enforce symmetry (helps a lot) ----------------------
        a = (a + a[::-1]) * 0.5
        a /= a.sum()

        # ----- 8. occasional stochastic perturbations -----------------
        r = rng.random()
        if r < 0.025:
            a = _random_block_perturb(a, rng)
        elif r < 0.045:
            a = _pairwise_swap_perturb(a, rng)

        # ----- 9. restart if stagnant ---------------------------------
        if time.monotonic() - last_improve > stagnation_limit:
            if rng.random() < 0.6:
                a = _init_arcsine(n)
            else:
                a = rng.gamma(shape=0.5, scale=1.0, size=n)   # a broader Dirichlet‑like start
                a = np.clip(a, 0.0, None)
                a /= a.sum()
            a += rng.normal(scale=0.001, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            momentum[:] = 0.0
            last_improve = time.monotonic()
            t = last_improve - start_time
            print(f"[{t:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Local refinement: simple hill‑climbing around the best vector
    # ------------------------------------------------------------------
    # Use any leftover time for a cheap stochastic search.
    while time.monotonic() < deadline:
        # propose a neighbour of the current best
        if rng.random() < 0.5:
            cand = _random_block_perturb(best, rng)
        else:
            cand = _pairwise_swap_perturb(best, rng)
        # enforce symmetry
        cand = (cand + cand[::-1]) * 0.5
        cand /= cand.sum()

        conv = _conv_self(cand, self_fft_len)
        cand_score = 2.0 * n * np.max(conv)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            t = time.monotonic() - start_time
            print(f"[{t:6.2f}s] refined best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to satisfy evaluate_sequence limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best.tolist()]
