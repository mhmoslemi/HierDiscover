"""
Simple step‑function optimiser.

The optimiser works in three stages:

1. **Diversified initialisation** – a small set of analytically‑motivated
   candidates (uniform, sinusoidal, quadratic, random Dirichlet and the
   previous best if available) are built and each is refined for a few
   seconds with a smooth‑max Adam optimiser.

2. **Main Adam optimisation** – the current best candidate is optimised for
   the remaining time budget.  The optimiser works on the probability
   simplex, uses a temperature‑controlled soft‑max surrogate for the
   `max` in the evaluation, enforces symmetry after each projection
   (the optimal shape is believed to be symmetric) and occasionally shakes
   the vector with a log‑normal perturbation to escape plateaus.

3. **High‑temperature fine‑tuning** – once the temperature has reached the
   final value the optimiser continues with a much sharper soft‑max
   (α≈300) and a slower learning‑rate decay, which effectively performs a
   projected sub‑gradient descent on the true `max` objective.

The whole routine is time‑bounded (≈ 1000 s) and returns the best sequence
found.  The code is self‑contained and only uses NumPy and SciPy.
"""

import time
import numpy as np
from scipy.optimize import linprog  # only imported for optional LP fallback


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥0, Σx=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry: v_i ← (v_i + v_{n‑1‑i}) / 2."""
    return (v + v[::-1]) / 2.0


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max over the convolution vector."""
    shift = conv.max()
    # exponent may underflow for very large α – guard against that.
    exps = np.exp(alpha * (conv - shift))
    s = exps.sum()
    if s == 0.0:                     # all values underflowed → fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exps / s


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    best_global_vec: np.ndarray,
    best_global_score: float,
):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best vector encountered and its score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = best_global_score

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.90, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule (soft‑max sharpness)
    alpha_start = 5.0
    alpha_end = 300.0
    # we roughly anticipate 500 000 updates – adjust increment lazily
    alpha = alpha_start
    alpha_inc = (alpha_end - alpha_start) / 600_000.0

    lr0 = 0.15                     # base learning‑rate

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ----- convolution & soft‑max weighting ---------------------------------
        conv = np.convolve(a, a)                     # length 2n‑1
        w = _softmax_weights(conv, alpha)

        # ----- gradient of the smooth‑max --------------------------------------
        grad = 2.0 * np.correlate(w, a, mode='valid')   # length n

        # ----- Adam step -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        lr = lr0 / (np.sqrt(iteration) + 1.0)          # gentle decay
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -------------------------------------
        a_new = _project_simplex(a_new)

        # ----- random perturbation (≈5 % of the steps) -------------------------
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.2)))   # log‑normal tweak
            a_new[idx] *= factor
            a_new = _project_simplex(a_new)

        # ----- enforce symmetry -------------------------------------------------
        a_new = _symmetrize(a_new)

        a = a_new

        # ----- evaluate ---------------------------------------------------------
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
            print(
                f"[improve] iter={iteration:,}, score={best_score:.6f}, α={alpha:.2f}"
            )

        # ----- temperature update ------------------------------------------------
        if alpha < alpha_end:
            alpha = min(alpha_end, alpha + alpha_inc)

        # ----- early exit if we are already very close to the theoretical bound
        theoretical_lb = 2.0 * n / (2 * n - 1)  # = 2n/(2n‑1)
        if best_score <= theoretical_lb * 1.02:
            print("[early stop] near‑optimal theoretical bound reached")
            break

        # ----- occasional progress report ---------------------------------------
        if iteration % 150_000 == 0:
            print(
                f"[status] iter={iteration:,} – best so far {best_score:.6f}, α={alpha:.2f}"
            )

    return best, best_score


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a handful of diverse start vectors (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. half‑cosine (positive, smooth, symmetric)
    x = np.linspace(0, np.pi, n)
    a = np.sin(x)                    # sin(0)=0, sin(π)=0, peak in middle
    a = np.maximum(a, 0.0)
    a /= a.sum()
    cand.append(a)

    # 3. quadratic “mountain”
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)           # symmetric, zero at the ends
    a = np.maximum(a, 0.0)
    a /= a.sum()
    cand.append(a)

    # 4. random Dirichlet (moderately spiky)
    a = rng.dirichlet(np.full(n, 0.5))
    cand.append(a)

    # 5. random Dirichlet (smooth)
    a = rng.dirichlet(np.full(n, 5.0))
    cand.append(a)

    # 6. previous best if present (upsampled / truncated)
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.size > 1:
                # interpolate to current size
                x_old = np.linspace(0, 1, prev.size)
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, prev)
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 0:
                    a /= a_sum
                    cand.append(a)
        except Exception:
            pass

    # make every candidate symmetric right away
    cand = [_symmetrize(c) for c in cand]
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
