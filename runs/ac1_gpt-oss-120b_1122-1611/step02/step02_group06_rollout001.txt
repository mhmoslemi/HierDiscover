import time, numpy as np, scipy.optimize as opt

def build_conv_matrix(a):
    n = len(a)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Add columns
    for j in range(n):
        A[j:j+n, j] = a
    return A
```

Then:

```
def solve_lp(a, M):
    A = build_conv_matrix(a)
    c = -np.ones(a.shape)  # maximize sum(b)
    bounds = [(0, None)] * len(a)
    res = opt.linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
```

"""
Propose a candidate step‑function using a two‑phase optimisation:

1. **Convex‑programming phase** – Repeatedly solve a linear program
   max Σb  subject to   (a * b)[k] ≤ M   for all shifts k,
   where M = max(a * a) and a is the current distribution on the simplex.
   The optimiser computes a new distribution g (the rescaled LP solution)
   and replaces a by g (or a convex combination with g).  This
   monotonically reduces the worst convolution value.

2. **Local refinement phase** – Starting from the best distribution found
   by the LP phase, run a projected Adam optimiser on a smooth
   approximation of the max‑convolution (the same surrogate used in the
   reference code).  This fine‑tunes the sequence and often squeezes
   the objective a little further.

Both phases respect the non‑negativity and ℓ₁‑normalisation constraints,
so `evaluate_sequence` can be called directly on the candidate.
The routine respects the 1000 s budget and prints progress.
"""

import time
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the matrix A such that A @ b = a * b (full convolution)
    for a fixed left operand `a`.  Shape of A is (2n‑1, n).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j is a shifted down by j positions
    for j in range(n):
        A[j:j + n, j] = a
    return A


# ----------------------------------------------------------------------
def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ b_j   s.t.   (a * b)_k ≤ M  ∀ k,  b_j ≥ 0 .
    Returns the optimal b (not normalised) or None on failure.
    """
    A = _build_conv_matrix(a)
    c = -np.ones_like(a)                # maximise Σb  →  minimise -Σb
    bounds = [(0.0, None)] * a.shape[0]
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    # use HiGHS interior‑point LP solver (fast for this size)
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_val, grad, conv, max_c).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    shifted = conv - max_c
    exp_w = np.exp(beta * shifted)
    w = exp_w / np.sum(exp_w)                # softmax weights

    # gradient: 2 * (w reversed) * a  (cross‑correlation)
    grad = 2.0 * np.convolve(w[::-1], a)[2 * n - 2 - np.arange(n)]

    smooth_val = (np.log(np.sum(exp_w)) + beta * max_c) / beta
    return smooth_val, grad, conv, max_c


# ----------------------------------------------------------------------
def _random_initial_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random feasible point on the simplex.
    Three strategies are mixed: uniform, Dirichlet(0.5) and a sparse equal‑weight vector.
    """
    choice = rng.integers(0, 3)
    if choice == 0:                         # uniform
        a = np.full(n, 1.0 / n)
    elif choice == 1:                       # Dirichlet, α = 0.5
        a = rng.dirichlet([0.5] * n)

        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / k

    return a

                # otherwise blend – a convex combination keeps the bound
                t = 0.5
                a = (1.0 - t) * a + t * g

            # evaluate
            cur_score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(cur_score) and cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[LP improve] restart={total_restarts} iter={lp_iter} score={best_score:.6f}")

            # simple stopping criteria for the inner LP loop
            if prev_M is not None and M >= prev_M * (1.0 - 1e-5):
                # progress stalled
                break
            prev_M = M

            # occasional random shake to escape tiny plateaus
            if lp_iter % 30 == 0 and rng.random() < 0.07:
                pos = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.15))
                a[pos] *= factor
                a = _project_simplex(a)

        # ---- Local Adam‑style refinement --------------------------------
        # Use the already‑found best sequence as warm‑start for a short Adam run.
        a_refine = best_seq.copy()
        m = np.zeros_like(a_refine)
        v = np.zeros_like(a_refine)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        base_eta = 0.15
        smooth_beta = 0.7
        max_smooth_beta = 25.0
        smooth_inc = 0.00007
        refine_iter = 0

        while time.monotonic() < deadline and refine_iter < 8000:
            refine_iter += 1

            smooth_val, grad, conv, max_c = _smooth_max_and_grad(a_refine, smooth_beta)

            # Adam update on the simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** refine_iter)
            v_hat = v / (1.0 - beta2 ** refine_iter)

            eta = base_eta / np.sqrt(refine_iter)
            a_refine = a_refine - eta * m_hat / (np.sqrt(v_hat) + eps)
            a_refine = _project_simplex(a_refine)

            # occasional multiplicative perturbation
            if rng.random() < 0.05:
                pos = rng.integers(0, n)
                a_refine[pos] *= np.exp(rng.normal(0.0, 0.12))
                a_refine = _project_simplex(a_refine)

            # evaluate and keep the best seen globally
            cur_score = float(evaluate_sequence(a_refine.tolist()))
            if np.isfinite(cur_score) and cur_score < best_score:
                best_score, best_seq = cur_score, a_refine.copy()
                print(f"[Adam improve] restart={total_restarts} iter={refine_iter} score={best_score:.6f}")

            # slowly tighten the smooth‑max temperature
            if smooth_beta < max_smooth_beta:
                smooth_beta = min(max_smooth_beta, smooth_beta + smooth_inc)

            # early termination if we are already very close to the theoretical bound
            theoretical_lb = 2.0 * n / (2 * n - 1)
            if best_score <= theoretical_lb * 1.02:
                print("[early stop] near‑optimal bound reached")
                break

        # after each restart, check remaining time; if only a few seconds left, stop
        if time.monotonic() + 2.0 >= deadline:
            break

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {total_restarts} restarts")
    if best_seq is None:
        # should never happen – fallback to a uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    return [float(x) for x in best_seq]
