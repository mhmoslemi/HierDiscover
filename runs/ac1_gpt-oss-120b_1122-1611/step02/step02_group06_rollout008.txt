"""
A two‑phase optimiser for the step‑function evaluation problem.

Phase 1 – fast Adam optimisation on a smooth‐max surrogate (as in the
reference code) to obtain a decent starting point.
Phase 2 – a Frank‑Wolfe style refinement: solve a linear programme that,
for the current sequence `a`, maximises the ℓ1‑norm of a partner vector
`g` under the constraint `a * g ≤ M` where `M` is the current maximum of
`a*a`.  After normalising `g` we mix it with `a` (full replacement or a
convex combination) and keep the best candidate.  Repeating the LP step
rapidly flattens the self‑convolution, driving the evaluation score
towards the theoretical lower bound `2 n/(2 n‑1) ≈ 1.001` for `n=1024`.

The code respects the 1000 s time budget, prints progress, and returns
the best non‑negative sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]          # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(a*a) and its gradient.

    Returns (smooth_val, grad, conv, max_c) where
        smooth_val = (1/β)·log Σ exp(β·c_k)
        grad       = ∂/∂a smooth_max
    """
    conv = np.convolve(a, a)          # length 2n‑1
    max_c = np.max(conv)

    shifted = conv - max_c
    exp_w = np.exp(beta * shifted)
    w = exp_w / np.sum(exp_w)        # softmax weights

    # grad = 2 * (w reversed) * a   (cross‑correlation)
    conv2 = np.convolve(w[::-1], a)  # length 3n‑2
    idx = 2 * a.size - 2 - np.arange(a.size)
    grad = 2.0 * conv2[idx]

    smooth_val = (np.log(np.sum(exp_w)) + beta * max_c) / beta
    return smooth_val, grad, conv, max_c


# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float):
    """
    Solve the linear programme:
        max  Σ g_j
        s.t. a * g ≤ M   (elementwise convolution)
             g_j ≥ 0.
    Returns (g, sum(g)) if successful, otherwise (None, None).
    """
    n = a.shape[0]
    L = 2 * n - 1

    # Build the Toeplitz‐like matrix A where A @ g = a * g
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a   # shift a by j positions

    b = np.full(L, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_j  → minimise –Σ g_j
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return None, None
    g = res.x
    return g, g.sum()


# ----------------------------------------------------------------------

    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smooth‑max surrogate -----------------------
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_eta = 0.2
    beta = 0.5                # start smooth
    beta_max = 30.0
    beta_inc = 0.0001

    # a modest number of Adam steps (fast) – we stop early if time runs out
    for it in range(1, 20001):
        if time.monotonic() >= deadline:
            break

        _, grad, conv, max_c = _smooth_max_and_grad(a, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = base_eta / np.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # occasional logging
        if it % 2000 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a.copy()
                print(f"[adam] iter={it}, score={best_score:.6f}")

        # increase smoothness gradually
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based Frank‑Wolfe refinement ------------------------
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = conv.max()

        # solve the LP; if it cannot improve we stop
        g, sum_g = _lp_step(a, M)
        if g is None or sum_g <= 1.0 + 1e-8:
            # no feasible improvement
            break

        # normalise g to unit ℓ1‑norm
        g_norm = g / sum_g

        # try a few mixing ratios and keep the best
        best_local_score = np.inf
        best_local_a = None
        for t in (0.2, 0.5, 1.0):           # t=1.0 == full replacement
            cand = (1.0 - t) * a + t * g_norm
            cand = _project_simplex(cand)   # guard against numerical drift
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_local_score:
                best_local_score = score
                best_local_a = cand

        if best_local_score < best_score:
            best_score = best_local_score
            best = best_local_a.copy()
            print(f"[lp‑improve] new best score={best_score:.6f}")

        a = best_local_a

        # occasional random shake to help escape plateaus
        if rng.random() < 0.05:
            pos = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[pos] *= factor
            a = _project_simplex(a)

        # early stop if we are already very close to the theoretical bound
        theoretical_lb = 2.0 * n / (2 * n - 1)   # ≈ 1.0005 for n=1024
        if best_score <= theoretical_lb * 1.01:
            print("[early stop] near‑optimal bound reached")
            break

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}")
    return [float(x) for x in best]
