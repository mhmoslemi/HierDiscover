Pseudo:

```
def project_simplex(v):
    # v: any real vector
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid sub‑gradient + simplex projection optimizer.

Goal
----
Minimise  2·n·max(a*a) / (∑a)²  for a non‑negative step‑function a.
Since the score is scale‑invariant we keep the vector normalised
(∑a = 1) during the whole search.

Method
------
1.   Choose a working size n (uses the provided height_sequence_1 if
     available, otherwise a modest 2048‑point vector).
2.   Initialise a by a one‑parameter “Beta‑shape’’ family
     a_i ∝ (xs·(n‑xs+1))**e   with xs=i+½.
     A fine grid of exponents e (covering edge‑heavy, uniform and
     centre‑heavy shapes) is scanned – the best point on this grid
     becomes the starting vector.
3.   Refine the start with a projected sub‑gradient descent on the
     maximum self‑convolution:
        – compute the full self‑convolution via FFT,
        – collect the (few) indices attaining the maximal value,
        – build the sub‑gradient w = Σ_k partner_vector(a,k),
        – apply a momentum‑averaged additive update
              a ← a – η·w,
          and project back onto the probability simplex.
        – The step size η decays as η₀/√t.
        – occasional tiny random mass‑swaps help escape plateaus.
4.   Stagnation handling – if no improvement for 30 s the current
     vector is re‑initialised (arcsine shape + small jitter) and the
     momentum reset.
5.   The best vector seen at any time is stored and finally returned.
   All values are clipped to the limits enforced by ``evaluate_sequence``.

The algorithm runs within the allotted budget (≈1000 s) and prints
progress messages showing the current best score.
"""

import time
import numpy as np
from scipy.optimize import linprog  # imported for completeness, not used in the fast path


# --------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# --------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Next power‑of‑two length that can hold a full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real, non‑negative vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)[k] w.r.t. a.
    Returns a vector w such that w_i = a_{k-i} (0 if the index is out of range).
    """
    n = a.size
    w = np.zeros_like(a)
    if k < n:
        # i = 0 … k
        w[: k + 1] = a[k :: -1]          # a[k], a[k‑1], …, a[0]
    else:
        # i = k‑(n‑1) … n‑1
        start = k - (n - 1)
        length = (2 * n - 1) - k
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _init_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine shape: a_i ∝ 1/√((i+½)(n‑i‑½))."""
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


def _score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of the objective (sum(a) is assumed to be 1)."""
    conv = _conv_self(a, fft_len)
    return 2.0 * a.size * conv.max()

# Main entry point required by the competition harness
# --------------------------------------------------------------------

    """
    Return a list of non‑negative heights that (empirically) minimise the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds (leaving a small safety margin) and uses a
    hybrid grid‑search → sub‑gradient → simplex‑projection optimisation.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 2048          # modest default size
        base = None

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Phase 1 – grid search over a simple 1‑parameter family
    # ------------------------------------------------------------------
    best_a = _init_arcsine(n)
    best_score = _score(best_a, fft_len)

    # exponent sweep: a_i ∝ ( (i+½)*(n‑i‑½) )**e
    # e = -0.9 … 0.6  (covers edge‑heavy up to centre‑heavy)
    exponents = np.linspace(-0.9, 0.6, 71)
    xs = np.arange(n, dtype=np.float64) + 0.5
    prod = xs * (n - xs + 1.0)          # symmetric term inside the power

    for e in exponents:

        if s == 0.0:

        a /= s

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a = best_a.copy()
    momentum = np.zeros_like(a)
    beta = 0.9               # momentum coefficient

        # ----- momentum smoothing -------------------------------------------
        momentum = beta * momentum + (1.0 - beta) * w

        # ----- step‑size schedule (η = η₀/√t) ---------------------------------
        eta = eta0 / np.sqrt(iteration)

        # ----- additive update + projection onto the simplex -----------------
        a_new = a - eta * momentum
        a = _project_simplex(a_new)

        # ----- occasional stochastic perturbations (≈2 % of the iterations) --
        if rng.random() < 0.02:
            # small random swap of mass between two coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = rng.random() * 0.2 * a[i]   # move up to 20 % of a[i]
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        # ----- evaluate current score ----------------------------------------
        cur_score = 2.0 * n * max_val  # sum(a) == 1 by construction
        if cur_score < best_score:

            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - (deadline - budget_s):6.2f}s] new best score = {best_score:.6f}")

        if time.monotonic() - last_improve > stagnation_limit:

            momentum[:] = 0.0
            last_improve = time.monotonic()
            print(f"[{last_improve - (deadline - budget_s):6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to respect evaluate_sequence limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
