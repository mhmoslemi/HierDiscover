def compute_grad_l2_fft(a, conv, u, n_fft):

    d = conv - u

"""
Propose a low‑score height sequence for the self‑convolution objective.

Algorithm
---------
1.  Initialise a number of candidate probability vectors (uniform, arcsine,
    triangular, a few random Dirichlet draws, and the previous best if
    available).  Evaluate them quickly using FFT‑based convolution and keep
    the best few as seeds.

2.  For each seed run two short optimisation phases:
    * Phase‑A – minimise the L2 distance of the self‑convolution to the
      perfectly uniform distribution (the ideal is a flat autocorrelation).
      Gradient of the loss is computed analytically via FFTs and the
      updates are performed with Adam on the probability simplex.
    * Phase‑B – minimise a smooth approximation of the max‑operator
      (soft‑max) using the same Adam machinery.  The temperature μ is
      increased linearly from a small value to a large one, turning the
      surrogate gradually into the true L∞‑objective.

    Both phases keep track of the best sequence encountered and abort
    early if the target score (≈ 1.5025) is reached.

3.  After the gradient‑based passes a cheap simulated‑annealing style
    fine‑tuning moves small amounts of probability mass between random
    indices.  Moves that improve the true score are always accepted,
    otherwise they are accepted with probability exp(−Δ/​T) where the
    temperature T is slowly cooled.

All vectors are projected onto the Euclidean probability simplex after
each update, guaranteeing non‑negativity and unit‑sum.  Convolutions and
gradients are performed with FFTs (linear‑time in practice) for speed.
The routine stops as soon as the allocated wall‑clock budget expires and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, n_fft: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real input). Returns length 2·n‑1."""
    fft_a = np.fft.rfft(a, n=n_fft)
    prod = fft_a * fft_a
    conv_full = np.fft.irfft(prod, n=n_fft)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _grad_l2_fft(a: np.ndarray, conv: np.ndarray,
                 u: np.ndarray, n_fft: int) -> np.ndarray:
    """
    Gradient of  L2 loss  L = ∑ (conv - u)²  w.r.t. a.
    Formula: ∇L = 4 * ( (conv-u) * a_rev )  (restricted to n entries).
    """
    d = conv - u
    fft_d = np.fft.rfft(d, n=n_fft)
    rev_a = a[::-1]
    fft_rev = np.fft.rfft(rev_a, n=n_fft)
    corr_full = np.fft.irfft(fft_d * fft_rev, n=n_fft)
    start = a.size - 1
    return 4.0 * corr_full[start : start + a.size]


# ----------------------------------------------------------------------
def _grad_softmax_fft(a: np.ndarray, conv: np.ndarray,
                     mu: float, n_fft: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate:
      f_mu(a) = log‑sum‑exp(mu·conv) / mu
    Using the identity ∂f/∂conv = softmax(mu·conv),
    the chain rule gives  ∇a = 2·corr(softmax, a).
    """
    cmax = conv.max()
    w = np.exp(mu * (conv - cmax))
    w /= w.sum()
    fft_w = np.fft.rfft(w, n=n_fft)
    rev_a = a[::-1]
    fft_rev = np.fft.rfft(rev_a, n=n_fft)
    corr_full = np.fft.irfft(fft_w * fft_rev, n=n_fft)
    start = a.size - 1
    return 2.0 * corr_full[start : start + a.size]


# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective value for unit‑sum a using only the convolution."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence discovered before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budget (keep a small safety margin)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Problem size – 1024 gives a good trade‑off between expressivity
    # and per‑iteration cost.
    n = 1024
    # FFT length sufficient for linear convolutions of length up to 3·n‑2.
    n_fft = _next_pow2(3 * n - 2)

    # Uniform target for the convolution (used in the L2 phase)
    u_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper to compute the exact objective quickly
    def _exact_score(a: np.ndarray) -> float:
        conv = _conv_fft(a, n_fft)
        return _score_from_conv(conv, n)

    # ------------------------------------------------------------------
    # Build a diverse pool of starting points.
    patterns = []

    patterns.append(np.full(n, 1.0 / n, dtype=np.float64))

    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)

                    prev = tmp
                else:

            patterns.append(prev)

    scores = [(p, _exact_score(p)) for p in patterns]

    # ---------------------------------------------------------------
    def _run_l2_phase(a0: np.ndarray, max_iters: int) -> np.ndarray:
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)

                break

            grad = _grad_l2_fft(a, conv, u_target, n_fft)

            # Adam step
            lr = 0.12 / np.sqrt(t)          # a modest decaying step‑size
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Projection onto the simplex
            a = _project_simplex(a)

            # Periodic exact evaluation
            if t % 250 == 0:
                sc = _exact_score(a)
                nonlocal best_seq, best_score
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[L2] it={t}, score={best_score:.6f}")
                if best_score <= 1.5025:
                    return a
        return a

    # Phase‑B : smooth‑max (soft‑max) minimisation.
    # ---------------------------------------------------------------
    def _run_softmax_phase(a0: np.ndarray, max_iters: int) -> np.ndarray:
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        mu0, mu_final = 0.5, 120.0     # temperature schedule
        for t in range(1, max_iters + 1):
            if time.monotonic() >= deadline:
                break
            conv = _conv_fft(a, n_fft)

            mu = mu0 + (mu_final - mu0) * (t / max_iters)  # linear increase
            grad = _grad_softmax_fft(a, conv, mu, n_fft)

            # Adam step
            lr = 0.07 / np.sqrt(t)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Projection onto the simplex
            a = _project_simplex(a)

            # Occasional perturbation to inject diversity
            if rng.random() < 0.004:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = min(a[i], 0.03) * rng.random()
                    a[i] -= delta
                    a[j] += delta
                    a = _project_simplex(a)

            # Periodic exact evaluation
            if t % 200 == 0:
                sc = _exact_score(a)
                nonlocal best_seq, best_score
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[SM] it={t}, mu={mu:.1f}, score={best_score:.6f}")
                if best_score <= 1.5025:
                    return a
        return a

    # ------------------------------------------------------------------
    # Run the two‑phase optimiser from each of the top seeds.
    # ---------------------------------------------------------------
    for seed_vec in top_seeds:
        if time.monotonic() >= deadline:
            break
        a_l2 = _run_l2_phase(seed_vec, max_iters=4000)
        if time.monotonic() >= deadline:
            break
        _ = _run_softmax_phase(a_l2, max_iters=8000)

    # ------------------------------------------------------------------
    # Phase‑C : cheap simulated‑annealing style fine‑tuning.
    # ---------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    T0 = 0.04  # initial temperature
    while time.monotonic() < deadline:

        elapsed = time.monotonic() - start_time
        T = T0 * max(0.0, 1.0 - elapsed / (budget_s - 0.5))

        # propose a small transfer of mass between two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        a_new[j] += delta

            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[SA] score={best_score:.6f}, T={T:.5f}")

        if best_score <= 1.5025:
            break

    # ------------------------------------------------------------------
    # Final sanity check – ensure the returned list complies with the
    # evaluator (non‑negative, normalised).  The projection guarantees it.
    final_seq = [float(x) for x in best_seq.tolist()]
    final_score = float(evaluate_sequence(final_seq))
    print(f"[final] best_score={final_score:.6f} (reported by evaluator)")
    return final_seq
