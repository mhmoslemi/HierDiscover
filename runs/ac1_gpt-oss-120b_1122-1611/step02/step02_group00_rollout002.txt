def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    max_time = budget_s

    # Get initial sequence
    if 'height_sequence_1' in globals() and isinstance(globals()['height_sequence_1'], (list, np.ndarray)):
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.size
        if n < 5000:
            n = 5000
            init_seq = np.pad(init_seq, (0, n - init_seq.size))
        else:
            n = init_seq.size
    else:
        n = 5000
        init_seq = None

    # Initialize w
    if init_seq is not None:
        w = init_seq.copy()
        w = np.maximum(w, 0.0)
        # Normalize sum to 1
        total = np.sum(w)
        if total == 0:
            w = np.full(n, 1.0 / n)
        else:
            w = w / total
    else:
        # Use Catalan sequence
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # Project to simplex (ensures sum=1)
    w = simplex_projection(w)

    best_w = w.copy()
    best_score = compute_score(best_w)
    print(f"Init n={n}, score={best_score:.6f}")

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.3   # base learning rate
    m = np.zeros(n)
    v = np.zeros(n)

    # iteration and counters
    it = 0
    no_improve = 0
    # As we need to cap time
    while time.monotonic() - start_time < max_time - 5:
        # compute convolution
        b = convolve_fft(w, w)
        max_val = np.max(b)
        max_idxs = np.flatnonzero(b >= max_val - 1e-15)
        i_max = int(rng.choice(max_idxs))

        # subgradient
        grad = np.zeros(n)
        # Determine indices j where w contributes
        # For each j: w_j contributes to b[i_max] with weight 2 * w[i_max - j]
        # valid j range
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        idx = i_max - j
        grad[j] = 2.0 * w[idx]

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / np.sqrt(it)  # optionally decays slowly

        w = simplex_projection(w_new)

            best_w = w.copy()

"""
Propose a step‑function sequence with a low self‑convolution peak.

Algorithm
---------
1. Initialise the probability vector `w` (sum = 1) from the best known
   construction (`height_sequence_1` if present, otherwise the Catalan
   sequence).  The vector is forced to be symmetric to reduce the search
   space.

2. Optimise `w` with an Adam‑style projected gradient descent on the
   objective
        f(w) = max_k (w*w)[k] .
   The sub‑gradient of the active maximal index(es) is
        ∂f/∂w_j = 2 * Σ_{i∈I_max} w_{i-j},
   where `I_max` is the (small) set of convolution indices attaining the
   maximum.  After each update the vector is projected onto the probability
   simplex (non‑negative, sum = 1) and re‑symmetrised.

3. Every few thousand iterations a cheap “block perturbation’’ (random
   scaling of a contiguous block) is tried; if it improves the score we
   adopt it and reset the Adam moments.

4. If no improvement is seen for many iterations the current best vector
   is perturbed and the optimiser is restarted – this helps escape local
   minima.

5. After the time budget is exhausted a short local‑search phase swaps small
   amounts of mass between symmetric positions to polish the result.

6. Finally the vector is scaled to the conventional ℓ¹‑norm
   `sum = sqrt(2·n)` (the evaluation is scale‑invariant) and returned as a
   plain Python list.

The whole procedure respects the 1000 s time limit, uses only NumPy and
standard library, and keeps all helper functions at top level as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negatives to zero (purely numerical)
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation: multiply a random contiguous block
    by a log‑normal factor, optionally symmetrise.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def compute_score(w: np.ndarray) -> float:
    """
    Score for a probability vector w (sum = 1):
        score = 2·n·max_k (w*w)[k] .
    """
    n = w.size
    b = convolve_fft(w, w)
    return 2.0 * n * np.max(b)


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The optimisation runs for up to ``budget_s``
    seconds, using an Adam projected gradient descent on the max‑convolution
    objective with occasional random jumps and a final local polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    max_time = float(budget_s)

    # ------------------------------------------------------------------
    # Determine problem size and initialise w (probability vector)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = np.asarray(eval('height_sequence_1'), dtype=np.float64)
            n = init.size
            if n < 2:
                raise ValueError
        except Exception:
            init = None
            n = 5000
    else:
        init = None
        n = 5000

    # If we have a previous construction, use it; otherwise use Catalan.
    if init is not None:
        w = np.maximum(init, 0.0)
        # Normalise to sum = 1, then project onto simplex for safety.
        w = w / np.maximum(np.sum(w), 1e-12)
        w = simplex_projection(w)
    else:
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # Enforce symmetry from the start (the optimal solution is believed to be symmetric)
    w = symmetrize(w)
    w = simplex_projection(w)

    best_w = w.copy()
    best_score = compute_score(best_w)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser parameters
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.3                      # initial stepsize (will decay as 1/sqrt(t))
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    iter_cnt = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < max_time - 5.0:
        # 1) compute convolution and locate active maximal indices
        b = convolve_fft(w, w)
        max_val = np.max(b)
        # Gather all indices that are numerically maximal (helps when a plateau exists)
        active = np.flatnonzero(b >= max_val - 1e-15)

        # 2) sub‑gradient: average over all active indices
        grad = np.zeros_like(w)
        for i_max in active:
            j_start = max(0, i_max - (n - 1))
            j_end = min(n - 1, i_max)
            j = np.arange(j_start, j_end + 1, dtype=int)
            idx = i_max - j
            grad[j] += w[idx]
        grad = 2.0 * grad / max(1, len(active))

        # 3) Adam update (with a mild decay of the base stepsize)
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        lr = lr0 / np.sqrt(iter_cnt)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        # Projection onto the simplex guarantees feasibility.
        w = simplex_projection(w_new)

        # Occasionally enforce exact symmetry (helps reduce dimensionality).
        if rng.random() < 0.30:
            w = symmetrize(w)
            w = simplex_projection(w)

        # 4) evaluate current candidate
        cur_score = compute_score(w)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iter_cnt:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # 5) Random block perturbation every few thousand iterations
        if iter_cnt % 2000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand = symmetrize(cand)
            cand = simplex_projection(cand)
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[block] iter={iter_cnt:6d}, new best={best_score:.6f}")

        # 6) Stagnation reset
        if no_improve > 50000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            w = symmetrize(w)
            w = simplex_projection(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iter_cnt:6d} – re‑initialising from perturbed best")

    # ------------------------------------------------------------------
    # Local polishing phase (simple swap of small mass between symmetric positions)
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < max_time - 1.0:
        i, j = rng.choice(n, size=2, replace=False)
        # Ensure we move mass symmetrically: i ↔ j and their mirrors.
        i_m = n - 1 - i
        j_m = n - 1 - j
        # Determine a small amount to transfer (up to 5 % of the source mass)
        source_mass = best_w[i] + best_w[i_m]
        if source_mass <= 0.0:
            continue
        delta = rng.random() * 0.05 * source_mass
        # Build a candidate vector
        cand = best_w.copy()
        cand[i] -= delta / 2.0
        cand[i_m] -= delta / 2.0
        cand[j] += delta / 2.0
        cand[j_m] += delta / 2.0
        # Guard against negativity caused by rounding
        if np.any(cand < -1e-12):
            continue
        cand = np.maximum(cand, 0.0)
        cand = symmetrize(cand)
        cand = simplex_projection(cand)

        cand_score = compute_score(cand)
        if cand_score < best_score - 1e-9:
            best_w = cand.copy()
            best_score = cand_score
            print(f"[local] iter={iter_cnt:6d}, improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the conventional sqrt(2·n) ℓ¹‑norm
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)
    # Clip to the range allowed by evaluate_sequence (0 … 1000)
    np.clip(final_seq, 0.0, 1000.0, out=final_seq)

    # Evaluate once more for reporting
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best≈{best_score:.6f}, evaluate={final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
