def _max_conv_and_subgrad(a):
    conv = np.convolve(a,a)
    max_c = conv.max()
    idxs = np.where(conv >= max_c - 1e-12)[0]  # tolerance
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in idxs:
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        i = np.arange(i_min, i_max+1)
        j = k - i
        grad[i] += 2 * a[j]
    grad /= len(idxs)
    return max_c, grad

"""
Propose a step‑function height sequence by (i) a projected‐Adam optimiser
    that uses the *exact* max‑convolution as the objective (sub‑gradient),
    (ii) occasional random “shakes’’ to escape flat regions, and
    (iii) a final short refinement phase.
The optimisation works directly on the probability simplex (∑a_i = 1,
a_i ≥ 0); the evaluation score is 2·n·max(conv(a,a)) because the
normalisation cancels out.
The algorithm explores several random restarts (including the previous
solution if available) and keeps the best sequence found before the
time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _random_initial_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random feasible point on the simplex.
    Mix three strategies: uniform, Dirichlet(0.5) and a sparse uniform vector.
    """
    choice = rng.integers(0, 3)
    if choice == 0:                     # uniform
        a = np.full(n, 1.0 / n)
    elif choice == 1:                   # Dirichlet α=0.5 (moderately sparse)
        a = rng.dirichlet([0.5] * n)
    else:                               # sparse uniform support
        k = rng.integers(max(10, n // 30), max(20, n // 5))
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0 / k
    a = _project_simplex(a)
    return a

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << ((L - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:L]
    # numerical noise may produce tiny negative values – clip
    np.clip(conv, 0.0, None, out=conv)
    return conv


# ----------------------------------------------------------------------
def _max_conv_and_subgrad(a: np.ndarray):
    """
    Compute max(conv(a,a)) and a sub‑gradient of that max w.r.t. a.
    The gradient corresponds to the (average) gradient of all maximal
    convolution entries.  Returns (max_c, grad).
    """
    conv = _conv_fft(a)
    max_c = conv.max()
    # collect all indices that are (numerically) maximal
    tol = 1e-12
    idxs = np.where(conv >= max_c - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in idxs:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i = np.arange(i_min, i_max + 1)
        j = k - i
        grad[i] += 2.0 * a[j]
    grad /= max(1, len(idxs))          # average if several ties
    return max_c, grad

                print(f"[improve] restart={restart_counter} it={iter_in} "
                      f"score={best_score:.6f}")

            # Adam update (projected onto simplex afterwards)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** iter_in)
            v_hat = v / (1.0 - beta2 ** iter_in)

            eta = base_eta / np.sqrt(iter_in)
            a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

            # enforce feasibility (simplex)
            a = _project_simplex(a)

            # ---------------------------------------------------------
            # occasional “shake’’ – a random multiplicative perturbation
            # helps to leave flat plateaux (e.g. the uniform fixed point)
            if rng.random() < 0.04:   # 4 % chance per iteration
                pos = rng.integers(0, n)
                factor = float(np.exp(rng.normal(0.0, 0.18)))
                a[pos] *= factor
                a = _project_simplex(a)

            # ---------------------------------------------------------
            # early exit from this restart if no improvement for a while
            if iter_in % 4000 == 0:
                if iter_in - last_improve_iter > 20000:
                    break

            # ---------------------------------------------------------
            # very occasional hard restart inside the same run
            if iter_in % 25000 == 0:
                a = _random_initial_a(n, rng)
                m.fill(0.0)
                v.fill(0.0)
                last_improve_iter = iter_in

    # ------------------------------------------------------------------
    # final short refinement on the best candidate found so far
    if best_a is not None:
        a = best_a.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-12
        base_eta = 0.1
        for it in range(1, 20001):
            max_c, grad = _max_conv_and_subgrad(a)
            score = 2.0 * n * max_c
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[refine] it={it} score={best_score:.6f}")

            # Adam step with smaller learning rate
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            eta = base_eta / np.sqrt(it)
            a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a)

            # tiny random shake at the very end to possibly escape the last
            # local trap (probability decays with iteration)
            if rng.random() < 0.01 * np.exp(-it / 5000):
                pos = rng.integers(0, n)
                a[pos] *= float(np.exp(rng.normal(0.0, 0.12)))
                a = _project_simplex(a)

    print(f"[final] best_score={best_score:.6f} (n={n})")
    return [float(x) for x in best_a]
