"""
propose_candidate
-----------------

A fast projected‑gradient search on the probability simplex with
momentum and enforced symmetry.  The objective is

    F(w) = 2·n·max_k (w*w)[k]    (∑w = 1, w ≥ 0)

which is exactly the quantity returned by ``evaluate_sequence`` after the
final scaling step.  The algorithm works as follows

1. Initialise a probability vector `w`.  If a previously found sequence
   (`height_sequence_1`) is available we reuse it, otherwise we start
   from the Catalan distribution.

2. Enforce symmetry   w = (w + w[::-1]) / 2   – the optimum can be taken
   symmetric without loss of optimality.

3. Repeatedly
   * compute the linear convolution b = w * w   (FFT, O(n log n));
   * locate the active set of indices where b attains its maximum;
   * build the sub‑gradient of the max‑convolution (average over the
     active set);
   * update the vector with a momentum term and a diminishing step size;
   * project back onto the probability simplex;
   * symmetrise the result again.

   Every few hundred iterations a cheap random “block perturbation’’ is
   applied to the current best vector; if it improves the score the
   best vector is updated.

4. After the time budget expires the best vector is rescales to the
   conventional √(2 n) L¹‑norm and returned as a plain Python list.

The combination of momentum, symmetric restriction and occasional
global perturbations typically pushes the bound below the Catalan
baseline (~1.571) and often reaches the target 1.5025.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be defined at top level.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` terms of a_i = C(2i,i)/4**i (Catalan numbers / 4**i)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution using real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negatives caused by rounding.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    # Duchi et al. (2008) algorithm.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries equal (or numerically zero) – return uniform.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation (used for global jumps)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetric averaging (kept from the reference implementation)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def symmetric_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Block perturbation followed by explicit symmetrisation."""
    cand = block_perturbation(base, rng)
    cand = (cand + cand[::-1]) * 0.5
    return cand


def subgradient_of_active_max(w: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Return the (averaged) sub‑gradient of max_k (w*w)[k] for the active set."""
    n = w.size
    grad = np.zeros_like(w)
    for i in active_idx:
        # Determine the range of j such that 0 ≤ j < n and 0 ≤ i-j < n
        j_start = max(0, i - (n - 1))
        j_end = min(i, n - 1)
        # Indices j_start … j_end correspond to w[j]
        j_range = np.arange(j_start, j_end + 1, dtype=int)
        partner = w[i - j_range]          # w[i-j]
        grad[j_range] += 2.0 * partner
    grad /= float(active_idx.size)
    return grad

# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) whose
    `evaluate_sequence` score is as low as possible.

    The routine performs a projected gradient descent with momentum on the
    probability simplex, enforces symmetry, and periodically applies a
    random block perturbation to escape local minima.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose the length n.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 1:
                raise ValueError
        except Exception:
            n = 8000
    else:
        n = 8000                    # a reasonably large size

    # ------------------------------------------------------------------
    # Initialise w (probability vector, ∑w = 1).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            # The provided sequence may have been scaled to √(2n); re‑normalise.
            w = init / np.sum(init)
            # If the length does not match n, resample linearly.
            if w.size != n:
                x_old = np.linspace(0, 1, w.size)
                x_new = np.linspace(0, 1, n)
                w = np.interp(x_new, x_old, w)
        except Exception:
            w = catalan_sequence(n)
            w = w / np.sum(w)
    else:
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # Enforce symmetry right away.
    w = (w + w[::-1]) * 0.5

    # ------------------------------------------------------------------
    # State for the optimiser.
    # ------------------------------------------------------------------
    momentum = np.zeros_like(w)
    beta = 0.9                     # momentum coefficient
    eta0 = 1.0                     # initial step size
    best_w = w.copy()
    # Initial score (no scaling yet – corresponds to evaluate_sequence after final scaling)
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0
    block_interval = 600           # how often to try a global perturbation
    reset_limit = 25000            # iterations without improvement before a reset

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # 1) convolution and active set
        b = convolve_fft(w, w)
        max_val = np.max(b)
        # use a small tolerance to collect all (near) maximisers
        active = np.flatnonzero(b >= max_val - 1e-14)

        # 2) sub‑gradient of the active max
        grad = subgradient_of_active_max(w, active)

        # 3) momentum update
        momentum = beta * momentum + grad

        # 4) step size schedule (classical 1/√t decay)
        eta = eta0 / np.sqrt(iteration + 1.0)

        # 5) projected gradient step
        w_new = w - eta * momentum
        w_new = simplex_projection(w_new)

        # 6) enforce symmetry (the optimum can be taken symmetric)
        w_new = (w_new + w_new[::-1]) * 0.5

        w = w_new

        # 7) evaluate current candidate
        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # Occasional global perturbation to escape plateaux.
        # ------------------------------------------------------------------
        if iteration % block_interval == 0 and iteration > 0:
            cand = symmetric_block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)           # back to probability vector
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                momentum[:] = 0.0
                print(f"[block] iter={iteration:6d}, new best score={best_score:.6f}")

        # ------------------------------------------------------------------
        # If we have been stuck for a long time, reset from a perturbed best.
        # ------------------------------------------------------------------
        if no_improve > reset_limit:
            w = symmetric_block_perturbation(best_w, rng)
            w = w / np.sum(w)
            momentum[:] = 0.0
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant but we output the
    # conventional √(2 n) L¹‑norm.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # will call it again on the returned list).

    except Exception:
        final_score = float('nan')
    print(f"[final] length={n}, best score (raw)={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
