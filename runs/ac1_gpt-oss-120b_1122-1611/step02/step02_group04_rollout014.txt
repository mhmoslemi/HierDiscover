def run_sparse_opt(n, rng, end_time):
    # Choose random K between min(20, n) and maxK
    minK = max(5, int(np.sqrt(n))) # maybe 32
    maxK = max(minK+1, int(2*np.sqrt(n))) # approx 64
    K = rng.integers(minK, maxK+1)
    positions = rng.choice(n, size=K, replace=False)
    positions.sort()
    # initialize weights Dirichlet
    weights = rng.exponential(1.0, size=K)
    weights /= weights.sum()
    # local best scores
    best_weights = weights.copy()
    best_positions = positions.copy()
    best_local_score = 2 * n * compute_max_conv(weights, positions, n) # compute via fast function
    # gradient descent on weights only
    eta0 = 0.5
    for t in range(5000):
        # compute conv and max idx
        conv, max_idx, max_val = compute_full_conv(weights, positions, n)
        # gradient w.r.t weights
        mask = (np.add.outer(positions, positions) == max_idx).astype(np.float64)
        grad = 2.0 * mask @ weights
        # step-size schedule
        step = eta0 / (np.sqrt(t+1) + 1.0)
        weights = weights * np.exp(-step * grad)
        # renorm
        sumw = weights.sum()
        if sumw == 0:
            weights = np.full_like(weights, 1.0 / K)
        else:
            weights /= sumw
        # evaluate new score
        score = 2 * n * max_val
        if score < best_local_score:
            best_local_score = score
            best_weights = weights.copy()
            best_positions = positions.copy()
        # occasional position mutation
        if rng.random() < 0.02:
            # mutate one random position
            i = rng.integers(K)
            new_pos = rng.integers(0, n)
            if new_pos in positions:
                continue
            old_pos = positions[i]
            positions[i] = new_pos
            positions.sort()
            # recompute conv after position change, but keep weights unchanged
            conv2, max_idx2, max_val2 = compute_full_conv(weights, positions, n)
            score2 = 2 * n * max_val2
            # accept if better or with small prob
            if score2 < score or rng.random() < 0.1:
                # keep changed pos
                # maybe update best if improved
                if score2 < best_local_score:
                    best_local_score = score2
                    best_weights = weights.copy()

    # after iterations, construct full vector

def run_adam_opt(start_seq, rng, end_time):
    a = np.copy(start_seq)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0/a.size)
    else:
        a = a / a.sum()
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = 0.05
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iteration = 0
    tau_factor = 0.5
    tau_decay = 0.9995
    while time.monotonic() < end_time:
        iteration += 1
        conv = np.convolve(a, a)
        current_max = conv.max()
        tau = max(current_max * tau_factor, 1e-12)
        shifted = (conv - current_max) / tau
        w = np.exp(shifted - np.max(shifted))  # stable
        w_sum = w.sum()
        w = w / w_sum
        grad = 2.0 * np.correlate(w, a, mode='full')[len(a)-1:len(a)-1+len(a)]
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad*grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = lr_base / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        # projection onto simplex
        a_candidate = _project_simplex(a_candidate)
        if rng.random() < 0.005:
            noise = rng.normal(0.0, 0.01, size=a_candidate.shape)
            a_candidate = a_candidate + noise
            a_candidate = _project_simplex(np.maximum(a_candidate, 0.0))
        # evaluate
        score = float(evaluate_sequence(a_candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_candidate.copy()
            print(f"[Adam-improve] iter={iteration} score={best_score:.6f}")
        a = a_candidate
        tau_factor *= tau_decay
    print(f"[Adam-final] best {best_score:.6f} after {iteration} iters")
    return best, best_score

"""Search for a step-function height sequence minimizing the evaluation score.

The search combines two complementary strategies:
1. A fast sparse optimisation that searches over a small set of support
   positions using an exponentiated‑gradient (EG) update.  Any improvement
   found is lifted to a full‑length vector and evaluated with the official
   ``evaluate_sequence``.
2. A projected‑gradient Adam optimiser that works on the full vector.
   Several independent Adam runs are performed with random starts; the
   best result over all runs is returned.

Both methods respect the non‑negativity and simplex constraints used by
``evaluate_sequence`` and operate under the given time budget.
"""
import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(weights: np.ndarray, pos: np.ndarray, n: int):
    """Return full convolution array, arg‑max index and max value."""
    # shape (K, K) of sums and products
    sums = np.add.outer(pos, pos)                 # (K, K) int
    prods = np.outer(weights, weights)            # (K, K) float
    conv = np.zeros(2 * n - 1, dtype=np.float64)
    np.add.at(conv, sums.ravel(), prods.ravel())
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    return conv, max_idx, max_val


def _sparse_optimize(n: int, rng: np.random.Generator, deadline: float):
    """Fast optimisation on a sparse support using exponentiated‑gradient."""
    # choose a support size between sqrt(n) and 2*sqrt(n)
    min_k = max(10, int(np.sqrt(n)))
    max_k = max(min_k + 1, int(2.0 * np.sqrt(n)))
    best_score = float('inf')
    best_vec = None

    while time.monotonic() < deadline:
        K = rng.integers(min_k, max_k + 1)
        pos = rng.choice(n, size=K, replace=False)
        pos.sort()
        # Dirichlet (exponential) initialization of weights
        w = rng.exponential(1.0, size=K)
        w /= w.sum()

        # keep a copy of best local solution
        best_local_score = 2.0 * n * _full_convolution(w, pos, n)[2]
        best_local_w = w.copy()
        best_local_pos = pos.copy()

        # main inner loop (a few thousand EG steps)
        eta0 = 0.8
        for it in range(2000):
            conv, max_idx, max_val = _full_convolution(w, pos, n)
            # mask of pairs that contribute to the current maximum
            mask = (np.add.outer(pos, pos) == max_idx).astype(np.float64)
            grad = 2.0 * mask @ w                     # shape (K,)

            step = eta0 / (np.sqrt(it + 1) + 1.0)
            w = w * np.exp(-step * grad)
            s = w.sum()
            if s == 0.0:
                w = np.full_like(w, 1.0 / K)
            else:
                w /= s

            cur_score = 2.0 * n * max_val
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_w = w.copy()
                best_local_pos = pos.copy()

            # occasional position mutation
            if rng.random() < 0.02:
                i = rng.integers(K)
                new_p = rng.integers(0, n)
                if new_p in pos:
                    continue
                old = pos[i]
                pos[i] = new_p
                pos.sort()
                conv2, max_idx2, max_val2 = _full_convolution(w, pos, n)
                new_score = 2.0 * n * max_val2
                # accept if better or with a tiny probability to escape plateaus
                accept = new_score < cur_score or rng.random() < 0.05
                if accept:
                    if new_score < best_local_score:
                        best_local_score = new_score
                        best_local_w = w.copy()
                        best_local_pos = pos.copy()
                else:
                    pos[i] = old
                    pos.sort()

            if time.monotonic() >= deadline:
                break

        # construct dense vector and evaluate with the official wrapper
        vec = np.zeros(n, dtype=np.float64)
        vec[best_local_pos] = best_local_w
        score = float(evaluate_sequence(vec.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_vec = vec.copy()
            print(f"[sparse] new best {best_score:.6f} (K={K})")

    return best_vec, best_score


def _adam_optimize(start_vec: np.ndarray, rng: np.random.Generator, deadline: float,
                   init_name: str = "adam"):
    """Projected‑Adam on the full vector."""
    a = np.copy(start_vec)
    a = np.maximum(a, 0.0)
    sum_a = a.sum()
    if sum_a == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = _project_simplex(a)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[{init_name}] start score {best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = 0.07
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature for the soft‑max surrogate
    tau_factor = 1.0
    tau_decay = 0.9995

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        conv = np.convolve(a, a)
        cur_max = conv.max()
        tau = max(cur_max * tau_factor, 1e-12)

        # stable soft‑max weights
        shift = (conv - cur_max) / tau
        # subtract max for numerical stability
        shift = shift - shift.max()
        w = np.exp(shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient of the smooth‑max term
        grad = 2.0 * np.correlate(w, a, mode='full')[len(a)-1:len(a)-1+len(a)]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr_base / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)

        # occasional shake‑up to jump out of flat plateaux
        if rng.random() < 0.004:
            noise = rng.normal(0.0, 0.010, size=a_candidate.shape)
            a_candidate = a_candidate + noise
            a_candidate = _project_simplex(np.maximum(a_candidate, 0.0))

        score = float(evaluate_sequence(a_candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_candidate.copy()
            print(f"[{init_name}] iter={iteration:,} improved {best_score:.6f}")

        a = a_candidate
        tau_factor *= tau_decay
        # optional slight acceleration of decay
        if iteration % 4000 == 0:
            tau_factor *= 0.98
            tau_factor = max(tau_factor, 5e-7)

        # early exit if we are already close to the theoretical lower bound
        lower = 2.0 * len(a) / (2 * len(a) - 1)
        if best_score <= lower * 1.02:
            break

    print(f"[{init_name}] final best {best_score:.6f} after {iteration:,} iters")
    return best, best_score

"""Search for a step‑function height sequence that minimises the
`evaluate_sequence` score.

The search proceeds in two stages:

1. **Sparse EG optimisation** – a small support (≈ 2·√n positions) is
   optimised with an exponentiated‑gradient (EG) update that directly
   reduces the worst self‑convolution entry.  The best sparse solution
   is lifted to a full‑length vector and evaluated with the official
   wrapper.

2. **Full‑vector Adam optimisation** – a few independent projected‑Adam
   runs start either from the best sparse candidate or from random
   Dirichlet points.  The Adam optimiser works on the smooth log‑sum‑exp
   surrogate of the max‑convolution and projects onto the probability
   simplex after each step.  The best result over all runs is returned.

Both stages respect the non‑negativity and simplex constraints required
by `evaluate_sequence`.  The algorithm stops when the supplied time
budget (default 1000 s) is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution(weights: np.ndarray, pos: np.ndarray, n: int):
    """
    Compute the full self‑convolution of a sparse vector.
    Returns (conv, max_index, max_value).
    """
    sums = np.add.outer(pos, pos)                 # shape (K, K) – integer sums
    prods = np.outer(weights, weights)            # shape (K, K) – outer product
    conv = np.zeros(2 * n - 1, dtype=np.float64)
    np.add.at(conv, sums.ravel(), prods.ravel())
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    return conv, max_idx, max_val

def _sparse_optimize(n: int, rng: np.random.Generator, deadline: float):
    """
    Fast optimisation on a sparse support using exponentiated‑gradient.
    Returns a dense candidate vector and its official score.
    """
    min_k = max(10, int(np.sqrt(n)))
    max_k = max(min_k + 1, int(2.0 * np.sqrt(n)))
    best_score = float('inf')
    best_vec = None

    while time.monotonic() < deadline:
        K = rng.integers(min_k, max_k + 1)
        pos = rng.choice(n, size=K, replace=False)
        pos.sort()
        w = rng.exponential(1.0, size=K)
        w /= w.sum()

        best_local_score = 2.0 * n * _full_convolution(w, pos, n)[2]
        best_local_w = w.copy()
        best_local_pos = pos.copy()

        eta0 = 0.8
        for it in range(2000):
            # --- current convolution and gradient of the hard max ---
            conv, max_idx, max_val = _full_convolution(w, pos, n)
            mask = (np.add.outer(pos, pos) == max_idx).astype(np.float64)
            grad = 2.0 * mask @ w

            # --- EG update of the weights ---
            step = eta0 / (np.sqrt(it + 1) + 1.0)
            w = w * np.exp(-step * grad)
            s = w.sum()
            w = w / s if s > 0 else np.full_like(w, 1.0 / K)

            # --- evaluate new max after the update ---
            conv2, max_idx2, max_val2 = _full_convolution(w, pos, n)
            cur_score = 2.0 * n * max_val2
            if cur_score < best_local_score:

                best_local_pos = pos.copy()

                if new_p in pos:
                    continue
                old_p = pos[i]
                pos[i] = new_p
                pos.sort()
                conv3, max_idx3, max_val3 = _full_convolution(w, pos, n)

                accept = new_score < cur_score or rng.random() < 0.05
                if accept:
                    if new_score < best_local_score:
                        best_local_score = new_score
                        best_local_w = w.copy()
                        best_local_pos = pos.copy()
                else:
                    pos[i] = old_p
                    pos.sort()

            if time.monotonic() >= deadline:

        vec = np.zeros(n, dtype=np.float64)

        n_runs = max(1, int(remaining // 40.0))   # roughly 40 s per run
        per_run = remaining / n_runs
        for run_id in range(n_runs):
            run_deadline = time.monotonic() + per_run
            if run_id == 0:
                start = best_seq.copy()
                name = f"adam_{run_id}"
            else:
                start = rng.random(n)
                start = _project_simplex(start)
                name = f"adam_{run_id}"
            vec, score = _adam_optimize(start, rng, run_deadline, init_name=name)
            if np.isfinite(score) and score < best_score:
                best_seq = vec.copy()
                best_score = score
                print(f"[global] new global best {best_score:.6f}")

    # --------------------------------------------------------------
    # Final short Adam polishing if any time is left
    if time.monotonic() < deadline - 2.0:
        vec, score = _adam_optimize(best_seq, rng, deadline, init_name="adam_final")
        if np.isfinite(score) and score < best_score:
            best_seq = vec.copy()
            best_score = score

    print(f"[final] best score={best_score:.6f}")
    return [float(x) for x in best_seq]
