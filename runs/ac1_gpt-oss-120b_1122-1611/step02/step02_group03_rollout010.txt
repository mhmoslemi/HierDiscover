def _smooth_weights(conv, mu):
    cmax = conv.max()
    exp_args = mu * (conv - cmax)
    w = np.exp(exp_args)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / conv.shape[0])
    else:
        w = w / w_sum
    return w

def _lp_refine(a, M):
    n = a.shape[0]
    # Build convolution matrix
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
    except Exception:
        return None
    if b.value is None:
        return None
    return np.asarray(b.value, dtype=np.float64)

def propose_candidate(seed=42, budget_s=1100):
    """
    Hybrid optimization for the self‑convolution objective.

    * A rich collection of parametric initial patterns (uniform, power‑law,
      exponential, arcsine, triangular, random Dirichlet) is generated.
    * The best of them seeds a projected Adam optimiser that minimises a
      smooth approximation of ``max(a*a)``.
    * Every few thousand Adam steps a small linear‑programming refinement
      (maximise the ℓ₁‑mass of a partner vector under the current convolution
      bound) is performed; the result is mixed into the current iterate.
    * The procedure is run repeatedly with random restarts until the
      wall‑clock budget expires, keeping the best sequence seen.
    """

    patterns.append(tent)

    arcsine = arcsine / arcsine.sum()
    patterns.append(arcsine)

        shape = shape / shape.sum()

        exp_shape = exp_shape / exp_shape.sum()
        patterns.append(exp_shape)

    # random Dirichlet
    patterns.append(rng.dirichlet(np.ones(n)).astype(np.float64))

    # previous global best if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                # Resize by simple interpolation / truncation.
                if prev.size < n:
                    pad = np.full(n, 0.0, dtype=np.float64)
                    pad[:prev.size] = prev
                    prev = pad
                else:
                    prev = prev[:n]
            prev = _project_simplex(prev)
            patterns.append(prev)
        except Exception:
            pass

    best_score = float('inf')
    total_iters = 0

    # hyper‑parameters for Adam
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.25         # a little larger than the previous 0.2
    mu0 = 2.0              # start with a very smooth max
    mu_scale = 50000.0     # controls how fast mu grows
    eval_interval = 15
    lp_interval = 2000
    t_lp = 0.25            # convex‑combination coefficient for LP step

    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a random starting pattern
        a = rng.choice(patterns).copy()
        a = _project_simplex(a)    # ensure sum = 1, non‑negative

        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # run a bounded inner optimisation (the bound is generous;
        # the outer time check will break the loop when needed)
        inner_limit = 4000
        for it in range(inner_limit):
            if time.monotonic() >= deadline:
                break

            total_iters += 1

            # ------ convolution and smooth‑max ----------
            conv = np.convolve(a, a)            # length 2n‑1
            M = conv.max()

            w = _smooth_weights(conv, mu)

            # ------ gradient of smooth‑max -------------
            corr_full = np.correlate(w, a, mode='full')
            grad = 2.0 * corr_full[n - 1:n - 1 + n]

            # ------ Adam update ------------------------
            lr = base_lr / np.sqrt(total_iters)
            m = beta1 * m + (1.0 - beta1) * grad

            a = _project_simplex(a)

            if rng.random() < 0.004:
                idx = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.15))
                a[idx] *= factor
                a = _project_simplex(a)

            if total_iters % eval_interval == 0:
                score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[improve] iter={total_iters} score={best_score:.6f}")

                if best_score <= 1.5025:
                    print("[early exit] target reached")
                    return [float(x) for x in best_seq]

            # ------ LP refinement (every few thousand steps) --
            if total_iters % lp_interval == 0:
                # solve the LP that finds the vector with maximal ℓ₁‑mass
                # under the current convolution bound.
                b = _lp_refine(a, M)
                if b is not None and np.isfinite(b).all():
                    # scale b to have the same ℓ₁‑mass as a (which is 1)
                    scale = a.sum() / b.sum() if b.sum() > 0 else 0.0
                    b = b * scale
                    # convex combination – larger t accelerates the effect
                    a = (1.0 - t_lp) * a + t_lp * b
                    a = _project_simplex(a)

    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform sequence")

    print(f"[final] total_iters={total_iters} best_score={best_score:.6f}")
    return [float(x) for x in best_seq]

"""
Hybrid optimisation for the self‑convolution objective.

The routine builds a large palette of hand‑crafted and random start
vectors (uniform, tent, arcsine‑shaped, power‑law, exponential‑decay,
previous best, Dirichlet).  Each start is projected onto the probability
simplex (Σaᵢ = 1, aᵢ ≥ 0).

The core optimiser is a projected Adam method that minimises a smooth
approximation of max(a∗a) via a soft‑max (temperature μ).  The temperature
starts very smooth (μ≈2) and grows slowly so the algorithm first follows
a smooth landscape and later focuses on the true peak.

Every few thousand Adam steps a tiny linear‑programming refinement is
performed:

    maximise   Σ bⱼ
    subject to  a∗b ≤ M   (component‑wise) ,   bⱼ ≥ 0

where M = max(a∗a) for the current iterate.  The LP returns a vector b
that pushes mass into slack locations of the convolution.  b is
renormalised and mixed into the current iterate by a convex combination.
The LP is tiny (≈2 n constraints, n variables) and solved with CVXPY.

Random perturbations (multiplicative jumps) are injected occasionally to
escape plateaus.  The best sequence seen so far is kept and reported
as soon as it beats the target score (≈ 1.5025) or when the time budget
expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional import of cvxpy – the LP refinement will be skipped if it is
# not available.
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:                     # pragma: no cover
    _HAS_CVX = False
    cp = None

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_weights(conv: np.ndarray, mu: float) -> np.ndarray:
    """
    Soft‑max weights approximating the gradient of max(conv).
    The exponential is stabilised by subtracting the maximum entry.
    """
    cmax = conv.max()
    exp_args = mu * (conv - cmax)
    w = np.exp(exp_args)
    s = w.sum()
    if s == 0.0:                     # numerical safety
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w / s
    return w


def _lp_refine(a: np.ndarray, M: float):
    """
    Solve
        maximise   Σ bᵢ
        subject to  a∗b ≤ M  (component‑wise),  b ≥ 0
    where a is fixed and M = max(a∗a).
    Returns a feasible b (numpy array) or None on failure.
    """
    if not _HAS_CVX:                # pragma: no cover
        return None

    n = a.shape[0]
    # Build the linear convolution matrix B of shape (2n‑1, n)
    B = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        B[j:j + n, j] = a                     # B[k, j] = a_{k‑j}
    b = cp.Variable(n, nonneg=True)
    constraints = [B @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False,
                    max_iters=5000)
    except Exception:               # pragma: no cover
        return None

    if b.value is None:
        return None
    return np.asarray(b.value, dtype=np.float64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Search for a non‑negative step‑function height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence discovered before the
    time budget expires.

    Parameters
    ----------
    seed : int, default 42

    Returns
    -------
    list[float]
        The candidate sequence.
    """
    rng = np.random.default_rng(seed)
    # reserve a couple of seconds for final printing / cleanup
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Problem dimension – 1024 provides a good balance between expressivity
    # and per‑iteration cost (the LP refinement stays cheap).
    n = 1024

    # ------------------------------------------------------------------
    # Assemble a diverse set of initial patterns
    patterns = []

    # (a) Uniform
    patterns.append(np.full(n, 1.0 / n))

    # (b) Simple tent (linear rise then fall)

                           np.linspace(2, 1, n - half)])
    tent = tent / tent.sum()
    patterns.append(tent)

    # (c) Arcsine‑shaped – heavy at the ends, light in the centre
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = arcsine / arcsine.sum()
    patterns.append(arcsine)

    # (d) Power‑law families  a_i ∝ ((i+½)(n‑i‑½))^{-p}
    for p in (0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        shape = 1.0 / ((i + 0.5) * (n - i - 0.5)) ** p
        shape = shape / shape.sum()
        patterns.append(shape)

    # (e) Exponential decay from the edges, mirrored
    for alpha in (0.001, 0.002, 0.005, 0.01, 0.02):
        exp_shape = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        exp_shape = exp_shape / exp_shape.sum()
        patterns.append(exp_shape)

    # (f) Random Dirichlet (stochastic diversity)

    # (g) Previous global best, if present

        except Exception:

    best_seq = None
    best_score = float('inf')
    total_iters = 0

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.25                # slightly larger than the previous run
    mu0 = 2.0                     # start with a very smooth max

    # ------------------------------------------------------------------

    while time.monotonic() < deadline:
        # pick a start pattern at random and project onto the simplex
        a = rng.choice(patterns).copy()

        v = np.zeros_like(a)

        for it in range(inner_limit):
            if time.monotonic() >= deadline:

            total_iters += 1

            M = conv.max()
            mu = mu0 * (1.0 + total_iters / mu_scale)
            w = _smooth_weights(conv, mu)

            # ------------------------------------------------------------------

            grad = 2.0 * corr_full[n - 1:n - 1 + n]

            lr = base_lr / np.sqrt(total_iters)

            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            if total_iters % eval_interval == 0:

                if np.isfinite(score) and score < best_score:
