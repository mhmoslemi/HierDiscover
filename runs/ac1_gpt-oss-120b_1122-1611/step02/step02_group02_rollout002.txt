def _partner_vector(a, k):
    n = a.size
    w = np.zeros_like(a)
    if k < n:
        w[: k+1] = a[k :: -1]
    else:
        start = k - (n-1)
        length = (2*n-1) - k
        rev = a[::-1]
        w[start:] = rev[:length]
    return w

max self‑convolution.  The algorithm keeps the vector normalized (∑a=1)
and uses a symmetric arcsine‑like initialization.  At each iteration it
identifies the near‑max convolution entries (tolerance decreasing over
time), builds the sub‑gradient (sum of the corresponding partner
vectors) and performs an Adam‑style update on the log‑coordinates

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real inputs)."""
    fa = rfft(a, n=fft_len)
    fc = fa * fa
    conv = irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """Gradient contribution of the convolution entry (a*a)[k].
    Returns the vector w with w_i = a_{k-i} (zero if out of range)."""
    n = a.size
    w = np.zeros_like(a)
    if k < n:
        # indices i=0..k  map to a[k-i]
        w[: k + 1] = a[k :: -1]
    else:
        # indices i=k-(n-1)..n-1 map to a[k-i]
        start = k - (n - 1)
        length = (2 * n - 1) - k
        rev = a[::-1]
        w[start:] = rev[:length]
    return w

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out = np.full_like(out, 1.0 / n)
    else:
        out /= out_sum
    # re‑symmetrise
    out = (out + out[::-1]) / 2.0
    out /= out.sum()
    return out

def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one coordinate to another."""

    else:
        out /= out_sum
    # enforce symmetry
    out = (out + out[::-1]) / 2.0
    out /= out.sum()
    return out

    xs = np.arange(n, dtype=np.float64) + 0.5

    else:
        best_a /= total
    return [float(x) for x in best_a]
