def propose_candidate(seed=0, budget_s=1000):
   import time, numpy as np
   rng = np.random.default_rng(seed)
   deadline = time.monotonic() + budget_s - 1
   n=1024
   best_seq = None
   best_score = float('inf')
   # Evaluate uniform baseline
   uniform = np.full(n, 1.0/n)
   best_score = evaluate_sequence(uniform.tolist())
   best_seq = uniform.copy()
   # loops: sample param combos
   while time.monotonic() < deadline:
        # choose random class
        choice = rng.integers(0,3)
        if choice==0:
            # Beta shape
            a = rng.uniform(0.05, 2.5)
            b = rng.uniform(0.05, 2.5)
            i = np.arange(n, dtype=np.float64)
            arr = (i+0.5)**(a-1) * (n - i - 0.5)**(b-1)
        elif choice==1:
            # Exponential shape
            lam = rng.uniform(-3, 3)
            i = np.arange(n, dtype=np.float64)
            arr = np.exp(-lam * i)
        else:
            # Cosine modulated Beta
            a = rng.uniform(0.05, 2.5)
            b = rng.uniform(0.05, 2.5)
            B = rng.uniform(0, 0.9)
            i = np.arange(n, dtype=np.float64)
            base = (i+0.5)**(a-1) * (n - i - 0.5)**(b-1)
            mod = 1 + B * np.cos(2*np.pi * i / (n-1))
            arr = base * mod
        # ensure positivity
        arr = np.maximum(arr, 0.0)
        if arr.sum() < 1e-12:
            continue
        arr = arr / arr.sum()
        score = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = arr.copy()
            print(f"[improve] new best={best_score:.6f}")
        # early exit if target reached
        if best_score <= 1.5025:
            break
   # If time left maybe do simple gradient descent from best_seq for finetuning (optional)
   # we can implement a few subgradient steps
   if best_seq is not None and time.monotonic() < deadline:
        a = best_seq.copy()
        # simple subgradient descent
        lr = 0.1
        total_iters = 0
        while time.monotonic() < deadline and total_iters < 20000:
            conv = np.convolve(a, a)
            max_idx = int(conv.argmax())
            # gradient for max
            grad = np.zeros_like(a)
            i_start = max(0, max_idx - (n-1))
            i_end = min(n-1, max_idx)
            idxs = np.arange(i_start, i_end+1)
            j_vals = max_idx - idxs
            grad[idxs] = 2 * a[j_vals]
            total_iters += 1
            # update
            a = a - (lr / np.sqrt(total_iters)) * grad
            # project to simplex
            a = _project_simplex(a)
            # occasional evaluate
            if total_iters % 100 == 0:
                score = float(evaluate_sequence(a.tolist()))
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[improve] subgrad iter={total_iters}, score={best_score:.6f}")
        # end subgrad
   if best_seq is None:
        best_seq = np.full(n, 1.0/n)
        best_score = float(evaluate_sequence(best_seq.tolist()))
   print(f"[final] best_score={best_score:.6f}")
   return [float(x) for x in best_seq]

def build_conv_matrix(a):
    n = a.shape[0]
    rows = 2*n - 1

    for j in range(n):

"""
A hybrid optimiser for the self‑convolution bound.

The search proceeds in three stages:

1. **Initialisation** – a smooth “arcsine” shape (known to be a good
   starting point) is normalised to a probability vector.

2. **Linear‑programming refinement** – given the current distribution
   `a` (∑a = 1) we compute `M = max(a*a)`.  The linear programme

        maximise   ∑ g_i
        subject to (a ∗ g)[k] ≤ M   for all k
                   g_i ≥ 0

   finds a distribution `g` that can carry as much total mass as possible
   without increasing the worst convolution value.  After normalising `g`
   we move a little towards it (`a ← (1‑t)·a + t·g`).  Repeating this step
   flattens the convolution and quickly lowers the bound.

3. **Smooth‑max Adam fine‑tuning** – the final stage refines the vector
   using the same soft‑max surrogate of the max as in the previous
   solution, but with a different schedule and a smaller learning‑rate.
   Occasional random “shake‑ups” help escape plateaus.

The routine respects the 1000 s wall‑clock budget, keeps the best
candidate seen and returns it as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Search for a non‑negative height sequence (length 1024) that minimises
    `evaluate_sequence`.  Returns the best sequence discovered before the
    allotted wall‑clock time expires.
    """
    rng = np.random.default_rng(seed)
    n = 1024
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ---- 1. initialise with an arcsine‑shaped distribution -----------------
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = a / a.sum()                     # probability simplex
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print("[init] score =", best_score)

    # helper to keep the current best
    def _maybe_update(candidate):
        nonlocal best_seq, best_score
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"[improve] new best = {best_score:.6f}")
        return sc

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement (up to a few hundred LP solves)
    # ------------------------------------------------------------------
    lp_iter = 0
    max_lp_iters = 120                     # generous but respects budget
    while time.monotonic() < deadline and lp_iter < max_lp_iters:
        # current worst convolution value
        conv = np.convolve(a, a)          # length 2n‑1
        M = conv.max()

        # Build the full convolution matrix A where (A @ g)[k] = (a∗g)[k]
        # A is Toeplitz with first column = [a, 0, …, 0] and first row = [a[0],0,…,0]
        col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([[a[0]], np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(col, row)            # shape (2n‑1, n)

        # Linear programme: maximise ∑g_i  ⇔  minimise −∑g_i
        c = -np.ones(n, dtype=np.float64)               # objective
        b_ub = np.full(2 * n - 1, M, dtype=np.float64) # upper bound M for each k
        bounds = [(0.0, None) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={"presolve": True})
        if not res.success:
            # LP failed (rare); abort the LP phase.
            print("[lp] failure:", res.message)
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print("[lp] zero mass – aborting LP phase")
            break

        # normalise to stay on the simplex
        g = g0 / sum_g0

        # convex combination – a small step towards g
        t = 0.25 / (1 + 0.02 * lp_iter)   # decaying step size
        a = (1.0 - t) * a + t * g
        a = _project_simplex(a)

        _maybe_update(a)

        if best_score <= 1.5025:
            return [float(x) for x in best_seq]

        lp_iter += 1

    # ------------------------------------------------------------------
    # 3. Adam optimisation with a soft‑max surrogate of the max (few‑thousand steps)
    # ------------------------------------------------------------------
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    total_iters = 0

    while time.monotonic() < deadline:
        total_iters += 1

        # full self‑convolution
        conv = np.convolve(a, a)
        cmax = conv.max()

        # soft‑max over the convolution (temperature μ grows slowly)
        mu = 3.0 + 0.004 * total_iters
        exp_args = mu * (conv - cmax)
        w = np.exp(exp_args)
        w /= w.sum()                       # ∂/∂conv of the smooth max

        # gradient = 2 * correlation(w, a)
        corr_full = np.correlate(w, a, mode='full')
        grad = 2.0 * corr_full[n - 1:n - 1 + n]

        # Adam update (Euclidean space)
        lr = 0.12 / np.sqrt(total_iters)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** total_iters)
        v_hat = v / (1.0 - beta2 ** total_iters)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the probability simplex
        a = _project_simplex(a)

        # occasional random “shake‑up” to leave plateaus
        if rng.random() < 0.008:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[idx] *= factor
            a = _project_simplex(a)

        # evaluate periodically (every ~30 iterations)
        if total_iters % 30 == 0:
            _maybe_update(a)
            if best_score <= 1.5025:
                break

    # ------------------------------------------------------------------
    # Return the best sequence seen (as plain Python floats)
    # ------------------------------------------------------------------
    print("[final] best_score =", best_score)
    return [float(x) for x in best_seq]
