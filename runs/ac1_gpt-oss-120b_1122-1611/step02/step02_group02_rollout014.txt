"""
Gradient‑based flattening of the self‑convolution on the probability simplex.

The algorithm works on a probability vector `a` (non‑negative, sum = 1) of
length ``n`` (taken from the provided ``height_sequence_1`` when
available, otherwise a modest power‑of‑two fallback).  The loss is a
smooth approximation of the ℓ∞‑norm of the convolution:
    L(a) =  (1/α)·log∑_k exp(α·(a*a)[k])  .
For α→∞ this tends to the true maximum.  Its gradient can be written
in closed form using a single FFT‑based convolution, which makes each
iteration cheap (≈ 2 FFT’s).  An exponentiated‑gradient (natural‑gradient)
update with momentum is applied, with a slowly decreasing step size.
Occasional random perturbations and restarts avoid stagnation.

The routine runs for the allotted time (≈ 1000 s) and returns the best
sequence found.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_len_full_conv(n_a: int, n_b: int) -> int:
    """Length of the FFT to compute a full linear convolution of lengths
    ``n_a`` and ``n_b``."""
    return _next_pow2(n_a + n_b - 1)


def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear (full) convolution of real vectors `a` and `b` using FFT.
    Returns an array of length ``len(a) + len(b) - 1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)
    return conv[: a.size + b.size - 1]


def _init_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    It already gives a fairly flat self‑convolution.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a


def _random_dirichlet(rng: np.random.Generator, n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex via Gamma (Dirichlet) sampling."""
    x = rng.gamma(shape=alpha, scale=1.0, size=n)
    return x / x.sum()


def _block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 1e-15, None)
    out /= out.sum()
    return out


def _pairwise_swap(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 1e-15, None)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a tiny safety margin) and uses a
    smooth‑max gradient descent on the simplex.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # fallback size – a power of two works nicely with the FFT
        n = 8192
        base = None

    # ------------------------------------------------------------------
    # Pre‑computations
    # ------------------------------------------------------------------
    len_w = 2 * n - 1                # length of a*a
    fft_len = _fft_len_full_conv(n, len_w)   # enough for both convolutions

    # ------------------------------------------------------------------
    # Keep the best solution seen so far
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 0.5                # initial step size for the exponentiated update
    eta_final = 0.02          # final step size after the schedule ends
    decay_start = 0.15        # fraction of total time after which we start strong decay
    beta_mom = 0.85           # momentum coefficient
    stagnation_limit = 30.0   # seconds without improvement → restart
    p_start = 1.0
    p_end = 10.0              # exponent for the soft‑max weighting (higher → sharper)
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_budget = deadline - start_time
    momentum = np.zeros(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Main optimisation loop (restarts on stagnation)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- (re)initialise -------------------------------------------------
        if rng.random() < 0.7:
            a = _init_arcsine(n)
        else:
            a = _random_dirichlet(rng, n, alpha=0.5)
        # small jitter to avoid perfect symmetry
        a += rng.normal(scale=1e-4, size=n)
        a = np.clip(a, 1e-15, None)
        a /= a.sum()
        momentum.fill(0.0)
        iteration = 0
        last_improve = time.monotonic()

        # ----- inner optimisation ---------------------------------------------
        while time.monotonic() < deadline:
            iteration += 1

            # ---- compute self‑convolution ------------------------------------
            A_fft = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(A_fft * A_fft, n=fft_len)[:len_w]       # a*a
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv

            # ---- record improvement -----------------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

            # ---- smooth‑max weighting ---------------------------------------
            # p grows from p_start to p_end linearly with time
            elapsed_total = time.monotonic() - start_time
            frac_total = elapsed_total / total_budget
            p = p_start + (p_end - p_start) * frac_total

            # power weighting (conv > 0 because a ≥ 0 and sum a = 1)
            w = np.power(np.maximum(conv, 1e-300), p)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ---- gradient of the smooth‑max loss ----------------------------
            # gradient_i = 2 * sum_k w_k * a_{k-i}
            # which is the cross‑correlation of a with w.
            w_rev = w[::-1]                                 # reverse for convolution
            W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
            conv_rev = np.fft.ifft(A_fft * W_rev_fft).real
            conv_rev = conv_rev[: n + len_w - 1]            # length = 3n‑2
            idx = (len_w - 1) - np.arange(n)                # positions of the needed slice
            grad = 2.0 * conv_rev[idx]

            # ---- momentum smoothing -----------------------------------------
            momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

            # ---- step‑size schedule -----------------------------------------
            if frac_total < decay_start:
                eta = eta0 * (1.0 - frac_total / decay_start)
            else:
                # after early phase we decay more aggressively
                eta = eta0 * (1.0 - (frac_total - decay_start) /
                              (1.0 - decay_start)) * (eta_final / eta0)

            # ---- exponentiated‑gradient update -------------------------------
            factor = np.exp(-eta * momentum)
            a = a * factor
            a = np.clip(a, 1e-15, None)      # avoid exact zeros
            a /= a.sum()                     # renormalise to stay on the simplex

            # ---- occasional stochastic exploration ---------------------------
            r = rng.random()
            if r < 0.015:
                a = _block_perturb(a, rng)
            elif r < 0.025:
                a = _pairwise_swap(a, rng)

            # ---- stagnation test -------------------------------------------
            if time.monotonic() - last_improve > stagnation_limit:
                # break to a fresh restart
                print(f"[{time.monotonic() - start_time:6.2f}s] restart (stagnation)")
                break

        # end of inner loop – another restart will be attempted automatically

    # ----------------------------------------------------------------------
    # Return the best vector (clipped to satisfy evaluate_sequence's limits)
    # ----------------------------------------------------------------------
    if best_a is None:
        # extremely unlikely, but fallback to a uniform distribution
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
