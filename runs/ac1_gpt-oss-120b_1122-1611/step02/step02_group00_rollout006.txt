"""
propose_candidate
-----------------
Projected gradient descent on a *soft‑max* approximation of the
maximum self‑convolution value.

The original problem (scale‑invariant) is  

    minimise   max_k (w * w)[k]    subject to   w ≥ 0 ,  Σ w = 1 .

Because the pointwise maximum is non‑smooth we replace it by the
smooth “log‑sum‑exp’’ surrogate  

    S_β(w) = (1/β)·log ∑_k exp(β·(w*w)[k]),   β > 0 ,

whose gradient is the weighted sum of the individual gradients

    ∇S_β = 2·(p * w)[:n] ,   p_k = exp(β·(w*w)[k]) / Z .

Thus each iteration consists of:
  1. compute the self‑convolution b = w*w,
  2. form the soft‑max weights p,
  3. obtain the gradient via a second FFT‑based convolution,
  4. take a diminishing step and project back onto the probability simplex.

A small amount of random “block’’ perturbation is applied occasionally
to escape shallow local minima.  The algorithm keeps the best vector
encountered and finally rescales it to the conventional L¹‑norm
√(2 n) required by the evaluation harness.

The whole search runs for at most `budget_s` seconds (default 1100 s,
a few seconds are kept for the final scaling and reporting).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – defined at top level.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan‑type coefficients a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using real FFT.
    Equivalent to np.convolve(a, b) but O(n log n) for large n.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off errors
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # In the unlikely case all entries are equal, return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation used as a “jump’’.
    A random contiguous block is multiplied by a log‑normal factor and,
    with some probability, the whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

            a = catalan_sequence(n)
            w = a / np.sum(a)
    else:
        n = 6000
        a = catalan_sequence(n)
        w = a / np.sum(a)

    # ------------------------------------------------------------------
    # State for the optimiser.
    # ------------------------------------------------------------------
    best_w = w.copy()
    # current objective (scale‑invariant form): 2·n·max(b)
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)

    print(f"[init] length={n}, start score={best_score:.6f}")

    start_time = time.monotonic()
    max_time = max(0.0, float(budget_s) - 5.0)   # reserve a few seconds for finalising

    # gradient‑descent hyper‑parameters
    eta0 = 0.7               # base step‑size
    beta = 1.0               # start of soft‑max smoothing
    beta_max = 200.0
    beta_growth = 1.025      # multiplicative increase every `beta_inc_interval`
    beta_inc_interval = 2000
    momentum = 0.9           # for a simple heavy‑ball momentum term
    velocity = np.zeros_like(w)

    iteration = 0
    no_improve = 0

    while time.monotonic() - start_time < max_time:
        # --------------------------------------------------------------
        # 1) self‑convolution and current score
        # --------------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        current_score = 2.0 * n * max_b

        # record improvement
        if current_score < best_score - 1e-9:
            best_score = current_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 2) soft‑max weights for the smoothed max
        # --------------------------------------------------------------
        # stability: subtract the (large) max before exponentiation
        exp_arg = beta * (b - max_b)           # ≤ 0
        p = np.exp(exp_arg)
        sum_p = np.sum(p)
        if sum_p == 0.0:                       # extremely large β → underflow
            # fall back to a uniform weight (unlikely)
            p = np.full_like(b, 1.0 / b.size)
            sum_p = 1.0
        p /= sum_p                              # now Σ p = 1

        # --------------------------------------------------------------
        # 3) gradient of the smoothed objective: 2·(p * w)[:n]
        # --------------------------------------------------------------
        grad = 2.0 * convolve_fft(p, w)[:n]

        # --------------------------------------------------------------
        # 4) heavy‑ball momentum + projected step
        # --------------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1.0)
        velocity = momentum * velocity + eta * grad
        w_candidate = w - velocity
        w = simplex_projection(w_candidate)

        # --------------------------------------------------------------
        # 5) occasional random block perturbation to escape stagnation
        # --------------------------------------------------------------
        if iteration % 5000 == 0 and iteration > 0:
            pert = block_perturbation(best_w, rng)
            pert = pert / np.sum(pert)          # renormalise to a probability vector
            w = pert.copy()
            velocity.fill(0.0)
            print(f"[perturb] iter={iteration:6d}")

        # --------------------------------------------------------------
        # 6) increase β slowly – the objective becomes closer to the true max
        # --------------------------------------------------------------
        if iteration % beta_inc_interval == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_growth)

        # --------------------------------------------------------------
        # 7) reset if no improvement for a long stretch
        # --------------------------------------------------------------
        if no_improve > 40000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            velocity.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – use the conventional √(2 n) L¹‑norm.
    # The value of `evaluate_sequence` after this scaling equals
    # `best_score` computed above, but we call it once for safety.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum          # Σ final_seq = √(2 n)
    final_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"[final] length={n}, best_score={best_score:.6f}, "
          f"evaluate={final_score:.6f}")

    # Convert to plain Python list for the harness.
    return [float(x) for x in final_seq]
