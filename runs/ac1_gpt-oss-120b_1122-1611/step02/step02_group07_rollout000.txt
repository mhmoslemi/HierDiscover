"""propose_candidate

Hybrid stochastic optimizer for the step‑function bound.

* Starts from the previously discovered best sequence (``height_sequence_1``)
  or from a uniform distribution on the probability simplex.
* Uses an Adam‑style update on a smooth approximation of the L∞‑norm
  (log‑sum‑exp) of the autocorrelation ``a*a``.
* Every ``hard_interval`` iterations replaces the smooth max by the
  true hard‑max gradient (sub‑gradient of the L∞‑norm) to sharpen the
  descent.
* Periodic multiplicative “jitter’’ moves mass between coordinates.
* After a long stagnation period the iterates are re‑initialized by
  adding small Gaussian noise to the current best solution (a cheap
  restart that helps escape plateaus).
* The learning‑rate decays slowly (1/(1+it·lr_decay)) and the temperature
  for the smooth‑max is increased from ``beta_min`` up to ``beta_max``.
* The best vector found (projected onto the simplex) is returned.

The algorithm runs until the time budget is exhausted or the target
value (≈1.5025) is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The optimizer is a hybrid Adam + hard‑max sub‑gradient method with
    periodic random jitter and restart.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time handling – keep a safety margin of a half‑second.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # problem size – reuse the size of the warm‑start if possible.
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = a0.size
            a = a0.copy()
        except Exception:
            n = 1024
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 1024
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ensure we start on the simplex
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters (tuned empirically)
    base_lr = 1.0          # initial learning rate
    lr_decay = 1e-4        # slower decay than 1/√t
    beta_min = 10.0        # start temperature for soft‑max
    beta_max = 400.0       # final temperature (sharp max)
    beta_inc_iters = 300_000   # iterations over which beta is increased
    hard_interval = 30_000      # use exact max‑gradient every this many iters
    jitter_prob = 0.08          # probability of a multiplicative jitter step
    jitter_sigma = 0.20         # std‑dev of log‑normal jitter factor

    # restart parameters
    stagnation_limit = 150_000  # iterations without improvement before a restart

    # ------------------------------------------------------------------
    # bookkeeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    it = 0
    last_improve = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) autocorrelation (linear convolution)
        conv = np.convolve(a, a)          # length 2n‑1

        # --------------------------------------------------------------
        # 2) choose gradient source: smooth‑max or hard‑max
        if (it % hard_interval) == 0:
            # hard max – sub‑gradient of the L∞ norm
            k_max = int(np.argmax(conv))
            w = np.zeros_like(conv)
            w[k_max] = 1.0
            beta = beta_max               # for logging only
        else:
            # smooth‑max (log‑sum‑exp) with a temperature that rises gradually
            beta = beta_min + (beta_max - beta_min) * min(1.0, it / beta_inc_iters)
            max_c = conv.max()
            # stable softmax
            exps = np.exp(beta * (conv - max_c))
            w = exps / exps.sum()

        # --------------------------------------------------------------
        # 3) gradient of the (smooth) max‑convolution wrt a
        a_rev = a[::-1]
        grad_full = np.convolve(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]    # central n entries

        # --------------------------------------------------------------
        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (slow decay)
        lr = base_lr / (1.0 + it * lr_decay)

        a_next = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5) occasional multiplicative jitter on a random coordinate
        if rng.random() < jitter_prob:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, jitter_sigma)))
            a_next[idx] *= factor

        # --------------------------------------------------------------
        # 6) project back onto the simplex (non‑negative, sum = 1)
        a_next = _project_simplex(a_next)

        # --------------------------------------------------------------
        # 7) evaluate the candidate
        score = float(evaluate_sequence(a_next.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_next.copy()
            best_score = score
            last_improve = it
            print(f"[improve] iter={it:,}  score={best_score:.6f}  beta={beta:.1f}  lr={lr:.4f}")

        # --------------------------------------------------------------
        # 8) restart if we have been stagnant for a long stretch
        if it - last_improve > stagnation_limit:
            # add small Gaussian noise to the best solution and re‑project
            noise = rng.normal(0.0, 0.02, size=n)
            a = best + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            last_improve = it
            print(f"[restart] iter={it:,}  added noise, continue search")
        else:
            a = a_next

        # --------------------------------------------------------------
        # 9) periodic status report
        if it % 25_000 == 0:
            elapsed = time.monotonic() - (deadline - budget_s + 0.5)
            print(f"[status] iter={it:,}  best_score={best_score:.6f}  time={elapsed:.1f}s")

        # --------------------------------------------------------------
        # 10) early exit if we have hit the target bound
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping early")
            break

    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    # Return a plain Python list of floats as required by the evaluator.
    return [float(x) for x in best]
