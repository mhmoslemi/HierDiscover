"""
Search for a step function with low max self‑convolution.
The algorithm uses a multiplicative‑weights (exponentiated‑gradient)
update driven by the sub‑gradient of the active convolution constraints.
The active set is defined by all convolution entries within a
fraction `ε` of the current maximum.  `ε` is annealed from a generous
value early on to a tiny value later, allowing a coarse‑to‑fine
optimisation.  Momentum smooths the update direction and occasional
random perturbations help escape plateaus.  The best sequence seen
throughout the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real non‑negative vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Compute the summed partner vector (sub‑gradient) for the active
    convolution indices.

    active – 1‑D array of length 2·n‑1 containing 1 where the constraint
    (a*a)[k] is considered active and 0 elsewhere.
    """
    n = a.size
    a_rev = a[::-1]

    # convolution length = (2n‑1) + n – 1 = 3n‑2
    conv_len = active.size + a_rev.size - 1
    fft_len = _next_pow2(conv_len)

    fa = np.fft.rfft(active, n=fft_len)
    fb = np.fft.rfft(a_rev, n=fft_len)
    prod = fa * fb
    conv = np.fft.irfft(prod, n=fft_len)[:conv_len]

    # the needed slice starts at index n‑1
    start = n - 1
    w = conv[start:start + n]
    return w


# ----------------------------------------------------------------------
# Simple stochastic perturbations
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from one coordinate to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]   # up to 30 % of a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def _init_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    The vector is normalised to sum to 1.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= total
    return a


def _init_uniform(n: int) -> np.ndarray:
    """Uniform distribution (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)

        base = None

    # ------------------------------------------------------------------

    a = _init_arcsine(n)
    a += rng.normal(scale=1e-3, size=n)   # break perfect symmetry
    a = np.clip(a, 0.0, None)

    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------

        a = a * factor
        a = np.clip(a, 1e-14, None)   # keep strictly positive numbers

        if r < 0.015:
            a = _random_block_perturb(a, rng)
        elif r < 0.025:
            a = _pairwise_swap_perturb(a, rng)

        conv = _conv_self(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        # ---- 7. record improvement ---------------------------------------
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best score = {best_score:.6f}")

        if time.monotonic() - last_improve > stagnation_limit:

            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best vector obtained, respecting the limits of evaluate_sequence
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

        # -----------------------------------------------------------------
        if frac < decay_start:
            eta = start_eta * (1.0 - frac / decay_start)      # gentle linear decay
        else:
            eta = start_eta * 0.1 * (1.0 - (frac - decay_start) / (1.0 - decay_start))

        # -----------------------------------------------------------------
        # 4. exponentiated‑gradient update (multiplicative weights)
        # -----------------------------------------------------------------
        expo = -eta * momentum
        expo = np.clip(expo, -40.0, 40.0)       # avoid overflow/underflow
        factor = np.exp(expo)
        a = a * factor
        a = np.clip(a, 1e-14, None)            # keep strictly positive numbers
        a /= a.sum()

        # -----------------------------------------------------------------
        # 5. occasional stochastic perturbations (helps escape plateaus)
        # -----------------------------------------------------------------
        r = rng.random()
        if r < 0.015:
            a = _random_block_perturb(a, rng)
        elif r < 0.025:
            a = _pairwise_swap_perturb(a, rng)

        # -----------------------------------------------------------------
        # 6. evaluate the new score
        # -----------------------------------------------------------------
        conv = _conv_self(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        # -----------------------------------------------------------------
        # 7. record improvement
        # -----------------------------------------------------------------
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best score = {best_score:.6f}")

        # -----------------------------------------------------------------
        # 8. restart if we have been stagnant for a while
        # -----------------------------------------------------------------
        if time.monotonic() - last_improve > stagnation_limit:
            # fresh start: mostly arcsine, occasionally a random Dirichlet‑like point
            if rng.random() < 0.6:
                a = _init_arcsine(n)
            else:
                a = rng.gamma(shape=1.0, size=n)
                a /= a.sum()
            a += rng.normal(scale=1e-3, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            momentum.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart (stagnation)")

    # -----------------------------------------------------------------
    # Return the best vector seen (clipped to the limits of evaluate_sequence)
    # -----------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
