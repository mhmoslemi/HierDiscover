"""
Propose a low‑evaluation step‑function using a two‑phase optimizer:

1. **Linear‑programming phase** – given a current sequence `a`,
   compute its worst self‑convolution value `M`.  Solve the LP

        max   Σ b_j
        s.t.  (a * b)_k ≤ M   for all k = 0 … 2n−2
              b_j ≥ 0

   The LP tries to increase the ℓ₁‑mass while keeping the convolution
   bounded by the current worst entry.  After solving we normalise the
   optimiser `b` to sum‑one and, if it improves the true objective,
   replace `a` with it.  Repeating this step pushes the maximal
   convolution downward.

2. **Adam‑based smooth‑max phase** – once the LP phase slows down,
   fine‑tune the sequence with the Adam optimiser on the smooth‑max
   surrogate of the self‑convolution (the same surrogate used in the
   reference implementation).  The sequence stays on the probability
   simplex (non‑negative, sum = 1).

The algorithm works for any length; it starts from the globally stored
`height_sequence_1` if present, otherwise from a random Dirichlet point.
Running time is limited by the supplied budget (default 1000 s).  The
best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation of the objective 2·n·max(conv)/(sum a)²."""
    conv = np.convolve(a, a)
    return float(2.0 * a.size * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _build_A_ub(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build the sparse matrix A such that (A @ b)[k] = (a * b)[k]
    for all k = 0 … 2·n‑2.
    """
    n = a.size
    # rows[i, j] = i + j      (i for a, j for b)
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(a, n)
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, max_conv: float) -> np.ndarray | None:
    """
    One LP step: maximise Σ b_j subject to (a * b)_k ≤ max_conv.
    Returns the normalised optimiser `b` (sum = 1) or None on failure.
    """
    n = a.size
    A_ub = _build_A_ub(a)
    b_ub = np.full(2 * n - 1, max_conv, dtype=np.float64)
    c = -np.ones(n)                     # minimise -Σb  => maximise Σb
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None
    return b / s                         # normalise to ℓ₁‑norm = 1


# ----------------------------------------------------------------------
def _adam_fine_tune(a: np.ndarray,
                    deadline: float,
                    best_score: float,
                    best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (identical to the reference
    implementation) for the remaining time.  Starts from `a`.
    Returns the best sequence and its score.
    """
    rng = np.random.default_rng(0)
    n = a.size
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_base = 0.05
    tau_factor = 0.5
    tau_decay = 0.9999

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # full convolution
        conv = np.convolve(a, a)
        current_max = conv.max()
        tau = max(current_max * tau_factor, 1e-12)

        # smooth‑max weights (stable)
        shifted = (conv - current_max) / tau
        w = np.exp(shifted - shifted.max())  # subtract max for stability
        w /= w.sum()

        # gradient of smooth‑max: 2 * correlation(w, a)
        grad = 2.0 * np.correlate(w, a, mode='full')[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr_base / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)

        # occasional tiny random shake‑up
        if rng.random() < 0.005:
            a_candidate += rng.normal(0.0, 0.015, size=n)
            a_candidate = _project_simplex(a_candidate)

        score = _objective(a_candidate)
        if score < best_score:
            best_score = score
            best_seq = a_candidate.copy()
            print(f"[adam improve] iter={iteration:,} score={best_score:.6f}")

        a = a_candidate
        tau_factor *= tau_decay
        if iteration % 5000 == 0:
            tau_factor = max(tau_factor * 0.99, 1e-6)

        # early stop if we are essentially at the theoretical bound
        lower = 2.0 * n / (2 * n - 1)
        if best_score <= lower * 1.01:
            print("[adam early stop] close to theoretical bound")
            break

    return best_seq, best_score

    """

    # ------------------------------------------------------------------

    try:

        if known.ndim == 1 and known.size > 0:
            n = known.size
            a = known.copy()
        else:
            raise ValueError
    except Exception:
        n = 1024
        a = rng.random(n)

    # normalise to a probability simplex (sum = 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:

    else:

    # book‑keeping of the best found sequence
    best_seq = a.copy()

    # Phase 1 – Linear‑programming updates

    lp_iterations = 0
