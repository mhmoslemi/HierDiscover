"""
Hybrid projected‑Adam + high‑temperature soft‑max + final exponentiated‑gradient
search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works entirely on the probability simplex (∑a_i = 1,
a_i ≥ 0) so the objective is scale‑invariant.
A fast FFT‑based convolution makes each iteration cheap, allowing
hundreds of thousands of updates inside the 1000 s budget.

* Phase 1 – projected Adam:
    * a soft‑max with temperature µ approximates the “max” in the
      objective.  µ starts small (smooth landscape) and rises quickly
      to a large value (≈ 𝛼·10³) so that later updates follow the true
      sub‑gradient of the peak.
    * Adam’s adaptive step‑size together with Euclidean projection onto
      the simplex keeps the iterate feasible.
    * The best sequence seen so far (exact objective) is stored.

* Phase 2 – exponentiated‑gradient fine‑tuning:
    * µ is set to a huge value so that the weight vector w becomes a
      one‑hot at the current maximal convolution index.
    * The update a ← a·exp(‑η·∇) (with η decaying) automatically
      respects positivity; normalisation restores the simplex.
    * Occasional random perturbations help to escape tiny plateaus.

The whole search stops before the supplied time budget
and returns the best sequence found.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(a: np.ndarray, b: np.ndarray, N: int) -> np.ndarray:
    """Linear convolution a*b using a size‑N FFT (real input)."""
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    conv = np.fft.irfft(C, N)
    return conv[: a.shape[0] + b.shape[0] - 1]


def _objective(a: np.ndarray, conv: np.ndarray = None) -> float:
    """Exact objective 2·n·max(conv)/sum(a)^2.  If conv not supplied it is computed."""
    n = a.shape[0]
    if conv is None:
        N = 1 << ((2 * n - 1).bit_length())
        conv = _fft_convolve(a, a, N)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42,
                      budget_s: float = 1100,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    self‑convolution objective.  Returns the best sequence discovered
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – 1024 gives a good trade‑off between expressiveness
    # and per‑iteration cost.
    n = 1024

    # ------------------------------------------------------------------
    # helper: generate a few diverse starting patterns.
    patterns = []

    # uniform
    patterns.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine – heavy at the ends
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)

    # simple triangular (tent)
    half = n // 2
    tent = np.concatenate([np.linspace(1, 2, half, endpoint=False),
                           np.linspace(2, 1, n - half)])
    tent /= tent.sum()
    patterns.append(tent)

    # sine‑shaped (smooth bump in the centre)
    sine = np.sin(np.pi * (i + 0.5) / n)
    sine /= sine.sum()
    patterns.append(sine)

    # random Dirichlet
    patterns.append(rng.dirichlet(np.ones(n)).astype(np.float64))

    # previous best if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                # simple linear interpolation / truncation
                if prev.size < n:
                    # interpolate to larger size
                    x_old = np.linspace(0, 1, prev.size)
                    x_new = np.linspace(0, 1, n)
                    prev = np.interp(x_new, x_old, prev)
                else:
                    prev = prev[:n]
            prev = _project_simplex(prev)
            patterns.append(prev)
        except Exception:
            pass  # ignore malformed previous variable

    # ------------------------------------------------------------------
    # pick the best of the starter patterns (using the exact objective)
    best_seq = None
    best_score = float("inf")
    for pat in patterns:
        score = _objective(pat)
        if score < best_score:
            best_score = score
            best_seq = pat.copy()
    print(f"[init] best starter score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation – projected Adam with a temperature schedule.
    # Allocation of the time budget:
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # 1 s safety margin

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Temperature schedule (soft‑max parameter µ)
    mu_start = 0.5          # smooth at the beginning
    mu_max = 800.0          # very close to a true max at the end
    mu_ramp = 200_000       # number of iterations needed to reach mu_max

    # Learning‑rate schedule for Adam
    lr_base = 0.5           # larger than the original 0.2

    # Pre‑compute FFT length (covers both convolution and correlation)
    N_fft = 1 << ((3 * n - 2).bit_length())   # ≥ 3n‑2

    # Adam state
    m = np.zeros_like(best_seq)
    v = np.zeros_like(best_seq)

    total_iters = 0
    last_improve_iter = 0

    # ------------------------------------------------------------------
    # Phase 1 – projected Adam
    while time.monotonic() < deadline:
        total_iters += 1

        # ----- convolution a*a (linear, length 2n‑1) -----
        conv_full = _fft_convolve(best_seq, best_seq, N_fft)[: 2 * n - 1]
        conv_full = np.maximum(conv_full, 0.0)        # guard against tiny negatives
        max_conv = conv_full.max()

        # ----- temperature (soft‑max) -----
        # linear ramp up, then clamp
        mu = mu_start + (mu_max - mu_start) * min(1.0, total_iters / mu_ramp)

        # ----- soft‑max weights w (approximation of the sub‑gradient of the max) -----
        exp_args = mu * (conv_full - max_conv)
        w = np.exp(exp_args)
        w_sum = w.sum()
        w /= w_sum

        # ----- gradient = 2 * correlation(w, a) (indices n‑1 … n‑1+n‑1) -----
        a_rev = best_seq[::-1]
        corr_full = _fft_convolve(w, a_rev, N_fft)  # length 3n‑2
        grad = 2.0 * corr_full[n - 1: n - 1 + n]

        # ----- Adam update on the simplex -----
        lr = lr_base / np.sqrt(total_iters)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** total_iters)
        v_hat = v / (1.0 - beta2 ** total_iters)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        best_seq = best_seq - step
        best_seq = _project_simplex(best_seq)

        # ----- occasional random shake‑up (helps escape shallow plateaux) -----
        if rng.random() < 0.007:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.16))
            best_seq[idx] *= factor
            best_seq = _project_simplex(best_seq)

        # ----- evaluation of the *exact* objective (every ~30 steps) -----
        if total_iters % 30 == 0:
            cur_score = _objective(best_seq, conv_full)
            if cur_score < best_score:
                best_score = cur_score
                last_improve_iter = total_iters
                print(f"[improve] iter={total_iters:,} score={best_score:.6f}")
                # early stop if we already beat the target
                if best_score <= 1.5025:
                    print("[early exit] target reached during Adam phase")
                    return [float(x) for x in best_seq]

        # ----- restart if stuck for a long time -----
        if total_iters - last_improve_iter > 120_000:
            # re‑initialise from the current best but shake it a bit
            best_seq = _project_simplex(best_seq + 1e-4 * rng.standard_normal(n))
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = total_iters
            print("[restart] no improvement for a while – re‑initialising Adam state")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient fine‑tuning with the *exact* max gradient
    print("[phase‑2] starting exponentiated‑gradient fine‑tuning")
    # recompute a_rev once per loop (cheap)
    while time.monotonic() < deadline:
        total_iters += 1

        # exact convolution & maximal index
        conv_full = _fft_convolve(best_seq, best_seq, N_fft)[: 2 * n - 1]
        conv_full = np.maximum(conv_full, 0.0)
        max_idx = int(conv_full.argmax())

        # exact sub‑gradient (soft‑max with µ→∞ ⇢ one‑hot at max_idx)
        # gradient_i = a_{max_idx‑i} + a_i   if 0 ≤ max_idx‑i < n else 0
        idxs = np.arange(n)
        valid = (max_idx - idxs >= 0) & (max_idx - idxs < n)
        grad = np.zeros_like(best_seq)
        grad[valid] = best_seq[max_idx - idxs[valid]] + best_seq[idxs[valid]]

        # exponentiated‑gradient step (η decays like 1/√t)
        eta = 0.6 / np.sqrt(total_iters)
        best_seq = best_seq * np.exp(-eta * grad)
        # avoid complete collapse to zero
        best_seq = np.maximum(best_seq, 1e-20)
        best_seq /= best_seq.sum()   # projection onto simplex (simple normalisation)

        # occasional random jitter to break ties
        if rng.random() < 0.005:
            i, j = rng.choice(n, size=2, replace=False)
            delta = 0.02 * best_seq[i] * rng.random()
            best_seq[i] = max(best_seq[i] - delta, 0.0)
            best_seq[j] += delta
            best_seq /= best_seq.sum()

        # periodic exact evaluation
        if total_iters % 20 == 0:
            cur_score = _objective(best_seq, conv_full)
            if cur_score < best_score:
                best_score = cur_score
                last_improve_iter = total_iters
                print(f"[improve‑phase2] iter={total_iters:,} score={best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early exit] target reached")
                    return [float(x) for x in best_seq]

        # tiny safety break if we run out of time
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # final fallback – if nothing was ever accepted, return uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"[final] total_iters={total_iters:,} best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
