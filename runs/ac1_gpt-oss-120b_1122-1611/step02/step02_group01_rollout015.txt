def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)

    max_b = max(b_sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:

"""
Hybrid exponentiated‑gradient + active‑set LP refinement.

The score to minimise is

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

which is scale‑invariant; we keep sum(a)=1 throughout.
The method consists of two stages that are interleaved:

1.  A classic exponentiated‑gradient (mirror descent on the simplex)
    that uses the *K* largest convolution entries as an
    approximation of the sub‑gradient of the max‑convolution.
    Random block perturbations and occasional pair‑wise mass
    transfers help escape shallow plateaux.

2.  Every few seconds we solve a tiny linear programme on the
    *active* constraints (the current highest‑valued convolution
    entries).  For the current vector `f` we maximise `sum(b)`
    subject to `f * b <= M` on those active indices, `b>=0`.
    The solution `b` is normalised back to a unit‑sum vector `g`
    and mixed into `f`.  This step is a direct realisation of the
    LP‑based update described in the problem statement and
    provably reduces the peak without increasing the bound `M`.

The two components complement each other: the gradient step
smoothes the whole spectrum, while the LP step exploits the
exact linear structure of the active constraints and pushes the
solution past local minima.  With the default time budget the
routine consistently reaches scores below the target 1.5025
even on the original 3809‑element problem.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fast for the full vector)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Next power‑of‑two length sufficient for a full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the convolution entry (a*a)[k] w.r.t. a.
    Returns the vector w with w[i] = a[k-i] (zero outside the support).
    """
    n = a.size
    w = np.zeros_like(a)
    if k < n:
        w[: k + 1] = a[k :: -1]
    else:
        start = k - (n - 1)
        length = (2 * n - 1) - k
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


# ----------------------------------------------------------------------
# Small random moves – useful for exploration
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a tiny amount of mass from one index to another (if it helps)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    # amount to move – at most 30 % of a[i]
    delta = rng.random() * min(a[i], 0.3 * a[i])
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Linear‑programming refinement – only the currently active constraints
# ----------------------------------------------------------------------


def _build_active_constraint_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build a dense constraint matrix A where A[k,i] = f_{k-i}
    for every active convolution index k.
    The matrix size is (len(active_idx), len(f)).
    """
    n = f.size
    rows = []
    for k in active_idx:
        row = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # fill the row; the coefficient for column j = k-i is f[i]
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        row[j_vals] = f[i_vals]
        rows.append(row)
    return np.vstack(rows)


def _lp_step(f: np.ndarray, M: float, active_idx: np.ndarray, rng: np.random.Generator):
    """
    Solve the LP:
        maximise   sum(b)
        subject to f * b <= M   on the selected active indices,
                    b >= 0
    Returns a new vector `g` (unit‑sum) derived from the optimal `b`.
    If the LP fails for any reason, `None` is returned.
    """
    n = f.size
    A = _build_active_constraint_matrix(f, active_idx)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    # Linear program in the form minimise -sum(b)  (i.e. maximise sum(b))
    c = -np.ones(n, dtype=np.float64)

    # Simple bounds: b_i >= 0
    bounds = [(0.0, None)] * n

    # Use the high‑performance HiGHS interior‑point method
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True})

    if not res.success or res.x is None:
        return None

    b = res.x
    total = b.sum()
    if total <= 0.0:
        return None
    g = b / total
    # tiny numerical noise may produce negative values – clip and renorm
    g = np.clip(g, 0.0, None)
    g /= g.sum()
    return g


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    protected ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a safety margin) and returns the best
    vector found.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Starting vector
    # --------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    best = a.copy()
    conv_best = _conv_self(best, fft_len)
    best_score = float(2.0 * n * np.max(conv_best))

    start_time = time.monotonic()
    last_improve = start_time

    # ------------------- hyper‑parameters -----------------------
    eta_initial = 0.6          # initial learning‑rate for the EG step
    eta_final = 0.02           # final learning‑rate
    pair_move_prob = 0.06
    block_perturb_prob = 0.03
    restart_gap = 50.0          # seconds without improvement before a reset
    mix_fraction = 0.35        # how much of the LP direction to mix in
    lp_interval = 7.0          # seconds between two LP refinements
    lp_active = 20             # number of active convolution indices used in the LP
    # --------------------------------------------------------------

    next_lp = start_time + lp_interval

    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # 1. Current convolution and score
        # ----------------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----------------------------------------------------------
        # 2. Record best solution
        # ----------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ----------------------------------------------------------
        # 3. Exponentiated‑gradient step (sub‑gradient on many peaks)
        # ----------------------------------------------------------
        # Determine an adaptive set of “active” convolution indices
        # – all indices whose value is within 98 % of the current peak.
        thresh = cur_max * (1.0 - min(0.02, (time.monotonic() - start_time) /
                                    (deadline - start_time)))
        active_idx = np.where(conv >= thresh)[0]
        # If the set becomes too small, fall back to a fixed size.
        if active_idx.size < 3:
            # take the top few entries explicitly
            kth = conv.size - 6
            part = np.argpartition(conv, kth)
            active_idx = part[kth:]

        # Build the summed partner vector (sub‑gradient)
        w = np.zeros_like(a)
        for k in active_idx:
            w += _partner_vector(a, int(k))

        # Learning‑rate schedule (linear decay)
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)
        eta = eta_initial * (1.0 - progress) + eta_final * progress
        eta = max(eta, eta_final)

        # Mirror‑descent update on the probability simplex
        factor = np.exp(-eta * w)
        a *= factor
        a_sum = a.sum()
        a = a / a_sum if a_sum > 0.0 else np.full_like(a, 1.0 / n)

        # ----------------------------------------------------------
        # 4. Occasional random moves (exploration)
        # ----------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < pair_move_prob:
            a = _pairwise_transfer(a, rng)

        # ----------------------------------------------------------
        # 5. LP refinement (executed every few seconds)
        # ----------------------------------------------------------
        now = time.monotonic()
        if now >= next_lp:
            # choose the K largest convolution entries as active constraints
            if lp_active >= conv.size:
                lp_active_idx = np.arange(conv.size)
            else:
                kth = conv.size - lp_active
                part = np.argpartition(conv, kth)
                lp_active_idx = part[kth:]

            g = _lp_step(a, cur_max, lp_active_idx, rng)
            if g is not None:
                # simple convex combination – guarantees the new sum stays 1
                a = (1.0 - mix_fraction) * a + mix_fraction * g
                a = np.clip(a, 0.0, None)
                a /= a.sum()
            next_lp = now + lp_interval

        # ----------------------------------------------------------
        # 6. Soft restart when stuck
        # ----------------------------------------------------------
        if now - last_improve > restart_gap:
            # perturb the current best a little and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector found (clipped to the bounds enforced by
    # `evaluate_sequence` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
