"""
Hybrid projected‑Adam optimiser for the non‑negative step‑function problem.

The evaluation metric is   2·n·max(conv(a,a)) / (sum(a)²) .
Since the metric is scale‑invariant we keep the vector on the
probability simplex (ℓ₁‑norm = 1).  At each iteration the true
“max‑convolution’’ index is located (using an FFT‑based convolution)
and the exact sub‑gradient of that entry is taken:
    ∂/∂a_j  c_k = 2·a_{k‑j}   (zero when k‑j is out of range).

An Adam update is performed on this sub‑gradient and the result is
projected back onto the simplex.  To avoid getting stuck in the basin
of the uniform solution we use many random restarts with a variety of
initialisations (uniform, Dirichlet, and sparse equal‑weight supports).
Occasional multiplicative “shakes’’ add stochasticity, and a very
simple simulated‑annealing acceptance rule allows uphill moves early
on.

The routine runs for the allotted time budget (default 1000 s) and
returns the best sequence found.  The final sequence is already
normalised (sum = 1) and satisfies the non‑negativity constraints,
so `evaluate_sequence` will return the same value as the internally
computed objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]                # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _random_initial_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random feasible vector on the simplex.
    Three styles are mixed:
        * uniform
        * Dirichlet with a random concentration α∈[0.1,0.5] (sparse)
        * sparse equal‑weight support (size randomly chosen)
    """
    style = rng.integers(0, 3)
    if style == 0:                      # uniform
        a = np.full(n, 1.0 / n)
    elif style == 1:                    # Dirichlet, random α (sparse)
        alpha = rng.uniform(0.1, 0.5)
        a = rng.dirichlet([alpha] * n)
    else:                               # sparse equal‑weight support
        # choose support size between 20 % and 80 % of n
        k = rng.integers(int(0.2 * n), int(0.8 * n) + 1)
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0 / k
    # ensure exact feasibility
    a = _project_simplex(a)
    return a


# ----------------------------------------------------------------------
def _max_convolution_and_grad(a: np.ndarray):
    """
    Compute max(conv(a,a)) together with its exact sub‑gradient.

    Returns (max_val, k_max, grad) where
        max_val   – the maximal convolution entry,
        k_max     – index of that entry,
        grad      – ∂/∂a max(conv) (shape (n,)).
    The convolution is computed with an FFT for speed.
    """
    n = a.shape[0]
    # FFT length: next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2 * n - 1]

    k_max = int(conv.argmax())
    max_val = float(conv[k_max])

    # gradient of the (ordered‑pair) sum at index k_max:
    g = np.zeros_like(a)
    if k_max < n:
        # indices 0 … k_max contribute
        g[: k_max + 1] = 2.0 * a[k_max::-1]
    else:
        offset = k_max - (n - 1)                # first index that participates
        count = 2 * n - 1 - k_max               # number of terms on the right side
        # slice of length `count` taken from the right‑hand side of `a`
        g[offset:] = 2.0 * a[n - 1 : n - count - 1 : -1]

    return max_val, k_max, g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function will stop a few seconds
        before this limit to guarantee a return).

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # safety margin and absolute deadline
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # problem size – we follow the reference size
    n = 1024

    # ------------------------------------------------------------------
    # initialise from a previously discovered construction if available,
    # otherwise generate a random feasible point.
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.concatenate([a, np.zeros(n - a.size)])
        a = _project_simplex(a)
    else:
        a = _random_initial_a(n, rng)

    # ------------------------------------------------------------------
    # Adam bookkeeping
    m = np.zeros(n)          # first moment
    v = np.zeros(n)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # step‑size schedule (global factor divided by √t)
    base_eta = 0.4

    # bookkeeping of the best solution seen
    max_c, _, _ = _max_convolution_and_grad(a)   # initial max conv
    best = a.copy()
    best_score = 2.0 * n * max_c                 # = evaluate_sequence(best)
    print(f"[init] n={n}, score={best_score:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # exact max‑convolution and its sub‑gradient
        max_c, k_max, grad = _max_convolution_and_grad(a)

        # Adam update (projected onto the simplex afterwards)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        eta = base_eta / np.sqrt(iteration)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # Project back onto the simplex (ensures sum = 1, a≥0)
        a = _project_simplex(a)

        # --------------------------------------------------------------
        # occasional multiplicative shake‑up (helps escape plateaux)
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.18)))
            a[idx] *= factor
            a = _project_simplex(a)

        # --------------------------------------------------------------
        # compute the true objective (fast because sum(a)=1)
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # simple simulated‑annealing acceptance for occasional uphill moves
        # (temperature decays linearly with elapsed time)
        elapsed = time.monotonic() - (deadline - (budget_s - 5.0))
        total = deadline - (deadline - (budget_s - 5.0))
        temperature = max(1e-6, (budget_s - elapsed) / budget_s) * 1e-2
        if cur_score > best_score and rng.random() < np.exp((best_score - cur_score) / temperature):
            # accept a worsening move (already applied)
            pass
        elif cur_score > best_score:
            # reject: revert to previous a (simple rollback)
            a = best.copy()
            m.fill(0.0)
            v.fill(0.0)

        # --------------------------------------------------------------
        # random restart every few ten‑thousand iterations
        if iteration % 20000 == 0:
            a = _random_initial_a(n, rng)
            m.fill(0.0)
            v.fill(0.0)

        # --------------------------------------------------------------
        # early stop if we have already reached the target bound
        if best_score <= 1.5025:
            print("[early stop] target bound reached")
            break

    # ------------------------------------------------------------------
    # final tiny local refinement by tiny mass swaps (helps polishing)
    for _ in range(2000):
        i, j = rng.choice(n, size=2, replace=False)
        # move a tiny amount from i to j
        delta = min(best[i], 0.002 * best[i])
        if delta == 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        max_c, _, _ = _max_convolution_and_grad(cand)
        cand_score = 2.0 * n * max_c
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # optional log
            # print(f"[local improve] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # make sure we stay within the hard limits imposed by evaluate_sequence
    best = np.clip(best, 0.0, 1000.0)

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    return best.tolist()
