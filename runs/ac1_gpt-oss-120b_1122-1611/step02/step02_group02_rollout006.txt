def solve_lp_max_sum(a, M):
    n = len(a)
    # Build constraint matrix A_ub shape (2*n-1, n)
    # Use sp.diags for each diagonal
    # For each offset i in range(n):
    # diag entries = a[i] (repeat n times)
    # However some diagonals exceed shape: For offset i, diag length = n (since 2n-1 - i >= n)
    # So we create a sparse matrix using list of (data, offset).
    data = np.vstack([np.full(n, a[i]) for i in range(n)]) # shape (n, n)
    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    # b_ub all equal M
    b_ub = np.full(2*n-1, M)
    # Bounds
    bounds = [(0, None)] * n
    # Objective: maximize sum -> minimize -sum
    c = -np.ones(n)
    # Solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

"""
Hybrid exponential‑gradient optimizer with a soft‑max weighted gradient.

The score   S(a) = 2·n·max_k (a*a)[k] / (∑a)² is scale‑invariant, therefore we keep
∑a = 1 throughout the search.  A discrete arcsine density (the continuous arcsine
distribution has a perfectly flat self‑convolution) is used as a sensible start.
At each iteration we

1. compute the full self‑convolution by FFT,
2. turn the convolution values into a probability distribution via a temperature‑
   controlled soft‑max (parameter μ) – this yields a smooth approximation of the
   active‑set of constraints,
3. obtain the (sub‑)gradient   w = Σ_k p_k·∂(a*a)[k]/∂a   efficiently as a
   cross‑correlation of the soft‑max weights p and the current a (again by FFT),
4. apply an exponentiated‑gradient update with a slowly decaying step size η,
   together with exponential momentum,
5. occasionally perturb the current point to escape plateaus,
6. keep the best vector seen and restart from a fresh start if no improvement
   occurs for a while.

The temperature μ is increased linearly during the run, so the algorithm first
optimises a smooth surrogate and gradually focuses on the true maximum.
The whole loop runs for the allotted time (≈ 1000 s) and returns the best
non‑negative step‑function found.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real inputs)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _corr_weighted(p: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute w_i = Σ_k p_k * a_{k-i}   (cross‑correlation of p with a)
    via FFT. The result has length len(a).
    """
    # reverse a for cross‑correlation
    a_rev = a[::-1]
    fp = np.fft.rfft(p, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(fp * fa, n=fft_len)
    start = a.size - 1           # index where the length‑n “valid” part begins
    return conv[start : start + a.size]


def _init_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine shape:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    Normalised to sum = 1.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one coordinate to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out
