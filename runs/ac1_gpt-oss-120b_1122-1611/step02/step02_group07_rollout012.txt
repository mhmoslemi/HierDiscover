def _project_simplex(v):
    """project onto simplex sum=1, v >=0"""
    # This implementation similar to previous.
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # return uniform
        return np.full_like(v, 1./n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _differential_evolution(a0, n, rng, time_limit):
    # set pop size
    pop_size = 15
    # initialize population
    pop = np.empty((pop_size, n), dtype=np.float64)
    # first individual = a0
    pop[0] = a0
    # rest random Dirichlet
    for i in range(1, pop_size):
        # sample exponential, then normalize
        exp = rng.exponential(scale=1.0, size=n)
        pop[i] = exp / exp.sum()
    # evaluate
    scores = np.empty(pop_size)
    for i in range(pop_size):
        scores[i] = _score(pop[i])
    best_idx = scores.argmin()
    best = pop[best_idx].copy()
    best_score = scores[best_idx]
    # DE parameters
    F = 0.8
    CR = 0.9
    generation = 0
    while time.time() < time_limit:
        generation += 1
        for i in range(pop_size):
            # choose r1,r2,r3 distinct from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

def _gradient_optimize(a_init, n, rng, deadline):
    a = a_init.copy()
    # adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    base_lr = 0.6
    beta_min=5.0
    beta_max=300.0
    beta_inc_iters=200_000
    best = a.copy()
    best_score = _score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution via FFT
        conv = _full_convolution_fft(a)
        # smooth max
        beta = beta_min + (beta_max - beta_min) * min(1.0, it / beta_inc_iters)
        max_c = conv.max()
        exps = np.exp(beta * (conv - max_c))
        w = exps / exps.sum()
        # gradient via convolution of w with a_rev
        a_rev = a[::-1]
        # convolution length: len(w) + len(a_rev) - 1 = (2n-1)+n-1=3n-2
        Lgrad = len(w) + n - 1
        size_g = 1 << (Lgrad - 1).bit_length()
        fw = np.fft.rfft(w, size_g)
        frev = np.fft.rfft(a_rev, size_g)
        grad_full = np.fft.irfft(fw * frev, size_g)[:Lgrad]
        grad = 2.0 * grad_full[n-1: n-1 + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = base_lr / np.sqrt(it)
        a_next = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # occasional shake
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.12))
            a_next[idx] *= factor
        a_next = _project_simplex(a_next)
        # evaluate approximated score
        cur_score = _score(a_next)
        if cur_score < best_score:
            best_score = cur_score
            best = a_next.copy()
            # print progress
            print(f"[improve] iter={it:,} approx_score={best_score:.6f} beta={beta:.1f}")
        a = a_next
        # early exit if reaching target approx
        if best_score <= 1.5025:
            print("[target] approximated target reached during gradient")
            break
    return best, best_score

"""
Hybrid stochastic optimizer for the step‑function bound.

The algorithm works in two phases:

1. **Differential Evolution (global search)** – a small population of
   probability vectors (entries ≥ 0, sum = 1) is evolved on the simplex.
   Evolution uses mutation, binomial crossover and a projection onto the
   simplex.  The fitness is the cheap proxy
   ``score = 2·n·max(a*a)`` which is computed with an FFT‑based convolution.

2. **Adam refinement (local search)** – the best individual from the
   DE phase is fine‑tuned with an Adam optimiser on a smooth (log‑sum‑exp)
   approximation of the max‑convolution.  Gradients are obtained via
   FFT‑based convolutions as well and the iterate is kept on the simplex
   using the same projection routine.  Periodic multiplicative “shake‑up”
   steps help escape shallow minima.

The final vector is returned as a plain Python list; it is also stored in
the global variable ``height_sequence_1`` for possible warm‑starts in
later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive – uniform fallback
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Return the full linear convolution a * a using an FFT.
    The result length is 2·len(a)‑1.
    """
    n = a.shape[0]
    L = 2 * n - 1
    size = 1 << (L - 1).bit_length()          # next power of two ≥ L
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:L]
    return conv


# ----------------------------------------------------------------------
def _score_approx(a: np.ndarray) -> float:
    """
    Fast proxy for ``evaluate_sequence`` when sum(a)=1.
    Returns 2·n·max(a*a) (the denominator is 1 after the simplex projection).
    """
    conv = _full_convolution_fft(a)
    return 2.0 * a.shape[0] * np.max(conv)


# ----------------------------------------------------------------------
def _differential_evolution(a0: np.ndarray,
                            n: int,
                            rng: np.random.Generator,
                            deadline: float):
    """
    Very small differential‑evolution run on the simplex.
    Returns the best individual found and its approximate score.
    """
    pop_size = 15
    pop = np.empty((pop_size, n), dtype=np.float64)

    # initialise ---------------------------------------------------------
    pop[0] = a0.copy()
    for i in range(1, pop_size):
        exp = rng.exponential(scale=1.0, size=n)
        pop[i] = exp / exp.sum()

    scores = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        scores[i] = _score_approx(pop[i])

    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    F = 0.8                     # differential weight
    CR = 0.9                    # crossover probability

    while time.time() < deadline:
        for i in range(pop_size):
            # three distinct donors, none equal to i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

            mutant = pop[r1] + F * (pop[r2] - pop[r3])

            # binomial crossover
            cross = rng.random(n) < CR
            if not np.any(cross):                     # guarantee at least one term
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, pop[i])

            # keep on the simplex
            trial = _project_simplex(trial)

            trial_score = _score_approx(trial)

            # selection
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = float(trial_score)
                    best = trial.copy()

        # early stop if we already beat the target
        if best_score <= 1.5025:
            print("[DE] target reached, stopping early")
            break

    return best, best_score

def _adam_refine(a_init: np.ndarray,
                 n: int,
                 rng: np.random.Generator,
                 deadline: float):
    """
    Adam optimiser on a smooth (log‑sum‑exp) approximation of the max‑convolution.
    Uses FFT‑based convolutions for speed.
    Returns the best vector seen and its approximate score.
    """
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    base_lr = 0.6
    beta_min = 5.0
    beta_max = 300.0          # sharper than the reference implementation
    beta_inc_iters = 200_000

    best = a.copy()
    best_score = _score_approx(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # full autocorrelation (FFT)
        conv = _full_convolution_fft(a)

        # smooth‑max (log‑sum‑exp)
        beta = beta_min + (beta_max - beta_min) * min(1.0, it / beta_inc_iters)
        mx = conv.max()
        exps = np.exp(beta * (conv - mx))
        w = exps / exps.sum()                     # soft‑max weights, length 2n‑1

        # gradient = 2 * (w * a_rev)  (convolution)
        a_rev = a[::-1]
        Lgrad = w.shape[0] + n - 1                # = 3n‑2
        size_g = 1 << (Lgrad - 1).bit_length()
        fw = np.fft.rfft(w, size_g)
        frev = np.fft.rfft(a_rev, size_g)
        grad_full = np.fft.irfft(fw * frev, size_g)[:Lgrad]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]  # middle n entries

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / np.sqrt(it)
        a_next = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional multiplicative shake‑up
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.12)))   # ≈ ±12 %
            a_next[idx] *= factor

        # back onto the simplex
        a_next = _project_simplex(a_next)

        # track progress
        cur_score = _score_approx(a_next)
        if cur_score < best_score:
            best_score = float(cur_score)
            best = a_next.copy()
            print(f"[improve] iter={it:,} approx_score={best_score:.6f} beta={beta:.1f}")

        a = a_next

        if best_score <= 1.5025:
            print("[Adam] target reached, stopping early")
            break

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a step‑function height vector that minimises ``evaluate_sequence``.

    The method first runs a lightweight differential‑evolution global search,
    then refines the best candidate with an Adam optimiser that works on a
    smooth approximation of the max‑convolution (log‑sum‑exp).  FFT‑based
    convolutions keep the per‑iteration cost low even for n=1024.

    Returns
    -------
    list[float]
        The best height sequence found (non‑negative, sum≈1).
    """
    rng = np.random.default_rng(seed)

    # reserve a small safety margin to guarantee an answer before timeout
    deadline = time.time() + max(0.0, float(budget_s) - 5.0)

    # problem dimension ----------------------------------------------------
    n = 1024

    # warm‑start ------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a0.size != n:
                raise ValueError
        except Exception:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a0 = _project_simplex(a0)

    # ---------------------------------------------------------------------
    # Phase 1 – global exploration via differential evolution
    de_deadline = time.time() + (deadline - time.time()) * 0.30   # ~30 % of remaining time
    best, best_score = _differential_evolution(a0, n, rng, de_deadline)
    print(f"[DE] best approximate score after global search = {best_score:.6f}")

    # ---------------------------------------------------------------------
    # Phase 2 – local refinement with Adam on smooth‑max
    best, best_score = _adam_refine(best, n, rng, deadline)
    print(f"[final approx] best approximate score after refinement = {best_score:.6f}")

    # ---------------------------------------------------------------------
    # Exact evaluation (the only place we call the provided function)
    final_seq = best.tolist()
    final_score = float(evaluate_sequence(final_seq))
    print(f"[final] exact evaluate_sequence score = {final_score:.6f}")

    # store for possible warm‑start in a future run
    globals()["height_sequence_1"] = final_seq

    return final_seq
