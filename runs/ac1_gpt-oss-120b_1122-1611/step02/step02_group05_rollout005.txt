def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("expect 1‑D vector")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0

    if not np.any(cond):

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            a_tmp = a.copy()
            a_tmp[idx] = max(0.0, a_tmp[idx] + delta)
            a_tmp = _project_to_simplex(a_tmp)
            a = a_tmp
            t = max(min_t, t * 0.5)
            continue

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            # degenerate solution – random perturb instead
            idx = rng.integers(0, n)
            delta = rng.normal(0.0, 0.01)
            a_tmp = a.copy()
            a_tmp[idx] = max(0.0, a_tmp[idx] + delta)
            a_tmp = _project_to_simplex(a_tmp)
            a = a_tmp
            t = max(min_t, t * 0.5)
            continue

        # --------------------------------------------------------------
        # normalise LP solution to a probability vector
        g = b / sum_b

        # --------------------------------------------------------------
        # form a convex combination and evaluate
        a_candidate = (1.0 - t) * a + t * g
        # (the combination automatically keeps the ℓ₁‑norm = 1)

        score_candidate = float(evaluate_sequence(a_candidate.tolist()))

        if np.isfinite(score_candidate) and score_candidate < best_score - 1e-9:
            # improvement – accept
            best_score = score_candidate
            best = a_candidate.copy()
            a = a_candidate
            print(f"[improve] iter={iteration:,}, t={t:.4f}, score={best_score:.6f}")
            # slowly increase the mixing weight, but keep it bounded
            t = min(max_t, t * 1.1)
            no_improve = 0
        else:
            # no improvement – shrink the step size
            t = max(min_t, t * 0.5)
            no_improve += 1

        # occasional random perturbation to escape flat regions
        if no_improve >= 30 and rng.random() < 0.3:
            idx = rng.integers(0, n)
            delta = rng.normal(0.0, 0.02)
            a_tmp = a.copy()
            a_tmp[idx] = max(0.0, a_tmp[idx] + delta)
            a_tmp = _project_to_simplex(a_tmp)
            tmp_score = float(evaluate_sequence(a_tmp.tolist()))
            if np.isfinite(tmp_score) and tmp_score < best_score:
                best_score = tmp_score
                best = a_tmp.copy()
                a = a_tmp
                print(f"[random] iter={iteration:,}, score={best_score:.6f}")
                no_improve = 0
                t = max(min_t, t * 0.8)

        # safety break if we are very close to the deadline
        if iteration % 10 == 0 and time.monotonic() > deadline - 1.0:
            break

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    # clip to the allowed range before returning
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
