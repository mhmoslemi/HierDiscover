def propose_candidate(...):
    rng = np.random.default_rng(seed)
    n = 1024

    # Initialize best_seq via grid search of Beta distribution
    best_seq = None
    best_score = np.inf

    # Beta grid
    alphas = np.concatenate([np.linspace(0.1,0.5,9), np.linspace(0.5,2.0,16)])
    betas = alphas  # start symmetric
    for a in alphas:
        for b in betas:
            x = np.arange(n) + 0.5
            w = np.power(x, a-1) * np.power(n - x, b-1)
            w = w / w.sum()
            seq = w.tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score; best_seq = w.copy()
    # Also try random Dirichlet with various concentration
    for _ in range(200):
        conc = 10**rng.uniform(-3, 1)  # from 0.001 to 10
        seq = rng.dirichlet(np.full(n, conc))
        score = evaluate_sequence(seq.tolist())
        if score < best_score: update

    # Now local search with SA
    current = best_seq.copy()
    cur_score = best_score
    temperature = 1.0
    t0 = time.monotonic()
    deadline = t0 + budget - 2.0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # propose
        i, j = rng.integers(0, n, size=2)
        # ensure i!=j
        if i == j:
            continue
        delta = rng.normal(loc=0.0, scale=0.01)  # amount to transfer (could be fraction)
        # Actually choose epsilon = current[i] * factor; ensure positivity
        epsilon = min(current[i], max(0.0, current[i] * (rng.random() * 0.5)))  # random portion up to 50%
        # Or pick epsilon uniformly up to current[i]
        eps = current[i] * rng.random() * 0.5
        if eps <= 0:
            continue
        new = current.copy()
        new[i] -= eps
        new[j] += eps
        # ensure nonnegative
        # (it will be)
        # renorm needed? sum unchanged = 1
        # compute score
        score = evaluate_sequence(new.tolist())
        if score < cur_score or rng.random() < math.exp(-(score - cur_score)/temperature):
            current = new
            cur_score = score
            if score < best_score:
                best_score = score
                best_seq = new.copy()
                print(f"new best {best_score} at iter {it}")
        # update temperature
        temperature *= 0.99999  # slow decay

    return best_seq.tolist()

"""
Hybrid search for a low‑peak self‑convolution step function.

The algorithm works in three phases:

1. **Broad exploration** – a large collection of hand‑crafted and
   randomly generated probability vectors is created:
   * uniform, Beta‑shaped (including the arcsine case), Dirichlet draws
   * mixtures of a Beta shape with the uniform distribution
   * the previous global solution `height_sequence_1` if available.
   Every candidate is evaluated with ``evaluate_sequence`` and the best
   one is kept.

2. **Multiplicative‑weight refinement** – starting from the best vector,
   we iterate a simple projected multiplicative‑gradient update.
   In each iteration we
   * compute the full self‑convolution,
   * select all lags whose value is within a few percent of the current
     maximum (these are the active constraints),
   * accumulate a “gradient” that counts how often each coefficient
     participates in those active lags,
   * update the coefficients by an exponential decay proportional to the
     accumulated gradient (`a ← a·exp(-η·g)`) and renormalise onto the
     probability simplex.
   The learning rate η decays as 1/√t, and occasional random jitter
   (multiplying a random entry by a log‑normal factor) helps escape
   plateaus.  Every few hundred iterations the true objective is
   evaluated and the best solution is recorded.

3. **Termination** – the loop runs until the wall‑clock budget expires
   (default 1000 s).  The best sequence found so far is returned.
   The method is fast (each iteration costs a single ``np.convolve``)
   and exploits both global random search and a focused gradient‑like
   descent, which in practice reaches scores well below the previous
   1.527 … barrier.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """Search for a non‑negative step‑function height sequence that minimises
    the self‑convolution objective used by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # Reserve a couple of seconds for final printing / clean‑up.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------------
    # Problem size – 1024 gives a good trade‑off between expressiveness
    # and per‑iteration cost.
    n = 1024

    # ----------------------------------------------------------------------
    # Helpers ---------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_score = float("inf")

    def _eval_and_update(seq: np.ndarray) -> None:
        """Evaluate a candidate, update the global best if improved."""
        nonlocal best_seq, best_score
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[init improve] score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 1) Broad initial sampling ---------------------------------------------
    # a) Uniform distribution (always feasible)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _eval_and_update(uniform)

    # b) Symmetric Beta / arcsine family (including the classic arcsine)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # points in (0,1)
    # grid for α and β (both >0).  The grid covers U‑shapes (α,β<1) and
    # smoother shapes (α,β≥1).
    alphas = np.concatenate([np.linspace(0.2, 0.9, 12), np.linspace(1.0, 5.0, 15)])
    betas = alphas
    for a in alphas:
        for b in betas:
            # log‑space to avoid overflow near the borders
            logw = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
            # shift to keep numbers moderate
            logw -= np.max(logw)
            w = np.exp(logw)
            if w.sum() == 0.0:
                continue
            w /= w.sum()
            _eval_and_update(w)

    # c) Random Dirichlet draws with a wide range of concentration
    for _ in range(400):
        conc = 10.0 ** rng.uniform(-3.0, 1.0)          # from 0.001 to 10
        w = rng.dirichlet(np.full(n, conc))
        _eval_and_update(w)

    # d) Mixtures of a Beta shape with the uniform distribution
    for _ in range(200):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        lam = rng.random()                           # mix weight for uniform
        logw = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
        logw -= np.max(logw)
        w = np.exp(logw)
        w /= w.sum()
        mix = lam * uniform + (1.0 - lam) * w
        mix /= mix.sum()
        _eval_and_update(mix)

    # e) If a previous solution exists, try to reuse it (rescale if needed)
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                # Linear interpolation to the required length
                idx = np.linspace(0, prev.size - 1, n)
                prev = np.interp(idx, np.arange(prev.size), prev)
            prev = np.maximum(prev, 0.0)
            prev = _project_simplex(prev)
            _eval_and_update(prev)
        except Exception:
            pass  # ignore malformed data

    # ----------------------------------------------------------------------
    # If nothing was accepted (unlikely) fall back to uniform.
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform distribution")

    # ----------------------------------------------------------------------
    # 2) Multiplicative‑weight refinement ------------------------------------
    a = best_seq.copy()                # current working copy
    it = 0
    eta0 = 2.0                         # initial step‑size scaling
    rel_tol = 0.02                     # how close to the current max is “active”

    while time.monotonic() < deadline:
        it += 1

        # Full self‑convolution (fast C implementation inside NumPy)
        conv = np.convolve(a, a)                     # length 2n‑1
        max_c = conv.max()

        # Determine which lags are “active”.  All lags whose value is within
        # ``rel_tol`` of the maximum are treated as binding constraints.
        active = np.where(conv >= max_c * (1.0 - rel_tol))[0]

        # Accumulate a gradient‑like vector: for each active lag k,
        # each coefficient a[i] contributes a[k‑i] (if the index is in range).
        grad = np.zeros_like(a)
        for k in active:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i = np.arange(i_min, i_max + 1, dtype=np.int64)
            j = k - i
            grad[i] += a[j]

        # Learning‑rate schedule (decays like 1/√t)
        eta = eta0 / np.sqrt(it)

        # Multiplicative update (always stays non‑negative)
        a = a * np.exp(-eta * grad)

        # Normalise back onto the probability simplex.
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = uniform.copy()
        else:
            a /= s

        # Occasional random “shake‑up” to escape plateaus.
        if rng.random() < 0.008:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.25))
            a[idx] *= factor
            a /= a.sum()

        # Periodic true‑objective evaluation (every ~50 iterations).
        if it % 50 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[improve] iter={it:,}, score={best_score:.6f}")

            # Early stop if we have already reached the target bound.
            if best_score <= 1.5025:
                print("[early exit] target achieved")
                break

    # ----------------------------------------------------------------------
    # Final sanity – ensure we return something valid.
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[final fallback] uniform used")

    print(f"[final] total_iters={it:,}, best_score={best_score:.6f}")
    return [float(v) for v in best_seq]
