"""
Hybrid linear‑programming / convex‑combination optimizer.

The code starts from a (down‑sampled) version of the best known
construction `height_sequence_1`.  In each iteration it solves a linear
program

    maximise   1ᵀg
    subject to  f * g ≤ M  (component‑wise)
                g ≥ 0

where `f` is the current step‑function, `M = max(f*f)` and `*` denotes
linear convolution.  The LP tells us how much extra L¹‑mass we can
add without raising the current worst convolution peak.  After the LP
solution is scaled back to unit sum we blend it with the current
vector (the blend factor is chosen by a cheap line‑search).  This
strictly reduces the peak `M` and therefore the score
`2·n·M`.  Random block‑perturbations are injected when progress stalls.

The algorithm works with a moderate length (default 600) so that the
LP (≈ 1200 constraints × 600 variables) is fast, yet the objective
`2·n·max(conv)` is length‑independent and can approach the theoretical
lower bound 1.  The routine respects the security checks of
`evaluate_sequence` and returns the best vector found before the time
budget expires.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def _downsample_seq(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Down‑sample `seq` to length ``target_n`` by summing adjacent bins."""
    old_n = seq.size
    if old_n == target_n:
        return seq.copy()
    # map each old index to a new bin
    bins = np.floor(np.arange(old_n) * target_n / old_n).astype(int)
    bins = np.clip(bins, 0, target_n - 1)
    new = np.zeros(target_n, dtype=seq.dtype)
    np.add.at(new, bins, seq)
    return new


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix ``A`` that satisfies  ``A @ g == np.convolve(f,g)``.
    `f` must be non‑negative.
    """
    n = f.size
    # first column: f padded with zeros to length 2n‑1
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row: f[0] followed by zeros (size n)
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return toeplitz(c, r)  # shape (2n‑1, n)


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise sum(g)   subject to   f*g ≤ M , g ≥ 0.
    Returns the optimal `g` (may be None if the solver fails).
    """
    n = f.size
    A = _build_conv_matrix(f)               # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=f.dtype)   # RHS of inequalities
    c = -np.ones(n, dtype=f.dtype)           # linprog minimises, so -sum(g)
    bounds = [(0.0, None)] * n

    # Use HiGHS – the default method in recent SciPy versions.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Run the LP‑based optimizer for roughly `budget_s` seconds
    (a 5 s safety margin is kept).  Returns a list of non‑negative
    heights that achieve a low score under `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1. Initialise a non‑negative, unit‑sum vector.
    # --------------------------------------------------------------
    target_n = 600                     # moderate size – fast LP
    if "height_sequence_1" in globals():

            else:
                init = None
        except Exception:
            init = None
    else:
        init = None

    if init is None:

        init = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    init = np.clip(init, 0.0, None)
    s = init.sum()
    if s == 0.0:

    n = f.size

                best_alpha = a

            continue

        # ------------------------------------------------------------------

        f = np.clip(f, 0.0, None)
        f_sum = f.sum()
        if f_sum == 0.0:
            f[:] = 1.0 / n
        else:
            f /= f_sum

        cur_score = float(2.0 * n * best_Mnew)

            best_score = cur_score
            best = f.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 6. If we have been idle for a while, perform a soft restart.
        # ------------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > 45.0:

            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            f *= noise
            f = np.clip(f, 0.0, None)
            f_sum = f.sum()
            if f_sum == 0.0:

            else:
                f /= f_sum
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – re‑starting from best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best vector found (clipped to the bounds used by
    # `evaluate_sequence` for safety).  The function expects a plain
    # Python list of floats.
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
