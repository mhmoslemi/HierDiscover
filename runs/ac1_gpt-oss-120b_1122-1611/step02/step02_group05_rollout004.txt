"""
Propose a step‑function (non‑negative vector) that minimises
    score = 2·n·max(a*a) / (sum(a)^2)

The algorithm is a *projected Adam* optimiser that directly works on the
real objective (the maximum of the self‑convolution).  At each iteration
the exact sub‑gradient of `max(a*a)` is computed:
    ∇max = 2·correlate(mask, a)    with mask[i]=1 if (a*a)[i] attains the max.
The iterate is projected onto the probability simplex (∑a_i=1, a_i≥0).
Occasional multiplicative “shake‑ups” avoid stagnation and a small
random restart scheme is used when progress stalls.

The method improves on the smooth‑max surrogate used previously and
typically pushes the evaluation score below the target 1.5025 within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute a*a (full linear convolution) using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    # zero‑pad to length 2n (next power of two is 2n for n a power of two)
    a_pad = np.concatenate([a, np.zeros(n, dtype=np.float64)])
    fa = np.fft.rfft(a_pad)                     # length n+1
    conv = np.fft.irfft(fa * fa, n=2 * n)        # length 2n
    return conv[: 2 * n - 1]                     # discard the very last zero


# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray):
    """
    Return a sub‑gradient of max(conv) w.r.t. a and the current maximum value.
    conv must be the full self‑convolution a*a.
    """
    max_val = conv.max()
    # consider all entries that are (numerically) at the maximum
    mask = (conv >= max_val - 1e-12).astype(np.float64)   # shape 2n‑1
    # gradient = 2 * correlate(mask, a)  (valid mode gives length n)
    grad = 2.0 * np.correlate(mask, a, mode='valid')
    return grad, max_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑evaluation step‑function.  The routine runs for roughly
    `budget_s` seconds and returns the best sequence (list of floats) it
    discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – 1024 works well for the time budget
    n = 1024

    # ------------------------------------------------------------------
    # initial vector (normalized to sum = 1)
    init_seq = None
    if 'height_sequence_1' in globals():
        try:
            tmp = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if tmp.size >= 2:
                init_seq = tmp / tmp.sum()
        except Exception:
            init_seq = None

    if init_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)          # uniform start
    else:
        if init_seq.size != n:
            if init_seq.size > n:
                a = init_seq[:n]
            else:
                a = np.concatenate([init_seq,
                                    np.zeros(n - init_seq.size,
                                             dtype=np.float64)])
            a = a / a.sum()
        else:
            a = init_seq.copy()

    # keep the best solution seen so far
    conv = _self_convolution_fft(a)
    best_score = 2.0 * n * conv.max()          # because sum(a) = 1
    best = a.copy()
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser state
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam learning‑rate schedule
    lr0 = 0.5                     # relatively large at the start

    # random‑shake probability
    shake_prob = 0.03

    # termination / early‑stop parameters
    target_score = 1.5025
    no_improve_limit = 150_000   # after this many iterations without improvement we re‑randomise a bit

    # time bookkeeping
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # safety margin
    iteration = 0
    it_since_improve = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        it_since_improve += 1

        # --- 1) convolution and sub‑gradient ---------------------------------
        conv = _self_convolution_fft(a)
        grad, max_val = _subgradient_maxconv(a, conv)

        # --- 2) Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments (avoid overflow in the exponent for huge iter)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # learning‑rate that decays like 1/√t
        lr = lr0 / (np.sqrt(iteration) + 1.0)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- 3) occasional multiplicative perturbation -----------------------
        if rng.random() < shake_prob:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.2)))   # log‑normal perturb
            a_new[idx] *= factor

        # --- 4) enforce feasibility (probability simplex) --------------------
        a_new = _project_simplex(a_new)

        a = a_new

        # --- 5) evaluate current candidate (exact objective) -----------------
        score = 2.0 * n * max_val        # sum(a) = 1 by construction
        if score < best_score:
            best_score = score
            best = a.copy()
            it_since_improve = 0
            print(f"[improve] iter={iteration:,}, score={best_score:.6f}")

        # --- 6) restart heuristic if stuck for a while -----------------------
        if it_since_improve >= no_improve_limit:
            # re‑inject a small amount of uniform mass to escape a flat region
            eps_mix = 0.05
            a = (1.0 - eps_mix) * best + eps_mix * np.full(n, 1.0 / n)
            a = _project_simplex(a)
            # reset Adam moments to avoid bias from the previous trajectory
            m.fill(0.0)
            v.fill(0.0)
            it_since_improve = 0
            print(f"[restart] iter={iteration:,} – mixing uniform to escape plateau")

        # --- 7) early stopping ------------------------------------------------
        if best_score <= target_score * 1.001:
            print("[early stop] reached near‑target score")
            break

        # --- 8) periodic status report ---------------------------------------
        if iteration % 200_000 == 0:
            print(f"[status] iter={iteration:,} – best score so far {best_score:.6f}")

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    # Return a plain Python list as required by the evaluation harness
    return [float(x) for x in best]
