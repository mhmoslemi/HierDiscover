def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # determine n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 4096  # fallback (maybe 2048)
    # init a
    a0 = _init_arcsine(n)
    # jitter
    a0 += rng.normal(scale=0.001, size=n)
    a0 = np.clip(a0, 0, None)
    a0 = a0 / a0.sum()
    # log transform
    z = np.log(a0 + 1e-12)

    # precompute lengths
    fft_len_self = _fft_len_for_n(n) # pow2 >=2n-1
    L = 2*n-1
    fft_len_grad = 1 << ((3*n-2 - 1).bit_length())
    # parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t_iter = 0
    alpha0 = 0.1   # base learning rate
    # gamma schedule
    gamma_start = 5.0
    gamma_end = 200.0

    best_a = a0.copy()
    best_score = _score_of_a(best_a, n, fft_len_self)
    last_improve = time.monotonic()
    stagnation_limit = 25.0

    start_time = time.monotonic()
    total_budget = deadline - start_time

    while time.monotonic() < deadline:
        t_iter += 1
        # compute a = softmax(z)
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()
        # self-convolution
        conv = _conv_self_fft(a, fft_len_self)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        # record best
        if score < best_score:
            best_score = score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] best score {best_score:.6f}")
        # compute gamma
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_budget
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        # compute softmax weights over conv
        # using gamma*conv, but subtract max for stability
        max_gc = np.max(gamma * conv)
        weight_exp = np.exp(gamma * conv - max_gc)
        w = weight_exp / weight_exp.sum()
        # gradient using convolution with w_rev
        w_rev = w[::-1]
        conv_grad = _linear_convolve(a, w_rev, fft_len_grad)  # length = n+L-1
        # extract d_i
        slice_start = L - n
        slice_end = L
        d = 2.0 * conv_grad[slice_start:slice_end][::-1]
        # compute grad_z
        dot = np.dot(d, a)
        grad_z = a * (d - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        # step size decay
        eta = alpha0 * (1.0 - frac)  # linear decay to zero
        z -= eta * m_hat / (np.sqrt(v_hat) + eps)
        # occasional jitter
        if rng.random() < 0.005:  # 0.5%

"""
Smooth‑max gradient descent on the simplex (Adam + annealed soft‑max).

The score to minimise is  

    S(a) = 2·n·max_k (a∗a)[k] / (∑a)^2 ,

and it is invariant under scaling.  We therefore keep the vector on the
probability simplex (∑a = 1) at all times.  The variables are
re‑parameterised as a = softmax(z); this guarantees non‑negativity and
normalisation without any extra projection steps.

A smoothed version of the max (log‑sum‑exp) is used:

    Φγ(a) = (1/γ)·log( Σ_k exp(γ·(a∗a)[k]) ).

When γ → ∞, Φγ → max_k (a∗a)[k].  Starting with a small γ and
increasing it linearly gives a smooth landscape at first and then
focuses on the true peak later.  The gradient of Φγ w.r.t. a is

    ∇_a Φγ = 2· Σ_k w_k·p_k   ,   w_k = softmax_k(γ·(a∗a)[k])
    p_k[i] = a_{k‑i} (zero outside the support)

The sum of weighted “partner” vectors p_k can be obtained with a single
FFT‑based convolution (a with the reversed weight vector).  Adam is
used to update the unconstrained vector z; occasional random jitter
helps escape plateaus and a simple stagnation‑restart mechanism explores
different basins.

The routine runs for the allotted time (≈ 1000 s) and returns the best
sequence seen so far.  All values are clipped to the limits required
by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper for the next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a (real, non‑negative) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _linear_convolve(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗b using real FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Initial shapes
# ----------------------------------------------------------------------


def _init_arcsine(n: int) -> np.ndarray:
    """
    Discrete approximation of the arcsine density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    The shape yields a near‑constant self‑convolution and is a strong
    starting point for further optimisation.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Scoring (fast version – identical to evaluate_sequence for normalised sum)
# ----------------------------------------------------------------------


def _score_of_a(a: np.ndarray, n: int, fft_len_self: int) -> float:
    conv = _conv_self_fft(a, fft_len_self)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds (leaving a small safety margin) and uses a
    smooth‑max Adam optimiser on the simplex.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        n = 4096          # safe fallback size
    # --------------------------------------------------------------
    # Initialise a on the simplex (arcsine + jitter)
    # --------------------------------------------------------------
    a0 = _init_arcsine(n)
    a0 += rng.normal(scale=0.001, size=n)       # break perfect symmetry
    a0 = np.clip(a0, 0.0, None)
    a0 /= a0.sum()
    # log‑parameterisation: a = softmax(z)
    z = np.log(a0 + 1e-12)

    # --------------------------------------------------------------
    # Pre‑computations for FFT lengths
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)                # for a∗a
    L = 2 * n - 1                                      # length of a∗a
    fft_len_grad = _next_pow2(3 * n - 2)               # for a∗w_rev

    # --------------------------------------------------------------
    # Adam state
    # --------------------------------------------------------------
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.15               # initial learning‑rate for z

    # --------------------------------------------------------------
    # Soft‑max (γ) schedule
    # --------------------------------------------------------------
    gamma_start = 5.0
    gamma_end = 200.0

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_score = _score_of_a(best_a, n, fft_len_self)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # safety margin
    last_improve = start_time
    stagnation_limit = 25.0          # seconds without improvement → restart
    it = 0

    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---------- a = softmax(z) ----------
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()                     # always ∑a = 1

        # ---------- self‑convolution ----------
        conv = _conv_self_fft(a, fft_len_self)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best = {best_score:.6f}")

        # ---------- annealed soft‑max over convolution ----------
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = elapsed / total
        gamma = gamma_start + (gamma_end - gamma_start) * frac

        # soft‑max of the scaled convolution (stable)
        scaled = gamma * conv
        w = _softmax(scaled)          # w_k ≥ 0, Σ w_k = 1

        # ---------- gradient of the smoothed max ----------
        # partner vectors summed with weight w can be obtained via a single FFT
        w_rev = w[::-1]                               # reverse order
        conv_grad = _linear_convolve(a, w_rev, fft_len_grad)   # length = n+L-1
        # extract d_i = 2· Σ_k w_k·a_{k-i}
        slice_start = L - n
        slice_end = L
        d = 2.0 * conv_grad[slice_start:slice_end][::-1]      # length n

        # ---------- gradient w.r.t. the softmax logits ----------
        dot = np.dot(d, a)
        grad_z = a * (d - dot)               # Jᵀ·d where J = diag(a)-a aᵀ

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple linear decay of the base step size
        lr = step0 * (1.0 - frac)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- occasional random jitter ----------
        if rng.random() < 0.008:                     # ~0.8 % of iterations
            # add a tiny Gaussian perturbation to a random sub‑block of z
            blk_start = rng.integers(0, n)
            blk_len = rng.integers(1, max(2, n // 100))
            blk_end = min(n, blk_start + blk_len)
            z[blk_start:blk_end] += rng.normal(scale=0.05, size=blk_end - blk_start)

        # ---------- stagnation check ----------
        if time.monotonic() - last_improve > stagnation_limit:
            # restart with a fresh point (mostly arcsine, sometimes random)
            if rng.random() < 0.6:
                a_restart = _init_arcsine(n)
            else:
                a_restart = rng.gamma(shape=1.0, size=n)
            a_restart = np.clip(a_restart, 0.0, None)
            a_restart /= a_restart.sum()
            a_restart += rng.normal(scale=0.001, size=n)
            a_restart = np.clip(a_restart, 0.0, None)
            a_restart /= a_restart.sum()
            z = np.log(a_restart + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart (stagnation)")

    # --------------------------------------------------------------
    # Return the best sequence found (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
