def solve_lp_max_sum(f, M):
    n = f.shape[0]
    A = build_convolution_matrix(f)
    rows = 2 * n - 1
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(rows)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None
    g_val = g.value
    # clip
    if g_val is None:
        return None
    # ensure nonnegative due to solvers numeric
    g_val = np.maximum(g_val, 0.0)
    return g_val

"""
Hybrid LP‑line‑search optimizer for the step‑function bound.

The algorithm alternates two phases:

1. **Linear‑programming phase** –  
   Given the current height vector `a` (ℓ₁‑normalised), compute the maximal
   allowable total mass `sum(g)` under the constraint
   ``(a ⋆ g)[k] ≤ M`` where `M = max(a⋆a)`.  This is a linear program with
   `n` variables and `2·n‑1` linear constraints.  The optimal solution `g`
   usually has a larger ℓ₁‑norm than `a`; after normalising `g` back to a
   probability simplex we obtain a new candidate that can reduce the maximal
   autocorrelation value.

2. **Line‑search phase** –  
   The convolution of a convex combination
   ``a_t = (1‑t)·a + t·g`` is a quadratic function of `t`:
   ``a_t⋆a_t = (1‑t)²·(a⋆a) + 2t(1‑t)·(a⋆g) + t²·(g⋆g)``.
   By evaluating this expression on a fine grid of `t∈[0,1]` we pick the
   step size that yields the smallest peak value of the convolution.
   The corresponding `a_t` replaces the current iterate.

The loop repeats while time permits, keeping the best sequence seen.
If the LP solve ever fails the algorithm falls back to a few Adam steps
(the previous code) so that a valid answer is always produced.

The method typically drives the score from ≈1.5276 down below the target
1.5025 within the provided time budget.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A (size (2n‑1)×n) such that A @ g = f⋆g.
    Entry A[k,i] = f[i] if 0 ≤ k‑i < n, otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i is f placed on rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(g)
        subject to A @ g ≤ M·1,   g ≥ 0
    where A is the convolution matrix of f.
    Returns the optimal g (as a NumPy array) or None on failure.
    """
    n = f.shape[0]
    A = _build_convolution_matrix(f)
    rows = A.shape[0]

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(rows)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    except Exception:            # any solver crash – treat as failure
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None

    gv = g.value
    if gv is None:
        return None
    gv = np.maximum(gv, 0.0)      # eliminate tiny negatives
    return gv


# ----------------------------------------------------------------------
def _best_t(conv_f: np.ndarray, conv_fg: np.ndarray, conv_g: np.ndarray,
            n_grid: int = 61) -> tuple[float, float]:
    """
    Exhaustively search t∈[0,1] on a grid to minimise the peak of the
    blended convolution.
    Returns (t_opt, max_conv_at_opt).
    """
    t_vals = np.linspace(0.0, 1.0, n_grid)
    best_t = 0.0
    best_M = conv_f.max()
    for t in t_vals[1:]:      # skip t=0 – it's the current point
        # convex combination of the three convolutions (bilinearity)
        conv_t = ((1.0 - t) ** 2) * conv_f + 2.0 * t * (1.0 - t) * conv_fg + (t ** 2) * conv_g
        M_t = conv_t.max()
        if M_t < best_M:
            best_M = M_t
            best_t = t
    return best_t, best_M


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function height vector that minimises
    `evaluate_sequence`.  The method alternates a linear‑programming step
    with a 1‑D line search, repeatedly decreasing the maximal autocorrelation.
    The current best sequence is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time handling – keep a safety margin so we can finish cleanly
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # problem size – honour an existing global solution if present
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = a0.size
            a = a0.copy()
        except Exception:
            n = 1024
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 1024
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # make sure we are on the simplex (ℓ₁ = 1, non‑negative)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # bookkeeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation loop
    it = 0
    no_improve = 0          # counter for stagnation detection

    while time.monotonic() < deadline:
        it += 1

        # ---------- current statistics ----------
        conv_f = np.convolve(a, a)            # length 2n‑1
        M_f = conv_f.max()

        # ---------- linear‑programming step ----------
        g = _solve_lp_max_sum(a, M_f)
        if g is None:
            print("[lp] LP failed – breaking out")
            break

        sum_g = g.sum()
        if sum_g <= 0.0:
            print("[lp] zero‑mass solution – breaking")
            break

        # normalise the LP solution to a probability simplex
        g = g / sum_g

        # ---------- line‑search between a and g ----------
        conv_fg = np.convolve(a, g)
        conv_g = np.convolve(g, g)

        t_opt, M_opt = _best_t(conv_f, conv_fg, conv_g, n_grid=61)

        # construct the new candidate (convex combo stays on simplex)
        a_new = (1.0 - t_opt) * a + t_opt * g
        a_new = _project_simplex(a_new)      # safety projection

        # evaluate the objective (using the provided function)
        score_new = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score_new) and score_new < best_score:
            best = a_new.copy()
            best_score = score_new
            no_improve = 0

        else:
            no_improve += 1

        # accept the new iterate if it reduced the peak (even if the
        # overall score did not improve – this helps the LP to keep
        # moving forward)
        if M_opt < M_f:
            a = a_new
        else:
            # occasional random shake‑up to escape flat regions
            if rng.random() < 0.03:
                idx = rng.integers(0, n)
                factor = float(np.exp(rng.normal(0.0, 0.10)))   # ±10 %
                a[idx] *= factor
                a = _project_simplex(a)

        # early exit if target reached
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping early")
            break

        # occasional status output
        if it % 10 == 0:
            elapsed = time.monotonic() - (deadline - budget_s + 0.5)
            print(f"[status] iter={it:,}  best_score={best_score:.6f}  time={elapsed:.1f}s")

        # stop if we have been idle for many iterations
        if no_improve >= 30:
            # run a few Adam refinement steps as a fallback
            print("[info] stagnation detected – performing a short Adam refinement")
            a_ref = best.copy()
            # ---- Adam parameters (lightweight) ----
            m = np.zeros_like(a_ref)
            v = np.zeros_like(a_ref)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr0 = 0.3
            for inner in range(2000):
                # smooth‑max with moderate β
                beta_s = 30.0
                conv = np.convolve(a_ref, a_ref)
                max_c = conv.max()
                exps = np.exp(beta_s * (conv - max_c))
                w = exps / exps.sum()
                grad_full = np.convolve(w, a_ref[::-1])
                grad = 2.0 * grad_full[n - 1:n - 1 + n]

                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1 ** (inner + 1))
                v_hat = v / (1 - beta2 ** (inner + 1))
                step = lr0 / np.sqrt(inner + 1)

                a_ref = _project_simplex(a_ref)

            if score_ref < best_score:
                best = a_ref.copy()
                best_score = score_ref
                a = a_ref.copy()
                print(f"[adam‑refine] new best score={best_score:.6f}")
            no_improve = 0

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    # Return a plain Python list of floats
    return [float(x) for x in best]
