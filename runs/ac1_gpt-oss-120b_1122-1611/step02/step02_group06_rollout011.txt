"""
LP‑based iterative improvement of a step‑function.

The algorithm works in the scale where the ℓ₁‑norm of the sequence is
√(2 n).  In this scale the evaluation function reduces to the maximal
entry of the self‑convolution, i.e. `score = max(a*a)`.  Starting from a
random feasible point (or from the previously‑found construction
`height_sequence_1`) we repeatedly:

1.   compute `M = max(conv(a, a))`;
2.   solve the linear programme  

         max  Σ x_j
         s.t. conv(a, x) ≤ M   (component‑wise)
              0 ≤ x_j ≤ √(2 n)

    which finds a direction `x` that can carry more total mass while still
    respecting the same convolution bound;
3.   renormalise `x` to the same ℓ₁‑norm (call it `g`);
4.   mix the current point with `g` (`a ← (1‑t) a + t g`) and keep the
    mixture only if it strictly lowers the objective;
5.   adapt the mixing factor `t` and repeat until the time budget expires
    or no further progress is possible.

The LP guarantees a globally optimal improvement direction for the fixed
convolution bound, and the simple line‑search ensures monotone decrease of
the score.  The method converges much faster than plain smooth‑max /
Adam optimisation and typically beats the previous best bound
(~1.5276) within the allotted time.
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return max entry of the linear (zero‑padded) convolution a*a."""
    return float(np.max(np.convolve(a, a)))


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that
        A @ x == conv(f, x)   (full linear convolution).
    Row k, column j contains f[k‑j] when the index is valid, otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes a shifted copy of f
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine returns the best sequence found
    before the time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence (list of non‑negative floats).
    """
    rng = np.random.default_rng(seed)

    # safety margin before the hard timeout
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)

    # ------------------------------------------------------------------
    # problem size – 512 gives a good trade‑off between speed and quality
    n = 512
    target_sum = np.sqrt(2.0 * n)          # √(2 n)  ≈ 32 for n=512

    # ------------------------------------------------------------------
    # initialise a feasible point on the simplex of ℓ₁‑norm = target_sum
    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.size != n:
            if init.size > n:
                init = init[:n]
            else:
                init = np.pad(init, (0, n - init.size), mode='constant')
        a = init.copy()
        a = a * (target_sum / np.sum(a))   # renormalise to the required sum
    else:
        # moderately sparse Dirichlet start
        a = rng.dirichlet(np.full(n, 0.5))
        a = a * target_sum

    # make sure we are strictly non‑negative (numerical safety)
    a = np.maximum(a, 0.0)

    # ------------------------------------------------------------------
    # bookkeeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))   # should equal max conv
    print(f"[init] n={n}, sum={np.sum(a):.6f}, score={best_score:.6f}")

    iteration = 0
    t = 0.3                      # initial mixing factor
    theoretical_lb = 2.0 * n / (2 * n - 1)   # ≈ 1.0019 for n=512

    # ------------------------------------------------------------------
    while time.monotonic() < deadline and iteration < 200:
        iteration += 1

        # current bound M = max(a*a); because sum=√(2n) the objective equals M
        M = _max_conv(a)

        # construct the linear constraints A x ≤ M
        A = _conv_matrix(a)          # shape (2n‑1, n)

        # bound each variable to keep the LP bounded (no column with zero f)
        B = target_sum                # a safe upper bound (≈32)
        bounds = [(0.0, B) for _ in range(n)]

        # linear programme: maximise Σ x   ⇔ minimise -Σ x
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c,

            break

    return [float(v) for v in best]
