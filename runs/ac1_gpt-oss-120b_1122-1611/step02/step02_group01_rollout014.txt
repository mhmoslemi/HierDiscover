import time, numpy as np, math, sys
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def _fft_len_for_n(n):
    size = 2*n -1
    return 1 << ((size-1).bit_length())

def _conv_self(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa*fa
    conv = np.fft.irfft(fc, n=fft_len)[:2*a.size-1]
    return conv

def _conv_pair(a,b,fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    fc = fa*fb
    conv = np.fft.irfft(fc, n=fft_len)[:2*a.size-1]
    return conv

def _build_conv_matrix(f):
    """Return the (2n-1) x n sparse matrix A where A @ g = conv(f,g)."""
    n = f.size
    # i index for f (0..n-1), j index for g (0..n-1)
    i = np.arange(n, dtype=np.int64).reshape(n,1)   # column vector
    j = np.arange(n, dtype=np.int64).reshape(1,n)   # row vector
    rows = (i + j).ravel()          # shape (n*n,)
    cols = np.tile(np.arange(n, dtype=np.int64), n)   # repeat 0..n-1 n times
    data = np.repeat(f, n)          # each f[i] repeated n times
    # shape is (2*n-1, n)
    return coo_matrix((data, (rows, cols)), shape=(2*n-1, n), dtype=np.float64).tocsr()

def _score_from_a(a, fft_len):
    """Assumes sum(a) == 1, returns 2*n*max(conv)."""
    n = a.size
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())

def _project_to_simplex(v):
    # v >= 0? Actually algorithm handles any v.
    # From Duchi et al. 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    return conv

    """Constructs CSR matrix A where A @ g = conv(f,g)."""
    n = f.size
    i = np.arange(n, dtype=np.int64).reshape(n, 1)      # column vector
    j = np.arange(n, dtype=np.int64).reshape(1, n)      # row vector

    """Euclidean projection of v onto the probability simplex."""

                    shift_i[i:i + n] = a
                    shift_j = np.zeros(2 * n - 1, dtype=a.dtype)
                    shift_j[j:j + n] = a
                    new_conv = conv + delta * (shift_j - shift_i)
                    new_score = 2.0 * n * float(new_conv.max())
                    if new_score < cur_score:
                        a[i] -= delta
                        a[j] += delta
                        a = np.clip(a, 0.0, None)
                        a /= a.sum()
                        conv = new_conv
                        cur_score = new_score

        # soft restart if stagnated
        now = time.monotonic()
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.11, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            elapsed = now - start_time
            print(f"[restart] re‑starting from best score {best_score:.6f} (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # Return the best vector found (clipped to the limits used by
    # evaluate_sequence for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
