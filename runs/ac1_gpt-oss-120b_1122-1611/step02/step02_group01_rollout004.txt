def _project_simplex(v):
    # ensure v is np.ndarray float.
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1  # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # initialize a
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                total = a.sum()
                if total > 0:
                    a /= total
                else:
                    a = np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # make sure a sum =1, nonneg
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # precompute fft length
    fft_len = _fft_len_for_n(n)

    best = a.copy()
    conv_best = _conv_self(best, fft_len)
    best_score = float(2.0 * n * conv_best.max())

    start_time = time.monotonic()
    last_improve = start_time

    # hyperparams
    eta_start = 0.7
    eta_end = 0.02
    max_active = 30          # maximum active indices
    eps_start = 0.10
    eps_end = 0.001
    block_prob = 0.03
    restart_gap = 40.0
    max_frac_transfer = 0.25 # fraction of source mass to move

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)
        # compute convolution
        conv = _conv_self(a, fft_len)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # determine active set
        eps_rel = eps_start * (1.0 - progress) + eps_end * progress
        threshold = max_conv * (1.0 - eps_rel)
        # find indices above threshold
        active_idx = np.where(conv >= threshold)[0]
        if active_idx.size > max_active:
            # keep only top max_active by conv value
            # use partition to get top indices
            kth = active_idx.size - max_active
            # active_idx is unsorted? it's sorted ascending because np.where returns sorted indices. Simpler: take based on conv[active_idx]
            # Use argpartition on conv[active_idx] to get top max_active
            sub = active_idx
            # find top max_active among sub
            values = conv[sub]
            top_k_idx = np.argpartition(values, -max_active)[-max_active:]
            active_idx = sub[top_k_idx]

        # compute aggregated partner vector
        w = np.zeros_like(a)
        for k in active_idx:
            w += _partner_vector(a, int(k))

        # learning rate schedule
        eta = eta_start * (1.0 - progress) + eta_end * progress
        # multiplicative update
        factor = np.exp(-eta * w)
        a = a * factor
        # renormalize to sum 1
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional block perturbation
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        # deterministic pairwise move targeting max convolution index
        # compute partner vector for k_max
        k_max = int(conv.argmax())
        w_kmax = _partner_vector(a, k_max)
        # product contributions
        contrib = a * w_kmax
        i_star = int(np.argmax(contrib))
        # find a low contribution index
        # candidate indices where w_kmax small
        # optionally choose index with minimal product
        j_star = int(np.argmin(contrib))
        if i_star != j_star:
            # max transfer fraction
            max_delta = min(a[i_star], max_frac_transfer * a[i_star])
            if max_delta > 0:
                # choose delta maybe half of max delta to be safe
                delta = max_delta * 0.5
                # compute new convolution after move
                part_i = _partner_vector(a, i_star)
                part_j = _partner_vector(a, j_star)
                new_conv = conv + delta * (part_j - part_i)
                new_max = new_conv.max()
                if new_max < max_conv:
                    a[i_star] -= delta
                    a[j_star] += delta
                    # renormalize (should be same sum)
                    s = a.sum()
                    if s == 0.0:
                        a = np.full_like(a, 1.0 / n)
                    else:
                        a /= s
                    conv = new_conv
                    max_conv = new_max
                    cur_score = 2.0 * n * max_conv
                    # maybe update best if improved
                    if cur_score < best_score:
                        best_score = cur_score
                        best = a.copy()
                        last_improve = time.monotonic()
                        print(f"[{elapsed:6.2f}s] new best after pair move = {best_score:.6f}")

        # restart if no improvement for a while
        now = time.monotonic()
        if now - last_improve > restart_gap:
            # reinitialize a from best with slight perturbation
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – from best {best_score:.6f}")

    # finalize best
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Hybrid exponential‑gradient + deterministic pairwise descent.

* Keeps the sum of the step‑function equal to 1 (objective is scale‑free).
* At each iteration the K largest convolution entries (or all entries within
  a shrinking relative tolerance) are used to build a sub‑gradient of the
  L∞‑norm of the self‑convolution.  An exponentiated‑gradient (multiplicative)
  update followed by a normalisation yields a feasible non‑negative vector.
* A cheap deterministic “mass‑transfer” step targets the current maximum
  convolution index: mass is moved from the index that contributes the most
  to that peak towards an index that contributes the least, and the move is
  kept only if it reduces the global maximum.
* Small random block perturbations explore the neighbourhood, and a soft
  restart (with a mild log‑normal jitter) prevents stagnation.
* The algorithm runs until the time budget (≈ 1000 s) expires and returns the
  best vector encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Length of the zero‑padded FFT that can hold the full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real non‑negative vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of (a*a)[k] with respect to a.
    Returns a vector w where w[i] = a[k-i] (0 outside the support).
    """
    n = a.size
    w = np.zeros_like(a)
    if k < n:
        # overlapping part i = 0 … k
        w[: k + 1] = a[k :: -1]          # a[k], a[k‑1], …, a[0]
    else:
        # overlapping part i = k-(n‑1) … n‑1
        start = k - (n - 1)               # first i that has a partner
        length = (2 * n - 1) - k          # number of overlapping entries
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Linear contribution of ``a[idx]`` to the full self‑convolution.
    The result has length 2*n‑1 and satisfies:
        (a*a)[k] = Σ_i a[i] * (contribution_vector(i))[k]
    """
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx : idx + n] = a
    return v


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``score = 2 * n * max(conv(a,a)) / (sum(a)**2)``.
    The routine runs for roughly ``budget_s`` seconds (leaving a small safety
    margin) and returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    # leave a safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Initial vector (prefer the supplied global if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                total = a.sum()
                if total > 0.0:
                    a /= total
                else:
                    raise ValueError
            else:
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # fallback: modest size uniform start
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # ------------------------------------------------------------------
    # 2. Book‑keeping
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    best = a.copy()
    conv_best = _conv_self(best, fft_len)
    best_score = float(2.0 * n * conv_best.max())

    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3. Hyper‑parameters (tuned experimentally)
    # ------------------------------------------------------------------
    eta_start = 0.7          # initial learning‑rate for EG step
    eta_end = 0.02           # final learning‑rate
    K_target = 25            # target number of active convolution entries
    eps_start = 0.08         # start relative tolerance for the active set
    eps_end = 0.001          # final tolerance
    block_perturb_prob = 0.04
    restart_gap = 30.0        # seconds without improvement before a soft restart
    max_frac_transfer = 0.25  # maximal fraction of source mass moved in pair‑move

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        progress = elapsed / (deadline - start_time)

        # --------------------------------------------------------------
        # a) Current convolution and score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # b) Keep the best solution
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # c) Build the active set (indices whose convolution is close to the max)
        # --------------------------------------------------------------
        eps_rel = eps_start * (1.0 - progress) + eps_end * progress
        threshold = max_conv * (1.0 - eps_rel)
        active_idx = np.where(conv >= threshold)[0]

        # limit the number of active indices for speed
        if active_idx.size > K_target:
            # keep the K_target largest values
            vals = conv[active_idx]
            top_k = np.argpartition(vals, -K_target)[-K_target:]
            active_idx = active_idx[top_k]

        # --------------------------------------------------------------
        # d) Exponentiated‑gradient update using the aggregated sub‑gradient
        # --------------------------------------------------------------
        w = np.zeros_like(a)
        for k in active_idx:
            w += _partner_vector(a, int(k))

        # learning‑rate schedule (linear decay)
        eta = eta_start * (1.0 - progress) + eta_end * progress
        factor = np.exp(-eta * w)
        a = a * factor
        # renormalise to keep Σa = 1
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --------------------------------------------------------------
        # e) Occasional random block perturbation (exploration)
        # --------------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # f) Deterministic pair‑wise mass transfer targeting the worst peak
        # --------------------------------------------------------------
        # index of the current maximal convolution entry
        k_max = int(conv.argmax())
        # contribution vector of each position to this peak
        wk = _partner_vector(a, k_max)                     # length n
        # product a[i] * wk[i] measures the absolute contribution of i
        contrib = a * wk
        i_star = int(np.argmax(contrib))
        j_star = int(np.argmin(contrib))

        if i_star != j_star:
            max_delta = min(a[i_star], max_frac_transfer * a[i_star])
            if max_delta > 0.0:
                # try moving half of the admissible mass (a small step improves stability)
                delta = max_delta * 0.5

                # contribution vectors of the two positions (full convolution contribution)
                contrib_i = _contribution_vector(a, i_star)
                contrib_j = _contribution_vector(a, j_star)

                new_conv = conv + delta * (contrib_j - contrib_i)
                new_max = float(new_conv.max())

                if new_max < max_conv:
                    # accept the move
                    a[i_star] -= delta
                    a[j_star] += delta
                    # renormalise (sum is unchanged analytically, but we keep numerical sanity)
                    s = a.sum()
                    if s == 0.0:
                        a = np.full_like(a, 1.0 / n)
                    else:
                        a /= s
                    conv = new_conv
                    max_conv = new_max
                    cur_score = 2.0 * n * max_conv
                    if cur_score < best_score:
                        best_score = cur_score
                        best = a.copy()
                        last_improve = time.monotonic()
                        print(f"[{elapsed:6.2f}s] new best after pair‑move = {best_score:.6f}")

        # --------------------------------------------------------------
        # g) Soft restart when stuck
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            # perturb the current best slightly and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – re‑starting from best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best vector found, clipped to the bounds used by the judge.
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
