def solve_lp_max_sum(f, M):
    n = len(f)
    # Build A matrix
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        # for each i, j such that i+j=k => b_j coefficient f_i
        # So for each i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # j = k - i
        # So for each i, set A[k, j] = f[i]
        # Let's vectorize: i indices array
        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        A[k, j_vals] = f[i_vals]
    # Constraint: A @ b <= M
    # Also sign constraints: b >= 0 (implicitly bound)
    # Use linprog: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback: maybe use some simple solution
        return None

def solve_lp_max_sum(f, M):
    n = len(f)
    rows = 2*n - 1

    # Build sparse matrix A (rows x n)
    from scipy.sparse import csr_matrix

    # Pre-allocate lists
    row_inds = []
    col_inds = []
    data = []

    # Only indices where f_i > 0
    for i, fi in enumerate(f):
        if fi <= 0:
            continue
        # rows i..i+n-1 correspond to conv positions where b_j appears
        start = i
        end = i + n  # exclusive
        # Append n entries
        # row indices: start to end-1
        row_inds.extend(range(start, end))
        col_inds.extend(range(n))
        data.extend([fi] * n)

    A = csr_matrix((data, (row_inds, col_inds)), shape=(rows, n))

    # b_ub = M vector of length rows
    b_ub = np.full(rows, M)

    c = -np.ones(n)  # maximize sum(b) => minimize -sum(b)

    bounds = [(0, None) for _ in range(n)]

    # solve using hiGHs method
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

def propose_candidate(seed=123, budget_s=1000.0, n=1024):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.5

    target_sum = math.sqrt(2 * n)

    def normalize(vec):
        vec = np.maximum(vec, 0)
        total = np.sum(vec)
        if total == 0:
            vec = np.full_like(vec, target_sum / n)
            total = np.sum(vec)
        vec = vec * (target_sum / total)
        return vec

    # may generate initial candidates: if have warm start use it, else uniform.
    # set initial best.

    best_seq = None
    best_score = float('inf')

    # we will run multiple attempts within time.
    attempt = 0
    while time.monotonic() < deadline:
        attempt += 1
        # choose start: warm start or random Dirichlet
        if attempt == 1 and 'height_sequence_1' in globals():
            a = np.asarray(globals()['height_sequence_1'], dtype=float)
            if a.size != n:
                a = np.full(n, target_sum / n)
        else:

            alpha = np.ones(n) * 0.5  # encourages sparsity?
            a = rng.gamma(shape=alpha, scale=1.0)

        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            print(f"[attempt {attempt}] init score={best_score:.6f}")

        # inner iteration planning: split time budget (remaining) between phases.

        # Phase1: LP steps
        max_lp_iters = 30
        for lp_iter in range(max_lp_iters):
            # check deadline
            if time.monotonic() >= deadline:
                break
            conv = np.convolve(a, a)

The method is deterministic apart from the random initialisations and runs

import time

from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                               # all non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _project_to_target_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the simplex with prescribed ℓ₁‑norm = target_sum."""
    w = _project_simplex(v)
    return w * target_sum


# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   ∑ b_j
        subject to  (f * b)[k] ≤ M  for all k
                    b_j ≥ 0
    Returns the optimal b (may be None on failure).
    """
    n = f.size
    rows = 2 * n - 1

    # Build the convolution matrix A where (A @ b)[k] = (f * b)[k]
    i = np.arange(n).reshape(-1, 1)          # column vector (n,1)
    j = np.arange(n).reshape(1, -1)          # row vector    (1,n)
    row_idx = (i + j).ravel()                # shape (n*n,)
    col_idx = j.ravel()                      # shape (n*n,)
    data = np.repeat(f, n)                   # each f_i appears n times

    A = csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))

    # Linear programme: minimise   -∑b   ↆ   maximise   ∑b
    c = -np.ones(n, dtype=float)             # objective
    b_ub = np.full(rows, M, dtype=float)    # RHS of inequality constraints
    bounds = [(0, None)] * n                 # b_j ≥ 0

    # Using the HiGHS interior‑point solver (fast for this size)
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True})

    if result.success:
        return result.x
    return None
