"""
Propose a low‑score step‑function using a hybrid of
* cheap parametric sweeps (Beta‑distribution shapes, power‑law shapes)
* a fast exponentiated‑gradient refinement that focuses on the largest
  convolution peaks.
The routine respects the 1000 s budget, works for any provided
`height_sequence_1`, and finally returns a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog  # only imported – not used in the fast path


# ----------------------------------------------------------------------
# Low‑level convolution helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Next power‑of‑two length sufficient for a linear convolution of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a (real, non‑negative) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)[k] with respect to a.
    Returns a vector w with w[i] = a[k-i] (0 if index out of range).
    """
    n = a.size
    w = np.zeros_like(a)

    if k < n:
        # i = 0 … k  →  a[k-i] = a[k], a[k-1], …, a[0]
        w[: k + 1] = a[k :: -1]
    else:
        # i = k-(n-1) … n-1  →  a[k-i] = a[n-1], …, a[0]
        start = k - (n - 1)
        length = (2 * n - 1) - k
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


# ----------------------------------------------------------------------
# Simple perturbation utilities used by the fine‑tuning stage
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Parametric families (Beta‑law and power‑law) used in the coarse search
# ----------------------------------------------------------------------


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on a grid of size n.
    a_i ∝ x_i^{α‑1} (1‑x_i)^{α‑1},   x_i = (i+0.5)/n.
    """
    eps = 1e-12
    xs = np.arange(n, dtype=np.float64) + 0.5
    xs = xs / n
    xs = np.clip(xs, eps, 1.0 - eps)
    shape = np.power(xs, alpha - 1.0) * np.power(1.0 - xs, alpha - 1.0)
    shape = np.clip(shape, 0.0, np.inf)
    s = shape.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return shape / s


def _power_shape(n: int, gamma: float) -> np.ndarray:
    """
    Power‑law shape a_i ∝ ((i+½)*(n‑i‑½))^{γ},
    which reproduces the classic arcsine law for γ = –½.
    """
    eps = 1e-12
    idx = np.arange(n, dtype=np.float64) + 0.5
    vals = (idx * (n - idx) + eps) ** gamma
    vals = np.clip(vals, 0.0, np.inf)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / s

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative height sequence that minimizes the
    `evaluate_sequence` score.  The method consists of:

    1. Load the provided `height_sequence_1` (if any) and keep it as a
       reference point.
    2. Perform cheap parametric sweeps over two one‑parameter families
       (symmetric Beta densities and a simple power‑law) and keep the
       best candidate found.
    3. Refine the best candidate with a fast exponentiated‑gradient
       algorithm that repeatedly pushes mass away from the currently

    4. Return the best vector seen (clipped to the limits imposed by
       `evaluate_sequence`).
    """

    # Establish a safe deadline (a couple of seconds before the hard limit)

        best_score = 2.0 * n * np.max(_conv_self(best_seq, _fft_len_for_n(n)))
        print(f"[init] fallback to uniform, n={n}, score≈{best_score:.6f}")

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 1️⃣  PARAMETRIC SWEEP ------------------------------------------------
    # ------------------------------------------------------------------
    # the arcsine law corresponds to α = ½ (Beta) or γ = –½ (power‑law)

    # 3️⃣  RETURN THE BEST VECTOR (clipped to the limits of `evaluate_sequence`)
    # ------------------------------------------------------------------

    return [float(x) for x in final_seq.tolist()]
