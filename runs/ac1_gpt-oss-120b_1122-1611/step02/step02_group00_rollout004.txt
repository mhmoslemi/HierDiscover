"""
propose_candidate
-----------------

Adam‑based projected gradient descent on a smooth approximation of the
max‑auto‑convolution objective.

The objective for a non‑negative vector `w` (∑w_i = 1) is

    F(w) = max_k (w * w)_k

which is convex but non‑differentiable.  We replace the max by a
soft‑max with a temperature parameter α :

    f_α(w) = (1/α)·log(∑_k exp(α·(w*w)_k))

The gradient of f_α is

    ∇f_α(w)_i = 2·∑_k s_k·w_{k−i}
    where s_k = exp(α·b_k)/∑_j exp(α·b_j)  (soft‑max weights)
          b = w * w  (full linear convolution).

Each iteration therefore consists of:
1. Computing `b = w*w`.
2. Building the soft‑max weights `s`.
3. Computing the gradient via one extra convolution.
4. Adam update.
5. Projection back onto the probability simplex.

A homotopy on α (slowly increasing from a small value to ~30) gives a
smooth start and focuses on the true maximum later on.  Random
block‑perturbations are applied when progress stalls, and the best
vector seen so far is returned (scaled to the conventional
√(2 n) L¹‑norm so that ``evaluate_sequence`` receives a nicely sized
list).

The routine works for any length; if a previous construction is
available in the global variable ``height_sequence_1`` it is used as a
starting point, otherwise a Catalan distribution is used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers a_i = C(2i,i)/4**i (recurrence)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full mode)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negatives caused by round‑off.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    # Duchi et al. (2008) – O(n log n) due to sorting.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # Numerical fallback – uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation (used for jumps)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetric averaging (helps for even‑odd patterns)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main optimisation routine required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) whose
    ``evaluate_sequence`` score is (hopefully) below the target.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length and an initial vector.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.ndim != 1 or init.size < 2:
                raise ValueError
            n = init.size
            w = np.maximum(init, 0.0)
        except Exception:
            # fallback to Catalan
            n = 8000
            w = catalan_sequence(n)
    else:
        n = 8000
        w = catalan_sequence(n)

    # Normalise to a probability vector (∑w = 1).
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w_sum

    # --------------------------------------------------------------
    # Optional quick search over a one‑parameter Beta‑family to get a
    # better starting point (symmetrical a=b<1 gives a U‑shape).
    # --------------------------------------------------------------
    best_local_w = w.copy()
    best_local_score = 2.0 * n * np.max(convolve_fft(w, w))

    exponents = np.linspace(0.05, 0.95, 19)  # a in (0,1)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a in exponents:
        # w_i ∝ x_i^{a-1} * (1-x_i)^{a-1}
        temp = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
        temp_sum = temp.sum()
        if temp_sum == 0.0:
            continue
        temp /= temp_sum
        score = 2.0 * n * np.max(convolve_fft(temp, temp))
        if score < best_local_score:
            best_local_score = score
            best_local_w = temp.copy()
    if best_local_score < best_local_w.max():  # sanity check
        w = best_local_w.copy()
        print(f"[beta‑init] n={n}, score={best_local_score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser settings.
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.20                # base learning rate
    m = np.zeros_like(w)     # first moment
    v = np.zeros_like(w)     # second moment

    # Smoothing (soft‑max) parameter.
    alpha = 3.0
    alpha_max = 30.0
    alpha_inc_factor = 1.003  # multiplicative increase per iteration

    best_w = w.copy()
    best_score = 2.0 * n * np.max(convolve_fft(w, w))
    print(f"[init] n={n}, start score={best_score:.6f}")

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0
    check_interval = 500          # how often we evaluate the true score
    perturb_interval = 8000       # occasional random jump
    max_no_improve = 20000        # after this we reset to a perturbed best

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 2.0):
        # ----- (1) compute current self‑convolution -----
        b = convolve_fft(w, w)          # length 2n‑1
        max_b = np.max(b)

        # ----- (2) smooth‑max and its soft‑max weights -----
        b_max = max_b
        # stable soft‑max
        shifted = alpha * (b - b_max)
        exp_shift = np.exp(shifted)
        s = exp_shift / np.sum(exp_shift)   # shape (2n‑1,)

        # ----- (3) gradient of the smooth objective -----
        w_rev = w[::-1]
        conv_rev = convolve_fft(w_rev, s)   # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]   # slice to length n

        # ----- (4) Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        w_new = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- (5) projection onto the simplex -----
        w = simplex_projection(w_new)

        # ----- (6) occasional housekeeping -----
        # Increase smoothing parameter gradually.
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc_factor)

        # Evaluate true objective only every `check_interval` iterations.
        if iteration % check_interval == 0:
            cur_score = 2.0 * n * np.max(convolve_fft(w, w))
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_w = w.copy()
                no_improve = 0
                print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
            else:
                no_improve += check_interval

        # Random block perturbation to escape plateaus.
        if iteration % perturb_interval == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[perturb] iter={iteration:6d}, new best={best_score:.6f}")

        # Hard reset if no improvement for a long time.
        if no_improve > max_no_improve:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

    # --------------------------------------------------------------
    # Final scaling – we output a vector whose L¹‑norm equals √(2 n)
    # (the scaling does not affect the score but keeps the numbers
    # within a comfortable range for the evaluator).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * (target_sum / best_w.sum())

    # Final sanity check (the harness will also do it).
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best score (pre‑scale)={best_score:.6f}, "
          f"evaluate_sequence={final_score:.6f}")

    return [float(x) for x in final_seq]
