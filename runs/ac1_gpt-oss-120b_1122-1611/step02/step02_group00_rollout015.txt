    # set length n

    # Ensure n is odd? not required.

    # Initialize w via Catalan.
    seq = catalan_sequence(n)
    w = seq / seq.sum()

    # Precompute N for FFT
    L = 2 * n - 1
    N = 1 << ((L - 1).bit_length())

    # hyperparameters
    eta0 = 0.6
    alpha = 10.0
    alpha_inc = 1.001
    alpha_max = 5000.0
    beta_mom = 0.5

    best_w = w.copy()
    b = convolve_fft(w, w, N)
    best_score = 2.0 * n * b.max()
    best_iter = 0

    # store previous w for momentum
    w_prev = w.copy()
    grad_prev = np.zeros_like(w)

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - start_time < max(0.0, budget_s - 5.0):
        # conv w*w
        # compute FFT once
        wf = np.fft.rfft(w, N)
        b_full = np.fft.irfft(wf * wf, N)[:L]
        max_b = b_full.max()
        score = 2.0 * n * max_b

        # record best
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            best_iter = iteration
            no_improve = 0
            print(f"[improve] iter={iteration} score={best_score:.6f}")
        else:
            no_improve += 1

        # compute smoothing weights
        diff = b_full - max_b  # non-positive
        # compute exponentials
        exp_vals = np.exp(alpha * diff)
        # if all zeros (underflow) fallback to uniform weighting on max indices
        if exp_vals.sum() == 0.0:
            # use uniform weights over indices with max_b
            max_idxs = np.flatnonzero(b_full >= max_b - 1e-15)
            exp_vals[max_idxs] = 1.0
        p = exp_vals / exp_vals.sum()

        # compute gradient via convolution p * rev_w
        rev_w = w[::-1]
        p_f = np.fft.rfft(p, N)
        rev_w_f = np.fft.rfft(rev_w, N)
        conv_full = np.fft.irfft(p_f * rev_w_f, N)[:len(p)+len(rev_w)-1]  # length = 3n-2
        grad = 2.0 * conv_full[n-1:n-1+n]  # slice length n

        # momentum
        if iteration == 0:
            w_new = w - eta0 / np.sqrt(iteration + 1) * grad
        else:
            delta = w - w_prev
            w_new = w - eta0 / np.sqrt(iteration + 1) * grad + beta_mom * delta

        # projection onto simplex
        w_new = simplex_projection(w_new)

        # update previous
        w_prev = w
        w = w_new

        # adapt alpha
        alpha = min(alpha * alpha_inc, alpha_max)

        # occasional random perturbation
        if iteration % 2000 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            # evaluate cand
            wf_c = np.fft.rfft(cand, N)
            b_c = np.fft.irfft(wf_c * wf_c, N)[:L]
            cand_score = 2.0 * n * b_c.max()
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                w = cand.copy()
                print(f"[block] iter={iteration} new best {best_score:.6f}")

        # reset if stuck
        if no_improve > 30000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            w_prev = w.copy()
            no_improve = 0
            print(f"[reset] iter={iteration} reset to perturbed best")

        iteration += 1

    # final scaling and evaluation
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best score={best_score:.6f}, eval after scaling={final_score:.6f}")

    return [float(x) for x in final_seq]

"""
Propose a step‑function (non‑negative sequence) with a low self‑convolution peak.

The algorithm is a smoothed projected sub‑gradient descent on the
quantity   F(w)=2·n·max_k (w*w)[k]   where w≥0, ∑w=1.
Instead of a hard max we use a softmax (log‑sum‑exp) smoothing:
    p_k ∝ exp(α·((w*w)[k]−max(w*w))),
so the gradient is a weighted average of all convolution gradients.
A momentum term and occasional random “block” perturbations help escape
local minima.  The iterate is projected onto the probability simplex at
each step (Euclidean projection).  The method runs until the allotted
time (≈1000 s) expires, keeping the best vector seen and finally scaling
it to the conventional √(2 n) ℓ¹‑norm required by `evaluate_sequence`.

The code is self‑contained; only NumPy and the provided
`evaluate_sequence` are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    First `length` Catalan‐type coefficients a_i = C(2i,i)/4**i.
    Recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) is numerically stable.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray, N: int) -> np.ndarray:
    """
    Linear convolution via real FFT of size N (power of two, N ≥ len(a)+len(b)-1).
    Result is clipped to non‑negative values.
    """
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[: a.size + b.size - 1]
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump: multiply a random contiguous block by a factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                      # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) whose
    `evaluate_sequence` score is as low as possible (target < 1.5025).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.  Use the previous construction if available,
    # otherwise pick a size that still fits into a 32768‑point FFT.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n0 = int(np.asarray(height_sequence_1).size)
            # Aim for roughly double the previous length but cap at 16000
            n = min(16000, max(8000, n0 * 2))
        except Exception:
            n = 12000
    else:
        n = 12000

    # --------------------------------------------------------------
    # Initialise with the Catalan distribution (probability vector).
    # --------------------------------------------------------------
    a = catalan_sequence(n)
    w = a / np.sum(a)                     # ∑w = 1

    # Pre‑compute FFT size (next power of two ≥ 2·n−1)
    L = 2 * n - 1
    N = 1 << ((L - 1).bit_length())

    # Track the best vector seen so far
    b_init = convolve_fft(w, w, N)
    best_score = 2.0 * n * np.max(b_init)
    best_w = w.copy()
    best_iter = 0
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters for the smoothed sub‑gradient descent
    # --------------------------------------------------------------
    eta0 = 0.6            # initial step size factor
    alpha = 10.0          # start smoothing temperature
    alpha_inc = 1.001     # geometric increase per iteration
    alpha_max = 5000.0    # upper bound for α
    beta_mom = 0.5        # momentum coefficient
    target = 1.5025       # goal to beat
    start_time = time.monotonic()
    iteration = 0
    stagnation = 0       # counts iterations without improvement
    w_prev = w.copy()

    # Main optimisation loop
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ---- 1) current convolution and score ----
        wf = np.fft.rfft(w, N)
        b_full = np.fft.irfft(wf * wf, N)[:L]
        max_b = np.max(b_full)
        cur_score = 2.0 * n * max_b

        # ---- 2) keep the best vector ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            best_iter = iteration
            stagnation = 0
            print(f"[improve] iter={iteration:6d} score={best_score:.6f}")
            if best_score < target:
                # excellent – we can stop early
                break
        else:
            stagnation += 1

        # ---- 3) smoothed gradient of the max‑convolution ----
        diff = b_full - max_b                     # ≤0, max at 0
        exp_vals = np.exp(alpha * diff)          # softmax weights (unnormalised)
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0:                        # numerical fallback (should not happen)
            max_idxs = np.flatnonzero(b_full >= max_b - 1e-15)
            exp_vals[max_idxs] = 1.0
            sum_exp = float(max_idxs.size)
        p = exp_vals / sum_exp                    # probabilities, ∑p = 1

        rev_w = w[::-1]
        p_f = np.fft.rfft(p, N)
        rev_w_f = np.fft.rfft(rev_w, N)
        conv_full = np.fft.irfft(p_f * rev_w_f, N)[:len(p) + len(rev_w) - 1]
        # gradient = 2·∑_k p_k·w_{k‑i}
        grad = 2.0 * conv_full[n - 1 : n - 1 + n]

        # ---- 4) momentum update + projection onto simplex ----
        eta = eta0 / np.sqrt(iteration + 1.0)
        delta = w - w_prev
        w_new = w - eta * grad + beta_mom * delta
        w_new = simplex_projection(w_new)

        # ---- 5) prepare for next iteration ----
        w_prev = w
        w = w_new

        # ---- 6) slowly tighten the smoothing (larger α) ----
        alpha = min(alpha * alpha_inc, alpha_max)

        # ---- 7) occasional random jump (block perturbation) ----
        if iteration % 2000 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_f = np.fft.rfft(cand, N)
            b_cand = np.fft.irfft(cand_f * cand_f, N)[:L]
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                w = cand.copy()
                w_prev = w.copy()
                print(f"[block] iter={iteration:6d} new best score={best_score:.6f}")

        # ---- 8) reset if stagnating for a long stretch ----
        if stagnation > 30000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            w_prev = w.copy()
            stagnation = 0
            print(f"[reset] iter={iteration:6d} reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – bring ℓ¹‑norm to √(2 n) (the evaluation is
    # scale‑invariant, but this size matches the traditional convention).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(
        f"[final] length={n}, best iter={best_iter}, "
        f"best score={best_score:.6f}, final evaluate={final_score:.6f}"
    )
    return [float(x) for x in final_seq]
