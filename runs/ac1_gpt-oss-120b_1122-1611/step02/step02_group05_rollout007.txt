"""
Propose a step‑function (probability vector) that minimises
`evaluate_sequence`.  The search consists of three stages:

1. **Soft‑max Adam optimisation** – we minimise a smooth approximation of
   the max‑convolution `S_α(a) = (1/α)·log Σ exp(α·(a*a)[k])`.  The temperature
   `α` is linearly increased from 2 to a fairly high value (≈80) so that the
   surrogate becomes close to the true max.  Adam updates keep the search
   stable; after each step the vector is projected onto the probability simplex.

2. **Hard‑max (sub‑gradient) Adam optimisation** – once the temperature is
   large we switch to the exact sub‑gradient of the max‑convolution
   (a one‑hot weight).  This fine‑tunes the solution and can further flatten
   the autocorrelation peaks.

3. **Random restarts + occasional perturbations** – the whole two‑phase
   optimiser is run from several random initial probability vectors
   (Dirichlet draws with a tunable concentration) as long as time permits.
   The best sequence seen overall is returned.

All operations respect the constraints of `evaluate_sequence`
(non‑negative, sum ≈ 1).  The routine runs for the supplied time budget
(~1000 s) and prints progress reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate S_α(a) = (1/α)·log Σ exp(α·(a*a)[k]).
    Returns ∇S_α(a) = 2·correlate(w, a) where w is the soft‑max weighting.
    """
    conv = np.convolve(a, a)                     # length 2n‑1
    shift = conv.max()
    # compute stable soft‑max weights
    exps = np.exp(alpha * (conv - shift))
    w = exps / exps.sum()
    # correlate w with a (valid mode gives length n)
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, conv.max()


# ----------------------------------------------------------------------
def _hardmax_gradient(a: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] : pick the index attaining the
    maximum and compute ∂/∂a_i max = 2·a_{k*‑i} (zero when partner index out of range).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    idx = np.arange(n)
    partner = k_star - idx
    valid = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    grad[valid] = 2.0 * a[partner[valid]]
    return grad, conv.max()

            else:
                a0 = np.concatenate([init_seq,
                                     np.full(n - init_seq.size, 0.0, dtype=np.float64)])
            a0 = a0 / np.sum(a0)
        else:
            a0 = init_seq.copy()

    # --------------------------------------------------------------
    # best solution tracking
    best = a0.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # time bookkeeping
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    iteration_global = 0

    # --------------------------------------------------------------
    # Helper to evaluate a candidate and possibly update the best record.
    def _maybe_update(a_vec):
        nonlocal best, best_score, iteration_global
        score = float(evaluate_sequence(a_vec.tolist()))
        iteration_global += 1
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_vec.copy()
            print(f"[improve] iter={iteration_global:,} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase‑1 : soft‑max Adam optimisation
    def _run_phase1(a_start, max_iters, alpha_end):
        a = a_start.copy()
        # Adam hyper‑parameters
        beta1, beta2, eps = 0.90, 0.999, 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        lr0 = 0.12
        alpha = 2.0
        inc_alpha = (alpha_end - alpha) / max_iters if max_iters > 0 else 0.0

        for it in range(1, max_iters + 1):
            if time.monotonic() >= deadline:
                break
            # gradient of the smooth‑max surrogate
            grad, _ = _softmax_gradient(a, alpha)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            lr = lr0 / (np.sqrt(it) + 1.0)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto the simplex
            a = _project_simplex(a)

            # occasional random shake‑up to escape plateaus
            if rng.random() < 0.03:
                idx = rng.integers(0, n)
                factor = float(np.exp(rng.normal(0.0, 0.2)))
                a[idx] = max(0.0, a[idx] * factor)
                a = _project_simplex(a)

            # update best record
            _maybe_update(a)

            # increase temperature
            if alpha < alpha_end:
                alpha = min(alpha_end, alpha + inc_alpha)

    # --------------------------------------------------------------
    # Phase‑2 : hard‑max (exact sub‑gradient) Adam optimisation
    def _run_phase2(a_start, max_iters):
        a = a_start.copy()
        beta1, beta2, eps = 0.90, 0.999, 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        lr0 = 0.08

        for it in range(1, max_iters + 1):
            if time.monotonic() >= deadline:
                break
            # exact sub‑gradient of the max‑convolution entry
            grad, _ = _hardmax_gradient(a)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            lr = lr0 / (np.sqrt(it) + 1.0)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto simplex
            a = _project_simplex(a)

            # tiny random jitter (helps escape shallow local minima)
            if rng.random() < 0.02:
                i, j = rng.integers(0, n, size=2)
                delta = min(a[i], 0.001)
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

            # update best record
            _maybe_update(a)

            # early‑stop if we are already within 2 % of the theoretical lower bound
            theoretical_lb = 2.0 * n / (2 * n - 1)
            if best_score <= theoretical_lb * 1.02:
                print("[early stop] near theoretical bound")
                break

    # --------------------------------------------------------------
    # -------------- MAIN SEARCH LOOP --------------------------------
    # Run the two‑phase optimiser on the initial vector
    _run_phase1(a0, max_iters=200_000, alpha_end=80.0)
    _run_phase2(best, max_iters=300_000)

    # -----------------------------------------------------------------
    # After the first run we spend the remaining time on several
    # random restarts (Dirichlet draws with concentration < 1 to encourage
    # sparse, irregular patterns).
    restart_counter = 0
    while time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        # We need roughly 150 s for another full two‑phase run; skip if not enough time.
        if time_left < 120.0:
            break
        restart_counter += 1

        # random Dirichlet – concentration 0.4 gives a fairly sparse start
        gamma = rng.gamma(0.4, 1.0, size=n)
        a_random = gamma / gamma.sum()
        print(f"[restart #{restart_counter}] starting new random optimisation")

        # phase‑1 with a slightly shorter schedule (as we already have a good solution)
        _run_phase1(a_random, max_iters=150_000, alpha_end=80.0)
        # phase‑2 – also slightly shorter
        _run_phase2(best, max_iters=200_000)

        # occasional status report
        print(f"[status] after restart #{restart_counter} – best score {best_score:.6f}")

    # -----------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {iteration_global:,} total iterations")
    # Return a plain Python list as required by the harness
    return [float(x) for x in best]
```
