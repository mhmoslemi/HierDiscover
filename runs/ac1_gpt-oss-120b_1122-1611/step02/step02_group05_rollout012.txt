import time
import numpy as np

def _normalize(v):
    """Clip negatives, renorm to sum 1 (probability simplex)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # worst case fallback to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s

def propose_candidate(seed: int = 123, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.
    --------------------------------------------------------------------
    The algorithm combines:
      * a smooth‑max surrogate `S_alpha = (1/α)·log Σ exp(α·c_k)` of the true
        max over the self‑convolution `c = a*a`;
      * Adam optimisation on the surrogate gradient `∇S = 2·correlate(w,a)`,
        where `w` are soft‑max weights of the convolution entries;
      * an annealing schedule for the temperature `α` (gradually makes the
        surrogate sharper);
      * FFT‑based convolution / correlation for speed (≈ O(N log N));
      * projection onto the probability simplex after each step.
    The loop runs until the time budget (≈ 1000 s) is exhausted and the
    best sequence found (according to the exact `evaluate_sequence`) is
    returned.
    """
    rng = np.random.default_rng(seed)

    # Problem size – use a longer sequence than before to give more degrees
    n = 2048        # you can adjust; 2048 is a good trade‑off

    # initialise a on the simplex – Dirichlet(0.1) gives a diverse start
    a = rng.dirichlet(np.full(n, 0.1))

    # keep track of the best solution seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n} initial score={best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule (smooth‑max)
    alpha_start, alpha_end = 2.0, 200.0
    alpha = alpha_start
    # we will increase α linearly, a small increment per iteration
    # approximate maximal number of iterations (we do not know in advance)
    # use a very small step so that we never overshoot too early.
    alpha_inc = (alpha_end - alpha_start) / 500_000.0

    # learning‑rate schedule
    lr0 = 0.2

    # Determine FFT length (next power of two ≥ 3·n‑2)
    conv_len = 2 * n - 1
    grad_len = conv_len + n - 1          # 3·n‑2
    N_fft = 1 << ((grad_len - 1).bit_length())
    # safety: ensure N_fft >= conv_len as well
    assert N_fft >= grad_len

    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    iteration = 0
    last_report = time.monotonic()
    while time.monotonic() < deadline:
        iteration += 1

        # --- self‑convolution via FFT ---
        fft_a = np.fft.rfft(a, N_fft)
        conv_full = np.fft.irfft(fft_a * fft_a, N_fft)
        conv = conv_full[:conv_len]          # linear convolution a*a, length 2n‑1

        # --- smooth‑max weights ---
        shift = conv.max()
        # (conv - shift) ≤ 0, exponentials stay bounded
        exps = np.exp(alpha * (conv - shift))
        w = exps / exps.sum()                # length 2n‑1, sums to 1

        # --- gradient = 2 * correlate(w, a) ---
        # correlate(w, a, mode='valid') = conv(w, a[::-1])[n-1 : n-1+n]
        fft_w = np.fft.rfft(w, N_fft)
        fft_a_rev = np.fft.rfft(a[::-1], N_fft)
        corr_full = np.fft.irfft(fft_w * fft_a_rev, N_fft)
        grad = 2.0 * corr_full[n-1 : n-1 + n]   # length n

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # diminishing learning rate (sqrt decay)
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto simplex (clip + renorm)
        a = _normalize(a_new)

        # occasional random shake‑up to escape plateaus
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[idx] *= factor
            a = _normalize(a)

        # evaluate the exact objective occasionally (every 200 iters)
        if iteration % 200 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a.copy()
                print(f"[improve] iter={iteration:,} score={best_score:.6f}")

        # temperature annealing
        if alpha < alpha_end:
            alpha = min(alpha_end, alpha + alpha_inc)

        # periodic status output (once per second)
        now = time.monotonic()
        if now - last_report > 1.0:
            print(f"[status] iter={iteration:,} α={alpha:.2f} best={best_score:.6f}")
            last_report = now

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    return [float(x) for x in best]

Propose a step‑function height sequence that minimises ``evaluate_sequence``.

Algorithm (high‑level)
----------------------
* Initialise a probability vector ``a`` (non‑negative, sum = 1).  If a
  previous best sequence ``height_sequence_1`` exists it is used,
  otherwise a Dirichlet draw gives a diverse start.
* Repeatedly minimise a *smooth‑max* surrogate of the true objective
  ``max_k (a*a)[k]``:
        S_α(a) = (1/α)·log Σ_k exp(α·c_k) ,   c = a*a
  Its gradient is   ∇S_α(a) = 2·correlate(w, a)   with
        w_k ∝ exp(α·c_k)   (soft‑max weights).
* The surrogate is optimised with the Adam optimiser; after each step the
  vector is projected onto the probability simplex (clipping negatives and
  renormalising).
* Temperature ``α`` is annealed from a very smooth value (2.0) up to a
  sharp one (≈200) so that the surrogate gradually approaches the true max.
* FFT‑based convolution / correlation (O(N log N) with a power‑of‑two length)
  makes each iteration fast, allowing many thousands of iterations within
