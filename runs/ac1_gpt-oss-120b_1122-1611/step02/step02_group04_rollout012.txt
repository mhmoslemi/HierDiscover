def compute_conv(a):
    n = a.shape[0]
    m = 1 << (2*n-1).bit_length()  # next power of 2 >= 2n-1
    fft_a = np.fft.rfft(a, n=m)  # rfft returns half-plus-one length

"""
Propose a step‑function height sequence that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)

The optimizer works on the probability simplex (sum = 1, a_i ≥ 0) and
combines two ideas:

1. **Smooth‑max surrogate** – early in the run we minimise a
   log‑sum‑exp upper bound on the max convolution.  This gives a smooth
   gradient that can be followed reliably.

2. **Hard max (sub‑gradient) phase** – after the surrogate temperature
   has become small we switch to the true max (a single delta weight at
   the worst convolution entry).  The update is performed with a
   mirror‑descent (exponentiated‑gradient) step, which automatically
   respects the simplex constraints.

A small amount of random jitter is added regularly to help escape
plateaus.  The algorithm keeps the best sequence seen so far and stops
as soon as the target score (≈1.5025) is reached or the time budget is
exhausted.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑2 FFT (real output)."""
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    fft_a = np.fft.fft(a, n=m)
    conv_full = np.fft.ifft(fft_a * fft_a).real
    return conv_full[: 2 * n - 1]


def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative sequence that yields a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  The function stops a little before this.
    **kwargs
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time limit (tiny safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # decide the length of the vector; reuse a known good one if present
    try:
        known = np.asarray(height_sequence_1, dtype=np.float64)
        if known.ndim == 1 and known.size > 0:
            n = known.size
            a = known.copy()
        else:
            raise ValueError
    except Exception:  # noqa: BLE001
        n = 1024
        a = rng.random(n)

    # ------------------------------------------------------------------
    # make a a feasible point on the simplex
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    # best solution bookkeeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # hyper‑parameters
    #   – smooth‑max (log‑sum‑exp) phase
    tau_factor = 0.7            # relative temperature (as fraction of current max)
    tau_decay = 0.99994         # exponential decay per iteration
    #   – mirror‑descent (exponentiated‑gradient) phase
    eta = 0.25                  # initial step size for EG updates
    eta_decay = 0.99995
    #   – general
    max_iter = 10 ** 9          # effectively unlimited; loop terminates by time
    iter_count = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline and iter_count < max_iter:
        iter_count += 1

        # --------------------------------------------------------------
        # convolution (fast FFT)
        conv = _conv_fft(a)

        # --------------------------------------------------------------
        # decide whether we are still in the smooth‑max regime
        use_smooth = (tau_factor > 1e-4) and (iter_count < 250_000)

        if use_smooth:
            cur_max = conv.max()
            # temperature – never let it become exactly zero
            tau = max(cur_max * tau_factor, 1e-12)

            # stable soft‑max weights
            shifted = (conv - cur_max) / tau
            # clip exponent to avoid overflow (exp(>50) ≈ 5e21 is safe)
            shifted = np.clip(shifted, -50.0, 50.0)
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / w_sum
        else:
            # hard max – a single delta at the worst position
            k_max = int(conv.argmax())
            w = np.zeros_like(conv)
            w[k_max] = 1.0

        # --------------------------------------------------------------
        # gradient of the (smooth) max term:
        #   grad_i = 2 * Σ_k w_k * a_{k-i}
        # This is exactly the correlation of w with a.
        grad = 2.0 * np.correlate(w, a, mode="full")[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # Mirror‑descent (exponentiated‑gradient) update.
        #   a_i ← a_i * exp(-η * grad_i)   then renormalise.
        # η is gradually decreased.
        exp_factor = np.exp(-eta * grad)
        a = a * exp_factor
        s = a.sum()
        if s == 0.0:

        else:
            a = a / s

        # --------------------------------------------------------------
        # occasional random jitter to shake out of plateaus

            a = np.maximum(a + noise, 0.0)

        # evaluate the true objective

        if np.isfinite(score) and score < best_score:

            best = a.copy()

        # update schedules

        # --------------------------------------------------------------
        # early exit if target reached
        if best_score <= 1.5025:
            print("[target reached]")
            break

    # ------------------------------------------------------------------
    total_iters = iter_count
    print(f"[final] best score={best_score:.6f} after {total_iters:,} iterations")
    return [float(x) for x in best]
