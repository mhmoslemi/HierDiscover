#!/usr/bin/env python
"""
Annealed‐softmax Adam optimizer for the self‑convolution peak problem.

The algorithm works on a probability vector w (∑ w = 1, w ≥ 0) and
minimises  max (w * w) .  The objective is scale‑invariant, therefore the
final sequence is simply rescaled to the conventional √(2 n) ℓ¹‑norm.

Key ideas
---------
* Use a soft‑max (log‑sum‑exp) approximation of the max to obtain a smooth
  gradient: ∇L ≈ 2·conv(q, w_rev) with q_i ∝ exp(β·b_i) where b = w*w.
* Increase the inverse temperature β linearly during the run (deterministic
  annealing).  Early on the gradient is an average over many shifts,
  later it concentrates on the current worst shift.
* Apply the Adam optimiser (first‑ and second‑moment estimates) to obtain
  adaptive step sizes on the simplex.
* After each Adam update project back onto the probability simplex and
  explicitly enforce symmetry (w_i = w_{n‑1‑i}), which is natural for this
  convolution problem.
* Occasionally perform a cheap random block perturbation to escape shallow
  local minima, keeping any improvement.
* The whole search runs for the allotted budget (default 1100 s, leaving a
  few seconds for final scaling) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan probabilities a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for lengths up to ~10⁵)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # clip tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, ∑x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) perturbation used as a random jump.
    A random contiguous block is multiplied by a random factor.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose `evaluate_sequence`
    score is as low as possible.  The routine runs a constrained Adam optimisation
    with an annealed soft‑max approximation of the max‑convolution.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.  Re‑use a previously discovered construction if
    # available; otherwise fall back to a default of 8000 entries.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 8000
    else:
        n = 8000

    # --------------------------------------------------------------
    # Initialise w.
    # Prefer the previously found sequence (if any); otherwise use the
    # Catalan distribution.  Normalise to a probability vector (∑w = 1)
    # and enforce symmetry.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.size != n:
                # truncate or pad with zeros to match the chosen length
                if base.size > n:
                    base = base[:n]
                else:
                    base = np.pad(base, (0, n - base.size), mode='constant')
            w = base / np.sum(base)
            print(f"[init] Loaded height_sequence_1 of length {n}")
        except Exception:
            w = catalan_sequence(n)
            w = w / np.sum(w)
            print(f"[init] Fallback to Catalan, length {n}")
    else:
        w = catalan_sequence(n)
        w = w / np.sum(w)
        print(f"[init] Started from Catalan, length {n}")

    # Enforce symmetry and project onto the simplex (just in case)
    w = 0.5 * (w + w[::-1])
    w = simplex_projection(w)

    # --------------------------------------------------------------
    # Helper to compute the current (unscaled) objective:
    #   obj = 2 * n * max(w*w)
    # --------------------------------------------------------------
    def current_obj(v):
        b = convolve_fft(v, v)
        return 2.0 * n * np.max(b)

    best_w = w.copy()
    best_score = current_obj(w)
    print(f"[init] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Adam parameters
    # --------------------------------------------------------------
    lr = 0.2                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(w)        # first moment
    v = np.zeros_like(w)        # second moment

    # --------------------------------------------------------------
    # Annealing schedule for the soft‑max temperature β.
    # β grows from β_start to β_end as the run proceeds.
    # --------------------------------------------------------------
    beta_start = 0.01
    beta_end = 20000.0

    start_time = time.time()
    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() - start_time < max(0.0, float(budget_s) - 5.0):
        # (1) convolution of w with itself
        b = convolve_fft(w, w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b

        # (2) record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # (3) annealed soft‑max over convolution entries
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (budget_s * 0.9))   # ramp during first 90 %
        beta = beta_start + frac * (beta_end - beta_start)

        # soft‑max probabilities q_i ∝ exp(β·(b_i - max_b))
        shift = b - max_b
        exp_vals = np.exp(beta * shift)            # max exponent = 0
        q = exp_vals / np.sum(exp_vals)

        # (4) gradient of the smooth objective
        #    grad = 2·conv(q , w_rev) restricted to length n
        w_rev = w[::-1]
        grad = 2.0 * convolve_fft(q, w_rev)[n - 1 : n - 1 + n]

        # (5) Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # (6) parameter update on the simplex
        w_new = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        w = simplex_projection(w_new)

        # (7) enforce symmetry (helps reduce the peak)
        w = 0.5 * (w + w[::-1])

        # (8) occasional random block perturbation (global jump)
        if iteration % 3000 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = current_obj(cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best score={best_score:.6f}")

        # (9) reset if we stall for a long stretch
        if no_improve > 35000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            w = 0.5 * (w + w[::-1])
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant; we output the
    # conventional √(2 n) ℓ¹‑norm so the numbers stay comfortably sized.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum(best_w)=1 → simple scaling

    # sanity check using the provided evaluator
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best unscaled score={best_score:.6f}, "
          f"evaluate_sequence={final_score:.6f}")

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
