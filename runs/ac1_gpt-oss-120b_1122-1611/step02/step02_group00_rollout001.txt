"""
propose_candidate
-----------------

A hybrid search that combines a cheap parametric “power‑law” scan,
a smoothed soft‑max gradient descent on the convolution peak,
and occasional random block perturbations.  The objective
`2·n·max(w*w)` (with sum w = 1) is invariant under scaling, so we work
with a probability vector `w`.  The gradient of a soft‑max
approximation of the max‑convolution is obtained by a single FFT
convolution, and each iteration is projected back onto the probability
simplex.  The schedule increases the soft‑max temperature (α) and
decreases the step‑size (η) over time, while random jumps help escape
local minima.  The best sequence found is finally rescaled to the
conventional √(2 n) L¹‑norm before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑scaled terms a_i = C(2i,i)/4**i.
    The recurrence a_{i+1}=a_i·(2i+1)/(2i+2) is numerically stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution using a real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two for speed
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Small negative round‑off errors are harmless; clip to 0.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, ∑x = 1}.  Implements the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random perturbation – multiply a random contiguous block
    by a log‑normal factor, optionally symmetrize."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def power_law_sequence(n: int, beta: float) -> np.ndarray:
    """Return a probability vector w_i ∝ (i+1)^{-beta}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-beta)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Main function required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is as low as possible.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.  If a previous construction exists we reuse its
    # length, otherwise pick a moderately large value.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 1:
                raise ValueError
        except Exception:
            n = 8000
    else:
        n = 8000

    # --------------------------------------------------------------
    # 1) Coarse parametric scan over power‑law exponents.
    # --------------------------------------------------------------
    best_w = None
    best_score = np.inf
    betas = np.linspace(0.3, 1.2, 46)      # roughly step 0.02
    for beta in betas:
        w = power_law_sequence(n, beta)
        b = convolve_fft(w, w)
        score = 2.0 * n * np.max(b)
        if score < best_score:
            best_score = score
            best_w = w.copy()
    # also try a few random Dirichlet starts
    for _ in range(8):
        w = rng.random(n)
        w /= w.sum()
        b = convolve_fft(w, w)
        score = 2.0 * n * np.max(b)
        if score < best_score:
            best_score = score
            best_w = w.copy()
    # keep Catalan as a safety net
    cat = catalan_sequence(n)
    w_cat = cat / cat.sum()
    b_cat = convolve_fft(w_cat, w_cat)
    score_cat = 2.0 * n * np.max(b_cat)
    if score_cat < best_score:
        best_score = score_cat
        best_w = w_cat.copy()

    print(f"[init] n={n}, best init score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Smoothed soft‑max gradient descent on the max‑convolution.
    # --------------------------------------------------------------
    w = best_w.copy()
    best_w = w.copy()
    best_score = 2.0 * n * np.max(convolve_fft(w, w))

    start_time = time.monotonic()
    max_time = max(0.0, float(budget_s) - 5.0)   # keep a small safety margin
    iteration = 0
    no_improve = 0

    # hyper‑parameters
    eta0 = 0.6               # initial step size coefficient
    alpha = 5.0              # start temperature for soft‑max
    alpha_max = 200.0        # final temperature
    alpha_mult = 1.001       # multiplicative increase per iteration

    while time.monotonic() - start_time < max_time:
        # ----- convolution and soft‑max -----
        b = convolve_fft(w, w)                 # full length 2n‑1
        max_b = b.max()
        # soft‑max weights (stable)
        exp_vals = np.exp(alpha * (b - max_b))  # ≤ 1, no overflow
        soft = exp_vals / exp_vals.sum()       # sums to 1

        # ----- gradient of the soft‑max approximation -----
        # grad_j = 2 * Σ_i soft_i * w_{i‑j}
        # can be obtained as a correlation; we use convolution with reversed w.
        conv_full = convolve_fft(soft, w[::-1])            # length 3n‑2
        grad = 2.0 * conv_full[n - 1:2 * n - 1]             # slice of length n

        # ----- projected gradient step -----
        eta = eta0 / np.sqrt(iteration + 1.0)
        w_new = w - eta * grad
        w = simplex_projection(w_new)

        # ----- evaluate current candidate -----
        b_curr = convolve_fft(w, w)
        curr_score = 2.0 * n * np.max(b_curr)

        if curr_score < best_score - 1e-12:
            best_score = curr_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ----- occasional random block jump -----
        if iteration % 2400 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best score={best_score:.6f}")
            else:
                # with small probability accept a non‑improving jump to escape flat regions
                if rng.random() < 0.03:
                    w = cand.copy()
                    print(f"[accept‑jump] iter={iteration:6d}")

        # ----- reset if stagnating for a long stretch -----
        if no_improve > 30000:
            w = block_perturbation(best_w, rng)
            w /= w.sum()
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        # ----- increase soft‑max temperature slowly -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mult)

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – map the probability vector back to the classic
    # √(2 n) L¹‑norm (the evaluation is scale‑invariant).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()

    # Verify via the provided evaluator (optional but useful for logging)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(
        f"[final] n={n}, best score (raw)={best_score:.6f}, "
        f"after scaling evaluate={final_score:.6f}"
    )

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
