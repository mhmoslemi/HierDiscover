"""
Hybrid Adam optimiser for flattening the auto‑convolution of a
probability vector.

The objective evaluated by `evaluate_sequence` is
    2·n·max_k (p*p)[k]   with  p_i ≥ 0 , Σ p_i = 1.
Hence we only need to find a probability vector whose auto‑correlation
has a small ∞‑norm.

The algorithm works in two phases:

* Phase 1 – smooth‑max Adam:
  minimise a smooth approximation
      S_α(p) = (1/α)·log Σ_k exp(α·(p*p)[k])
  using Adam on the simplex (Euclidean projection).  The temperature
  α is slowly increased from 1 to a large value (≈200) so that the
  surrogate becomes increasingly sharp.

* Phase 2 – hard‑max Adam:
  after the smooth phase we switch to a true max‑gradient
  (weights w are a one‑hot vector at the current maximal convolution
  index).  This drives the remaining peaks down.

All convolutions are performed with FFTs (real‑valued rfft/irfft) so that
the method comfortably handles sequences of length up to a few ten‑thousands
within the 1000 s budget.

Random multiplicative perturbations (≈5 % of steps) are injected to
escape plateaus.  The best sequence seen (according to `evaluate_sequence`)
is recorded and returned at the end.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a*a using real FFT of length L (L ≥ 2·len(a)-1).
    Returns the first 2·n‑1 entries.
    """
    Fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Fa * Fa, n=L)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _fft_correlation(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Compute 2·correlate(w, a, mode='valid') using FFT.
    w has length 2·n‑1, a has length n.
    Returns a vector of length n.
    """
    a_rev = a[::-1]
    Fw = np.fft.rfft(w, n=L)
    Fa_rev = np.fft.rfft(a_rev, n=L)
    corr_full = np.fft.irfft(Fw * np.conj(Fa_rev), n=L)
    n = a.shape[0]
    start = n - 1
    end = start + n
    return 2.0 * corr_full[start:end]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # problem size – 2048 gives a good balance between expressiveness
    # and speed (FFT based kernels keep the inner loop cheap).
    n = 2048

    # --------------------------------------------------------------
    # initialise a probability vector a (sum = 1)
    init_seq = None
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if cand.size >= 2:
                init_seq = cand / np.sum(cand)
        except Exception:
            init_seq = None

    if init_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    else:
        if init_seq.size != n:
            if init_seq.size > n:
                a = init_seq[:n]
            else:
                a = np.concatenate([init_seq,
                                    np.zeros(n - init_seq.size, dtype=np.float64)])
            a = a / np.sum(a)
        else:
            a = init_seq.copy()

    # keep the best solution ever seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam state
    beta1, beta2, eps = 0.90, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing temperature schedule (soft‑max)
    alpha_start, alpha_end = 1.0, 200.0          # more aggressive than previous
    # number of iterations we devote to the smooth phase
    smooth_iters = 350_000
    # linear increase per iteration
    alpha_inc = (alpha_end - alpha_start) / smooth_iters

    # learning‑rate schedule
    lr0 = 0.20

    # FFT length – same for convolution and correlation
    L_fft = _next_pow2(3 * n - 2)   # ≥ (2n‑1)+(n‑1)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    iteration = 0
    last_improve_iter = 0
    eval_interval = 10            # evaluate every few iterations (cheap)

    while time.monotonic() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # 1) convolution a*a
        conv = _fft_convolve(a, L_fft)               # length 2n‑1

        # ----------------------------------------------------------
        # 2) determine weighting w
        if iteration <= smooth_iters:
            # smooth‑max (softmax) weighting
            alpha = alpha_start + alpha_inc * iteration
            shift = conv.max()
            exps = np.exp(alpha * (conv - shift))
            w = exps / exps.sum()
        else:
            # hard max – one‑hot at the current maximal index
            w = np.zeros_like(conv)
            max_idx = int(conv.argmax())
            w[max_idx] = 1.0
            alpha = alpha_end

        # ----------------------------------------------------------
        # 3) gradient  g = 2·correlate(w, a)  (valid mode)
        grad = _fft_correlation(w, a, L_fft)

        # ----------------------------------------------------------
        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----------------------------------------------------------
        # 5) projection onto the simplex
        a_new = _project_simplex(a_new)

        # ----------------------------------------------------------
        # 6) occasional random multiplicative shake‑up
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.2)))   # log‑normal perturb
            a_new[idx] *= factor
            a_new = _project_simplex(a_new)

        a = a_new

        # ----------------------------------------------------------
        # 7) periodic evaluation
        if iteration % eval_interval == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a.copy()
                last_improve_iter = iteration
                print(f"[improve] iter={iteration:,}, score={best_score:.6f}")

        # ----------------------------------------------------------
        # 8) early‑stop if we have not improved for a long while
        if iteration - last_improve_iter > 200_000:
            # perform a stronger random perturbation to try escaping
            perturb = np.exp(rng.normal(0.0, 0.5, size=n))
            a = a * perturb
            a = _project_simplex(a)
            last_improve_iter = iteration
            print(f"[restart] iter={iteration:,}")

        # ----------------------------------------------------------
        # occasional status report
        if iteration % 100_000 == 0:
            print(f"[status] iter={iteration:,} – best score so far {best_score:.6f}")

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    return [float(x) for x in best]
