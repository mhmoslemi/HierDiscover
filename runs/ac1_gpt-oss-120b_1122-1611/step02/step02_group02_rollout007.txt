def _proj_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid projected‑gradient / soft‑max descent for the min‑max self‑convolution
problem.

The score to minimise is   2·n·max_k (a*a)[k]   with the scale‑invariant
constraint   Σ a_i = 1 , a_i ≥ 0 .
The algorithm proceeds in three phases

  1.  A cheap parametric sweep (symmetric Beta densities) supplies a good
      deterministic start that already beats the “arcsine’’ reference.

  2.  A continuous optimisation phase uses a smooth approximation of the
      max‑operator:
          f_α(a) = (1/α)·log∑_k exp(α·(a*a)[k]) .
      Its gradient is 2·corr(w,a) where w are the soft‑max weights.
      After each gradient step we project back onto the probability simplex.
      The temperature α grows quadratically from a small value to a large
      value, so the early iterations smooth many peaks while the later ones
      concentrate on the true maximum.

  3.  Occasional stochastic perturbations (block scaling, pairwise swaps)
      and a stagnation‑restart mechanism help to escape plateaus.

The implementation is fully vectorised, uses only NumPy and the
already‑provided ``evaluate_sequence`` for final reporting, and respects the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_self(n: int) -> int:
    """Length of FFT that can hold the linear self‑convolution of length‑n vector."""
    return _next_pow2(2 * n - 1)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (real inputs)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max (log‑sum‑exp) weights for the convolution entries.
    The max entry receives weight 1; all others are ≤ 1.
    """
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0.0:
        # numerical safety – fall back to uniform weights
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _gradient_from_weights(weights: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max objective:
        ∇_a  ∑_k w_k (a*a)[k]  =  2 * corr(weights, a)  (valid mode)
    Returns a vector of length len(a).
    """
    # np.correlate with mode='valid' yields exactly the needed sum:
    #   g_i = Σ_k w_k * a_{k-i}  (zero when out of range)
    g = np.correlate(weights, a, mode='valid')
    return 2.0 * g


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x ≥ 0 , Σ x = 1}
    Implements the algorithm of Duchi et al. (2008) – O(n log n).
    """
    # work on a copy to avoid side effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform simplex point
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: force sum = 1 (may be off by 1e‑15)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / w.shape[0]
    else:
        w /= w_sum
    return w


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta‑density shape on the discrete grid.
    a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1}.
    Returned vector is already normalised to sum‑1.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    # avoid overflow for large exponents – work in log‑space if needed
    a = np.exp((alpha - 1.0) * np.log(xs) + (beta - 1.0) * np.log(n - xs + 1.0))
    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= total
    return a


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 12))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a small safety margin) and uses a hybrid
    projected‑gradient / soft‑max descent approach, starting from a
    parametric Beta‑shape sweep.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Determine problem size and initialise a candidate vector
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # fallback to a moderate size if the global is missing / malformed
        n = 4096
        base = None

    # ------------------------------------------------------------------
    # 1) Parameter sweep over symmetric Beta(α,α) shapes (including arcsine)
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_self(n)

    best_a = None
    best_score = np.inf

    # a few deterministic candidates
    alpha_grid = np.linspace(0.20, 1.80, 23)   # includes α=0.5 (arcsine)
    for a_val in alpha_grid:
        a = _beta_shape(n, a_val, a_val)
        conv = _conv_self(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    # a few random Dirichlet draws for extra diversity
    for _ in range(30):
        a = rng.gamma(shape=1.0, size=n)
        a /= a.sum()
        conv = _conv_self(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    # tiny jitter to break perfect symmetry (helps the gradient phase)
    best_a += rng.normal(scale=5e-4, size=n)
    best_a = np.clip(best_a, 0.0, None)
    best_a /= best_a.sum()

    # ------------------------------------------------------------------
    # Keep the best solution seen so far (may be improved later)
    # ------------------------------------------------------------------
    a = best_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len).max()
    print(f"[0.00s] initial sweep best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.5)   # small safety margin

    # step‑size schedule parameters
    eta0 = 0.38          # initial additive step size
    eta_min = 0.004      # eventual step size
    alpha_start = 0.4    # soft‑max temperature at the beginning
    alpha_end = 80.0     # temperature at the end

    # momentum for gradient descent
    beta_mom = 0.86
    momentum = np.zeros_like(a)

    # stagnation handling
    last_improve = time.monotonic()
    stagnation_limit = 28.0   # seconds without improvement → restart

    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = elapsed / total                      # fraction of total budget

        # --------------------------------------------------------------
        # 1) compute self‑convolution and current score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 2) keep the best solution so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) compute soft‑max weights and gradient
        # --------------------------------------------------------------
        alpha = alpha_start + (alpha_end - alpha_start) * (frac ** 2)   # quadratically growing
        weights = _softmax_weights(conv, alpha)                       # length 2n‑1, sum = 1
        grad = _gradient_from_weights(weights, a)                     # shape (n,)

        # --------------------------------------------------------------
        # 4) momentum update
        # --------------------------------------------------------------
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # --------------------------------------------------------------
        # 5) step‑size schedule (linear decay)
        # --------------------------------------------------------------
        eta = eta0 * (1.0 - frac) + eta_min * frac

        # --------------------------------------------------------------
        # 6) additive gradient step followed by projection onto the simplex
        # --------------------------------------------------------------
        a_new = a - eta * momentum
        a = _project_onto_simplex(a_new)

        # --------------------------------------------------------------
        # 7) occasional stochastic exploration (≈2 % of iterations)
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.014:
            a = _random_block_perturb(a, rng)
        elif r < 0.024:
            a = _pairwise_swap_perturb(a, rng)

        # --------------------------------------------------------------
        # 8) restart if stagnation persists for a while
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > stagnation_limit:
            # fresh start – keep the best known point and add a small jitter
            a = best + rng.normal(scale=5e-4, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            momentum[:] = 0.0
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to satisfy evaluate_sequence limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
