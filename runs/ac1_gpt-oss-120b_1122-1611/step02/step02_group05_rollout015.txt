"""
Propose a step‑function height sequence that minimizes `evaluate_sequence`.

The algorithm has two phases:

1. **Warm start** – generate a handful of analytically‑motivated candidates
   (uniform, √‑shaped, 1/√‑shaped, half‑sine) and keep the best one.
2. **Hybrid optimisation**
   * **Phase A (Adam on a smooth‑max surrogate)** – a few hundred thousand
     Adam steps with an increasingly sharp soft‑max (α up to 200) drive the
     convolution maximum down quickly.
   * **Phase B (simulated annealing on the exact objective)** – the remaining
     time is spent in a simple, symmetry‑preserving stochastic search that
     moves probability mass between index pairs.  The objective is evaluated
     with an FFT‑based convolution (O(n log n)) so that tens of millions of
     proposals can be examined within the 1000 s budget.

Both phases respect the probability‑simplex constraint (non‑negative,
sum‑to‑one) and keep the best sequence ever seen.  When the time budget is
exhausted (or the target value 1.5025 is reached) the best sequence is
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Return max_{k} (a*a)[k] using an FFT of length `fft_len`
    (must be ≥ 2*len(a)-1).  The routine works for non‑negative a.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    # only the first 2*len(a)-1 entries correspond to the linear convolution
    return float(conv[: 2 * a.shape[0] - 1].max())


# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective used by `evaluate_sequence` (avoids the internal
    clipping because all entries are safely inside [0,1000]).
    """
    max_c = _max_convolution(a, fft_len)
    s = float(np.sum(a))
    # the guard `s >= 0.01` is always satisfied in our construction
    return 2.0 * n * max_c / (s * s)

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (length 1024) that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – 1024 gives a good trade‑off between quality and speed
    n = 1024
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two (2048)

    # ------------------------------------------------------------------
    # 1) build a few analytically motivated starting points
    starters = []

    # uniform
    starters.append(np.full(n, 1.0 / n, dtype=np.float64))

    # √‑shape (low at edges, high in centre)
    half = n // 2
    left_sqrt = np.sqrt(np.arange(half, dtype=np.float64) + 1.0)
    sqrt_shape = np.concatenate([left_sqrt, left_sqrt[::-1]])
    starters.append(sqrt_shape / sqrt_shape.sum())

    # 1/√‑shape (high at edges, low in centre)
    left_inv = 1.0 / np.sqrt(np.arange(half, dtype=np.float64) + 1.0)
    inv_shape = np.concatenate([left_inv, left_inv[::-1]])
    starters.append(inv_shape / inv_shape.sum())

    # half‑sine shape (zero at edges, max in centre)
    x = np.arange(1, n + 1, dtype=np.float64)
    sine_shape = np.sin(np.pi * x / (n + 1))
    starters.append(sine_shape / sine_shape.sum())

    # if a previous construction exists, reuse it (normalised)
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if prev.size > 0:
                if prev.size != n:
                    if prev.size > n:
                        prev = prev[:n]
                    else:
                        prev = np.concatenate([prev,
                                               np.zeros(n - prev.size, dtype=np.float64)])
                prev = np.maximum(prev, 0.0)
                total = prev.sum()
                if total > 0.0:
                    starters.append(prev / total)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # pick the best of the warm starts (using the exact objective)
    best_vec = None
    best_score = float('inf')
    for cand in starters:
        sc = _score(cand, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    print(f"[init] chosen warm start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # helper to compute the objective with the official evaluator
    # (used only for the final reporting – the fast version is used inside the loop)
    def exact_score(v):
        try:
            return float(evaluate_sequence(v.tolist()))
        except Exception:
            # fallback to the fast version
            return _score(v, n, fft_len)

    # ------------------------------------------------------------------
    # time management
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # safety margin
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # --------------------------------------------------------------
    # Phase A – Adam optimisation on a smooth‑max surrogate
    # --------------------------------------------------------------
    # we allocate roughly 20 % of the total budget to this phase
    phase_a_deadline = start_time + 0.2 * (deadline - start_time)

    a = best_vec.copy()                     # current iterate
    m = np.zeros_like(a)                    # Adam first moment
    v = np.zeros_like(a)                    # Adam second moment
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.20                               # a slightly larger initial step size
    iteration = 0

    # temperature for the smooth‑max – we start very smooth and sharpen dramatically
    alpha_start = 2.0
    alpha_end = 200.0
    alpha = alpha_start
    # linear schedule (≈ 200 k steps, can stop earlier if time runs out)
    alpha_inc = (alpha_end - alpha_start) / 200_000.0

    while time.monotonic() < phase_a_deadline:
        iteration += 1

        # full self‑convolution (needed for the smooth‑max)
        conv = np.convolve(a, a)                # length 2n‑1
        # smooth‑max weights
        shift = conv.max()
        exps = np.exp(alpha * (conv - shift))
        w = exps / exps.sum()
        # gradient of the surrogate
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        if rng.random() < 0.07:

            factor = float(np.exp(rng.normal(0.0, 0.25)))   # log‑normal shake‑up
            a_new[idx] *= factor
            a_new = _project_simplex(a_new)

        a = a_new

        # evaluate with the fast routine – keep the overall best vector
        cur_score = _score(a, n, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[adam improve] iter={iteration:,} score={best_score:.6f}")

        # temperature schedule
        if alpha < alpha_end:
            alpha = min(alpha_end, alpha + alpha_inc)

        # simple progress report every 100 k iterations
        if iteration % 100_000 == 0:
            print(f"[adam status] iter={iteration:,} best={best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.5025:
            print("[early exit] target reached during Adam phase")
            break

    # ------------------------------------------------------------------
    # --------------------------------------------------------------
    # Phase B – Simulated annealing on the exact objective
    # --------------------------------------------------------------
    a = best_vec.copy()                     # start SA from the best Adam result
    cur_score = best_score
    best_score = cur_score
    best_vec = a.copy()

    # parameters for SA
    init_temp = 0.02
    min_temp = 1e-7
    total_sa_time = deadline - time.monotonic()
    iters = 0

    # pre‑compute index ranges for edge ↔ centre moves (left half only – symmetry preserved)
    edge_max = n // 8          # first 1/8 of the left half are “edges”
    centre_min = n // 4        # centre starts at 1/4 of the left half
    half_n = n // 2

    while time.monotonic() < deadline:
        iters += 1
        elapsed = time.monotonic() - start_time
        t_frac = elapsed / (deadline - start_time)          # 0 → 1

        # temperature linearly decays
        temperature = init_temp * (1.0 - t_frac) + min_temp

        # move size (fraction of source mass) also decays
        max_move_frac = 0.25 * (1.0 - t_frac) + 0.005

        # pick source (edge) and destination (centre) with high probability
        if rng.random() < 0.75:
            src = rng.integers(0, edge_max)                # edge index on left half
            dst = rng.integers(centre_min, half_n)          # centre index on left half
        else:
            src = rng.integers(0, half_n)
            dst = rng.integers(0, half_n)

        if src == dst:
            continue

        # maximal transferable mass from the source pair
        delta_max = a[src]
        if delta_max <= 0.0:
            continue
        delta = delta_max * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        # propose a symmetric move (maintain a[i] = a[n‑1‑i])
        new_a = a.copy()
        new_a[src] -= delta
        new_a[n - 1 - src] -= delta
        new_a[dst] += delta
        new_a[n - 1 - dst] += delta

        # compute exact score (fast FFT version)
        new_score = _score(new_a, n, fft_len)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temperature):
            a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = new_a.copy()
                print(f"[sa improve] iter={iters:,} score={best_score:.6f}")

        # occasional status line
        if iters % 200_000 == 0:
            print(f"[sa status] iter={iters:,}  best={best_score:.6f}  cur={cur_score:.6f}")

        # stop if the target has been reached
        if best_score <= 1.5025:
            print("[early exit] target reached during SA phase")
            break

    # ------------------------------------------------------------------
    # final report
    final_eval = exact_score(best_vec)
    print(f"[final] best score (official) = {final_eval:.6f} after {iters:,} SA iterations")
    return [float(x) for x in best_vec]
