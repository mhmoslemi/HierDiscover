def propose_candidate(...):
    seed etc.
    n = 1024
    S = np.sqrt(2*n)
    rng = np.random.default_rng(seed)
    best = None
    best_score = inf
    deadline = time.monotonic() + budget_s - 5.0
    while time.monotonic() < deadline:
        # random initial
        a = rng.dirichlet(np.full(n, 0.1))    # many non-zero
        a = a * S
        # simple iterative refinement
        iter = 0
        step = 0.5
        while time.monotonic() < deadline and iter < 5000:
            # compute conv and max index
            conv = np.convolve(a, a)
            M = conv.max()
            # compute gradient for all max indices
            max_indices = np.where(np.isclose(conv, M, atol=1e-12))[0]
            grad = np.zeros_like(a)
            for k in max_indices:
                # grad_j contribution for this k
                idx = k - np.arange(n)
                mask = (idx >= 0) & (idx < n)
                grad[mask] += 2 * a[idx[mask]]
            grad /= len(max_indices)  # average
            # Exponentiated gradient update
            a = a * np.exp(-step * grad)
            # Renormalize to sum S
            a = a * (S / a.sum())
            # Evaluate
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best = a.copy()
                best_score = score
                print("[improve] score=...", best_score)
            # step schedule
            step = max(step * 0.9995, 0.001)
            # occasional random shake
            if rng.random() < 0.05:
                # move small mass between random indices
                i = rng.integers(n)
                j = rng.integers(n)
                delta = min(a[i] * 0.1, a[i])   # small fraction
                a[i] -= delta
                a[j] += delta
                a = a * (S / a.sum())
            iter += 1
        # random restart again
    return best.tolist()

"""
Optimisation via iterative linear‑programming (LP) and convex combination.

The idea follows the reference scheme: for a current step‑function `f`
(normalised to unit ℓ₁‑norm) we compute its self‑convolution and the
current worst entry `M`.  Keeping `M` fixed we solve a linear program

    max   Σ b_j
    s.t.  b_j ≥ 0,    (f * b)_k ≤ M   (∀k),

which finds a direction `b` that spreads the mass of `f` into the
coordinates where `f` is small while respecting the same worst
convolution bound.  Normalising `b` to a unit sum yields a candidate
`g`.  Updating `f ← (1‑t)·f + t·g` (with a modest step size `t`) strictly
decreases the maximal convolution value and therefore the evaluation
score.  Repeating this inner LP step a few times gives a strong local
improvement; the outer loop performs many random restarts and keeps the
best sequence found.

All arithmetic stays on the probability simplex (ℓ₁‑norm = 1); the
returned sequence is scaled to the conventional sum = √(2 n) before
being handed to `evaluate_sequence`.  The algorithm respects the 1000 s
budget, prints progress, and never returns an empty list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    scale_sum = np.sqrt(2.0 * n)

        return a_prob * scale_sum

            a = _project_simplex(a)

    return final_seq
