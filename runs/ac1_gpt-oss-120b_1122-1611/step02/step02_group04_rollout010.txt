"""
Propose a candidate step‑function sequence with a low evaluate_sequence score.

The algorithm is a hybrid of two ideas:

1. **Simplex‑projected gradient descent** (Adam on a smooth‐max surrogate)
   is run for a short warm‑up.  This quickly reaches the best known
   sequence (≈1.5275 for n=1024).

2. **Linear‑programming “lifting’’ step** (the method described in the
   problem statement).  Given the current distribution `a` with
   `M = max(a*a)`, we solve

        max   Σ g_i
        s.t.  a * g  ≤  M   (convolution, component‑wise)
              g_i ≥ 0

   The LP pushes mass onto indices where `a` is small while guaranteeing
   that the convolution with `a` does not exceed the current worst value
   `M`.  Mixing the obtained `g` back into `a` (= convex combination)
   reduces the worst self‑convolution and thus the objective.
   The LP is solved with `scipy.optimize.linprog` (HiGHS).  Several
   mixing steps are performed until the time budget expires.

Both phases keep the best sequence seen and perform occasional random
perturbations to escape plateaus.  All vectors are kept on the probability
simplex (non‑negative, sum = 1) because scaling does not affect the
evaluate_sequence value.
"""

import time
import numpy as np

# optional SciPy import – the LP phase is the main improvement driver
try:
    from scipy.optimize import linprog
    _HAS_LP = True
except Exception:  # pragma: no cover
    _HAS_LP = False

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _smooth_max_step(a: np.ndarray,
                     lr_base: float,
                     beta1: float,
                     beta2: float,
                     m: np.ndarray,
                     v: np.ndarray,
                     iteration: int,
                     tau_factor: float,
                     tau_decay: float,
                     rng: np.random.Generator) -> tuple:
    """
    One Adam‑based update on the log‑sum‑exp surrogate of max(a*a).
    Returns the new `a`, updated moment vectors and the new tau factor.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)

    # temperature based on current max (never zero)
    current_max = conv.max()
    tau = max(current_max * tau_factor, 1e-12)

    # numerically stable soft‑max weights
    shifted = (conv - current_max) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w / w_sum

    # gradient of the smooth‑max term: 2 * correlation(w, a)
    grad = 2.0 * np.correlate(w, a, mode='full')[n - 1:n - 1 + n]

    # Adam moment update
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** iteration)
    v_hat = v / (1.0 - beta2 ** iteration)

    step = lr_base / (np.sqrt(v_hat) + 1e-8)
    a_next = a - step * m_hat

    # project back onto the simplex
    a_next = _project_simplex(a_next)

    # occasional shake‑up
    if rng.random() < 0.005:
        noise = rng.normal(0.0, 0.015, size=n)
        a_next = a_next + noise
        a_next = np.maximum(a_next, 0.0)
        a_next = _project_simplex(a_next)

    # decay temperature
    tau_factor *= tau_decay
    return a_next, m, v, tau_factor


# ----------------------------------------------------------------------
def _lp_mix_step(a: np.ndarray,
                 mixing_t: float,
                 rng: np.random.Generator) -> np.ndarray:
    """
    Perform one linear‑programming improvement step.

    * Solve   max Σ g_i   s.t.   a * g ≤ M,   g ≥ 0
    * Normalise g to a probability vector.
    * Return the convex combination (1‑t)·a + t·g_norm.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()

    # Build the Toeplitz matrix T where (T @ g)[d] = Σ_i a_{d‑i} g_i.
    # Each column j contains a copy of `a` shifted by `j` positions.
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a

    # Linear program data: maximise Σ g_i  ⇔ minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve with HiGHS (fast for dense medium‑sized problems)
    res = linprog(c, A_ub=T, b_ub=b_ub,
                  bounds=(0, None), method='highs',
                  options={'presolve': True, 'time_limit': 0.5})

    if not res.success:
        # If the LP fails, fall back to a tiny random move.
        g_norm = _project_simplex(a + rng.normal(0.0, 0.001, size=n))
    else:
        g = res.x
        if g.sum() == 0.0:
            g_norm = a.copy()
        else:
            g_norm = g / g.sum()

    a_new = (1.0 - mixing_t) * a + mixing_t * g_norm
    a_new = _project_simplex(a_new)
    return a_new
