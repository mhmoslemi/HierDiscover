"""Propose a low‑score height sequence for the self‑convolution objective.

The algorithm works in two stages.

1️⃣ **Stage 1 – Adam on a smooth approximation of the max‑convolution.**
   We minimise a soft‑max version of the quantity
   `max_i (a*a)[i]` while staying on the probability simplex.
   The temperature μ grows exponentially so the optimisation
   first follows a smooth landscape and later focuses on the true peak.
   An Adam‑style adaptive step‑size together with Euclidean projection
   gives fast descent.

2️⃣ **Stage 2 – Simulated‑annealing hill‑climbing.**
   Starting from the best vector found in stage 1 we perform a large number
   of tiny mass‑transfer moves (pick i,j, move a small fraction of a[i] to a[j])
   and accept them either if they improve the exact objective
   `2·n·max(a*a)` or with a probability that slowly drops to zero.
   A modest “shake‑up’’ (random multiplicative perturbation of one entry)
   is inserted now and then to escape plateaus.

Both stages respect the constraints required by `evaluate_sequence`
(non‑negative entries, sum≈1).  The whole search stops after the supplied
wall‑clock budget (≈ 1000 s) and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0:
        return np.full_like(x, 1.0 / x.size)
    return e / s


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution using an FFT of length a power of two."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=L)
    c = np.fft.irfft(A * A, n=L)[: 2 * n - 1]    # keep only the needed part
    # numerical noise may produce tiny negatives – clip to zero
    np.maximum(c, 0, out=c)
    return c


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by `evaluate_sequence` (fast version)."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


def _smooth_max_gradient(a: np.ndarray, mu: float) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) of the self‑convolution.
    Returns a vector of length len(a).
    """
    conv = np.convolve(a, a)               # exact convolution – cheap for n≈1k
    cmax = conv.max()
    # soft‑max weights (stable)
    w = np.exp(mu * (conv - cmax))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    n = a.size
    # correlation of w with a gives the required gradient
    corr = np.correlate(w, a, mode="full")
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function height sequence that minimises
    `evaluate_sequence`.  The routine respects the time budget `budget_s`
    (seconds) and returns the best sequence found.

    Parameters
    ----------
    seed : int, default 42
        Random seed for reproducibility.
    budget_s : float, default 1100
        Wall‑clock time allowed for the search.

    Returns
    -------
    list[float]
        The candidate sequence (list of floats, sum ≈ 1, all entries ≥ 0).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # timing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    start_time = time.monotonic()
    total_budget = deadline - start_time

    # --------------------------------------------------------------
    # problem size – 1024 is a good trade‑off (you may increase it)
    n = 1024

    # --------------------------------------------------------------
    # build a collection of diverse starting points
    patterns = []

    # uniform
    patterns.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine‑shaped (heavy at the ends)
    idx = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)

    # simple triangular (tent) shape
    half = n // 2
    tent = np.concatenate(
        [np.linspace(1, 2, half, endpoint=False), np.linspace(2, 1, n - half)]
    )
    tent /= tent.sum()
    patterns.append(tent)

    # random Dirichlet
    patterns.append(rng.dirichlet(np.ones(n)).astype(np.float64))

    # previous best, if available – resize/interpolate if needed
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                # linear interpolation / truncation to the required length
                if prev.size < n:
                    x_old = np.linspace(0.0, 1.0, prev.size)
                    x_new = np.linspace(0.0, 1.0, n)
                    prev = np.interp(x_new, x_old, prev)
                else:
                    prev = prev[:n]
            prev = _project_simplex(prev)
            patterns.append(prev)
        except Exception:
            pass  # ignore malformed previous result

    # --------------------------------------------------------------
    # pick the best among the handcrafted starts (fast evaluation)
    best_a = None
    best_score = float("inf")
    for p in patterns:
        p = _project_simplex(p)
        sc = compute_score(p)
        if sc < best_score:
            best_score = sc
            best_a = p.copy()
    print(f"[initial] best start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # ------- Stage 1 : Adam on a smooth approximation ----------------
    a = best_a.copy()
    m = np.zeros_like(a)    # first moment
    v = np.zeros_like(a)    # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # hyper‑parameters
    lr0 = 0.15                # initial learning‑rate factor
    mu_start = 0.5
    mu_end = 4000.0
    phase1_max_iters = 600_000   # iteration count that reaches mu_end

    phase1_deadline = time.monotonic() + 0.65 * (deadline - start_time)

    it_phase1 = 0
    while time.monotonic() < phase1_deadline:
        it_phase1 += 1
        # temperature schedule (exponential increase)
        mu = mu_start * (mu_end / mu_start) ** (min(it_phase1, phase1_max_iters) / phase1_max_iters)

        # smooth‑max gradient
        grad = _smooth_max_gradient(a, mu)

        # Adam update (learning‑rate decays as 1/√t)
        lr = lr0 / math.sqrt(it_phase1)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it_phase1)
        v_hat = v / (1.0 - beta2 ** it_phase1)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        # occasional exact evaluation of the true objective
        if it_phase1 % 50 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[phase‑1 improve] iter={it_phase1:,}, score={best_score:.6f}")
            if best_score <= 1.5025:
                print("[phase‑1 early exit] target reached")
                break

    # --------------------------------------------------------------
    # ------- Stage 2 : Simulated‑annealing hill‑climbing -------------
    a = best_a.copy()
    current_score = compute_score(a)

    # annealing schedules (linear in elapsed fraction)
    step_max = 0.25    # at the beginning we may move up to 25 % of a[i]
    step_min = 0.001   # final fine‑grained moves
    temp_max = 0.02    # early willingness to accept uphill moves
    temp_min = 1e-5    # near‑zero acceptance at the end
    total_iters = it_phase1

    while time.monotonic() < deadline:
        total_iters += 1
        elapsed = time.monotonic() - start_time
        used_frac = elapsed / total_budget

        # linear schedules
        step = step_max * (1.0 - used_frac) + step_min * used_frac
        temperature = temp_max * (1.0 - used_frac) + temp_min * used_frac

        # ----- propose a small mass‑transfer move -----
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if a[i] == 0.0:
            continue
        eps_move = step * a[i] * rng.random()
        if eps_move <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= eps_move
        a_candidate[j] += eps_move
        # numerical safety: re‑project only if something went negative
        if a_candidate[i] < 0.0:
            a_candidate = _project_simplex(a_candidate)

        cand_score = compute_score(a_candidate)
        delta = cand_score - current_score

        if delta < 0.0 or rng.random() < math.exp(-delta / (temperature + 1e-12)):
            a = a_candidate
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                print(f"[phase‑2 improve] iter={total_iters:,}, score={best_score:.6f}")
                if best_score <= 1.5025:
                    print("[phase‑2 early exit] target reached")
                    break

        # ----- occasional larger “shake‑up’’ -----
        if rng.random() < 0.002:
            idx = rng.integers(0, n)
            factor = math.exp(rng.normal(0.0, 0.3))
            a[idx] *= factor
            a = _project_simplex(a)
            cur = compute_score(a)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"[shake improve] iter={total_iters:,}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # final verification (uses the provided `evaluate_sequence`)
    try:
        final_score = float(evaluate_sequence(best_a.tolist()))
        print(f"[final] best_score (official) = {final_score:.6f}")
    except Exception:
        # should never happen – fall back to the fast score
        final_score = best_score
        print(f"[final] best_score (fast) = {final_score:.6f}")

    return [float(x) for x in best_a.tolist()]
