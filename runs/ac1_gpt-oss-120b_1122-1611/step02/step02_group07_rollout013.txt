from scipy.optimize import linprog

def solve_lp(f, M):
    n = len(f)
    # Build constraint matrix for conv(f,b) <= M
    rows = 2*n-1
    # using list comprehension to fill each row's data
    A = np.zeros((rows, n))
    for k in range(rows):
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # b index j = k - i
        # For i as above, coefficient = f[i]
        # So assign A[k,i] = f[i] if j in range
        # Actually b[j] coefficient, not i. Let's build correctly:
        # conv[k] = sum_i f[i] * b[j] where j = k - i
        # So coefficient for variable b[j] is f[i]. So each column j corresponds to i = k - j
        # Equivalent: for each i in i_range, j = k - i, set A[k, j] = f[i].
        for i in range(i_min, i_max+1):
            j = k - i
            A[k, j] = f[i]
    # b_ub vector = M repeated
    b_ub = np.full(rows, M)
    c = -np.ones(n)  # minimize negative sum -> maximize sum
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback to previous f perhaps
        return None

import time
import numpy as np
from scipy.optimize import linprog

def _build_convolution_matrix(f):
    """Return the dense matrix A where (A @ g) equals the full convolution f * g."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j corresponds to f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A

def _solve_minimax_cross(f):
    """
    Solve:
        minimize    t
        subject to  (f * g)[k] <= t   for all k
                    sum(g) = 1
                    g >= 0
    Returns (g, t). If LP fails, returns (None, None).
    """
    n = f.shape[0]
    A = _build_convolution_matrix(f)            # shape (2n-1, n)
    rows = A.shape[0]

    # Append column for the scalar variable t (coefficient -1 in each inequality)
    A_ub = np.hstack([A, -np.ones((rows, 1), dtype=np.float64)])  # (rows, n+1)
    b_ub = np.zeros(rows, dtype=np.float64)

    # Equality: sum(g) = 1   (t does not participate)
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise t  -> c = [0,…,0,1]
    c = np.concatenate([np.zeros(n, dtype=np.float64), np.array([1.0])])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Use HiGHS interior‑point/dual‑simplex solver (fast for moderate size)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success:
        return None, None
    g = res.x[:n]
    t_val = res.x[n]
    return g, t_val

LP‑based iterative optimizer for the step‑function bound.

The algorithm works on the probability simplex (ℓ₁‑normalised and
non‑negative).  Starting from a (possibly random) distribution `f`,
we solve the linear programme

    minimise    t
    subject to  (f * g)[k] ≤ t   for all k
                ∑_i g_i = 1
                g_i ≥ 0

The LP returns the vector `g` that gives the smallest possible worst‑case
cross‑convolution with the current iterate `f`.  After each LP solve we
blend the new point into the current iterate with a decaying step size,
occasionally apply a small multiplicative shake‑up to avoid stagnation,
and keep the best sequence seen so far (according to `evaluate_sequence`).

The loop stops when the time budget is exhausted or the target bound
1.5025 is reached.  The function returns the best sequence found as a
plain list of Python floats.
"""

def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix `A` such that (A @ g) == np.convolve(f, g).
    Shape of `A` is (2·len(f)‑1, len(f)).
    """

    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j corresponds to f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A

    """
    Solve the linear programme

        minimise    t
        subject to  (f * g)[k] ≤ t   for every k = 0 … 2n‑2
                    ∑ g_i = 1
                    g_i ≥ 0

    Returns (g, t) on success, (None, None) otherwise.
    """

    A_conv = _build_convolution_matrix(f)       # (2n‑1, n)

    rows = A_conv.shape[0]

    # Append the scalar variable `t` (coefficient –1 in each inequality)
    A_ub = np.hstack([A_conv, -np.ones((rows, 1), dtype=np.float64)])  # (rows, n+1)
    b_ub = np.zeros(rows, dtype=np.float64)

    # Equality: sum(g) = 1 (t does not appear)
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise t → c = [0,…,0,1]
    c = np.concatenate([np.zeros(n, dtype=np.float64), np.array([1.0])])

    # Bounds: g_i ≥ 0, t ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Solve with HiGHS (fast for medium‑size LPs)
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return None, None

    g = res.x[:n]
    t_val = res.x[n]
    return g, t_val


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function height vector that minimises `evaluate_sequence`.

    The method repeatedly solves a linear programme that finds the vector
    with the smallest possible worst‑case cross‑convolution with the current
    iterate.  The new vector is blended back into the iterate, a tiny random
    shake‑up is applied from time to time, and the best sequence seen is
    recorded.  The routine respects the global ``height_sequence_1`` if it
    exists and uses it as a warm start.
    """
    rng = np.random.default_rng(seed)

    # ---- time handling -------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # 0.5 s safety margin

    # ---- problem size --------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = a0.size
            a = a0.copy()
        except Exception:
            n = 1024
            a = rng.exponential(scale=1.0, size=n)
    else:
        n = 1024
        a = rng.exponential(scale=1.0, size=n)

    # ---- initialise on the simplex (ℓ₁ = 1) ----------------------------
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total

    # ---- keep the best result -----------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ---- main optimisation loop ----------------------------------------
    it = 0
    max_iters = 200                     # generous upper bound, loop also respects time
    blend = 0.5                         # initial blending factor
    no_improve = 0                      # counter for stagnation handling

    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # ----- LP: minimise max (a * g) ---------------------------------
        g, t_val = _solve_minimax_cross(a)
        if g is None:
            print(f"[lp] iteration {it}: LP failed – aborting")
            break

        # ----- normalise g (numerical drift) ----------------------------
        sum_g = g.sum()
        if sum_g <= 0.0:
            print(f"[lp] iteration {it}: zero‑sum g – aborting")
            break
        g = g / sum_g

        # ----- evaluate candidate ----------------------------------------
        score_g = float(evaluate_sequence(g.tolist()))
        if np.isfinite(score_g) and score_g < best_score:
            best = g.copy()
            best_score = score_g
            no_improve = 0
            print(f"[improve] iter={it:,} score={best_score:.6f}  t={t_val:.6f}")
        else:
            no_improve += 1

        # ----- early exit if target reached -------------------------------
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping")
            break

        # ----- blending --------------------------------------------------
        a = (1.0 - blend) * a + blend * g
        a = a / a.sum()                     # re‑normalise to stay on the simplex

        # ----- slowly decay blending factor -------------------------------
        blend = max(0.05, blend * 0.95)

        # ----- occasional random shake‑up (≈10 % of iterations) ----------
        if rng.random() < 0.10:
            mask = rng.random(n) < 0.05
            factor = np.exp(rng.normal(0.0, 0.15, size=n))
            a[mask] *= factor[mask]
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # ----- restart after prolonged stagnation ------------------------
        if no_improve >= 15:
            # perturb the current best to try a new direction
            a = best.copy()
            perturb = np.exp(rng.normal(0.0, 0.20, size=n))
            a *= perturb
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            print("[restart] stagnation – re‑randomising around best")

        # ----- progress report --------------------------------------------
        if it % 5 == 0:
            elapsed = time.monotonic() - (deadline - budget_s + 0.5)
            print(f"[status] iter={it:,}  best={best_score:.6f}  elapsed={elapsed:.1f}s")

    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    # Return a plain Python list – the evaluation function expects a list.
    return [float(x) for x in best]
