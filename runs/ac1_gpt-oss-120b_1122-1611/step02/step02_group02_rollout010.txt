"""
Propose a height sequence that (empirically) minimizes the evaluation
function.  The algorithm consists of three stages:

1. **Quick shape scan** –  test a few simple parametric families
   (uniform, arcsine, and a power‑law   a_i ∝ [(i+½)(n‑i‑½)]^{-p})
   for several candidate lengths.  The best‑scoring shape becomes the
   start point.

2. **Smooth max‑convolution sub‑gradient** –  the true objective is
   `2·n·max(a*a)`.  We approximate the max by a smooth “soft‑max”
   weight
        w_k = exp(α·(c_k‑c_max)),
   where `c = a*a`.  The gradient of the weighted sum
        Σ_k w_k·c_k
   with respect to `a` is the convolution of `a` with the reversed weight
   vector; it can be obtained with a single FFT.

3. **Adam‑style projected gradient descent** –  the gradient is fed to an
   Adam optimizer, the step is taken, then the vector is projected back
   onto the probability simplex (non‑negative entries that sum to 1).
   The learning‑rate and the soft‑max sharpness `α` are annealed.
   Stagnation triggers a cheap restart (arcsine plus jitter).

The whole loop runs until the allotted time (≈ 1000 s) expires and the
best vector found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors x and y using a pre‑allocated FFT length."""
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    return conv[: x.size + y.size - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    w_i ≥ 0, Σ w_i = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _init_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1/√((i+½)(n‑i‑½))."""
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a


def _shape_powerlaw(n: int, p: float) -> np.ndarray:
    """Family a_i ∝ [(i+½)(n‑i‑½)]^{-p}  (p=0.5 gives the arcsine)."""
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = (xs * (n - xs + 1.0)) ** p
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a


def _fast_score(a: np.ndarray, fft_len_self: int) -> float:
    """Fast compute of 2·n·max(a*a) assuming Σ a = 1."""
    conv = _conv_fft(a, a, fft_len_self)
    return 2.0 * a.size * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds and uses a smooth‑max gradient with an Adam optimiser.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    # Leave a small safety margin for the harness (≈1 s)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Determine a set of candidate problem sizes
    # ------------------------------------------------------------------
    candidate_ns = []
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.all(np.isfinite(base)):
            candidate_ns.append(base.size)
    except Exception:
        pass

    # Add a few canonical sizes (powers of two) as fall‑backs
    for n_try in (1024, 2048, 4096, 8192):
        if n_try not in candidate_ns:
            candidate_ns.append(n_try)

    # ------------------------------------------------------------------
    # 1) Quick scan of simple parametric shapes
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_vec = None
    best_n = None

    for n in candidate_ns:
        fft_len_self = _next_pow2(2 * n - 1)

        # Uniform – a simple baseline
        a_uni = np.full(n, 1.0 / n, dtype=np.float64)
        sc_uni = _fast_score(a_uni, fft_len_self)
        if sc_uni < best_score:
            best_score, best_vec, best_n = sc_uni, a_uni.copy(), n

        # Arcsine
        a_arc = _init_arcsine(n)
        sc_arc = _fast_score(a_arc, fft_len_self)
        if sc_arc < best_score:
            best_score, best_vec, best_n = sc_arc, a_arc.copy(), n

        # Power‑law family  p ∈ [0.2, 0.9]
        for p in np.linspace(0.2, 0.9, 13):
            a_pow = _shape_powerlaw(n, p)
            sc_pow = _fast_score(a_pow, fft_len_self)
            if sc_pow < best_score:
                best_score, best_vec, best_n = sc_pow, a_pow.copy(), n

    # If for some reason no vector was found (should never happen) fall back:
    if best_vec is None:
        best_n = 2048
        best_vec = _init_arcsine(best_n)

    # ------------------------------------------------------------------
    # 2) Adam‑style projected gradient descent with smooth‑max gradient
    # ------------------------------------------------------------------
    n = best_n
    a = best_vec.copy()                       # current iterate (always Σa = 1)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # FFT parameters (single length used for both convolutions)
    L_self = 2 * n - 1                 # length of a*a
    L_grad = 3 * n - 2                 # length of a * (reversed weight)
    fft_len = _next_pow2(L_grad)       # big enough for both products
    base_index = L_self - 1            # for extracting the gradient slice

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # Schedule parameters
    lr0 = 0.25                     # initial learning‑rate (before decay)
    alpha_start = 5.0             # soft‑max sharpness start
    alpha_end = 30.0              # soft‑max sharpness end
    stagnation_seconds = 20.0
    last_improve = time.monotonic()

    # Print initial status
    print(f"[0.00s] initial best score = {best_score:.6f} (n={n})")

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # a) FFT of the current vector (shared for both convolutions)
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, n=fft_len)

        # --------------------------------------------------------------
        # b) Self‑convolution (full) and max entry
        # --------------------------------------------------------------
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: L_grad]
        conv_self = conv_full[:L_self]                # a*a
        max_val = conv_self.max()
        cur_score = 2.0 * n * max_val                # Σa = 1 ⇒ no denominator

        # --------------------------------------------------------------
        # c) Record improvement
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # d) Compute smooth‑max weights and gradient via FFT
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time

        # learning‑rate schedule (linear decay to a tiny floor)
        lr = lr0 * (1.0 - frac) + 1e-4

        # soft‑max sharpness schedule (grows linearly)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # soft‑max weights (shape = L_self)
        weights = np.exp(alpha * (conv_self - max_val))

        # gradient = Σ_k w_k ∂(a*a)_k/∂a  =  a * (reversed weights)
        rev_w = weights[::-1]            # length L_self
        rev_w_fft = np.fft.rfft(rev_w, n=fft_len)
        grad_full = np.fft.irfft(a_fft * rev_w_fft, n=fft_len)[: L_grad]
        # extract the n‑dimensional gradient slice
        grad = grad_full[base_index - np.arange(n)]

        # --------------------------------------------------------------
        # e) Adam update + projection onto the simplex
        # --------------------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step
        a = _project_simplex(a)

        # --------------------------------------------------------------
        # f) Occasional stochastic perturbations (helps escape plateaus)
        # --------------------------------------------------------------
        if rng.random() < 0.012:
            # block scaling
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a[start:start + width] *= factor
            a = _project_simplex(a)
        elif rng.random() < 0.004:
            # pairwise mass swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.random() * 0.3 * a[i]
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)

        # --------------------------------------------------------------
        # g) Restart on stagnation
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > stagnation_seconds:
            a = _init_arcsine(n)
            a += rng.normal(scale=0.002, size=n)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best vector found (clipped to the limits enforced by
    # evaluate_sequence).  All entries are already non‑negative and sum to 1.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
