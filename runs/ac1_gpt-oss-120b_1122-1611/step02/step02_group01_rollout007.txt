    fa = np.fft.rfft(a, n=fft_len)

    corr = np.fft.irfft(fa * np.conj(fw), n=fft_len)

Gradient‑based search for a non‑negative step‑function that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works on the probability simplex (sum(a)=1).  Each iteration
does the following:

1. Compute the self‑convolution a*a (FFT, O(n log n)).
2. Build a soft‑max over the convolution entries:
   w_k = exp(α·(conv_k – max_conv)).  For large α this concentrates on the
   current peak, for small α it smooths the gradient.
3. Obtain a smooth sub‑gradient of the (soft) maximum via the linear
   cross‑correlation of a with the weight vector w (FFT again).
4. Apply an Adam‑style exponentiated‑gradient (mirror descent) update:
   a ← a · exp(−η·update)  and renormalise to sum=1.
5. Occasionally perform cheap random block perturbations and pair‑wise
   mass transfers to escape shallow minima.
6. If no improvement has been seen for a while, restart from the best
   solution with a small random shake.

The schedule for the learning‑rate η decays linearly, while the soft‑max
temperature α slowly grows, focusing the search on the dominant peak as
optimization proceeds.  The routine runs for roughly ``budget_s`` seconds
(leaving a small safety margin) and returns the best vector found.
"""

import numpy as np


def _fft_len_for_n(n: int) -> int:
    """Length of the next power‑of‑two sufficient for a linear self‑convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_corr(n: int) -> int:
    """Length of the next power‑of‑two sufficient for the correlation a⋆w."""
    size = 3 * n - 2            # (len(a) + len(w) - 1) with len(w)=2n-1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _corr_grad(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the correlation of ``a`` with the weight vector ``w``:

        g_i = Σ_k w_k * a_{k-i}   (0 ≤ i < len(a))

    The linear correlation is obtained via an FFT (O(n log n)).
    """
    fa = np.fft.rfft(a, n=fft_len)

    start = len(w) - 1               # index that corresponds to shift 0
    return corr[start:start + a.size]
