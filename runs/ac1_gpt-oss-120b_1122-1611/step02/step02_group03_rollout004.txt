"""
Propose a low‑score height sequence for the self‑convolution objective.

Algorithm
---------
1. **Problem size** – we work with n = 2048 (a good trade‑off between
   expressive power and per‑iteration cost).  Convolutions and correlations are
   performed with FFTs (O(n log n)).

2. **Initial patterns** – several hand‑crafted shapes (uniform, arcsine‑like,
   decreasing power‑law, random Dirichlet) are generated and later used as
   restart points.

3. **Smooth‑max Adam optimisation** – at each iteration the full self‑convolution
   `c = a*a` is computed by FFT.  A soft‑max with slowly increasing temperature
   `μ` builds a smooth approximation of `max(c)`.  The gradient of this
   surrogate is `∇a = 2·corr(w, a)` where `w` are the soft‑max weights; the
   correlation is also evaluated by FFT.  An Adam update is taken and the
   iterate is projected back onto the probability simplex (ℓ₁ = 1, aᵢ ≥ 0).

   * `μ` starts small (≈ 0.5) and grows linearly to ≈ 30, moving the optimisation
     from a smooth landscape toward the true max.
   * The Adam learning‑rate is `0.4 / √t`, decaying slowly.
   * Every few iterations a cheap FFT‑based objective estimate is recorded;
     every few thousand iterations the *real* `evaluate_sequence` function is
     called to keep the true best sequence.

4. **Restarts** – after a bounded number of inner steps the algorithm restarts
   from a randomly chosen initial pattern.  All restarts share the same time
   budget, and the best sequence ever seen (according to the true objective)
   is returned.

5. **Termination** – the loop stops when the wall‑clock budget expires or the
   target score (≈ 1.5025) is reached.  The final best sequence is printed and
   returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(x: np.ndarray, mu: float) -> np.ndarray:
    """
    Soft‑max over a vector `x` with temperature `mu`.
    Returns a probability vector that sums to 1.
    """
    c = x.max()
    exp_args = mu * (x - c)          # always ≤ 0, stabilises exp
    w = np.exp(exp_args)
    s = w.sum()
    if s == 0.0:                     # numerical safety
        return np.full_like(w, 1.0 / w.size)
    w /= s
    return w


def _fast_objective(a: np.ndarray, n: int, L_conv: int) -> float:
    """
    Fast (FFT‑based) evaluation of the objective.
    The sequence `a` is assumed to satisfy sum(a) ≈ 1.
    Returns 2·n·max(conv(a, a)).
    """
    A = np.fft.rfft(a, n=L_conv)
    conv = np.fft.irfft(A * A, n=L_conv)[: 2 * n - 1]
    return 2.0 * n * conv.max()

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Temperature schedule for the soft‑max.
    mu_start = 0.5
    mu_end = 30.0
    mu_slope = (mu_end - mu_start) / max(1, budget_s * 200)   # roughly reach mu_end near the end

    # ------------------------------------------------------------------
    # Main optimisation loop.
    while time.monotonic() < deadline:
        # Pick a start pattern at random (implicit restart).
        a = rng.choice(patterns).copy()
        a = _project_simplex(a)          # ensure feasibility (ℓ₁=1, a≥0)

        # Adam accumulators.
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # How many inner steps per restart?  The cheap FFT lets us afford many.
        inner_limit = 20000

        for it in range(inner_limit):
            if time.monotonic() >= deadline:
                break

            total_iters += 1

            # ----------------------------------------------------------
            # 1. Self‑convolution (FFT, full mode).
            A_fft = np.fft.rfft(a, n=L_conv)
            conv = np.fft.irfft(A_fft * A_fft, n=L_conv)[:conv_len]

            # ----------------------------------------------------------
            # 2. Soft‑max weights (smooth approximation of max(conv)).
            mu = mu_start + mu_slope * total_iters
            w = _softmax_weights(conv, mu)

            # ----------------------------------------------------------
            # 3. Gradient = 2·corr(w, a)  (FFT‑based correlation).
            W_fft = np.fft.rfft(w, n=L_corr)
            A_fft_corr = np.fft.rfft(a, n=L_corr)
            corr_full = np.fft.irfft(W_fft * np.conj(A_fft_corr), n=L_corr)[:corr_len]
            grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # length n

            # ----------------------------------------------------------
            # 4. Adam update (followed by simplex projection).
            lr = 0.4 / np.sqrt(total_iters)            # decaying step‑size
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----------------------------------------------------------
            # 5. Projection onto the probability simplex.
            a = _project_simplex(a)

            # ----------------------------------------------------------
            # 6. Occasional random shake‑up to escape plateaus.
            if rng.random() < 0.005:
                idx = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.18))
                a[idx] *= factor
                a = _project_simplex(a)

            # ----------------------------------------------------------
            # 7. Cheap FFT‑based estimate (every 50 iterations).
            if total_iters % 50 == 0:
                est = _fast_objective(a, n, L_conv)
                if est < best_est:
                    best_est = est
                    best_seq = a.copy()
                    print(f"[est] iter={total_iters}, est_score={best_est:.6f}")

            # ----------------------------------------------------------
            # 8. True evaluation using the provided function (every ~2000 it).
            if total_iters % 2000 == 0:
                true_score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(true_score) and true_score < best_score:
                    best_score = true_score
                    best_seq = a.copy()
                    print(f"[true] iter={total_iters}, score={best_score:.6f}")
                    if best_score <= 1.5025:
                        print("[early target reached]")
                        return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # If we never obtained a true score (unlikely), fall back to the cheapest
    # estimate we have and evaluate it once.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    if best_score == float('inf'):
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(
        f"[final] total_iters={total_iters}, best_est={best_est:.6f}, best_score={best_score:.6f}"
    )
    return [float(x) for x in best_seq]
