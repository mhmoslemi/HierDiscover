"""
Propose a step‑function height sequence with a low evaluation score.

The algorithm consists of two phases that run inside the given time budget.

Phase 1 – Adam on a soft‑max surrogate.
    * The objective 2·n·max(conv)/(∑a)² is invariant to scaling,
      therefore we keep the vector on the probability simplex (∑a = 1,
      a_i ≥ 0).
    * A smooth “log‑sum‑exp” approximation of the max is used:
          S_τ(a) = (1/τ)·log Σ_k exp(τ·(a*a)_k)
      Its gradient is a weighted correlation between the weight vector
      (soft‑max over the convolution entries) and the current a.
    * Adam updates with a decaying temperature τ move the vector towards
      a distribution whose self‑convolution is as flat as possible.

Phase 2 – Direct max‑gradient descent with a cooling schedule.
    * The index k* where the convolution attains its maximum is located.
    * The exact gradient of the worst convolution entry is
          ∂(a*a)_{k*}/∂a_i = 2·a_{k*‑i}   (if the index exists, else 0)
    * A simple projected gradient step using this gradient is applied
      repeatedly (the step size slowly cools).  This sharpens the result
      obtained by the smooth surrogate.
    * Occasionally a small random shake‑up is performed to escape plateaus.

Both phases keep the best feasible vector seen so far.
The loop stops early if the theoretical lower bound 2·n/(2·n‑1) is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]                    # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        v_hat = v / (1.0 - beta2 ** iteration)

        step = step_size / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)

        # tiny random shuffle every few hundred steps
        if rng.random() < 0.003:
            noise = rng.normal(0.0, 0.008, size=n)
            a_candidate = a_candidate + noise
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        # evaluate
        score = float(evaluate_sequence(a_candidate.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best = a_candidate.copy()
                max_no_improve = 0
                print(f"[improve‑max] iter={iteration:,} score={best_score:.6f}")
            else:
                max_no_improve += 1

        a = a_candidate
        step_size *= step_cool
        if step_size < 5e-5:
            step_size = 5e-5

        # early stopping if we have not improved for a while
        if max_no_improve > 150000:
            print("[early‑stop‑max] stagnation")
            break

        # stop if we are already essentially at the theoretical bound
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        if best_score <= lower_bound * 1.01:
            print("[early‑stop‑bound] reached lower bound region")
            break

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    return [float(x) for x in best]
```
