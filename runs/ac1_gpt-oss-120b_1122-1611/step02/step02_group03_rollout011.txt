"""
Propose a step‑function height sequence that minimises the self‑convolution
upper bound used by `evaluate_sequence`.

The algorithm has two phases.

1. **Linear‑programming (LP) phase** – starting from a simplex point `a`,
   solve the LP
        maximise   Σ b_j
        subject to  b_j ≥ 0,
                    (a * b)_k ≤ M   for all k,
   where `M = max(a*a)`.  The LP finds a direction `b` that can be added
   to `a` without increasing the current worst convolution value.
   A convex combination `a ← (1‑t)·a + t·b̂` (with `b̂` normalised to sum 1)
   strictly reduces the maximum coefficient.  A few such updates already
   give a noticeable drop.

2. **Smooth‑max Adam phase** – after the LP updates the iterate is refined
   with a projected Adam optimiser that minimises a smooth approximation
   of `max(conv)`.  The soft‑max temperature is increased slowly, the step
   size decays, and tiny random perturbations help escape plateaus.

Both phases keep the vector on the probability simplex (all entries
non‑negative and summing to 1).  The best sequence found before the time
budget expires is returned.  If the target score 1.5025 is reached early,
the function exits immediately.
"""

import numpy as np

    _HAS_CVX = True

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, mu: float) -> np.ndarray:
    """Soft‑max (with temperature `mu`) over the convolution coefficients."""
    cmax = conv.max()
    exp_args = mu * (conv - cmax)          # stabilise the exponentials
    w = np.exp(exp_args)
    s = w.sum()
    if s == 0.0:
        # fall back to a uniform weight vector
        return np.full_like(conv, 1.0 / conv.size)
    return w / s

        while time.monotonic() < deadline and lp_step < max_lp_steps:
            # current worst self‑convolution value
            conv_a = np.convolve(a, a)
            M = conv_a.max()

            # build the Toeplitz matrix that realises the convolution a * b
            # B has shape (2n‑1, n); row k contains a shifted copy of `a`.
            B = np.zeros((2 * n - 1, n), dtype=np.float64)
            for j in range(n):
                B[j:j + n, j] = a

            # Linear programme:
            #   maximise  Σ b_j
            #   s.t.      B @ b ≤ M
            #             b ≥ 0
            b = cp.Variable(n, nonneg=True)
            constraints = [B @ b <= M * np.ones(2 * n - 1)]
            prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True,
                           max_iters=2000, abstol=1e-9, reltol=1e-9,
                           verbose=False)
            except Exception:      # pragma: no cover
                break

            if b.value is None:
                break

            b_sol = np.maximum(b.value, 0.0)
            sum_b = b_sol.sum()
            if sum_b <= 0.0:
                break

            # normalise the direction onto the simplex
            g = b_sol / sum_b

            # convex combination – this reduces the current maximum
            a = (1.0 - t_lp) * a + t_lp * g
            a = _project_simplex(a)          # numerical safety

            # occasional cheap evaluation
            total_iters += 1
            if total_iters % 2 == 0:
                score = evaluate_sequence(a.tolist())
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[LP improve] step={lp_step + 1}, iter={total_iters}, "
                          f"score={best_score:.6f}")

            # decay the mixing weight so later steps become finer
            t_lp = max(0.05, t_lp * 0.85)
            lp_step += 1

            if best_score <= 1.5025:
                print("[early exit] target reached during LP phase")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Phase 2 – Smoothed‑max Adam optimisation (the method that previously
    # achieved ~1.527).  Here we start from the improved LP point and run
    # longer with a slightly different schedule.
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    while time.monotonic() < deadline:
        # full self‑convolution
        conv = np.convolve(a, a)

        # smooth approximation of the max – temperature slowly rises
        mu = 5.0 * (1.0 + total_iters / 40000.0)
        w = _softmax_weights(conv, mu)   # ∂/∂conv of the smooth max

        # gradient of the smooth max (2·corr(w, a))
        corr_full = np.correlate(w, a, mode='full')
        grad = 2.0 * corr_full[n - 1: n - 1 + n]

        # Adam update on the simplex
        lr = 0.2 / np.sqrt(total_iters + 1.0)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (total_iters + 1))
        v_hat = v / (1.0 - beta2 ** (total_iters + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the probability simplex
        a = _project_simplex(a)

        # tiny random shake‑up occasionally (helps jump off plateaus)
        if rng.random() < 0.005:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[idx] *= factor
            a = _project_simplex(a)

        total_iters += 1

        # evaluate every few iterations (cheap enough)
        if total_iters % 15 == 0:
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[Adam improve] iter={total_iters}, score={best_score:.6f}")

            if best_score <= 1.5025:
                print("[early exit] target reached in Adam phase")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Return the best sequence found.
    print(f"[final] total_iters={total_iters}, best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
