The method combines two phases:

1. **Smooth‑max Adam** –  a soft‑max surrogate of the max‑convolution is
   optimised with Adam while the vector is kept on the probability simplex.
   The temperature (α) is increased from a very smooth regime (≈0.5) to a
   sharp one (≈200) so that the surrogate focuses on the actual maximum.

2. **Hard‑max sub‑gradient** –  after the first half of the time budget the
   exact index of the maximal convolution value is used.  The sub‑gradient of
   that term is obtained analytically and an exponential‑gradient (mirror
   descent on the simplex) update is performed.  This fine‑tunes the
   distribution and can push the max‑convolution a little further down.

Both phases use a projection onto the simplex (or a multiplicative update with
renormalisation) to guarantee feasibility (non‑negative entries summing to
1).  Random multiplicative perturbations are applied occasionally to avoid
getting stuck in plateaus.  The best sequence seen during the whole run is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x | x_i≥0, Σx_i=1}. """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal entry of a*a w.r.t. a.
    Parameters
    ----------
    a : np.ndarray (n,)
        Current probability vector (sum = 1, non‑negative).
    conv : np.ndarray (2*n-1,)
        Full self‑convolution a*a.
    Returns
    -------
    grad : np.ndarray (n,)
        Sub‑gradient (2 * a_{k‑i} for all i that participate in the maximal sum).
    """
    n = a.shape[0]
    k = int(conv.argmax())                     # index of the maximal entry
    start_i = max(0, k - (n - 1))
    end_i = min(n - 1, k)
    i = np.arange(start_i, end_i + 1)
    j = k - i
    grad = np.zeros_like(a)
    grad[i] = 2.0 * a[j]                       # derivative of a_i * a_j (both orders)
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Generate a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine runs for roughly the given time budget
    (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – a longer vector gives more freedom while keeping the
    # runtime acceptable (FFT based convolution is O(n log n)).
    n = 2048                                           # try a larger size than before

    # ------------------------------------------------------------------
    # initialise a probability vector (sum = 1)
    a = None
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if cand.ndim > 1:
                cand = cand.ravel()
            cand = np.maximum(0.0, cand)               # clip negatives (should not happen)
            if cand.size != n:
                if cand.size > n:
                    cand = cand[:n]
                else:
                    cand = np.concatenate([cand,
                                           np.zeros(n - cand.size, dtype=np.float64)])
            if cand.sum() < 1e-12:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a = cand / cand.sum()
        except Exception:
            a = None
    if a is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)      # uniform start

    # best solution seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam hyper‑parameters for the smooth‑max phase
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max surrogate
    alpha_start = 0.5
    alpha_end = 200.0
    alpha = alpha_start

    # learning‑rate schedule (Adam)
    lr0 = 0.20

    # ------------------------------------------------------------------
    # time management
    start_time = time.monotonic()
    # we keep a small safety margin (2 s) before the harness kills us
    deadline = start_time + max(0.0, float(budget_s) - 2.0)
    # split the budget roughly 60 % / 40 % between the two phases
    phase1_deadline = start_time + 0.60 * (deadline - start_time)

    # pre‑compute FFT length (next power of two ≥ 2n‑1) for fast convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    iteration = 0
    phase2_iter = 0

    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        phase1 = now < phase1_deadline

        # --------------------------------------------------------------
        # 1) full self‑convolution via FFT (fast, O(n log n))
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_conv = conv.max()

        if phase1:
            # ----------------------------------------------------------
            # smooth‑max surrogate (soft‑max) → gradient = 2·correlate(w, a)
            shift = max_conv
            # exponentials are stable because conv - shift ≤ 0
            exps = np.exp(alpha * (conv - shift))
            w = exps / exps.sum()                     # weight vector (sums to 1)

            # correlation (valid) – length n
            # np.correlate is already reasonably fast for our size
            grad = 2.0 * np.correlate(w, a, mode='valid')

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            # learning‑rate decays gently (≈ 1/√t)
            lr = lr0 / (np.sqrt(iteration) + 1.0)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto the simplex
            a_new = _project_simplex(a_new)

            # occasional random shake‑up (≈ 3 % of steps)
            if rng.random() < 0.03:
                idx = rng.integers(0, n)
                factor = float(np.exp(rng.normal(0.0, 0.2)))   # log‑normal perturbation
                a_new[idx] *= factor
                a_new = _project_simplex(a_new)

            # update temperature (linear ramp based on elapsed fraction)
            fraction = (now - start_time) / (phase1_deadline - start_time)
            alpha = alpha_start + (alpha_end - alpha_start) * fraction

        else:
            # ----------------------------------------------------------
            # hard‑max sub‑gradient phase with exponential‑gradient update
            grad = _max_convolution_subgradient(a, conv)

            # step size decays like 1/√t
            eta = 0.5 / (np.sqrt(phase2_iter + 1.0))
            a_new = a * np.exp(-eta * grad)               # multiplicative update
            a_new = a_new / a_new.sum()                   # renormalise (stays on simplex)

            # a tiny random redistribution now and then (≈2 % of steps)
            if rng.random() < 0.02:
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 != i2:
                    # move a tiny amount from i1 to i2
                    delta = min(a_new[i1], 0.001 * rng.random())
                    a_new[i1] -= delta
                    a_new[i2] += delta
                    a_new = a_new / a_new.sum()

            phase2_iter += 1

        a = a_new

        # --------------------------------------------------------------
        # evaluate the current candidate (cheap, O(n))
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
            print(f"[improve] iter={iteration:,} phase={'1' if phase1 else '2'} "
                  f"score={best_score:.6f}")

        # occasional status dump
        if iteration % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[status] iter={iteration:,}, time={elapsed:.1f}s, "
                  f"best={best_score:.6f}")

        # ------------------------------------------------------------------
        # simple early‑stop: we cannot beat the trivial lower bound 2n/(2n‑1)
        # by a large margin; if we are already within 0.5 % of it we stop.
        theoretical_lb = 2.0 * n / (2 * n - 1)
        if best_score <= theoretical_lb * 1.005:
            print("[early stop] close to theoretical lower bound")
            break

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    # Convert to a plain Python list as required by the evaluation harness
    return [float(x) for x in best]
