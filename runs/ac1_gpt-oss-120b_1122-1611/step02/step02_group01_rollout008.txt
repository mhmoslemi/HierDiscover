def proj_simplex(v):
    """Project vector v onto probability simplex {x >=0, sum=1}"""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

The score  

\[
\text{score}= \frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is *scale invariant* – we can keep `sum(a)=1` at all times and minimise  

\[
\Phi(a)=2\,n\;\max (a*a) .
\]

For a fixed vector `a` the self‑convolution `c = a*a` is easy to obtain by an FFT.
The non‑smooth `max` can be approximated by a *soft‑max*:

\[
p_k=\frac{e^{\alpha\,c_k}}{\sum_j e^{\alpha\,c_j}}\qquad (\alpha>0)
\]

When `α` is small the whole convolution contributes, when `α` is large the weight
concentrates on the current peaks.  
The gradient of the smoothed objective  

\[
F(a)=\sum_k p_k\,c_k
\]

is

\[
\nabla_a F = 2\; (a\!\star\!p) ,
\]

i.e. twice the (linear) cross‑correlation of `a` and `p`.  
Both the convolution and the correlation are obtained with a single FFT
pair, so each optimisation step costs only a few milliseconds even for
`n≈4000`.

The optimisation consists of three phases

1. **Mirror‑descent on the simplex** – multiplicative update  
   `a ← a·exp(-η·∇F)` followed by normalisation.  
   `η` decays while `α` grows, thus we first smooth the objective and later
   focus on the true maximum.

2. **Exploration** – occasional random block perturbations and pair‑wise
   mass transfers (the cheap local moves used in the previous solution).

3. **Restart** – if no improvement is observed for a while we restart from the
   current best solution with a tiny random perturbation.

All updates respect the non‑negativity and the `∑a = 1` constraint, so the
sequence can be handed directly to `evaluate_sequence`.  
The programme runs for the whole time budget (≈ 1000 s) and returns the best
vector it has seen.

```python
"""
Propose a step‑function that minimises

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

using a smooth‑max surrogate, mirror‑descent on the probability simplex,
and occasional random local moves.  The algorithm works for any length
(`height_sequence_1` is used when available) and fits comfortably into the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, ∑x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_transfer(a: np.ndarray, rng: np.random.Generator,
                       max_frac: float = 0.30) -> np.ndarray:
    """
    Try a random transfer of a small amount of mass from i → j.
    The move is kept only if it improves the current score.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = min(a[i], max_frac * a[i]) * rng.random()
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # sum unchanged, just re‑normalise against numerical drift
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using an FFT of length L.
    Returns the first 2·len(a)‑1 entries.
    """
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * a.size - 1]
    # tiny negative values from rounding are harmless
    return conv


def _gradient_via_softmax(a: np.ndarray, conv: np.ndarray,
                         alpha: float, L: int) -> np.ndarray:
    """
    Compute the gradient of the soft‑max surrogate
          F(a)=∑_k p_k·(a*a)_k ,   p_k∝exp(α·(a*a)_k)
    using a single FFT pair.
    The result is 2·(a ⋆ p), i.e. twice the cross‑correlation of a and p.
    """
    # soft‑max weights (stable)
    cmax = conv.max()
    ex = np.exp(alpha * (conv - cmax))
    p = ex / ex.sum()

    # FFT of a (already available from the convolution step) could be reused,
    # but for clarity we recompute it here.
    fa = np.fft.rfft(a, n=L)
    fp = np.fft.rfft(p, n=L)

    # cross‑correlation via conj(FA)·FP  (circular, but length L is large enough)
    corr = np.fft.irfft(np.conj(fa) * fp, n=L)[: a.size]
    grad = 2.0 * corr
    return grad


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) – sum(a) is assumed to be 1."""
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the protected
    ``evaluate_sequence`` score.  The routine runs for approximately
    ``budget_s`` seconds (a few seconds safety margin) and returns the best
    vector found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1.  Initialise a – use the previously found construction if present
    # --------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
            else:
                raise ValueError
        except Exception:
            a = np.full(default_n, 1.0 / default_n, dtype=np.float64)
    else:
        a = np.full(default_n, 1.0 / default_n, dtype=np.float64)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s

    n = a.size

    # --------------------------------------------------------------
    # 2.  Pre‑compute FFT length (must accommodate a (size n) and p (size 2n‑1))
    # --------------------------------------------------------------
    L = _next_pow2(3 * n)          # ≥ 3·n  →  enough for both convolutions

    # --------------------------------------------------------------
    # 3.  Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self_fft(best, L)
    best_score = _score_from_conv(conv_best, n)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # Hyper‑parameters (tuned heuristically)
    eta_start = 0.8
    eta_end = 0.02
    alpha_start = 0.5
    alpha_end = 5000.0
    block_prob = 0.02
    pair_prob = 0.07
    restart_gap = 30.0                # seconds without improvement → restart
    max_frac_transfer = 0.30

    # --------------------------------------------------------------
    # 4.  Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- (a) current convolution and score -------------
        conv = _conv_self_fft(a, L)
        cur_score = _score_from_conv(conv, n)

        # ----- (b) keep the best solution seen so far ----------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- (c) compute soft‑max weights and gradient -------
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)   # ∈[0,1]
        alpha = alpha_start * (1.0 - progress) + alpha_end * progress
        eta = eta_start * (1.0 - progress) + eta_end * progress

        grad = _gradient_via_softmax(a, conv, alpha, L)

        # ----- (d) mirror‑descent update (always stays on simplex)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a_sum

        # ----- (e) occasional exploratory perturbations --------
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < pair_prob:
            a_try = _pairwise_transfer(a, rng, max_frac_transfer)
            # evaluate the trial move (full convolution – cheap enough)
            conv_try = _conv_self_fft(a_try, L)
            score_try = _score_from_conv(conv_try, n)
            if score_try < cur_score:
                a = a_try
                cur_score = score_try
                # best will be updated at the top of the next iteration

        # ----- (f) soft restart if we are stuck ---------------
        now = time.monotonic()
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – re‑starting from best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector found (clip to the limits enforced by
    # evaluate_sequence for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
