"""
Propose a step‑function that minimizes

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm is a hybrid of three ideas:

1. Fast exponentiated‑gradient (EG) descent that uses the K largest
   convolution entries to build a sub‑gradient (the “partner vectors”).
   This aggressively flattens the peak of the self‑convolution.

2. A periodic linear‑programming (LP) move.  For the current sequence
   `f` we solve
        max   Σ g_i
        s.t.  (f * g) ≤ M   (all convolution indices)
              g_i ≥ 0
   where M = max(f*f).  The LP finds a feasible direction `g` whose
   sum is as large as possible while respecting the same peak bound.
   After normalising `g` to unit sum we mix it with `f` (mixing weight
   `t`) and keep the improvement if it reduces the score.

3. Light‑weight local search: occasional random block perturbations
   and pairwise mass transfers to escape shallow minima.

The solver runs for roughly ``budget_s`` seconds (leaving a few seconds
safety margin) and returns the best vector seen.  All numbers are kept
non‑negative and stay inside the bounds enforced by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – top‑level as required
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Return the smallest power‑of‑two length that can hold a full linear
    convolution of two vectors of length ``n``.
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a (real, non‑negative) using FFT.
    Result length is ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a convolution index ``k`` (0‑based) return the vector ``w`` such that
        w[i] = a[k-i]  (0 when the index is out of range).

    This is exactly the gradient of the term (a*a)[k] with respect to ``a``.
    """
    n = a.size
    w = np.zeros_like(a)

    if k < n:
        # overlapping part: i = 0 … k
        w[: k + 1] = a[k :: -1]          # a[k], a[k‑1], …, a[0]
    else:
        # overlapping part: i = k-(n‑1) … n‑1
        start = k - (n - 1)               # first i that has a partner
        length = (2 * n - 1) - k          # number of overlapping entries
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    Afterwards the vector is re‑normalised to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Return the contribution of entry ``a[idx]`` to the full self‑convolution.
    The result is a vector of length ``2*n-1`` with

        v[k] = a[k-idx]  (0 outside the support).

    It is simply a shifted copy of ``a``.
    """
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx : idx + n] = a
    return v


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative height sequence that minimizes the protected
    ``evaluate_sequence`` score.  The routine runs for roughly ``budget_s``
    seconds (leaving a safety margin) and returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose an initial vector (use the global if it exists)
    # ------------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
            else:
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size
        total = a.sum()
        if total <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

    # ------------------------------------------------------------------
    # 2. Book‑keeping
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)
    best = a.copy()
    cur_conv = _conv_self(best, fft_len)
    best_score = float(2.0 * n * cur_conv.max())
    start_time = time.monotonic()
    last_improve = start_time

    # Hyper‑parameters (tuned heuristically)
    eta_initial = 0.7          # EG learning‑rate at the start
    eta_final = 0.02           # EG learning‑rate near the end
    K_active = 30              # number of top convolution entries used for the EG sub‑gradient
    block_perturb_prob = 0.02 # chance to apply a random block perturbation each iteration
    pair_move_prob = 0.07      # chance to attempt a pair‑wise mass transfer each iteration
    lp_every = 30.0            # seconds between LP moves
    t_mix = 0.25               # mixing weight when we accept an LP direction
    restart_gap = 45.0         # seconds without improvement before a soft restart
    max_frac_transfer = 0.30   # maximal fraction of mass moved in a pair move
    max_lp_iters = 12          # total number of LP moves we are willing to spend time on

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    lp_done = 0
    next_lp_time = start_time + lp_every

    while time.monotonic() < start_time + max(0.0, budget_s - 5.0):
        # --------------------------------------------------------------
        # (a) current convolution and score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # --------------------------------------------------------------
        # (b) keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # (c) multi‑peak exponentiated‑gradient step
        # --------------------------------------------------------------
        # pick K largest convolution entries (indices)
        if K_active >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            kth = conv.size - K_active
            partition = np.argpartition(conv, kth)
            active_idx = partition[kth:]

        # aggregated sub‑gradient
        w = np.zeros_like(a)
        for k_idx in active_idx:
            w += _partner_vector(a, int(k_idx))

        # linear decay learning‑rate schedule
        elapsed = time.monotonic() - start_time
        progress = elapsed / (budget_s - 5.0)
        eta = eta_initial * (1.0 - progress) + eta_final * progress
        eta = max(eta, eta_final)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * w)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --------------------------------------------------------------
        # (d) occasional random block perturbation
        # --------------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # (e) pair‑wise mass transfer (fine local search)
        # --------------------------------------------------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], max_frac_transfer * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    # Quick update of convolution via contribution vectors
                    contrib_i = _contribution_vector(a, i)
                    contrib_j = _contribution_vector(a, j)
                    new_conv = conv + delta * (contrib_j - contrib_i)
                    new_score = 2.0 * n * float(new_conv.max())
                    if new_score < cur_score:
                        a[i] -= delta
                        a[j] += delta
                        a = np.clip(a, 0.0, None)
                        a /= a.sum()
                        conv = new_conv
                        cur_score = new_score
                        cur_max = float(conv.max())

        # --------------------------------------------------------------
        # (f) periodic LP move (global direction)
        # --------------------------------------------------------------
        now = time.monotonic()
        if lp_done < max_lp_iters and now >= next_lp_time:
            # Build the full convolution matrix A (shape (2n-1, n)).
            # Row k corresponds to partner_vector(f, k).
            A = np.empty((2 * n - 1, n), dtype=np.float64)
            A.fill(0.0)
            for col in range(n):
                A[col : col + n, col] = a  # fills the Toeplitz structure

            # RHS: M on every row
            M = cur_max
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            # Linear program: maximize Σ g_i  <=> minimize -Σ g_i
            c = -np.ones(n, dtype=np.float64)

            # No explicit upper bounds – the A‑matrix already bounds all variables.
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True, "time_limit": 30},
            )
            if res.success:
                g = res.x
                sum_g = g.sum()
                if sum_g > 0.0:
                    # Normalise to unit sum (the same scale we use throughout)
                    g = g / sum_g

                    # Mix candidate and keep it if it improves the score
                    a_candidate = (1.0 - t_mix) * a + t_mix * g
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate /= a_candidate.sum()
                    conv_cand = _conv_self(a_candidate, fft_len)
                    cand_score = 2.0 * n * float(conv_cand.max())
                    if cand_score < cur_score:
                        a = a_candidate
                        conv = conv_cand
                        cur_score = cand_score
                        cur_max = conv.max()
                        # successful LP move – we reset the LP timer a bit later
                        next_lp_time = now + lp_every
                # regardless of success we count the attempt
                lp_done += 1

        # --------------------------------------------------------------
        # (g) soft restart when we are stuck
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # modest multiplicative log‑normal noise on the whole vector
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            print(
                f"[{now - start_time:6.2f}s] restart – re‑starting from best score {best_score:.6f}"
            )
            # after a restart we postpone the next LP a little
            next_lp_time = now + lp_every

    # ------------------------------------------------------------------
    # 4. Return the best vector (clipped to the bounds enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
