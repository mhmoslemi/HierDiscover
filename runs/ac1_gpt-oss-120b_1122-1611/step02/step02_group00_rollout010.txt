def cross_corr(a, b):
    N = 1 << (len(a) + len(b) - 2).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    # Note: correlation = ifft(Fa * conj(Fb))
    corr_full = np.fft.irfft(Fa * np.conj(Fb), N)
    L = len(a) + len(b) - 1
    corr = corr_full[:L]  # linear correlation
    return corr

"""
Propose a better step‑function by minimizing the maximum of the
self‑convolution using a smoothed max‑approximation.
The objective
    max_k (w*w)[k]   (w ≥ 0,  ∑w = 1)
is convex.  We replace the max by a soft‑max
    f(w) = (1/β)·log Σ_k exp(β·(w*w)[k])
whose gradient can be written in closed form as a weighted convolution.
A projected Adam / sub‑gradient descent on the probability simplex is
performed, combined with occasional random block perturbations
(to escape shallow local minima).  The algorithm works for any
length (default: the length of the previously found best sequence, or
10 000) and runs for the whole time‑budget (≈1000 s).  The returned
sequence is finally scaled to the conventional √(2 n) ℓ¹‑norm, which
does not change the evaluation score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First `length` Catalan numbers a_i = C(2i,i)/4**i.
    Recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) is stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # clip tiny negatives
    return c


def cross_corr_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation (a ⋆ b) using real FFT.
    Returns the full correlation of length len(a)+len(b)-1,
    with lags from -(len(b)-1) … len(a)-1."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    # correlation = irfft( Fa * conj(Fb) )
    corr_full = np.fft.irfft(Fa * np.conj(Fb), N)
    return corr_full[:n]


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump – multiply a random block by a random factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetric averaging (helps for even lengths)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main search routine required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) with a low
    `evaluate_sequence` score.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 10000
    else:
        n = 10000

    # --------------------------------------------------------------
    # Initialise w on the simplex.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            w0 = np.asarray(height_sequence_1, dtype=np.float64)
            if w0.size != n:
                # pad or truncate to the chosen length
                if w0.size < n:
                    pad = n - w0.size
                    left = pad // 2
                    right = pad - left
                    w0 = np.pad(w0, (left, right), mode='constant')
                else:
                    w0 = w0[:n]
            w = w0.copy()
        except Exception:
            w = catalan_sequence(n)
    else:
        w = catalan_sequence(n)

    # Normalise to a probability vector (∑w = 1)
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w.sum()

    # --------------------------------------------------------------
    # Book‑keeping.
    # --------------------------------------------------------------
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)      # same as evaluate_sequence after scaling
    best_w = w.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Parameters for the smoothed‑max (softmax) approximation
    beta0 = 0.5          # start with a very smooth max
    beta_max = 200.0    # final beta (almost exact max)
    # Rough guess of total iterations we can afford; used for beta schedule
    target_iters = 200_000
    beta_inc = (beta_max - beta0) / target_iters

    # Adam‑style moments (optional – works well with simplex projection)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr_base = 0.5       # base learning rate (scaled by 1/√t in the code)
    iteration = 0
    no_improve = 0

    start_time = time.monotonic()
    max_time = max(0.0, float(budget_s) - 5.0)   # keep a few seconds buffer

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < max_time:
        # ---- 1) current convolution and score ---------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b

        if score < best_score - 1e- nine:   # strict improvement
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ---- 2) smoothed‑max weighting ---------------------------------------
        # beta grows slowly up to beta_max
        beta = min(beta_max, beta0 + beta_inc * iteration)

        # compute p ∝ (b / max_b)^beta  (soft‑max weights)
        if max_b <= 0.0:
            p = np.full_like(b, 1.0 / b.size)
        else:
            # divide by max to keep numbers in [0,1]
            ratio = b / max_b
            # power‑law weighting (equivalent to soft‑max with log‑space)
            p = np.power(ratio, beta, dtype=np.float64)
            # add a tiny epsilon to avoid a completely zero vector when beta is large
            if p.sum() == 0.0:
                p = np.full_like(b, 1.0 / b.size)
        p_sum = p.sum()
        p /= p_sum

        # ---- 3) gradient of the smoothed objective ---------------------------
        # grad_j = 2 * Σ_k p_k * w_{k-j}
        corr = cross_corr_fft(p, w)               # full correlation (len = 3n‑2)
        # take the first n entries (lags –(n‑1)…0) and reverse them
        grad = 2.0 * np.flip(corr[:n])

        # ---- 4) Adam update on the simplex ------------------------------------
        iteration += 1
        # learning rate decays as 1/√t (helps stability)
        lr = lr_base / np.sqrt(iteration)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        w_tmp = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        w = simplex_projection(w_tmp)

        # ---- 5) occasional random jumps ---------------------------------------
        if iteration % 2000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[block] iter={iteration:6d}, new best score={best_score:.6f}")

        # ---- 6) reset after long stagnation ------------------------------------
        if no_improve > 30_000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

    # ------------------------------------------------------------------
    # Final scaling – return the standard √(2 n) ℓ¹‑norm version.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # One final validation (the harness will also call evaluate_sequence)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Convert to plain list.
    return [float(x) for x in final_seq]
