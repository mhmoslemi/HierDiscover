"""
propose_candidate
-----------------

Entropic mirror‑descent with a smooth approximation of the max‑convolution
objective.  The original problem is

    minimise   max_k (w * w)[k]    subject to   w_i ≥ 0 ,  Σ w_i = 1,

where “*” is the linear discrete convolution.  The objective is invariant to
scaling, therefore we keep the vector on the probability simplex.  The
non‑smooth max is replaced by a log‑sum‑exp surrogate

    F_α(w) = max(b) + (1/α)·log Σ exp(α·(b‑max(b)))   ,   b = w*w ,

which converges to max(b) as α → ∞.  Its gradient is

    ∂F_α/∂w_i = 2· Σ_j w_j·p_{i+j}   with   p_k ∝ exp(α·(b_k‑max(b))).

The gradient is a correlation that can be evaluated with a single FFT‑based
convolution.  An entropic (mirror‑descent) update

    w ← w ⊙ exp(‑η·∇F_α) ,   w ← w / Σ w

keeps all coordinates strictly positive and enforces the simplex
constraint automatically.  After each step we symmetrise the vector
(w := (w+w[::-1])/2) because any optimal solution can be taken
palindromic – this reduces the effective dimension and prevents
off‑centre spikes.

The algorithm starts from several sensible initialisations
(Catalan, uniform, sine‑squared, and the previously‑found best sequence)
and performs many iterations within the time budget.  The smoothing
parameter α is increased slowly (α ← min(α·1.02, 500)) so that the
 optimisation first follows a smooth landscape and later focuses on the
true maximum.  Occasionally a cheap block‑perturbation (the same routine
used in the reference solution) is applied to the current best candidate
to escape possible local minima.

Finally the best probability vector is rescaled to the conventional
‖a‖₁ = √(2 n) normalisation and returned as a plain Python list; this

# ----------------------------------------------------------------------
# Helper functions – top level as required by the harness.

def catalan_sequence(length: int) -> np.ndarray:
    """Return the first `length` Catalan numbers divided by 4**i.
    The recurrence a_{i+1}=a_i·(2·i+1)/(2·i+2) is numerically stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast up to n≈30 000)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off artefacts
    np.maximum(c, 0.0, out=c)
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random jump: multiply a random contiguous block by a log‑normal
    factor and optionally symmetrise.  The result stays non‑negative.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

        log_w -= eta * grad
        w_new = np.exp(log_w)
        w_new /= np.sum(w_new)                # renormalise to Σ w = 1
        # enforce symmetry (palindromic) – reduces effective dimensionality
        w_new = (w_new + w_new[::-1]) / 2.0
        w = w_new

        # ---- schedule / occasional actions ----
        if iteration % epoch_len == 0 and iteration > 0:
            # make the surrogate sharper
            alpha = min(alpha * alpha_inc, max_alpha)

        if no_improve > max_no_improve:
            # random restart from a different initialisation
            w = init_fns[rng.integers(len(init_fns))]()
            no_improve = 0
            print(f"[restart] iter={iteration:6d}, new init, α={alpha:.2f}")

        if iteration % block_interval == 0 and iteration > 0 and best_w is not None:
            # try a cheap random jump from the current best
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand = (cand + cand[::-1]) / 2.0   # keep symmetry
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                no_improve = 0
                print(f"[block] iter={iteration:6d}, new best score={best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling to the conventional √(2 n) L¹‑norm.
    # (The evaluation function is scale‑invariant, this step is only for
    # readability of the returned numbers.)
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
