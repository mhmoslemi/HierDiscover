"""
Propose a non‑negative step‑function (list of heights) with a low evaluation score.

Algorithm
---------
1. Choose a problem size `n` (here 2048 – a good trade‑off of quality vs. speed).
2. Initialise a probability vector `a` on the simplex (Dirichlet + uniform mix).
3. Main optimisation loop (≈ 1000 s):
   * Compute the full self‑convolution `c = a * a` by FFT (O(n log n)).
   * **Warm‑up phase** (first ~20 % of the time):
        – Use a smooth‑max surrogate `S_α(c) = (1/α)·log Σ exp(α·c_k)`.
        – Gradient = `2·correlate(w, a)` where `w` are soft‑max weights.
        – Adam update + projection onto the simplex.
        – Temperature `α` is increased linearly to sharpen the surrogate.
   * **Fine phase** (remaining time):
        – Identify the maximal convolution index `k*`.
        – Use the exact sub‑gradient `∇ = 2·correlate(e_{k*}, a)`,
          which reduces to a simple index shift (`∇_i = 2·a_{k*‑i}` if in range).
        – Adam update + projection.
   * Occasionally apply a random multiplicative shake‑up to escape plateaus.
   * Keep the best vector seen (score = `2·n·max(c)` because `sum(a)=1`).

The use of FFT makes each iteration cheap, allowing many more steps than the
previous implementation and a more aggressive refinement of the peak
convolution.  The combination of a smooth‑max warm‑up and a hard‑max
sub‑gradient fine‑tuning usually drives the evaluation score below the
target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper: next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a using real FFTs."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[: 2 * n - 1]


def _gradient_smoothmax(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate.
    Returns 2·correlate(w, a) in “valid” mode (length = len(a)).
    Implemented via FFT for speed.
    """
    n = a.shape[0]
    a_rev = a[::-1]

    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv = np.fft.irfft(W * A_rev, n=L)

    start = n - 1
    end = start + n
    corr = conv[start:end]
    return 2.0 * corr


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Generate a step‑function (list of non‑negative floats) that minimises
    `evaluate_sequence`.  The routine runs for roughly the given time budget
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # problem size – a larger n gives more degrees of freedom.
    # 2048 works well within the 1000 s budget.
    n = 2048

    # --------------------------------------------------------------
    # initialise a probability vector on the simplex.
    # Mix a Dirichlet draw (to break symmetry) with the uniform vector.
    dir_alpha = 0.5
    a = rng.dirichlet(np.full(n, dir_alpha))
    a = 0.5 * a + 0.5 * np.full(n, 1.0 / n)  # gentle smoothing
    a = _project_simplex(a)                 # ensure exact feasibility

    # keep the best ever seen
    conv = _full_convolution_fft(a)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser state
    beta1, beta2 = 0.90, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smooth‑max temperature schedule
    alpha_start, alpha_end = 2.0, 30.0
    alpha = alpha_start

    # learning‑rate schedule
    lr0 = 0.12

    # split the time budget: warm‑up (smooth‑max) ≈ 20 %,
    # fine‑tuning (hard‑max) the rest.
    start_time = time.monotonic()
    total_budget = float(budget_s)
    warmup_deadline = start_time + 0.20 * total_budget

    iteration = 0
    while time.monotonic() < start_time + max(0.0, total_budget - 2.0):
        iteration += 1

        # --------------------------------------------------------------
        # current objective value (no extra convolution needed)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[improve] iter={iteration:,}, time={elapsed:.1f}s, score={best_score:.6f}")

        # --------------------------------------------------------------
        # decide which gradient to use
        now = time.monotonic()
        if now < warmup_deadline:
            # ---- smooth‑max (soft max) gradient --------------------
            shift = max_conv
            exps = np.exp(alpha * (conv - shift))  # shift for numerical stability
            w = exps / exps.sum()
            grad = _gradient_smoothmax(w, a)

            # linearly increase temperature
            alpha = min(alpha_end,
                        alpha_start + (alpha_end - alpha_start) *
                        ((now - start_time) / (warmup_deadline - start_time)))
        else:
            # ---- hard‑max (sub‑gradient) ---------------------------
            k_star = int(np.argmax(conv))
            # gradient_i = 2 * a_{k*‑i} when the index is inside range
            i = np.arange(n)
            j = k_star - i
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[j[mask]]

        # --------------------------------------------------------------
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random shake‑up to jump out of plateaus
        if rng.random() < 0.03:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.3)))  # log‑normal factor
            a_new[idx] *= factor

        # projection back onto the simplex (keeps sum = 1, a_i ≥ 0)
        a_new = _project_simplex(a_new)

        a = a_new

        # recompute convolution for the next iteration
        conv = _full_convolution_fft(a)

        # early exit if we are already at the target region
        if best_score <= 1.5026:
            print("[early stop] reached target vicinity")
            break

        # occasional status line (keeps the output readable)
        if iteration % 20000 == 0:
            print(f"[status] iter={iteration:,}, best_score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    # return a plain Python list as required by the evaluation harness
    return [float(x) for x in best]
