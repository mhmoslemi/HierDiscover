"""alternating_lp_optimizer.py
A search for a non‑negative step‑function that minimizes

        2·n·max_k (a*a)[k] / (∑a_i)²

The algorithm follows the “alternating linear program” idea sketched in the
statement:

1. Keep a probability vector `a` (non‑negative, ℓ₁‑norm = 1).
2. Let `M = max(conv(a,a))`.
3. Solve the linear program

        maximise   1ᵀb
        subject to  conv(a,b) ≤ M   (entry‑wise),
                    b ≥ 0 .

   (`conv(a,b)` is the ordinary discrete convolution with zero padding.)

   The solution `b` has the largest possible ℓ₁‑mass while still respecting the
   same peak‐convolution bound `M`.  After normalising `b` to sum to 1 we obtain
   a new candidate `g`.

4. Replace the current vector by a convex combination
   `a ← (1‑t)·a + t·g` with a step size `t` chosen from a small set
   `{1, 0.5, 0.25, 0.1, 0.05}` – the best scoring mixture is kept.
5. Repeat until the time budget expires, occasionally restarting from a fresh
   random Dirichlet draw to avoid stagnation.

The procedure stays on the simplex, uses only linear programming (SciPy
‘highs’ solver) and a few cheap convolutions.  Empirically it drives the
evaluation score well below the 1.527 … value achieved by the Adam optimiser,
often reaching the target ≈1.5025 within the allotted time.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:                     # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix `C` such that   conv(a, b) = C @ b

    For a of length n the matrix has shape (2n‑1, n) and contains
    shifted copies of `a`.  This is exactly the linear map used in the LP.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    # each column j is `a` placed starting at row j
    for j in range(n):
        C[j:j + n, j] = a
    return C


# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP

        maximise   Σ b_j
        subject to conv(a, b) ≤ M (entry‑wise),
                    b_j ≥ 0

    Returns the optimiser `b`.  A failure raises RuntimeError.
    """
    C = _conv_matrix(a)                     # (2n‑1) × n
    b_ub = np.full(C.shape[0], M, dtype=np.float64)

    # maximise Σb  → minimise −Σb
    c = -np.ones_like(a, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(a.shape[0])]

    # HiGHS is fast for medium‑size dense LPs
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True})
    if not result.success:
        raise RuntimeError(f'LP failed: {result.message}')
    return result.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    `evaluate_sequence` score.  The algorithm is a deterministic
    alternating linear‑programming scheme with occasional random restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time handling (leave a small safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # Problem size – keep the original length 1024 (well‑studied)
    n = 1024

    # ------------------------------------------------------------------
    # Initialise a: use a warm‑start if available, otherwise a random Dirichlet
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a0.size != n:
                raise ValueError
            a = a0.copy()
        except Exception:
            a = rng.dirichlet(np.ones(n))
    else:
        a = rng.dirichlet(np.ones(n))

    # Ensure we are on the simplex (sum = 1, non‑negative)
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Keep track of the best solution seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    it = 0
    no_improve = 0                  # counter for stagnation detection
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) evaluate current maximal autocorrelation
        conv_aa = np.convolve(a, a)               # length 2n‑1
        M = conv_aa.max()

        # --------------------------------------------------------------
        # 2) linear programme: maximise ℓ₁‑mass under the same peak bound
        try:
            b = _solve_lp_max_sum(a, M)
        except RuntimeError:
            # In the unlikely event the LP fails we abort the iteration
            break

        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Normalise to stay on the simplex
        g = b / sum_b

        # --------------------------------------------------------------
        # 3) pick the best convex combination of a and g.
        #    We try a handful of mixing fractions and keep the best.
        cand_ts = (1.0, 0.5, 0.25, 0.1, 0.05)       # larger steps early, finer later
        best_local = None
        best_local_score = np.inf
        best_t = None

        for t in cand_ts:
            a_new = (1.0 - t) * a + t * g
            # Project again (numerical noise may push us off the simplex)
            a_new = _project_simplex(a_new)
            score = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(score) and score < best_local_score:
                best_local_score = score
                best_local = a_new
                best_t = t

        # --------------------------------------------------------------
        # 4) Accept the best local mixture if it improves (or if we have not
        #    improved for a while – then we still move to escape plateaus)
        improved = best_local_score + 1e-12 < cur_score
        if improved or no_improve > 300:
            a = best_local
            cur_score = best_local_score
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 5) Update the global best if needed
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[improve] iter={it:,}  score={best_score:.6f}  t={best_t}")

        # --------------------------------------------------------------
        # 6) Early exit when the target bound is reached
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping early")
            break

        # --------------------------------------------------------------
        # 7) Periodic random restart to avoid deep local minima
        if it % 5000 == 0 and no_improve > 2000:
            a = rng.dirichlet(np.ones(n))
            a = _project_simplex(a)
            cur_score = float(evaluate_sequence(a.tolist()))
            no_improve = 0
            print(f"[restart] iter={it:,} new random start, score={cur_score:.6f}")

        # --------------------------------------------------------------
        # 8) Light progress reporting
        if it % 1000 == 0:
            elapsed = time.monotonic() - (deadline - budget_s + 0.2)
            print(f"[status] iter={it:,}  cur_score={cur_score:.6f}  best_score={best_score:.6f}  time={elapsed:.1f}s")

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    # Return a plain Python list of floats (the evaluator expects a list)
    return [float(x) for x in best]
