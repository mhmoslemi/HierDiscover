"""
Mirror‑descent construction for a step‑function with minimal
self‑convolution peak.

The evaluation metric is
    2·n·max_k (a*a)[k] / (∑a)^2 .
Because the score is homogeneous of degree 0 we keep the mass
normalised (∑a = 1) at all times.  The algorithm works on a vector
`a` of length `n` (the length of the supplied `height_sequence_1`,
or 2048 as a fallback) and proceeds as follows:

1. Initialise `a` with a smooth “arcsine’’ density (the analytical
   minimiser in the continuous case) plus a tiny random jitter.

2. In each iteration:
   * Compute the linear self‑convolution `c = a*a` by FFT.
   * Form a smooth approximation of the max‑operator:
         w_k ∝ exp(γ·(c_k – max(c)))     (γ grows from 0.5 → 2000)
     The vector `w` is a probability distribution over convolution
     indices, giving larger weight to the largest entries.
   * The sub‑gradient of the smoothed maximum is the weighted sum of
     the partner vectors `∂(a*a)[k]/∂a`.  This can be obtained in
     O(n log n) via a cross‑correlation performed with FFT:
         g = Σ_k w_k·partner_k(a)
   * Apply a mirror‑descent (exponentiated‑gradient) update with a
     linearly decaying step‑size η(t) (0.6 → 0.02) and exponential
     momentum β = 0.9:
         m ← β·m + (1‑β)·g
         a ← a·exp(‑η·m)
     Afterwards clip tiny values, renormalise to sum = 1 and keep
     `a` non‑negative.

3. Occasionally (≈2 % of iterations) perform a cheap stochastic
   perturbation (block scaling or pairwise mass swap) to escape
   plateaus.

4. If the best score has not improved for `stagnation_limit` seconds
   the search restarts from a fresh arcsine or Dirichlet sample
   (preserving the current best solution).

The loop runs until the allotted wall‑clock time (≈ 1000 s) expires.
The best vector seen is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n(n: int) -> int:
    """FFT length needed for a full linear convolution of two length‑n vectors."""
    size = 2 * n - 1
    return _next_pow2(size)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real‑FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    conv_len = x.size + y.size - 1
    fft_len = _next_pow2(conv_len)
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[:conv_len]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _init_arcsine(n: int) -> np.ndarray:
    """
    Initialise with a discrete arcsine density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    The shape yields a very flat self‑convolution in the interior.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a small safety margin) and uses a
    mirror‑descent optimiser with a smoothed max objective.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size & initial vector
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 2048                     # modest fallback size
    a = _init_arcsine(n)
    a += rng.normal(scale=0.001, size=n)   # break perfect symmetry
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    # pre‑compute FFT length for the self‑convolution
    fft_len_self = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Tracking best solution
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len_self)
    best_score = 2.0 * n * np.max(conv)   # sum(a)=1 ⇒ score = 2·n·max(conv)
    best = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # safety margin
    print(f"[0.00s] initial score = {best_score:.6f}")

    # optimisation hyper‑parameters
    beta = 0.9                      # momentum coefficient
    m = np.zeros_like(a)           # momentum buffer
    stagnation_limit = 30.0        # seconds without improvement → restart
    last_improve = start_time

    # schedules (linear in elapsed fraction)
    gamma_start, gamma_end = 0.5, 2000.0
    eta_start, eta_end = 0.6, 0.02

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        if total <= 0:
            break
        frac = elapsed / total

        # ----- 1. self‑convolution -----
        conv = _conv_self(a, fft_len_self)
        max_conv = conv.max()

        # ----- 2. smoothed max weights -----
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        # stabilise the exponential by subtracting the maximum
        w_raw = np.exp(gamma * (conv - max_conv))
        w = w_raw / w_raw.sum()                # probability on indices 0 … 2n‑2

        # ----- 3. gradient of the smoothed max (cross‑correlation) -----
        # partner vectors summed with weight w are exactly the cross‑correlation
        # between a and w (no need to materialise each partner vector).
        w_rev = w[::-1]                         # reverse for cross‑correlation
        corr = _conv_fft(a, w_rev)              # length = n + 2n‑1 – 1 = 3n‑2
        start_idx = w.size - 1                 # aligns lag = 0
        g = corr[start_idx : start_idx + n]     # shape (n,)

        # ----- 4. momentum smoothing -----
        m = beta * m + (1.0 - beta) * g

        # ----- 5. step‑size schedule -----
        eta = eta_start + (eta_end - eta_start) * frac

        # ----- 6. exponentiated‑gradient (mirror) update -----
        a = a * np.exp(-eta * m)
        a = np.clip(a, 1e-14, None)            # avoid exact zeros
        a /= a.sum()                           # keep Σa = 1

        # ----- 7. occasional stochastic exploration -----
        r = rng.random()
        if r < 0.015:
            a = _random_block_perturb(a, rng)
        elif r < 0.025:
            a = _pairwise_swap_perturb(a, rng)

        # ----- 8. evaluate current score (fast version) -----
        conv = _conv_self(a, fft_len_self)
        cur_score = 2.0 * n * np.max(conv)

        # ----- 9. record improvement -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best = {best_score:.6f}")

        # ----- 10. restart if stagnated -----
        if time.monotonic() - last_improve > stagnation_limit:
            # fresh start: mix arcsine and random Dirichlet
            if rng.random() < 0.6:
                a = _init_arcsine(n)
            else:
                a = rng.gamma(shape=1.0, size=n)
                a /= a.sum()
            a += rng.normal(scale=0.001, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return best vector (clipped to satisfy the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
