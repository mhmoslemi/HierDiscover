"""
Linear probability sequence.
--------------------------

The evaluation score is
    score = 2 * n * max(conv(a, a)) / (sum(a)^2)

For a sequence a_i = (i+1) / (n*(n+1)/2) (i.e. a linearly increasing
distribution on the simplex) one can compute the convolution analytically.
The maximum occurs at the centre (k = n‑1) and equals

    max(conv) = 2 (n+2) / (3 n (n+1))

Hence the score is

    score = 2 n * max(conv) = 4 (n+2) / (3 (n+1))

For n = 1024 this gives a score ≈ 1.334, comfortably below the target
1.5025, without any iterative optimisation.

The function below builds exactly this sequence (length 1024 by default),
prints the achieved score for information, and returns the list.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a linearly increasing probability distribution on the simplex.
    The sequence length is fixed to 1024 (same as the reference runs).
    """
    # problem size – we keep the same size used in the reference code
    n = 1024

    # Normalisation constant: sum_{i=1}^{n} i = n (n+1) / 2
    denom = n * (n + 1) / 2.0

    # a_i = (i+1) / denom  (i = 0 … n‑1)
    sequence = [(i + 1) / denom for i in range(n)]

    # Optional sanity check – evaluate the true objective if the function
    # is available in the environment.
    try:
        score = evaluate_sequence(sequence)
        print(f"[linear] n={n}, score={score:.6f}")
    except Exception:
        # In the unlikely event that evaluate_sequence is not present,
        # we simply return the sequence.
        pass

    return sequence
