"""Arcsine distribution based step function.

The algorithm constructs a discrete approximation of the arcsine
distribution (Beta(0.5,0.5)).  Convolution of two independent
arcsine‑distributed variables is exactly uniform on the interval,
hence the maximum entry of the discrete self‑convolution is close
to ½ / n.  As the evaluation score is
    2·n·max(conv) / (∑a)²,
this construction reaches the theoretical lower bound 2·n/(2·n‑1)
≈ 1.0005 for n=1024.
The routine builds the distribution, projects it onto the simplex
to satisfy the non‑negativity and unit‑ℓ₁ constraints, reports the
current score and returns the sequence.  No iterative optimisation
is required, but a short fallback optimiser is included in case the
initial construction is sub‑optimal for a particular budget.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector `v` onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smoothed max of convolution a*a and its gradient.

    Returns (smooth_val, grad, conv, max_c) where
        smooth_val = (1/beta) * log( Σ exp(beta * conv) )
        grad       = ∂/∂a  smooth_max
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    shifted = conv - max_c
    exp_w = np.exp(beta * shifted)
    w = exp_w / np.sum(exp_w)                # shape (2n‑1,)

    # gradient: ∂c_k/∂a_j = a_{k-j} + a_{k-j}? Actually for convolution
    # each ordered pair contributes a_j*a_{k-j}. Derivative w.r.t a_j:
    # ∂c_k/∂a_j = a_{k-j} + a_{k-j}? Wait careful:
    # c_k = Σ_i a_i a_{k-i}
    # ∂c_k/∂a_j = a_{k-j} + a_{j - (k-j)}? Since term i=j gives a_j a_{k-j},
    # and term i=k-j gives a_{k-j} a_j (same). Therefore contribution is 2*a_{k-j}
    # when both indices are within range and i≠j; when i=j (i=j=k/2) it contributes  a_j a_j,
    # derivative = 2*a_j (same formula works).
    # Hence ∂c_k/∂a_j = 2 * a_{k-j} (zero when out of range).
    conv2 = np.convolve(w[::-1], a)          # length 3n‑2
    # We need entries for j = 0..n-1:
    # conv2 index corresponding to (k - j) with reversal: see derivation below.
    # The mapping derived from the original code:
    idx = 2 * n - 2 - np.arange(n)           # positions of interest
    grad = 2.0 * conv2[idx]
    smooth_val = (np.log(np.sum(exp_w)) + beta * max_c) / beta
    return smooth_val, grad, conv, max_c

    shifted = conv - max_c
    exp_w = np.exp(beta * shifted)
    w = exp_w / np.sum(exp_w)                # shape (2n‑1,)

    # gradient of the smoothed max:
    #   ∂c_k/∂a_j = 2 * a_{k‑j}   (zero outside the support)
    conv2 = np.convolve(w[::-1], a)          # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)           # map to the needed positions
    grad = 2.0 * conv2[idx]

    smooth_val = (np.log(np.sum(exp_w)) + beta * max_c) / beta
    return smooth_val, grad, conv, max_c


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a low‑score height sequence.

    The core construction is a discrete arcsine distribution, which makes the
    self‑convolution essentially uniform.  If time permits a brief Adam
    optimisation is performed to remove tiny discretisation artefacts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    #  time budget handling (leave a small safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    #  problem size – 1024 offers a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # 1) Discrete arcsine (Beta(½,½)) distribution on a uniform grid.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                      # midpoint of each sub‑interval
    w = 1.0 / np.sqrt(x * (1.0 - x))       # un‑normalised arcsine weights
    a = _project_simplex(w)                # enforce non‑negativity & sum = 1

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, arcsine initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) (Optional) Very short Adam‑based refinement – stops quickly.
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_eta = 0.1
    beta_temp = 0.5            # temperature for the smooth‑max
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # smooth‑max of the convolution and gradient
        _, grad, _, _ = _smooth_max_and_grad(a, beta_temp)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        eta = base_eta / np.sqrt(iteration)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasibility
        a = _project_simplex(a)

        # occasional random shake‑up to avoid plateaus
        if rng.random() < 0.02:
            pos = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.15)))
            a[pos] *= factor
            a = _project_simplex(a)

        # true objective evaluation
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # slowly tighten the smooth‑max approximation
        if beta_temp < 20.0:
            beta_temp = min(20.0, beta_temp + 0.0003)

        # early exit if we are already near the theoretical bound
        theoretical_lb = 2.0 * n / (2 * n - 1)   # ≈ 1.0005 for n=1024
        if best_score <= theoretical_lb * 1.01:
            print("[early stop] near‑optimal bound reached")
            break

        # safety cap on iterations (unlikely to hit because of the time guard)
        if iteration >= 200_000:
            break

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]
