"""
Propose a step‑function height sequence with a lower evaluation score.

Algorithm
---------
1.   Build a diverse collection of simple “seed” sequences (uniform,
     linear, sine, a grid of Beta‑shaped PDFs, and the previously
     discovered best sequence if available).  All seeds are projected
     onto the probability simplex so that they sum to 1.
2.   Evaluate the seeds with the supplied ``evaluate_sequence`` and keep
     the best one as the starting point.
3.   **Phase 1 – smooth‑max Adam optimisation**  
     * Use the smooth approximation Sα(a)= (1/α)·log Σ exp(α·(a*a)[k])  
       with a temperature α that is slowly increased from 2 to 120.  
     * Adam (β1=0.90, β2=0.999) updates the probability vector, each
       step being projected back onto the simplex.  
     * Every iteration a cheap score ``2·n·max(a*a)`` is computed (the
       denominator of the true objective is 1 because the vector sums
       to 1).  Whenever a new record is found it is printed.
     * A small random multiplicative “shake‑up’’ (5 % of the steps)
       helps escape plateaus.
4.   **Phase 2 – direct sub‑gradient mass‑transfer refinement**  
     * The exact convolution is computed, the index of the maximal
       entry is located and the sub‑gradient (2 · a reflected at that
       index) is formed.  
     * Mass is moved from the index with the largest gradient
       contribution to the one with the smallest contribution (step
       size decreasing as 1/√t).  The vector is re‑projected onto the
       simplex.  
     * The cheap score is monitored and the global best is updated.
5.   The routine stops when the 1000 s budget (minus a small safety
     margin) is exhausted or when the target score ≤ 1.5025 is reached.
6.   The best probability vector found is returned as a plain Python
     list – it will be accepted by ``evaluate_sequence`` because the
     sum is 1 and all entries are ≥ 0.

The combination of a rich set of initial guesses, a smooth‑max Adam
search with a higher temperature schedule, and a focused sub‑gradient
phase makes it very likely to push the evaluation below the target
1.5025 within the allotted time.
"""

import time

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution of a real vector via an FFT (linear convolution)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

    n = 1024

    # ------------------------------------------------------------------
    # Helper to build Beta‑shaped PDFs on the discrete grid
    def beta_pdf(alpha_, beta_):
        x = (np.arange(n) + 0.5) / n
        raw = (x ** (alpha_ - 1)) * ((1 - x) ** (beta_ - 1))
        raw = np.where(np.isfinite(raw), raw, 0.0)
        if raw.sum() == 0:
            return None
        return raw / raw.sum()

    # ------------------------------------------------------------------
    # 1) generate a pool of candidate initialisations
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # linear ramps (increasing / decreasing)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    candidates.append(ramp / ramp.sum())
    candidates.append(ramp[::-1] / ramp.sum())

    # sine‑shaped (positive on the whole interval)
    x = (np.arange(n) + 0.5) / n
    sine = np.sin(np.pi * x)
    candidates.append(sine / sine.sum())

    # a modest grid of Beta PDFs (including the arcsine case α=β=0.5)
    beta_params = [0.3, 0.5, 0.8, 1.2, 2.0, 3.0]
    for a_ in beta_params:
        for b_ in beta_params:
            pdf = beta_pdf(a_, b_)
            if pdf is not None:
                candidates.append(pdf)

    # previous best sequence if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.size:
                # normalise, then fit to current length
                prev = prev / prev.sum()
                if prev.size > n:
                    prev = prev[:n]
                elif prev.size < n:
                    prev = np.concatenate([prev, np.full(n - prev.size, 0.0)])
                prev = prev / prev.sum()
                candidates.append(prev)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) pick the best seed according to the true objective
    best = None
    best_score = float("inf")
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_score = score
            best = cand.copy()
    print(f"[init] n={n}, best seed score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – Adam optimisation on the smooth‑max surrogate
    a = best.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2, eps = 0.90, 0.999, 1e-8
    lr0 = 0.12

    alpha_start, alpha_end = 2.0, 120.0
    # linear ramp – we do not know the exact number of iterations,
    # so we use a generous upper bound.
    max_est_iters = 300_000
    alpha_inc = (alpha_end - alpha_start) / max_est_iters
    alpha = alpha_start

    start_time = time.monotonic()
    total_deadline = start_time + max(0.0, float(budget_s) - 0.5)
    phase1_deadline = start_time + 0.7 * (float(budget_s) - 0.5)

    iteration = 0
    while time.monotonic() < phase1_deadline:
        iteration += 1

        # convolution (linear)
        conv = _convolve_self_fft(a)               # length 2n‑1
        max_conv = conv.max()

        # smooth‑max weights
        shift = max_conv
        exps = np.exp(alpha * (conv - shift))
        w = exps / exps.sum()

        # gradient of smooth‑max: 2·correlate(w, a)
        grad = 2.0 * np.correlate(w, a, mode="valid")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _project_simplex(a_new)

        # occasional random perturbation (≈5 % of steps)
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.2)))  # log‑normal
            a_new[idx] *= factor
            a_new = _project_simplex(a_new)

        a = a_new

        # cheap score (exact sum is 1, so evaluate = 2·n·max_conv)
        score = 2.0 * n * max_conv
        # occasional full evaluation for safety
        if iteration % 5_000 == 0:
            full = float(evaluate_sequence(a.tolist()))
            if np.isfinite(full):
                score = full

        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[improve] iter={iteration:,}, score={best_score:.6f}")

        # temperature schedule
        if alpha < alpha_end:
            alpha = min(alpha_end, alpha + alpha_inc)

        # early exit if we already beat the target
        if best_score <= 1.5025:
            print("[early stop] target reached in Phase 1")
            break

        if iteration % 100_000 == 0:
            print(f"[status] iter={iteration:,} – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – direct sub‑gradient mass‑transfer refinement
    iter2 = 0
    eta0 = 0.02                # base step size for the mass transfer
    while time.monotonic() < total_deadline:
        iter2 += 1
        conv = _convolve_self_fft(a)
        max_conv = conv.max()
        max_idx = int(conv.argmax())

        # sub‑gradient of the max entry (without the outer factor 2)
        grad = np.zeros_like(a)
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        grad[i[mask]] = a[j[mask]]

        # pick the most and the least responsible indices
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # step size diminishing as 1/√t
        step = eta0 / np.sqrt(iter2)
        delta = step * a[i_high]          # amount of mass to move
        if delta > 0:
            a[i_high] -= delta
            a[i_low]  += delta
            a = _project_simplex(a)       # re‑enforce feasibility

        # cheap score
        score = 2.0 * n * max_conv
        if iter2 % 10_000 == 0:
            full = float(evaluate_sequence(a.tolist()))
            if np.isfinite(full):
                score = full

        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[sub‑improve] sub‑iter={iter2:,}, score={best_score:.6f}")

        if best_score <= 1.5025:
            print("[early stop] target reached in Phase 2")
            break

        if iter2 % 100_000 == 0:
            print(f"[sub‑status] sub‑iter={iter2:,} – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}")
    # Return a plain Python list (the harness expects that format)
    return [float(x) for x in best]
