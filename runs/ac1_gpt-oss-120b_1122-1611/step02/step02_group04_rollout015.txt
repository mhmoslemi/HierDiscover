def propose_candidate(seed=..., budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.1

    # Determine length n
    n = 1024
    try:
        known = np.asarray(height_sequence_1, dtype=np.float64)
        if known.ndim == 1 and known.size > 0:
            n = known.size
    except:
        pass

    # parameter: number of restarts
    # Each restart will perform certain number of gradient steps.

    best = None; best_score = np.inf
    # Initialize from known if exists else random
    # Also generate one random restart separate

    def init_sequence():
        # maybe use random dirichlet of parameter alpha = 1 (flat)
        # generate random positive numbers and project to simplex.
        x = rng.random(n)
        x = _project_simplex(x)
        return x

    # start with known if present
    sequences_to_try = []
    if 'height_sequence_1' in globals():
        # use known
        sequences_to_try.append(np.array(height_sequence_1, dtype=np.float64))
    # add random
    for i in range(5):
        sequences_to_try.append(init_sequence())

    # loop over these initial sequences, run gradient descent.
    for init_idx, a in enumerate(sequences_to_try):
        # ensure a non-negative and sum to 1
        a = np.maximum(a, 0)
        if a.sum() == 0:
            a = np.full(n, 1.0/n)
        else:
            a = _project_simplex(a)

        # evaluate initial
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best = a.copy(); best_score = score

        # set hyperparameters for gradient descent
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr_base = 0.05
        tau_factor = 0.5
        tau_decay = 0.9995
        iteration = 0
        # run until deadline
        while time.monotonic() < deadline:
            iteration += 1
            conv = np.convolve(a, a)
            current_max = conv.max()
            # compute softmax weights
            tau = max(current_max * tau_factor, 1e-15)
            shifted = (conv - current_max) / tau
            w = np.exp(shifted - shifted.max())  # shifting again for stability
            w /= w.sum()
            grad = 2.0 * np.correlate(w, a, mode='full')[n-1:n-1+n]
            # Adam
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** iteration)
            v_hat = v / (1 - beta2 ** iteration)
            step = lr_base / (np.sqrt(v_hat) + eps)
            a_candidate = a - step * m_hat
            # occasionally random shake
            if rng.random() < 0.01:
                a_candidate += rng.normal(0.0, 0.01, size=n)

            # Project onto simplex
            a_candidate = _project_simplex(a_candidate)

            # evaluate
            cand_score = evaluate_sequence(a_candidate.tolist())
            if np.isfinite(cand_score) and cand_score < best_score:
                best = a_candidate.copy()
                best_score = cand_score
                # print improvement

            # Update a
            a = a_candidate

            # update temperature
            tau_factor *= tau_decay

            # periodically attempt LP improvement
            if iteration % 10000 == 0:
                # solve LP to maximize sum(b) with conv(a,b) <= current_max + 1e-12
                dual = solve_lp(a, current_max)
                # check improvement
                if dual is not None:
                    # mix a and dual heavily
                    mix = 0.5
                    a_lp = (1 - mix) * a + mix * dual
                    a_lp = _project_simplex(a_lp)
                    lp_score = evaluate_sequence(a_lp.tolist())
                    if np.isfinite(lp_score) and lp_score < best_score:
                        best = a_lp.copy()
                        best_score = lp_score
                    a = a_lp

            if best_score <= lower_bound * 1.01:

Now implement solve_lp.

solve_lp takes current a (array length n) and bound M (float). Solve LP:

max sum_b b_j

subject to:
- b_j >= 0

"""
Hybrid optimizer for the step‑function evaluation problem.

The goal is to minimise
    2·n·max_k (a*a)_k / (∑a_i)^2 ,
with a_i ≥ 0 and ∑a_i > 0 (scale‑invariant, so we keep the simplex).

Approach
--------
1. Initialise several candidates (the known best sequence, plus a few random
   Dirichlet draws) and keep the overall best.
2. For each candidate run a projected Adam optimiser that minimises a smooth
   surrogate of the max‑convolution:
        S_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)                (smooth max)
   To accelerate convergence we blend the gradient of S_λ with the gradient
   of the L2‑energy Σ_k (a*a)_k², encouraging a flat convolution profile.
   The blending weight α starts small and is increased gradually.
3. Periodically (every ~20 000 iterations) we solve a linear programme
   ``max Σ b_i  s.t.  (a*b)_k ≤ M`` (M = current max‑convolution).  The LP
   provides a promising direction; after projection onto the simplex we
   replace the current iterate if it improves the true objective.
4. Random “shake‑ups’’ and occasional Adam‑state resets help escape plateaus.
5. The loop respects the 1000 s budget and stops early if the theoretical
   lower bound 2·n/(2n‑1) is approached.

The function returns the best sequence found within the time limit.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _solve_lp(a: np.ndarray, M: float):
    """
    Solve  max Σ b_i   s.t.   (a * b)_k ≤ M  for all k,  b_i ≥ 0.
    Returns the optimal b (may be None on failure).
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    n = a.size
    rows = 2 * n - 1
    # Build the constraint matrix A (rows = conv positions, cols = b variables)
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_range = k - i_range
        A[k, j_range] = a[i_range]

    # Right‑hand side is constant M for every convolution constraint
    b_ub = np.full(rows, M, dtype=np.float64)

    # Maximise Σ b_i  → minimise -Σ b_i
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 10.0})

    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function height sequence with a low evaluation score.
    Returns the best sequence found within the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # overall deadline (tiny safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # decide length n – use known best if available, otherwise 1024
    try:
        known = np.asarray(height_sequence_1, dtype=np.float64)
        if known.ndim == 1 and known.size > 0:
            n = int(known.size)
        else:
            raise ValueError
    except Exception:
        known = None
        n = 1024

    # ------------------------------------------------------------------
    # bookkeeping for the best solution
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Helper to evaluate and update the global best
    def _maybe_update(candidate):
        nonlocal best_seq, best_score
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"[improve] score={best_score:.6f}")
        return sc

    # ------------------------------------------------------------------
    # Prepare initial seeds: known (if any) + a few random draws
    initial_seeds = []

    if known is not None:
        a0 = np.maximum(known, 0.0)
        a0 = _project_simplex(a0)
        initial_seeds.append(a0)

    # add random Dirichlet starts
    for _ in range(3):
        start = rng.random(n)
        start = _project_simplex(start)
        initial_seeds.append(start)

    # ------------------------------------------------------------------
    # Main optimisation loop over the various restarts
    for restart_idx, a in enumerate(initial_seeds):
        # Ensure feasibility
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = _project_simplex(a)

        # Evaluate the initial candidate
        _maybe_update(a)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr_base = 0.05

        # temperature parameters for the smooth‑max surrogate
        tau_factor = 0.5          # relative to current max
        tau_decay = 0.9998

        # blending weight for max‑surrogate vs L2‑energy
        alpha = 0.2               # starts with more L2 influence
        alpha_inc = 3e-7          # per iteration increase; will cap at 1

        # counters for stagnation handling
        last_improve_iter = 0
        iteration = 0
        reset_interval = 200_000   # reset Adam moments every so often

        while time.monotonic() < deadline:
            iteration += 1

            # --------------------------------------------------------------
            # self‑convolution (full length = 2n‑1)
            conv = np.convolve(a, a)

            # --------------------------------------------------------------
            # smooth‑max temperature (never zero)
            cur_max = conv.max()
            tau = max(cur_max * tau_factor, 1e-12)

            # --------------------------------------------------------------
            # soft‑max weights for the max‑surrogate
            shifted = (conv - cur_max) / tau
            # subtract max for numeric stability
            shifted -= shifted.max()
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

            # --------------------------------------------------------------
            # gradient components
            #  * max‑surrogate gradient
            grad_max = 2.0 * np.correlate(w, a, mode='full')[n - 1:n - 1 + n]
            #  * L2‑energy gradient (flattens the whole convolution)
            grad_l2 = 4.0 * np.correlate(conv, a, mode='full')[n - 1:n - 1 + n]

            # --------------------------------------------------------------
            # blend the two gradients
            if iteration % 100 == 0:        # slowly increase emphasis on the max term
                alpha = min(1.0, alpha + alpha_inc * 100)
            grad = alpha * grad_max + (1.0 - alpha) * grad_l2

            # --------------------------------------------------------------
            # Adam update (bias‑corrected moments)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            step = lr_base / (np.sqrt(v_hat) + eps)
            a_candidate = a - step * m_hat

            # --------------------------------------------------------------
            # occasional random perturbation to escape plateaus
            if rng.random() < 0.01:
                a_candidate += rng.normal(0.0, 0.015, size=n)

            # --------------------------------------------------------------
            # project back onto the simplex
            a_candidate = _project_simplex(a_candidate)

            # --------------------------------------------------------------
            # evaluate objective
            sc = _maybe_update(a_candidate)

            # --------------------------------------------------------------
            # keep a as the latest iterate
            a = a_candidate

            # --------------------------------------------------------------
            # temperature annealing
            tau_factor *= tau_decay
            if tau_factor < 1e-6:
                tau_factor = 1e-6

            # --------------------------------------------------------------
            # every few ten‑thousands of iterations try a linear‑program step
            if iteration % 20000 == 0:
                lp_b = _solve_lp(a, cur_max * (1.0 + 1e-12))
                if lp_b is not None:
                    lp_b = _project_simplex(lp_b)
                    lp_sc = _maybe_update(lp_b)
                    # if the LP gives a better point, continue from there
                    if lp_sc < sc:
                        a = lp_b
                        # reset Adam moments to avoid stale gradients
                        m.fill(0.0)
                        v.fill(0.0)

            # --------------------------------------------------------------
            # reset Adam moments occasionally (helps escape flat regions)
            if iteration % reset_interval == 0:
                m.fill(0.0)
                v.fill(0.0)

            # --------------------------------------------------------------
            # early stop if we are essentially at the theoretical lower bound
            lower_bound = 2.0 * n / (2 * n - 1)
            if best_score <= lower_bound * 1.01:
                print("[early stop] near theoretical bound")
                return best_seq.tolist()

            # --------------------------------------------------------------
            # safety exit before the deadline expires
            if time.monotonic() > deadline - 0.2:
                break

        # after this restart, continue with the next
        # (the best sequence overall is stored in best_seq)

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
