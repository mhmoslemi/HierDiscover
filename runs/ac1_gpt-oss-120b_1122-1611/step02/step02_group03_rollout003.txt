def _project_onto_simplex(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:

    else:
        return np.full_like(a, 1.0 / len(a))

def _grad_from_max(a, k):
    n = len(a)

    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k - idx]

"""
Search for a non‑negative step‑function that minimises `evaluate_sequence`.

Algorithm
---------
* Use a relatively large number of coefficients (default n=4096).  
* Initialise many diverse start patterns (uniform, arcsine, triangular,
  sparse Dirichlet samples and the previous best solution if available).  
* For each restart run a projected Adam optimiser that works on the *true*
  maximum of the auto‑convolution (sub‑gradient of max, not a smooth
  surrogate).  Convolutions are computed with an FFT, so each iteration
  costs O(n log n).  
* After the Adam step the iterate is re‑projected onto the probability
  simplex by a simple non‑negative normalisation.  
* Occasionally a random multiplicative perturbation is applied to help
  escape plateaus.  
* The internal objective = 2·n·max(conv) (sum is kept equal to 1) is
  evaluated every iteration; the best sequence seen so far is kept.
* The search stops either when the time budget expires or when the target
  score (1.5025) is reached.

import time
import numpy as np


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Simple non‑negative projection onto the probability simplex
    (set negative entries to zero and renormalise)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback – uniform distribution
    return np.full_like(v, 1.0 / v.shape[0])


def _resize_sequence(seq, n: int) -> np.ndarray:
    """Linearly interpolate a given sequence to length n."""
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, seq)

    patterns.append(arcsine)

    # symmetric triangular (tent)

                           np.linspace(2, 1, n - half)])

    patterns.append(tent)

    # random Dirichlet – dense

    patterns.append(rng.dirichlet(np.full(n, 0.5)).astype(np.float64))

    # previous best solution, if available
    if 'height_sequence_1' in globals():

            prev = _project_onto_simplex(prev)

    # ------------------------------------------------------------------

    n_fft = 1 << ((2 * n - 2).bit_length())

    beta2 = 0.999
