"""
Propose a candidate step‑function height vector.

The algorithm is a multi‑restart projected mirror‑descent on the
probability simplex.  Each restart starts from a random Dirichlet
distribution and runs a smoothed‑max Adam optimisation, where the
smooth max is a log‑sum‑exp of the self‑convolution.  The smoothing
parameter β is increased from 5 up to 200 so that the objective
approaches the true ℓ_∞ norm of the convolution.  Gradient steps are
performed in the log‑space of the probabilities (exponentiated‑gradient
mirror descent) and the iterate is projected onto the simplex after
each step.  Random multiplicative shake‑ups are injected periodically
to escape shallow minima.  The best solution seen across all restarts
is returned and stored in the global variable ``height_sequence_1`` for
future warm‑starts.

The objective being minimised is
    2·n·max_k (a*a)[k]   (the sum of a is forced to 1)
which is exactly the value returned by ``evaluate_sequence`` for
non‑negative vectors.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho_arr.size == 0:
        # all entries non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a height vector that minimises ``evaluate_sequence``.
    The routine runs several Adam‑based mirror‑descent restarts,
    each on the simplex, and returns the best vector found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time handling (keep a small safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # problem size – keep the size that was used in the reference run
    n = 1024

    # pre‑compute FFT lengths (next power of two for speed)
    L1 = 1 << ((2 * n - 1).bit_length())                 # for a*a
    L2 = 1 << ((3 * n - 2).bit_length())                 # for w * a_rev

    # ------------------------------------------------------------------
    # hyper‑parameters
    beta_min = 5.0               # start temperature for smooth‑max
    beta_max = 200.0             # final temperature (very sharp max)
    beta_inc_iters = 400_000     # iterations over which β grows
    base_lr = 0.8                # base learning‑rate for Adam
    shake_prob = 0.07            # probability of a multiplicative shake‑up
    shake_sigma = 0.15           # log‑normal standard deviation

    # ------------------------------------------------------------------
    # storage for the best solution
    best_a = None
    best_score = np.inf
    total_iters = 0
    restart_id = 0

    # ------------------------------------------------------------------
    # helper to compute the objective (2·n·max(conv))
    def _objective_from_a(vec: np.ndarray) -> float:
        Fa = np.fft.rfft(vec, n=L1)
        conv = np.fft.irfft(Fa * Fa, n=L1)[:2 * n - 1]
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        restart_id += 1

        # ---- initialise a fresh starting point (Dirichlet, moderately
        #      sparse to encourage structure)
        a = rng.dirichlet(np.full(n, 0.2))

        # --------------------------------------------------------------
        # Adam state (mirrored in log‑space later)
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-12

        it_restart = 0
        while True:
            now = time.monotonic()
            if now >= deadline:
                break

            total_iters += 1
            it_restart += 1

            # ---------- 1) compute self‑convolution via FFT ----------
            Fa = np.fft.rfft(a, n=L1)
            conv = np.fft.irfft(Fa * Fa, n=L1)[: 2 * n - 1]

            # ---------- 2) smooth‑max weights ----------
            beta = beta_min + (beta_max - beta_min) * min(1.0,
                                                         total_iters / beta_inc_iters)
            max_c = conv.max()
            exps = np.exp(beta * (conv - max_c))          # stable exp
            w = exps / exps.sum()                         # soft‑max (size 2n‑1)

            # ---------- 3) gradient of the smooth max ----------
            a_rev = a[::-1]
            Fw = np.fft.rfft(w, n=L2)
            Fa_rev = np.fft.rfft(a_rev, n=L2)
            grad_full = np.fft.irfft(Fw * Fa_rev, n=L2)
            start = n - 1
            grad = 2.0 * grad_full[start : start + n]     # shape (n,)

            # ---------- 4) Adam update in log‑space (mirror descent) ----------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** total_iters)
            v_hat = v / (1.0 - beta2 ** total_iters)

            lr = base_lr / np.sqrt(total_iters)           # diminishing step
            # mirror‑descent step: work with log‑variables implicitly via
            # exponentiated update (no explicit log needed)
            a_next = a * np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps))

            # ---------- 5) occasional random multiplicative shake‑up ----------
            if rng.random() < shake_prob:

                factor = float(np.exp(rng.normal(0.0, shake_sigma)))
                a_next[idx] *= factor

            # ---------- 6) projection onto the simplex ----------
            a_next = _project_simplex(a_next)

            # ---------- 7) evaluate current objective ----------
            # (use the fast FFT‑based computation, no extra call to
            #  evaluate_sequence is needed)
            score = _objective_from_a(a_next)

            if score < best_score:
                best_score = score
                best_a = a_next.copy()
                print(f"[restart {restart_id}] iter={total_iters:,} "
                      f"score={best_score:.6f}  β={beta:.1f}")

                # stop early if we have already reached the target bound
                if best_score <= 1.5025:
                    print("[target] desired bound reached – terminating")
                    break

            # prepare for next iteration
            a = a_next

            # occasional progress report
            if total_iters % 100_000 == 0:
                elapsed = time.monotonic() - (deadline - budget_s + 0.5)
                print(f"[status] total_iters={total_iters:,} "
                      f"best_score={best_score:.6f}  elapsed={elapsed:.1f}s")

        if best_score <= 1.5025:
            break

    # ------------------------------------------------------------------
    # store the best sequence globally for possible warm‑starts later
    if best_a is not None:
        globals()["height_sequence_1"] = best_a.tolist()
        print(f"[final] best score={best_score:.6f} after {total_iters:,} "
              f"iterations (restarts={restart_id})")
        # Double‑check with the provided evaluator (optional but cheap)
        try:
            val = evaluate_sequence(best_a.tolist())
            print(f"[verify] evaluator reports {val:.6f}")
        except Exception:
            pass
        return [float(x) for x in best_a]
    else:
        # fallback – uniform vector (should never happen)
        return [1.0 / n] * n
