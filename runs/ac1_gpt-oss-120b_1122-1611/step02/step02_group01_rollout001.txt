def simplex_projection(v):
    """Project onto simplex {x >= 0, sum = 1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]   # descending sorted
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]   # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _random_block_perturb_big(a, rng, max_width_factor=0.2, std=0.35):
   width = rng.integers(2, max(3, int(a.size * max_width_factor)))
   start = rng.integers(0, a.size - width + 1)
   factor = np.exp(rng.normal(0, std))
   out = a.copy()
   out[start:start+width] *= factor
   out = np.clip(out, 0.0, None)
   # re-normalise to sum 1.
   out_sum = out.sum()
   if out_sum == 0: out[:] = 1.0/a.size
   else: out /= out_sum
   return out

"""
A fast gradient‑based optimizer for the step‑function attack.

The objective  2·n·max(conv(a,a)) / sum(a)²  is invariant to scaling,
so we keep the vector on the probability simplex (sum = 1, aᵢ ≥ 0).
The algorithm iteratively smooths the “max‑convolution’’ with a
soft‑max (temperature τ) and performs a projected gradient descent step
on the simplex using a correlation that can be computed with two FFTs
per iteration.  The temperature and the step size are annealed so that
the method first performs coarse reductions of the peak and then
focuses on the dominant convolution index.  Occasional random block
perturbations and soft restarts avoid stagnation, while a final greedy
pair‑wise mass‑transfer sweep cleans up any remaining local irregularities.

The routine runs for the allotted time budget (default ≈ 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative vector ``a`` using FFT.
    Returns an array of length ``2*len(a)-1``.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(A * A, n=fft_len)
    # Keep only the linear part
    return conv_full[: 2 * a.size - 1]


def _weighted_gradient(a: np.ndarray, weights: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the (soft‑max) gradient of ``max(conv(a,a))`` w.r.t. ``a``.
    The gradient is the correlation of ``a`` with ``weights`` (length
    ``2*len(a)-1``) divided by ``sum(weights)``.  ``fft_len`` must be a
    power‑of‑two ≥ len(a)+len(weights)-1.
    """
    A = np.fft.rfft(a, n=fft_len)
    W = np.fft.rfft(weights, n=fft_len)
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len)
    # Linear correlation length = n + (2n‑1) – 1 = 3n‑2
    needed = a.size + weights.size - 1
    corr_full = corr_full[:needed]
    # Gradient is the first n entries of this correlation
    return corr_full[: a.size]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        # Already on the simplex
        return v
    u = np.sort(v)[::-1]            # sort descending
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safeguard
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _random_block_perturb_big(a: np.ndarray, rng: np.random.Generator,
                              max_width_frac: float = 0.2,
                              log_std: float = 0.35) -> np.ndarray:
    """
    Apply a multiplicative log‑normal perturbation to a random contiguous
    block of ``a`` and re‑normalise to the simplex.
    """
    n = a.size
    width = rng.integers(2, max(3, int(n * max_width_frac) + 1))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, log_std))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    ``evaluate_sequence``.  A simple annealed gradient descent with
    soft‑max smoothing is used, complemented by random block
    perturbations, soft restarts and a final greedy pair‑wise sweep.
    The function returns the best vector (list of floats) found before
    the allotted time expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose an initial vector (use the provided one if sensible)
    # --------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                total = a.sum()
                if total > 0.0:
                    a = a / total
                else:
                    a = np.full(cand.size, 1.0 / cand.size, dtype=np.float64)
            else:
                a = None
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # --------------------------------------------------------------
    # Pre‑compute FFT sizes
    # --------------------------------------------------------------
    conv_len = 2 * n - 1                    # length of a*a
    corr_len = n + conv_len - 1             # needed length for correlation = 3n-2
    fft_len = _next_pow2(max(conv_len, corr_len))

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self_fft(best, fft_len)
    best_score = float(2.0 * n * conv_best.max())
    last_improve = time.monotonic()
    start_time = time.monotonic()

    # Hyper‑parameters (tuned empirically)
    step_initial = 0.10          # start step size for gradient descent
    step_final = 0.001           # final step size near the end
    tau_initial = 0.50           # initial temperature for soft‑max
    tau_final = 0.02             # final temperature
    block_perturb_prob = 0.02    # chance of a block perturbation each iter
    restart_gap = 45.0           # seconds without improvement → soft restart
    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # 1. Current convolution and score
        # ----------------------------------------------------------
        conv = _conv_self_fft(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----------------------------------------------------------
        # 2. Record best
        # ----------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            conv_best = conv.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best = {best_score:.6f}")

        # ----------------------------------------------------------
        # 3. Annealing schedules
        # ----------------------------------------------------------
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)
        step = step_initial * (1.0 - progress) + step_final * progress
        tau = tau_initial * (1.0 - progress) + tau_final * progress
        if tau < 1e-8:
            tau = 1e-8

        # ----------------------------------------------------------
        # 4. Soft‑max weights for the current peak
        # ----------------------------------------------------------
        max_conv = conv.max()
        # subtract max to avoid overflow; scale by τ
        weights = np.exp((conv - max_conv) / tau)
        sum_weights = weights.sum()

        # ----------------------------------------------------------
        # 5. Gradient (correlation of a with weights)
        # ----------------------------------------------------------
        g = _weighted_gradient(a, weights, fft_len) / sum_weights

        # ----------------------------------------------------------
        # 6. Projected gradient step (move opposite to the gradient)
        # ----------------------------------------------------------
        a_new = a - step * g
        a = _simplex_projection(a_new)

        # ----------------------------------------------------------
        # 7. Random block perturbation (exploration)
        # ----------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb_big(a, rng)

        # ----------------------------------------------------------
        # 8. Soft restart if stuck
        # ----------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > restart_gap:
            # perturb best slightly and continue
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _simplex_projection(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart from best (score {best_score:.6f})")

    # ------------------------------------------------------------------
    # Final greedy pair‑wise mass‑transfer sweep (few seconds)
    # ------------------------------------------------------------------
    a = best.copy()
    conv = conv_best.copy()
    cur_score = best_score
    remaining = deadline - time.monotonic()
    # Use at most the remaining time (or a hard cap of ~30 s)
    sweep_time = min(remaining, 30.0)
    sweep_end = time.monotonic() + sweep_time
    # Pre‑compute contribution vectors for faster updates
    # contribution of unit mass at position i is the vector a shifted by i
    # (length 2n‑1). Rather than recompute each time, we will just recompute
    # conv from scratch for each accepted move – the number of accepted
    # moves is tiny, so the overhead is negligible.
    while time.monotonic() < sweep_end:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Transfer up to 15 % of the source mass (small local move)
        max_delta = min(a[i], 0.15 * a[i])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # Re‑normalise (the sum stays 1, but numerical drift may appear)
        a_candidate = _simplex_projection(a_candidate)
        conv_candidate = _conv_self_fft(a_candidate, fft_len)
        cand_score = 2.0 * n * conv_candidate.max()
        if cand_score < cur_score:
            a = a_candidate
            conv = conv_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = a.copy()
                conv_best = conv.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] improved by pairwise to {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best vector (clipped to the safety bounds of evaluate_sequence)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
