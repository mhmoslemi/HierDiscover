def convolve_fft(a,b):
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a,N)
    Fb = np.fft.rfft(b,N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc,N)[:n]
    c = np.maximum(c,0.0)
    return c

import time
import numpy as np

def convolve_fft(a, b):
    """Linear convolution via FFT (real)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c

def simplex_projection(v):
    """Project onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def block_perturbation(base, rng):
    """Random block scaling and optional symmetrisation."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)

"""
Smooth‑max projected gradient descent with momentum.

The algorithm works on the probability simplex (∑w = 1, w ≥ 0).  
It minimises the L∞ norm of the self‑convolution w∗w by
optimising the smooth approximation
    f(w)=μ·log∑_k exp((w∗w)[k]/μ) .
The gradient of f is 2·conv(p, w_rev) where p is the soft‑max
over the convolution values.  A diminishing step‑size together with
momentum and a cheap block‑perturbation escape heuristic are used.
The final sequence is re‑scaled to the traditional √(2 n) ℓ₁‑norm,
which does not change the score returned by `evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – defined at top level as required by the harness.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation, used as a jump operator."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    `evaluate_sequence` score is (hopefully) better than the previous best
    (~1.518).  The method optimises a smooth approximation of the L∞
    convolution peak with projected momentum gradient descent.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.  Re‑use a previously found construction if it
    # exists; otherwise fall back to a moderately large size (≈8000).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=np.float64)
            n = init_seq.size
            if n < 2:
                raise ValueError
        except Exception:
            init_seq = None
            n = 8000
    else:
        init_seq = None
        n = 8000

    # --------------------------------------------------------------
    # Initialise w on the simplex.  If a previous candidate is present we
    # reuse it (clipping negatives and normalising); otherwise start from
    # the classic central‑binomial distribution a_i = C(2i,i)/4**i.
    # --------------------------------------------------------------
    if init_seq is not None:
        w = np.maximum(init_seq, 0.0)
        s = np.sum(w)
        if s < 1e-12:
            w = np.full(n, 1.0 / n)
        else:
            w = w / s
    else:
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2 * i)
        w = a / np.sum(a)

    # --------------------------------------------------------------
    # State variables.
    # --------------------------------------------------------------
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = float(2.0 * n * np.max(b))
    best_iter = 0
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters.
    # --------------------------------------------------------------
    eta0 = 0.6               # base learning rate
    beta = 0.9               # momentum coefficient
    mu_factor = 0.5          # start with μ = mu_factor·max(b)
    mu_factor_min = 0.04
    mu_decay = 0.9995        # exponential decay per iteration
    momentum = np.zeros_like(w)

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 30000
    block_interval = 2500    # try a random jump every few thousand steps
    reset_interval = 20000   # fall back to a perturbed best if stuck

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # 1) convolution and current score
        b = convolve_fft(w, w)
        maxb = np.max(b)
        score = 2.0 * n * maxb

        # 2) record best
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            best_iter = iteration
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # 3) smooth‑max weights
        mu = maxb * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        shifted = (b - maxb) / mu          # max(b) → 0, others ≤ 0
        exp_term = np.exp(shifted)        # safely in [0,1]
        p = exp_term / np.sum(exp_term)   # softmax over the 2n‑1 entries

        # 4) gradient = 2·conv(p, w_rev) (slice gives a length‑n vector)
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # 5) step‑size and momentum update
        eta = eta0 / np.sqrt(iteration + 1.0)
        momentum = beta * momentum - eta * grad
        w_candidate = w + momentum

        # 6) project back onto the simplex
        w = simplex_projection(w_candidate)

        # 7) slowly reduce the smoothing factor
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        # 8) occasional cheap random jump
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                momentum = np.zeros_like(w)
                no_improve = 0
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # 9) restart from a perturbed best if stagnating
        if no_improve > reset_interval:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            momentum = np.zeros_like(w)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Short pure sub‑gradient cleanup phase (focus on the active peak).
    # --------------------------------------------------------------
    sub_iter = 0
    eta0_sub = 0.5
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 0.5):
        b = convolve_fft(w, w)
        maxb = np.max(b)
        max_idxs = np.flatnonzero(b >= maxb - 1e-15)
        i_max = int(rng.choice(max_idxs))

        # sub‑gradient for the active index i_max
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        grad[j] = 2.0 * w[i_max - j]          # 2·w shifted

        eta = eta0_sub / np.sqrt(sub_iter + 1.0)
        w = simplex_projection(w - eta * grad)

        # check improvement
        score = 2.0 * n * np.max(convolve_fft(w, w))
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[sub‑improve] sub_iter={sub_iter:6d}, score={best_score:.6f}")

        sub_iter += 1
        if sub_iter > 20000:
            break

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but we output the
    # conventional √(2 n) ℓ₁‑norm for readability.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best iter={best_iter}, best score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
