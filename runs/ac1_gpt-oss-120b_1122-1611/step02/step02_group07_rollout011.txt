    from scipy.linalg import toeplitz

    rng = np.random.default_rng(seed)

    # outer while time left
    inner_iter = 0
    while time.monotonic() < deadline:
        # compute convolution and M
        conv_f = np.convolve(a, a)
        M = conv_f.max()
        # build Toeplitz matrix A
        c = np.concatenate([a, np.zeros(n - 1)])
        r = np.concatenate([a[:1], np.zeros(n - 1)])

        # solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True})
        if not res.success:
            # fallback: random perturb
            idx = rng.integers(0, n)
            a[idx] = max(0.0, a[idx] + rng.normal(scale=0.01))
            a = _project_simplex(a)
            continue

        g_opt = res.x
        sum_g = g_opt.sum()
        if sum_g < 1e-12:
            # can't improve
            break
        g_norm = g_opt / sum_g

        # try step size line search
        improved = False
        for t in [0.5, 0.3, 0.7, 1.0, 0.1]:
            candidate = (1 - t) * a + t * g_norm
            candidate = _project_simplex(candidate)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate.copy()
                best_score = score
                a = candidate
                improved = True

            idx = rng.integers(0, n)

    return [float(x) for x in best]

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation function

    2·n·max_k (a*a)[k] / (∑a_i)² .

The core of the search is an *alternating linear‑program / mixing*
procedure inspired by the state‑of‑the‑art iteration described in the
challenge text:

1.  Start from a simplex‑projected vector (the previous best sequence
    if available, otherwise the uniform distribution).

2.  Let M = max convolution of the current vector with itself.
    Solve the linear programme

        maximise   Σ g_i
        subject to  (f * g)[k] ≤ M   for all k,
                    g_i ≥ 0.

    The constraint matrix (f * g) ≤ M is a Toeplitz matrix built from f,
    so a dense matrix of size (2n‑1)×n is used.

3.  The LP returns a vector g₀ with (usually) a larger ℓ₁‑norm than the
    current f.  Rescale it to the simplex:
        g = g₀ / Σ g₀ .

4.  Mix the two vectors:  f←(1‑t)·f + t·g   (t∈(0,1]).  The step size t is
    chosen from a short list by line‑search; the best feasible mix that
    improves the score is kept.  The mixture stays on the simplex, so the
    evaluation function can be called directly.

5.  Repeat steps 2‑4 until the time budget is exhausted or no further
    improvement is possible.  Random multiplicative perturbations are
    injected when the LP does not yield progress, helping the search to
    escape shallow plateaux.

If SciPy is unavailable the routine falls back to a simple random‑search
using Dirichlet draws.

The algorithm respects the 1000 s budget, prints progress, and returns
the best sequence found as a list of floats.

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            A_ub = toeplitz(col, row)           # shape (2n‑1, n)

            # ----- 3) linear programme: maximise Σ g_i  (i.e. minimise –Σ g_i) -----
            lp_res = linprog(c=-np.ones(n, dtype=np.float64),
                             A_ub=A_ub,
                             b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                             bounds=[(0, None)] * n,
                             method='highs',
                             options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})

            if not lp_res.success or lp_res.x is None:
                # LP failed – fall back to a tiny random shake‑up and continue
                idx = rng.integers(0, n)
                a[idx] = max(0.0, a[idx] + rng.normal(scale=0.01))
                a = _project_simplex(a)
                continue

            g_opt = lp_res.x
            sum_g = g_opt.sum()
            if sum_g < 1e-12:
                # nothing to improve – break out
                break

            # Normalise to the simplex (unit ℓ₁)
            g = g_opt / sum_g

            # ----- 4) line‑search on the mixing parameter t ------------------------
            improved = False
            old_score = best_score
            for t in t_candidates:
                cand = (1.0 - t) * a + t * g
                cand = _project_simplex(cand)   # safety projection
                cand_score = _score(cand)
                if np.isfinite(cand_score) and cand_score < best_score:
                    # accept the improvement
                    a = cand
                    best = cand.copy()
                    best_score = cand_score
                    improved = True
                    print(f"[improve] iter={it:,}  t={t:.2f}  score={best_score:.6f}")
                    break

            if not improved:
                # If none of the candidate steps improve, apply a mild random
                # multiplicative perturbation (≈5 % chance) to help escape.
                if rng.random() < 0.05:
                    idx = rng.integers(0, n)
                    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈ ±12 %
                    a[idx] *= factor
                    a = _project_simplex(a)

        else:  # -----------------------------------------------------------------
            # Simple random‑search fallback (no SciPy)
            for _ in range(20):
                # draw a random Dirichlet point and evaluate
                rand_vec = rng.dirichlet(np.ones(n))
                sc = _score(rand_vec)
                if sc < best_score:
                    best = rand_vec.copy()
                    best_score = sc
                    a = rand_vec.copy()
                    print(f"[rand‑improve] iter={it:,}  score={best_score:.6f}")
                    break

        # ----- 5) early exit if the target is reached ---------------------------
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping early")
            break

        # ----- 6) occasional progress report ------------------------------------
        if it % 10 == 0:
            elapsed = time.monotonic() - (deadline - budget_s + 0.5)
            print(f"[status] iter={it:,}  best={best_score:.6f}  elapsed={elapsed:.1f}s")

    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    # return a plain Python list as required by the evaluator
    return [float(x) for x in best]
