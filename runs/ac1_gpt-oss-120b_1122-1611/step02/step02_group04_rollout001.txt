"""
Propose a low‑score step‑function height sequence.

Algorithm
---------
* Work on the probability simplex (non‑negative entries that sum to 1);
  this makes the objective scale‑invariant.
* For a given length `n` run several independent Adam optimisations.
  The smooth‑max surrogate of the worst self‑convolution entry
  Sλ(a) = (1/λ)·log∑exp(λ·(a*a)_k) is used.
  Its gradient is a correlation between the soft‑max weights and the current
  vector, which we compute with `np.correlate`.
* The temperature (inverse λ) starts relatively large (smooth landscape)
  and is multiplied by a factor < 1 each iteration, so the surrogate sharpens
  gradually.
* After each Adam step we project back onto the simplex with the Euclidean
  projection algorithm.
* Occasionally a small Gaussian shake‑up is added to escape plateaus.
* The best sequence seen across all restarts and all lengths is kept.
* A cheap hill‑climbing refinement (mass transfer between two coordinates)
  is performed at the end.
* The function respects the 1000 s time budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x >= 0, sum(x) = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _adam_optimize(rng: np.random.Generator,
                   n: int,
                   a_init: np.ndarray,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Run Adam optimisation for a single restart.

    Returns the best vector found and its score (2·n·max(a*a)).
    """

    a = _project_simplex(a)

    # initial best
    conv = np.convolve(a, a)
    best = a.copy()
    best_score = 2.0 * n * conv.max()

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base = 0.07                # a little larger than the previous code
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature handling (smooth‑max)
    tau_factor = 2.0               # start with a relatively high temperature
    tau_decay = 0.9999

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ----- convolution and temperature -----
        conv = np.convolve(a, a)
        max_conv = conv.max()
        tau = max(max_conv * tau_factor, 1e-12)

        # ----- soft‑max weights (stable) -----
        shifted = (conv - max_conv) / tau
        # subtract max for numerical stability
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # ----- gradient of the smooth‑max surrogate -----
        grad = 2.0 * np.correlate(w, a, mode='full')[n - 1:n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr_base / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat

        # ----- projection back onto the simplex -----
        a_candidate = _project_simplex(a_candidate)

        # ----- occasional random shake‑up -----
        if rng.random() < 0.006:
            a_candidate += rng.normal(0.0, 0.012, size=n)
            a_candidate = _project_simplex(a_candidate)

        # ----- evaluate candidate (exact objective) -----
        cand_conv = np.convolve(a_candidate, a_candidate)
        cand_score = 2.0 * n * cand_conv.max()

        if np.isfinite(cand_score) and cand_score < best_score:
            best_score = cand_score
            best = a_candidate.copy()
            print(f"[improve] n={n} iter={iteration:,} score={best_score:.6f}")

        # ----- prepare next iteration -----
        a = a_candidate
        tau_factor *= tau_decay

        # optional early stop when we are already close to the theoretical bound
        lower_bound = 2.0 * n / (2 * n - 1)
        if best_score <= lower_bound * 1.02:
            break

        # safety break if we have exhausted the time budget
        if iteration % 10000 == 0 and time.monotonic() >= deadline:
            break

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function height sequence with a low evaluation score.
    Uses several Adam‑based optimisations at increasing lengths and a final
    hill‑climbing refinement.
    """
    rng = np.random.default_rng(seed)

    # overall deadline, leave a tiny safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.15)

    # ------------------------------------------------------------------
    # try to reuse the previously known best sequence if it exists
    known_seq = None
    try:
        known_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if known_seq.ndim != 1 or known_seq.size == 0:
            known_seq = None
    except Exception:
        known_seq = None

    # ------------------------------------------------------------------
    # bookkeeping for the global best
    best_overall = None
    best_score_overall = float("inf")

    # ------------------------------------------------------------------
    # lengths we will explore (larger length → more freedom)
    lengths = [1024, 2048, 4096]

    for n in lengths:
        # stop if we have run out of time
        if time.monotonic() >= deadline:
            break

        # allocate roughly a third of the remaining budget to this length
        remaining = deadline - time.monotonic()
        sub_deadline = time.monotonic() + remaining * 0.33

        restart = 0
        while time.monotonic() < sub_deadline:
            # initialise a fresh candidate
            if known_seq is not None and known_seq.size == n and restart == 0:
                a0 = known_seq.copy()
            else:
                a0 = rng.random(n)

            a0 = _project_simplex(a0)

            # run a single Adam optimisation
            best_local, score_local = _adam_optimize(rng, n, a0, sub_deadline)

            if score_local < best_score_overall:
                best_score_overall = score_local
                best_overall = best_local.copy()
                print(f"[global improve] n={n} restart={restart} score={best_score_overall:.6f}")

            restart += 1

            # break early if we are close to the theoretical bound
            lower = 2.0 * n / (2 * n - 1)
            if best_score_overall <= lower * 1.02:
                break

            # give the outer loop a chance to re‑check the deadline
            if time.monotonic() + 0.5 >= deadline:
                break

    # ------------------------------------------------------------------
    # cheap hill‑climbing refinement on the best sequence found
    if best_overall is not None:
        a = best_overall.copy()
        n = a.shape[0]
        for step in range(8000):
            if time.monotonic() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # propose moving a small amount from i to j
            delta = (rng.random() - 0.5) * 0.015  # up to +/-0.0075
            if a[i] - delta < 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            # evaluate
            score_prop = 2.0 * n * np.max(np.convolve(a_prop, a_prop))
            if score_prop < best_score_overall:
                best_score_overall = score_prop
                best_overall = a_prop.copy()
                a = a_prop
                print(f"[refine] step={step} score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # fallback – should never happen, but keep a valid answer
    if best_overall is None:
        n = 1024
        best_overall = np.full(n, 1.0 / n)
        best_score_overall = 2.0 * n * np.max(np.convolve(best_overall, best_overall))
        print("[fallback] using uniform distribution")

    print(f"[final] best score={best_score_overall:.6f}, length={len(best_overall)}")
    return [float(x) for x in best_overall]
