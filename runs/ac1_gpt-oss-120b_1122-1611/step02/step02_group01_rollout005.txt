"""
Softmax‑Adam with temperature annealing.

We treat the step heights a_i as a probability vector (sum = 1) and
parameterise them by a soft‑max of real variables s_i :

    a_i = exp(s_i) / Σ_j exp(s_j)

The objective 2·n·max(conv(a,a)) is scale‑invariant, so we keep
Σ a_i = 1 throughout.  Instead of the non‑smooth max we minimise a
smooth “temperature‑softmax” of the convolution:

    L(s,τ) = 2·n·τ·log Σ_k exp(conv_k(s)/τ)

For a given temperature τ the gradient w.r.t. s can be written in
closed form using a single FFT‑based correlation, giving an
O(N log N) update.  An Adam optimiser drives s while τ is linearly
annealed from a large initial value (exploration) to a small final
value (exploitation).  Occasionally we perform cheap random
perturbations (block scaling, pairwise mass transfer) and a soft
restart if no improvement is seen for a while.  The best vector
encountered is returned as a plain Python list.

The routine respects all checks of `evaluate_sequence` and runs for
the allotted budget (≈1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_three_n(n: int) -> int:
    """
    Return the next power‑of‑two length that can hold a full linear
    correlation of a length‑n vector with a length‑(2·n‑1) vector.
    The required size is (n + (2·n‑1) – 1) = 3·n – 2.
    """
    size = 3 * n - 2
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT of length ``fft_len``.
    Returns a vector of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may create tiny negatives; clip to zero.
    return np.maximum(conv_full, 0.0)


def _correlation(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear correlation (a ⋆ w)[j] = Σ_i a[i]·w[i+j] for j=0…len(a)-1.
    Implemented via FFT: irfft(FA·conj(FW)).
    """
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(fa * np.conj(fw), n=fft_len)
    return corr_full[: a.size]  # only the first n entries are needed


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Contribution of the entry a[idx] to the full self‑convolution.
    Returns a vector of length 2·n‑1 where the copy of ``a`` is placed
    starting at position ``idx``.
    """
    n = a.size
    out = np.zeros(2 * n - 1, dtype=a.dtype)
    out[idx : idx + n] = a
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the protected
    ``evaluate_sequence`` score.  A soft‑max Adam optimiser with annealed
    temperature is used, supplemented by occasional block perturbations,
    pairwise mass transfers and soft restarts.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Initialise the base vector (use the global if present)
    # --------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
                pass
            else:
                raise ValueError
        except Exception:
            base = None
    else:
        base = None

    if base is None:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = base.size
        a = np.clip(base, 0.0, None)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 2. Parameterisation (soft‑max) and Adam state
    # --------------------------------------------------------------
    eps = 1e-12
    s_vec = np.log(a + eps)            # log‑soft‑max parameters
    m = np.zeros_like(s_vec)           # 1st‑moment estimate
    v = np.zeros_like(s_vec)           # 2nd‑moment estimate
    t_iter = 0                         # Adam time step

    # --------------------------------------------------------------
    # 3. FFT planning
    # --------------------------------------------------------------
    fft_len = _fft_len_for_three_n(n)   # big enough for both conv & corr

    # --------------------------------------------------------------
    # 4. Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len)
    best_score = float(2.0 * n * np.max(conv_best))
    start_time = time.monotonic()
    last_improve = start_time

    # --------------------------------------------------------------
    # 5. Hyper‑parameters
    # --------------------------------------------------------------
    lr_initial = 0.02                  # Adam base learning‑rate
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8

    tau_start = 0.20                   # initial temperature (smooth max)
    tau_end = 0.005                    # final temperature
    block_perturb_prob = 0.02
    pair_move_prob = 0.07
    max_frac_transfer = 0.30
    restart_gap = 45.0                 # seconds without improvement

    # --------------------------------------------------------------
    # 6. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < start_time + max(0.0, float(budget_s) - 5.0):
        # ---- (a) obtain probability vector from parameters ----
        s_shift = np.max(s_vec)               # stability for exp
        a = np.exp(s_vec - s_shift)
        a_sum = a.sum()
        a /= a_sum if a_sum != 0 else np.full_like(a, 1.0 / n)

        # ---- (b) self‑convolution and current score -------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- (c) track the best solution -----------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- (d) temperature schedule (linear decay) -----------
        elapsed_total = time.monotonic() - start_time
        progress = elapsed_total / (budget_s - 5.0)
        tau = tau_start * (1.0 - progress) + tau_end * progress
        tau = max(tau, tau_end)

        # ---- (e) smooth‑max weights (softmax over the convolution) --
        max_conv = conv.max()
        w_exp = np.exp((conv - max_conv) / tau)     # stability shift
        w = w_exp / w_exp.sum()

        # ---- (f) gradient of the smooth objective (via correlation) ---
        corr = _correlation(a, w, fft_len)          # length n
        grad_a = (4.0 * n / tau) * corr

        # ---- (g) back‑prop to the soft‑max parameters ------------
        dot = np.dot(a, grad_a)
        grad_s = a * (grad_a - dot)

        # ---- (h) Adam parameter update -------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_s
        v = beta2 * v + (1.0 - beta2) * (grad_s ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        lr = lr_initial * (1.0 - progress) + 0.001 * progress   # slight decay
        s_vec -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---- (i) occasional random block perturbation ----------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)
            s_vec = np.log(a + eps)      # re‑initialise parameters

        # ---- (j) occasional pair‑wise mass transfer ------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], max_frac_transfer * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    # Linear effect on the convolution
                    contrib_i = _contribution_vector(a, i)
                    contrib_j = _contribution_vector(a, j)
                    new_conv = conv + delta * (contrib_j - contrib_i)
                    new_score = 2.0 * n * float(np.max(new_conv))
                    if new_score < cur_score:
                        a[i] -= delta
                        a[j] += delta
                        a_sum = a.sum()
                        a = a / a_sum if a_sum != 0 else np.full_like(a, 1.0 / n)
                        conv = new_conv
                        cur_score = new_score
                        cur_max = float(np.max(conv))
                        s_vec = np.log(a + eps)  # keep the parametrisation in sync

        # ---- (k) soft restart if stuck -------------------------
        now = time.monotonic()
        if now - last_improve > restart_gap:
            a = best.copy()
            # add modest log‑normal noise to the probabilities
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            a = a / a_sum if a_sum != 0 else np.full_like(a, 1.0 / n)
            s_vec = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – re‑starting from best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to the bounds enforced by the
    # evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
