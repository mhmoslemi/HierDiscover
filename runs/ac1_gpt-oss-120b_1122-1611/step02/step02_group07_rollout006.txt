"""
propose_candidate
-----------------
A stochastic optimizer that directly minimises the true objective
2·n·max(a*a) (sum(a) is kept equal to 1 by a simplex projection).
The algorithm uses an Adam‑style update with the exact sub‑gradient of the
max‑convolution (instead of a smooth log‑sum‑exp approximation) and
periodic random shake‑ups.  When the current solution stops improving for
a while it is re‑initialised around the best point found so far, which
helps escaping local minima.  The routine runs until the given time budget
expires and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0                # (cumulative sum) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:                        # all non‑positive → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function (length 1024) that minimises
    ``evaluate_sequence``.  The optimiser works on the exact max‑convolution
    gradient, uses Adam for adaptive steps, and injects occasional random
    perturbations.  When progress stalls a restart around the current best
    point is performed.  The best sequence found before the time budget ends
    is returned as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time handling – keep a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # problem size – the historic experiments used 1024
    n = 1024

    # ------------------------------------------------------------------
    # initialise a (warm‑start if possible, otherwise uniform)
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a0.size != n:
                raise ValueError
            a = a0.copy()
        except Exception:
            a = rng.dirichlet(np.ones(n))            # random simplex point
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)    # uniform vector

    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # learning‑rate schedule (base_lr decays as 1/sqrt(t))
    base_lr = 0.25

    # ------------------------------------------------------------------
    # bookkeeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    it = 0
    stagnation = 0                     # iterations since last improvement

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) autocorrelation of the current vector
        conv = np.convolve(a, a)                     # length 2n‑1

        # --------------------------------------------------------------
        # 2) index of the (first) maximal entry – this gives the exact sub‑gradient
        k_star = int(conv.argmax())
        # (the value conv[k_star] is not needed here because the objective
        #  is 2·n·conv[k_star] and the denominator is constant =1)

        # --------------------------------------------------------------
        # 3) compute sub‑gradient of max(conv) w.r.t. a
        #    ∂/∂a_i  conv[k_star] = a_{k_star-i} + a_{k_star-i}  (i≠k_star-i)
        #    = 2·a_{k_star-i},  otherwise = a_i
        grad = np.zeros_like(a)

        i_start = max(0, k_star - (n - 1))
        i_end   = min(n - 1, k_star)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_star - i_range

        mask = i_range != j_range

        # i == j  →  a_i
        grad[i_range[~mask]] = a[i_range[~mask]]

            m.fill(0.0)

            stagnation = 0
            continue

        a = a_next

            print(f"[status] iter={it:,}  best_score={best_score:.6f}  elapsed={elapsed:.1f}s")

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    return [float(x) for x in best]
