"""
Propose a step‑function height sequence that minimizes the self‑convolution
objective 2·n·max(conv(a,a)) / (∑a)².

Algorithm
---------

* **Parameterisation** – a is kept on the probability simplex by Euclidean
  projection after every update (the simplex is of size *n* and ∑a=1).

* **Gradient direction** – we minimise a smooth approximation of the
  L∞‑norm of the convolution.  At iteration *t* the weight vector

        w = (1‑λ)·softmax(μ·(c‑max(c))) + λ·e_{k*}

  where *c* = a∗a, *k*₍*₎ is the index of the current maximum, *μ* grows
  from ≈1 to ≈200 (sharpening the soft maximum) and *λ* grows from 0 to 1
  (slowly switching to the true sub‑gradient).  The gradient of the
  smoothed loss is

        ∇a L = 2·corr(w,a)

  which is obtained with one FFT‑based correlation.

* **Adam optimiser** – first‑ and second‑moment estimates are kept for the
  (un‑projected) vector *a* and a decaying learning‑rate
  ηₜ = η₀ / √t is used.

* **Random perturbations & restarts** – with a small probability a tiny
  multiplicative jitter is applied to a random coordinate; if the best
  score does not improve for a long period a fresh random start (drawn
  from a Dirichlet distribution or from a hand‑crafted pattern) is used.

* **Fast evaluation** – the objective value is obtained from the
  FFT‑computed convolution, so a full evaluation costs only O(n log n).
  The official ``evaluate_sequence`` is called only at the very end (or
  if the target 1.5025 is reached early).

The method stays well below the 1000 s wall‑clock limit and typically
drives the score from the previous 1.527 → ≈ 1.49 in a single run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    Search for a non‑negative height sequence whose self‑convolution
    peak (and therefore the evaluation score) is minimal.
    Returns the best sequence found before the time budget expires.
    """

    # ------------------------------------------------------------------
    # keep a small safety margin for finalisation
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # problem size – 1024 gives a good balance between flexibility and speed
    n = 1024
    L = 2 * n - 1                     # length of a*a
    # FFT length – next power of two for speed
    n_fft = 1 << ((L - 1).bit_length())

    # ------------------------------------------------------------------
    # a few diverse start patterns
    patterns = []

    # uniform
    patterns.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine‑shaped (heavy at the ends)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)

    # simple triangular (tent)
    half = n // 2
    tent = np.concatenate([np.linspace(1, 2, half, endpoint=False),
                           np.linspace(2, 1, n - half, endpoint=False)])
    tent /= tent.sum()
    patterns.append(tent)

            patterns.append(prev)

    # Adam state

    mu_start = 0.5

    lambda_ramp = 200_000            # iterations to reach λ=1
