def propose_candidate(seed=123, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.05
    n = determine_n
    # if height_sequence_1 known, use its n else 1024 maybe larger.
    if known exists:
        n = known.shape[0]
        a = known.copy()
    else:
        n = 2048 maybe? but initial is 1024. We'll use 1024 maybe for speed.

    # ensure simplex
    a = np.maximum(a, 0)
    a = _project_simplex(a)

    # compute initial best
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(...)

    # Adam variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9, beta2=0.999, eps=1e-8, lr=0.05

    # temperature schedule
    tau_factor = 0.3  # maybe smaller for more weight
    tau_decay = 0.9999

    # parameters for phases
    phase_cutoff = start + 0.6 * budget_s

    # iteration
    iteration = 0

    while time.monotonic() < deadline:
        iteration+=1
        # conv via fft
        conv = _convolve_fft(a)
        max_conv = conv.max()
        # determine phase
        now = time.monotonic()
        if now < phase_cutoff:
            # smooth mode
            tau = max(max_conv * tau_factor, 1e-12)
            shifted = (conv - max_conv) / tau
            w = np.exp(shifted)
            ws = w.sum()
            if ws == 0:
                w = np.full_like(conv, 1.0/conv.size)
            else:
                w /= ws
            # gradient
            grad = 2.0 * np.correlate(w, a, mode='full')[n-1:n-1+n]
        else:
            # max mode
            k_max = int(np.argmax(conv))
            # compute gradient: a_shift = zeros length n
            # For i in [0,n):
            #   j = k_max - i
            # if 0 <= j < n: grad_i = a[j] else 0
            # Using vectorized: indices = k_max - np.arange(n)
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = a[idx[mask]]
            grad = 2.0 * grad
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat
        # clip negative to zero before projection
        a_candidate = np.maximum(a_candidate, 0.0)
        # project onto simplex
        a_candidate = _project_simplex(a_candidate)
        # occasional shake
        if rng.random() < 0.001:
            noise = rng.normal(0.0, 0.02, size=n)
            a_candidate += noise
            a_candidate = np.maximum(a_candidate, 0)
            a_candidate = _project_simplex(a_candidate)
        # evaluate
        score = float(evaluate_sequence(a_candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_candidate.copy()
            print(f"[improve] iter={iteration:,} score={best_score:.6f}")

        # assign a = a_candidate

        a = a_candidate

        # adjust tau_factor
        tau_factor *= tau_decay
        # Ensure tau not too tiny
        if iteration % 2000 == 0:

        # Early stop if close to bound (1 + little). Bound = 2*n/(2*n-1)
        lower_bound = 2.0 * n / (2.0 * n - 1.0)

"""
Propose a candidate step‑function using a fast FFT‑based convolution,
Adam optimisation with a smooth‑max surrogate that gradually sharpens,
followed by a direct “max‑only” subgradient phase.
The algorithm periodically perturbs the current iterate to escape plateaus
and restarts from the best solution seen so far when stagnation is detected.
All iterates are projected onto the probability simplex (non‑negative,
unit ℓ₁‑norm) so the evaluation function’s denominator stays constant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _convolve_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length that is a power of two.
    Returns a real array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    # Numerical noise can produce tiny negatives – clip them.
    return np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate S_λ(a) = (1/λ)·log∑exp(λ·(a*a)_k).
    Implements 2·∑_k w_k·a_{k‑i} via a convolution (FFT) between w and the
    reversed a vector, returning a vector of length len(a).
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    # reverse a for correlation
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)
    # gradient_i corresponds to entry (i + n - 1) of the full convolution
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad

    Search for a non‑negative height sequence with a low evaluation score.

    Parameters

    seed : int

        Time budget in seconds (the function stops a fraction of a second

    **kwargs : dict

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    rng = np.random.default_rng(seed)

    try:

    except Exception:
        n = 1024
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    # best‑ever bookkeeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_base = 0.07                     # a modest base learning rate

    tau_factor = 0.4                    # start with 40 % of the current max
