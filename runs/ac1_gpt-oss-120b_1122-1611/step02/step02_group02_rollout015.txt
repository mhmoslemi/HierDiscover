"""
Hybrid exponential‑mirror descent with occasional linear‑programming
flattening.

The objective to minimise is  2·n·max_k (a*a)[k]  with a_i ≥ 0,
∑a_i = 1.
We keep the sum exactly 1 (the value of the score is scale‑invariant).
The main loop is an exponentiated‑gradient (mirror descent w.r.t. the
KL‑divergence) that follows a (sub)gradient of the ‑max‑convolution.
The gradient is assembled from the partner‑vectors of the largest
convolution entries (the “active set”).  A momentum term gives a
smooth direction, the step‑size decays slowly.

Every few seconds we solve a tiny linear programme:
 maximise  ∑b_i   subject to   a⋆b ≤ M   (only the currently active

The LP finds a direction that can support more mass while keeping the
convolution under the current upper bound M.  After normalising the LP
solution we mix it into the current iterate – this often produces a
large “flattening” jump that plain gradient steps miss.

Random block perturbations and pairwise mass swaps are used
occasionally to escape plateaus, and a full restart is triggered if no
improvement is seen for a while.

The routine runs for roughly `budget_s‑1.5` seconds (leaving a safety
margin) and returns the best sequence found.
"""

def _fft_len_for_n(n: int) -> int:
    """Next power‑of‑two length that accommodates a full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution (real, non‑negative) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)[k] w.r.t. a.
    Returns w where w_i = a_{k-i} (0 outside the range).
    """
    n = a.size
    w = np.zeros_like(a)
    if k < n:
        # i = 0 … k
        w[: k + 1] = a[k :: -1]          # a[k], a[k‑1], …, a[0]
    else:
        # i = k‑(n‑1) … n‑1
        start = k - (n - 1)
        length = (2 * n - 1) - k
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _init_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    Afterwards normalised to sum = 1.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


def _build_lp_matrix(a: np.ndarray, idx: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build a sparse matrix A such that (A @ b)[r] = (a * b)[k_r],
    where k_r = idx[r] are the convolution rows we wish to enforce.
    """
    n = a.size
    rows = []
    cols = []
    data = []

    for r, k in enumerate(idx):
        # j runs over columns that appear in the k‑th convolution sum
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows.extend([r] * length)
        cols.extend(range(j_start, j_end + 1))
        # coefficients are a_{k-j} in reverse order of j
        i_start = k - j_end
        i_end = k - j_start
        coeffs = a[i_start : i_end + 1][::-1]   # reverse to match j order
        data.extend(coeffs.tolist())

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(idx), n))
    return A

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds (leaving a tiny safety margin) and uses a hybrid
    exponentiated‑gradient / occasional linear‑programming scheme.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    # ------------------------------------------------------------------
    # Determine problem size and initialise a candidate vector
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        # fallback size – a few thousand entries is a good trade‑off
        n = 3000
        base = None

    # start from the classic arcsine shape (symmetric) with a tiny jitter
    a = _init_arcsine(n)
    a += rng.normal(scale=0.001, size=n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Keep the best solution seen so far
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len)
    best_score = 2.0 * n * conv.max()
    best = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # optimisation parameters
    momentum = np.zeros_like(a)
    beta = 0.85                # momentum decay
    eta0 = 1.5                 # initial exponential‑gradient step size
    top_k = 30                 # active‑set size for the sub‑gradient
    last_improve = time.monotonic()
    stagnation_limit = 25.0    # seconds without improvement → restart
    lp_interval = 18.0         # seconds between LP attempts
    last_lp = time.monotonic()
    t_lp = 0.35                # mixing weight when an LP solution is used

    start_time = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # --------------------------------------------------------------
        # 1) compute self‑convolution and identify the active set
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        M = conv.max()

        # indices of the `top_k` largest convolution entries (sorted)
        if top_k >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            part = np.argpartition(-conv, top_k - 1)[:top_k]
            active_idx = part[np.argsort(-conv[part])]

        # --------------------------------------------------------------
        # 2) build the summed partner vector – a (sub)gradient of max‑conv
        # --------------------------------------------------------------
        w = np.zeros_like(a)
        for k in active_idx:
            w += _partner_vector(a, int(k))

        # momentum smoothing
        momentum = beta * momentum + (1.0 - beta) * w

        # --------------------------------------------------------------
        # 3) step‑size schedule (smooth decay)
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)
        # simple power‑law decay
        eta = eta0 / (1.0 + 6.0 * frac)

        # --------------------------------------------------------------
        # 4) exponentiated‑gradient update on the probability simplex
        # --------------------------------------------------------------
        factor = np.exp(-eta * momentum)
        a = a * factor
        a = np.clip(a, 1e-15, None)          # avoid exact zeros
        a /= a.sum()

        # --------------------------------------------------------------
        # 5) occasional stochastic exploration
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.015:
            a = _random_block_perturb(a, rng)
        elif r < 0.025:
            a = _pairwise_swap_perturb(a, rng)

        # --------------------------------------------------------------
        # 6) evaluate the current score (fast version)
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # 7) keep the best vector seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 8) restart if stagnation is observed
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > stagnation_limit:
            if rng.random() < 0.6:
                a = _init_arcsine(n)
            else:
                a = rng.gamma(shape=1.0, size=n)
                a /= a.sum()
            a += rng.normal(scale=0.001, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            momentum[:] = 0.0
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart (stagnation)")

        # --------------------------------------------------------------
        # 9) periodic linear‑programming flattening step
        # --------------------------------------------------------------
        if time.monotonic() - last_lp > lp_interval:
            last_lp = time.monotonic()
            # Build a small LP using only the current active rows
            try:
                A = _build_lp_matrix(a, active_idx)
                b_ub = np.full(A.shape[0], M, dtype=np.float64)
                c = -np.ones(n, dtype=np.float64)           # maximise sum(b)
                bounds = [(0.0, None)] * n

                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True, "time_limit": 5.0},
                )
                if res.success:
                    b = res.x
                    sum_b = b.sum()
                    if sum_b > 1e-12:
                        # Normalise the LP solution to have unit mass
                        g = b / sum_b
                        # Mix it with the current iterate
                        cand = (1.0 - t_lp) * a + t_lp * g
                        cand = np.clip(cand, 0.0, None)
                        cand /= cand.sum()
                        # Evaluate the mixed candidate
                        conv_cand = _conv_self(cand, fft_len)
                        cand_score = 2.0 * n * conv_cand.max()
                        if cand_score < best_score:
                            best_score = cand_score
                            best = cand.copy()
                            a = cand                     # keep the improvement
                            momentum[:] = 0.0            # reset momentum after a jump
                            last_improve = time.monotonic()
                            print(
                                f"[{last_improve - start_time:6.2f}s] LP step improved score = {best_score:.6f}"
                            )
                # If LP fails or gives no improvement we simply continue.
            except Exception as exc:
                # Any numerical issue – ignore and proceed.
                print(f"[{time.monotonic() - start_time:6.2f}s] LP error: {exc}")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to satisfy evaluate_sequence's limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
