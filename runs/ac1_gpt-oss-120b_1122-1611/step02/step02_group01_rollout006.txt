def project_simplex(v):
    # Ensures non-negative sum to 1
    if v.sum() == 1 and np.all(v>=0): return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho  (maybe slightly different)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All non-positive, return uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step function that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a smooth‑max (log‑sum‑exp) surrogate, projected gradient descent on the
probability simplex and occasional random perturbations/restarts.
The algorithm works for any length `n` (it uses the global
`height_sequence_1` if present) and stops automatically before the time
budget expires.  The code is pure NumPy (FFT based convolutions) and
contains no closures, lambda‑expressions or file/network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Return the next power‑of‑two length that can hold a full linear
    convolution of two vectors of length ``n``.
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_corr(m: int, n: int) -> int:
    """
    Return the next power‑of‑two length that can hold the linear
    convolution of a length‑``m`` vector with a length‑``n`` vector.
    Used for the correlation w * a_rev needed for the gradient.
    """
    size = m + n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Result length is 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _corr_w_a(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the vector r with
        r[i] = Σ_k w[k] * a[k-i]   (out‑of‑range indices give 0)
    This is the cross‑correlation of ``w`` with ``a`` and yields the
    gradient of the smooth‑max of the convolution.
    ``w`` has length 2*n-1, ``a`` length n.
    """
    # reverse a for the correlation
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    prod = fw * fa
    corr_full = np.fft.irfft(prod, n=fft_len)[: len(w) + len(a) - 1]
    offset = len(w) - 1                # index of zero shift
    # extract r[i] = corr_full[offset - i] for i = 0 .. n-1
    r = corr_full[offset - np.arange(len(a))]
    return r


def _softmax_weights(conv: np.ndarray, alpha: float):
    """
    Compute a soft‑max weighting of the convolution entries.
    Returns (w, approx_max) where

        w[i] = exp(alpha * (conv[i] - max(conv))) / Σ_j ...

    and approx_max = (log Σ_j exp(alpha * conv[j])) / alpha,
    which is a smooth approximation of max(conv).
    """
    cmax = conv.max()
    # shift to avoid overflow/underflow
    shifted = conv - cmax
    exp_scaled = np.exp(alpha * shifted)
    sum_exp = exp_scaled.sum()
    w = exp_scaled / sum_exp
    approx_max = cmax + np.log(sum_exp) / alpha
    return w, approx_max


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # Fast O(n log n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]               # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = _project_simplex(out)
    return out

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the protected
    ``evaluate_sequence`` score.  The routine runs for roughly ``budget_s``
    seconds (leaving a safety margin) and returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose an initial vector
    # --------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                total = a.sum()
                if total == 0.0:
                    a = np.full(cand.size, 1.0 / cand.size)
                else:
                    a = a / total
                n = a.size
            else:
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Pre‑compute FFT lengths
    # --------------------------------------------------------------
    fft_len_self = _fft_len_for_n(n)                     # for a*a
    fft_len_corr = _fft_len_for_corr(2 * n - 1, n)       # for w * a_rev

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len_self)
    best_score = 2.0 * n * conv_best.max()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # Hyper‑parameters
    alpha_start = 5.0          # smooth‑max temperature at start
    alpha_end = 50.0           # temperature at the end (more “max‑like”)
    base_lr = 0.5              # initial learning rate for gradient step
    min_lr = 1e-4              # lower bound for learning rate
    max_attempts = 12          # line‑search attempts per iteration
    block_perturb_prob = 0.015
    restart_gap = 40.0         # seconds without improvement before a restart

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1. Current convolution and score (exact, not the smooth proxy)
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # --------------------------------------------------------------
        # 2. Record the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3. Smooth‑max surrogate, gradient computation
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        total_time = deadline - start_time
        progress = min(1.0, elapsed / total_time)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        w, _ = _softmax_weights(conv, alpha)          # w sums to 1

        # gradient of the smooth max (factor 2 comes from ∂conv/∂a_i = 2 a_{k-i})
        grad = 2.0 * _corr_w_a(w, a, fft_len_corr)

        # --------------------------------------------------------------
        # 4. Projected gradient step with back‑tracking line search
        # --------------------------------------------------------------
        # learning‑rate schedule (decays towards the end)
        lr = base_lr * (1.0 - progress) + min_lr * progress
        improved = False

        for attempt in range(max_attempts):
            a_candidate = _project_simplex(a - lr * grad)

            # evaluate candidate (exact objective)
            conv_cand = _conv_self(a_candidate, fft_len_self)
            cand_score = 2.0 * n * conv_cand.max()

            if cand_score < cur_score:
                a = a_candidate
                improved = True
                break            # accept first improving step
            else:
                lr *= 0.5        # shrink step size and retry

        if not improved:
            # no improvement – keep the current point but still apply a tiny step
            # to escape possible plateaus (handled later by perturbations)
            pass

        # --------------------------------------------------------------
        # 5. Occasional random block perturbation (exploration)
        # --------------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 6. Soft restart if we are stuck for a while
        # --------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > restart_gap:
            a = best.copy()
            # modest multiplicative log‑normal noise on all entries
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = _project_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – re‑starting from best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best vector (clipped to the bounds enforced by
    # ``evaluate_sequence`` – non‑negative and ≤ 1000).
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
