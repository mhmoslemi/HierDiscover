def _lp_boost(a, M, mix, rng):
    n = a.shape[0]
    L = 2*n - 1
    # Build constraint matrix A (L x n)
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a
    c = -np.ones(n, dtype=np.float64)   # minimise -sum(b)
    bounds = [(0, None)] * n
    # Solve using Highs (default)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=bounds, method='highs')
    except Exception:
        return a
    if not res.success:
        return a
    b = res.x
    total = b.sum()
    if total <= 0:
        return a
    b_norm = b / total
    a_new = (1 - mix) * a + mix * b_norm
    a_new = _project_simplex(a_new)
    return a_new

def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    n = 1024   # we could try to increase but keep at 1024 for safety
    # initial sequence
    if 'height_sequence_1' in globals():
        seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if seq.size != n:
            if seq.size > n:
                seq = seq[:n]
            else:
                seq = np.concatenate([seq, np.zeros(n - seq.size)])
        a = np.maximum(seq, 0.0)
        sum_a = a.sum()
        if sum_a < 1e-12:
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a
    else:
        a = np.full(n, 1.0/n, dtype=np.float64)

    best = a.copy()
    conv = np.convolve(a, a)
    best_score = 2.0 * n * conv.max()  # sum=1
    print(f"[init] n={n}, score={best_score:.6f}")

    # Adam parameters
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1t = beta1
    b2t = beta2

    # temperature schedule
    alpha_start = 5.0
    alpha_end = 200.0
    alpha = alpha_start
    # number of iterations to reach alpha_end (smooth phase)
    smooth_iters = 300_000     # adjust later based on time
    alpha_inc = (alpha_end - alpha_start) / smooth_iters if smooth_iters>0 else 0.0

    # learning rate schedule
    lr0 = 0.2     # initial step size

    iteration = 0
    last_improve_iter = 0
    # main loop
    while time.monotonic() < deadline:
        iteration += 1

        # compute convolution for gradient and to know bound
        conv = np.convolve(a, a)
        max_conv = conv.max()

        # choose gradient
        if iteration <= smooth_iters:
            # smooth max gradient
            shift = max_conv
            exps = np.exp(alpha * (conv - shift))
            w = exps / exps.sum()
            grad = 2.0 * np.correlate(w, a, mode='valid')
            # update alpha gradually
            if alpha < alpha_end:
                alpha = min(alpha_end, alpha + alpha_inc)
        else:
            # hard max gradient (indicator at peak)
            k = int(conv.argmax())
            w = np.zeros_like(conv)
            w[k] = 1.0
            grad = 2.0 * np.correlate(w, a, mode='valid')

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        # bias correction
        m_hat = m / (1.0 - b1t)
        v_hat = v / (1.0 - b2t)
        # increase bias correction factors
        b1t *= beta1
        b2t *= beta2

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        # occasional random shake‑up (≈7 % of steps)
        if rng.random() < 0.07:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.3)))
            a_new[idx] *= factor
            a_new = _project_simplex(a_new)

        a = a_new

        # evaluate new candidate
        conv_new = np.convolve(a, a)
        score = 2.0 * n * conv_new.max()
        if score < best_score:
            best_score = score
            best = a.copy()
            last_improve_iter = iteration
            print(f"[improve] iter={iteration:,} score={best_score:.6f}")

        # occasional LP boost (every 80 k iterations)
        if _HAS_LINPROG and iteration % 80_000 == 0:
            # mixing coefficient t becomes smaller over time
            mix = max(0.1, 0.5 * (1.0 - iteration / 600_000))
            a = _lp_boost(a, max_conv, mix, rng)
            # recompute after boost for tracking
            conv = np.convolve(a, a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best = a.copy()
                last_improve_iter = iteration
                print(f"[lp‑boost] iter={iteration:,} score={best_score:.6f}")

        # early stopping when we have not improved for a while
        if iteration - last_improve_iter > 200_000:
            # add a larger random shake to try escaping
            a = _project_simplex(a + rng.normal(scale=0.01, size=n))
            last_improve_iter = iteration
            print(f"[restart] iter={iteration:,}")

        # occasional status
        if iteration % 100_000 == 0:
            print(f"[status] iter={iteration:,} best={best_score:.6f}")

        # target reached approx?
        if best_score <= 1.5025 * 1.01:
            print("[early stop] near target reached")
            break

    # final fine‑tuning with tiny simulated annealing moves
    while time.monotonic() < deadline:
        # pick two indices and move a tiny amount of mass
        i, j = rng.integers(0, n, size=2)
        if a[i] < 1e-12:

        delta = min(a[i], rng.random() * 0.02)   # up to 2 % of a[i]

        a_candidate[i] -= delta

            best = a_candidate.copy()
            a = a_candidate
            print(f"[anneal‑improve] score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    return [float(x) for x in best]

"""
Propose a step‑function height sequence with a low evaluation score.

The algorithm works in three phases:

1. **Smooth‑max Adam optimisation** – the objective `max(conv(a,a))`
   is approximated by a soft‑max with a temperature that is raised
   from a low to a high value.  Adam together with a Euclidean projection
   onto the probability simplex keeps the iterates feasible.

2. **Hard‑max (sub‑gradient) fine‑tuning** – once the temperature is high
   the soft‑max becomes an almost exact max, i.e. the gradient reduces to
   the sub‑gradient of the true max‑convolution.  A smaller learning rate
   is used and occasional “shake‑ups’’ keep the method from stagnating.

3. **Optional LP‑boost & simulated‑annealing** – every few‑ten‑thousand
   iterations a tiny linear‑programming step (via `scipy.optimize.linprog`)
   maximises the ℓ₁‑norm of a vector `g` under the current convolution bound.
   The current iterate is mixed with the normalised LP solution.
   After the time budget is almost exhausted a final random‑move annealing
   attempts a last improvement.

All iterates are projected onto the probability simplex so that the
sequence stays non‑negative and sums to one; the evaluation score is
`2·n·max(conv)`, which is exactly the value returned by `evaluate_sequence`
when the sum equals one (the function is scale‑invariant).  The routine
runs for the supplied time budget (≈ 1000 s) and returns the best sequence
found.
"""

import time
import numpy as np

# optional linear‑programming support
try:
    from scipy.optimize import linprog
    _HAS_LINPROG = True
except Exception:                     # pragma: no cover
    _HAS_LINPROG = False


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_gradient(a: np.ndarray,
                         conv: np.ndarray,
                         alpha: float) -> np.ndarray:
    """Gradient of the soft‑max surrogate of `max(conv(a,a))`."""
    shift = conv.max()
    # exponentials are taken of non‑positive numbers → stable
    exps = np.exp(alpha * (conv - shift))
    w = exps / exps.sum()
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def _hard_max_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the true max‑convolution (indicator at the peak)."""
    k = int(conv.argmax())
    w = np.zeros_like(conv)
    w[k] = 1.0
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def _lp_boost(a: np.ndarray,
              M: float,
              mix: float,
              rng: np.random.Generator) -> np.ndarray:
    """
    Solve
        maximise   sum(b)
        s.t.       a * b <= M   (element‑wise convolution)
                   b >= 0
    and return a convex combination (1‑mix)·a + mix·b̂,
    where b̂ is the normalised LP solution.
    """
    if not _HAS_LINPROG:
        return a
    n = a.shape[0]
    L = 2 * n - 1

    # build the Toeplitz constraint matrix A (L × n)
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a

    # maximise sum(b)  →  minimise –sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:                 # pragma: no cover
        return a
    if not res.success:
        return a

    b = res.x
    tot = b.sum()
    if tot <= 0.0:
        return a
    b_norm = b / tot
    a_new = (1.0 - mix) * a + mix * b_norm
    return _project_simplex(a_new)


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (a step‑function) that minimises
    `evaluate_sequence`.  The routine respects the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – 1024 gives a good trade‑off between quality and speed.
    n = 1024

    # initialise a feasible probability vector -------------------------------------------------
    if 'height_sequence_1' in globals():

            tmp = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if tmp.size != n:
                # truncate or pad with zeros

                    tmp = tmp[:n]
                else:
                    tmp = np.concatenate([tmp,
                                          np.zeros(n - tmp.size, dtype=np.float64)])
            a = np.maximum(tmp, 0.0)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
        except Exception:              # pragma: no cover
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    best = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Adam hyper‑parameters -------------------------------------------------
    beta1, beta2, eps = 0.90, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1t, b2t = beta1, beta2            # bias‑correction terms

    # temperature schedule for the smooth‑max -------------------------------------------------
    alpha_start, alpha_end = 5.0, 200.0
    alpha = alpha_start
    smooth_iters = 300_000              # after this we use the hard max
    alpha_inc = (alpha_end - alpha_start) / smooth_iters

    # learning‑rate schedule -------------------------------------------------
    lr0 = 0.20

    # time control -------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # MAIN OPTIMISATION LOOP
    while time.monotonic() < deadline:

        conv = np.convolve(a, a)

        if iteration <= smooth_iters:

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - b1t)
        v_hat = v / (1.0 - b2t)
        b1t *= beta1
        b2t *= beta2

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        # --- occasional random shake‑up --------------------------------------------
        if rng.random() < 0.07:                     # about 7 % of steps
            idx = rng.integers(0, n)

        a = a_new

        conv_new = np.convolve(a, a)
        score = 2.0 * n * conv_new.max()
        if score < best_score:

            if score_lp < best_score:

        # --- stagnation handling ----------------------------------------------------
        if iteration - last_improve > 200_000:
            # inject a moderate random perturbation to escape a plateau
            a = _project_simplex(a + rng.normal(scale=0.01, size=n))
            last_improve = iteration
            print(f"[restart] iter={iteration:,}")

        # --- status output -----------------------------------------------------------
        if iteration % 100_000 == 0:
            print(f"[status] iter={iteration:,} best={best_score:.6f}")

        # --- early stopping if we are already very close to the target -------------

    # FINAL SIMULATED‑ANNEALING STYLE TUNING (tiny random moves)
    while time.monotonic() < deadline:

            print(f"[anneal‑improve] score={best_score:.6f}")
