def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 2.0)

    # pick length n based on budget (more length may give better lower bound)
    n = 2048 if budget_s > 600 else 1024

    L = 2 * n - 1  # convolution length

    # target uniform convolution
    target_conv = np.full(L, 1.0 / L, dtype=np.float64)
    target_fft = np.fft.rfft(target_conv)  # length n
    target_mag = np.sqrt(np.abs(target_fft))

    # Build initial patterns (same as before) but sized n
    patterns = []
    patterns.append(np.full(n, 1.0 / n))
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)
    half = n // 2
    tent = np.concatenate([np.linspace(1, 2, half, endpoint=False),
                           np.linspace(2, 1, n - half)])
    tent /= tent.sum()
    patterns.append(tent)
    patterns.append(rng.dirichlet(np.ones(n)))
    if 'height_sequence_1' in globals():
        # similar resizing
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                if prev.size < n:
                    pad = np.full(n, 1.0 / n, dtype=np.float64)
                    pad[:prev.size] = prev
                    prev = pad
                else:
                    prev = prev[:n]
            prev = _project_simplex(prev)
            patterns.append(prev)
        except Exception:
            pass

    best_seq = None
    best_score = np.inf
    total_iters = 0

    # Choose initial candidate
    a = rng.choice(patterns).copy()
    a = _project_simplex(a)

    # Main loop of alternating projection
    while time.monotonic() < deadline:
        total_iters += 1

        # FFT of a zero-padded to length L
        A = np.fft.rfft(a, n=L)

        # Impose magnitude constraint
        # avoid division by zero: if magnitude zero keep zero.
        phase = np.angle(A)
        A = target_mag * np.exp(1j * phase)

        a_full = np.fft.irfft(A, n=L)

        a = a_full[:n]

"""
Propose a low‑score step‑function height sequence.

This implementation uses an *alternating projection* (Gerchberg–Saxton) scheme
that directly enforces the desired shape of the self‑convolution in the
frequency domain.

* A target convolution is taken to be the uniform distribution on the
  possible sums {0,…,2·n‑2}.  Its discrete Fourier transform determines the
  magnitude that the Fourier transform of the candidate sequence must have
  (|A(ω)|² = FFT(target)).
* Starting from a diverse set of feasible seeds (uniform, arcsine‑shaped,
  triangular, random Dirichlet, and the previous best if available) the
  algorithm iterates:
    1. Compute the DFT of the zero‑padded candidate.
    2. Replace its magnitude by the target magnitude while keeping the phase.
    3. Inverse‑transform back to the time domain.
    4. Enforce the primal constraints (non‑negativity, sum‑to‑one) by simple
       clipping and Euclidean projection onto the probability simplex.
* Every few iterations the true objective (`evaluate_sequence`) is evaluated.
  The best sequence seen so far is kept and returned when the time budget
  expires (or earlier if the target score 1.5025 is reached).

The method works well for moderate lengths (1024 – 2048).  Larger `n`
gives a lower theoretical floor (≈ 1) while still being cheap because each
iteration uses only one FFT and one inverse FFT (≈ O(n log n)).  Random
perturbations are injected occasionally to escape shallow plateaus.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1.0) > 0)[0]
    if rho.size == 0:
        # all non‑positive – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence discovered before the
    time budget expires.

    Parameters
    ----------
    seed : int, default 42
        Random seed for reproducibility.
    budget_s : float, default 1100
        Wall‑clock time (seconds) allocated for the search.  A safety margin
        of a few seconds is kept for finalisation and printing.

    Returns
    -------
    list[float]
        The candidate sequence (list of floats, all entries ≥ 0, sum ≈ 1).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time handling
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical floor.
    # Use n=2048 for a generous budget, otherwise fall back to 1024.
    n = 2048 if budget_s > 600 else 1024
    L = 2 * n - 1                     # length of the linear convolution

    # ------------------------------------------------------------------
    # Target uniform convolution (the ideal flat distribution)
    target_conv = np.full(L, 1.0 / L, dtype=np.float64)
    target_fft = np.fft.rfft(target_conv)      # length n
    target_mag = np.sqrt(np.abs(target_fft))   # magnitude we want for A

    # ------------------------------------------------------------------
    # Build a collection of diverse feasible starting points.
    patterns = []

    # Uniform distribution (feasible by construction)
    patterns.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Arcsine‑shaped – heavy tails, low centre
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)

    # Simple triangular (tent) shape
    half = n // 2
    tent = np.concatenate([np.linspace(1, 2, half, endpoint=False),
                           np.linspace(2, 1, n - half)])
    tent /= tent.sum()
    patterns.append(tent)

    # Random Dirichlet vector – stochastic diversity
    patterns.append(rng.dirichlet(np.ones(n)).astype(np.float64))

    # Re‑use the previous best solution if present.
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                if prev.size < n:
                    # pad at the beginning with a tiny uniform tail
                    pad = np.full(n, 1.0 / n, dtype=np.float64)
                    pad[:prev.size] = prev
                    prev = pad
                else:
                    prev = prev[:n]
            prev = _project_simplex(prev)
            patterns.append(prev)
        except Exception:
            pass  # ignore malformed global variable

    # ------------------------------------------------------------------
    # Search state
    best_seq = None
    best_score = float('inf')
    total_iters = 0

    # Choose an initial candidate at random from the pattern pool
    a = rng.choice(patterns).copy()
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Main optimisation loop – alternating projection
    #   (frequency‑domain magnitude ↔ time‑domain simplex)
    while time.monotonic() < deadline:
        total_iters += 1

        # 1. FFT of the zero‑padded candidate (length L)
        A = np.fft.rfft(a, n=L)

        # 2. Impose the target magnitude while preserving the current phase
        phase = np.angle(A)
        A = target_mag * np.exp(1j * phase)

        # 3. Inverse FFT back to the time domain (real signal)
        a_full = np.fft.irfft(A, n=L)

        # 4. Keep only the first n entries – the support constraint
        a = a_full[:n]

        # 5. Enforce non‑negativity and the ℓ₁‑normalisation (probability simplex)
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # 6. Occasional random shake‑up to avoid stagnation
        if rng.random() < 0.006:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[idx] *= factor
            a = _project_simplex(a)

        # 7. Periodic evaluation of the genuine objective.
        #    Checking every few iterations limits overhead.
        if total_iters % 10 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[improve] iter={total_iters}, n={n}, score={best_score:.6f}")
                # Early termination if we beat the target.
                if best_score <= 1.5025:
                    print("[early exit] target reached")
                    return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Fallback – should never happen but guarantees a return value.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] total_iters={total_iters}, best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
