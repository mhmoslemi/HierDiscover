"""
Smoothed max gradient descent with projection.

The algorithm minimizes
    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

by keeping the vector normalized to sum(a)=1 (scale‑invariant) and
iteratively reducing a smooth approximation of the max‑convolution
peak.  The smooth max is the log‑sum‑exp (aka soft‑max) of the
self‑convolution, which yields a differentiable surrogate whose
gradient is a weighted sum of the “partner” vectors of every
convolution entry.  At each iteration we

* compute the self‑convolution,
* form softmax weights with a temperature β that grows from a small
  value (smooth) to a large one (sharp),
* obtain the gradient via a single FFT‑based correlation,
* take a projected gradient step (mirrored onto the probability simplex
  by Duchi’s projection),
* occasionally perform inexpensive pair‑wise mass transfers and
  log‑normal block perturbations to escape shallow minima,
* restart from the current best after a period without improvement.

All operations are O(n log n) and run comfortably within the
1000 s budget.  The routine returns the best vector seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (real‑to‑real)."""
    L = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    fz = fx * fy
    conv = np.fft.irfft(fz, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2*len(a)-1)."""
    return _conv_fft(a, a)


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k = exp(beta * conv_k) / sum_k exp(beta * conv_k),
    using a numerically stable shift.
    """
    # shift by maximum to avoid overflow
    shifted = beta * conv
    max_shift = shifted.max()
    exp_shifted = np.exp(shifted - max_shift)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given probabilities w (length 2*n-1) representing the
    soft‑max of the convolution, return the gradient
    ∇_a  ( (1/beta) log Σ exp(beta * conv) )
    = 2 * (w ⋆ a)  (correlation of w with a).
    The result has length n.
    """
    n = a.size
    w_rev = w[::-1]
    # full linear convolution length = (2*n-1) + n - 1 = 3*n-2
    full = _conv_fft(w_rev, a)  # length 3n-2
    # slice that corresponds to indices i = 0 … n‑1
    start = 2 * n - 2
    grad = 2.0 * full[start : start + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the algorithm of
    Duchi, Shalev‑Shwartz, Singer, and Chandra (ICML 2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑d arrays")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not cond.any():
        # all entries are zero or negative; return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Contribution of a single entry a[idx] to the full self‑convolution.
    Returns a vector of length 2*n‑1 where
        v[k] = a[idx]   if k = idx + i for some i with 0 ≤ i < n,
        else 0.
    In other words it is the sequence a shifted by ``idx``.
    """
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx : idx + n] = a
    return v


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    protected ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a safety margin) and returns the best
    vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose an initial length and vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                total = a.sum()
                if total <= 0.0:
                    n = cand.size
                    a = np.full(n, 1.0 / n, dtype=np.float64)
                else:
                    a = a / total
            else:
                raise ValueError
        except Exception:
            # fallback
            n = 3809
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 3809
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure exact simplex property
    a = _project_simplex(a)

    # bookkeeping
    best = a.copy()
    conv_best = _conv_self(best)
    best_score = float(2.0 * n * conv_best.max())

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5‑s safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned for the 1000‑s budget)
    # ------------------------------------------------------------------
    eta_start = 1.5          # step size at the beginning (projected GD)
    eta_end = 0.02           # final step size
    beta_start = 0.3         # start temperature for the soft‑max
    beta_end = 150.0         # final temperature (approaches true max)
    block_prob_start = 0.08
    block_prob_end = 0.003
    pair_prob_start = 0.07
    pair_prob_end = 0.01
    max_frac_transfer = 0.25 # max fraction of a[i] that can be moved
    restart_gap = 40.0       # seconds without improvement before a soft restart
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1. Current convolution and score
        # --------------------------------------------------------------
        conv = _conv_self(a)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # --------------------------------------------------------------
        # 2. Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3. Smoothing schedule (soft‑max temperature) and step size
        # --------------------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        progress = elapsed / total_time
        beta = beta_start + (beta_end - beta_start) * progress
        eta = eta_start + (eta_end - eta_start) * progress

        # --------------------------------------------------------------
        # 4. Soft‑max weights and gradient (correlation)
        # --------------------------------------------------------------
        w = _smooth_weights(conv, beta)          # shape (2*n-1)
        grad = _gradient_from_weights(a, w)      # shape (n,)

        # --------------------------------------------------------------
        # 5. Projected gradient descent step
        # --------------------------------------------------------------
        a_tmp = a - eta * grad
        a = _project_simplex(a_tmp)

        # --------------------------------------------------------------
        # 6. Occasional block perturbation (global exploration)
        # --------------------------------------------------------------
        block_prob = block_prob_start + (block_prob_end - block_prob_start) * progress
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 7. Optional pair‑wise mass transfer (local fine‑tuning)
        # --------------------------------------------------------------
        pair_prob = pair_prob_start + (pair_prob_end - pair_prob_start) * progress
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], max_frac_transfer * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    # contribution vectors before the move
                    contrib_i = _contribution_vector(a, i)
                    contrib_j = _contribution_vector(a, j)
                    # Updated convolution after moving delta mass from i to j
                    new_conv = conv - delta * contrib_i + delta * contrib_j
                    new_score = 2.0 * n * float(new_conv.max())
                    if new_score < cur_score:
                        # accept the move
                        a[i] -= delta
                        a[j] += delta
                        a = _project_simplex(a)          # re‑normalise
                        conv = new_conv
                        cur_score = new_score
                        cur_max = float(conv.max())

        # --------------------------------------------------------------
        # 8. Soft restart when stuck for a while
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # apply a modest multiplicative log‑normal noise to all entries
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _project_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – re‑starting from best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to the bounds enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Smoothed‑max gradient descent with simplex projection.

The score to minimise is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

Scale invariance allows us to keep the vector normalised to
`sum(a) = 1`.  The algorithm replaces the non‑smooth `max` by a
smooth approximation `log‑sum‑exp` (soft‑max) with a temperature β.
The gradient of this surrogate is a weighted sum of the “partner”
vectors of every convolution entry; it can be obtained in O(n log n)
by a single FFT‑based correlation.

Each iteration therefore:

1. computes the self‑convolution `conv = a * a`,
2. builds soft‑max weights `w_k = exp(β·conv_k) / Σ exp(β·conv)`,
3. obtains the gradient `g = 2·(w ⋆ a)` (correlation of `w` with `a`),
4. takes a projected gradient step `a ← Π_simplex(a − η·g)`,
5. occasionally performs cheap block perturbations and pair‑wise
   mass transfers,
6. restarts from the best vector after a stagnation period.

All operations are O(n log n); with a modest `n ≈ 3800` the loop
runs many thousands of times within the 1000 s budget.  The function
returns the best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    fz = fx * fy
    conv = np.fft.irfft(fz, n=L)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·len(a)‑1)."""
    return _conv_fft(a, a)


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights w_k = exp(beta * conv_k) / Σ exp(beta * conv_k).
    Numerically stabilised by subtracting the maximum.
    """
    shifted = beta * conv
    max_shift = shifted.max()
    exp_shifted = np.exp(shifted - max_shift)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate:
        ∇_a ( 1/β log Σ exp(β·conv) ) = 2·(w ⋆ a)

    Returns a vector of length n.
    """
    n = a.size
    w_rev = w[::-1]
    full = _conv_fft(w_rev, a)               # length 3·n‑2
    start = 2 * n - 2
    grad = 2.0 * full[start : start + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1} (Duchi et al., ICML 2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑d arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not cond.any():
        # all non‑positive → return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:                     # numerical safety
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    out = out / s if s > 0 else np.full(n, 1.0 / n)
    return out


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Contribution of a single entry a[idx] to the self‑convolution.
    Returns a vector of length 2·n‑1 where the pattern `a` is shifted by `idx`.
    """
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx : idx + n] = a
    return v


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    protected ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a safety margin) and returns the best
    vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose an initial vector (use the supplied one if it looks sane)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                total = a.sum()
                a = a / total if total > 0 else np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            # fallback to a modest uniform start
            n = 3809
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 3809
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Small random perturbation of the initial point to avoid exact symmetry
    a *= np.exp(rng.normal(0.0, 0.05, size=a.size))
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best)
    best_score = float(2.0 * n * conv_best.max())

    start_time = time.monotonic()
    total_time = budget_s - 5.0                       # keep a safety margin
    deadline = start_time + total_time
    last_improve = start_time

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned for the ~1000 s budget)
    # ------------------------------------------------------------------
    eta_start = 1.5          # initial projected‑gradient step size
    eta_end = 0.02           # final step size
    beta_start = 0.3         # initial temperature for soft‑max
    beta_end = 8000.0        # final temperature (approaches true max)

    block_prob_start = 0.08
    block_prob_end = 0.003
    pair_prob_start = 0.07
    pair_prob_end = 0.01

    max_frac_transfer = 0.25   # maximal fraction of a[i] that may be moved
    restart_gap = 40.0        # seconds without improvement before a restart

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1. Current self‑convolution and score
        # --------------------------------------------------------------
        conv = _conv_self(a)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # --------------------------------------------------------------
        # 2. Record the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3. Progress‑dependent schedules
        # --------------------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        progress = elapsed / total_time

        # exponential schedule for β (smooth → sharp)
        beta = beta_start * (beta_end / beta_start) ** progress
        # linear decay for the step size η
        eta = eta_start + (eta_end - eta_start) * progress

        # --------------------------------------------------------------
        # 4. Compute soft‑max weights and the corresponding gradient
        # --------------------------------------------------------------
        w = _smooth_weights(conv, beta)          # length 2·n‑1, sums to 1
        grad = _gradient_from_weights(a, w)       # length n

        # --------------------------------------------------------------
        # 5. Projected gradient descent step (mirrored onto the simplex)
        # --------------------------------------------------------------
        a_tmp = a - eta * grad
        a = _project_simplex(a_tmp)

        # --------------------------------------------------------------
        # 6. Occasional global block perturbation (exploration)
        # --------------------------------------------------------------
        block_prob = block_prob_start + (block_prob_end - block_prob_start) * progress
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 7. Optional pair‑wise mass transfer (fine‑grained local search)
        # --------------------------------------------------------------
        pair_prob = pair_prob_start + (pair_prob_end - pair_prob_start) * progress
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], max_frac_transfer * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    contrib_i = _contribution_vector(a, i)
                    contrib_j = _contribution_vector(a, j)
                    new_conv = conv - delta * contrib_i + delta * contrib_j
                    new_score = 2.0 * n * float(new_conv.max())
                    if new_score < cur_score:
                        a[i] -= delta
                        a[j] += delta
                        a = _project_simplex(a)
                        conv = new_conv
                        cur_score = new_score
                        cur_max = float(conv.max())

        # --------------------------------------------------------------
        # 8. Soft restart if we have been idle for a while
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # mild log‑normal noise on all entries
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _project_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – re‑starting from best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to the bounds enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
