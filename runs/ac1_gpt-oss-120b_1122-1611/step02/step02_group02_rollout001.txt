def grad_softmax(a, conv, beta):
    maxc = conv.max()
    # use stable softmax compute exp(beta*(conv - maxc))
    exp_terms = np.exp(beta * (conv - maxc))
    w = exp_terms / exp_terms.sum()
    # compute gradient = 2 * (a_rev * w) slice
    a_rev = a[::-1]
    # FFT length for conv of length len(a_rev)+len(w)-1 = n + (2n-1) -1 = 3n-2
    l = len(a_rev) + len(w) -1
    # find next power of 2.
    fft_len = 1 << ((l - 1).bit_length())
    fa = np.fft.rfft(a_rev, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    prod = fa * fw
    conv_grad_full = np.fft.irfft(prod, n=fft_len)[:l]
    g = 2 * conv_grad_full[n-1:n-1 + a.size]
    return g

"""
Propose a candidate step‑function (non‑negative heights) that minimizes

        2·n·max_k (a * a)[k] / (∑ a)^2 .

The algorithm is a *mirrored exponentiated‑gradient* method that works on a
smooth approximation of the “max’’ – a soft‑max (or p‑norm) of the
self‑convolution.  At each iteration we

* compute the linear convolution a*a by FFT,
* build soft‑max weights w_k  ∝ exp(β·conv_k),
* obtain the (sub‑)gradient g = 2·(a_rev * w)   (a_rev = a reversed) by a
  second FFT; this is exactly the gradient of the smoothed objective,
* update a with an exponentiated‑gradient step a_i ← a_i·exp(−η·g_i),
* renormalise a to keep ∑a = 1,
* occasionally apply cheap stochastic perturbations (block scaling,
  pairwise mass swaps) to escape plateaus,
* keep the best vector ever seen.

β (soft‑max sharpness) and the step‑size η are increased / decreased
according to a simple schedule.  When no improvement is observed for a
while we restart from the current best with a small jitter instead of
discarding everything.

The whole routine runs for the allotted time (≈1000 s) and finally returns
the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n(n: int) -> int:
    """
    Length of the zero‑padded FFT that is sufficient for both the
    self‑convolution (length 2·n‑1) and the gradient convolution
    (length 3·n‑2).
    """
    needed = max(2 * n - 1, 3 * n - 2)
    return _next_pow2(needed)


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear self‑convolution a*a using FFT.
    Returns a vector of length 2·n‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_softmax(a: np.ndarray, conv: np.ndarray,
                     beta: float, fft_len: int) -> np.ndarray:
    """
    Given a (length n) and its self‑convolution `conv` (length 2·n‑1),
    compute the gradient of the soft‑max objective
        Sβ(a)= (1/β)·log Σ_k exp(β·conv_k)
    with respect to a.  The gradient is
        g = 2· (a_rev  *  w)        (linear convolution)
    where w_k = exp(β·conv_k) / Σ_j exp(β·conv_j) and a_rev = a reversed.
    The returned vector has length n.
    """
    # --- soft‑max weights -------------------------------------------------
    max_c = np.max(conv)                     # for numerical stability
    exp_scaled = np.exp(beta * (conv - max_c))
    w = exp_scaled / np.sum(exp_scaled)      # shape (2·n‑1,)

    # --- convolution a_rev * w -------------------------------------------
    a_rev = a[::-1]                          # reversed a, length n
    # FFTs
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    prod = fa_rev * fw
    # linear convolution length = n + (2·n‑1) – 1 = 3·n‑2
    conv_full = np.fft.irfft(prod, n=fft_len)[: 3 * a.size - 2]
    # The gradient entries are the central n entries of conv_full
    start = a.size - 1
    g = 2.0 * conv_full[start : start + a.size]
    return g


def _init_arcsine(n: int) -> np.ndarray:
    """
    Initialise with a discrete arcsine density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    The shape yields a naturally flat self‑convolution.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    delta = rng.random() * 0.3 * a[i]          # up to 30 % of a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds (leaving a small safety margin) and uses a
    mirrored exponentiated‑gradient / soft‑max descent.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initialise a candidate vector
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        # fallback size – a power of two makes FFTs slightly cheaper
        n = 4096
        base = None

    # start from the smooth arcsine shape with a tiny random jitter
    a = _init_arcsine(n)
    a += rng.normal(scale=0.001, size=n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    # pre‑computed FFT length (covers both convolutions we need)
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Keep the best solution observed so far
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    best_score = 2.0 * n * np.max(conv)      # sum(a)=1 ⇒ denominator = 1
    best = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # optimisation hyper‑parameters
    eta0 = 1.2               # initial step‑size for exponentiated update
    beta_start = 0.5         # soft‑max starts smooth
    beta_end = 8.0           # becomes sharp near the end
    decay_start = 0.15       # fraction of budget after which η decays fast
    stagnation_limit = 30.0  # seconds without improvement → restart
    momentum = np.zeros_like(a)
    beta_mom = 0.85           # momentum coefficient for the gradient

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)  # safety margin
    last_improve = start_time
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # compute current self‑convolution and score
        # --------------------------------------------------------------
        conv = _conv_self_fft(a, fft_len)
        cur_score = 2.0 * n * np.max(conv)

        # --------------------------------------------------------------
        # record improvement
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # time‑dependent schedules
        # --------------------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # step size η
        if frac < decay_start:
            eta = eta0 * (1.0 - frac / decay_start)          # gentle decay
        else:
            eta = eta0 * 0.1 * (1.0 - (frac - decay_start) / (1.0 - decay_start))

        # soft‑max sharpness β – increase slowly, then accelerate
        beta = beta_start + (beta_end - beta_start) * (frac ** 2)

        # --------------------------------------------------------------
        # compute soft‑max weights and gradient
        # --------------------------------------------------------------
        g = _gradient_softmax(a, conv, beta, fft_len)

        # momentum smoothing (helps stability)
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g

        # --------------------------------------------------------------
        # exponentiated‑gradient update (mirror descent with KL‑divergence)
        # --------------------------------------------------------------
        factor = np.exp(-eta * momentum)
        a = a * factor
        a = np.clip(a, 1e-14, None)       # avoid exact zeros
        a /= a.sum()                     # enforce Σa = 1

        # --------------------------------------------------------------
        # occasional stochastic exploration (≈2 % of the iterations)
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.015:
            a = _random_block_perturb(a, rng)
        elif r < 0.025:
            a = _pairwise_swap_perturb(a, rng)

        # --------------------------------------------------------------
        # restart if we have been stuck for a while
        # --------------------------------------------------------------
        if now - last_improve > stagnation_limit:
            # start from the best seen so far, but add a fresh jitter
            a = best.copy()
            a += rng.normal(scale=0.002, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            momentum[:] = 0.0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to satisfy evaluate_sequence's limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
