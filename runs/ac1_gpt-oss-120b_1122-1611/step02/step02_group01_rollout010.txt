-----------------

Search for a non‑negative step function that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a temperature‑controlled mirror‑descent on a smooth
approximation of the max‑norm.  At each iteration we

* compute the full self‑convolution with an FFT,
* form a soft‑max weighting of the K largest convolution entries,
* accumulate a weighted sum of the corresponding “partner” vectors,
* apply an exponentiated‑gradient (mirror‑descent) update on the
  probability simplex, and
* occasionally perform random block perturbations, pair‑wise mass
  transfers and a soft‑restart when progress stalls.

The temperature (β) and learning‑rate (η) are scheduled linearly so
that the algorithm focuses more sharply on the current peak as time
progresses.  The method works on any initial vector – we start from
the supplied ``height_sequence_1`` if it exists, otherwise from a

returning the best sequence seen.

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold the full linear
    convolution of two length‑n vectors.
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Contribution of a single entry ``a[idx]`` to the full
    self‑convolution (linear).  Length is ``2*n-1``.
    """
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx : idx + n] = a
    return v

        total = deadline - start_time
        # linear interpolation
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- soft‑max weighting of the largest convolution entries ----------
        if K_active >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            kth = conv.size - K_active
            active_idx = np.argpartition(conv, kth)[kth:]

        # Compute weighted partner vectors without allocating many temporaries
        grad = np.zeros_like(a)
        rev_a = a[::-1]                      # reversed copy once per iteration
        max_conv = conv.max()
        for k in active_idx:
            w = np.exp(beta * (conv[int(k)] - max_conv))   # 0 <= w <= 1
            if w == 0.0:
                continue
            k = int(k)
            if k < n:
                # indices 0 … k inclusive
                # partner_i = a[k - i] → slice a[k::-1]
                grad[: k + 1] += w * a[k :: -1]
            else:
                start = k - (n - 1)                     # first i with a partner
                length = (2 * n - 1) - k                # number of overlapping entries
                grad[start:] += w * rev_a[:length]

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- mirror‑descent update (exponentiated gradient) ----------------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---- occasional random block perturbation -------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ---- optional pair‑wise mass transfer ------------------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # move at most 30 % of the donor's mass
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    # effect on the convolution
                    contrib_i = _contribution_vector(a, i)
                    contrib_j = _contribution_vector(a, j)
                    new_conv = conv + delta * (contrib_j - contrib_i)
                    new_score = 2.0 * n * np.max(new_conv) / (np.sum(a) ** 2)
                    if new_score < cur_score:
                        a[i] -= delta
                        a[j] += delta
                        a = np.clip(a, 0.0, None)
                        a /= a.sum()
                        conv = new_conv
                        cur_score = new_score
                        cur_max = np.max(conv)

        # ---- soft restart when stuck ---------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # modest multiplicative log‑normal jitter on every coordinate
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
