"""
Propose a step‑function (probability vector) with a low evaluate_sequence score.

The algorithm is a *projected exponentiated‑gradient* optimizer that directly
minimises a smooth approximation of the worst self‑convolution entry.
Key ideas
---------
* Work on the simplex by using the multiplicative (KL‑mirror) update
      a_i ← a_i · exp(‑η·∂/∂a_i)   and renormalise.
* Approximate the nondifferentiable max by a soft‑max with temperature τ.
  τ is tied to the current maximal convolution value and slowly decayed,
  so the surrogate becomes sharper as the optimisation proceeds.
* All convolutions and correlations are performed with FFTs (O(m log m))
  where m ≥ 3 · n; this makes each iteration cheap enough to explore many
  random restarts and hyper‑parameter settings.
* The search is time‑bounded (default 1000 s).  It repeatedly
  - draws a promising initial distribution (uniform, power‑law, Dirichlet,
    or a mixture of decreasing/increasing shapes),
  - runs a few hundred exponentiated‑gradient steps,
  - keeps the best vector ever seen,
  - occasionally injects a small random shake‑up or re‑initialises to
    escape possible plateaus.
* Early stopping when the theoretical lower bound 2 n/(2 n‑1) is reached.

The function returns the best sequence (list of non‑negative floats)
found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute stable soft‑max weights w_k ∝ exp((conv_k - max)/τ).

    Parameters
    ----------
    conv : np.ndarray
        Full self‑convolution (length 2 n‑1).
    tau : float
        Temperature > 0.

    Returns
    -------
    np.ndarray
        Normalised weight vector of the same shape as ``conv``.
    """
    cmax = conv.max()
    # shift to avoid overflow, the largest exponent becomes 0
    shifted = (conv - cmax) / tau
    # numerical guard – the shift already makes the max exponent 0
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    return w


def _correlation_fft(w: np.ndarray, a_fft: np.ndarray, m: int, n: int) -> np.ndarray:
    """
    Linear correlation (w ⋆ a) for full length, using the already‑computed FFT
    of ``a`` (length m).  Returns the slice of length n that corresponds to the
    gradient formula 2·Σ_k w_k·a_{k‑i}.
    """
    # FFT of w (zero‑padded to length m)
    w_fft = np.fft.rfft(w, m)
    # cor = ifft( w_fft * conj(a_fft) )
    corr_full = np.fft.irfft(w_fft * np.conj(a_fft), m)
    # The required slice starts at index (2·n‑2) (see derivation in analysis)
    start = 2 * n - 2
    return 2.0 * corr_full[start:start + n]


def _make_initial_vector(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Produce a diverse initial probability vector of length ``n``.
    The choice is randomised among several families that are known to give
    reasonably flat self‑convolutions.
    """
    choice = rng.integers(0, 7)
    if choice == 0:          # uniform
        a = np.full(n, 1.0 / n)
    elif choice == 1:        # power law ~ 1/√(i+1)
        a = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 1.0)
    elif choice == 2:        # random power‑law with shift
        p = rng.uniform(0.3, 0.8)               # exponent ≈ 0.5
        s = rng.uniform(0.0, 5.0)               # shift
        a = 1.0 / np.power(np.arange(n, dtype=np.float64) + s + 1.0, p)
    elif choice == 3:        # mixture of decreasing and its reverse (U‑shape)
        base = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 1.0)
        mix = rng.random()
        a = mix * base + (1.0 - mix) * base[::-1]
    elif choice == 4:        # symmetric beta‑like shape
        alpha = rng.uniform(0.5, 3.0)
        beta  = rng.uniform(0.5, 3.0)
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    elif choice == 5:        # Dirichlet with moderate concentration
        concentration = rng.uniform(0.2, 1.0)
        a = rng.dirichlet(np.full(n, concentration))
    else:                    # use the global best if it exists, else random
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)
            if a.size != n:
                raise ValueError
            a = np.maximum(a, 0.0)
        except Exception:
            a = rng.random(n)
    # Normalise onto the simplex
    a_sum = a.sum()
    if a_sum == 0.0 or not np.isfinite(a_sum):
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function (height sequence) that minimises
    ``evaluate_sequence``.  The function respects the given time budget and
    returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function will stop a little before this).

    Returns
    -------
    list[float]
        The best height sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ---------- time handling ----------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ---------- problem size ----------
    try:
        known = np.asarray(height_sequence_1, dtype=np.float64)
        if known.ndim == 1 and known.size > 0:
            n = known.size
        else:
            raise ValueError
    except Exception:
        n = 1024

    # pre‑compute FFT length (≥ 3 · n for linear correlation)
    m = _next_pow2(3 * n)

    # ---------- bookkeeping ----------
    best = None
    best_score = np.inf
    total_iters = 0

    # ------------------------------------------------------------------
    # Helper: evaluate and possibly improve the current candidate
    # ------------------------------------------------------------------
    def _maybe_update(candidate):
        nonlocal best, best_score, total_iters
        total_iters += 1
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = candidate.copy()
            print(f"[improve] iter={total_iters:,} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main search loop – several short optimisation runs with diverse starts
    # ------------------------------------------------------------------
    # hyper‑parameters for the exponentiated‑gradient scheme
    eta0 = 0.5           # initial learning rate (multiplicative)
    tau_factor = 0.7    # τ = τ_factor * current_max   (relative temperature)
    tau_decay = 0.9995  # geometric decay of τ each iteration
    max_local_iters = 2500   # iterations per restart (adjustable)

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # initialise a fresh probability vector
        a = _make_initial_vector(rng, n)

        # keep a copy of the current vector for the inner loop
        eta = eta0
        local_iter = 0

        while time.monotonic() < deadline and local_iter < max_local_iters:
            local_iter += 1
            total_iters += 1

            # ----- FFT of a (used for both conv and correlation) -----
            a_fft = np.fft.rfft(a, m)

            # ----- self‑convolution (linear) -----
            conv_full = np.fft.irfft(a_fft * a_fft, m)
            conv = conv_full[: 2 * n - 1]

            # ----- temperature (smooth‑max) handling -----
            cur_max = conv.max()
            tau = max(cur_max * tau_factor, 1e-12)

            # ----- soft‑max weights -----
            w = _softmax_weights(conv, tau)

            # ----- gradient via correlation (FFT) -----
            grad = _correlation_fft(w, a_fft, m, n)   # shape (n,)

            # ----- exponentiated‑gradient update -----
            # Work in log‑space for stability
            log_a = np.log(a + 1e-30) - eta * grad
            a = np.exp(log_a)
            # renormalise onto the simplex
            s = a.sum()
            if s == 0.0 or not np.isfinite(s):
                a = np.full(n, 1.0 / n)
            else:
                a /= s

            # ----- occasional perturbation to escape plateaus -----
            if rng.random() < 0.001:
                noise = rng.normal(0.0, 0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()

            # ----- evaluate true objective occasionally (every iteration is cheap) -----
            _maybe_update(a)

            # ----- schedule updates -----
            eta *= 0.9999                 # slowly shrink step size
            tau_factor *= tau_decay       # cool the temperature

            # early stop if we are already at the theoretical bound
            lower_bound = 2.0 * n / (2 * n - 1)
            if best_score <= lower_bound * 1.01:
                print("[early stop] reached near‑optimal lower bound")
                break

        # after a local run, decide whether to keep going or re‑initialise
        # (the outer ``while`` already checks the global deadline)

    # ------------------------------------------------------------------
    # fallback in the extremely unlikely case that no candidate was recorded
    if best is None:
        best = np.full(n, 1.0 / n)
        best_score = float(evaluate_sequence(best.tolist()))
        print(f"[fallback] uniform score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f} after {total_iters:,} iterations")
    # Convert to plain Python floats for the harness
    return [float(x) for x in best]
