"""
Propose a candidate step‑function sequence.

The algorithm mixes three ideas:

1. **Projected Adam on a smooth‑max surrogate** – identical in spirit to the
   previous solution but uses a more aggressive temperature schedule and runs
   several independent restarts with diverse initialisations.

2. **Occasional linear‑programming boost** – given the current sequence `a`
   and its worst self‑convolution value `M`, we solve a small LP that maximises
   the ℓ₁‑norm of a partner sequence `b` under the constraints
   `a * b ≤ M`.  Scaling `b` back to the simplex and mixing it with `a`
   provably never increases `M` and often yields a strict improvement.

3. **Simple local perturbations** – a tiny random shake‑up is added from time
   to time to escape plateaus.

The search runs for the allotted time (default 1000 s).  Several vector
lengths (1024 and 2048) are tried; the best sequence overall is returned.
All operations respect the non‑negative simplex constraint, which makes the
objective scale‑invariant.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _smooth_max_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Given a convolution vector `conv`, return the soft‑max weights
    w_k ∝ exp((conv_k - max(conv))/temperature).
    """
    if temperature <= 0:
        temperature = 1e-12
    shift = (conv - conv.max()) / temperature
    w = np.exp(shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w / w_sum


# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, max_conv: float, deadline: float) -> tuple[np.ndarray, bool]:
    """
    Perform one LP‑based improvement step.
    Returns the new sequence and a flag indicating whether the step succeeded.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    # Build the Toeplitz‑like matrix A with entries a_i for the convolution a*b
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i iterates over valid a‑indices, j = k - i is the b‑index
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsc()

    # maximize sum(b)  <=> minimize -sum(b)
    c = -np.ones(n)
    ub = np.full(2 * n - 1, max_conv)

    # give the LP a tiny time budget so we never overshoot the global deadline
    time_left = deadline - time.monotonic()
    if time_left < 0.1:
        return a, False
    # SciPy's HiGHS accepts a 'time_limit' option (seconds)
    linprog_opts = {"time_limit": max(0.0, time_left - 0.05), "presolve": True}
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=[(0, None)] * n,
            method="highs",
            options=linprog_opts,
        )
    except Exception:  # pragma: no cover
        return a, False

    if not res.success:
        return a, False
    b = res.x
    sum_b = b.sum()
    if sum_b <= 1e-12:
        return a, False

    # Scale back to the simplex and mix with current a
    b_simplex = b / sum_b
    mix = 0.2  # moderate mixing factor
    a_new = (1 - mix) * a + mix * b_simplex
    a_new = _project_simplex(a_new)
    return a_new, True

                max_conv = conv.max()

                # ---- smooth‑max gradient ----
                w = _smooth_max_weights(conv, temperature)
                # correlation between w and a gives ∂/∂a of the smooth‑max
                grad = 2.0 * np.correlate(w, a, mode="full")[n - 1 : n - 1 + n]

                # ---- Adam update & projection ----
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)

                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)

                step = base_lr / (np.sqrt(v_hat) + eps)
                a_candidate = a - step * m_hat
                a_candidate = _project_simplex(a_candidate)

                # occasional random shake‑up (helps escape plateaus)
                if rng.random() < 0.008:
                    noise = rng.normal(0.0, 0.02, size=n)
                    a_candidate = _project_simplex(np.maximum(a_candidate + noise, 0.0))

                # evaluate true objective
                score = float(evaluate_sequence(a_candidate.tolist()))
                if np.isfinite(score) and score < best_local_score:
                    best_local_score = score
                    best_local = a_candidate.copy()
                    print(
                        f"[len={n}] iter={it:,} new best={best_local_score:.6f}"
                    )

                # accept the candidate (greedy)
                a = a_candidate

                # temperature cooling
                temperature = max(temperature * temp_decay, 1e-12)

                # LP improvement step
                if it % lp_interval == 0:
                    a_lp, ok = _lp_improve(a, max_conv, deadline)
                    if ok:
                        a = a_lp
                        # re‑evaluate (it is cheap compared to the LP itself)
                        score_lp = float(evaluate_sequence(a.tolist()))
                        if np.isfinite(score_lp) and score_lp < best_local_score:
                            best_local_score = score_lp
                            best_local = a.copy()
                            print(
                                f"[len={n}] LP step iter={it:,} improved score={best_local_score:.6f}"
                            )

                # early stop if we are already at the theoretical bound
                lower_bound = 2.0 * n / (2 * n - 1)
                if best_local_score <= lower_bound * 1.01:
                    break

            # after this start, keep the best seen
        return best_local, best_local_score

    # ------------------------------------------------------------------
    # Try a couple of lengths – larger n gives a tighter theoretical bound.
    # ------------------------------------------------------------------
    lengths = [1024, 2048]  # order matters: start with the faster one
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        seq, sc = _optimise_length(n)
        if seq is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_overall = seq.copy()
            print(f"[overall] length={n} best overall score={best_score:.6f}")

    # fallback (should never happen)
    if best_overall is None:
        best_overall = np.array([1.0])
        best_score = float(evaluate_sequence(best_overall.tolist()))
        print("[fallback] using trivial sequence")

    print(f"[final] best score={best_score:.6f} (length={best_overall.size})")
    return [float(x) for x in best_overall]
