def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _grad_exact(a: np.ndarray, kmax: int) -> np.ndarray:
    """Sub‑gradient of max{a*a}[k] at the index kmax."""
    n = a.shape[0]
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    i_valid = idx[mask]
    j_valid = j[mask]
    grad = np.zeros_like(a)
    # diagonal (i=j) contributes a_i, off‑diagonal contributes 2*a_j
    diag = i_valid == j_valid
    # off‑diagonal
    off = ~diag
    grad[i_valid[off]] = 2.0 * a[j_valid[off]]
    # diagonal
    grad[i_valid[diag]] = a[i_valid[diag]]
    return grad

            if a0.size != n:
                raise ValueError
            a = a0.copy()
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = _project_simplex(a)   # ensures sum = 1 and non‑negative

    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters
    base_lr = 0.5
    beta_min = 5.0
    beta_max = 200.0     # sharper approximation of the true max
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # pre‑compute FFT lengths
    L_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))      # ≥2n‑1
    L_grad = 1 << int(np.ceil(np.log2(3 * n)))          # ≥3n‑2

    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 1) linear convolution a*a via FFT (fast, O(n log n))

            a_rev_pad[:n] = a[::-1]

            W = np.fft.rfft(w_pad)
            A_rev = np.fft.rfft(a_rev_pad)
            grad_full = np.fft.irfft(W * A_rev, n=L_grad)
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]
        else:

            kmax = int(np.argmax(conv))
            grad = _grad_exact(a, kmax)

        # 4) Adam update

        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / np.sqrt(it)
        a_next = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            factor = float(np.exp(rng.normal(0.0, 0.13)))   # log‑normal ~ ±13 %
            a_next[idx] *= factor

        # 6) projection onto the simplex
        a_next = _project_simplex(a_next)

        # ------------------------------------------------------------------
        # 7) evaluate
        score = float(evaluate_sequence(a_next.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_next.copy()
            best_score = score
            last_improve = it
            print(f"[improve] iter={it:,} score={best_score:.6f} beta={beta:.1f}")

        # early exit if we have already reached the target
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping early")
            break

        a = a_next

        # ------------------------------------------------------------------
        # periodic status output
        if it % 50000 == 0:
            elapsed = time.monotonic() - start
            print(f"[status] iter={it:,} best={best_score:.6f} time={elapsed:.1f}s beta={beta:.1f}")

        # ------------------------------------------------------------------
        # restart with a small random jitter if we have been stuck for a while
        if it - last_improve > 200000:
            noise = rng.normal(0.0, 0.02, size=n)
            a = best * (1.0 + noise)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"[restart] iter={it:,} – jitter after stagnation")

    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    return [float(x) for x in best]

"""
Proposes a step‑function height vector that minimises
``evaluate_sequence``.  The method starts from an arcsine‑type
distribution (the continuous solution of a∗a≈const), runs a smooth‑max
Adam optimiser for most of the budget and finishes with an exact‑max
sub‑gradient phase.  All updates are projected onto the ℓ₁‑simplex
so the sequence stays non‑negative and has unit sum.  Occasional
log‑normal perturbations and a stagnation‑restart help to escape local
minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:                      # all entries non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _grad_exact(a: np.ndarray, kmax: int) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] at the index kmax."""
    n = a.shape[0]
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    i_valid = idx[mask]
    j_valid = j[mask]
    grad = np.zeros_like(a)
    # diagonal terms (i=j) contribute a_i, off‑diagonal contribute 2*a_j
    diag = i_valid == j_valid
    off = ~diag
    grad[i_valid[off]] = 2.0 * a[j_valid[off]]
    grad[i_valid[diag]] = a[i_valid[diag]]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)

    # ---------- timing ----------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # ---------- problem size ----------
    n = 1024

    # ---------- initial vector ----------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a0.size != n:

            a = a0.copy()

            a = None
    else:
        a = None

        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))

    a = _project_simplex(a)          # ensures sum = 1 and non‑negative

    # ---------- Adam state ----------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # ---------- hyper‑parameters ----------
    base_lr = 0.5
    beta_min = 5.0
    beta_max = 200.0                 # sharper approximation to the true max

    # ---------- bookkeeping ----------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT sizes: power‑of‑two ≥ needed linear‑convolution lengths
    L_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))   # ≥2n‑1
    L_grad = 1 << int(np.ceil(np.log2(3 * n)))       # ≥3n‑2

    it = 0
    last_improve = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # 1) linear convolution a*a via FFT (fast O(n log n))
        a_pad = np.zeros(L_conv, dtype=np.float64)
        a_pad[:n] = a

        conv_full = np.fft.irfft(A * A, n=L_conv)
        conv = conv_full[: 2 * n - 1]

        elapsed = time.monotonic() - start
        time_frac = elapsed / (budget_s - 0.5)               # in [0,1]

        a_next = _project_simplex(a_next)

        # 7) evaluate

            best_score = score
            last_improve = it
            print(f"[improve] iter={it:,} score={best_score:.6f} beta={beta:.1f}")

        # early termination if the target is already reached
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping early")
            break

        a = a_next

        # periodic status output
        if it % 50000 == 0:

            last_improve = it
            print(f"[restart] iter={it:,} – jitter after stagnation")

    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
