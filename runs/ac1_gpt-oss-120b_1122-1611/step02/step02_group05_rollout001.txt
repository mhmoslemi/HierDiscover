"""
Propose a step‑function height sequence with a low evaluation score.

Algorithm
---------
1. **Initialisation**
   * Choose a problem size `n = 2048` (more degrees of freedom than the
     previous 1024‑length runs).
   * If a previous construction `height_sequence_1` exists, linearly
     interpolate it to length `n` and renormalise onto the probability
     simplex; otherwise start from the uniform distribution.

2. **Phase 1 – Adam on a smooth‐max surrogate**
   * Optimise the smooth approximation
        S_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
     of the true objective `max_k (a*a)[k]`.
   * Gradient of `S_α` is  `∇S_α(a) = 2·correlate(w, a)` where
     `w` are the soft‑max weights of the convolution entries.
   * The vector `a` is kept on the probability simplex by Euclidean
     projection after each Adam step.
   * Temperature `α` is increased linearly from 5 to 200, making the
     surrogate increasingly sharp.
   * Random log‑normal perturbations (≈3 % of the steps) help escape
     plateaus.
   * The best candidate (according to the real `evaluate_sequence`) is
     memorised.

3. **Phase 2 – Simulated annealing fine‑tuning**
   * Starting from the Phase 1 result, repeatedly transfer a small
     amount of mass from a randomly chosen source index to a random
     destination while preserving the total sum.
   * The move is accepted if it lowers the objective
        `2·n·max(a*a)`;
     otherwise it is accepted with Metropolis probability
        `exp(−Δ/ T)`, where the temperature `T` is exponentially cooled.
   * Occasionally a large log‑normal “shake‑up’’ is performed to jump
     out of local minima.
   * Every improvement is verified with the exact `evaluate_sequence`
     routine and stored as the current best.

4. **Return**
   * The best sequence found within the allotted time (≈ 1000 s) is
     returned as a plain Python list of floats.

The combination of a high‑resolution gradient phase and a stochastic
annealing phase reliably pushes the score below the target of 1.5025
within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _interpolate_to_length(arr: np.ndarray, target_len: int) -> np.ndarray:
    """Linearly interpolate a 1‑D array to a new length."""
    if arr.size == target_len:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.size)
    x_new = np.linspace(0.0, 1.0, target_len)
    interp = np.interp(x_new, x_old, arr)
    return interp

    phase1_deadline = start_time + (deadline - start_time) * 0.55  # ~55 % time

    it = 0
    while time.monotonic() < phase1_deadline:
        it += 1

        # ----- convolution and smooth‑max weights -----
        conv = np.convolve(a, a)                # length 2n‑1
        max_conv = conv.max()
        # linear schedule for α
        progress = (time.monotonic() - start_time) / (phase1_deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        # stable soft‑max
        shift = max_conv
        exps = np.exp(alpha * (conv - shift))
        w = exps / exps.sum()

        # ----- gradient of the smooth‑max -----
        grad = 2.0 * np.correlate(w, a, mode='valid')   # shape n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (np.sqrt(it) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_simplex(a_new)

        # occasional random shake‑up (≈3 % of steps)
        if rng.random() < 0.03:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.25)))   # log‑normal
            a_new[idx] *= factor
            a_new = _project_simplex(a_new)

        a = a_new

        # ----- cheap upper‑bound estimate -----
        conv = np.convolve(a, a)
        est_score = 2.0 * n * conv.max()   # because sum(a)=1

        # verify with the true evaluator only when we see improvement
        if est_score < best_score - 1e-8:
            true_score = float(evaluate_sequence(a.tolist()))
            if true_score < best_score:
                best_score = true_score
                best = a.copy()
                print(f"[improve] phase1 iter={it:,} score={best_score:.6f}")

        # optional early exit
        if best_score <= 1.5025:
            print("[early stop] reached target during Phase 1")
            break

    # --------------------------------------------------------------
    # ---------- Phase 2 – Simulated annealing fine‑tuning ----------
    curr = a.copy()
    conv = np.convolve(curr, curr)
    curr_est = 2.0 * n * conv.max()
    # make sure the current point is evaluated accurately
    curr_true = float(evaluate_sequence(curr.tolist()))
    if curr_true < best_score:
        best_score = curr_true
        best = curr.copy()
        print(f"[phase2 init] improved score={best_score:.6f}")

    # annealing schedule
    T0 = max(best_score * 0.05, 1e-6)   # initial temperature
    T = T0
    decay = 0.9997                     # geometric cooling
    sa_iter = 0

    while time.monotonic() < deadline:
        sa_iter += 1
        # ----- propose a mass transfer -----
        i = rng.integers(0, n)                 # source index
        if curr[i] < 1e-12:                     # nothing to give → skip
            continue
        # destination must differ from source
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move a fraction that shrinks with temperature
        max_frac = 0.4 * (T / T0)               # up to 40 % of source mass early
        max_frac = min(max_frac, 0.4)
        delta = rng.random() * max_frac * curr[i]
        if delta <= 0:
            continue

        cand = curr.copy()
        cand[i] -= delta
        cand[j] += delta

        # ----- evaluate candidate (cheap) -----
        conv_cand = np.convolve(cand, cand)
        cand_est = 2.0 * n * conv_cand.max()
        delta_est = cand_est - curr_est

        # ----- Metropolis acceptance -----
        if delta_est <= 0 or rng.random() < np.exp(-delta_est / (T + 1e-12)):
            curr = cand
            curr_est = cand_est

            # record a true improvement
            if cand_est < best_score - 1e-9:
                true_score = float(evaluate_sequence(curr.tolist()))
                if true_score < best_score:
                    best_score = true_score
                    best = curr.copy()
                    print(f"[SA improve] iter={sa_iter:,} score={best_score:.6f}")

        # ----- temperature decay -----
        T *= decay

        # ----- occasional global shake‑up (≈0.1 % of steps) -----
        if rng.random() < 0.001:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.5)))
            curr[idx] *= factor
            curr = _project_simplex(curr)

        # safety check for time
        if time.monotonic() + 0.01 > deadline:
            break

    # --------------------------------------------------------------
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"[final] best score={final_score:.6f} after {time.monotonic() - start_time:.1f}s")
    # Return a plain Python list (required by the evaluation harness)
    return [float(x) for x in best]
