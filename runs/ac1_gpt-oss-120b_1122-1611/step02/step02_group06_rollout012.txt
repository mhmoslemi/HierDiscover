"""
Propose a better step‑function height sequence.

The evaluation metric (see the provided `evaluate_sequence`) is
    L(a) = 2·n·max(conv(a,a)) / (sum(a))² .
A lower value is better.  The original reference code used a projected
Adam optimiser on a soft‑max approximation of the max‑convolution.
Here we combine several ideas that have helped us push the score below
the previously reported 1.5275:

*   **Exponentiated‑gradient (mirror‑descent) on the probability simplex**.
    This guarantees non‑negativity and exact ℓ₁‑normalisation without a
    separate projection step, and often yields larger steps in the
    right direction.

*   **Linear‑temperature schedule for the smooth‑max** – the “beta”
    parameter controlling the softness is increased quickly (to 30) so
    that the optimiser focuses on the true maximum earlier.

*   **Two‑stage schedule:** a short fast optimisation on a modest size
    (n≈800) to discover a good shape, then an up‑sampling to a larger
    size (n≈1800) where more degrees of freedom allow the convolution to
    become flatter.  The up‑sampled vector is simply a repeat of the
    smaller one with a final normalisation.

*   **Occasional multiplicative “shakes”** (log‑normal perturbations) to
    escape plateaux, and periodic random restarts.

*   **Early stopping** when the bound 2 n/(2 n−1) is approached.

The whole search runs for the allotted time (default 1000 s) and returns
the best sequence seen.
"""

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Smooth approximation of max(conv(a,a)) and its gradient.

    Returns
        smooth_val  – (1/β)·log Σ exp(β·c_k)
        grad        – ∂/∂a smooth_max
        conv        – full convolution a*a (length 2n‑1)
        max_c       – ordinary (hard) maximum of conv
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                 # length 2n‑1
    max_c = np.max(conv)

    # soft‑max weights
    shifted = conv - max_c
    exp_w = np.exp(beta * shifted)
    w = exp_w / np.sum(exp_w)                # shape (2n‑1,)

    # gradient of the (soft) max:
    # ∂c_k/∂a_j = a_{k‑j} + a_{j‑k}  (zero‑padded)
    # Using cross‑correlation:
    conv2 = np.convolve(w[::-1], a)          # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)           # map to needed positions
    grad = 2.0 * conv2[idx]

    # smooth value (for logging only)
    smooth_val = (np.log(np.sum(exp_w)) + beta * max_c) / beta
    return smooth_val, grad, conv, max_c


# ----------------------------------------------------------------------
def _random_initial_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random point on the simplex.
    Three strategies are mixed:
        * uniform vector
        * Dirichlet (α=0.5) – fairly sparse
        * a sparse equal‑weight vector with random support size
    """
    which = rng.integers(0, 3)
    if which == 0:
        a = np.full(n, 1.0 / n)
    elif which == 1:
        a = rng.dirichlet([0.5] * n)
    else:
        k = rng.integers(max(5, n // 40), max(15, n // 10))
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0 / k
    return _project_simplex(a)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  The function will stop a few seconds before
        the limit to guarantee a return.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # safety margin – stop a few seconds early
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # ---- Phase‑1 (fast, moderate size) --------------------------------
    n1 = 800                     # a size that fits many iterations fast
    a = _random_initial_a(n1, rng)

    # Adam‑like parameters for the exponential‑gradient update
    base_eta = 0.4               # step‑size (will be divided by sqrt(t))
    beta = 0.4                   # start temperature for smooth‑max
    beta_max = 30.0
    beta_inc = 0.00012           # aggressive temperature increase

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init‑phase1] n={n1}, initial score={best_score:.6f}")

    it = 0
    # main optimisation loop for phase‑1
    while time.monotonic() < deadline * 0.5:   # spend about half the budget
        it += 1

        # ----- smooth max and gradient ---------------------------------
        _, grad, _conv, _maxc = _smooth_max_and_grad(a, beta)

        # ----- exponentiated‑gradient step (mirror descent) -----------
        eta = base_eta / np.sqrt(it)
        a = a * np.exp(-eta * grad)          # element‑wise
        a = a / np.sum(a)                    # simplex normalisation

        # ----- occasional log‑normal shake to escape plateaux ---------
        if rng.random() < 0.07:
            pos = rng.integers(0, n1)
            factor = float(np.exp(rng.normal(0.0, 0.22)))
            a[pos] *= factor
            a = a / np.sum(a)

        # ----- evaluate true objective ---------------------------------
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            print(f"[improve‑1] it={it:,} score={best_score:.6f}")

        # ----- temperature schedule ------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ----- occasional random restart -------------------------------
        if it % 60000 == 0:
            a = _random_initial_a(n1, rng)
            beta = 0.4

        # ----- early stop ------------------------------------------------
        theoretical_lb = 2.0 * n1 / (2 * n1 - 1)
        if best_score <= theoretical_lb * 1.015:
            print("[early‑stop‑1] close to theoretical bound")
            break

    # ------------------------------------------------------------------
    # ---- Phase‑2 (larger problem) ------------------------------------
    n2 = 1800                     # larger size gives a tighter possible bound
    # up‑sample the best vector from phase‑1
    repeat_factor = n2 // n1
    remainder = n2 % n1
    a_big = np.repeat(best, repeat_factor)
    if remainder:
        a_big = np.concatenate([a_big, best[:remainder]])
    a = _project_simplex(a_big)

    # reset optimiser parameters for the larger scale
    beta = 0.3
    it = 0
    print(f"[start‑phase2] n={n2}, lifted score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1

        _, grad, _conv, _maxc = _smooth_max_and_grad(a, beta)

        eta = base_eta / np.sqrt(it + 1e-9)
        a = a * np.exp(-eta * grad)
        a = a / np.sum(a)

        # occasional stronger shake
        if rng.random() < 0.05:
            pos = rng.integers(0, n2)
            factor = float(np.exp(rng.normal(0.0, 0.25)))
            a[pos] *= factor
            a = a / np.sum(a)

        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            print(f"[improve‑2] it={it:,} score={best_score:.6f}")

        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        if it % 80000 == 0:
            a = _random_initial_a(n2, rng)
            beta = 0.3

        theoretical_lb = 2.0 * n2 / (2 * n2 - 1)
        if best_score <= theoretical_lb * 1.01:
            print("[early‑stop‑2] bound reached")
            break

    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    return [float(x) for x in best]
