def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    import scipy.linalg, scipy.optimize

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)

    n = 1024

    target_sum = np.sqrt(2*n)

    if 'height_sequence_1' in globals():
        init = np.array(height_sequence_1, dtype=float)
        if init.size != n:
            # trim/pad
            if init.size > n:
                init = init[:n]
            else:
                init = np.concatenate([init, np.zeros(n - init.size)])
        # Normalize to target sum
        init_sum = np.sum(init)
        if init_sum > 0:
            init = init * (target_sum / init_sum)
        else:
            init = np.full(n, target_sum / n)
        f = init.copy()
    else:
        # uniform
        f = np.full(n, target_sum / n)

    # Best record
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, score={best_score:.6f}")

    # Hyperparameters
    t = 0.5
    min_t = 0.01
    decay = 0.9995   # per iteration
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # Compute convolution and bound M
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        # Build Toeplitz matrix for convolution f * b
        col = np.concatenate([f, np.zeros(n-1)])
        row = np.concatenate([np.array([f[0]]), np.zeros(n-1)])
        A = scipy.linalg.toeplitz(col, row)   # shape (2*n-1, n)

        # Solve LP: max sum(b)  => min -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(2*n-1, M)

        # Use highs method
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                    method='highs', options={'presolve': True})

        if not res.success:
            # fallback: perturb and continue
            print(f"[linprog fail] iter={iteration} {res.message}")
            # small random shake
            pos = rng.integers(0, n)
            factor = np.exp(rng.normal(0, 0.2))
            f[pos] *= factor
            f = f * (target_sum / np.sum(f))
            continue

        b = res.x
        sum_b = np.sum(b)

        # scale to target sum
        if sum_b > 1e-12:
            scale = target_sum / sum_b
        else:
            scale = 0.0
        g = b * scale

        # convex combination
        f = (1 - t) * f + t * g

        # occasionally random shake to escape local minima
        if rng.random() < 0.02:
            idx = rng.integers(0, n)
            f[idx] = f[idx] * np.exp(rng.normal(0, 0.3))
            f = f * (target_sum / np.sum(f))

        # enforce nonnegativity (should be)
        f = np.clip(f, 0, None)

        # evaluate
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = f.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # decay mixing coefficient
        t = max(min_t, t * decay)

        # early stop if we hit target bound
        if best_score <= 1.5025:
            print("[early exit] target reached")
            break

    print(f"[final] best_score={best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]

"""
Propose a step‑function sequence using a *linear‑programming driven* iterative
scheme.  The idea follows the classical “exchange” method:

1. Start from a random feasible vector `f` (non‑negative, fixed ℓ₁‑norm).
2. Let `M = max(conv(f,f))`.  All entries of the convolution are bounded by `M`.
3. Solve a linear program
       maximise   Σ b_j
       subject to conv(f,b) ≤ M   (linear in `b`)
                   b ≥ 0
   This finds a new candidate `b` that has the *largest possible* ℓ₁‑norm
   while still respecting the current peak bound `M`.
4. Scale `b` so that its ℓ₁‑norm equals the target norm and mix it with `f`
   via a convex combination `f ← (1‑t) f + t·b`.  The peak of `conv(f,f)` never
   increases; with a decreasing mixing weight `t` the bound steadily drops.
5. Repeat until the time budget expires, occasionally applying random
   “shakes’’ to escape plateaus and restarting from a fresh random point
   when progress stalls.

The LP is built from a Toeplitz matrix that represents the convolution
operator.  Using `scipy.optimize.linprog` (HiGHS backend) yields a fast
solution for `n = 1024`.  The algorithm is completely independent of the
previous Adam‑based approach and typically pushes the evaluation score
well below the 1.5275 obtained earlier.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # shift to zero‑sum simplex and use the standard algorithm
    # (see Duchi et al. 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(n) + 1
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # all entries are zero – fall back to uniform distribution
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_initial_a(n: int, rng: np.random.Generator, target_sum: float) -> np.ndarray:
    """Return a random feasible point on the simplex of radius `target_sum`."""
    # Dirichlet with α = 0.5 yields a moderately sparse distribution
    a = rng.dirichlet(np.full(n, 0.5))
    a = a * target_sum
    # Occasionally try an extreme sparse pattern
    if rng.random() < 0.2:
        k = rng.integers(max(5, n // 30), max(15, n // 8))
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = target_sum / k
    return a

    Search for a non‑negative step‑function height sequence with a low
    evaluation score.  The method uses a sequence of linear programs
    as described above and returns the best vector found before the time
    budget runs out.
    """
    rng = np.random.default_rng(seed)

    # ---------- problem size ----------
    n = 1024                     # matches the reference run
    target_sum = float(np.sqrt(2 * n))

    # ---------- initialise ----------
    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=float)
        if init.size != n:

        init_sum = init.sum()
        if init_sum > 0:
            f = init * (target_sum / init_sum)
        else:
            f = np.full(n, target_sum / n)
    else:
        f = _random_initial_a(n, rng, target_sum)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ---------- hyper‑parameters ----------
    t = 0.5                     # convex‑combination weight
    t_min = 0.02
    decay = 0.9995              # geometric decay of `t` each iteration
    shake_prob = 0.03           # probability of a random multiplicative shake
    restart_interval = 300      # iterations without improvement → restart
    last_improve = 0
    iteration = 0

    # safety margin before the hard deadline
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        iteration += 1

        # ---- current peak -------------------------------------------------
        conv_f = np.convolve(f, f)            # length 2n‑1
        M = float(np.max(conv_f))

        # ---- build Toeplitz matrix for conv(f, b) ≤ M --------------------
        col = np.concatenate([f, np.zeros(n - 1)])
        row = np.concatenate([[f[0]], np.zeros(n - 1)])
        A = toeplitz(col, row)                # shape (2n‑1, n)

        # ---- linear program: maximise Σ b_j  (i.e. minimise –Σ b_j) ------
        c = -np.ones(n)                       # because linprog minimises
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n               # b_j ≥ 0

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # LP failure – perform a small random shake and continue
            print(f"[linprog fail] iter={iteration} ({res.message})")
            idx = rng.integers(0, n)
            f[idx] *= np.exp(rng.normal(0.0, 0.2))
            f = f * (target_sum / np.sum(f))
            continue

        b = res.x
        sum_b = np.sum(b)

        # scale the LP solution to the target ℓ₁‑norm
        if sum_b > 1e-12:
            scale = target_sum / sum_b
        else:
            scale = 0.0
        g = b * scale

        # ---- convex update ------------------------------------------------
        f = (1.0 - t) * f + t * g

        # ---- occasional random shake ------------------------------------
        if rng.random() < shake_prob:
            pos = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.18)))
            f[pos] *= factor
            f = f * (target_sum / np.sum(f))

        # ---- enforce feasibility -----------------------------------------
        # (after the shake the sum may have drifted)
        f = _project_onto_simplex(f, target_sum)

        # ---- evaluate current candidate -----------------------------------
        # The exact score can be recomputed from `M` and the known ℓ₁‑norm,
        # which is faster than calling `evaluate_sequence` each time.
        score = 2.0 * n * M / (target_sum ** 2)

        # ---- keep the best ------------------------------------------------
        if score < best_score:
            best_score = score
            best = f.copy()
            last_improve = iteration
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # ---- mixing weight decay -----------------------------------------
        t = max(t_min, t * decay)

        # ---- restart if stuck --------------------------------------------
        if iteration - last_improve > restart_interval:
            # keep the best found so far, but give the search a fresh start
            f = _random_initial_a(n, rng, target_sum)
            last_improve = iteration
            print(f"[restart] iter={iteration}")

        # ---- early exit if we already beat the target --------------------
        if best_score <= 1.5025:
            print("[early exit] target bound reached")
            break

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # return a plain Python list of floats as required
    return [float(x) for x in best]
