def solve_lp(a, M):
    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n-1)])
    row = np.concatenate([a[:1], np.zeros(n-1)])
    A_ub = toeplitz(col, row) # shape (2n-1, n)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n) # maximize sum => minimize negative sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0:
        return None
    g = b_opt / sum_b
    return g

            alpha = 0.02

        a = np.maximum(a, 0)
        a_sum = a.sum()

            a = np.full(n, 1.0 / n)

"""
Propose a low‑score height sequence for the self‑convolution objective.

The core idea is an alternating linear‑program / line‑search scheme:
1.  Start from a heavy‑ends “arcsine” distribution (known to improve the
    max‑self‑convolution).
2.  Given the current vector a (sum = 1, a_i ≥ 0) compute M = max(a*a).
3.  Solve a linear program that **maximises** the ℓ₁‑norm of a
    partner vector b under the linear constraints   a * b ≤ M
    (convolution is linear in the second argument).  The optimum b has
    sum(b) > 1, i.e. there is slack in the bound.
4.  Normalise b to a probability vector g = b / sum(b).  Because
    a * g ≤ M / sum(b) the cross‑convolution is strictly smaller than M.
5.  Mix the current a with g.  The mixing coefficient α is chosen by a
    cheap 1‑dimensional golden‑section search that minimises the
    new max‑self‑convolution
        (1‑α)²·(a*a) + 2α(1‑α)·(a*g) + α²·(g*g).
    This guarantees a monotonic decrease of the objective.
6.  Occasionally inject a small random perturbation to escape plateaux.
7.  Keep the best sequence seen (according to the provided
    `evaluate_sequence` function) and stop as soon as the target
    score (≈ 1.5025) is reached or the wall‑clock budget (≈ 1000 s) is
    exhausted.

The algorithm works entirely on the probability simplex, never needs
to scale the vector (the objective is scale‑invariant) and uses only
standard SciPy linear programming facilities.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the Toeplitz matrix that maps a vector b to the
    discrete convolution a * b (full length 2·len(a)-1).

    The i‑th row corresponds to the coefficient vector for
    (a*b)[i] = Σ_j a[j]·b[i‑j]  (out‑of‑range indices are zero).
    """
    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    return toeplitz(col, row)


def _optimal_alpha(conv_aa: np.ndarray,
                   conv_ag: np.ndarray,
                   conv_gg: np.ndarray,
                   current_max: float) -> tuple[float, float]:
    """
    One‑dimensional minimax line‑search.

    Find α ∈ [0, 1] that minimises
        max_k  ((1‑α)²·conv_aa[k] + 2α(1‑α)·conv_ag[k] + α²·conv_gg[k])

    Returns the best α and the associated value.
    """
    lo, hi = 0.0, 1.0
    phi = (np.sqrt(5.0) - 1.0) * 0.5           # 0.618...
    best_alpha = 0.0
    best_val = current_max

    for _ in range(30):                        # enough for 1e‑9 precision
        t1 = lo + (1.0 - phi) * (hi - lo)
        t2 = lo + phi * (hi - lo)

        mix1 = ((1.0 - t1) ** 2) * conv_aa + (2.0 * t1 * (1.0 - t1)) * conv_ag + (t1 ** 2) * conv_gg
        mix2 = ((1.0 - t2) ** 2) * conv_aa + (2.0 * t2 * (1.0 - t2)) * conv_ag + (t2 ** 2) * conv_gg

        f1 = mix1.max()
        f2 = mix2.max()

        if f1 < f2:
            hi = t2
            if f1 < best_val:
                best_val = f1
                best_alpha = t1
        else:
            lo = t1
            if f2 < best_val:
                best_val = f2
                best_alpha = t2

    return best_alpha, best_val


def propose_candidate(seed: int = 42, budget_s: float = 1100) -> list[float]:
    """
    Search for a non‑negative step‑function height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence discovered before the
    time budget expires.

    Parameters
    ----------
    seed : int, default 42
        Random seed for reproducibility.
    budget_s : float, default 1100
        Wall‑clock time (seconds) allocated for the search.  A small
        safety margin is kept for finalisation.

    Returns
    -------
    list[float]
        The candidate sequence (list of floats, sum ≈ 1, entries ≥ 0).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time management
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Problem size – 1024 is a good compromise.
    n = 1024

    # ------------------------------------------------------------------
    # Initial pattern: arcsine‑shaped (heavy at the ends, light in the middle)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = np.maximum(a, 0.0)
    a = a / a.sum()                     # enforce ℓ₁ = 1

    # ------------------------------------------------------------------
    # Keep the best solution seen so far.
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] score={best_score:.6f}")

    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop.
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1. Current self‑convolution and its maximum.
        conv_aa = np.convolve(a, a)          # length 2n‑1
        M = conv_aa.max()

        # --------------------------------------------------------------
        # 2. Linear programme: maximise Σ b_j  subject to a * b ≤ M, b ≥ 0.
        #    The matrix delivering a * b is a Toeplitz matrix.
        A_ub = _build_conv_matrix(a)         # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Objective = -Σ b (linprog minimises ⇒ maximise Σ b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n           # b_j ≥ 0

        # Solve – use the high‑performance HiGHS interior‑point method.
        lp_res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                         bounds=bounds, method='highs',
                         options={'presolve': True})
        if not lp_res.success:
            print(f"[lp fail] iter={iteration} – falling back to current best")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            print(f"[lp zero] iter={iteration} – no feasible mass increase")
            break

        # --------------------------------------------------------------
        # 3. Normalise to obtain a feasible probability vector g.
        g = b / sum_b   # Σ g = 1, and a * g ≤ M / sum_b

        # --------------------------------------------------------------
        # 4. Cross‑convolution and self‑convolution of the candidate.
        conv_ag = np.convolve(a, g)
        conv_gg = np.convolve(g, g)

        # --------------------------------------------------------------
        # 5. Choose mixing coefficient α by minimising the new peak.
        alpha, new_max = _optimal_alpha(conv_aa, conv_ag, conv_gg, current_max=M)

        # If the line‑search does not improve, fall back to a tiny step.
        if alpha <= 0.0 or new_max >= M - 1e-12:
            alpha = 0.02
            # recompute the mixed maximum (cheap approximation)
            mixed = ((1.0 - alpha) ** 2) * conv_aa + (2.0 * alpha * (1.0 - alpha)) * conv_ag + (alpha ** 2) * conv_gg
            new_max = mixed.max()
            if new_max >= M:
                # No improvement possible – perform a random shake‑up.
                if rng.random() < 0.03:
                    idx = rng.integers(0, n)
                    factor = np.exp(rng.normal(0.0, 0.2))
                    a[idx] *= factor
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                continue

        # --------------------------------------------------------------
        # 6. Update the current distribution.
        a = (1.0 - alpha) * a + alpha * g
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        # --------------------------------------------------------------
        # 7. Periodically evaluate the true objective.
        if iteration % 5 == 0 or time.monotonic() >= deadline:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[improve] iter={iteration}, score={best_score:.6f}")

            if best_score <= 1.5025:
                print("[early exit] target reached")
                return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # Return the best sequence found.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
