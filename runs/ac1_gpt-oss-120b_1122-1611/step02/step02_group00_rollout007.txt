"""
Propose a candidate step function.

The algorithm uses a *soft‑max‑smoothed* gradient of the objective
   f(w) = max_k (w * w)[k]   (w ≥ 0, Σ w = 1)

Instead of a plain sub‑gradient that looks at a single index attaining
the maximum, we replace the max by a smooth approximation
   f_γ(w)=γ·log Σ_k exp((w*w)[k]/γ)

Its gradient is a convex combination of all shifted copies of w,
where the weights are the soft‑max probabilities of the convolution
entries.  The update is performed with an *exponentiated‑gradient*
(step w_i ← w_i·exp(−η·∂f_γ/∂w_i))  followed by a renormalisation
onto the probability simplex.  This respects the simplex geometry
and converges faster than a plain Euclidean projection.

To avoid stagnation the method periodically
* symmetrises the vector (w ← (w+w[::-1])/2),
* applies a cheap random block perturbation,
* and restarts from the best known solution after a long stretch without
  improvement.

All convolutions are computed with an FFT‑based routine (O(n log n)).
The whole search runs for the allotted time budget (≈1000 s) and
returns the best sequence found, scaled to the conventional
√(2 n) L¹‑norm required by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level as required by the harness.

def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers divided by 4**i (a_i = C(2i,i)/4**i)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # clip tiny negative round‑off errors
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation used as a jump."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:  # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max weights p_k ∝ exp(conv_k / γ).  Returns a vector that sums to 1."""
    # shift by the maximum to avoid overflow
    shift = conv - np.max(conv)
    # exponentiate with scaling by gamma; gamma>0
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        # fall back to uniform weights (should rarely happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute ∇f_γ(w) where f_γ is the soft‑max smoothed max‑convolution.
    The gradient is   2 * Σ_k p_k * shift_k(w) .
    """
    n = w.size
    # 1. self‑convolution
    b = convolve_fft(w, w)                     # length 2n‑1
    # 2. soft‑max weights over the convolution entries
    p = _softmax_weights(b, gamma)             # same length 2n‑1
    # 3. reverse the weight vector to obtain the appropriate shift pattern
    p_rev = p[::-1]
    # 4. convolution w * p_rev  → length 3n‑2
    conv = convolve_fft(w, p_rev)
    # 5. reverse the result to align indices and take the first n entries
    grad = 2.0 * conv[::-1][:n]
    return grad, np.max(b)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step function (list of floats) whose
    `evaluate_sequence` score is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # Choose the length of the candidate.
    # Re‑use a previously discovered construction if available,
    # otherwise fall back to a few thousand entries.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
            w = prev / np.sum(prev)                 # normalise to a probability vector
        except Exception:
            n = 6000
            w = catalan_sequence(n)
            w = w / np.sum(w)
    else:
        n = 6000
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # Keep the best vector seen so far (always a probability vector)
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)
    best_w = w.copy()
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Algorithm hyper‑parameters
    # --------------------------------------------------------------
    eta0 = 0.5                # initial step size for exponentiated‑gradient
    gamma_factor = 0.08       # γ = gamma_factor * max(b)  (soft‑max temperature)
    sym_interval = 1200       # how often to enforce symmetry
    block_interval = 3000     # how often to try a random block jump
    no_improve_limit = 30000 # after this many iterations without gain we reset
    max_exponent = 50.0       # clipping for the exponent in EG updates

    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 4.0):
        # (a) evaluate current objective
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b

        # (b) record improvement
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # (c) compute the soft‑max‑smoothed gradient
        gamma = max(gamma_factor * max_b, 1e-12)
        grad, _ = _smoothed_gradient(w, gamma)

        # (d) exponentiated‑gradient update (with clipping of the exponent)
        eta = eta0 / np.sqrt(iteration + 1.0)
        exponent = -eta * grad
        np.clip(exponent, -max_exponent, max_exponent, out=exponent)
        w = w * np.exp(exponent)
        # renormalise to the simplex
        w_sum = np.sum(w)
        if w_sum == 0.0:                 # numerical safety
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # (e) occasional symmetry enforcement
        if iteration % sym_interval == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])       # make the vector (approximately) symmetric

        # (f) occasional random block perturbation – keep it if it improves
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best score={best_score:.6f}")

        # (g) long‑run stagnation handling – reset with a perturbed best
        if no_improve > no_improve_limit:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but we output the
    # traditional √(2 n) L¹‑norm so the numbers are comfortable.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum   # best_w already sums to 1

    # sanity check with the official evaluator (will be silent in the final run)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best score={best_score:.6f}, "
              f"after scaling evaluate={final_score:.6f}")
    except Exception:
        # evaluate_sequence might not be imported in a dry‑run environment
        print(f"[final] length={n}, best score={best_score:.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in final_seq]
