"""
propose_candidate
-----------------

Entropic (multiplicative) gradient descent on a smoothed‐max of the
self‑convolution, with FFT‑based convolutions.  The objective is

    f(w) = max_k (w * w)[k]       (w ≥ 0,  Σ w = 1)

which is convex.  The pointwise max is approximated by a
soft‑max (log‑sum‑exp) with a temperature that grows as √iteration,
turning the problem into a (smooth) convex optimisation that can be
handled by simple first‑order updates.

At each iteration:
    1. compute the linear convolution b = w * w (FFT, O(n log n));
    2. build a probability vector α over the 2 n−1 convolution
       entries by α_k ∝ exp(β·(b_k−max b));
    3. the gradient of the smoothed objective is
           ∇f ≈ 2·correlation(w,α)
       which we compute with another FFT convolution of w reversed
       and α;
    4. update multiplicatively: w ← w·exp(−η·∇f) and renormalise;
    5. occasional symmetric averaging and cheap random block
       perturbations (the same routine used in the reference
       solution) are applied to escape plateaus.

The algorithm works for any length ≥ 2; if a previous construction
(`height_sequence_1`) is available its length is reused, otherwise a
default length of 8000 is chosen.  After the search finishes the best
vector is scaled to the conventional √(2 n) ℓ¹‑norm (the score is
scale‑invariant) and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers C_i / 4**i.
    Recurrence a_{i+1}=a_i·(2 i+1)/(2 i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution using a real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Numerical noise can produce tiny negative numbers; clip harmlessly.
    np.maximum(c, 0.0, out=c)
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump: multiply a random block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and ~8 % of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetric smoothing
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max:   g = 2·corr(w, α)
    where α is a probability vector of length 2·n−1.
    Implemented via a second FFT convolution:
        corr(w,α)[j] = sum_i w[i]·α[i+j]  (j = 0…n−1)
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = 3·n−2
    start = w.shape[0] - 1                     # index that aligns with j=0
    return 2.0 * conv[start:start + w.shape[0]]


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is as low as possible.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a length.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
            w = prev / np.sum(prev)          # normalise to a probability vector
        except Exception:
            n = 8000
            w = catalan_sequence(n)
            w = w / np.sum(w)
    else:
        n = 8000
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = float(2 * n * np.max(b))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # Hyper‑parameters (tuned empirically)
    eta0 = 5.0               # base step size for multiplicative update
    beta0 = 0.5              # temperature for soft‑max (grows as √iter)
    beta_max = 200.0        # safety cap
    reset_patience = 25000  # iterations without improvement before a reset
    block_every = 3000      # apply a cheap random jump every this many iters
    symmetrize_every = 500  # enforce symmetry regularly

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # --------------------------------------------------------------
        # 1. self‑convolution and current (hard) score
        # --------------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 2. Smooth‑max (soft‑max) weights α
        # --------------------------------------------------------------
        beta = beta0 * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        # shift by the maximum for numerical stability
        diff = b - max_b
        exp_term = np.exp(beta * diff)          # max element becomes 1.0
        alpha = exp_term / np.sum(exp_term)     # length 2·n−1, sums to 1

        # --------------------------------------------------------------
        # 3. Gradient of the smoothed objective
        # --------------------------------------------------------------
        grad = correlation_grad(w, alpha)       # shape (n,)

        # --------------------------------------------------------------
        # 4. Multiplicative update (entropic mirror descent)
        # --------------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1)
        w = w * np.exp(-eta * grad)
        # avoid all‑zero vectors after extreme updates
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 5. Periodic symmetrisation (the optimal solution is symmetric)
        # --------------------------------------------------------------
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= np.sum(w)

        # --------------------------------------------------------------
        # 6. Occasional cheap random jump (helps escape plateaus)
        # --------------------------------------------------------------
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = float(2 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best score={best_score:.6f}")

        # --------------------------------------------------------------
        # 7. Reset if stuck for a long time
        # --------------------------------------------------------------
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but we keep the
    # conventional √(2 n) ℓ¹‑norm for nice looking numbers.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # sanity check with the provided evaluator (optional, for logging)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:            # the harness will call evaluate_sequence itself
        final_score = float('nan')
    print(f"[final] length={n}, best score (raw)={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
