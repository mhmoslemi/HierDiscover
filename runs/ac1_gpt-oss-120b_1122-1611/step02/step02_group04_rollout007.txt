"""
Propose a step‑function sequence with very low evaluation score.

The algorithm consists of three parts:

1. **Arcsine initialisation** – a discrete analogue of the continuous
   arcsine density f(x) ∝ 1/√(x·(N‑x)).  Its self‑convolution is (almost)
   uniform, i.e. the maximal auto‑correlation is close to the theoretical
   lower bound 2·N/(2·N‑1).

2. **Two‑phase Adam optimisation**  
   *Phase A* (first ~10 % of the time): minimise the L2 distance between the
   self‑convolution and the uniform target (which keeps the whole profile
   flat).  
   *Phase B* (remaining time): minimise a smooth approximation of the
   max‑convolution using a log‑sum‑exp surrogate.  Both phases use the same
   Adam optimiser and a Euclidean projection onto the probability simplex
   (∑a_i = 1, a_i ≥ 0).

3. **Keep the best sequence** – after each iteration the true (secure)
   `evaluate_sequence` is called; the best sequence seen so far is kept.
   If the lower bound is reached (within 1 %), the search stops early.

The whole routine respects the 1000 s budget, uses only NumPy and runs on a
single thread.  It works for any length `n` (defaults to the length of the
global `height_sequence_1` if provided, otherwise 1024) and returns a list
compatible with `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_initial(n: int) -> np.ndarray:
    """
    Return a length‑`n` probability vector that approximates the discrete
    arcsine distribution f(i) ∝ 1/√((i+1)*(n-i)).
    """
    # i runs from 0 to n‑1
    i = np.arange(1, n + 1)                # 1 … n
    a = 1.0 / np.sqrt(i * (n - i + 1))      # 1/√(i·(n‑i+1))
    a = a / a.sum()                         # normalise to sum = 1
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function height sequence with a small
    evaluation score.  Returns the best list found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ----------- timing -------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ----------- problem size -------------------------------------------
    try:
        # try to reuse the length of the previously known best sequence
        known = np.asarray(height_sequence_1, dtype=np.float64)
        if known.ndim == 1 and known.size > 0:
            n = int(known.size)
        else:
            raise ValueError
    except Exception:
        n = 1024

    # ----------- initial point (arcsine) --------------------------------
    a = _arcsine_initial(n).astype(np.float64)

    # sanity – projection (should already be on the simplex)
    a = _project_simplex(a)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_base_phase_a = 0.10   # larger step while flattening L2‑error
    lr_base_phase_b = 0.05   # finer step for the max‑surrogate

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for the two phases
    total_time = budget_s
    phase_a_fraction = 0.12   # roughly the first 12 % of the budget
    phase_a_deadline = time.monotonic() + phase_a_fraction * total_time

    # parameters for the smooth‑max surrogate
    tau_factor = 0.5          # initial temperature relative to current max
    tau_decay = 0.9999

    iteration = 0
    lower_bound = 2.0 * n / (2 * n - 1)

    while time.monotonic() < deadline:
        iteration += 1

        # ------------------ convolution & basic statistics ----------------
        conv = np.convolve(a, a)                 # length 2n‑1
        sum_a = a.sum()                          # should be 1 (projected simplex)

        # ------------------ target (uniform) for L2 phase ----------------
        target = (sum_a * sum_a) / (2 * n - 1)    # = 1/(2n‑1) after normalisation

        # ------------------ gradient computation -------------------------
        if time.monotonic() < phase_a_deadline:
            # Phase A – minimise L2 distance to the uniform target
            err = conv - target
            grad = 4.0 * np.correlate(err, a, mode='full')[n - 1:n - 1 + n]
            lr_base = lr_base_phase_a
        else:
            # Phase B – minimise a smooth approximation of the max entry
            current_max = conv.max()
            tau = max(current_max * tau_factor, 1e-12)
            shifted = (conv - current_max) / tau      # safely shifted
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / w_sum
            grad = 2.0 * np.correlate(w, a, mode='full')[n - 1:n - 1 + n]
            lr_base = lr_base_phase_b

        # ------------------ Adam update ---------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr_base / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat

        # ------------------ simplex projection ---------------------------
        a_candidate = _project_simplex(a_candidate)

        # ------------------ occasional random shake ----------------------
        if rng.random() < 0.004:   # a little noise to escape tiny plateaus
            noise = rng.normal(0.0, 0.018, size=n)
            a_candidate = a_candidate + noise
            a_candidate = _project_simplex(np.maximum(a_candidate, 0.0))

        # ------------------ evaluate true objective ----------------------
        score = float(evaluate_sequence(a_candidate.tolist()))

        # ------------------ bookkeeping ---------------------------------
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_candidate.copy()
            print(f"[improve] iter={iteration:,} score={best_score:.6f}")

        # early stop if we are essentially at the theoretical optimum
        if best_score <= lower_bound * 1.01:
            print("[early stop] reached near‑optimal bound")
            break

        # ------------------ prepare next iteration ----------------------
        a = a_candidate

        # temperature cooling (only relevant in phase B)
        tau_factor *= tau_decay
        if iteration % 4000 == 0:
            # keep tau from collapsing too fast
            tau_factor = max(tau_factor, 1e-6)

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    return [float(x) for x in best]
