"""
Propose a step‑function sequence that minimises
    2·n·max_k (a*a)_k / (∑ a_i)²

The optimiser uses:
* Adam on the simplex (projected Euclidean gradient)
* A smooth‑max surrogate whose temperature λ is increased aggressively
* FFT‑based self‑convolution and correlation for speed
* Enforced symmetry a[i]=a[n‑1‑i]
* Random jitter when progress stalls
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear self‑convolution of a using a real FFT."""
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    return conv


def _full_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Full correlation (w ⋆ a) of two real vectors using FFT.
    The length of the result is len(w)+len(a)-1.
    """
    n_w = w.shape[0]
    n_a = a.shape[0]
    m = _next_pow2(n_w + n_a - 1)
    W = np.fft.rfft(w, n=m)
    A = np.fft.rfft(a, n=m)
    corr = np.fft.irfft(W * np.conj(A), n=m)[: n_w + n_a - 1]
    return corr
