def simplex_projection(v):
    """Project vector v onto probability simplex (sum=1, v>=0)."""
    # from https://github.com/jaason/python-simplex-projection ... 
    n = v.shape[0]; u = np.sort(v)[::-1]; cssv = np.cumsum(u) - 1; 
    rho = np.where(u > cssv / np.arange(1,n+1))[0][-1]; 
    theta = cssv[rho] / (rho + 1); 
    w = np.maximum(v - theta, 0); return w

def _conv_fft(a,b,fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    fc = fa * fb
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: a.size + b.size - 1]

"""
Propose a step‑function that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a smooth‐approximation of the max (log‑sum‑exp) and
multiplicative mirror‑descent on the probability simplex.
The gradient of the soft‑max approximation is obtained cheaply
with one additional FFT‑based convolution per iteration.
Random block perturbations, occasional pairwise mass transfers,
and a soft‑restart scheme improve exploration.
The routine runs for the allotted time budget and returns the
best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative vector ``a`` using FFT.
    Result length = 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two vectors ``a`` and ``b`` using FFT.
    Returns an array of length ``len(a)+len(b)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    fc = fa * fb
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: a.size + b.size - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, Σ x_i = 1}.  Implements the algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: if rounding errors make sum slightly off, renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_simplex(out)

    Search for a non‑negative step‑function that minimises the protected
    ``evaluate_sequence`` score.  The routine respects the 1000 s time
    budget (leaving a small safety margin) and returns the best vector
    found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Initialise the candidate vector

    default_n = 256
    if "height_sequence_1" in globals():

            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():

        except Exception:                     # pragma: no cover
            a = None
    else:
        a = None

    else:

        else:
            a = a / total                       # normalise to sum = 1

    fft_len_grad = _next_pow2(3 * n - 2)          # for a * w_rev

    best = a.copy()
    conv_best = _conv_self(best, fft_len_self)

    start_time = time.monotonic()
    last_improve = start_time

    # hyper‑parameters (tuned empirically)
    eta_start = 0.8           # initial learning‑rate for EG step
    eta_end = 0.02            # final learning‑rate
    beta_start = 0.5          # start smoothness for log‑sum‑exp
    beta_end = 12.0           # end – close to true maximum
    block_perturb_prob = 0.02

    max_frac_transfer = 0.30  # max fraction of mass moved in a pair move
    restart_gap = 35.0        # seconds without improvement before a soft restart

    # --------------------------------------------------------------

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    iteration = 0
    while time.monotonic() < deadline:

        # ---- self‑convolution and current score --------------------
        conv = _conv_self(a, fft_len_self)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- keep best --------------------------------------------
        if cur_score < best_score:

            best = a.copy()

        # ---- schedule for learning‑rate and beta -------------------
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)

        beta = beta_start + (beta_end - beta_start) * progress

        # shift by cur_max to avoid large exponentials

        w = exp_shift / exp_shift.sum()          # length 2*n-1, sums to 1

        # partner vectors summed with weight w <=> 2 * (a * w_rev) slice
        w_rev = w[::-1]
        conv_aw = _conv_fft(a, w_rev, fft_len_grad)          # length 3*n-2

        # clip exponent to avoid under/overflow

        exp_arg = np.clip(exp_arg, -50.0, 50.0)

        a = a * factor
        a = _project_simplex(a)                         # enforce sum=1, non‑neg.

                i, j = rng.integers(0, n, size=2)
                if i == j:

                max_delta = min(a[i], max_frac_transfer * a[i])
                if max_delta <= 0.0:

                delta = rng.random() * max_delta
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_simplex(a_candidate)

                conv_cand = _conv_self(a_candidate, fft_len_self)
                cand_score = 2.0 * n * conv_cand.max()

                    a = a_candidate
                    cur_score = cand_score
                    cur_max = conv_cand.max()
                    # accept the first improving move
                    break

        now = time.monotonic()

            a = a * noise
            a = _project_simplex(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # Return the best vector found (clipped to the bounds enforced by
    # ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
