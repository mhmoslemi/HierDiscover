"""
Iterative LP‑based flattening of the self‑convolution.

The algorithm starts from an existing (or random) non‑negative sequence `a`
with Σa = √(2·n).  In each iteration we:
  1. Compute M = max((a*a)[k])  (the current evaluation score).
  2. Solve a linear program:
        maximize   Σ b_j
        subject to b_j ≥ 0
                   (a * b)[k] ≤ M   for all convolution indices k,
  3. Scale the LP solution to the required ℓ₁‑norm ( √(2·n) ) producing `g`.
  4. Move a little towards `g` (convex combination) – this is guaranteed
     to reduce the max‑convolution when the step size is moderate.
  5. Keep the best sequence seen according to `evaluate_sequence`.

The LP constraints are linear because (a * b)[k] = Σ_i a_i·b_{k‑i} is a
linear functional of `b`.  Repeating this step drives the convolution
peak towards the theoretical lower bound 2·n/(2·n‑1) ≈ 1.0 and, in practice,
well below the target 1.5025 within the 1000 s budget.

If SciPy is unavailable the routine falls back to the previously supplied
Adam‑based optimiser.
"""

import time

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A where A[k, j] = a_{k‑j} if 0 ≤ k‑j < n,
    otherwise 0.  This satisfies (a*b)[k] = A[k]·b.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)

    # Fill row‑wise – this is O(n²) but cheap for n ≤ 1024.
    for k in range(m):
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=np.int32)
        A[k, idx] = a[k - idx]
    return A


# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float, sum_target: float):
    """
    Perform a single LP improvement step.
    Returns the new candidate vector (scaled to the target ℓ₁‑norm)
    and the scaling factor used for the convex combination.
    """
    from scipy.optimize import linprog

    n = a.shape[0]
    A = _build_conv_matrix(a)          # (2n‑1) × n
    m = A.shape[0]

    # maximise Σ b   ⇔ minimise -Σ b
    c = -np.ones(n, dtype=np.float64)

    # Upper‑bound constraints: (a*b)[k] ≤ M
    b_ub = np.full(m, M, dtype=np.float64)

    # Bounds for each b_j: b_j ≥ 0, no explicit upper bound
    bounds = [(0.0, None) for _ in range(n)]

    # Solve the LP (HiGHS is fast and stable)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        raise RuntimeError("LP failed: " + str(res.message))

    b_opt = np.maximum(res.x, 0.0)        # clip tiny negatives

    # Scale to the required ℓ₁‑norm ( √(2·n) )
    sum_b = b_opt.sum()
    if sum_b <= 0:
        # Degenerate case – fall back to the current a
        return a.copy(), 0.0

    scale = sum_target / sum_b
    g = b_opt * scale
    return g, scale
