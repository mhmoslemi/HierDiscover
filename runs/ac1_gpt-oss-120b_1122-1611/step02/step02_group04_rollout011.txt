"""
Hybrid linear‑programming + stochastic refinement for the
self‑convolution minimisation problem.

The core idea follows the method described in the statement:
given a current sequence *f* (non‑negative, sum fixed) we compute
M = max(f*f).  The linear programme

    maximise   Σ g_i
    subject to g_i ≥ 0,
               (f * g)_k ≤ M   for all k = 0,…,2n‑2

has a feasible solution *f* itself, but often a strictly larger
ℓ₁‑norm.  By scaling this optimiser back to the original ℓ₁‑norm
and mixing it with *f* the maximal convolution value is reduced.
The routine repeatedly solves this LP (using SciPy’s ``highs`` solver)
and accepts the convex combination when it improves the true
objective

        score = 2·n·max(f*f) / (Σ f_i)² .

If SciPy is unavailable a fallback Adam‑based projected gradient
descent (the algorithm from the previous run) is used.

The search runs until the time budget expires (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(seq: np.ndarray) -> float:
    """Fast computation of the evaluation score (identical to evaluate_sequence)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_c = conv.max()
    s = seq.sum()
    if s < 0.01:   # safeguard – matches the wrapper’s behaviour
        return float('inf')
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑score step‑function height sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    **kwargs
        Ignored (kept for compatibility).

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # overall deadline (tiny safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # choose a length; prefer the known best if present, otherwise 1024
    try:
        known = np.asarray(height_sequence_1, dtype=np.float64)
        if known.ndim == 1 and known.size > 0:
            n = known.size
            a = known.copy()
        else:
            raise ValueError
    except Exception:
        n = 1024
        a = rng.random(n)

    # make the initial vector feasible (non‑negative, sum to 1)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = _project_simplex(a)

    best = a.copy()
    best_score = _score(best)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # try to use SciPy's LP solver; otherwise fall back to Adam search
    try:
        from scipy.optimize import linprog
        SCIPY_AVAILABLE = True
    except Exception:   # pragma: no cover
        SCIPY_AVAILABLE = False

    # ------------------------------------------------------------------
    # parameters used by both methods
    iteration = 0
    lower_bound = 2.0 * n / (2 * n - 1)   # theoretical lower bound

    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        # ---- LP‑based refinement ------------------------------------------------
        t = 0.2            # convex‑combination weight (1‑t)*a + t*g
        while time.monotonic() < deadline:
            iteration += 1

            # current convolution bound
            conv_f = np.convolve(a, a)
            M = conv_f.max()

            # build the (2n‑1) × n constraint matrix A where
            #   (A @ g)[k] = Σ_i a_i g_{k-i}   (zero‑padding outside)
            A = np.zeros((2 * n - 1, n), dtype=np.float64)
            for i in range(n):
                A[i:i + n, i] = a  # fill column i with a shifted downwards

            b = np.full(2 * n - 1, M, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)       # maximise Σ g_i  ⇔ minimise -Σ g_i

            # give the LP solver a tiny time budget per call so we respect the global deadline
            lp_time = max(0.0, deadline - time.monotonic())
            # (highs) tolerates a `time_limit` argument via the `options` dict
            res = linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': lp_time, 'presolve': True})

            if res.status != 0:          # non‑optimal (unbounded, infeasible, timeout,…)
                # In practice this rarely happens; we simply stop the LP phase.
                print("[lp] solver failed or timed‑out, status:", res.status)
                break

            g = res.x
            sum_g = g.sum()
            sum_a = a.sum()
            if sum_g == 0.0:
                break

            # scale g back to the same ℓ₁‑norm as a (sum_g ≥ sum_a)
            scale = sum_a / sum_g
            g_scaled = g * scale

            # convex combination with the current iterate
            a_new = (1.0 - t) * a + t * g_scaled
            a_new = np.maximum(a_new, 0.0)

            # renormalise to the original ℓ₁‑norm (numerical drift)
            a_new = a_new / a_new.sum() * sum_a

            # occasionally jitter to escape plateaus
            if rng.random() < 0.008:
                noise = rng.normal(0.0, 0.008, size=n)
                a_new = a_new + noise
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum() * sum_a

            cur_score = _score(a_new)

            # keep the best ever seen
            if np.isfinite(cur_score) and cur_score < best_score:
                best_score = cur_score
                best = a_new.copy()
                print(f"[improve] iter={iteration:,} score={best_score:.6f}")

            # accept the new point only if it does not increase the objective;
            # otherwise retain the old a (the LP is guaranteed not to worsen)
            if cur_score <= _score(a) * (1.0 + 1e-12):
                a = a_new

            # early exit if we are already close to the lower bound
            if best_score <= lower_bound * 1.01:
                print("[early stop] near theoretical bound")
                break

    else:
        # ---- Adam‑based projected gradient descent (fallback) ---------------
        # Hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr_base = 0.05
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        tau_factor = 0.5
        tau_decay = 0.9999

        while time.monotonic() < deadline:
            iteration += 1

            conv = np.convolve(a, a)
            current_max = conv.max()
            tau = max(current_max * tau_factor, 1e-12)

            shifted = (conv - current_max) / tau
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / w_sum

            # gradient of the smooth‑max surrogate
            grad = 2.0 * np.correlate(w, a, mode='full')[n - 1:n - 1 + n]

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            step = lr_base / (np.sqrt(v_hat) + eps)
            a_candidate = a - step * m_hat
            a_candidate = _project_simplex(a_candidate)

            if rng.random() < 0.005:
                noise = rng.normal(0.0, 0.015, size=n)
                a_candidate = a_candidate + noise
                a_candidate = np.maximum(a_candidate, 0.0)
                a_candidate = _project_simplex(a_candidate)

            score = float(evaluate_sequence(a_candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a_candidate.copy()
                print(f"[improve] iter={iteration:,} score={best_score:.6f}")

            a = a_candidate
            tau_factor *= tau_decay
            if iteration % 5000 == 0:
                tau_factor = max(tau_factor * 0.99, 1e-6)

            # early stop based on bound
            lower = 2.0 * n / (2 * n - 1)
            if best_score <= lower * 1.01:
                print("[early stop] close to bound")
                break

    # ------------------------------------------------------------------
    # final sanity check using the secure wrapper (ensures the same score)
    final_seq = [float(x) for x in best]
    final_score = float(evaluate_sequence(final_seq))
    print(f"[final] best score={final_score:.6f} after {iteration:,} iterations")
    return final_seq
