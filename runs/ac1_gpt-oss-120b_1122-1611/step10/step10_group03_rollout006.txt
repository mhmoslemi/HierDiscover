def _softmax_gradient(a, L, smoothing):
    # a: shape (n,)
    # compute conv = a*a via FFT length Lc maybe same as L
    A = np.fft.rfft(a, n=L)

"""
Propose a step‑function (non‑negative sequence) that minimises
    2 * n * max(a*a) / (sum(a)**2)

The optimiser performs many short random‑restart runs.  Each run starts
from a random sparse probability vector and then refines it by
projected Adam on a smooth approximation of the ∞‑norm:
    f_ε(a) = ε·log(∑_k exp((a*a)_k/ε))
The gradient of f_ε is computed analytically with FFT‑based
correlation, which makes each iteration O(n log n).  The smoothing
parameter ε is gradually reduced, turning the smooth objective into
the true max‑convolution.  After the Adam phase a few stochastic
mass‑transfer moves are applied for extra fine‑tuning.  The best
sequence seen over all runs is returned.

The code respects the 1 000 s time budget, works on the probability
simplex (sum = 1) and uses only NumPy, math and time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be at top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    # shift by the largest value to avoid numerical problems
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Re‑normalise (projector already guarantees sum==1, but guard against round‑off)
    s = w.sum()
    if s <= 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum(a)=1)."""
    n = a.size
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _softmax_gradient(a: np.ndarray, fft_len: int, smoothing: float):
    """
    Returns:
        max_conv      – the true maximum of the convolution a*a,
        grad          – gradient of the smoothed objective w.r.t. a,
        conv          – full convolution array (length 2*n-1).
    """
    n = a.size
    # FFT of a (shared for convolution and correlation)
    A = np.fft.rfft(a, n=fft_len)

    # ------ linear convolution a*a ------------------------------------
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()

    # ------ soft‑max weights (smooth max) ----------------------------
    # ε = smoothing * max_conv (never zero)
    eps = max(smoothing * max_conv, 1e-14)
    # shift by max for numerical stability
    shifted = conv - max_conv
    w = np.exp(shifted / eps)
    w_sum = w.sum()
    if w_sum == 0.0:          # should never happen, but guard
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum            # soft‑max probabilities, sum to 1

    # ------ gradient = 2 * (a ⋆ w) (cross‑correlation) -------------
    # cross‑correlation can be obtained as conj(FFT(a)) * FFT(w)
    W = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(np.conj(A) * W, n=fft_len)
    grad = 2.0 * corr_full[:n]       # only the first n entries are needed
    return max_conv, grad, conv


def _adam_projected_descent(a0: np.ndarray,
                            time_budget: float,
                            rng: np.random.Generator,
                            *,
                            init_smoothing: float = 0.5,
                            min_smoothing: float = 1e-4,
                            smooth_decay: float = 0.95,
                            base_lr: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector using projected Adam on a smoothed max‑convolution.
    Returns the best vector seen (on the true objective) and its score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)          # enough for correlation a ⋆ w
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    it = 0
    smoothing = init_smoothing

    while time.monotonic() - start < time_budget:
        it += 1
        max_conv, grad, _ = _softmax_gradient(a, fft_len, smoothing)

        # true objective for monitoring
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # ---- Adam update -------------------------------------------------
        lr = base_lr / (math.sqrt(it) + 1.0)   # slowly decreasing
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # projection onto simplex
        a_candidate = _project_onto_simplex(a_candidate)

        # accept if it improves, otherwise Metropolis‑style acceptance
        max_conv_cand, _, _ = _softmax_gradient(a_candidate, fft_len, smoothing)
        cand_score = 2.0 * n * max_conv_cand
        if cand_score < cur_score:
            a = a_candidate
        else:
            # temperature linearly decreasing to 0
            T = max(1e-4,
                    (time_budget - (time.monotonic() - start)) / time_budget)
            accept_prob = math.exp(-(cand_score - cur_score) / T)
            if rng.random() < accept_prob:
                a = a_candidate

        # decay smoothing parameter every few iterations
        if it % 40 == 0:
            smoothing = max(smoothing * smooth_decay, min_smoothing)

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            jitter = a + rng.normal(scale=0.005, size=n)
            a = _project_onto_simplex(np.maximum(jitter, 0.0))

    # final evaluation on the plain max‑convolution
    final_score = _score_from_a(best, fft_len)
    return best, final_score


def _stochastic_local_search(a0: np.ndarray,
                             time_budget: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap mass‑transfer neighbourhood search (similar to the
    “stochastic local refinement” in the reference code).  It is run
    after the Adam phase to squeeze a little more performance.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)
    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        # pick a source index with relatively large probability
        src = int(rng.choice(np.where(a > 0)[0]))
        # pick a distinct target
        tgt = int(rng.integers(0, n))
        while tgt == src:
            tgt = int(rng.integers(0, n))

        # move a random fraction of the source mass
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.12)))  # 0.1 % – 12 %
        delta = a[src] * frac
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        cand_score = _score_from_a(cand, fft_len)
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand
        else:
            # occasional simulated‑annealing acceptance
            T = max(1e-4,
                    (time_budget - (time.monotonic() - start)) / time_budget)
            if rng.random() < math.exp(-(cand_score - best_score) / T):
                a = cand

        # small random jitter every few thousand steps
        if it % 3000 == 0:
            width = max(1, n // 200)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.02))
            jitter = a.copy()
            jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(jitter)

    return best, best_score


def _run_one_restart(n: int,
                    total_budget: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Create a random sparse start of length ``n`` and improve it.
    Returns the best sequence discovered and its true score.
    """
    # ---- initialise a sparse uniform vector -------------------------
    # target average number of representations ≈ 1 ⇒ m ≈ sqrt(2n)
    m_target = max(2, int(np.sqrt(2.0 * n) * rng.uniform(0.7, 1.3)))
    m_target = min(m_target, n)
    indices = rng.choice(n, size=m_target, replace=False)
    a0 = np.zeros(n, dtype=np.float64)
    a0[indices] = 1.0 / m_target

    # add a tiny random perturbation before optimisation
    a0 += rng.normal(scale=0.002, size=n)
    a0 = _project_onto_simplex(np.maximum(a0, 0.0))

    # ---- split the allocated budget ---------------------------------
    # 70 % for Adam‑descent, 30 % for cheap stochastic polishing
    adam_budget = 0.70 * total_budget
    local_budget = total_budget - adam_budget

    # ---- Adam phase --------------------------------------------------
    a_mid, _ = _adam_projected_descent(a0, adam_budget, rng)

    # ---- Local stochastic refinement ---------------------------------
    a_best, best_score = _stochastic_local_search(a_mid, local_budget, rng)

    return a_best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The search consists of many short random
    restarts, each refining a sparse random start with projected Adam
    on a soft‑max approximation of the ∞‑norm and a final stochastic
    mass‑transfer polish.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Global best – initialise from the previous run if available,
    # otherwise from a uniform vector of moderate size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        if best.sum() > 0.0:
            best = best / best.sum()
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        n0 = 4096
        best = np.full(n0, 1.0 / n0, dtype=np.float64)

    best_score = _score_from_a(best, _next_pow2(3 * best.size))
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # ------------------------------------------------------------------
    # Lengths we will explore – powers of two up to 16384
    # ------------------------------------------------------------------
    length_pool = [1024, 2048, 3072, 4096, 6144, 8192, 12288, 16384]

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    restart_id = 0
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        # leave a little safety buffer for the final evaluation
        if remaining < 5.0:
            break

        # ----------------------------------------------------------------
        # Choose a length for the next restart (biased towards current best)
        # ----------------------------------------------------------------
        if rng.random() < 0.55:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # ensure we have enough time for a meaningful run
        max_run = min(40.0, remaining * 0.7)          # at most 40 s per restart
        if max_run < 5.0:
            break

        restart_id += 1
        print(f"[{time.monotonic() - start_time:6.2f}s] restart #{restart_id} – n = {n}")

        # ----------------------------------------------------------------
        # Run a single optimisation episode
        # ----------------------------------------------------------------
        a_opt, score_opt = _run_one_restart(n, max_run, rng)

        # ----------------------------------------------------------------
        # Update global best if improvement was achieved
        # ----------------------------------------------------------------
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # Final processing – clip to evaluator limits and compute the exact
    # score with the provided ``evaluate_sequence`` function.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)        # evaluator caps values anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # make the sequence available for a possible next run
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
