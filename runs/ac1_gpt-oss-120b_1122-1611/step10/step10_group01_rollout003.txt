"""
Propose a non‑negative step‑function (list of heights) that minimizes the
``evaluate_sequence`` objective.  The method consists of three parts:

1. **Smart seeding** – generate a handful of parametric families (uniform,
   exponential, power‑law, random Dirichlet) together with the previously
   discovered construction (if available).  The best candidate according to
   the exact objective is taken as the start point.

2. **Projected sub‑gradient descent on the ℓ∞‑convolution** – repeatedly
   identify the lag(s) attaining the current maximum of the linear
   convolution, compute the exact sub‑gradient of that maximum and take a
   step in the opposite direction while staying on the probability simplex
   (projection onto the simplex).  The step size follows a diminishing‑step
   schedule (ηₜ = η₀/√t).  A small random multiplicative jitter is applied
   periodically to escape plateaus.

3. **Guided stochastic refinement** – repeatedly move a tiny fraction of mass
   from a source index that contributes most to the current maximal lag to a
   target index that contributes the least.  Moves are accepted only if they
   do not increase the exact objective (with occasional acceptance of equal
   scores).  This phase continues until the time budget is exhausted.

All operations are performed with FFT‑based linear convolution (O(n log n)).
The routine respects the 1000 s budget, prints progress, and finally returns
the best sequence found (clipped to the evaluator’s limits).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level only, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise (numerical safety)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv_full[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a∗a)[k]`` for the set ``active`` of
    indices attaining (or approximating) the current maximum.
    For each active lag k, the contribution to ∂max/∂a_i is a_{k-i}
    counted twice (convolution symmetry).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)           # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value for a simplex‑normalised vector (sum = 1)."""
    conv = _conv_fft(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Determine problem size and initial candidates
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        n = a0.size
        if n == 0:
            n = 4096
    else:
        a0 = None
        n = 4096

    start_candidates = []

    # uniform (fallback)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    start_candidates.append(uniform)

    # previous construction (if any)
    if a0 is not None and a0.size == n:
        start_candidates.append(a0 / a0.sum())

    # exponential families (left‑skewed)
    for beta in [0.0, 1.0 / n, 3.0 / n, 5.0 / n, 10.0 / n]:
        exp_vec = np.exp(-beta * np.arange(n, dtype=np.float64))
        exp_vec /= exp_vec.sum()
        start_candidates.append(exp_vec)

    # power‑law families a_i ∝ (i+1)^{-α}
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 2.0]:
        p_vec = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
        p_vec /= p_vec.sum()
        start_candidates.append(p_vec)

    # few sparse Dirichlet draws (concentration < 1)
    for _ in range(6):
        conc = rng.uniform(0.1, 0.7)
        dir_vec = rng.dirichlet(np.full(n, conc))
        start_candidates.append(dir_vec)

    # --------------------------------------------------------------
    # 1) Timing set‑up
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # leave a small safety margin for the final call to evaluate_sequence
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Pick the best initial candidate (exact score)
    # --------------------------------------------------------------
    fft_len = _next_pow2(2 * n)          # enough for a*a linear convolution
    best_seq = None
    best_score = np.inf
    for cand in start_candidates:
        cand = np.clip(cand, 0.0, None)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        sc = _score_from_a(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = _score_from_a(best_seq, fft_len)

    # small random jitter to break possible symmetries
    jitter_noise = np.exp(rng.normal(0.0, 0.018, size=n))
    a = _project_onto_simplex(best_seq * jitter_noise)

    # initial evaluation
    conv = _conv_fft(a, fft_len)
    cur_score = 2.0 * n * conv.max()
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()
    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:.2f}s] initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Main optimisation loop (sub‑gradient + stochastic moves)
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # decide which move to perform (≈55 % sub‑gradient, ≈45 % stochastic)
        # ------------------------------------------------------------------
        if rng.random() < 0.55:
            # ---------- sub‑gradient step ----------
            max_val = conv.max()
            idx_max = int(np.argmax(conv))

            # active set – treat exact ties and also a few neighbours
            eps = 1e-12
            active = np.where(np.abs(conv - max_val) <= eps)[0]
            if active.size < 3:
                top_k = min(5, conv.size)
                top_idx = np.argpartition(-conv, top_k)[:top_k]
                active = np.unique(np.concatenate([active, top_idx]))

            grad = _gradient_from_active(a, active)

            # direction that preserves the sum (mean‑zero)
            direction = -grad + np.mean(grad)

            # diminishing step size
            eta = 0.5 / np.sqrt(iteration)

            a_new = a + eta * direction
            a_new = _project_onto_simplex(a_new)

            conv_new = _conv_fft(a_new, fft_len)
            new_score = 2.0 * n * conv_new.max()

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] iter={iteration} sub‑grad new best = {best_score:.6f}")

            # accept monotone moves (helps stability)
            if new_score <= cur_score + 1e-12:
                a = a_new
                conv = conv_new
                cur_score = new_score

        else:
            # ---------- stochastic tiny transfer ----------
            # pick source from indices that drive the current max lag
            idx_max = int(np.argmax(conv))
            grad_max = _gradient_from_active(a, np.array([idx_max]))
            src = int(np.argmax(grad_max))

            # pick a target among the “quiet” indices (low gradient)
            low_thresh = np.percentile(grad_max, 15)
            low_candidates = np.where(grad_max <= low_thresh)[0]
            if low_candidates.size:
                tgt = int(rng.choice(low_candidates))
            else:
                tgt = int(rng.integers(0, n))

            # ensure a valid move
            if src == tgt or a[src] <= 0.0:
                src = int(rng.integers(0, n))
                tgt = int(rng.integers(0, n))
                while tgt == src:
                    tgt = int(rng.integers(0, n))
                if a[src] <= 0.0:
                    continue

            # log‑uniform fraction between 0.1 % and 5 %
            frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
            delta = a[src] * frac
            a_new = a.copy()
            a_new[src] -= delta
            a_new[tgt] += delta   # sum unchanged, non‑negative by construction

            conv_new = _conv_fft(a_new, fft_len)
            new_score = 2.0 * n * conv_new.max()

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] iter={iteration} random move new best = {best_score:.6f}")

            if new_score <= cur_score + 1e-12:
                a = a_new
                conv = conv_new
                cur_score = new_score

        # ------------------------------------------------------------------
        # occasional broader jitter to escape plateaus
        # ------------------------------------------------------------------
        if iteration % 2000 == 0:
            width = max(2, n // 80)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            jitter = a.copy()
            jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(jitter)
            conv = _conv_fft(a, fft_len)
            cur_score = 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # 4) Final evaluation and return
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # respect evaluator caps
    final_score = evaluate_sequence(best_seq.tolist())
    total_time = time.monotonic() - start_time
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    return [float(v) for v in best_seq]
