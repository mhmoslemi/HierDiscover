def _smooth_max_gradient(a, fft_len, beta):
    conv = _conv_fft(a, fft_len)
    max_conv = conv.max()
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: a.size + w.size - 1]
    term2 = conv_aw[:a.size]
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: a.size + w_rev.size - 1]
    m_len = w.size
    term1 = conv_aw_rev[m_len - 1 - np.arange(a.size)]
    grad = term1 + term2

def _smooth_max_step_adam(a, fft_len, beta, lr, m, v, t, beta1=0.9, beta2=0.999, eps=1e-8):
    grad = _smooth_max_gradient(a, fft_len, beta)

"""Improved multi‑scale optimizer for the max‑convolution objective.

The routine runs several short “exploration” optimisations on a modest
length (2048) with random starts, keeps the best pattern, interpolates it
to a larger length (8192) and then performs a longer three‑phase
optimisation (smooth‑max with Adam, hard‑max sub‑gradient, stochastic
local refinement).  The smoother phase uses a soft‑max (log‑sum‑exp)
approximation of the max, whose gradient is computed by FFT.  Random
jitter, source‑target mass transfers and occasional index swaps help the
search escape plateaus.  The whole search respects the 1000 s budget and
returns the best sequence found.
"""

# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max_k (a*a)[k] for active indices."""
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i0 = max(0, k - (n - 1))
        i1 = min(n, k + 1)          # exclusive
        i = np.arange(i0, i1, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) gradient – used by Adam
# ----------------------------------------------------------------------


def _smooth_max_gradient(a: np.ndarray, fft_len: int, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation
        g(a) = log‑sum‑exp(beta * (a*a - max(a*a)) )
    with respect to a (kept on the simplex).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    max_conv = conv.max()
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                               # now ∑ w = 1

    # ----- term2 = (a * w)[0:n] -----
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: n + len(w) - 1]          # full linear conv
    term2 = conv_aw[:n]

    # ----- term1 = (a * w_rev)[m‑1‑k] -----
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: n + len(w_rev) - 1]
    m = len(w)
    term1 = conv_aw_rev[m - 1 - np.arange(n)]

    return term1 + term2


def _smooth_max_step_adam(a: np.ndarray,
                          fft_len: int,
                          beta: float,
                          lr: float,
                          m: np.ndarray,
                          v: np.ndarray,
                          t: int,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam step on the smoothed‑max objective.
    Returns updated a, first‑moment vector m and second‑moment vector v.
    """
    grad = _smooth_max_gradient(a, fft_len, beta)

    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** (t + 1))
    v_hat = v / (1.0 - beta2 ** (t + 1))

    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = _project_onto_simplex(a_new)
    return a_new, m, v


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient step
# ----------------------------------------------------------------------


def _hard_max_step(a: np.ndarray,
                   fft_len: int,
                   step_factor: float,
                   iteration: int) -> np.ndarray:
    """
    Projected additive sub‑gradient step for the true max objective.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    step = step_factor / np.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


# ----------------------------------------------------------------------
# Stochastic local refinement (mass transfer + occasional swap)
# ----------------------------------------------------------------------


def _local_refinement_step(a: np.ndarray,
                           fft_len: int,
                           rng: np.random.Generator,
                           src: int,
                           tgt: int) -> tuple[np.ndarray, int, int]:
    """
    Tiny directed mass‑transfer move.
    Returns (new_sequence, new_src, new_tgt).
    """
    n = a.size
    if a[src] <= 0.0:
        src = int(rng.integers(0, n))
    if src == tgt:
        tgt = int(rng.integers(0, n))

    # transfer fraction in [0.1 % , 5 %] (log‑uniform)
    frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    return a_new, src, tgt


def _swap_step(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exchange the values at two random indices (keeps the sum unchanged)."""
    n = a.size
    i, j = rng.choice(n, size=2, replace=False)
    a_swapped = a.copy()
    a_swapped[i], a_swapped[j] = a[j], a[i]
    return a_swapped


# ----------------------------------------------------------------------
# Core optimiser for a fixed length
# ----------------------------------------------------------------------


def _run_optimizer(n: int,
                   init_a: np.ndarray,
                   time_limit: float,
                   rng: np.random.Generator,
                   name: str = "run") -> tuple[np.ndarray, float]:
    """
    Three‑phase optimiser for a fixed length ``n``.

    Returns the best probability vector (sum = 1) and its score
    ``2 * n * max(conv)`` discovered within ``time_limit`` seconds.
    """
    start = time.monotonic()
    deadline = start + time_limit

    fft_len = _next_pow2(3 * n)

    a = init_a.copy()
    # ----- bookkeeping -----
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()
    best = a.copy()
    print(f"[{0.0:6.2f}s] {name} initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam descent
    # ------------------------------------------------------------------
    phase1_deadline = start + 0.5 * (deadline - start)
    it = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta_start, beta_end = 15.0, 800.0   # schedule for smoothing
    while time.monotonic() < phase1_deadline:
        elapsed = time.monotonic() - start
        frac = (elapsed) / (phase1_deadline - start)
        beta = beta_start + frac * (beta_end - beta_start)

        lr = 0.25 / np.sqrt(it + 1.0)   # base learning rate
        a, m, v = _smooth_max_step_adam(a, fft_len, beta, lr, m, v, it)

        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score, best = score, a.copy()
            print(f"[{elapsed:6.2f}s] {name} new best (smooth) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached in smooth phase")
                break
        it += 1
        if it % 250 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] {name} smooth‑phase it={it} best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max sub‑gradient descent
    # ------------------------------------------------------------------
    a = best.copy()
    phase2_deadline = start + 0.8 * (deadline - start)
    it = 0
    while time.monotonic() < phase2_deadline:
        cand = _hard_max_step(a, fft_len, step_factor=0.25, iteration=it)
        conv_cand = _conv_fft(cand, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score - 1e-12:
            best_score, best = score_cand, cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] {name} new best (hard) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached in hard‑max phase")
                break

        # accept non‑worsening move
        conv_cur = _conv_fft(a, fft_len)
        if score_cand <= 2.0 * n * conv_cur.max():
            a = cand
        it += 1
        if it % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] {name} hard‑max it={it} best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic refinement (mass‑transfer + swaps + jitter)
    # ------------------------------------------------------------------
    a = best.copy()
    it = 0
    src = int(rng.integers(0, n))
    tgt = int(rng.integers(0, n))
    while time.monotonic() < deadline:
        # occasional full‑swap move
        if it % 5000 == 0 and it > 0:
            cand = _swap_step(a, rng)
        else:
            cand, src, tgt = _local_refinement_step(a, fft_len, rng, src, tgt)

        conv_cand = _conv_fft(cand, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score - 1e-12:
            best_score, best = score_cand, cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] {name} new best (local) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached in local phase")
                break

        # keep improving candidate
        if score_cand < 2.0 * n * conv.max():
            a = cand

        it += 1
        if it % 2000 == 0:
            # jitter a random window
            width = max(1, n // 200)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.02))
            a_jitter = a.copy()
            a_jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jitter)

        if it % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] {name} local it={it} best={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Top‑level entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    The routine performs several quick optimisations on a modest length,
    lifts the best pattern to a larger length and finishes with a long
    three‑phase optimisation.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # leave a 5‑second safety margin for the grader
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 0) small‑scale exploration (length 2048)
    # ------------------------------------------------------------------
    small_n = 2048
    small_runs = 4
    small_time_per_run = 30.0                         # seconds
    best_small = None
    best_small_score = np.inf

    for run_idx in range(small_runs):
        now = time.monotonic()
        if now + small_time_per_run + 30.0 > deadline:
            break   # keep at least 30 s for the final large run
        # random start on the simplex
        init = rng.exponential(scale=1.0, size=small_n)
        init /= init.sum()
        best_a, score = _run_optimizer(
            n=small_n,
            init_a=init,
            time_limit=small_time_per_run,
            rng=rng,
            name=f"small{run_idx}"
        )
        if score < best_small_score:
            best_small_score, best_small = score, best_a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] "
                  f"new global best from small run {run_idx}: {score:.6f}")

    # If we never ran a small optimisation (unlikely) fall back to uniform
    if best_small is None:
        best_small = np.full(small_n, 1.0 / small_n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) lift to a larger size (8192) via linear interpolation
    # ------------------------------------------------------------------
    large_n = 8192
    x_small = np.arange(small_n)
    x_large = np.linspace(0, small_n - 1, large_n)
    lifted = np.interp(x_large, x_small, best_small)
    lifted = _project_onto_simplex(lifted)

    # ------------------------------------------------------------------
    # 2) long optimisation on the large problem
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:   # ensure we have a tiny margin
        best_large, _ = _run_optimizer(
            n=large_n,
            init_a=lifted,
            time_limit=remaining,
            rng=rng,
            name="large"
        )
    else:
        best_large = lifted

    # ------------------------------------------------------------------
    # 3) final evaluation and return
    # ------------------------------------------------------------------
    best_large = np.clip(best_large, 0.0, 1000.0)
    final_score = evaluate_sequence(best_large.tolist())
    total = time.monotonic() - start_time
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")

    # expose the best sequence for possible reuse in later calls
    globals()["height_sequence_1"] = best_large.tolist()
    return [float(x) for x in best_large]
