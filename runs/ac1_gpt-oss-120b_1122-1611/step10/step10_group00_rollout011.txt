"""
Hybrid LP‑guided step‑function optimizer.

* Work on the probability simplex (∑a = 1, a_i ≥ 0).  The evaluation
  objective (after the scale‑invariant reduction) is
      J(a) = 2·n·max(conv(a,a)).
* Initialise from a portfolio of diverse candidates (uniform,
  geometric, linear ramps, Gaussian bells, Dirichlet draws, the previous
  best construction if available) and keep the best.
* Main loop (until the time budget expires):
    – Compute the full auto‑convolution with a real FFT.
    – Let M = max(conv).  Build the sparse convolution matrix
      A such that (A @ b) = a * b for any non‑negative b.
    – Solve a linear programme
          maximise   ∑ b_j
          subject to A b ≤ M·1 ,   b ≥ 0 .
      The LP tries to pack as much mass as possible while keeping the
      convolution under the current peak M.
    – Normalise the LP solution to obtain a candidate g.
    – If J(g) < J(current) accept g, otherwise try a modest convex
      combination (t = 0.2) of current and g.
    – Keep the globally best vector seen.
    – After 60 s without improvement jitter the current solution with a
      tiny multiplicative Gaussian perturbation (helps escape plateaus).
* Finally return the best vector (clipped to the allowed range and
  re‑normalised).  The routine respects the 1000 s wall‑time limit.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that A @ b = a * b.
    Each diagonal offset d (0 ≤ d < n) contains the constant a[d].
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # data[i] = constant a[i] repeated n times (the length of each diagonal)
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = sp.diags(data, offsets, shape=(conv_len, n), format='csr')
    return A


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Choose a problem size (number of steps).
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(512, min(n0, 3000))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ----------------------------------------------------------------------
    # 2. Build a diverse portfolio of initial probability vectors.
    # ----------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps – decreasing / increasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells with a few widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – moderately and strongly concentrated
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if len(v) > n:

            extra = np.full(n - len(v), 0.0)
            v = np.concatenate([v, extra])
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # ----------------------------------------------------------------------
    # 3. Pick the best of the initial candidates.
    # ----------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        max_conv = conv.max()
        J = float(2.0 * n * max_conv)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = time.monotonic()

    # ----------------------------------------------------------------------
    # 4. Main optimisation loop (LP‑guided + jitter).
    # ----------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a small safety margin

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ------------------------------------------------------------------
        # 4.1  Evaluate current state
        # ------------------------------------------------------------------
        cur_conv = _conv_full(cur_vec, fft_len, conv_len)
        M = cur_conv.max()
        cur_J = float(2.0 * n * M)

        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.2  Stagnation handling – jitter after 60 s without improvement.
        # ------------------------------------------------------------------
        if now - last_improve > 60.0:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            # recompute after jitter
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            cur_J = float(2.0 * n * cur_conv.max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.3  Build the convolution matrix for the current vector.
        # ------------------------------------------------------------------
        A_full = _build_conv_matrix(cur_vec)           # shape (2n‑1, n)
        b_ub = np.full(conv_len, M, dtype=np.float64)  # right‑hand side

        # ------------------------------------------------------------------
        # 4.4  Solve the LP: maximise sum(b)  s.t.  A b ≤ M , b ≥ 0.
        # ------------------------------------------------------------------
        c_obj = -np.ones(n, dtype=np.float64)          # maximise sum(b)  → minimise –sum(b)
        try:
            res = linprog(c_obj,
                          A_ub=A_full,
                          b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'presolve': True})
        except Exception as exc:  # pragma: no cover – safety net
            print(f"[{elapsed:6.2f}s] LP raised exception: {exc}")
            break

            print(f"[{elapsed:6.2f}s] LP failed: {res.message}")
            noise = np.exp(rng.normal(0.0, 0.01, size=n))
            cur_vec = proj_simplex(cur_vec * noise)
            continue

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            # No useful direction found – terminate early.
            print(f"[{elapsed:6.2f}s] LP returned zero sum, ending.")
            break

        g = b_opt / sum_b                     # normalise to the simplex
        g_conv = _conv_full(g, fft_len, conv_len)
        M_g = g_conv.max()
        J_g = float(2.0 * n * M_g)

        # ------------------------------------------------------------------
        # 4.5  Acceptance step.
        # ------------------------------------------------------------------
        improved = False
        if J_g < cur_J - 1e-12:
            cur_vec = g
            cur_J = J_g
            improved = True
        else:
            # try a modest convex combination with the candidate
            t = 0.2
            blended = (1 - t) * cur_vec + t * g
            blended = proj_simplex(blended)
            blended_conv = _conv_full(blended, fft_len, conv_len)
            J_blend = float(2.0 * n * blended_conv.max())
            if J_blend < cur_J - 1e-12:
                cur_vec = blended
                cur_J = J_blend
                improved = True

        if not improved:
            # optional tiny random drift to keep the walk alive
            if iteration % 50 == 0:
                jitter = np.exp(rng.normal(0.0, 0.005, size=n))
                cur_vec = proj_simplex(cur_vec * jitter)

        iteration += 1

    # ----------------------------------------------------------------------
    # 5. Return the best solution found (after clipping & re‑normalisation).
    # ----------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
