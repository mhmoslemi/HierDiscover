def _gradient_from_active(a, active):
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]

def _smooth_max_gradient(a, p, fft_len):
    conv = _conv_fft(a, fft_len)
    eps = 1e-12
    w = np.maximum(conv, eps) ** (p - 1)
    L = fft_len
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    offset = len(w) - 1
    grad = 2.0 * corr[offset:offset + a.size]
    return grad
```

    return conv

def _gradient_from_active(a, active):
    """Exact sub‑gradient of max_k (a∗a)[k] for the given active indices."""
    n = a.size
    grad = np.zeros_like(a)
    for k in active:

        i_end = min(n, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad

def _smooth_max_gradient(a, p, fft_len):
    """Gradient of (∑ (a∗a)^p)^{1/p} (smooth approximation of max)."""
    conv = _conv_fft(a, fft_len)
    eps = 1e-12
    w = np.maximum(conv, eps) ** (p - 1)
    L = fft_len
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    offset = len(w) - 1
    grad = 2.0 * corr[offset : offset + a.size]
    return grad

        # --------------------------------------------------------------
        # Record the best for this length.
        # --------------------------------------------------------------
        if best_score < best_global_score:
            best_global_score = best_score
            best_global = best_seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] global best updated: score {best_global_score:.6f}")

    # ------------------------------------------------------------------
    # Final bookkeeping and return
    # ------------------------------------------------------------------
    if best_global is None:
        # Extremely unlikely fallback: a modest uniform vector.
        best_global = np.full(1024, 1.0 / 1024, dtype=np.float64)

    best_global = np.clip(best_global, 0.0, 1000.0)   # respect evaluator limits
    final_score = evaluate_sequence(best_global.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_global]
