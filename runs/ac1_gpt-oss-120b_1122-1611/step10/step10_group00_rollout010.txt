"""
Hybrid Adam‑based optimisation on the probability simplex.

* Works with any length `n` (default 2048, capped at 8192 for speed).
* Two‑phase approach:
    1. Quick Dirichlet sampling to obtain a good starting distribution.
    2. Adam optimiser on the log‑probabilities of the vector.
      – The objective is  J = 2·n·max(conv(a,a))  (scale‑invariant).
      – A smooth approximation of the max (soft‑max with a temperature β)
        yields a dense gradient:  g = ∂max(conv)/∂a.
      – Using the soft‑max weights we compute g via a single FFT‑based
        correlation, then obtain the gradient w.r.t. the logits
        (log‑probabilities) by the chain rule.
* The temperature β starts small (β≈5) and is linearly increased to β≈200,
  allowing early coarse updates and later precise minimisation of the true
  maximum.
* If no improvement has been seen for 30 s a random Gaussian jitter is
  added to the logits to escape stagnation.
* The best vector found during the run is returned (clipped to the allowed
  range and renormalised).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    `fft_len` must be a power of two ≥ conv_len.
    Returns the first `conv_len` entries of the linear convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`. Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size (length of the sequence)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Quick random sampling to obtain a decent start point
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_J = np.inf

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep 5 s safety margin
    # use roughly 15 % of the total time for pure sampling
    sampling_deadline = start_time + 0.15 * (deadline - start_time)

    while time.monotonic() < sampling_deadline:
        # Random concentration between very sparse and almost uniform
        alpha = rng.uniform(0.05, 2.0)
        a = rng.dirichlet(np.ones(n) * alpha)
        conv = _conv_full(a, fft_len, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_a = a.copy()

    print(f"[{time.monotonic() - start_time:.2f}s] Sampling phase best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimisation on the logits (log‑probabilities)
    # ------------------------------------------------------------------
    # Initialise logits from the best sampled distribution
    param = np.log(best_a + 1e-12)               # shape (n,)
    m = np.zeros_like(param)
    v = np.zeros_like(param)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # learning‑rate schedule (base value will be divided by sqrt(t))
    lr0 = 1.0

    # temperature (soft‑max sharpness) schedule
    beta_start = 5.0
    beta_end = 200.0

    last_improve = start_time
    best_vec = best_a.copy()
    best_J = float('inf')
    it = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- 3.1  Re‑normalise logits to obtain a probability vector ----
        max_param = np.max(param)
        exp_param = np.exp(param - max_param)          # improve numerical stability
        a = exp_param / exp_param.sum()                # sum(a) == 1

        # ---- 3.2  Evaluate current objective -------------------------
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        if cur_J < best_J - 1e-13:          # strict improvement
            best_J = cur_J
            best_vec = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- 3.3  Stagnation handling --------------------------------
        if now - last_improve > 30.0:
            # Random Gaussian jitter on logits, reset Adam moments
            param += rng.normal(scale=0.2, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stagnation – jitter applied")

        # ---- 3.4  Smooth approximation of the max (soft‑max) ----------
        beta = beta_start + (beta_end - beta_start) * (elapsed / (deadline - start_time))
        # stable soft‑max on the convolution values
        max_c = max_conv
        exps = np.exp(beta * (conv - max_c))          # all ≤ 1
        w = exps / exps.sum()                         # shape (conv_len,)

        # ---- 3.5  Gradient of the smooth max w.r.t. a -----------------
        # g = ∂max(conv)/∂a  (size n) = 2 * correlation(a, w)
        w_rev = w[::-1]                               # reversed weights
        L = _next_pow2(n + conv_len - 1)               # length for FFT‑based correlation
        fft_a = np.fft.rfft(a, L)
        fft_wrev = np.fft.rfft(w_rev, L)

        # Index where the reversed slice starts: conv_len-1 == 2*n-2
        start_idx = conv_len - 1
        g = 2.0 * corr_full[start_idx - np.arange(n)]

        # ---- 3.6  Gradient w.r.t. logits (softmax chain rule) ---------
        avg = np.dot(a, g)               # scalar ⟨a, g⟩
        grad_param = a * (g - avg)       # shape (n,)

        # ---- 3.7  Adam update ----------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_param
        v = beta2 * v + (1.0 - beta2) * (grad_param ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)           # decaying step size
        param -= lr * m_hat / (np.sqrt(v_hat) + eps)

    # ------------------------------------------------------------------
    # 4. Return the best solution found (final clipping & renormalisation)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
