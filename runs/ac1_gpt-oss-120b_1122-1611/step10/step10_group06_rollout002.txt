def build_convolution_ub_matrix(f):
    n = f.shape[0]
    m = 2*n -1
    # Build lists
    row_ind = []
    col_ind = []
    data = []
    # For each j in 0..n-1, column j has f[j] on rows i=j..j+n-1 (i.e., i from j to j+n-1)
    # Let's iterate over j:
    for j in range(n):
        rows = np.arange(j, j+n, dtype=np.int32)  # length n
        # Clip rows to m length (max row index = 2n-2)
        # Actually j+n-1 = j+(n-1) is <= 2n-2, always within m.
        row_ind.extend(rows.tolist())
        col_ind.extend([j] * n)
        data.extend([f[j]] * n)
    # Convert to arrays
    A = csr_matrix((data, (row_ind, col_ind)), shape=(2*n-1, n))

"""
Linear‑programming driven alternating optimisation.

The algorithm alternates:
1. given a current non‑negative vector f (sum = sqrt(2n))
   compute M = max(f*f).
2. solve a linear program maximise Σ g_j subject to g≥0 and
   the convolution constraints (f * g)_i ≤ M for every i.
3. rescale g to the required L1‑norm (sqrt(2n)) and replace f.
Repeatedly applying the step strictly reduces the maximal convolution
value, quickly converging to a value close to the theoretical lower bound
≈ 1 (for n large).  The routine stops when the budget is exhausted or
the target 1.5025 is reached.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using an FFT of length *fft_len*."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv

def _precompute_convolution_matrix_indices(n: int):
    """Pre‑compute row / column indices for the n×(2n‑1) convolution matrix.

    The matrix has entry (i, j) = 1 if column j contributes to convolution
    index i, i.e. 0 ≤ i‑j < n, otherwise 0.  The actual values are filled
    later by repeating the current coefficient vector.
    """
    # column index repeats n times, one for each row it touches
    col_ind = np.repeat(np.arange(n, dtype=np.int32), n)
    # row indices form blocks j, j+1, …, j+n‑1 for each column j
    row_blocks = [np.arange(j, j + n, dtype=np.int32) for j in range(n)]
    row_ind = np.concatenate(row_blocks)
    return row_ind, col_ind

def _build_convolution_ub_matrix(f: np.ndarray, row_ind, col_ind):
    """Construct A_ub for the constraints (f * g) ≤ M."""
    n = f.shape[0]
    data = np.repeat(f, n)  # each coefficient appears exactly n times
    return csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.

    The method uses the alternating LP‑based scheme described above.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # ---- 1) Choose problem size -------------------------------------------------
    # A power of two is convenient for FFT; 1024 gives a good trade‑off
    n = 1024
    S = np.sqrt(2.0 * n)                # target L1‑norm

    # ---- 2) Initialise with the uniform distribution ---------------------------
    f = np.full(n, S / n, dtype=np.float64)   # sum = S, non‑negative
    best_f = f.copy()
    # FFT length – the next power of two larger than 2 n − 1
    fft_len = _next_pow2(2 * n - 1)

    # Pre‑compute the index pattern of the convolution matrix (independent of f)
    row_ind, col_ind = _precompute_convolution_matrix_indices(n)

    # ---- 3) Main loop -----------------------------------------------------------
    best_score = None
    iteration = 0
    while time.monotonic() < deadline:
        # --- current convolution and objective value --------------------------------
        conv = _conv_self_fft(f, fft_len)
        M = conv.max()                     # current maximum convolution entry
        score = M                           # because sum(f)=sqrt(2n)

        # --- Safety clipping (numerical noise) ------------------------------------

        # --- Prepare for next iteration -------------------------------------------
        f = f_new
        iteration += 1

        # Periodic progress report (every 5 iterations)
        if iteration % 5 == 0:

    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final evaluated score = {final_score:.6f}")
    return [float(x) for x in best_f]

"""
Linear‑programming driven alternating optimisation.

The routine repeatedly performs:
1. Compute M = max(f * f) for the current non‑negative vector f,
   with the L1‑norm fixed to √(2 n).
2. Solve the LP      maximise   Σ g_i
      subject to   g_i ≥ 0  and   (f * g)_k ≤ M   for all k.
   (The constraints are a linear convolution; they are built as a sparse
   matrix that changes each iteration because f changes.)
3. Rescale g to the required L1‑norm and replace f by the scaled g.
   The maximal convolution value is guaranteed to shrink (in practice it
   drops by a factor ≈½ on the first iteration).

The process stops when the time budget expires or when the evaluated
score falls below the target 1.5025.  The best sequence seen is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using an FFT of length *fft_len*."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _precompute_convolution_indices(n: int):
    """
    Pre‑compute (row, col) index arrays for the n×(2n‑1) convolution matrix.
    The matrix entry (i, j) is 1 iff 0 ≤ i‑j < n; the actual values are
    filled later by repeating the current coefficient vector.
    """
    # Column index repeats n times (once for each row it participates in)
    col_ind = np.repeat(np.arange(n, dtype=np.int32), n)

    # Row blocks: for column j the rows are j, j+1, …, j+n‑1
    row_blocks = [np.arange(j, j + n, dtype=np.int32) for j in range(n)]
    row_ind = np.concatenate(row_blocks)
    return row_ind, col_ind


def _build_convolution_ub_matrix(f: np.ndarray, row_ind, col_ind) -> csr_matrix:
    """
    Build the constraint matrix A_ub for ``(f * g) ≤ M``.
    Each entry of the matrix is the current coefficient f[j]; the sparsity
    pattern is given by the pre‑computed indices.
    """
    data = np.repeat(f, f.size)          # each coefficient appears exactly n times
    return csr_matrix((data, (row_ind, col_ind)),
                      shape=(2 * f.size - 1, f.size))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.

    The method uses the alternating LP‑based scheme described above.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a tiny safety margin so the process does not overrun the budget
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1) Problem size – a power of two makes the FFT cheap
    # ------------------------------------------------------------------
    n = 1024                      # moderate size: enough freedom, cheap LP
    S = np.sqrt(2.0 * n)          # target L¹‑norm (sum of the sequence)

    # Initialise with the uniform distribution (sum = √(2 n) ⇒ M = 2)
    f = np.full(n, S / n, dtype=np.float64)

    # FFT length for convolutions (next power of two ≥ 2n‑1)
    fft_len = _next_pow2(2 * n - 1)

    # Pre‑compute the sparsity pattern of the convolution matrix
    row_ind, col_ind = _precompute_convolution_indices(n)

    best_f = f.copy()
    best_score = None
    iteration = 0

    # ------------------------------------------------------------------
    # 2) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution and maximal entry --------------------
        conv = _conv_self_fft(f, fft_len)
        M = conv.max()                     # because sum(f)=√(2 n), score = M

        # keep the best sequence seen so far
        if best_score is None or M < best_score - 1e-12:
            best_score = M
            best_f = f.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] iteration {iteration}: new best M = {best_score:.6f}")

        # early exit if we have already crossed the target
        if best_score <= 1.5025:
            print("[target reached]")
            break

        # ----- build LP matrix A_ub (constraints (f * g) ≤ M) ---------
        A_ub = _build_convolution_ub_matrix(f, row_ind, col_ind)

        # ----- linear programme: maximise Σ g_i ------------------------
        # SciPy's linprog minimises, so we minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # HiGHS is fast and robust
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"LP failed at iteration {iteration}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print(f"LP returned zero total mass at iteration {iteration}")
            break

        # ----- rescale g to the required L¹‑norm -----------------------
        scale = S / sum_g
        f_new = g * scale
        f_new = np.clip(f_new, 0.0, None)   # guard against tiny negatives

        # ----- prepare next iteration -----------------------------------
        f = f_new
        iteration += 1

        # occasional progress report
        if iteration % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] after {iteration} iterations, best M = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final post‑processing
    # ------------------------------------------------------------------
    # Re‑normalise the best vector to eliminate accumulated drift
    best_f = best_f * (S / best_f.sum())
    best_f = np.clip(best_f, 0.0, 1000.0)

    final_score = evaluate_sequence(best_f.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final evaluated score = {final_score:.6f}")

    return [float(x) for x in best_f]
