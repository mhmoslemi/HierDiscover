def _optimize(a0, time_budget, rng):
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_local = a.copy()
    best_score = float('inf')
    start = time.monotonic()
    iteration = 0
    gamma = 2000.0  # initial
    while time.monotonic() - start < time_budget:
        # Compute self-convolution
        a_fft = np.fft.rfft(a, n=fft_len)
        s = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        max_s = s.max()
        cur_score = 2.0 * n * max_s
        if cur_score < best_score:
            best_score = cur_score
            best_local = a.copy()
        # softmax weights
        w = _softmax_weights(s, gamma)
        # gradient
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=fft_len)
        a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        conv = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)[: 3 * n - 2]
        grad = 2.0 * conv[n - 1 : n - 1 + n]
        # Adam update
        iteration += 1
        t = iteration
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        a = a - 0.2 * m_hat / (np.sqrt(v_hat) + 1e-8)
        # simplex projection
        a = _project_onto_simplex(a)
        # increase gamma slowly
        if iteration % 1000 == 0:
            gamma = min(gamma * 1.05, 20000.0)
        # occasional jitter
        if iteration % 5000 == 0:
            width = max(1, n // 100)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.015))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)
        # occasional mass transfer
        if iteration % 3000 == 0:
            src = int(np.argmax(a))
            tgt = int(rng.integers(0, n))
            if src != tgt:

                delta = a[src] * frac
                a[src] -= delta
                a[tgt] += delta
                a = _project_onto_simplex(a)
    return best_local, best_score

"""
Proposes a step‑function (non‑negative sequence) that minimises

    2 * n * max(a * a) / (sum(a)**2)

The optimiser works on the probability simplex (sum = 1) and melds
two ideas:

* A smooth approximation of the “max” (soft‑max with a temperature that
  grows during the run) gives a differentiable objective.
* Adam updates on the simplex followed by Euclidean projection, together
  with occasional random mass‑transfer / jitter moves, explore the
  neighbourhood and avoid plateaux.

The routine performs many short optimisation runs (different random
starts, different lengths) until the time budget (≈ 1000 s) is exhausted.
The best sequence encountered is finally evaluated with the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must stay at top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _conv_full(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution using an FFT of length ``fft_len``."""
    af = np.fft.rfft(a, n=fft_len)
    bf = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(af * bf, n=fft_len)
    return conv


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a * a (full) using length ``fft_len``."""
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax_weights(s: np.ndarray, gamma: float) -> np.ndarray:
    """
    Returns a probability vector w_i = exp(gamma * (s_i - max(s))) / Σ.
    Numerical stability is ensured by subtracting max(s) first.
    """
    max_s = s.max()
    exp_vals = np.exp(gamma * (s - max_s))
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_from_softmax(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Exact gradient of the soft‑max approximated objective.
    grad_i = 2 * Σ_j w_{i+j} * a_j.
    Implemented via a single convolution (w * rev(a)).
    """
    n = a.size
    a_rev = a[::-1]
    conv = _conv_full(w, a_rev, fft_len)[: 3 * n - 2]   # length 3n‑2
    grad = 2.0 * conv[n - 1 : n - 1 + n]               # slice of length n
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value for a simplex‑normalised vector (sum = 1)."""
    n = a.size
    conv = _conv_self(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimiser for a *single* run (fixed length)
# ----------------------------------------------------------------------


def _optimise_one_run(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Optimise a single start ``a0`` for at most ``time_budget`` seconds.
    Returns the best sequence found in this run and its score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)          # enough for all convolutions we need
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # bookkeeping
    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    it = 0
    gamma = 1500.0                        # initial temperature (soft‑max sharpness)

    while time.monotonic() - start < time_budget:
        # ---------- self‑convolution ----------
        s = _conv_self(a, fft_len)       # length 2n‑1
        max_s = s.max()
        cur_score = 2.0 * n * max_s
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()

        # ---------- soft‑max weights ----------
        w = _softmax_weights(s, gamma)

        # ---------- gradient ----------
        grad = _gradient_from_softmax(a, w, fft_len)

        # ---------- Adam update ----------
        it += 1
        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.18                         # learning‑rate factor
        a = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---------- simplex projection ----------
        a = _project_onto_simplex(a)

        # ---------- temperature schedule ----------
        if it % 800 == 0:
            gamma = min(gamma * 1.07, 20000.0)   # slowly sharpen the soft‑max

        # ---------- occasional stochastic moves ----------
        if it % 3500 == 0:
            # random mass transfer between a large entry and a random target
            src = int(np.argmax(a))
            tgt = int(rng.integers(0, n))
            if src != tgt and a[src] > 0.0:
                frac = np.exp(rng.uniform(np.log(0.001), np.log(0.12)))
                delta = a[src] * frac
                a[src] -= delta
                a[tgt] += delta
                a = _project_onto_simplex(a)

        if it % 5500 == 0:
            # jitter a random contiguous block
            width = max(1, n // 120)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)

    return best, best_score
