"""
Propose a non‑negative step‑function that (approximately) flattens its
autocorrelation.  The algorithm works in three phases:

1. **Choose a length** – a short list of candidate lengths (256 … 4000)
   is tried.  For each length a few parametric families (uniform,
   Gaussian, power‑law, sparse Dirichlet) are generated and the one with
   the smallest “simplex‑score’’ 2·n·max(a∗a) is kept.

2. **Projected sub‑gradient descent** – the convex objective
   `f(a)=max_k (a∗a)_k` (with `∑a_i=1, a_i≥0`) is minimised directly.
   At each iteration we
   * compute the convolution,
   * build a mask of the active (maximal) lags,
   * obtain an exact sub‑gradient by a single FFT‑correlation,
   * take a diminishing step `η_t = η0/√t` and project back onto the
     probability simplex.
   The inexpensive FFT based operations allow tens of thousands of
   iterations within the 1000 s budget.

3. **Fine‑grained stochastic refinement** – after the gradient phase a
   cheap “mass‑transfer’’ hill‑climber (move a tiny fraction of mass
   from a high‑gradient index to a low‑gradient one) is run until time
   runs out.  Any improvement is recorded.

The best sequence seen over all lengths is returned (clipped to the
range [0,1000] required by `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """
    Exact 2·n·max(conv) for a simplex‑normalised vector (∑a=1).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _subgradient_max_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of f(a)=max_k (a∗a)_k.
    The active set is the set of lags attaining the current maximum.
    """
    n = a.size
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    maxv = conv.max()
    # mask of active lags (allow a tiny tolerance for numerical noise)
    mask = (conv >= maxv - 1e-12).astype(np.float64)

    # correlation of mask with a gives Σ_{k∈active} a_{k‑i}
    mask_fft = np.fft.rfft(mask, n=fft_len)
    a_fft = np.fft.rfft(a, n=fft_len)
    corr = np.fft.irfft(mask_fft * np.conj(a_fft), n=fft_len)
    offset = len(mask) - 1                      # =2n‑2
    grad = 2.0 * corr[offset: offset + n]       # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a 5‑second buffer for final bookkeeping
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 0) Candidate lengths to explore (shorter = faster, longer = richer)
    # ------------------------------------------------------------------
    candidate_lengths = [256, 512, 1024, 2048, 3809]  # 3809 is the length we already have
    if "height_sequence_1" in globals():
        seq1 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        seq1 = np.clip(seq1, 0.0, None)
        s = seq1.sum()
        if s > 0.0:
            seq1 = seq1 / s
        else:
            seq1 = np.full_like(seq1, 1.0 / seq1.size)
        # put its length first – we already know it works reasonably well
        candidate_lengths = [seq1.size] + [l for l in candidate_lengths if l != seq1.size]

    # Global best across all lengths
    global_best = None
    global_best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Loop over lengths
    # ------------------------------------------------------------------
    for n in candidate_lengths:
        # break if we are out of time
        if time.monotonic() >= deadline:
            break

        fft_len = _next_pow2(3 * n)          # safe for all linear convolutions

        # --------------------------------------------------------------
        # 1a) Generate a few initial candidates for this length
        # --------------------------------------------------------------
        init_candidates = []

        # uniform
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # Gaussian (symmetric around centre)
        mu = (n - 1) / 2.0
        sigma_vals = [n / 8.0, n / 6.0, n / 5.0, n / 4.0]
        for sigma in sigma_vals:
            idx = np.arange(n, dtype=np.float64)
            g = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
            if g.sum() > 0:
                g /= g.sum()
                init_candidates.append(g)

        # Power‑law
        alphas = [0.2, 0.5, 1.0, 1.5, 2.0]
        for alpha in alphas:
            pvec = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
            if pvec.sum() > 0:
                pvec /= pvec.sum()
                init_candidates.append(pvec)

        # Sparse Dirichlet draws (concentration < 1)
        for _ in range(3):
            sparse = rng.dirichlet(np.full(n, 0.3))
            init_candidates.append(sparse)

        # If we have a previous construction of the same length, add it
        if "height_sequence_1" in globals():
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq.size == n:
                seq = np.clip(seq, 0.0, None)
                if seq.sum() > 0:
                    seq = seq / seq.sum()
                else:
                    seq = np.full_like(seq, 1.0 / n)
                init_candidates.append(seq)

        # --------------------------------------------------------------
        # 1b) Pick the best initial candidate (cheap exact score)
        # --------------------------------------------------------------
        best_start = None
        best_start_score = np.inf
        for cand in init_candidates:
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0.0:
                continue
            cand = cand / cand.sum()
            sc = _score_from_a(cand, fft_len)
            if sc < best_start_score:
                best_start_score = sc
                best_start = cand.copy()
        if best_start is None:
            # should never happen; fallback to uniform
            best_start = np.full(n, 1.0 / n, dtype=np.float64)
            best_start_score = _score_from_a(best_start, fft_len)

        # --------------------------------------------------------------
        # 1c) Projected sub‑gradient descent on the simplex
        # --------------------------------------------------------------
        a = best_start.copy()
        cur_score = best_start_score
        best_local = a.copy()
        best_local_score = cur_score

        # Estimate per‑iteration cost (very rough) – we assume
        # 0.001 s × (n/2000).  Adjust later if needed.
        est_iter_time = 0.001 * (n / 2000.0)
        max_iters = int(max(1, (deadline - time.monotonic()) / est_iter_time))
        # Cap to a reasonable number to keep the loop responsive
        max_iters = min(max_iters, 30000)

        # Adam‑style parameters are not needed for pure sub‑gradient;
        # we use a simple diminishing step.
        eta0 = 0.20  # initial step size (tuned empirically)

        for it in range(1, max_iters + 1):
            if time.monotonic() >= deadline:
                break

            # sub‑gradient of the max‑convolution
            grad = _subgradient_max_conv(a, fft_len)

            # diminishing step size
            step = eta0 / np.sqrt(it)

            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            # evaluate
            score_new = _score_from_a(a_new, fft_len)

            # keep global best if improved
            if score_new < global_best_score - 1e-12:
                global_best_score = score_new
                global_best = a_new.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] length={n} iter={it} new global best = {global_best_score:.6f}")

            # keep the best for this length as well
            if score_new < best_local_score - 1e-12:
                best_local_score = score_new
                best_local = a_new.copy()

            # accept the move if it does not increase the true objective,
            # otherwise keep the old point (standard sub‑gradient rule)
            if score_new <= cur_score:
                a = a_new
                cur_score = score_new

            # occasional jitter to avoid plateaus (every 2000 steps)
            if it % 2000 == 0:
                jitter_factor = np.exp(rng.normal(0.0, 0.015))
                a = _project_onto_simplex(a * jitter_factor)

        # --------------------------------------------------------------
        # 1d) Local stochastic refinement (tiny mass transfers)
        # --------------------------------------------------------------
        a_refine = best_local.copy()
        cur_refine_score = best_local_score

        # simple heuristic: move a small fraction from the index with
        # largest sub‑gradient to one with smallest sub‑gradient.
        while time.monotonic() < deadline:
            # recompute sub‑gradient only periodically
            grad_ref = _subgradient_max_conv(a_refine, fft_len)
            src = int(np.argmax(grad_ref))
            tgt_candidates = np.where(grad_ref <= np.min(grad_ref) + 1e-12)[0]
            if tgt_candidates.size == 0:
                tgt = int(rng.integers(0, n))
            else:
                tgt = int(rng.choice(tgt_candidates))

            if src == tgt:
                tgt = int(rng.integers(0, n))

            # transfer a tiny random fraction (log‑uniform 0.1 %–3 %)
            frac = np.exp(rng.uniform(np.log(0.001), np.log(0.03)))
            delta = a_refine[src] * frac
            if delta <= 0.0:
                break

            cand = a_refine.copy()
            cand[src] -= delta
            cand[tgt] += delta

            # No need to renormalise – we kept the total mass
            cand_score = _score_from_a(cand, fft_len)

            if cand_score < global_best_score - 1e-12:
                global_best_score = cand_score
                global_best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] length={n} local move improved to {global_best_score:.6f}")

            if cand_score <= cur_refine_score:
                a_refine = cand
                cur_refine_score = cand_score
            else:
                # occasionally accept a worsening move with tiny probability
                if rng.random() < 0.001:
                    a_refine = cand
                    cur_refine_score = cand_score

        # end of this length ------------------------------------------------

    # ------------------------------------------------------------------
    # 2) Finalisation – clip to evaluator limits and compute the official score
    # ------------------------------------------------------------------
    if global_best is None:
        # should never happen – fall back to a simple uniform vector
        n_fallback = 512
        global_best = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # enforce the evaluator’s hard caps
    global_best = np.clip(global_best, 0.0, 1000.0)

    final_score = evaluate_sequence(global_best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final (official) score = {final_score:.6f}")

    # return as a plain Python list of floats
    return [float(x) for x in global_best]
