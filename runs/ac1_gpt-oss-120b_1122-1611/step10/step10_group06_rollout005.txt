"""
Propose a step‑function that (hopefully) beats the previous 1.513196 bound.

Algorithm summary
-----------------
The search works in three stages, all on the probability simplex
(· sum to 1, all entries ≥0).

1. **Adam‑mirrored gradient descent on a smoothed max objective.**
   We minimise a log‑sum‑exp surrogate of `max(conv)`.  The gradient of the
   surrogate w.r.t. the logits `z = log a` is `a * (g - <a,g>)`,
   where `g` is the usual convolution‑gradient.
   Adam’s adaptive moments give a well‑behaved step‑size schedule.

2. **Deterministic pairwise mass transfers.**
   For the current worst convolution entry we compute its exact sub‑gradient
   (the list of contributions).  Mass is moved from the index with the
   largest contribution to the index with the smallest one.
   Several trial fractions are tried (0.5 … 0.005) and the best improving
   move is applied.

3. **Simulated‑annealing style multi‑index reshuffles.**
   Randomly pick 2‑4 indices, take all mass on them and redistribute it
   according to a fresh Dirichlet sample.  Moves are accepted by a
   Metropolis rule with a linearly cooling temperature.

The three phases split the time budget roughly 40 % / 40 % / 20 %.
All intermediate vectors are projected back onto the simplex by a
simple renormalisation (the updates never create negative mass except
by rounding).  The best vector discovered is returned (clipped to the
evaluator’s 0…1000 range).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2*len(a)-1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_weighted(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Convolution of a (length n) with a weight vector w (length 2n‑1).
    Returns full linear convolution (length n + len(w) - 1 = 3n‑2).
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: a.size + w.size - 1]
    return full


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the convolution entry (a*a)[idx].
    Component i receives a[idx‑i] (or 0 when out of range).
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)          # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    return grad


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    mx = np.max(x)
    ex = np.exp(x - mx)
    return ex / ex.sum()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    `evaluate_sequence` score within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise starting vector (sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
    else:
        a0 = None

    if a0 is None or a0.size == 0:
        # default length – a power of two is FFT‑friendly
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a0.size
        a = np.clip(a0, 0.0, None)
        a = a / a.sum()                     # simplex normalisation

    fft_len = _next_pow2(3 * n)               # ≥ 3n‑2, safe for all convolutions

    # Helper to compute the exact external score (sum=1 ⇒ 2·n·max(conv))
    def _score(a_vec: np.ndarray) -> float:
        conv = _conv_self(a_vec, fft_len)
        return float(2.0 * n * conv.max())

    # Book‑keeping for the best solution
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    best_a = a.copy()
    best_score = _score(best_a)

    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam + log‑sum‑exp smoothing (≈40 % of time)
    # --------------------------------------------------------------
    phase1_end = start + 0.40 * (deadline - start)
    beta = 5.0                      # smoothing temperature
    log_a = np.log(a + 1e-16)       # logits (avoid log(0))
    m = np.zeros_like(a)           # first Adam moment
    v = np.zeros_like(a)           # second Adam moment
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.3
    it = 0

    while time.monotonic() < phase1_end:
        # ----------------------------------------------------------
        # (i) evaluate current state
        # ----------------------------------------------------------
        a = _softmax(log_a)                     # a_i ≥0 , ∑a_i = 1
        a_fft = np.fft.rfft(a, n=fft_len)

        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        max_c = conv.max()

        # ----------------------------------------------------------
        # (ii) smoothed (log‑sum‑exp) weights
        # ----------------------------------------------------------
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()           # length 2n‑1, sums to 1

        # ----------------------------------------------------------
        # (iii) gradient of the surrogate w.r.t. a
        # ----------------------------------------------------------
        w_fft = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: n + w.size - 1]
        grad = grad_full[:n]                    # length n

        # ----------------------------------------------------------
        # (iv) convert to gradient in logits (mirror descent)
        # ----------------------------------------------------------
        avg = np.dot(a, grad)                   # ⟨a,∇_a f⟩
        grad_z = a * (grad - avg)               # = ∂f/∂z

        # ----------------------------------------------------------
        # (v) Adam update on logits
        # ----------------------------------------------------------
        it += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z ** 2)

        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)

        lr = lr0 / math.sqrt(it)                # simple decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # ----------------------------------------------------------
        # (vi) evaluate score and keep the best
        # ----------------------------------------------------------
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f}")

            # early exit if we already beat the target
            if best_score <= 1.5025:
                print("[early stop] target reached in Adam phase")
                break

        # ----------------------------------------------------------
        # (vii) slowly sharpen the soft‑max
        # ----------------------------------------------------------
        beta = min(beta * 1.0015, 200.0)

        # occasional progress report
        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Adam iter={it}, beta={beta:.1f}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Deterministic pairwise mass transfer (≈40 % of time)
    # --------------------------------------------------------------
    a = best_a.copy()
    phase2_end = start + 0.80 * (deadline - start)
    it2 = 0
    # trial fractions (largest first – we try a few and keep the best)
    fractions = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    while time.monotonic() < phase2_end:
        conv = _conv_self(a, fft_len)
        max_val = conv.max()
        max_idx = int(np.argmax(conv))

        # gradient of the active (maximal) entry
        grad_max = _gradient_for_index(a, max_idx)

        src = int(np.argmax(grad_max))   # source: contributes most
        tgt = int(np.argmin(grad_max))   # target: contributes least

        # try a handful of fractions and keep the best improvement
        best_local = None
        best_local_score = best_score
        for frac in fractions:
            delta = a[src] * frac
            if delta <= 0.0 or delta > a[src]:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            # (mass moved, sum unchanged – still on simplex)
            cand = np.clip(cand, 0.0, None)   # safety

            cand_score = _score(cand)
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local = cand

        # accept the best local move if it improves the global best
        if best_local is not None:
            a = best_local
            if best_local_score < best_score - 1e-12:
                best_score = best_local_score
                best_a = best_local.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] new best (pair) = {best_score:.6f} (it={it2})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in pairwise phase")
                    break
        else:
            # no improving fraction – make a tiny random jitter to escape
            jitter = rng.normal(scale=0.001, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        it2 += 1
        if it2 % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] pairwise iter={it2}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated‑annealing multi‑index reshuffles (remaining time)
    # --------------------------------------------------------------
    a = best_a.copy()
    it3 = 0
    temp_start = 0.05 * best_score          # modest initial temperature
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac_time = (elapsed) / (deadline - start)
        temperature = temp_start * (1.0 - frac_time)

        # pick 2–4 distinct indices
        k = rng.integers(2, 5)                # number of indices to reshuffle
        idxs = rng.choice(n, size=k, replace=False)

        total_mass = a[idxs].sum()
        if total_mass == 0.0:
            # nothing to reshuffle – pick another move next iteration
            continue

        # fresh Dirichlet sample (α = 1 for each)
        new_weights = rng.dirichlet(np.ones(k)) * total_mass

        cand = a.copy()
        cand[idxs] = new_weights

        cand_score = _score(cand)
        cur_score = _score(a)

        # metropolis acceptance
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[{elapsed:6.2f}s] new best (SA)   = {best_score:.6f} (it={it3})")
            if best_score <= 1.5025:
                print("[early stop] target reached in SA phase")
                break

        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / max(1e-12, temperature)):
            a = cand

        it3 += 1
        if it3 % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] SA iter={it3}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
