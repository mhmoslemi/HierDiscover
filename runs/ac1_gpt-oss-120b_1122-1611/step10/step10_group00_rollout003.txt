def refine_exponentiated(vec, fft_len, conv_len, n, steps=2000):
    v = vec.copy()
    for t in range(steps):
        conv = _conv_full(v, fft_len, conv_len)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]

        grad = np.zeros_like(v)
        for p in active:
            start = max(0, p - (n - 1))
            end = min(p, n - 1)
            idx = np.arange(start, end + 1)
            grad[idx] += 2.0 * v[p - idx]

        # step schedule
        step = 0.2 / np.sqrt(t + 1)
        factor = np.exp(-step * grad)
        v = v * factor
        # enforce symmetry
        v = (v + v[::-1]) / 2
        s = v.sum()
        if s == 0:
            v[:] = 1.0 / n
        else:
            v /= s
    return v

"""
Simulated‑annealing + random‑move optimiser for the step‑function bound.

The algorithm works on the probability simplex (∑a = 1, a_i ≥ 0) because the
objective

    J(a) = 2·n·max(conv(a,a))

is scale‑invariant.  A diverse portfolio of deterministic shapes
(triangular, cosine, Hann, quadratic, …) and many Dirichlet draws is built,
the best of them is taken as the start and the following loop runs until the
time budget expires:

* With probability *p_noise* (≈ 0.35) a multiplicative Gaussian perturbation
  is applied; otherwise a random amount of mass is moved from one index to
  another.
* After every proposal the vector is forced to be symmetric
  (a ← (a + a[::-1]) / 2) and projected back onto the simplex.
* Acceptance follows the Metropolis rule with a linear cooling schedule.
* If no improvement is seen for *stagnation_seconds* the current best
  solution is jittered to escape a local minimum.
* When the main loop ends a short exponentiated‑gradient sub‑gradient
  refinement (the update used in the reference implementation) is performed.

The function returns the best sequence found as a plain ``list[float]``.
"""

import time
import numpy as np
from typing import List


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _objective(vec: np.ndarray, fft_len: int, conv_len: int, n: int) -> float:
    """Compute J = 2·n·max(conv) for a probability vector."""
    conv = _conv_full(vec, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Make a vector symmetric w.r.t. reversal, preserving the sum."""
    v = (v + v[::-1]) / 2.0
    return v


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex (sum to 1, non‑negative)."""
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def _initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Create a diverse set of starting probability vectors."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        cand.append(_normalize(v))

    # linear ramps (decreasing / increasing)
    cand.append(_normalize(np.arange(n, 0, -1, dtype=np.float64)))
    cand.append(_normalize(np.arange(1, n + 1, dtype=np.float64)))

    # triangular (peak in the centre)
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    v = 1.0 - np.abs(x) / ((n - 1) / 2.0)
    cand.append(_normalize(v))

    # quadratic (parabolic) shape
    v = 1.0 - (x / ((n - 1) / 2.0)) ** 2
    cand.append(_normalize(v))

    # cosine window (cosine goes to zero at the borders)
    v = np.cos(np.pi * x / (n - 1))
    cand.append(_normalize(v))

    # cosine‑squared (Hann) window
    v = np.cos(np.pi * x / (n - 1)) ** 2
    cand.append(_normalize(v))

    # Hann window (sin²)
    v = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n, dtype=np.float64) / (n - 1)))
    cand.append(_normalize(v))

    # Gaussian bells of various widths
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        cand.append(_normalize(v))

    # Dirichlet draws – a broad palette of concentrations
    for alpha in [0.02, 0.05, 0.1, 0.3, 0.6, 1.0]:
        for _ in range(2):
            v = rng.dirichlet(np.ones(n) * alpha)
            v = _symmetrize(v)
            cand.append(_normalize(v))

    # previous construction (if it exists) – resized if necessary
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(height_sequence_1, dtype=np.float64)
            if old.size != n:
                # linear interpolation onto the new grid
                old_x = np.linspace(0.0, 1.0, old.size)
                new_x = np.linspace(0.0, 1.0, n)
                v = np.interp(new_x, old_x, old)
            else:
                v = old.copy()
            v = np.clip(v, 0.0, None)
            v = _symmetrize(v)
            cand.append(_normalize(v))
        except Exception:
            pass

    return cand


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    Returns the best probability vector (sum = 1) found within the given time
    budget.  The implementation combines a temperature‑controlled Metropolis
    walk with occasional random jitter and a final exponentiated‑gradient
    refinement.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Problem size (capped at 8192 for speed, never smaller than 4096)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(height_sequence_1)
        except Exception:
            n0 = 0
    else:
        n0 = 0

    n_target = 4096
    n = max(min(max(n0, n_target), 8192), n_target) if n0 > 0 else n_target
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ----------------------------------------------------------------------
    # 2. Build initial portfolio and pick the best start point
    # ----------------------------------------------------------------------
    init_cands = _initial_candidates(n, rng)

    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = _objective(cand, fft_len, conv_len, n)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # 3. Main Metropolis‑annealing loop
    # ----------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time

    # schedule parameters
    T_start = best_J * 0.30      # 30 % of the current best as initial temperature
    T_end = 1e-8
    stagnation_seconds = 30.0
    p_noise = 0.35               # probability of a multiplicative‑noise move
    jitter_sigma = 0.12          # size of stochastic restart jitter

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear temperature cooling
        frac = elapsed / (deadline - start_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # ------------------------------------------------------------------
        # 3.1  Stagnation handling
        # ------------------------------------------------------------------
        if now - last_improve > stagnation_seconds:
            # small jitter around the best solution to escape a plateau
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cur_vec = best_vec * noise
            cur_vec = _symmetrize(cur_vec)
            cur_vec = _normalize(cur_vec)
            cur_J = _objective(cur_vec, fft_len, conv_len, n)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 3.2  Propose a new candidate
        # ------------------------------------------------------------------
        if rng.random() < p_noise:
            # multiplicative Gaussian perturbation
            sigma = 0.5 * (1.0 - frac) + 0.02 * frac   # decay from 0.5 to 0.02
            factor = np.exp(rng.normal(0.0, sigma, size=n))
            proposal = cur_vec * factor
        else:
            # move a random chunk of mass from i to j
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = cur_vec[i] * 0.5                 # never move more than half of the source mass
            delta = rng.uniform(0.0, max_delta)
            proposal = cur_vec.copy()
            proposal[i] -= delta
            proposal[j] += delta

        # enforce symmetry and simplex constraints
        proposal = _symmetrize(proposal)
        proposal = _normalize(proposal)

        # ------------------------------------------------------------------
        # 3.3  Evaluate the proposal
        # ------------------------------------------------------------------
        prop_J = _objective(proposal, fft_len, conv_len, n)

        # ------------------------------------------------------------------
        # 3.4  Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ------------------------------------------------------------------
        # 3.5  Global best update
        # ------------------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ----------------------------------------------------------------------
    # 4. Final exponentiated‑gradient refinement (sub‑gradient of max(conv))
    # ----------------------------------------------------------------------
    remaining = deadline + 5.0 - time.monotonic()   # use the 5 s safety buffer
    refine_steps = int(min(5000, max(200, remaining * 1000)))  # rough step budget
    vec = best_vec.copy()
    for t in range(refine_steps):
        conv = _conv_full(vec, fft_len, conv_len)
        max_conv = conv.max()
        # active set (indices achieving the maximum up to numerical tolerance)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]

        grad = np.zeros_like(vec)
        for p in active:
            start = max(0, p - (n - 1))
            end = min(p, n - 1)
            idx = np.arange(start, end + 1)
            grad[idx] += 2.0 * vec[p - idx]

        # step size decays like 1/√t
        step = 0.2 / np.sqrt(t + 1.0)
        factor = np.exp(-step * grad)
        vec = vec * factor
        vec = _symmetrize(vec)
        vec = _normalize(vec)

    refined_J = _objective(vec, fft_len, conv_len, n)
    if refined_J < best_J:
        best_J = refined_J
        best_vec = vec
        print(f"[{time.monotonic() - start_time:6.2f}s] Refined best J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # 5. Return the best vector (clipped to the allowed ranges)
    # ----------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
