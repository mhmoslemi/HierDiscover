"""
Propose a step‑function (non‑negative vector on the simplex) that minimises
``evaluate_sequence``.  The optimiser combines three complementary ideas:

1. **Adam‑accelerated smoothed‑max descent** – we minimise a smooth
   approximation of the ∞‑norm of the self‑convolution using an
   Adam optimiser on the log‑probabilities (mirror‑descent on the simplex).
   The smoothing parameter β is increased from a modest value to a large
   one so that the algorithm first sees a “soft‑max’’ landscape and then
   progressively sharpens to the true max.

2. **Hard‑max projected sub‑gradient descent** – after the smooth phase we
   switch to the exact sub‑gradient of the max‑convolution and keep the
   iterate on the simplex with a Euclidean projection.

3. **Simulated‑annealing local refinements** – the remaining budget is spent
   on cheap random mass‑transfer moves (directed, swap and jitter) that are
   sometimes accepted even when they increase the objective (temperature
   schedule).  This helps to escape plateaux missed by the deterministic
   phases.

Multiple random starts are tried and the best intermediate solution
initialises the long‑run optimisation.  All helper functions are top‑level,
use FFT for O(n log n) convolutions and respect the 1000 s time budget.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the active set
    (indices attaining the current maximum).  For each active index k
    the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# --------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp)
# --------------------------------------------------------------

def _smooth_max_gradient(a: np.ndarray, fft_len: int, beta: float) -> np.ndarray:
    """Gradient of the β‑smoothed max (log‑sum‑exp) of a*a."""
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    max_conv = conv.max()
    # soft‑max weights w_i = exp(β (conv_i – max_conv)) / Σ
    w = np.exp(beta * (conv - max_conv))
    w /= w.sum()

    # term2 = (a * w)   (first n entries)
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: a.size + w.size - 1]
    term2 = conv_aw[: a.size]

    # term1 = (a * rev(w)) reversed appropriately
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: a.size + w_rev.size - 1]
    m = w.size
    term1 = conv_aw_rev[m - 1 - np.arange(a.size)]

    return term1 + term2   # ∂/∂a of the smoothed max


# --------------------------------------------------------------
# Hard‑max sub‑gradient descent (projected)
# --------------------------------------------------------------

def _hard_max_step(a: np.ndarray, fft_len: int, step_size: float, iteration: int) -> np.ndarray:
    """
    Projected sub‑gradient step for the true max objective.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])
    grad = _gradient_from_active(a, active)
    # sub‑gradient points *up*, we move opposite to it
    step = step_size / math.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


# --------------------------------------------------------------
# Simulated‑annealing local moves
# --------------------------------------------------------------

def _mass_transfer_move(a: np.ndarray, rng: np.random.Generator,
                       max_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from a source index to a target index.
    Returns the new vector and the (log‑probability) move size.
    """
    n = a.size
    src = int(rng.integers(0, n))
    # ensure source has some mass
    while a[src] <= 0.0:
        src = int(rng.integers(0, n))
    tgt = int(rng.integers(0, n))
    while tgt == src:
        tgt = int(rng.integers(0, n))

    # log‑uniform fraction in [1e‑3, max_frac]
    frac = math.exp(rng.uniform(math.log(1e-3), math.log(max_frac)))
    delta = a[src] * frac
    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    return a_new, delta


def _jitter_move(a: np.ndarray, rng: np.random.Generator,
                 max_width: int = 128, sigma: float = 0.02) -> np.ndarray:
    """
    Multiply a random contiguous block by a random factor ≈ 1.
    """
    n = a.size
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = math.exp(rng.normal(0.0, sigma))
    a_jit = a.copy()
    a_jit[start:start + width] *= factor
    a_jit = _project_onto_simplex(a_jit)
    return a_jit


# --------------------------------------------------------------
# Main optimisation routine
# --------------------------------------------------------------

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search uses (i) Adam‑powered smoothed‑max descent, (ii) hard‑max
    sub‑gradient descent and finally (iii) a simulated‑annealing refinement.
    It respects the 1000 s time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0.  Choose a working length (power of two) and initialise a.
    # ------------------------------------------------------------------
    n = 4096                     # comfortable size for FFT, fits in memory
    fft_len = _next_pow2(3 * n)  # enough for linear convolution

    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            # linearly interpolate / truncate to the chosen length
            x_old = np.linspace(0, 1, a0.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, a0)
        else:
            a = a0.copy()
        a = _project_onto_simplex(a)
    else:
        # try a few random starts and keep the best (cheap eval)
        starts = []
        # very sparse Dirichlet (α=0.3) – encourages few large spikes
        for _ in range(5):
            v = rng.gamma(shape=0.3, scale=1.0, size=n)
            starts.append(v)
        # moderately sparse (α=0.7)
        for _ in range(5):
            v = rng.gamma(shape=0.7, scale=1.0, size=n)
            starts.append(v)
        # uniform start
        starts.append(np.full(n, 1.0 / n))

        best_start_score = np.inf
        best_start = None
        for v in starts:
            a_tmp = _project_onto_simplex(v)
            conv = _conv_fft(a_tmp, fft_len)
            score = 2.0 * n * conv.max()          # exact objective (no clipping)
            if score < best_start_score:
                best_start_score = score
                best_start = a_tmp
        a = best_start.copy()
        print(f"[init] chose start with score ≈ {best_start_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2. Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()
    best = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Phase A – Adam‑accelerated smooth‑max descent
    # ------------------------------------------------------------------
    phaseA_end = start_time + 0.35 * (deadline - start_time)
    # Adam hyper‑parameters
    m = np.zeros_like(a)           # 1st moment
    v = np.zeros_like(a)           # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.5                      # base learning rate for the log‑variables
    it = 0
    beta_start, beta_end = 5.0, 400.0   # smoothness schedule
    while time.monotonic() < phaseA_end:
        # linearly interpolate β
        frac = (time.monotonic() - start_time) / (phaseA_end - start_time)
        beta = beta_start + frac * (beta_end - beta_start)

        # gradient of the smoothed max
        grad = _smooth_max_gradient(a, fft_len, beta)

        # Adam update on the *log* of a (mirror‑descent on simplex)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / math.sqrt(it + 1)    # slowly decay learning rate
        log_a = np.log(a + eps_adam) - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = np.exp(log_a)
        a = _project_onto_simplex(a)

        # occasional evaluation
        if it % 250 == 0:
            conv = _conv_fft(a, fft_len)
            score = 2.0 * n * conv.max()
            if score < best_score - 1e-12:
                best_score = score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (smooth‑Adam) = {best_score:.6f}")
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] smooth‑Adam iter={it}, score≈{score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 4. Phase B – exact hard‑max projected sub‑gradient descent
    # ------------------------------------------------------------------
    a = best.copy()
    phaseB_end = start_time + 0.70 * (deadline - start_time)
    it = 0
    while time.monotonic() < phaseB_end:
        a_cand = _hard_max_step(a, fft_len, step_size=0.30, iteration=it)
        conv_cand = _conv_fft(a_cand, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        # keep global best
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (hard‑max) = {best_score:.6f}")

        # accept if not worse (simple descent)
        conv_cur = _conv_fft(a, fft_len)
        if score_cand <= 2.0 * n * conv_cur.max():
            a = a_cand

        it += 1
        if it % 300 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] hard‑max iter={it}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Phase C – simulated‑annealing stochastic refinement
    # ------------------------------------------------------------------
    a = best.copy()
    it = 0
    # temperature schedule: start relatively high, end near zero
    T_start = 0.12
    T_end = 0.001
    phaseC_end = deadline
    while time.monotonic() < phaseC_end:
        # linear temperature decay
        elapsed = time.monotonic() - start_time
        T = T_start + (T_end - T_start) * (elapsed / (deadline - start_time))

        # pick a move type probabilistically
        move_type = rng.random()
        if move_type < 0.65:                     # simple mass transfer
            a_new, delta = _mass_transfer_move(a, rng, max_frac=0.05)
        elif move_type < 0.85:                   # jitter a block
            a_new = _jitter_move(a, rng, max_width=256, sigma=0.018)
        else:                                    # swap two random entries
            a_swap = a.copy()
            i, j = rng.integers(0, n, size=2)
            a_swap[i], a_swap[j] = a_swap[j], a_swap[i]
            a_new = a_swap

        conv_new = _conv_fft(a_new, fft_len)
        score_new = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if score_new < best_score - 1e-12:
            best_score = score_new
            best = a_new.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f}")
        if score_new <= 2.0 * n * _conv_fft(a, fft_len).max():
            # always accept non‑worsening moves
            a = a_new
        else:
            # accept worsening move with probability exp(-(Δ)/T)
            prob = math.exp(-(score_new - 2.0 * n * _conv_fft(a, fft_len).max()) / max(T, 1e-12))
            if rng.random() < prob:
                a = a_new

        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA iter={it}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Finalise – clip to evaluator limits and evaluate once more
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
