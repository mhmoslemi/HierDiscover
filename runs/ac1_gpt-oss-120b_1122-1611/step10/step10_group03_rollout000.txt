"""
propose_candidate
=================

A global‑optimisation routine that searches for a non‑negative step function
(a probability vector) that minimises

    2 * n * max(a * a) / (sum(a)**2)

The algorithm is based on **smooth log‑sum‑exp relaxation** of the
“maximum convolution” term.  For a fixed smoothing parameter β the function

    f_β(a) = (1/β)·log∑ₖ exp(β·(a*a)[k])

is a smooth upper bound of max(a*a).  Its gradient can be computed
exactly via a single FFT‑based convolution, allowing fast projected
gradient descent on the probability simplex.  The schedule starts with a
small β (smooth landscape) and gradually increases it up to β≈200,
thereby focusing the optimisation on the true maximum.

The optimisation is performed many times with random restarts and
different vector lengths.  Each run uses
* exponential moving‑average (momentum) for faster convergence,
* Euclidean projection onto the simplex,
* occasional jitter when progress stalls,
* a final high‑β refinement of the best solution found.

The routine respects the 1000 s wall‑time limit, prints progress and
returns the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    # standard O(n log n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    n = a.size
    return conv[: 2 * n - 1]  # keep only the linear part


def _grad_from_weights(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) term.
    Implements  ∇_a  Σ_k w_k (a*a)[k]  = 2·(w ⋆ a)[i],
    where ⋆ denotes (non‑circular) cross‑correlation.
    The correlation is computed via FFT for speed.
    """
    # correlation w ⋆ a = conv(w, a_rev)
    a_rev_fft = np.fft.rfft(a[::-1], n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)
    n = a.size
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """
    Exact objective for a simplex‑normalised vector (sum(a)=1):
    2·n·max(a*a).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation of a single start (smooth log‑sum‑exp)
# ----------------------------------------------------------------------


def _smooth_optimize_one_run(
    a0: np.ndarray,
    time_budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth upper‑bound
    (1/β)·log∑ exp(β·conv) with an increasing β.
    Returns the best point seen during the run and its exact score.
    """
    a = a0.copy()
    n = a.size
    # FFT length large enough for convolution of length (2n‑1)+(n‑1)=3n‑2
    fft_len = _next_pow2(3 * n)

    # hyper‑parameters
    beta = 0.5
    beta_target = 200.0
    beta_mul = 1.02                # multiplicative increase each iteration
    step0 = 0.6                    # base step size (will be scaled by 1/√t)
    momentum = 0.9
    vel = np.zeros_like(a)

    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    it = 0
    last_improve = 0
    start = time.monotonic()
    deadline = start + time_budget

    while time.monotonic() < deadline:
        # ----- convolution and smooth weights -----
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        maxc = conv.max()

        # log‑sum‑exp weights (numerically stable)
        # w_k = exp(β·(conv_k - maxc)) / Σ_j exp(β·(conv_j - maxc))
        exp_vals = np.exp(beta * (conv - maxc))
        sum_exp = exp_vals.sum()
        # guarding against pathological zero (should never happen)
        if sum_exp == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_vals / sum_exp

        # ----- gradient of the smooth objective -----
        grad = _grad_from_weights(w, a, fft_len)

        # ----- step with momentum and simplex projection -----
        lr = step0 / math.sqrt(it + 1.0)
        vel = momentum * vel - lr * grad
        a_new = a + vel
        a_new = _project_onto_simplex(a_new)

        # ----- β schedule -----
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # ----- occasional evaluation & best‑track -----
        if it % 5 == 0:
            cur_score = _score_from_a(a_new, fft_len)
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best = a_new.copy()
                last_improve = it

        # ----- jitter if we are stuck for a long time -----
        if it - last_improve > 5000:
            jitter = rng.normal(scale=0.008, size=n)
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = _project_onto_simplex(a_new)
            vel[:] = 0.0        # reset momentum
            last_improve = it

        a = a_new
        it += 1

    return best, best_score

            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] REFINEMENT GLOBAL BEST = {best_score:.6f} (len={best.size})"
            )

    # ------------------------------------------------------------------
    # Final processing – clip to evaluator limits and compute the exact score
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)           # evaluator caps values anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(
        f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}"
    )

    # make the sequence available for a possible next run
    globals()["height_sequence_1"] = best.tolist()
    # Return plain Python floats
    return [float(x) for x in best]
```

This implementation replaces the previous exponentiated‑gradient / sub‑gradient scheme with a
smooth log‑sum‑exp relaxation, uses FFT‑based exact gradients, momentum, adaptive
step sizes and occasional jitter.  By gradually tightening the smoothing
parameter β the optimizer first explores a benign landscape and then hones in on
the true maximum, often yielding a lower `evaluate_sequence` value within the
same 1000 s budget.
