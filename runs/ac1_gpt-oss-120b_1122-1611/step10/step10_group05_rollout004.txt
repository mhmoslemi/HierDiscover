        else: a = a / s
    else:
        n = 2048
        a = np.full(n, 1.0/n)

    n = a.size
    fft_len = _next_pow2(3 * n)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # initial conv and score
    conv = _conv_fft(a, fft_len)
    cur_score = _objective_from_conv(conv, n)
    best = a.copy()
    best_score = cur_score
    print(f"[0.00s] init score = {best_score:.6f}")

    iteration = 0
    # parameters
    eta0 = 0.4   # initial step size
    eta_min = 1e-5

    top_k = 5
    eps = 1e-12

    while time.monotonic() < deadline:
        # compute active indices top_k
        if top_k >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            active_idx = np.argpartition(-conv, top_k)[:top_k]

        # compute grad
        grad = np.zeros_like(a)
        # loop over active indices
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n, k + 1)
            # indices i_start:i_end correspond to a_i
            # For those i, a[k-i] exists
            i = np.arange(i_start, i_end)
            grad[i] += a[k - i]
        grad *= 2.0

        # step size schedule
        elapsed = time.monotonic() - start_time
        t_frac = elapsed / (deadline - start_time)
        eta = eta0 * (1.0 - t_frac) + eta_min * t_frac

        # candidate update
        cand = a - eta * grad

        cand = _project_onto_simplex(cand)

        cand_score = _objective_from_conv(cand_conv, n)

            conv = cand_conv

"""
Projected sub‑gradient descent on the ℓ∞‑norm of the auto‑convolution.
The algorithm works on the probability simplex (∑a_i = 1) so the
objective 2·n·max(conv) is exactly the value returned by
``evaluate_sequence`` (the denominator is 1).

Key ideas
---------
* Start from the best known construction (global ``height_sequence_1``) or
  a uniform distribution.
* In each iteration compute the convolution via an FFT (O(n log n)).
* Take the `top_k` largest convolution entries (they dominate the max).
  The sub‑gradient of the maximum is the sum of the gradients of these
  entries; each gradient is simply 2·a shifted by the corresponding lag.
* Take a step opposite to the sub‑gradient, then project back onto the
  simplex (Euclidean projection, Duchi et al. 2008).  The step size
  decays linearly with the elapsed time.
* Occasionally apply a small random jitter to help escape plateaus.
* Keep the best sequence seen so far and stop early if the target
  (1.5025) is reached.

The routine respects the 1000 s time budget, prints progress and finally
returns a list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the value 2 * n * max(conv); the sum of a is assumed = 1."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function will stop about 5 s
        before the limit to be safe).

    Returns
    -------
    list[float]
        The best sequence found (values are already within the evaluator’s
        allowed range).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a (normalise to the probability simplex)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # fall‑back: a modest power‑of‑two length with uniform mass
        n0 = 2048
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(3 * n)        # enough for all linear convolutions

    # --------------------------------------------------------------
    # 1) Timing
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    cur_score = _objective_from_conv(conv, n)

    best = a.copy()
    best_score = cur_score
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Hyper‑parameters
    # --------------------------------------------------------------
    top_k = 5                     # number of dominant convolution entries
    eta0 = 0.4                    # initial step size (sub‑gradient step)
    eta_min = 1e-5                # final step size
    jitter_every = 2000           # iterations between random jitter moves
    eps_active = 1e-12            # tolerance when selecting the active set

    iteration = 0

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- active indices (largest convolution entries) ----
        if top_k >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            # ``np.argpartition`` is O(n) and gives the top‑k indices
            active_idx = np.argpartition(-conv, top_k)[:top_k]

        # ---- sub‑gradient of the max‑convolution ----
        grad = np.zeros_like(a)
        for k in active_idx:
            # contributions of a shifted by lag k
            i_start = max(0, k - (n - 1))
            i_end = min(n, k + 1)          # exclusive
            i = np.arange(i_start, i_end, dtype=int)
            grad[i] += a[k - i]
        grad *= 2.0                         # ∂C_k/∂a_i = 2·a_{k-i}

        # ---- step‑size schedule (linear decay) ----
        elapsed = time.monotonic() - start_time
        t_frac = elapsed / (deadline - start_time)   # in [0,1]
        eta = eta0 * (1.0 - t_frac) + eta_min * t_frac

        # ---- projected sub‑gradient step ----
        cand = a - eta * grad
        cand = _project_onto_simplex(cand)

        # ---- evaluate candidate ----
        cand_conv = _conv_fft(cand, fft_len)
        cand_score = _objective_from_conv(cand_conv, n)

        # ---- greedy acceptance (always keep improvement) ----
        if cand_score < cur_score - 1e-15:
            a = cand
            conv = cand_conv
            cur_score = cand_score

        # ---- track the best solution ever seen ----
        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f} (iter={iteration})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

        # ---- occasional random jitter to jump out of plateaus ----
        if iteration % jitter_every == 0 and iteration > 0:
            # pick a modest fraction of coordinates and perturb them
            jitter_frac = 0.05                     # 5 % of the entries
            k_jitter = max(2, int(jitter_frac * n))
            idxs = rng.choice(n, size=k_jitter, replace=False)
            factor = np.exp(rng.normal(0.0, 0.02, size=idxs.shape))
            pert = a.copy()
            pert[idxs] *= factor
            pert = _project_onto_simplex(pert)

            pert_conv = _conv_fft(pert, fft_len)
            pert_score = _objective_from_conv(pert_conv, n)

            if pert_score < best_score - 1e-12:
                best = pert.copy()
                best_score = pert_score
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (jitter) = {best_score:.6f}")

            if pert_score < cur_score:
                a = pert
                conv = pert_conv
                cur_score = pert_score

        iteration += 1

    # --------------------------------------------------------------
    # 5) Return the best sequence (respect evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
