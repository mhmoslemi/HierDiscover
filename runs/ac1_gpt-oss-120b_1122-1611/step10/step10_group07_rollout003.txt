"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(sequence).

Algorithm
---------
1. Initialise ----------------------------------------------------
   • If a previous solution (global name ``height_sequence_1``) is
     available, use it, otherwise start from a uniform vector.
   • Normalise to make the sum equal to 1 (the objective is scaling‑invariant).

2. Fast shape exploration ----------------------------------------
   • Sample a handful of parametrised families (uniform, power‑law,
     symmetric “U‑shape’’ a_i ∝ (i+1)^‑p+(n‑i)^‑p and Dirichlet blocks)
   • Keep the best candidate according to a cheap FFT‑based estimate of
     ``score = 2·n·max(conv)``.

3. Simulated‑annealing (global+directed moves) -------------------
   • Temperature T decays linearly from ``T0`` to zero.
   • At each iteration we compute the current active convolution indices
     (those attaining the maximal value) and the exact sub‑gradient
     ∂max/∂a_i.
   • Three move types are used
        – **directed transfer** : move a fraction of the mass from the
          index with the largest gradient to the index with the smallest.
        – **random transfer**   : move a random fraction of mass between
          two random positions.
        – **jitter**            : multiply the whole vector by a small
          Gaussian factor and re‑project onto the simplex.
   • Moves are accepted if they improve the score, otherwise with
     probability exp(−Δ/T) (Metropolis rule).  The best ever vector is
     stored.

4. Adam‑style mirror descent (fine tuning) ----------------------
   • Work in the log‑domain (exponentiated‑gradient) and apply Adam’s
     adaptive moments to the sub‑gradient of the max‑convolution.
   • After each update the vector is exponentiated and normalised to the
     simplex.

5. Return the best vector found (clipped to the evaluator’s limits).

The routine respects the 1000 s budget (leaving a 5 s safety margin),
uses only NumPy and the provided ``evaluate_sequence`` function and
never creates closures or file/network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k, the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score_fft(a: np.ndarray, fft_len: int) -> float:
    """
    Fast estimate of the evaluation function:
        score = 2 * n * max(conv(a,a))
    This is exactly the value returned by ``evaluate_sequence`` before
    the clipping of numbers (which we never exceed).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: float
        Total time budget (seconds).  Five seconds are kept as safety
        margin; the algorithm stops earlier if the budget is exhausted.

    Returns
    -------
    list[float]
        The best sequence found (already clipped to the evaluator’s limits).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` and normalise to sum = 1
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
    else:
        n = 3809                     # reasonable default (close to previous run)
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(3 * n)     # safe length for linear convolution

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best = a0.copy()
    best_score = _score_fft(best, fft_len)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 0 – fast parametric shape exploration (≈10% of time)
    # --------------------------------------------------------------
    phase0_end = start_time + 0.10 * (deadline - start_time)

    # --- uniform (already covered) ---------------------------------
    # Uniform already stored as ``best`` – nothing to do.

    # --- power‑law shapes -----------------------------------------
    while time.monotonic() < phase0_end:
        p = rng.uniform(0.05, 2.5)
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = idx ** (-p)
        seq /= seq.sum()
        score = _score_fft(seq, fft_len)
        if score < best_score:
            best_score = score
            best = seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (pow‑law) = {best_score:.6f}")
        # also try a symmetric “U‑shape’’ built from two power‑laws
        seq_u = (idx ** (-p) + (n - idx + 1) ** (-p))
        seq_u /= seq_u.sum()
        score_u = _score_fft(seq_u, fft_len)
        if score_u < best_score:
            best_score = score_u
            best = seq_u.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (U‑shape) = {best_score:.6f}")

        # break early if time runs out
        if time.monotonic() >= phase0_end:
            break

    # --- Dirichlet on coarse bins ----------------------------------
    K = 256 if n >= 4096 else max(8, n // 32)
    block = max(1, n // K)

    while time.monotonic() < phase0_end:
        alpha = rng.uniform(0.3, 2.0)
        bins = rng.dirichlet(np.full(K, alpha))
        if rng.random() < 0.5:
            bins = np.sort(bins)[::-1]          # enforce monotone decreasing
        seq = np.repeat(bins, block)
        if seq.size < n:
            seq = np.concatenate([seq, bins[: n - seq.size]])
        else:
            seq = seq[:n]
        seq /= seq.sum()
        score = _score_fft(seq, fft_len)
        if score < best_score:
            best_score = score
            best = seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (Dirichlet) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – Simulated Annealing with directed moves (≈65% time)
    # --------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    T0 = 0.02 * best_score                     # initial temperature proportional to score
    temp_start = start_time
    phase1_end = start_time + 0.75 * (deadline - start_time)

    it_sa = 0
    while time.monotonic() < phase1_end:
        # elapsed fraction determines temperature linearly
        elapsed = time.monotonic() - start_time
        frac = min(1.0, (elapsed - (deadline - start_time) * 0.75) /
                   ((deadline - start_time) * 0.25 + 1e-12))
        # linear cooling
        T = max(1e-8, T0 * (1.0 - (elapsed - start_time) / (deadline - start_time)))

        # compute convolution and sub‑gradient for the current max
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-7
        active = np.where(conv >= max_b - eps)[0]
        if active.size > 100:
            active = rng.choice(active, size=100, replace=False)
        grad = _gradient_from_active(a, active)

        # ----- choose move type -----------------------------------
        move_type = rng.random()
        max_frac = min(0.2, T / 0.01)          # larger temperature → larger transfers

        if move_type < 0.60:
            # Directed mass transfer guided by the gradient
            src = int(np.argmax(grad))
            tgt = int(np.argmin(grad))
        elif move_type < 0.90:
            # Random transfer between two positions
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                src = int(rng.integers(0, n))
            else:
                src = int(rng.choice(nonzero))
            tgt = int(rng.integers(0, n))
            while tgt == src:
                tgt = int(rng.integers(0, n))
        else:
            # Jitter (multiplicative noise)
            sigma = max_frac * 0.5
            a_cand = a * (1.0 + rng.normal(0.0, sigma, size=n))
            a_cand = np.clip(a_cand, 0.0, None)
            a_cand = _project_onto_simplex(a_cand)
            cand_score = _score_fft(a_cand, fft_len)
            delta = cand_score - cur_score
            accept = (delta < 0.0) or (rng.random() < math.exp(-delta / max(T, 1e-12)))
            if accept:
                a = a_cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best = a_cand.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] new best (jitter) = {best_score:.6f}")
            it_sa += 1
            continue  # go to next iteration

        # ----- directed / random transfer --------------------------
        src_mass = a[src]
        if src_mass <= 1e-15:
            it_sa += 1
            continue
        delta = src_mass * rng.random() * max_frac
        if delta <= 1e-15:
            it_sa += 1
            continue

        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta

        cand_score = _score_fft(a_cand, fft_len)
        delta_score = cand_score - cur_score

        # Metropolis acceptance
        if delta_score < 0.0 or rng.random() < math.exp(-delta_score / max(T, 1e-12)):
            a = a_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = a_cand.copy()
                now = time.monotonic() - start_time
                print(f"[{now:6.2f}s] new best (SA) = {best_score:.6f}")

        it_sa += 1
        # occasional progress report
        if it_sa % 5000 == 0:
            now = time.monotonic() - start_time
            print(f"[{now:6.2f}s] SA iter={it_sa}, cur={cur_score:.6f}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – Adam‑style mirror descent (≈15% time left)
    # --------------------------------------------------------------
    a = best.copy()
    cur_score = best_score

    # log‑domain variables for exponentiated‑gradient style updates
    log_a = np.log(a + 1e-15)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.12
    it_adam = 0

    while time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-7
        active = np.where(conv >= max_b - eps)[0]
        if active.size > 80:
            active = rng.choice(active, size=80, replace=False)
        grad = _gradient_from_active(a, active)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it_adam + 1))
        v_hat = v / (1.0 - beta2 ** (it_adam + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_a = log_a - step
        a = np.exp(log_a)
        a = a / a.sum()                # projection onto simplex (implicit)

        cand_score = _score_fft(a, fft_len)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = a.copy()
            now = time.monotonic() - start_time
            print(f"[{now:6.2f}s] new best (Adam) = {best_score:.6f}")

        cur_score = cand_score
        it_adam += 1

        if it_adam % 3000 == 0:
            now = time.monotonic() - start_time
            print(f"[{now:6.2f}s] Adam iter={it_adam}, cur={cur_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # Final exact evaluation (slow but accurate) – useful for debugging
    try:
        final_score = evaluate_sequence(best.tolist())
        print(f"Finished – final exact score = {final_score:.6f}")
    except Exception as ex:
        print(f"Finished – exact evaluation failed: {ex}")

    return [float(x) for x in best]
