"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a))²

where conv is the linear convolution.
The algorithm

1. builds an “arcsine‑like’’ distribution (the discrete analogue of the
   continuous Beta(½,½) law).  This distribution already provides a
   substantially lower ‑max‑convolution than the uniform sequence used
   in the reference implementation.

2. refines the start point with three optimisation phases that stay on
   the simplex (∑aᵢ=1):
      – smooth‑max (log‑sum‑exp) mirror descent,
      – exact hard‑max sub‑gradient descent,
      – a simulated‑annealing style stochastic mass‑transfer step.

All phases are performed with FFT‑based convolutions (O(n log n) per
iteration) and respect the 1000 s time budget.  The routine works for
any length that is a power of two (default n=4096) and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities – must stay at module level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the active set.
    For each active index k the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Phase‑1 – smooth‑max (log‑sum‑exp) update
# ----------------------------------------------------------------------


def _smooth_max_step(a: np.ndarray,
                     fft_len: int,
                     beta: float,
                     eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step on the β‑smoothed max objective.
    Returns a new probability vector (still on the simplex).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    m = conv.size
    max_conv = conv.max()
    # soft‑max weights
    w = np.exp(beta * (conv - max_conv))
    w /= w.sum()                               # sum to 1

    # term2 = (a * w) evaluated on the first n entries
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: n + m - 1]
    term2 = conv_aw[:n]

    # term1 = (a * w_rev) reflected
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: n + m - 1]
    term1 = conv_aw_rev[m - 1 - np.arange(n)]

    grad = term1 + term2                         # ∂/∂a of smoothed max

    # exponentiated‑gradient update on log‑probabilities
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Phase‑2 – exact hard‑max sub‑gradient descent
# ----------------------------------------------------------------------


def _hard_max_step(a: np.ndarray,
                   fft_len: int,
                   step_factor: float,
                   iteration: int) -> np.ndarray:
    """Projected sub‑gradient step for the true max‑convolution objective."""
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    step = step_factor / np.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


# ----------------------------------------------------------------------
# Helper to build the arcsine‑type start distribution
# ----------------------------------------------------------------------


def _arcsine_init(n: int) -> np.ndarray:
    """Return a probability vector whose shape approximates the continuous
    Beta(½,½) (arcsine) distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    # the 0.5 offset regularises the end‑point singularities
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimiser – runs the three phases inside the time budget
# ----------------------------------------------------------------------


def _run_optimizer(a0: np.ndarray,
                   budget_s: float,
                   start_time: float,
                   rng_seed: int) -> tuple[np.ndarray, float]:
    """Return the best vector found and its (2·n·max‑conv) score."""
    rng = np.random.default_rng(rng_seed)
    n = a0.size
    fft_len = _next_pow2(3 * n)

    best = a0.copy()
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------- Phase‑1 : smooth‑max ------------------- #
    phase1_end = start_time + 0.40 * (budget_s - 5.0)   # keep 5 s safety margin
    beta_start, beta_end = 5.0, 250.0
    eta0 = 0.7
    it = 0
    while time.monotonic() < phase1_end:
        elapsed = time.monotonic() - start_time
        frac = (elapsed) / (phase1_end - start_time)
        beta = beta_start + frac * (beta_end - beta_start)
        eta = eta0 / np.sqrt(it + 1.0)

        a0 = _smooth_max_step(a0, fft_len, beta, eta)

        conv = _conv_fft(a0, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a0.copy()
            print(f"[{elapsed:6.2f}s] new best (smooth) = {best_score:.6f}")

        it += 1
        if it % 1000 == 0:
            print(f"[{elapsed:6.2f}s] smooth‑iter={it}, best={best_score:.6f}")

    # ------------------- Phase‑2 : hard‑max ------------------- #
    a = best.copy()
    phase2_end = start_time + 0.75 * (budget_s - 5.0)
    it = 0
    while time.monotonic() < phase2_end:
        a_cand = _hard_max_step(a, fft_len, step_factor=0.30,
                                iteration=it)
        conv_cand = _conv_fft(a_cand, fft_len)
        score_cand = 2.0 * n * conv_cand.max()
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (hard) = {best_score:.6f}")

        # accept if it does not increase the objective
        conv_cur = _conv_fft(a, fft_len)
        if score_cand <= 2.0 * n * conv_cur.max():
            a = a_cand

        it += 1
        if it % 500 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] hard‑iter={it}, best={best_score:.6f}")

    # ------------------- Phase‑3 : stochastic refinement -------- #
    a = best.copy()
    phase3_end = start_time + (budget_s - 5.0) - 0.001
    it = 0
    while time.monotonic() < phase3_end:
        # temperature slowly decreases from 1 to ~0
        temperature = max(0.001,
                          1.0 - (time.monotonic() - start_time) /
                                (budget_s - 5.0))

        # choose source / target indices
        src = int(rng.integers(0, n))
        tgt = int(rng.integers(0, n))
        # fractional mass to move – log‑uniform between 0.1 % and 2 %
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.02)))
        delta = a[src] * frac
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new = _project_onto_simplex(a_new)

        conv_new = _conv_fft(a_new, fft_len)
        score_new = 2.0 * n * conv_new.max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best = a_new.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (stoch) = {best_score:.6f}")

        # Metropolis acceptance
        cur_score = 2.0 * n * _conv_fft(a, fft_len).max()
        if score_new <= cur_score:
            a = a_new
        else:
            prob = np.exp(-(score_new - cur_score) / temperature)
            if rng.random() < prob:
                a = a_new

        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] stoch‑iter={it}, best={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    # Desired length – a power of two makes FFT bookkeeping cheap.
    n = 4096

    # ---------- 0)  initialise -------------------------------------------------
    # use the arcsine‑type start; fall back to the global best if it exists
    a = _arcsine_init(n)

    if "height_sequence_1" in globals():
        alt = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        alt = np.clip(alt, 0.0, None)
        if alt.sum() > 0.0:
            alt = _project_onto_simplex(alt)
        else:
            alt = np.full_like(alt, 1.0 / alt.size)
        # adjust size to n (pad with zeros or truncate)
        if alt.size != n:
            if alt.size < n:
                alt = np.concatenate([alt, np.zeros(n - alt.size)])
            else:
                alt = alt[:n]
            alt = _project_onto_simplex(alt)
        # keep the better of the two as start point
        score_a = evaluate_sequence(a.tolist())
        score_alt = evaluate_sequence(alt.tolist())
        if score_alt < score_a:
            a = alt
            print(f"Starting from global best, score = {score_alt:.6f}")
        else:
            print(f"Starting from arcsine init, score = {score_a:.6f}")

    # ----------------------------------------------------------------------
    # If the initial candidate already beats the target we can stop early.
    # ----------------------------------------------------------------------
    init_score = evaluate_sequence(a.tolist())
    print(f"Initial evaluate_sequence score = {init_score:.6f}")
    if init_score <= 1.5025:
        return [float(x) for x in a]

    # ----------------------------------------------------------------------
    # Run the three‑phase optimiser inside the remaining time budget.
    # ----------------------------------------------------------------------
    remaining_budget = deadline - time.monotonic()
    if remaining_budget > 0.0:
        best_vec, best_score = _run_optimizer(a, remaining_budget,
                                              time.monotonic(),
                                              seed + 1)
    else:
        best_vec, best_score = a, init_score

    # ----------------------------------------------------------------------
    # Final post‑processing – clip to the limits enforced by evaluate_sequence
    # ----------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    final_score = evaluate_sequence(best_vec.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_vec]
