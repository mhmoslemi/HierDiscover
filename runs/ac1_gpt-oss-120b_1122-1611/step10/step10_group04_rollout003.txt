assistantfinal**Solution Overview**

The solver searches for a non‑negative step‑function `a = (a₀,…,a_{n‑1})` that minimizes  

\[
F(a)=\frac{2n\;\max_k (a*a)_k}{(\sum_i a_i)^2}.
\]

Because the objective is homogeneous we may keep `∑a_i = 1`.  
Thus the problem reduces to *minimising the maximum entry of the
auto‑convolution* `b = a*a` under the simplex constraint.

The algorithm combines three ideas that were not used together in the
reference implementation:

1. **Multiple random restarts** – a fresh random simplex point is tried
   many times; the best solution seen over all restarts is kept.
2. **Exact‑subgradient mirror descent** – at each iteration the active
   set (indices attaining the current maximum of `b`) is computed and
   its sub‑gradient is used in a multiplicative (exponentiated‑gradient)
   update.  This respects the simplex without an explicit projection.
3. **Adaptive sparsification** – every few iterations the smallest
   coefficients are set to zero, keeping only the `keep_k` largest
   entries.  A sparser vector reduces the number of collisions in the
   convolution and therefore lowers the maximum entry.

All operations are `O(n log n)` because the convolution is performed with
the FFT.  The routine respects the 1000 s time budget, prints progress,
and finally returns the best sequence (clipped to the limits imposed by
`evaluate_sequence`).  If a previous solution is available via the global
`height_sequence_1` it is used as the first restart; otherwise a fresh
random start is used.

```python
"""
Multi‑restart exponentiated‑gradient optimiser with adaptive sparsification.

*   Starts from the previous best sequence (if present) and many random
    simplex points.
*   Uses the exact sub‑gradient of max(a*a) and a mirror‑descent (EG)
    update on the probability simplex.
*   Every `prune_interval` iterations keeps only the `keep_k` largest
    entries (the rest are set to zero and the vector is re‑normalised).
*   All convolutions are computed with an FFT of length power‑of‑two,
    giving O(n log n) per iteration.
*   The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the active set ``active``.
    For each active index k the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _smooth_max_step(a: np.ndarray,
                     fft_len: int,
                     beta: float,
                     eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step on the *smoothed* max objective.
    Returns the updated probability vector (still on the simplex).
    """
    # convolution a*a
    conv = _conv_fft(a, fft_len)               # length 2n‑1
    max_conv = conv.max()
    # soft‑max weights (log‑sum‑exp trick)
    w = np.exp(beta * (conv - max_conv))
    w /= w.sum()

    # term2 = (a * w)[k] for k = 0 … n‑1
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)[: a.size + w.size - 1]
    term2 = conv_aw[: a.size]

    # term1 = (a * rev(w))[m‑1‑k]  (m = len(w))
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: a.size + w_rev.size - 1]
    m = w.size
    term1 = conv_aw_rev[m - 1 - np.arange(a.size)]

    grad = term1 + term2                       # ∂/∂a of weighted sum

    # exponentiated‑gradient update on log‑probabilities
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


def _exp_subgrad_step(a: np.ndarray,
                      fft_len: int,
                      eta: float,
                      eps: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent step using the *exact* sub‑gradient of the hard max.
    Returns (new_a, max_conv) where ``max_conv`` is the current
    maximum entry of the convolution (needed for the objective).
    """
    conv = _conv_fft(a, fft_len)
    max_conv = conv.max()
    # indices that are (almost) active
    active = np.where(conv >= max_conv - eps * max_conv)[0]
    grad = _gradient_from_active(a, active)

    # multiplicative (exponentiated) update
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new, max_conv


def _prune_to_keep(a: np.ndarray, keep_k: int) -> np.ndarray:
    """
    Zero‑out all but the ``keep_k`` largest entries of ``a`` and renormalise.
    """
    if keep_k >= a.size:
        return a
    # ``argpartition`` gives the positions of the k largest values (unsorted)
    idx = np.argpartition(-a, keep_k - 1)[:keep_k]
    mask = np.zeros_like(a, dtype=bool)
    mask[idx] = True
    a_pruned = np.where(mask, a, 0.0)
    total = a_pruned.sum()
    if total == 0.0:
        # fallback – revert to uniform distribution
        a_pruned = np.full_like(a, 1.0 / a.size)
    else:
        a_pruned /= total
    return a_pruned


# ----------------------------------------------------------------------
#  Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    The method uses many random restarts, exponentiated‑gradient mirror
    descent on the exact max‑convolution objective and adaptive
    sparsification.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Length of the sequence (power of two is convenient for FFT)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        init_seq = np.clip(init_seq, 0.0, None)
        total = init_seq.sum()
        if total > 0.0:
            init_seq = init_seq / total
        else:
            init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
        n = init_seq.size
    else:
        # default length – a power of two simplifies the FFT bookkeeping
        n = 2048
        init_seq = None

    fft_len = _next_pow2(2 * n)          # enough for a linear convolution

    # ------------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")               # objective = 2*n*max_conv (sum=1)

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # how many restarts we try (depends on remaining time)
    min_restart_time = 8.0                  # we need at least this much per restart
    restart = 0

    # Parameters for the inner optimisation
    phase1_iters = 400          # smooth‑max warm‑up
    phase2_iters = 2500         # exponentiated exact‑subgradient phase
    prune_interval = 200        # how often we sparsify
    keep_k = max(30, int(3 * np.sqrt(n)))   # size of the kept support

    # ------------------------------------------------------------------
    # 4) Main restart loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - min_restart_time:
        restart += 1
        # ----- initialise a -------------------------------------------------
        if restart == 1 and init_seq is not None:
            a = init_seq.copy()
        else:
            a = rng.random(n)
            a /= a.sum()                 # project onto the simplex

        # ----- Phase 1 – smooth‑max warm‑up ---------------------------------
        for it in range(phase1_iters):
            # linearly increase the smoothing parameter β
            beta = 30.0 + 0.1 * it
            eta = 0.6 / np.sqrt(it + 1.0)
            a = _smooth_max_step(a, fft_len, beta, eta)

            # occasional progress report
            if (it + 1) % 100 == 0 and time.monotonic() >= deadline:
                break

        # ----- Phase 2 – exact‑subgradient mirror descent --------------------
        for it in range(phase2_iters):
            # exponentiated‑gradient step on the real (hard) max
            eta = 0.5 / np.sqrt(it + 1.0)
            a, max_conv = _exp_subgrad_step(a, fft_len, eta)

            # current objective (sum is 1, so 2*n*max_conv)
            cur_score = 2.0 * n * max_conv

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] restart {restart} new best ≈ {best_score:.6f}")

                # early exit if we already beat the target
                if best_score <= 1.5025:
                    print("[early stop] target reached")
                    break

            # sparsify every ``prune_interval`` iterations
            if (it + 1) % prune_interval == 0:
                a = _prune_to_keep(a, keep_k)

            # time‑budget check
            if time.monotonic() >= deadline:
                break

        # break outer loop early if target already reached
        if best_score <= 1.5025:
            break

    # ------------------------------------------------------------------
    # 5) Final evaluation with the official ``evaluate_sequence`` function
    # ------------------------------------------------------------------
    if best_a is None:                     # should never happen, but be safe
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced inside ``evaluate_sequence`` (0 … 1000)
    best_a = np.clip(best_a, 0.0, 1000.0)

    final_score = evaluate_sequence(best_a.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_a]
```
