def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 5.0

    # get n
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # start from that sequence
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, None)
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ensure sum=1
    a = a / a.sum()
    # we will resize to ensure length is power of two? Not needed.
    fft_len = 1 << ((3 * n - 1).bit_length())  # next power of two >= 3n

    # Adam parameters
    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_score = float('inf')
    best_a = a.copy()
    # target uniform conv value:
    avg = 1.0 / (2 * n - 1)

    while time.monotonic() < deadline:
        # compute convolution
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new best score {best_score:.6f}")

            # maybe early stop if below target
            if best_score <= 1.5025:
                break

        # compute residual
        r = conv - avg

        # compute grad: conv of r with reversed a
        a_rev = a[::-1]
        r_fft = np.fft.rfft(r, n=fft_len)
        a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(r_fft * a_rev_fft, n=fft_len)[: 2 * n - 1]
        # slice to length n
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # gradient descent on a (minimize squared error)
        a = a - step
        a = np.clip(a, 0.0, None)
        # renormalize onto simplex
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # optional learning rate schedule: decay
        if t % 1000 == 0:
            lr *= 0.9

        # perhaps restart if stagnating
        if t % 5000 == 0 and best_score > 1.6:
            # random perturb
            a = a + rng.normal(scale=0.001, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m = np.zeros_like(a)
            v = np.zeros_like(a)

    # final best convert to list clipped to 0-1000 as evaluator expects
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return best.tolist()

"""
Propose a step‑function (non‑negative list) that minimises
evaluate_sequence(sequence) = 2·n·max(conv) / (sum(sequence)²).

The algorithm works on the probability simplex (the vector is renormalised
to sum 1 after each update, because the objective is homogeneous).
It consists of two stages:

1.  Gradient descent on the squared‑error between the self‑convolution
    a*a and the perfectly flat target 1/(2·n‑1).  The gradient can be
    computed with three FFTs per iteration, and an Adam optimiser is
    used for robust step‑size adaptation.

2.  A short simulated‑annealing refinement that performs random pairwise
    mass transfers, accepting moves that improve the objective or that
    are accepted according to a temperature schedule.  This helps to
    escape any remaining tiny local minima.

The whole routine respects the 1000 s budget, prints progress, and
returns the best sequence found (clipped to the limits of evaluate_sequence).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length *fft_len*."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    s = v.sum()
    if s <= 0.0:
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Choose the length and initialise the distribution
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)                     # safety
    else:
        # fallback: a power‑of‑two sized uniform vector (nice for FFT)
        a = np.full(4096, 1.0 / 4096, dtype=np.float64)

    a = _simplex_normalise(a)
    n = a.size
    fft_len = _next_pow2(3 * n)          # ≥ 3·n for all convolutions we need

    # target flat convolution (average value of a*a)
    avg_conv = 1.0 / (2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Adam optimiser state
    # ------------------------------------------------------------------
    lr = 0.25                     # initial learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)         # first moment
    v = np.zeros_like(a)         # second moment
    t = 0

    # bookkeeping
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)      # 5 s safety margin

    # initial score
    conv = _conv_self(a, fft_len)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)   # (a.sum()==1) → 2·n·max
    best_a = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Gradient‑descent phase (flatten the convolution)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- compute convolution and current score --------------------
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (GD) = {best_score:.6f}")

            # early stop if we beat the target
            if best_score <= 1.5025:
                print("[early stop] target reached during gradient phase")
                break

        # ---- gradient of the squared‑error objective -----------------
        # loss = Σ (conv - avg)^2
        r = conv - avg_conv                     # residual (length 2n‑1)

        # convolution of residual with the reversed a gives ∂loss/∂a (up to 2·)
        a_rev = a[::-1]
        r_fft = np.fft.rfft(r, n=fft_len)
        a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(r_fft * a_rev_fft, n=fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # length n

        # ---- Adam update --------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- move, enforce feasibility --------------------------------
        a = a - step
        a = np.clip(a, 0.0, None)               # non‑negativity
        a = _simplex_normalise(a)               # stay on simplex

        # ---- learning‑rate schedule ----------------------------------
        if t % 2000 == 0:
            lr *= 0.9                           # gentle decay

        # ---- occasional random jitter to avoid stagnation -----------
        if t % 5000 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            a = _simplex_normalise(np.clip(a + jitter, 0.0, None))

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing refinement (to escape tiny local minima)
    # ------------------------------------------------------------------
    a = best_a.copy()
    temp_start = 0.02 * best_score
    it_sa = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        fraction = elapsed / (deadline - start)
        temperature = max(0.0, temp_start * (1.0 - fraction))

        # current convolution
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * n * conv.max()

        # pick a source index proportional to a (large masses move more)
        prob = a ** 2
        prob_sum = prob.sum()
        if prob_sum > 0:
            prob /= prob_sum
            src = int(rng.choice(n, p=prob))
        else:
            src = rng.integers(0, n)

        # random target, different from source
        tgt = rng.integers(0, n)
        if tgt == src:
            tgt = (tgt + 1) % n

        # move a small fraction of src to tgt
        delta = a[src] * rng.uniform(0.001, 0.03)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _simplex_normalise(cand)

        # evaluate candidate
        cand_conv = _conv_self(cand, fft_len)
        cand_score = 2.0 * n * cand_conv.max()

        # update best
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new best (SA) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached during SA phase")
                break

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp(
            (cur_score - cand_score) / max(1e-12, temperature)
        ):
            a = cand

        it_sa += 1
        if it_sa % 4000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] SA iter={it_sa}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to the evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
