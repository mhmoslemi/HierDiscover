"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Strategy
--------
* **Initial pool** – generate a handful of parametric families
  (uniform, Gaussian, power‑law, half‑sine, sparse Dirichlet) and keep the
  best candidate (score = 2·n·max(conv) because we keep the sum = 1).

* **Exact sub‑gradient descent** – the objective  max_k (a∗a)[k] is
  non‑smooth.  For the currently active lag(s) we can compute the exact
  sub‑gradient

        g_i = 2·∑_{k∈active} a_{k‑i}   (indices out of range are ignored)

  and perform a projected sub‑gradient step on the simplex.
  The step size decays like η_t = η0 / sqrt(t).

* **Stochastic local search** – after the deterministic phase we
  perform a simulated‑annealing style walk:
        – move a random fraction of mass from a “high‑gradient” index
          (large contribution to the current maximum) to a “low‑gradient”
          index,
        – accept the move if it improves the exact objective or with a
          probability exp(‑Δ/Temp) where the temperature slowly cools.
  Periodic “jitter’’ (stretch/compress a short segment) helps to leave
  shallow plateaus.

All operations are carried out with FFT‑based convolutions (O(n log n)).
The whole routine checks the remaining time and stops a few seconds
before the budget expires, returning the best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of  max_k (a∗a)[k]  for a set ``active`` of lag
    indices where the current maximum is attained.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact (unnormalised) score = 2·n·max(conv) for a simplex‑normalised a."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:

    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    """

    # --------------------------------------------------------------
    start_candidates = []

        score_new = float(2.0 * n * conv_new.max())

            elapsed = time.monotonic() - start_time

    # --------------------------------------------------------------
    a = best.copy()

    temperature = 0.008                 # initial temperature
    iter_local = 0

            conv = _conv_fft(a, fft_len)
            max_val = conv.max()
            tol = max_val * 1e-12 + 1e-15

            grad_active = _gradient_from_active(a, active)
            src = int(np.argmax(grad_active))          # donor (most harmful)

            low_thr = np.percentile(grad_active, 15)
            low_cands = np.where(grad_active <= low_thr)[0]
            if low_cands.size > 0:
                tgt = int(rng.choice(low_cands))
            else:
                tgt = int(rng.integers(0, n))

        # ensure donor still has some mass
        if a[src] <= 0.0:
            src = int(rng.integers(0, n))
        if src == tgt:
            tgt = int(rng.integers(0, n))

        # random fraction of the donor's mass (geometric sampling)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))   # ≈0.1 %–5 %
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # projection not required – the total sum is unchanged and both are ≥0

        conv_cand = _conv_fft(cand, fft_len)
        cand_score = float(2.0 * n * conv_cand.max())
        delta_score = cand_score - cur_score

        # Metropolis acceptance
        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] local move #{iter_local} improved to {best_score:.6f}")

        # cooling schedule
        temperature *= 0.99995
        iter_local += 1

        # broader jitter every few thousand steps
        if iter_local % 4000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.02))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)

    # --------------------------------------------------------------
    # 2.6 Final bookkeeping and return
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)  # respect the evaluator’s caps
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
