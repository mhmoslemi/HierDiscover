"""
Hybrid exponentiated‑gradient / smoothed‑max optimisation.

* Works on the probability simplex (∑a = 1, a_i ≥ 0).  The objective
  J(a) = 2·n·max(conv(a,a)) is scale‑invariant, so we keep the
  simplex constraint after every move.
* At each iteration we compute the full auto‑convolution via a real FFT.
  To obtain a useful descent direction we replace the hard max by a
  smooth log‑sum‑exp approximation:
        F(a) = (1/λ)·log(∑_k exp(λ·conv_k(a)))
  (λ is increased during the run – a classic continuation strategy).
  The gradient of F is
        ∇F = 2·(a_rev  *  w)[n‑1:n‑1+n]      where
        w_k = softmax_k(λ·conv) .
  The convolution a_rev * w is also performed with an FFT.
* A multiplicative (exponentiated‑gradient) step
        a ← a·exp(−η·∇F)
  maintains non‑negativity; a Euclidean projection onto the simplex
  restores the sum‑to‑one constraint.
* Metropolis acceptance with a linearly cooling temperature lets the
  walk escape shallow plateaus.  After 20 s without improvement we apply
  a small random jitter to the current best solution to restart the walk.
* Several diverse initial points (uniform, geometric, Dirichlet draws,
  the previous `height_sequence_1` if present) are tested; the best of
  them becomes the starting vector.
* The routine stops after the allotted budget (default 1000 s) and
  returns the best sequence found.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found
    within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size (reasonable for FFT speed)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)                 # for a*a
    grad_fft_len = _next_pow2(3 * n - 2)                # for a_rev * w

    # ------------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (increasing / decreasing)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – low, medium, high concentration
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best of the initial candidates (using the exact objective)
    # ------------------------------------------------------------------
    def exact_J(vec: np.ndarray) -> float:
        conv = _conv_full_fft(vec, fft_len_conv, conv_len)
        return float(2.0 * n * conv.max())

    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        cur_J = exact_J(cand)
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (smooth‑max + exponentiated‑gradient + annealing)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # Temperature schedule (linear cooling from 5 % of current best)
    T_start = best_J * 0.05
    T_end = 1e-8

    # λ‑continuation parameters
    lam = 5.0                # start smoothness
    lam_max = 500.0
    lam_factor = 1.02        # geometric increase per 100 iterations

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear cooling of the Metropolis temperature
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + \
            T_end * (elapsed / (deadline - start_time))

        # --------------------------------------------------------------
        # 4.1  Compute smoothed objective and its gradient
        # --------------------------------------------------------------
        conv = _conv_full_fft(cur_vec, fft_len_conv, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 4.2  Stagnation handling – jitter after 20 s without improvement
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            # Small multiplicative Gaussian perturbation of the best solution
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 4.3  Smooth‑max weights (softmax) and gradient of the smooth objective
        # --------------------------------------------------------------
        # Stabilise the exponentials
        max_c = max_conv
        exp_args = lam * (conv - max_c)
        # avoid overflow/underflow
        w = np.exp(exp_args)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum          # softmax weights, sum to 1

        # gradient = 2 * (a_rev * w)[n‑1 : n‑1+n]
        a_rev = cur_vec[::-1]
        A_rev = np.fft.rfft(a_rev, grad_fft_len)
        W = np.fft.rfft(w, grad_fft_len)
        grad_full = np.fft.irfft(A_rev * W, n=grad_fft_len)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # --------------------------------------------------------------
        # 4.4  Exponentiated‑gradient proposal
        # --------------------------------------------------------------
        step = 0.5 / np.sqrt(iteration + 1.0)   # diminishing step size
        factor = np.exp(-step * grad)           # multiplicative update
        proposal = cur_vec * factor
        proposal_sum = proposal.sum()
        if proposal_sum == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= proposal_sum               # project onto simplex

        # --------------------------------------------------------------
        # 4.5  Evaluate the proposal (exact objective)
        # --------------------------------------------------------------
        prop_conv = _conv_full_fft(proposal, fft_len_conv, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # --------------------------------------------------------------
        # 4.6  Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # --------------------------------------------------------------
        # 4.7  Update λ (smoothness) occasionally
        # --------------------------------------------------------------
        if iteration % 100 == 0 and lam < lam_max:
            lam = min(lam_max, lam * lam_factor)

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (clip, renormalise and cast to list)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
