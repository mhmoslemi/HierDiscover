"""
Propose a high‑quality step‑function sequence.

Algorithm (three phases, fully self‑contained, no I/O):

1. **Differential‑Evolution (DE) global search**  
   Works on the probability simplex (∑a_i = 1, a_i ≥ 0).  
   The population is mutated with `a + F·(b‑c)`, crossed‑over and
   projected onto the simplex by simple truncation + renormalisation.
   The DE parameters (F, CR) are annealed – large early for exploration,
   small later for exploitation.  The best individual seen so far is
   kept globally.

2. **Exponentiated‑gradient refinement**  
   Uses the exact sub‑gradient of the max‑convolution (all active
   indices where the current maximum is attained) on the logarithmic
   coordinates.  A diminishing step‑size is applied and the vector is
   re‑normalised to the simplex.

3. **Simulated‑annealing (SA) local polishing**  
   Tiny directed mass‑transfer moves between two positions are accepted
   by a Metropolis rule with an exponentially decaying temperature.
   This phase cleans up any remaining “spikes”.

The routine respects the given time budget (≈5 s safety margin)
and prints periodic progress information.  The returned sequence is
already clipped to the limits imposed by ``evaluate_sequence``.
"""

import numpy as np
from numpy.fft import rfft, irfft

# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Fast projection onto the probability simplex:
    set negatives to zero, then renormalise.
    If the vector becomes all zeros, fall back to uniform.
    """
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s > 0.0:
        return v / s
    n = v.shape[0]
    return np.full_like(v, 1.0 / n)


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2·len(a)‑1.
    """
    a_fft = rfft(a, n=fft_len)
    conv = irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute the objective value 2·n·max(a*a) for a vector a that already
    satisfies ∑a = 1.
    """
    max_b = _conv_fft(a, fft_len).max()
    return 2.0 * n * max_b


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k, the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the evaluator after we return
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 0) Initialise problem size and a feasible start vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / a0.sum()
        n0 = a0.shape[0]
        # pad to the next power of two for a cheap FFT length
        n = _next_pow2(n0)
        if n != n0:
            a0 = np.pad(a0, (0, n - n0), constant_values=0.0)
    else:
        n = 1024                     # a modest default size
        a0 = rng.random(n)
        a0 /= a0.sum()

    fft_len = _next_pow2(2 * n)

    # --------------------------------------------------------------
    # 1) Differential Evolution (global exploration)
    # --------------------------------------------------------------
    # population size grows slowly with n – enough diversity without
    # overwhelming the time budget.
    popsize = max(15, n // 256)
    population = np.empty((popsize, n), dtype=np.float64)
    scores = np.empty(popsize, dtype=np.float64)

    # warm start as first individual
    population[0] = a0.copy()
    scores[0] = _score(population[0], n, fft_len)

    # random remainder (Dirichlet with alpha=1 → uniform on simplex)
    for i in range(1, popsize):
        x = rng.gamma(1.0, 1.0, size=n)
        population[i] = x / x.sum()
        scores[i] = _score(population[i], n, fft_len)

    best_idx = int(np.argmin(scores))
    best = population[best_idx].copy()
    best_score = float(scores[best_idx])

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:.2f}s] initial best = {best_score:.6f}")

    # DE parameters – they will be annealed during the run
    target_score = 1.5025
    phase1_end = start_time + 0.70 * (deadline - start_time)

    while time.monotonic() < phase1_end:
        # linearly shrink F,CR as we approach the deadline
        remaining = deadline - time.monotonic()
        total = deadline - start_time
        frac = remaining / total                     # 1 → start, 0 → end
        F = 0.4 + 0.5 * frac                         # from ~0.9 down to 0.4
        CR = 0.5 + 0.4 * frac                        # from ~0.9 down to 0.5

        for i in range(popsize):
            # three distinct indices different from i
            while True:
                idxs = rng.choice(popsize, size=3, replace=False)
                if i not in idxs:
                    break
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, None)

            # binomial crossover – ensure at least one mutant component
            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(0, n)] = True
            trial = np.where(cross, mutant, population[i])

            # cheap simplex projection
            trial = _project_to_simplex(trial)

            cand_score = _score(trial, n, fft_len)

            # selection
            if cand_score < scores[i]:
                population[i] = trial
                scores[i] = cand_score

            # global best update
            if cand_score < best_score:
                best_score = cand_score
                best = trial.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] new best (DE) = {best_score:.6f}")

        if best_score <= target_score:
            print("[early stop] target reached during DE")
            break

    # --------------------------------------------------------------
    # 2) Exponentiated‑gradient refinement (exact sub‑gradient)
    # --------------------------------------------------------------
    a = best.copy()
    it_eg = 0
    while time.monotonic() < deadline - 0.3:   # reserve a few seconds for SA
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-9
        active = np.where(conv >= max_b - eps)[0]
        grad = _gradient_from_active(a, active)

        eta = 0.5 / np.sqrt(it_eg + 1.0)       # diminishing step
        log_a = np.log(a + 1e-15)
        log_a -= eta * grad
        a = np.exp(log_a)
        a = _project_to_simplex(a)

        score = _score(a, n, fft_len)
        if score < best_score:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] new best (EG) = {best_score:.6f}")
            if best_score <= target_score:
                print("[early stop] target reached during EG")
                break

        it_eg += 1

    # --------------------------------------------------------------
    # 3) Simulated‑annealing local polishing
    # --------------------------------------------------------------
    a = best.copy()
    curr_score = best_score
    T = 0.012                     # start temperature
    decay = 0.99985               # exponential cooling
    it_sa = 0

    while time.monotonic() < deadline:
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-16:
            continue

        # mass transfer: a small random fraction of src
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.04)))  # ~[0.001,0.04]
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_to_simplex(cand)

        cand_score = _score(cand, n, fft_len)
        Δ = cand_score - curr_score

        # Metropolis acceptance
        if Δ < 0.0 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] new best (SA) = {best_score:.6f}")
                if best_score <= target_score:
                    print("[early stop] target reached during SA")
                    break

        # occasional jitter – random scaling of a short block
        if it_sa % 7000 == 0 and it_sa > 0:
            width = rng.integers(1, max(2, n // 80))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start:start + width] *= factor
            a = _project_to_simplex(a_jit)

        # cooling
        T *= decay
        it_sa += 1
        if it_sa % 3000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] SA it={it_sa}, T={T:.3e}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
