def build_conv_constraints(f):
    n = len(f)
    # number of constraints m = 2*n-1
    m = 2*n-1
    idx = []
    indptr = [0]
    data = []
    # For each k:
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # j = k - i
        # i from i_start..i_end
        # Append coefficients f[i] for variable at j
        count = i_end - i_start + 1
        # iterate i
        for i in range(i_start, i_end+1):
            j = k - i
            data.append(f[i])
            idx.append(j)
        indptr.append(indptr[-1] + count)
    A = scipy.sparse.csr_matrix((data, idx, indptr), shape=(m, n))
    return A

"""
Propose a step‑function sequence that lowers the max‑convolution score.

Algorithm (≈ 1000 s budget)

1.  Try a handful of lengths (1024, 2048, 4096 and the length of the
    previously found `height_sequence_1`).  For each length we start from a
    uniform distribution (or from the historic sequence when the size matches).

2.  A short, cheap *exponentiated‑gradient* (EG) phase quickly moves the
    distribution away from the uniform worst‑case.

3.  If the length ≤ 2048 we perform a *Frank‑Wolfe / linear‑program* step:
    for the current distribution `f` we solve
        maximise   Σ g_j
        subject to  (f * g)[k] ≤ M      for every k,
                    g_j ≥ 0,
        where M = max(f * f).
    The LP gives a feasible direction `g`.  A few convex‑combination
    candidates `(1‑t) f + t g` (t ∈ {1,½,⅓,¼,⅛}) are evaluated and the best
    improving one is kept.  The process is repeated a few times.

4.  A lightweight pairwise mass‑transfer walk (random source/target,
    small fractional moves) refines the solution for the remaining time.

5.  Finally a brief simulated‑annealing walk (Metropolis acceptance) is run
    starting from the best sequence seen overall.

All sequences are kept on the probability simplex (sum = 1).  Scaling does
not affect the objective, therefore the returned list can be used directly
with `evaluate_sequence`.  The routine prints progress and stops early if
the target bound 1.5025 is reached.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (full length 2·len(a)‑1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the convolution entry (a*a)[idx].
    Component i receives a[idx‑i] when the index is in range, else 0.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)  # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    return grad

                        best_seq = a.copy()
                else:
                    print(f"[LP] iter {lp_it} no improvement → stop LP")
                    break

        # ------------------------------------------------------------------
        # 4) pairwise random mass‑transfer walk (cheap downhill moves)
        # ------------------------------------------------------------------
        max_pair_iter = 5000
        it_pair = 0
        while it_pair < max_pair_iter and time.monotonic() < deadline:
            src, tgt = rng.choice(n, size=2, replace=False)
            if a[src] <= 0.0:
                it_pair += 1
                continue
            frac = rng.uniform(0.001, 0.02)
            delta = a[src] * frac
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = _simplex_normalise(cand)

            conv_cand = _conv_self(cand, fft_len)
            score_cand = float(2.0 * n * conv_cand.max())

            if score_cand < best_score - 1e-12:
                best_score = score_cand
                best_seq = cand.copy()
                print(f"[pair] new best {best_score:.6f} (it={it_pair})")
                a = cand
                cur_score = score_cand
            elif score_cand <= cur_score:
                a = cand
                cur_score = score_cand

            it_pair += 1

        # ------------------------------------------------------------------
        # 5) early exit if target already reached
        # ------------------------------------------------------------------
        if best_score <= 1.5025:
            print(f"[{n}] target bound reached → {best_score:.6f}")
            break

    # ------------------------------------------------------------------
    # 6) final short simulated‑annealing refinement (if time remains)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        n = a.size
        fft_len = _next_pow2(3 * n)
        sa_iters = 2000
        for it in range(sa_iters):
            if time.monotonic() >= deadline:
                break
            # temperature schedule (linear decay)
            elapsed = time.monotonic() - start
            frac = elapsed / (deadline - start) if deadline > start else 1.0
            temperature = 0.02 * best_score * (1.0 - frac)

            # current convolution and its maximiser
            conv = _conv_self(a, fft_len)
            max_idx = int(np.argmax(conv))
            grad_max = _gradient_for_index(a, max_idx)

            # source selection biased by squared gradient
            prob_src = grad_max ** 2
            prob_sum = prob_src.sum()
            if prob_sum > 0.0:
                prob_src /= prob_sum
                src = int(rng.choice(n, p=prob_src))
            else:
                src = rng.integers(0, n)

            # random target (different from source)
            tgt = rng.integers(0, n)
            if tgt == src:
                tgt = (tgt + 1) % n

            delta = a[src] * rng.uniform(0.001, 0.025)
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = _simplex_normalise(cand)

            conv_cand = _conv_self(cand, fft_len)
            score_cand = float(2.0 * n * conv_cand.max())
            cur_score = float(2.0 * n * conv.max())

            if score_cand < best_score - 1e-12:
                best_score = score_cand
                best_seq = cand.copy()
                a = cand
                print(f"[SA] new best {best_score:.6f} (it={it})")
                if best_score <= 1.5025:
                    break
            elif score_cand < cur_score or rng.random() < math.exp((cur_score - score_cand) / max(1e-12, temperature)):
                a = cand

    # ------------------------------------------------------------------
    # 7) final evaluation & return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform 1024‑vector
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    total_time = time.monotonic() - start
    print(f"\nFinished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_seq]
