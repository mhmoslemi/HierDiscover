"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence`` using a smoothed max–convolution gradient descent.
The algorithm builds a probability vector (sum = 1) and repeatedly
  * evaluates the auto‑convolution,
  * builds a soft‑max weighting of the convolution entries,
  * computes the gradient of the weighted sum via an FFT‑based correlation,
  * performs a multiplicative (exponentiated‑gradient) update,
  * projects back onto the simplex,
  * occasionally adds a tiny random jitter to escape plateaus.

Multiple short restarts are performed until the time budget is exhausted.
The best sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    a_f = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_f * a_f, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two (potentially different‑length) vectors."""
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: x.size + y.size - 1]
    return conv


def _gradient_smoothing(a: np.ndarray,
                        conv: np.ndarray,
                        alpha: float,
                        fft_len: int) -> np.ndarray:
    """
    Gradient of a smooth approximation of ``max(conv)``.
    ``conv``  – full auto‑convolution of ``a`` (length 2n‑1).
    ``alpha`` – temperature of the soft‑max (larger → sharper).
    Returns a vector of length n.
    """
    # soft‑max weights (numerically stable)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    w /= w.sum()                     # now sum(w)=1

    # correlation a ⋆ w  (equivalent to ∂/∂a of Σ_k w_k·conv_k)
    w_rev = w[::-1]                  # reversed weight vector
    corr_full = _conv_fft_two(a, w_rev, fft_len)

    offset = w_rev.size - 1          # = 2*n-2
    # extract exactly n entries: g_i = 2* Σ_k w_k a_{k-i}
    grad = 2.0 * corr_full[offset - np.arange(a.size)]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise (numerical safety)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / len(w))
    else:
        w /= s
    return w

            else:
                g_norm = np.zeros_like(grad)

            # ----- learning‑rate schedule (linear decay) --------------------
            progress = it / max_iters
            eta = eta0 * (1.0 - progress) + 1e-6 * progress

            # ---------- multiplicative (exponentiated‑gradient) update -------
            # clip the normalised gradient to avoid overflow in exp()
            g_norm = np.clip(g_norm, -5.0, 5.0)
            a = a * np.exp(-eta * g_norm)

            # ---------- project back onto the simplex -----------------------
            a = _project_simplex(a)

            # ---------- occasional jitter to avoid stagnation ---------------
            if (it + 1) % 500 == 0:
                sigma = 0.018
                jitter = np.exp(rng.normal(0.0, sigma, size=n))
                a = a * jitter
                a = _project_simplex(a)

        # --------------------------------------------------------------
        # one restart finished
        # --------------------------------------------------------------
        restart += 1

        # quick exit if target already reached
        if best_score <= 1.5025:
            break

    # --------------------------------------------------------------
    # 5) Final polishing – one short simulated‑annealing pass
    # --------------------------------------------------------------
    a = best_seq.copy()
    cur_conv = _conv_fft(a, fft_len)
    cur_score = 2.0 * n * cur_conv.max()
    T_start = 1e-2 * cur_score
    T_end = 1e-8 * cur_score
    total_anneal = 2000
    for it in range(total_anneal):
        if time.monotonic() >= deadline:
            break
        # current temperature (linear)
        temp = T_start * (1.0 - it / total_anneal) + T_end * (it / total_anneal)

        # pick src/tgt from gradient (same as before)
        grad = _gradient_smoothing(a, cur_conv, alpha, fft_len)
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_simplex(cand)

        cand_conv = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * cand_conv.max()

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            a, cur_conv, cur_score = cand, cand_conv, cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                elapsed = time.monotonic() - start_time
                print(
                    f"[{elapsed:6.2f}s] anneal improvement → best = {best_score:.6f}"
                )
        # occasional tiny jitter inside anneal
        if it % 400 == 0 and it > 0:
            jitter = np.exp(rng.normal(0.0, 0.01, size=n))
            a = a * jitter
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # 6) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(v) for v in best_seq]
