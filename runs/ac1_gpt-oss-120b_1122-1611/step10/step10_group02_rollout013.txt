assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the “step heights”) that minimises  

```
score(a) = 2·n·max(conv(a,a)) / ( Σ a )²
```

`conv(a,a)` is the linear convolution of `a` with itself.  
Because the objective is homogeneous, we can restrict the search to the
probability simplex `Σ a = 1`.  With this normalisation the objective reduces to  

```
score(a) = 2·n·max(conv(a,a))
```

Uniform weights give `score = 2`.  The best known construction reaches
≈ 1.513.  The new optimiser uses three ideas that are not present in the
previous code:

1. **Smooth approximation of the max** – we minimise the
   *log‑sum‑exp* of the convolution values,
   `f(a)= (1/β)·log Σ exp(β·conv_k)`.  Its gradient is a weighted sum of the
   individual convolution gradients, where the weights are a soft‑max over the
   convolution entries.  This gives a well‑behaved gradient even when many
   indices are almost maximal.

2. **Additive projected gradient descent with an Adam‑style step size** – a
   simple additive step `a ← a – η·∇f(a)` followed by Euclidean projection onto
   the simplex.  Adam’s adaptive moments make the method robust to the
   different scales of the gradient components.

3. **Cold‑start / warm‑start handling + simulated‑annealing polishing** – the
   optimiser starts from the provided `height_sequence_1` (if any) or from a
   uniform random vector, runs two gradient phases with increasing smoothness
   parameter `β`, and finishes with a short simulated‑annealing phase that
   makes tiny directed mass‑transfer moves.  The SA phase aggressively
   explores the neighbourhood of the best solution found by gradient descent.

All heavy work is performed with FFT‑based convolutions (`O(n log n)`);
for a typical length of a few thousand the whole search finishes well
within the 1000 s budget.

```python
"""
gradient_sa_optimiser.py

A three‑phase optimiser for the max‑convolution objective:

1. Smoothed‑max gradient descent (soft‑max weighting of the convolution).
   Uses an Adam‑style step size and projects onto the probability simplex.
2. A second gradient phase with a larger smoothing parameter (β) to focus
   on the true maximum.
3. Simulated‑annealing polishing (tiny directed mass transfers) to escape
   possible local minima.

The routine respects a 1000 s time budget, prints intermediate scores and
returns the best height sequence found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2·len(a)‑1.
    """
    a_fft = rfft(a, n=fft_len)
    conv = irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_lin(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two (possibly different) vectors a and b
    using an FFT of length ``fft_len``.
    Returns the convolution of length len(a)+len(b)-1.
    """
    a_fft = rfft(a, n=fft_len)
    b_fft = rfft(b, n=fft_len)
    conv = irfft(a_fft * b_fft, n=fft_len)[: a.size + b.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a soft‑max over the convolution values.
    Returns a probability vector of the same length as ``conv``.
    """
    # for numerical stability subtract the max
    max_c = conv.max()
    shifted = conv - max_c
    exp_term = np.exp(beta * shifted)
    s = exp_term.sum()
    if s == 0.0:                     # should not happen, but be safe
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / s


def _gradient_weighted_conv(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ w_k·conv_k (conv = a*a) w.r.t. a.
    w is a probability vector of length 2·n‑1.
    Returns a vector of length n.
    """
    a_rev = a[::-1]
    fft_len = _next_pow2(w.size + a_rev.size - 1)
    full = _conv_lin(w, a_rev, fft_len)          # length = 3·n‑2
    n = a.size
    # cross‑correlation slice that corresponds to ∂/∂a_i
    return full[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.

    The algorithm consists of:
      * Phase 1 – soft‑max gradient descent with a small β (smooth objective)
      * Phase 2 – the same but with a larger β (focus on the true max)
      * Phase 3 – simulated‑annealing polishing
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Initialise a feasible probability vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a = np.full_like(a0, 1.0 / a0.size)
        else:
            a = a0 / a0.sum()
    else:
        # moderate size for fast experiments – can be increased later
        n = 3000
        a = rng.random(n)
        a = a / a.sum()

    n = a.size
    fft_len = _next_pow2(2 * n)          # for a*a convolutions

    # ------------------------------------------------------------------
    # 1) Book‑keeping for the best solution
    # ------------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()    # identical to evaluate_sequence for Σa=1
    best = a.copy()
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)   # leave a safety margin
    deadline = start_time + total_budget
    target_score = 1.5025

    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Phase 1 – smoothed gradient descent (β grows from 0.5 to 5)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * total_budget
    beta_start = 0.5
    beta_end = 5.0
    step0 = 0.25                     # base step size for additive update
    t = 0
    m = np.zeros_like(a)            # Adam first moment
    v = np.zeros_like(a)            # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    while time.monotonic() < phase1_end:
        # current convolution and β schedule
        conv = _conv_fft(a, fft_len)
        elapsed = time.monotonic() - start_time
        phase_ratio = elapsed / total_budget
        beta = beta_start * (beta_end / beta_start) ** phase_ratio

        # soft‑max weights (smooth max)
        w = _softmax_weights(conv, beta)

        # gradient of the smoothed objective
        grad = _gradient_weighted_conv(a, w)

        # Adam update on the *additive* step (projected after the move)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0 / np.sqrt(t)   # additional decay
        a_candidate = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a_candidate = _project_onto_simplex(a_candidate)

        # cheap score estimate (exact, because Σa=1)
        conv_cand = _conv_fft(a_candidate, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = a_candidate.copy()
            print(f"[{elapsed:6.2f}s] new best (phase1) = {best_score:.6f}")
            if best_score <= target_score:
                print("[early stop] target reached in phase 1")
                a = a_candidate
                break
        a = a_candidate

        # occasional logging
        if t % 2000 == 0:
            print(f"[{elapsed:6.2f}s] phase1 it={t}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 2 – sharper gradient descent (β grows from 5 to 80)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * total_budget
    beta_start2 = 5.0
    beta_end2 = 80.0
    step0_2 = 0.12
    t = 0
    m.fill(0.0)
    v.fill(0.0)

    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        elapsed = time.monotonic() - start_time
        phase_ratio = (elapsed - 0.40 * total_budget) / (0.30 * total_budget)
        phase_ratio = max(0.0, min(1.0, phase_ratio))
        beta = beta_start2 * (beta_end2 / beta_start2) ** phase_ratio

        w = _softmax_weights(conv, beta)
        grad = _gradient_weighted_conv(a, w)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = step0_2 / np.sqrt(t)
        a_candidate = a - step * (m_hat / (np.sqrt(v_hat) + eps))
        a_candidate = _project_onto_simplex(a_candidate)

        conv_cand = _conv_fft(a_candidate, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = a_candidate.copy()
            print(f"[{elapsed:6.2f}s] new best (phase2) = {best_score:.6f}")
            if best_score <= target_score:
                print("[early stop] target reached in phase 2")
                a = a_candidate
                break
        a = a_candidate

        if t % 3000 == 0:
            print(f"[{elapsed:6.2f}s] phase2 it={t}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 3 – simulated‑annealing local refinement (remaining time)
    # ------------------------------------------------------------------
    a = best.copy()
    curr_score = best_score
    T = 0.018                     # start temperature
    decay = 0.9999                # exponential cooling
    it = 0

    while time.monotonic() < deadline:
        # --- tiny directed mass transfer ---------------------------------
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-15:
            continue
        # multiplicative fraction in [0.001, 0.04] (log‑uniform)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.04)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        conv_cand = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        Δ = cand_score - curr_score

        # Metropolis acceptance
        if Δ < 0 or rng.random() < np.exp(-Δ / max(T, 1e-15)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f}")

        # occasional block‑jitter to re‑stir the distribution
        if it % 5000 == 0 and it > 0:
            width = rng.integers(2, max(3, n // 60))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            a_j = a.copy()
            a_j[start:start + width] *= factor
            a = _project_onto_simplex(a_j)

        T *= decay
        it += 1
        if it % 4000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA it={it}, T={T:.2e}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence found (apply evaluator clipping)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # store the result for possible warm‑starts in later runs
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
```
