"""
Iterative Linear‑Program (LP) refinement for the step‑function problem.

The algorithm works on the simplex (∑a_i=1, a_i≥0).  For the current
distribution f we compute its maximal auto‑correlation M = max(f*f).  Holding
f fixed we solve a linear program

    max   Σ b_i
    s.t.  (f * b)_k ≤ M   for all k = 0 … 2n‑2
          0 ≤ b_i ≤ Bmax

which asks how much total mass can be placed in a new vector b without
exceeding the same convolution bound M.  If the optimum satisfies
Σb_i > 1 we obtain a “more spread out’’ candidate g = b / Σb.
Because the convolution scales quadratically, g usually has a strictly
smaller max‑auto‑correlation.  We accept the improvement (or a convex
mixture with the current f) and repeat the process until the time budget
runs out.

If the LP does not yield any improvement we fall back to a small random
jitter (a cheap simulated‑annealing style move) to escape potential
plateaus.  The best sequence found is finally returned as a plain Python
list; it is already normalised (∑a_i=1) and respects the evaluator’s
clipping limits.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """
    For a vector on the probability simplex (∑a_i=1) the value returned
    by ``evaluate_sequence`` equals 2·n·max(conv).  This helper avoids
    the slower call to ``evaluate_sequence`` during the search.
    """
    return 2.0 * n * conv.max()


def _solve_lp_max_mass(f: np.ndarray, M: float, Bmax: float = 1000.0,
                      solver: str = "ECOS") -> np.ndarray | None:
    """
    Solve the linear program
        max  Σ b_i
        s.t. conv(f, b) ≤ M   (component‑wise)
             0 ≤ b_i ≤ Bmax
    Returns the optimal b as a NumPy array, or ``None`` if the solver
    fails or declares the problem infeasible/unbounded.
    """
    n = f.size
    b = cp.Variable(n, nonneg=True)          # b ≥ 0
    f_param = cp.Parameter(n, nonneg=True)   # fixed convolution kernel
    f_param.value = f

    # Linear convolution f * b (length 2n‑1) – cvxpy will keep it linear in b
    conv_expr = cp.conv(f_param, b)

    # Build constraints
    constraints = [conv_expr <= M,
                   b <= Bmax]   # elementwise upper bound

    # Linear objective: maximize Σ b_i  <=> minimize –Σ b_i
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=solver,
                   warm_start=True,
                   max_iters=2000,
                   abstol=1e-6,
                   reltol=1e-6,
                   feastol=1e-6,
                   verbose=False)
    except Exception:          # any solver mishap → treat as failure
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return b.value

# Main search routine
# ----------------------------------------------------------------------

    * When the LP does not improve the solution we apply a cheap random jitter

        # default length – a power of two makes the FFT cheap
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(3 * n)          # safe length for linear convolution

    # ------------------------------------------------------------------
    # 1) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    cur_conv = _conv_fft(a, fft_len)
    cur_score = _objective_from_conv(cur_conv, n)

    best = a.copy()
    best_score = cur_score
    print(f"[init] n={n}  initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Time‑budget handling (keep a 5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    iteration = 0
    stagnation = 0                     # number of iterations without improvement

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # (a) current worst auto‑correlation M
        # --------------------------------------------------------------
        conv = _conv_fft(a, fft_len)
        M = conv.max()                 # this is the bound used in the LP

        # --------------------------------------------------------------
        # (b) solve the LP  max Σ b_i  s.t. (f * b) ≤ M
        # --------------------------------------------------------------
        b_opt = _solve_lp_max_mass(a, M, Bmax=1000.0, solver="ECOS")
        if b_opt is None:
            # LP failed – fall back to a random jitter move
            jitter = True
        else:
            sum_b = float(np.sum(b_opt))
            if sum_b <= 1e-12:
                jitter = True
            else:
                # Normalise to a probability vector (sum = 1)
                g = b_opt / sum_b
                g = np.clip(g, 0.0, None)
                g_sum = g.sum()
                if g_sum == 0.0:
                    jitter = True
                else:
                    g /= g_sum

                    # Evaluate the candidate (fast, using the FFT formula)
                    conv_g = _conv_fft(g, fft_len)
                    cand_score = _objective_from_conv(conv_g, n)

                    jitter = False

        # --------------------------------------------------------------
        # (c) Acceptance logic
        # --------------------------------------------------------------
        improved = False
        if not jitter:
            # Direct acceptance if strictly better than the global best
            if cand_score + 1e-12 < best_score:
                best = g.copy()
                best_score = cand_score
                a = g.copy()                # also move the current state
                improved = True
                stagnation = 0
                print(f"[{time.monotonic() - start_time:6.2f}s] "
                      f"LP improvement -> best = {best_score:.6f} (iter={iteration})")
            else:
                # Try a few convex mixtures with the current state.
                for mix in (0.5, 0.3, 0.1):
                    h = (1.0 - mix) * a + mix * g
                    h = np.clip(h, 0.0, None)
                    h /= h.sum()
                    conv_h = _conv_fft(h, fft_len)
                    h_score = _objective_from_conv(conv_h, n)
                    if h_score + 1e-12 < best_score:
                        best = h.copy()
                        best_score = h_score
                        a = h.copy()
                        improved = True
                        stagnation = 0
                        print(f"[{time.monotonic() - start_time:6.2f}s] "
                              f"mix={mix:.2f} -> best = {best_score:.6f} (iter={iteration})")
                        break

        # --------------------------------------------------------------
        # (d) If the LP (and mixtures) did not improve, perform a small
        #     random jitter to escape a possible plateau.
        # --------------------------------------------------------------
        if not improved:
            stagnation += 1
            if stagnation >= 8:   # every few stagnant iterations we jitter
                block_len = max(1, int(rng.integers(1, max(2, n // 80))))
                start = int(rng.integers(0, n - block_len + 1))
                factor = np.exp(rng.normal(0.0, 0.12, size=block_len))
                a_jitter = a.copy()
                a_jitter[start:start + block_len] *= factor
                a_jitter = np.clip(a_jitter, 0.0, None)
                a_jitter /= a_jitter.sum()
                conv_j = _conv_fft(a_jitter, fft_len)
                jitter_score = _objective_from_conv(conv_j, n)
                if jitter_score + 1e-12 < best_score:
                    best = a_jitter.copy()
                    best_score = jitter_score
                    a = a_jitter.copy()
                    stagnation = 0
                    print(f"[{time.monotonic() - start_time:6.2f}s] "
                          f"random jitter -> best = {best_score:.6f} (iter={iteration})")
                else:
                    # Keep the jittered vector as the current state only
                    # if it is not dramatically worse.
                    cur_score = _objective_from_conv(conv, n)
                    if jitter_score < cur_score * 1.02:
                        a = a_jitter.copy()

        iteration += 1

        # Early exit if we have already reached the target bound
        if best_score <= 1.5025:
            print("[early stop] target bound reached")
            break

    # ------------------------------------------------------------------
    # 3) Finalise and return the best sequence
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # respect evaluator caps
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start_time
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
