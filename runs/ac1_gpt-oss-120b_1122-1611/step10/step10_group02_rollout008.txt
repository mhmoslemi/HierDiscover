"""
propose_candidate
=================

A three‑phase optimiser that first explores a small family of analytically
motivated shapes (Beta‑type, symmetric exponentials and simple mixtures) on a
moderately large support, then refines the best candidate with an
exponentiated‑gradient / sub‑gradient descent on the simplex, and finally
does a short simulated‑annealing polishing.  All heavy work (convolutions)
uses an FFT, so we can afford thousands of iterations within the 1000 s
budget.  The routine works entirely on the simplex (∑a_i = 1); the final
list is returned unchanged – ``evaluate_sequence`` rescales it automatically.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
# Helper utilities (top level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # shift by the largest element to avoid numerical issues
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2·len(a)‑1.
    """
    a_fft = rfft(a, n=fft_len)
    conv = irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for a set of active indices.
    For each active k the contribution to ∂max/∂a_i is 2·a_{k-i}
    (with out‑of‑range entries ignored).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)  # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad


def _score_fft(a: np.ndarray, fft_len: int) -> float:
    """Score 2·n·max(conv) using the FFT‑based convolution."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    safety = 5.0                     # keep a few seconds free for the caller
    deadline = start_time + max(0.0, float(budget_s) - safety)

    # --------------------------------------------------------------
    # 0) Problem size – use a power‑of‑two length for fast FFTs.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        n = max(1024, a0.size)
        # optionally enlarge to a power of two
        n = 1 << ((n - 1).bit_length())
    else:
        n = 4096          # a comfortable size (FFT friendly)

    # initialise with a uniform distribution on the chosen length
    a = np.full(n, 1.0 / n)
    fft_len = _next_pow2(2 * n)

    # --------------------------------------------------------------
    # 1) Very fast exploratory search over simple parametric families.
    # --------------------------------------------------------------
    best = a.copy()
    best_score = _score_fft(best, fft_len)

    print(f"[0.00s] initial (uniform) score = {best_score:.6f}")

    # ---- Helper to evaluate a candidate and update the global best ----
    def try_candidate(candidate: np.ndarray, tag: str = ""):
        nonlocal best, best_score
        cand = np.clip(candidate, 0.0, None)
        cand /= cand.sum() if cand.sum() > 0 else 1.0
        score = _score_fft(cand, fft_len)
        if score < best_score - 1e-12:
            best_score = score
            best = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best {tag}= {best_score:.6f}")

    # ---- 1a) Beta‑type shapes  (symmetric Beta(p,p) density) ----------
    p_vals = np.linspace(0.55, 1.0, 10)          # p=1 gives uniform
    x = (np.arange(n) + 0.5) / n                 # centre of each bin
    for p in p_vals:
        if time.monotonic() >= deadline:
            break
        # pdf ∝ x^{p-1}(1-x)^{p-1}
        a_beta = (x ** (p - 1.0)) * ((1.0 - x) ** (p - 1.0))
        a_beta = np.maximum(a_beta, 0.0)
        try_candidate(a_beta, f"beta(p={p:.2f})")

    # ---- 1b) Symmetric exponential tails ------------------------------
    lam_vals = np.logspace(-4, -1, 12)  # from 1e‑4 to 1e‑1
    idx = np.arange(n)
    for lam in lam_vals:
        if time.monotonic() >= deadline:
            break
        a_exp = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
        try_candidate(a_exp, f"exp(lam={lam:.4g})")

    # ---- 1c) Simple mixtures of uniform and the above -----------------
    mix_weights = np.linspace(0.0, 1.0, 6)   # proportion of uniform part
    for w in mix_weights:
        if time.monotonic() >= deadline:
            break
        # mix uniform + Beta(p=0.7) as a representative
        a_beta = (x ** (-0.3)) * ((1.0 - x) ** (-0.3))   # p=0.7
        a_mix = w * (1.0 / n) + (1.0 - w) * a_beta
        try_candidate(a_mix, f"mix(w={w:.2f})")

    # ---- 1d) A few random starts for diversity -----------------------
    for _ in range(8):
        if time.monotonic() >= deadline:
            break
        a_rand = rng.random(n)
        a_rand /= a_rand.sum()
        try_candidate(a_rand, "rand")

    # At this point ``best`` holds the most promising shape we have seen.
    # --------------------------------------------------------------
    # 2) Phase‑1 – exponentiated‑gradient descent (sub‑gradient on max).
    #    Use the remaining time (≈55 % of the budget) for a fine‑grained
    #    descent with a slowly decreasing learning rate.
    # --------------------------------------------------------------
    phase1_end = start_time + 0.55 * (deadline - start_time)
    a = best.copy()
    m = np.zeros_like(a)      # momentum (Adam‑style first moment)
    v = np.zeros_like(a)      # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0
    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        # treat all near‑max entries as active (tolerance ~1e‑8 of the max)
        active = np.where(conv >= max_b - max_b * 1e-8)[0]
        grad = _gradient_from_active(a, active)

        # Adam‑style adaptive step on the *log* variables
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        it += 1
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        # base learning rate – empirically chosen
        lr = 0.35 / np.sqrt(it + 1.0)
        delta_log = lr * m_hat / (np.sqrt(v_hat) + eps)

        log_a = np.log(a + 1e-14) - delta_log
        a = np.exp(log_a)
        a = _project_onto_simplex(a)

        # evaluate the true objective every few iterations
        if it % 250 == 0:
            score = _score_fft(a, fft_len)
            if score < best_score - 1e-12:
                best_score = score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (exp‑grad) = {best_score:.6f}")

        # safety break
        if time.monotonic() >= phase1_end:
            break

    # --------------------------------------------------------------
    # 3) Phase‑2 – Simulated‑annealing polishing (remaining time).
    # --------------------------------------------------------------
    a = best.copy()
    curr_score = best_score
    T = 0.012                     # initial temperature
    decay = 0.99990               # exponential cooling
    it = 0
    while time.monotonic() < deadline:
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-15:
            continue
        # move a random fraction (log‑uniform) of mass from src to tgt
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)
        cand = _project_onto_simplex(cand)

        cand_score = _score_fft(cand, fft_len)
        delta_score = cand_score - curr_score

        # Metropolis acceptance
        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(T, 1e-12)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f}")

        # occasional jitter – rescale a short random block
        if it % 6000 == 0 and it > 0:
            width = rng.integers(1, max(2, n // 80))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            jitter = a.copy()
            jitter[start:start + width] *= factor
            a = _project_onto_simplex(jitter)

        T *= decay
        it += 1

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clipped to evaluator limits).
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # final verification – the official scorer will do the same,
    # but we report the exact value here for transparency.
    final_score = evaluate_sequence(best.tolist())
    elapsed = time.monotonic() - start_time
    print(f"Finished in {elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
