"""
Propose a step‑function (non‑negative vector) minimizing the
``evaluate_sequence`` score.

Algorithm
---------
1. **Adam on a smooth surrogate** – we minimise a log‑sum‑exp
   approximation of   max(conv(a,a)).  The variables are the logits
   `x` of a soft‑max, guaranteeing non‑negativity and unit sum.
   The smoothing parameter `beta` is increased slowly so that the
   surrogate morphs into the true max‑convolution.
2. **Greedy pairwise mass transfer** – after the Adam phase we
   repeatedly move a small fraction of mass from the index that most
   contributes to the current maximal convolution entry to the index
   that contributes the least.  This is an exact sub‑gradient move on
   the true max‑convolution.
3. **Simulated‑annealing refinement** – a final stochastic walk
   (Metropolis acceptance) with a linearly decaying temperature
   helps to escape any remaining shallow local minima.
4. **Multiple restarts** – if time permits we restart the whole
   procedure from a fresh random Dirichlet sample; the best sequence
   over all runs is returned.

All operations are performed with FFT‑based convolutions (O(n log n))
and the current best score is tracked throughout.  The routine respects
the 1000 s budget (with a 5 s safety margin) and prints progress
information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2*len(a)-1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_weighted(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Convolution of a (length n) with a weight vector w (length 2n‑1).
    Returns the full linear convolution (length n+len(w)-1 = 3n‑2).
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: a.size + w.size - 1]
    return full


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = v.max()
    e = np.exp(v - m)
    return e / e.sum()


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact evaluate_sequence score when sum(a)=1 (i.e. score = 2 n max(conv))."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Starting distribution (sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
    else:
        a0 = None

    # use a power‑of‑two length for nice FFTs – 4096 works well
    n = 4096
    if a0 is not None and a0.size > 0:
        # up‑sample / truncate to the chosen length
        if a0.size != n:
            # simple linear interpolation onto the new grid
            x_src = np.linspace(0, 1, a0.size, endpoint=False)
            x_dst = np.linspace(0, 1, n, endpoint=False)
            a0 = np.interp(x_dst, x_src, a0)
        a0 = np.clip(a0, 0.0, None)
        a0 = a0 / a0.sum()
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # FFT length sufficient for all convolutions we need (≥ 3n)
    fft_len = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # initialise best
    conv0 = _conv_self(a0, fft_len)
    best_score = _score_from_conv(conv0, n)
    best_a = a0.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Adam optimisation on a smoothed max‑convolution surrogate
    # ------------------------------------------------------------------
    # log‑it representation (soft‑max) guarantees feasibility
    x = np.log(a0 + 1e-16)          # logits
    m = np.zeros_like(x)           # first moment
    v = np.zeros_like(x)           # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12                      # base learning rate for Adam
    beta_start, beta_end = 2.0, 180.0   # smoothing schedule
    phase1_end = start + 0.55 * (deadline - start)
    it = 0
    while time.monotonic() < phase1_end:
        # ------- logits → probabilities -------
        mx = x.max()
        e = np.exp(x - mx)
        a = e / e.sum()                     # a ∈ simplex, sum = 1

        # ------- self‑convolution -------
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ------- smoothing parameter β (linear schedule) -------
        beta = beta_start + (beta_end - beta_start) * (it / 150000.0)
        if beta > beta_end:
            beta = beta_end

        # ------- soft‑max weights for the smoothed max -------
        c_max = conv.max()
        # shift for numerical stability
        exp_vals = np.exp(beta * (conv - c_max))
        w = exp_vals / exp_vals.sum()               # w has length 2n‑1

        # ------- gradient of smoothed objective w.r.t a -------
        w_fft = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: n + w.size - 1]
        grad_a = grad_full[:n]                     # same shape as a

        # ------- chain rule: ∂/∂x (soft‑max) -------
        grad_x = a * (grad_a - np.dot(grad_a, a))

        # ------- Adam update -------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ------- occasional exact evaluation -------
        if it % 1200 == 0:
            score = _score_from_conv(conv, n)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f} (it={it})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in Adam phase")
                    break
        it += 1

    # ------------------------------------------------------------------
    # 2) Targeted pairwise mass transfer (greedy sub‑gradient step)
    # ------------------------------------------------------------------
    a = best_a.copy()
    phase2_end = start + 0.85 * (deadline - start)
    it2 = 0
    while time.monotonic() < phase2_end:
        conv = _conv_self(a, fft_len)
        max_val = conv.max()
        max_idx = int(np.argmax(conv))

        # exact sub‑gradient for the active maximal entry
        grad_max = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n, max_idx + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad_max[i] = a[max_idx - i]

        src = int(np.argmax(grad_max))
        tgt = int(np.argmin(grad_max))

        # move a modest random fraction of the source mass
        frac = math.exp(rng.uniform(math.log(0.001), math.log(0.035)))
        delta = a[src] * frac
        if delta > a[src]:
            delta = a[src] * 0.5

        candidate = a.copy()
        candidate[src] -= delta
        candidate[tgt] += delta
        # re‑normalise (projection onto simplex)
        cand_sum = candidate.sum()
        if cand_sum > 0:
            candidate /= cand_sum
        else:
            candidate = np.full_like(candidate, 1.0 / n)

        conv_cand = _conv_self(candidate, fft_len)
        score_cand = _score_from_conv(conv_cand, n)

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = candidate.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (pair) = {best_score:.6f} (it={it2})")
            if best_score <= 1.5025:
                print("[early stop] target reached in pairwise phase")
                break

        # accept non‑worsening moves to keep a downhill walk
        if score_cand <= _score_from_conv(conv, n):
            a = candidate

        it2 += 1

        # occasional jitter to avoid stagnation
        if it2 % 2500 == 0:
            jitter = rng.normal(scale=0.0015, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()

        if it2 % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] pairwise iter={it2}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it3 = 0
    temp_start = 0.025 * best_score
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        fraction = elapsed / (deadline - start)
        temperature = temp_start * (1.0 - fraction)

        conv = _conv_self(a, fft_len)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        grad_max = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n, max_idx + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad_max[i] = a[max_idx - i]

        # source biased by squared gradient magnitude (more aggressive)
        prob_src = grad_max ** 2
        prob_sum = prob_src.sum()
        if prob_sum > 0:
            prob_src /= prob_sum
            src = int(rng.choice(n, p=prob_src))
        else:
            src = rng.integers(0, n)

        tgt = rng.integers(0, n)
        if tgt == src:
            tgt = (tgt + 1) % n

        delta = a[src] * rng.uniform(0.001, 0.032)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum
        else:
            cand = np.full_like(cand, 1.0 / n)

        conv_cand = _conv_self(cand, fft_len)
        score_cand = _score_from_conv(conv_cand, n)
        curr_score = _score_from_conv(conv, n)

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (SA)   = {best_score:.6f} (it={it3})")
            if best_score <= 1.5025:
                print("[early stop] target reached in SA phase")
                break

        # Metropolis acceptance
        if score_cand < curr_score or rng.random() < math.exp((curr_score - score_cand) / max(1e-12, temperature)):
            a = cand

        it3 += 1
        if it3 % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] SA iter={it3}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional additional restart if time still remains
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 1.0:   # at least one second for a tiny run
        # tiny fresh run (very few Adam steps) to possibly escape a flat region
        # initialise from a Dirichlet sample
        x = np.log(rng.exponential(scale=1.0, size=n) + 1e-16)
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        it_restart = 0
        restart_deadline = time.monotonic() + 0.07 * (deadline - start)   # ~7 % of remaining budget
        while time.monotonic() < restart_deadline:
            # logits → a
            mx = x.max()
            e = np.exp(x - mx)
            a = e / e.sum()
            a_fft = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

            beta = 5.0 + (beta_end - 5.0) * (it_restart / 2000.0)
            if beta > beta_end:
                beta = beta_end

            cmax = conv.max()
            exp_vals = np.exp(beta * (conv - cmax))
            w = exp_vals / exp_vals.sum()
            w_fft = np.fft.rfft(w, n=fft_len)
            grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: n + w.size - 1]
            grad_a = grad_full[:n]

            grad_x = a * (grad_a - np.dot(grad_a, a))

            it_restart += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
            m_hat = m / (1.0 - beta1 ** it_restart)
            v_hat = v / (1.0 - beta2 ** it_restart)
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
            x -= step

        # evaluate the short run
        conv = _conv_self(a, fft_len)
        score = _score_from_conv(conv, n)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (restart) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(v) for v in best]
