"""
Binary‑support simulated‑annealing optimizer.

The idea:  use a probability vector that is non‑zero on a small set S
of size *m* (all non‑zero entries are equal to 1/m).  For such a vector

    (a*a)[k] = cnt[k] / m**2 ,

where ``cnt[k]`` is the number of ordered pairs (i,j)∈S×S with i+j=k.
The evaluation function is

    score = 2·n·max_k cnt[k] / m**2 .

Thus reducing the maximum pair‑sum count of S directly reduces the
objective.  The algorithm searches for a set S with a small
max‑count using a fast O(m)‑per‑move incremental update of the
pair‑sum histogram and a simple simulated‑annealing schedule.
All operations are pure NumPy/Python and respect the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _compute_counts(support: np.ndarray, n: int) -> np.ndarray:
    """
    Return the ordered‑pair histogram ``cnt`` of size 2·n‑1 for a given
    support set (array of indices).  ``cnt[k]`` equals the number of
    ordered pairs (i,j)∈support×support with i+j = k.
    """
    # Outer sum of the support indices (shape m×m) and flatten.
    sums = np.add.outer(support, support).ravel()
    # Bin‑count over the possible range [0, 2·n‑2].
    cnt = np.bincount(sums, minlength=2 * n - 1)
    return cnt.astype(np.int32)


def _remove_element(cnt: np.ndarray, element: int, members: np.ndarray) -> None:
    """
    Decrease the histogram for the removal of ``element`` from the
    current support (``members`` includes the element itself).
    """
    for s in members:
        if s == element:
            cnt[2 * element] -= 1               # (element,element)
        else:
            cnt[element + s] -= 2               # (element,s) and (s,element)


def _add_element(cnt: np.ndarray, element: int, members: np.ndarray) -> None:
    """
    Increase the histogram for the insertion of ``element`` into the
    support (``members`` does *not* contain the element yet).
    """
    for s in members:
        cnt[element + s] += 2                   # (element,s) and (s,element)
    cnt[2 * element] += 1                       # (element,element)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The search works on a binary
    support set of size *m* and uses simulated‑annealing with
    O(m) incremental updates of the pair‑sum histogram.
    """
    rng = np.random.default_rng(seed)

    # ------------------- problem parameters ----------------------------
    n = 4096                           # length of the sequence (power of two)
    m = int(4 * np.sqrt(n))           # support size (≈256 for n=4096)
    if m > n:
        m = n

    # ------------------- initialise a random support ------------------
    support = rng.choice(n, size=m, replace=False)
    support = np.sort(support)        # keep order only for convenience
    in_set = np.zeros(n, dtype=bool)
    in_set[support] = True

    # histogram of ordered pair sums for the initial support
    cnt = _compute_counts(support, n)
    cur_max = int(cnt.max())
    cur_score = 2.0 * n * cur_max / (m * m)

    best_support = support.copy()
    best_max = cur_max
    best_score = cur_score

    print(f"[0.00s] initial max‑count = {cur_max}, score = {cur_score:.6f}")

    # ------------------- simulated annealing --------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    step = 0
    T0 = 2.0                     # initial temperature
    max_steps_est = 2_000_000   # used for linear cooling schedule

    # Convert support to a mutable Python list for easy element removal / addition
    support_list = list(support)

    while time.monotonic() < deadline:
        # linear cooling
        T = max(0.01, T0 * (1.0 - step / max_steps_est))

        # ------------------- propose a swap ---------------------------
        src_idx = int(rng.integers(0, m))          # position in the list
        src = support_list[src_idx]

        # pick a target not currently in the support
        tgt = int(rng.integers(0, n))
        while in_set[tgt]:
            tgt = int(rng.integers(0, n))

        # ---- apply removal of src (in‑place) ----------------------
        _remove_element(cnt, src, np.array(support_list, dtype=np.int32))

        # build the support after removal (list without src)
        support_without_src = support_list.copy()
        support_without_src.pop(src_idx)

        # ---- apply insertion of tgt --------------------------------
        _add_element(cnt, tgt, np.array(support_without_src, dtype=np.int32))

        # evaluate new maximum count
        new_max = int(cnt.max())
        delta = new_max - cur_max

        # ----- acceptance test ------------------------------------
        accept = False
        if delta <= 0:
            accept = True
        else:
            # metropolis criterion
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            # keep the move
            in_set[src] = False
            in_set[tgt] = True
            support_list = support_without_src
            support_list.append(tgt)

            cur_max = new_max
            cur_score = 2.0 * n * cur_max / (m * m)

            if cur_max < best_max:
                best_max = cur_max
                best_score = cur_score
                best_support = np.array(support_list, dtype=np.int32).copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] new best max‑count={best_max}, "

                if best_score <= 1.5025:
                    print("[early stop] target reached")
                    break

            _remove_element(cnt, tgt, np.array(support_without_src, dtype=np.int32))
            _add_element(cnt, src, np.array(support_list, dtype=np.int32))
            # the histogram and support stay as before
        step += 1

        # occasional progress report
        if step % 200_000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] step={step}, best_score={best_score:.6f}")

    # ------------------- build final sequence -------------------------
    a = np.zeros(n, dtype=np.float64)
    a[best_support] = 1.0 / m     # uniform weight on the chosen support

    # final evaluation using the provided evaluator
    final_score = evaluate_sequence(a.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in a]
