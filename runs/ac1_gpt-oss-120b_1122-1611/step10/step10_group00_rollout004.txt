"""
Hybrid multiplicative / additive gradient optimizer with soft‑max smoothing.

* Works on the probability simplex (∑a_i = 1, a_i ≥ 0).  The objective
  J(a) = 2·n·max(conv(a,a)) is invariant to scaling, so we keep a normalized.
* For the current vector we compute:
    – the full auto‑convolution (FFT, O(n log n));
    – an *active* indicator (indices where the convolution is maximal);
    – a soft‑max weight vector w_i ∝ exp(β·conv_i) that focuses on the
      largest entries while remaining smooth.
* Two descent directions are built from these weights:
    – a **multiplicative** (exponentiated‑gradient) step that pushes mass
      away from the active positions;
    – an **additive** step that follows the gradient of the smooth
      max‑approximation (the weighted correlation of w with a).
* Both proposals are projected onto the simplex and evaluated.
  The better one is accepted with a Metropolis rule (linear cooling).
* If no improvement is seen for a while we jitter the current state to
  escape plateaus.
* The algorithm runs for the allotted time (default 1000 s) and returns the
  best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    weight has length (2·n‑1), a has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # the valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (keep FFTs fast)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlations

    # --------------------------------------------------------------
    # 2. Build a diverse palette of initial probability vectors
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – various concentrations (including very sparse)
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best of the initial candidates (fast FFT evaluation)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        max_conv = conv.max()
        J = float(2.0 * n * max_conv)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    total_time = deadline - start_time

    # temperature schedule (linear cooling)
    T_start = best_J * 0.05          # 5 % of the current best
    T_end = 1e-8

    # soft‑max schedule
    beta_start = 1.0
    beta_end = 800.0

    cur_vec = best_vec.copy()
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)
    cur_J = float(2.0 * n * cur_conv.max())

    best_vec = cur_vec.copy()
    best_J = cur_J
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- temperature (linear) -----
        T = (T_start * (deadline - now) + T_end * (now - start_time)) / total_time

        # ----- compute current convolution & objective -----
        cur_conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = cur_conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ----- update global best -----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation handling -----
        if now - last_improve > 20.0:
            # small multiplicative Gaussian jitter
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            cur_J = float(2.0 * n * cur_conv.max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # ----- active set → sub‑gradient (multiplicative update) -----
        tol = max(1e-12, max_conv * 1e-12)
        active_mask = (cur_conv >= max_conv - tol).astype(np.float64)   # length conv_len
        grad_sub = _weighted_gradient(cur_vec, active_mask, fft_len_corr, n)

        # ----- soft‑max weights → smooth gradient (additive update) -----
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total_time)
        # shift to avoid overflow
        w_exp = np.exp(beta * (cur_conv - max_conv))
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w_norm = w_exp / w_sum
        grad_smooth = _weighted_gradient(cur_vec, w_norm, fft_len_corr, n)

        # ----- step sizes (decaying like 1/√t) -----
        step_mul = 0.6 / np.sqrt(iteration + 1.0)
        step_add = 0.3 / np.sqrt(iteration + 1.0)

        # ----- multiplicative proposal -----
        factor = np.exp(-step_mul * grad_sub)
        proposal_mul = cur_vec * factor
        proposal_mul = proj_simplex(proposal_mul)

        # ----- additive proposal (projected gradient descent) -----
        proposal_add = cur_vec - step_add * grad_smooth
        proposal_add = np.maximum(proposal_add, 0.0)   # keep non‑negative before projection
        proposal_add = proj_simplex(proposal_add)

        # ----- evaluate both proposals -----
        conv_mul = _conv_full(proposal_mul, fft_len, conv_len)
        J_mul = float(2.0 * n * conv_mul.max())

        conv_add = _conv_full(proposal_add, fft_len, conv_len)
        J_add = float(2.0 * n * conv_add.max())

        # pick the better proposal for the Metropolis test
        if J_mul < J_add:
            proposal = proposal_mul
            prop_J = J_mul
        else:
            proposal = proposal_add
            prop_J = J_add

        # ----- Metropolis acceptance -----
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return best solution (clip and renormalise for safety)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
