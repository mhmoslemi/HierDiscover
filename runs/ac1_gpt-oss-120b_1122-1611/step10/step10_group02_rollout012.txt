"""
propose_candidate
=================

A three‑stage optimiser that combines a **linear‑programming improvement
step** (the technique described in the state‑of‑the‑art paper) with
exponentiated‑gradient descent, additive sub‑gradient descent and a
simulated‑annealing refinement.

* Phase 0 – initialisation.  Uses the supplied `height_sequence_1` (if any),
  otherwise a uniform random start on a moderate size (`n≈2000`) that keeps
  the LP tractable.

* Phase 1 – LP‑step.  For the current distribution `f` we compute
  `M = max(f*f)`.  We then solve the linear program

        maximise   Σ b_i
        subject to  b_i ≥ 0,
                    (f * b)_k ≤ M  for every convolution index k.

  The optimiser returns a candidate `b`; after renormalising it to a
  probability vector we perform a tiny line‑search on the mixing factor
  `t∈{0.05,…,1.0}` and keep the best convex combination `f←(1‑t)f+tb`.

  Several iterations (≈10–15) are performed while time permits.

* Phase 2 – exponential‑gradient descent on the simplex (the log‑space version
  of the subgradient descent used previously).  Step‑size `η≈1/√it`
  gradually decays.

* Phase 3 – additive projected sub‑gradient descent (classical Euclidean
  projection onto the simplex).

* Phase 4 – simulated‑annealing local search with directed mass‑transfer
  moves and occasional “jitter” (random scaling of a short block).  The
  temperature follows an exponential cooling schedule.

All phases keep track of the best sequence seen so far and stop early as
soon as the target score `1.5025` is reached.  The routine respects the
overall time budget (1000 s minus a 5 s safety margin) and finally returns
the best sequence (clipped to the limits enforced by `evaluate_sequence`).
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of prescribed length."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that A @ b = f * b
    (linear convolution).  Entry A[k,i] = f[k‑i] if the index is valid,
    otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f
    return A


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] with respect to a,
    considering the set of active convolution indices.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Problem size and feasible initial vector
    # --------------------------------------------------------------
    TARGET_SCORE = 1.5025
    MAX_N = 2000                     # keep linear‑program tractable

    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / a0.sum()
        if a0.size > MAX_N:
            # simple truncation – keep the first part and renormalise
            a0 = a0[:MAX_N]
        a = a0.copy()
    else:
        n = MAX_N
        a = rng.random(n)
        a = a / a.sum()

    n = a.size
    fft_len = _next_pow2(2 * n)

    # --------------------------------------------------------------
    # 1) Book‑keeping
    # --------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()          # same as evaluate_sequence for sum=1
    best = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # 2) Phase 1 – linear‑programming improvement (≈15 % of time)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.15 * (deadline - start_time)
    lp_iter = 0
    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        M = conv.max()
        # Build convolution matrix for the current a
        A = _conv_matrix(a)                     # shape (2n‑1, n)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # Linear program: maximise Σ b_i   →   minimise –Σ b_i
        c_obj = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {lp_iter}: solver failed – {res.message}")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b == 0.0:
            break
        g = b_opt / sum_b                     # renormalise to a probability vector

        # tiny line‑search on mixing factor t
        t_candidates = (0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.85, 1.0)
        best_local_a = a
        best_local_score = best_score
        for t in t_candidates:
            cand = (1.0 - t) * a + t * g
            conv_cand = _conv_fft(cand, fft_len)
            score_cand = 2.0 * n * conv_cand.max()
            if score_cand < best_local_score:
                best_local_score = score_cand
                best_local_a = cand

        a = best_local_a
        if best_local_score < best_score - 1e-12:
            best_score = best_local_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] after LP iter {lp_iter} new best = {best_score:.6f}")
            if best_score <= TARGET_SCORE:
                print("[early stop] target reached in LP phase")
                break

        lp_iter += 1

    # --------------------------------------------------------------
    # 3) Phase 2 – exponentiated‑gradient descent (≈30 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    phase2_end = start_time + 0.45 * (deadline - start_time)
    it = 0
    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-8
        active = np.where(conv >= max_b - eps)[0]
        grad = _gradient_from_active(a, active)

        eta = 0.6 / np.sqrt(it + 1.0)            # diminishing step‑size
        log_a = np.log(a + 1e-15)
        log_a -= eta * grad
        a = np.exp(log_a)
        a = _project_onto_simplex(a)

        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (exp‑grad) = {best_score:.6f} (it={it})")
            if best_score <= TARGET_SCORE:
                print("[early stop] target reached in exp‑grad")
                break
        it += 1
        if it % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] exp‑grad it={it}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 3 – additive projected sub‑gradient descent (≈15 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    phase3_end = start_time + 0.65 * (deadline - start_time)
    it = 0
    while time.monotonic() < phase3_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-10
        active = np.where(conv >= max_b - eps)[0]
        grad = _gradient_from_active(a, active)

        step = 0.45 / np.sqrt(it + 1.0)
        cand = a - step * grad
        cand = _project_onto_simplex(cand)

        conv_cand = _conv_fft(cand, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        # accept non‑worsening moves
        if score_cand <= 2.0 * n * max_b:
            a = cand

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (sub‑grad) = {best_score:.6f} (it={it})")
            if best_score <= TARGET_SCORE:
                print("[early stop] target reached in sub‑grad")
                break

        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] sub‑grad it={it}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Phase 4 – simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    curr_score = best_score
    T = 0.012                     # starting temperature
    decay = 0.99985               # exponential cooling
    it = 0
    while time.monotonic() < deadline:
        # directed mass transfer – pick a source with relatively large value
        # and a target with a relatively small one.
        src_candidates = np.where(a > np.percentile(a, 80))[0]
        tgt_candidates = np.where(a < np.percentile(a, 20))[0]
        if src_candidates.size == 0 or tgt_candidates.size == 0:
            src = rng.integers(0, n)
            tgt = rng.integers(0, n)
        else:
            src = rng.choice(src_candidates)
            tgt = rng.choice(tgt_candidates)

        if src == tgt or a[src] <= 1e-15:
            continue

        # multiplicative fraction in [0.001, 0.05] (log‑uniform)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)
        cand = _project_onto_simplex(cand)

        conv_cand = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        Δ = cand_score - curr_score

        # Metropolis acceptance
        if Δ < 0.0 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f} (it={it})")
                if best_score <= TARGET_SCORE:
                    print("[early stop] target reached in SA")
                    break

        # occasional jitter – random scaling of a short block
        if it % 6000 == 0 and it > 0:
            width = rng.integers(1, max(2, n // 80))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            jitter = a.copy()
            jitter[start:start + width] *= factor
            a = _project_onto_simplex(jitter)

        T *= decay
        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA it={it}, T={T:.3e}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
