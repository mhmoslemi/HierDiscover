    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    best_a = None
    best_score = float('inf')
    n = 4096  # default length
    fft_len = _next_pow2(3 * n)

    # define total phases durations as fractions of remaining time
    # we will loop restarts until time exhausted
    while time.monotonic() < deadline:
        # generate random initial a
        if "height_sequence_1" in globals():
            # with probability 0.3 start from known best
            if rng.random() < 0.3:
                a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full_like(a, 1.0 / a.size)
                # If length differs, adjust by truncating or padding
                if a.size != n:
                    if a.size < n:
                        # pad with zeros
                        a = np.pad(a, (0, n - a.size), constant_values=0.0)
                    else:
                        a = a[:n]
            else:
                # random Dirichlet
                a = rng.random(n)
                a = np.abs(a)
                a = a / a.sum()
        else:
            a = np.full(n, 1.0 / n)

        # Phase 1: smooth-max gradient descent
        phase1_time = 0.25 * (deadline - time.monotonic())
        phase1_end = time.monotonic() + phase1_time
        beta_start, beta_end = 10.0, 500.0
        eta0 = 0.8
        it = 0
        while time.monotonic() < phase1_end:
            elapsed = time.monotonic() - start
            frac = (time.monotonic() - (phase1_end - phase1_time)) / phase1_time

            a = _smooth_max_step(a, fft_len, beta_cur, eta)

            it += 1

        # Phase 2: hard-max subgradient descent
        phase2_time = 0.25 * (deadline - time.monotonic())
        phase2_end = time.monotonic() + phase2_time
        it = 0
        while time.monotonic() < phase2_end:
            a_candidate = _hard_max_step(a, fft_len, step_factor=0.3, iteration=it)

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(seq) = 2·n·max(a*a)/(sum(a)^2).
The algorithm combines three complementary stages:

1. **Smoothed‑max mirror descent** – exponentiated‑gradient updates
   on the log‑probabilities using a temperature β that is raised from
   ≈10 to ≈800.  This quickly flattens the convolution while staying
   on the simplex.

2. **Hard‑max sub‑gradient descent with line‑search** – exact
   sub‑gradient of the true max‑convolution is computed; several
   step‑sizes are tried and the best improvement is kept.  This
   sharpens the solution beyond the smoothed approximation.

3. **Stochastic local refinement with simulated annealing** – tiny
   directed mass‑transfer moves (source → target) are performed.
   Moves that increase the objective are occasionally accepted with a
   temperature that decays to zero, allowing the search to escape shallow
   plateaus.  Periodic jitter and explicit symmetrisation help avoid
   local minima.

Multiple random restarts are performed within the 1000 s budget; the
best sequence seen overall is returned.  The code uses only NumPy and
FFT‑based convolutions (O(n log n) per iteration) and respects the
security checks of evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length fft_len."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max_k (a*a)[k] for the active set."""
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Phase‑1 – smoothed‑max exponentiated‑gradient step
# ----------------------------------------------------------------------


def _smooth_max_step(a: np.ndarray,
                     fft_len: int,
                     beta: float,
                     eta: float) -> np.ndarray:
    """One exponentiated‑gradient update for the smoothed max objective."""
    # a is already on the simplex
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    max_conv = conv.max()
    # smooth weights (log‑sum‑exp trick)
    w = np.exp(beta * (conv - max_conv))
    w /= w.sum()                               # ∑ w_i = 1

    # ----- gradient of the smoothed max -----
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: a.size + len(w) - 1]   # full linear conv
    term2 = conv_aw[: a.size]

    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: a.size + len(w_rev) - 1]
    m = len(w)
    term1 = conv_aw_rev[m - 1 - np.arange(a.size)]

    grad = term1 + term2                         # ∂/∂a of weighted sum

    # exponentiated‑gradient update on log‑probabilities
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Phase‑2 – hard‑max sub‑gradient descent with line‑search
# ----------------------------------------------------------------------


def _hard_max_step_line_search(a: np.ndarray,
                               fft_len: int,
                               base_step: float,
                               iteration: int) -> np.ndarray:
    """
    Projected additive step using the exact sub‑gradient.
    Several step‑sizes are tried (0.5×, 1×, 2×) and the best
    improvement is kept.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    # try a few scalings of the step length
    best_candidate = a
    cur_score = 2.0 * a.size * max_b
    best_score = cur_score
    # base step decreases like 1/sqrt(iter+1)
    step_base = base_step / np.sqrt(iteration + 1.0)

    for factor in (0.5, 1.0, 2.0):
        step = factor * step_base
        cand = a - step * grad
        cand = _project_onto_simplex(cand)
        conv_cand = _conv_fft(cand, fft_len)
        score_cand = 2.0 * a.size * conv_cand.max()
        if score_cand < best_score:
            best_score = score_cand
            best_candidate = cand

    return best_candidate


# ----------------------------------------------------------------------
# Phase‑3 – stochastic directed mass‑transfer refinement
# ----------------------------------------------------------------------


def _local_refinement_step(a: np.ndarray,
                           fft_len: int,
                           rng: np.random.Generator,
                           src: int,
                           tgt: int) -> tuple[np.ndarray, int, int]:
    """
    Tiny directed mass‑transfer move.
    Returns (new_sequence, new_src, new_tgt).
    """
    n = a.size
    if a[src] <= 0.0:
        src = int(rng.integers(0, n))
    if src == tgt:
        tgt = int(rng.integers(0, n))

    # transfer fraction in [0.1 % , 5 %] (log‑uniform)
    frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    return a_new, src, tgt

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:

    above, and keeps the best sequence ever observed.
    """

    # Helper to evaluate the hard objective (no clipping)

    while time.monotonic() < deadline:

        if remaining < 10.0:            # not enough time for another full run

            if it % 1200 == 0:
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] local iter={it}, best={best_score:.6f}")

        # after the inner loop we may have already exhausted the budget;
        # the outer while will break accordingly.

    # ------------------------------------------------------------------
    # Return the best sequence discovered (clip to evaluator limits)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best_a.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_a]
```

This implementation respects all constraints, uses only top‑level functions, runs for up to the allotted 1000 s, and combines several optimisation heuristics to explore the search space more aggressively than the previous baseline.
