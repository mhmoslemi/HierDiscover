def golden_section_search(f, a, b, tol, max_iter, time_check):
    best_x = None
    best_f = float('inf')
    c = b - (b - a) / phi
    d = a + (b - a) / phi
    fc = None; fd = None
    for i in range(max_iter):
        if time.time() > time_check(): break
        if fc is None: fc = f(c)
        if fd is None: fd = f(d)
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / phi
            fc = None
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / phi
            fd = None
        # keep track of best
        if fc is not None and fc < best_f:
            best_f = fc; best_x = c
        if fd is not None and fd < best_f:
            best_f = fd; best_x = d
        if b - a < tol:
            break
    return best_x, best_f

"""
Exponential‑decay search for a step‑function.

The objective is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the ratio is scale‑invariant we may normalise the
vector to sum = 1.  For a monotone decreasing exponential
a_i ∝ exp(-λ·i) the convolution has a single dominant peak and
the score depends smoothly on the decay λ.  The algorithm
simply searches the one‑dimensional λ‑space (golden‑section)
using the fast FFT‑based convolution, keeps the best vector,
and then applies a tiny stochastic mass‑transfer refinement.
The whole search respects the 1000 s budget and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Small helper utilities – the same ones used by the reference code
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a via FFT (full length 2*len(a)-1).
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_of_vector(a: np.ndarray, fft_len: int) -> float:
    """
    Return the evaluation score   2·n·max(a*a)   (sum(a)=1).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method builds a one‑parameter
    exponential‑decay family a_i ∝ exp(-λ·i), finds the optimal λ
    by golden‑section search, and finishes with a short stochastic
    refinement that transfers tiny amounts of mass between random
    positions.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Choose a length (use the known construction length if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 4096                     # default power of two – convenient for FFT

    fft_len = _next_pow2(3 * n)      # enough size for a full convolution

    # ------------------------------------------------------------------
    # 1) Time‑budget handling (keep a 5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2) Golden‑section search on the decay parameter λ
    # ------------------------------------------------------------------
    # λ is searched in a wide interval; the optimum is typically around
    # λ ≈ 1 / n (order of 10⁻⁴ for n≈4000).
    lam_low = 1e-6                 # very close to uniform
    lam_high = 5e-3                # fairly concentrated at the left edge

    phi = (1.0 + math.sqrt(5.0)) / 2.0          # golden ratio
    inv_phi = 1.0 / phi
    resphi = 2.0 - phi                         # = 1/phi² ≈ 0.381966

    # initialise the two interior points
    c = lam_high - resphi * (lam_high - lam_low)
    d = lam_low + resphi * (lam_high - lam_low)

    # evaluate the objective for a given λ
    def evaluate_lambda(lam: float) -> tuple[float, np.ndarray]:
        """Return (score, normalised vector) for decay λ."""
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()                     # enforce sum = 1
        score = _score_of_vector(a, fft_len)
        return score, a

    score_c, a_c = evaluate_lambda(c)
    score_d, a_d = evaluate_lambda(d)

    best_score = min(score_c, score_d)
    best_vec = a_c if score_c <= score_d else a_d
    best_lam = c if score_c <= score_d else d

    it = 0
    max_iter = 50                       # more than enough for λ‑tolerance
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        # shrink the interval depending on the function values
        if score_c < score_d:
            lam_high = d
            d = c
            score_d = score_c
            a_d = a_c
            c = lam_high - resphi * (lam_high - lam_low)
            score_c, a_c = evaluate_lambda(c)
        else:
            lam_low = c
            c = d
            score_c = score_d
            a_c = a_d
            d = lam_low + resphi * (lam_high - lam_low)
            score_d, a_d = evaluate_lambda(d)

        # keep track of the best solution seen so far
        if score_c < best_score:
            best_score = score_c
            best_vec = a_c.copy()
            best_lam = c
        if score_d < best_score:
            best_score = score_d
            best_vec = a_d.copy()
            best_lam = d

        # termination condition on interval width (absolute tolerance)
        if lam_high - lam_low < 1e-8:
            break

        # occasional progress report
        if it % 5 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] λ‑search iter={it}, best_score={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] λ‑search finished – best λ={best_lam:.8f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Stochastic directed mass‑transfer refinement (tiny steps)
    # ------------------------------------------------------------------
    a = best_vec.copy()
    n = a.size
    it_refine = 0
    while time.monotonic() < deadline:
        it_refine += 1
        # pick source with non‑zero mass
        src = int(rng.integers(0, n))
        if a[src] <= 0.0:
            # fall back to a random positive index
            positives = np.where(a > 1e-12)[0]
            if positives.size == 0:
                break
            src = int(rng.choice(positives))

        # pick a different target
        tgt = int(rng.integers(0, n))
        while tgt == src:
            tgt = int(rng.integers(0, n))

        # transfer a tiny fraction (log‑uniform between 0.1 % and 2 %)
        frac = math.exp(rng.uniform(math.log(0.001), math.log(0.02)))
        delta = a[src] * frac

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        # keep the simplex constraint (sum stays 1 automatically)
        score_new = _score_of_vector(a_new, fft_len)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_vec = a_new.copy()
            a = a_new
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] refinement iter={it_refine}, new best={best_score:.6f}")

        # accept the move only if it does not increase the score
        if score_new <= _score_of_vector(a, fft_len):
            a = a_new

        # stop early if we have already reached the target bound
        if best_score <= 1.5025:
            print("[early stop] target reached during refinement")
            break

    # ------------------------------------------------------------------
    # 4) Final evaluation with the provided ``evaluate_sequence`` helper
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_vec.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # Return the best sequence (already clipped to the permissible range)
    return [float(x) for x in best_vec]
