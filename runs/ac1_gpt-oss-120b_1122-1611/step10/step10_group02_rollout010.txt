"""
Propose a (near‑optimal) non‑negative step‑function for the max‑convolution
objective.

Algorithm
---------

1.  Multiple *restart lengths* are tried (512, 1024, 2048, 4096 and the length
    of an existing `height_sequence_1` if available).  For each length a fresh
    probability vector is generated.

2.  **Smooth‑max gradient descent**  
    The exact sub‑gradient of a hard max is replaced by a weight vector
    `w_k = exp( λ (c_k – max(c)) )`.  In the limit `λ → ∞` this becomes the
    active‑set gradient, while for moderate `λ` it uses many near‑max
    entries, giving a much more stable direction.  An exponentiated‑gradient
    step (`log‑a ← log‑a – η·∇`) is performed, followed by a Euclidean projection
    onto the probability simplex.  `λ` is slowly increased (up to ≈30) and the
    learning rate `η` follows the usual `η₀/√t` schedule.

3.  **Enhanced simulated‑annealing** – after the gradient phase the best
    vector is refined by a temperature‑controlled SA.  Four move types are used
    (mass transfer, index swap, local block scaling, cyclic permutation) so
    that the search can escape plateaus that trap a pure gradient method.

4.  The global best solution seen across all lengths is kept.  The loop stops
    early if the target score (1.5025) is reached or the time budget (≈1000 s)
    is exhausted.

All helper functions are top‑level, there are no closures or file I/O,
and the routine respects the prescribed time limit.  The final sequence is
clipped to the evaluator’s limits before being returned.
"""

import time
import numpy as np
from math import log, exp

# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length `fft_len`."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_smooth(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Exact gradient of  Σ_k w_k (a*a)[k] with respect to ``a``.
    Implements 2·(w * a_rev) where * is linear convolution and a_rev is a reversed.
    """
    a_rev = a[::-1]
    w_fft = np.fft.rfft(w, n=fft_len)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)
    # Linear convolution length = len(w)+len(a)-1 = 3*len(a)-2
    L = len(w) + len(a) - 1
    full = full[:L]                     # discard trailing padding
    start = len(a) - 1                  # index corresponding to shift 0
    grad = 2.0 * full[start:start + len(a)]
    return grad

                # random fraction in [0.001, 0.05] on a log‑scale
                frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
                delta = a[src] * frac
                cand[src] -= delta
                cand[tgt] += delta
            elif move == "swap":
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    cand[i], cand[j] = cand[j], cand[i]
            elif move == "scale":
                width = rng.integers(1, max(2, n // 20))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.028))
                cand[start:start + width] *= factor
            elif move == "rotate":
                shift = rng.integers(1, n)
                cand = np.roll(cand, shift)

            # re‑project onto the simplex (keeps sum = 1)
            cand = np.clip(cand, 0.0, None)
            cand = _project_onto_simplex(cand)

            conv_cand = _conv_fft(cand, fft_len_conv)
            cand_score = 2.0 * n * conv_cand.max()
            delta = cand_score - curr_score

            # Metropolis acceptance
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                a = cand
                curr_score = cand_score
                if cand_score < local_best_score - 1e-12:
                    local_best_score = cand_score
                    local_best_vec = cand.copy()
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_vec = cand.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] new global best (SA n={n}) = {best_score:.6f}")

            # cooling
            T *= decay
            sa_iters += 1
            if sa_iters % 2500 == 0:
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] n={n} SA it={sa_iters}, T={T:.3e}, best={best_score:.6f}")

        # early exit if the target is already met
        if best_score <= 1.5025:
            print(f"[{time.monotonic() - start_time:.2f}s] target reached (score={best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # 2) Return the best construction found (clip to evaluator limits)
    # ------------------------------------------------------------------
    if best_vec is None:
        # fallback – a plain uniform distribution (never happens in practice)
        n = 1024
        best_vec = np.full(n, 1.0 / n)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    final_score = evaluate_sequence(best_vec.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_vec]
