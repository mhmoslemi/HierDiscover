    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0) # safety margin

    # get initial sequence
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0, None)
        if a.sum() == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a.sum()
        n = a.size
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # compute initial score using fast conv
    conv = _conv_fft(a, _next_pow2(3 * n))
    best_score = 2.0 * n * conv.max()
    best = a.copy()
    print(f"[{0.0:.2f}s] initial best = {best_score:.6f}")

    # iteration loop
    iter = 0
    while time.monotonic() < deadline:
        # compute M
        conv = _conv_fft(a, _next_pow2(3 * n))
        M = conv.max()
        # solve LP to get b
        b = _solve_lp_max_sum(a, M, rng)  # function returns array or None
        if b is None:
            # fallback: break or random move
            break

        # normalize b and combine
        b_norm = b / b.sum()
        # try a few mixture fractions
        for t in (0.1, 0.25, 0.5):
            cand = (1 - t) * a + t * b_norm
            cand = np.clip(cand, 0, None)
            cand = cand / cand.sum()
            cand_score = 2.0 * n * _conv_fft(cand, _next_pow2(3 * n)).max()
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] new best (t={t}) = {best_score:.6f}")
                a = cand  # move forward
                break  # exit t loop after improvement
        else:
            # no improvement; maybe accept cand with minimal increase to continue progress
            # we can still set a = cand with smallest score among t values
            # compute scores for all t
            cand_scores = []
            cand_candidates = []
            for t in (0.1, 0.25, 0.5):
                cand = (1 - t) * a + t * b_norm

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.
Algorithm
---------
1. Initialise from the previous best (global variable ``height_sequence_1``)
   or from a uniform distribution.
2. Repeatedly improve the current sequence by a Frank–Wolfe style step:
   * Compute M = max (a*a) (the current worst convolution value).
   * Solve a small linear programme that maximises the total mass
     Σb_i subject to the *active* convolution constraints
     (f*b)[k] ≤ M for those k where (f*f)[k] is close to M.
   * Scale the LP‑solution so that all convolution constraints are obeyed.
   * Form a convex combination (1‑t)·a + t·b̂ and keep the best mixture.
3. When the time budget is almost exhausted perform a cheap stochastic
   local‑refinement (tiny directed mass transfers) to escape possible
   plateaus.
4. Return the best sequence found (clipped to the limits of the evaluator).

The LP is solved with ``scipy.optimize.linprog`` on a handful of
constraints, which makes each iteration fast while still guaranteeing
feasibility of the full convolution bound.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT of length ``fft_len``.
    Returns the convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Convenient wrapper that returns max(a*a)."""
    return float(_conv_fft(a, fft_len).max())


def _active_indices(conv: np.ndarray, rel_tol: float = 1e-6, max_act: int = 200) -> np.ndarray:
    """
    Return indices where ``conv`` is within ``rel_tol`` relative to its maximum.
    The result is truncated to at most ``max_act`` indices (randomly sampled
    if there are more).
    """
    max_val = conv.max()
    if max_val == 0:
        return np.arange(conv.size, dtype=int)
    active = np.where(conv >= max_val - rel_tol * max_val)[0]
    if active.size > max_act:
        rng = np.random.default_rng()
        active = rng.choice(active, size=max_act, replace=False)
    return active


def _solve_lp_active(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve a small linear programme:
        maximise   Σ b_i
        subject to (f * b)[k] ≤ M   for k in ``active_idx``
                    b_i ≥ 0.
    Returns the optimal b (may be None if the solver fails).
    """
    n = f.size
    m = active_idx.size
    if m == 0:
        return None

    # build the (small) constraint matrix A (shape m × n)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        A[row, i] = f[k - i]

    # linprog minimises c·x, so we minimise -1·b to maximise Σb
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(m, M, dtype=np.float64),
                  bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set of active indices.
    For each active index k we add a_{k-i} to ∂/∂a_i.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad

            tgt = int(rng.integers(0, n))

        frac = np.exp(rng.uniform(np.log(0.0005), np.log(0.02)))   # 0.05 %–2 %
        delta = a[src] * frac
        a[src] -= delta
        a[tgt] += delta
        a /= a.sum()

        cand_score = 2.0 * n * _max_convolution(a, fft_len)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached in local phase")
                break

        it_loc += 1
        if it_loc % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local iter={it_loc}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
