def _split_move(a):
    # choose index with large mass
    idx = int(np.argmax(a))
    mass = a[idx]
    if mass < 1e-6:
        return a.copy()
    # choose shift distance uniformly from 5 to n//4 (or ensure within bounds)
    max_shift = max(5, n // 10)
    shift = rng.integers(5, max_shift)
    # compute new positions
    left = idx - shift
    right = idx + shift
    if left < 0 or right >= n:
        return a.copy()
    # allocate fractions
    keep_frac = rng.uniform(0.2, 0.5)
    left_frac = (1 - keep_frac) * rng.uniform(0.4, 0.6)
    right_frac = (1 - keep_frac) - left_frac
    cand = a.copy()
    cand[idx] = mass * keep_frac
    cand[left] += mass * left_frac
    cand[right] += mass * right_frac
    cand = _simplex_normalise(cand)
    return cand
```

def _lp_step(a):
    conv = _conv_self(a, fft_len)
    M = conv.max()
    n = a.size
    # select top K indices
    K = min(40, conv.size)
    # indices sorted descending by conv value
    idxs = np.argpartition(-conv, range(K))[:K]   # unsorted but delim.
    # Build A_ub
    rows = []
    for k in idxs:
        # compute row: for j in [max(0,k-(n-1)) ... min(n-1,k)]
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        # length: end-start+1
        row = np.zeros(n)
        # indeces j start..end inclusive; coefficient for g_j is a[k-j]
        js = np.arange(start, end+1)
        row[js] = a[k - js]
        rows.append(row)
    if not rows:
        return None
    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M)
    c = -np.ones(n)   # maximize sum(g) => minimize -sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        if g.sum() > 1e-12:
            g_norm = g / g.sum()
            return g_norm
    return None
```

def _split_move(a, rng):
    idx = int(np.argmax(a))
    mass = a[idx]
    if mass < 1e-9:
        return a.copy()
    # choose shift distance
    max_shift = max(2, a.size // 10)
    shift = rng.integers(2, max_shift)
    left = idx - shift
    right = idx + shift
    # ensure left/right within bounds
    if left < 0 or right >= a.size:
        return a.copy()
    keep_frac = rng.uniform(0.2, 0.6)
    remaining = 1.0 - keep_frac

    right_frac = remaining - left_frac
    cand = a.copy()
    cand[idx] = mass * keep_frac
    cand[left] += mass * left_frac
    cand[right] += mass * right_frac

"""
Propose a step‑function (non‑negative simplex vector) with a low max‑convolution
score.

Algorithm
---------
1. **Initialisation** – start from the provided `height_sequence_1` if available,
   otherwise from a uniform distribution.  The vector is padded to the next power
   of two (nice for FFT) and normalised to unit L1‑norm (the objective is
   scale‑invariant).

2. **Phase 1 – Adam‑accelerated exponentiated‑gradient**.  
   We minimise a smooth surrogate of the max‑convolution:
   `L(a)= (1/β)·log Σ_k exp(β·(a*a)[k])`.  
   The gradient of `L` is obtained by one FFT‑based convolution with the
   soft‑max weights `w`.  Adam’s first/second moments are used and an
   *exponentiated* (multiplicative) update keeps the iterates on the simplex.
   The smoothing parameter `β` is increased slowly.

3. **Phase 2 – Linear‑programming refinement**.  
   From the current iterate `f` we compute `M = max(f*f)`.  A small LP
   (≤30 active convolution constraints) finds a feasible vector `g` with the
   same bound `f·g ≤ M` but maximal L1‑mass.  The pair `(f,g)` is mixed by a
   short line search (`t∈{0.1,…,0.9}`) and the best mixture is kept.

4. **Phase 3 – Aggressive local moves**.  
   Hill‑climbing with two kinds of moves:
   *pairwise mass transfer* (source = index with largest gradient of the
   active convolution entry, target = smallest) and *splitting* of a heavy
   atom into two neighbours.  A simple simulated‑annealing acceptance rule
   allows occasional uphill steps.

All phases respect the overall time budget (default 1000 s) with a 5 s safety
margin and keep the best sequence seen so far.  The final sequence is clipped
to the evaluator’s limits and returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (full length 2·len(a)‑1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the convolution entry (a*a)[idx].
    For each i the component is a[idx‑i] (zero when out of range).
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)          # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    return grad


def _smooth_grad_and_weights(a: np.ndarray, beta: float, fft_len: int):
    """
    Return (conv, weights, gradient) for the log‑sum‑exp surrogate.
    `conv` is the full self‑convolution, `weights` are the soft‑max probabilities
    and `gradient` is 2·(a * weights) (the factor 2 comes from d(a*a)/da).
    """
    a_fft = np.fft.rfft(a, n=fft_len)

    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]

    max_c = conv.max()
    # stable soft‑max
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()

    w_fft = np.fft.rfft(w, n=fft_len)
    grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: a.size + w.size - 1]
    grad = 2.0 * grad_full[: a.size]   # exact sub‑gradient factor 2
    return conv, w, grad


def _lp_improve(a: np.ndarray, M: float, top_k: int, fft_len: int):
    """
    Solve a tiny linear program:
        maximise  Σ g_j
        s.t.     (a * g)[k] ≤ M   for the `top_k` largest convolution entries
                 g ≥ 0
    Return a feasible vector `g` normalised to unit sum, or ``None`` if the LP
    fails or yields an all‑zero solution.
    """
    n = a.size
    conv = _conv_self(a, fft_len)
    # indices of the `top_k` biggest convolution values
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, np.arange(top_k))[:top_k]

    rows = []
    for k in idxs:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        js = np.arange(start, end + 1, dtype=int)
        row = np.zeros(n, dtype=np.float64)
        row[js] = a[k - js]                     # coefficient a_{k‑j}
        rows.append(row)

    if not rows:
        return None

    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)           # maximise Σ g_j  ⇔  minimise −Σ g_j
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    g = res.x
    if g.sum() <= 1e-12:
        return None
    return g / g.sum()                          # unit sum


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Load / initialise a starting distribution (unit L1‑norm)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
    else:
        # default to a modest uniform vector
        base = np.full(3809, 1.0 / 3809, dtype=np.float64)

    # pad to the next power of two (nice for FFT) – keep existing entries
    target_len = _next_pow2(base.size)
    if target_len != base.size:
        a = np.zeros(target_len, dtype=np.float64)
        a[: base.size] = base
    else:
        a = base.copy()

    a = _simplex_normalise(a)
    n = a.size
    fft_len = _next_pow2(3 * n)           # enough for all convolutions

    # ------------------------------------------------------------------
    # Helper to compute the exact evaluation score (sum(a)=1 ⇒ score=2·n·max)
    # ------------------------------------------------------------------
    def _score_from_conv(conv_arr: np.ndarray) -> float:
        return float(2.0 * n * conv_arr.max())

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety

    conv = _conv_self(a, fft_len)
    best_a = a.copy()
    best_score = _score_from_conv(conv)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam‑accelerated exponentiated‑gradient (≈35 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start + 0.35 * (deadline - start)

    beta = 5.0
    beta_max = 250.0
    beta_factor = 1.0018

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eta0 = 0.5                                   # base learning rate
    it = 0

    while time.monotonic() < phase1_end:
        conv, w, grad = _smooth_grad_and_weights(a, beta, fft_len)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        eta = eta0 / math.sqrt(it + 1.0)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a * np.exp(-step)                     # multiplicative EG update
        a = _simplex_normalise(a)

        # score
        score = _score_from_conv(conv)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (EG)  = {best_score:.6f} (it={it})")
            if best_score <= 1.5025:
                print("[early stop] target reached in EG phase")
                break

        # schedule
        beta = min(beta_max, beta * beta_factor)
        it += 1

        # occasional progress report
        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] EG iter={it}, β={beta:.1f}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming refinement (≈20 % of budget)
    # ------------------------------------------------------------------
    a = best_a.copy()
    phase2_end = start + 0.65 * (deadline - start)

    # We will attempt at most 5 LP refinements (they are comparatively expensive)
    lp_attempts = 0
    max_lp_attempts = 5
    while time.monotonic() < phase2_end and lp_attempts < max_lp_attempts:
        conv = _conv_self(a, fft_len)
        M = conv.max()
        g = _lp_improve(a, M, top_k=30, fft_len=fft_len)
        if g is None:
            break

        improved = False
        # line‑search over mixing coefficient t
        for t in np.linspace(0.1, 0.9, 9):
            cand = (1.0 - t) * a + t * g
            cand = _simplex_normalise(cand)
            conv_cand = _conv_self(cand, fft_len)
            score_cand = _score_from_conv(conv_cand)
            if score_cand < best_score - 1e-12:
                best_score = score_cand
                best_a = cand.copy()
                a = cand
                improved = True
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] LP mix t={t:.2f} → {best_score:.6f}")
                break
        if not improved:
            # if no improvement, still accept the mixture with smallest score
            # (helps escape shallow plateaus)
            t = 0.5
            cand = (1.0 - t) * a + t * g
            cand = _simplex_normalise(cand)
            conv_cand = _conv_self(cand, fft_len)
            score_cand = _score_from_conv(conv_cand)
            if score_cand < best_score:
                best_score = score_cand
                best_a = cand.copy()
                a = cand
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] LP fallback mix t={t:.2f} → {best_score:.6f}")

        lp_attempts += 1
        # small pause to avoid hogging the CPU
        if time.monotonic() < phase2_end and lp_attempts < max_lp_attempts:
            time.sleep(0.01)

    # ------------------------------------------------------------------
    # Phase 3 – Aggressive local moves + simulated annealing (remaining time)
    # ------------------------------------------------------------------
    a = best_a.copy()
    phase3_end = deadline
    it3 = 0
    temp_start = 0.02 * best_score

    while time.monotonic() < phase3_end:
        # current convolution and score
        conv = _conv_self(a, fft_len)
        curr_score = _score_from_conv(conv)

        # decide which move to make
        if rng.random() < 0.7:
            # ----- pairwise mass transfer -----
            max_idx = int(conv.argmax())
            grad_max = _gradient_for_index(a, max_idx)
            src = int(np.argmax(grad_max))
            tgt = int(np.argmin(grad_max))
            # log‑uniform fraction between 0.1 % and 2 %
            frac = np.exp(rng.uniform(np.log(0.001), np.log(0.02)))
            delta = a[src] * frac
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
        else:
            # ----- split a heavy atom into two neighbours -----
            idx = int(np.argmax(a))
            mass = a[idx]
            if mass < 1e-9:
                cand = a.copy()
            else:
                max_shift = max(2, n // 10)
                shift = rng.integers(2, max_shift)
                left = idx - shift
                right = idx + shift
                if left >= 0 and right < n:
                    keep_frac = rng.uniform(0.2, 0.6)
                    remaining = 1.0 - keep_frac
                    left_frac = remaining * rng.uniform(0.3, 0.7)
                    right_frac = remaining - left_frac
                    cand = a.copy()
                    cand[idx] = mass * keep_frac
                    cand[left] += mass * left_frac
                    cand[right] += mass * right_frac
                else:
                    cand = a.copy()

        # occasional jitter to keep the search fluid
        if rng.random() < 0.07:
            jitter = rng.normal(scale=0.0015, size=n)
            cand = _simplex_normalise(np.clip(cand + jitter, 0.0, None))

        cand = _simplex_normalise(cand)

        # evaluate candidate
        conv_cand = _conv_self(cand, fft_len)
        score_cand = _score_from_conv(conv_cand)

        # record best solution
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (it={it3})")
            if best_score <= 1.5025:
                print("[early stop] target reached in local phase")
                break

        # simulated‑annealing style acceptance
        fraction = (time.monotonic() - start) / (deadline - start)
        temperature = temp_start * (1.0 - fraction)
        if score_cand < curr_score or rng.random() < math.exp(
                (curr_score - score_cand) / max(1e-12, temperature)):
            a = cand

        it3 += 1

        # occasional reporting
        if it3 % 4000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] local iter={it3}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (clip to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
