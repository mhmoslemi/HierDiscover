"""
Advanced soft‑max based optimisation for the step‑function challenge.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) and
minimises a smooth approximation of the objective

    2 * n * max(a * a) / (∑a)²  =  2 * n * max(conv)

by replacing the max with a log‑sum‑exp (soft‑max) that becomes tighter
as a temperature parameter γ grows.  For a given γ the gradient can be
computed exactly using only a few FFTs, allowing fast Adam‑style updates.
γ is increased gradually so the optimisation first explores a smooth
landscape and then concentrates on the true max.

Each run consists of a fast Adam phase, occasional jitter to escape
plateaus and a final short stochastic “mass‑transfer’’ refinement.
Multiple random‑restart runs with different lengths are performed until
the global time budget (≈1000 s) is exhausted.  The best sequence found
is returned and made available as a global variable for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric helpers – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max objective and its exact gradient
# ----------------------------------------------------------------------


def _softmax_obj_and_grad(a: np.ndarray, gamma: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(a*a) and its gradient.
    The approximation is (1/γ)·log∑exp(γ·conv) where conv = a*a.
    Gradient = (1/γ)·( (w*a) + (rev(w)*a) )   where w_i ∝ exp(γ·conv_i).
    All convolutions are performed via FFT.
    """
    n = a.size
    # Use a single FFT length that comfortably fits a*a (2n‑1) and w*a (3n‑2).
    L = _next_pow2(3 * n)

    # FFT of a (same length used for all convolutions in this call)
    a_fft = np.fft.rfft(a, n=L)

    # ----- a * a -------------------------------------------------------
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]

    # ----- soft‑max weights --------------------------------------------
    maxc = conv_full.max()
    # scaled exponents are non‑positive → no overflow
    exp_vals = np.exp(gamma * (conv_full - maxc))
    Z = exp_vals.sum()
    w = exp_vals / Z                     # Σ w_i = 1

    # ----- w * a -------------------------------------------------------
    w_fft = np.fft.rfft(w, n=L)
    conv_w_full = np.fft.irfft(w_fft * a_fft, n=L)[: n + len(w) - 1]

    # ----- rev(w) * a --------------------------------------------------
    rev_w_fft = np.fft.rfft(w[::-1], n=L)
    conv_rev_w_full = np.fft.irfft(rev_w_fft * a_fft, n=L)[: n + len(w) - 1]

    # Gradient for the (1/γ)·log‑sum‑exp objective
    grad = (conv_w_full[:n] + conv_rev_w_full[:n]) / gamma

    # Smoothed max value (objective we are actually minimising)
    obj = maxc + np.log(Z) / gamma
    return float(obj), grad


# ----------------------------------------------------------------------
# Optimisation of a single start point
# ----------------------------------------------------------------------


def _optimize_one_run(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Adam optimiser on the soft‑max surrogate, occasional jitter and a short
    stochastic refinement.  Returns the best sequence (as a NumPy array) and
    its exact score 2·n·max(conv) (sum(a)=1).
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = 2.0 * n * _conv_fft(a, _next_pow2(2 * n - 1)).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    start = time.monotonic()
    deadline = start + time_budget

    it = 0
    gamma = 5.0                # start with a smooth surrogate
    gamma_inc_interval = 2000  # increase γ every this many iterations

    while time.monotonic() < deadline:
        # ---- anneal γ -------------------------------------------------
        if it % gamma_inc_interval == 0 and it > 0:
            gamma = min(gamma * 1.5, 200.0)

        # ---- compute surrogate objective and gradient -----------------
        obj, grad = _softmax_obj_and_grad(a, gamma)

        # ---- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step size that slowly decays; tuned for stability
        lr = 0.5 / math.sqrt(it + 1.0)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex (keeps sum=1, non‑negative)
        a = _project_onto_simplex(a)

        # ---- occasional jitter to shake out of plateaus ---------------
        if it % 500 == 0 and it > 0:
            width = max(1, n // 120)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a)

        # ---- cheap score check (every few iterations) -----------------
        if it % 250 == 0:
            conv = _conv_fft(a, _next_pow2(2 * n - 1))
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

        it += 1

    # ------------------------------------------------------------------
    # Short stochastic refinement (uses the exact objective)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        src = int(rng.integers(0, n))
        tgt = int(rng.integers(0, n))
        if src == tgt:
            continue
        # log‑uniform fraction in [0.1 % , 12 %]
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.12)))
        delta = a[src] * frac
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        conv = _conv_fft(cand, _next_pow2(2 * n - 1))
        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            a = cand.copy()
        else:
            # tiny simulated‑annealing acceptance to avoid dead‑ends
            temperature = max(1e-4, (deadline - time.monotonic()) /
                             (deadline - start))
            if rng.random() < math.exp(-(cand_score - best_score) / temperature):
                a = cand.copy()

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Multiple random‑restart runs are performed,
    each using the soft‑max Adam optimiser described above.  The best
    sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin so the harness does not cut us off
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise the global best – reuse previous construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        if best.sum() > 0.0:
            best = best / best.sum()
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        # start from a modestly sized uniform distribution
        best = np.full(2048, 1.0 / 2048, dtype=np.float64)

    best_score = 2.0 * best.size * _conv_fft(best,
                                            _next_pow2(2 * best.size - 1)).max()
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # ------------------------------------------------------------------
    # Lengths to explore (powers of two are cheap for the FFT)
    # ------------------------------------------------------------------
    length_pool = [1024, 2048, 4096, 6144, 8192]

    # each restart gets at least this much time (adjustable)
    min_run_time = 20.0

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # Pick a length – bias towards the current best size
        # --------------------------------------------------------------
        if rng.random() < 0.6:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # --------------------------------------------------------------
        # Build an initial point for this run
        # --------------------------------------------------------------
        if n == best.size:
            # jitter the current best to leave the current basin of attraction
            a0 = best.copy()
            noise = rng.normal(scale=0.008, size=n)
            a0 = np.maximum(a0 + noise, 0.0)
            a0 = _project_onto_simplex(a0)
        else:
            # moderately sparse Dirichlet start (α < 1 favours sparsity)
            alpha = np.full(n, 0.5)
            a0 = rng.dirichlet(alpha).astype(np.float64)

        # --------------------------------------------------------------
        # Run a short optimisation on this start
        # --------------------------------------------------------------
        remaining = deadline - time.monotonic()
        run_budget = min(min_run_time, remaining)
        if run_budget < 1.0:
            break

        a_opt, score_opt = _optimize_one_run(a0, run_budget, rng)

        # --------------------------------------------------------------
        # Update the global best if improved
        # --------------------------------------------------------------
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # Final processing – enforce evaluator limits and compute the exact score
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # evaluator caps values anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # store for a possible next call
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
