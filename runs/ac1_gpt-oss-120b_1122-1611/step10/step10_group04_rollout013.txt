def propose_candidate(seed=..., budget_s=1000.0):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 5.0

    # Choose length n
    n = 4096
    a = np.full(n, 1.0/n)
    fft_len = 1 << ((2*n - 2).bit_length())
    # Define function to compute score
    def compute_score(arr):
        conv = np.fft.irfft(np.fft.rfft(arr, fft_len) * np.fft.rfft(arr, fft_len), fft_len)[:2*n-1]
        maxb = conv.max()
        return 2.0 * n * maxb, maxb, conv

    score, maxb, conv = compute_score(a)
    best_a = a.copy()
    best_score = score

    # Simulated annealing parameters
    T0 = 0.01  # initial temperature (scale relative to objective difference)
    T_end = 1e-6
    total_iter_est = 0
    iter_counter = 0

    while time.monotonic() < deadline:
        # compute temperature based on progress
        elapsed = time.monotonic() - start
        progress = elapsed / (deadline - start)
        T = T0 * ( (T_end/T0) ** progress )  # exponential cooling

        # Choose src and tgt
        # Compute gradient from active (active set indices of max conv)
        # For efficiency compute infrequently
        if iter_counter % 10 == 0:
            maxb = conv.max()
            eps = maxb * 1e-6
            active = np.where(conv >= maxb - eps)[0]
            grad = _gradient_from_active(a, active)
            # Choose src with largest grad, tgt with smallest grad
            src = int(np.argmax(grad))
            # Pick tgt among indices with low gradient (below 25th percentile)
            thresh = np.percentile(grad, 25)
            low_idx = np.where(grad <= thresh)[0]
            if low_idx.size > 0:

            else:

        # Transfer fraction

        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.3)))

"""
Simulated‑annealing optimiser for the max‑convolution objective.

* initialise from the best known sequence (if any) or from the uniform
  distribution.
* repeatedly perform a directed mass‑transfer move (source → target) with
  a log‑uniform step size.
* accept every improving move; otherwise accept with probability
  exp(−Δ/temperature). Temperature decays exponentially over the budget.
* the source is chosen from the indices that have the largest
  sub‑gradient of the current max‑convolution (computed from the active
  set). The target is taken among indices with the smallest gradient,
  which tends to flatten the convolution.
* after a long streak without improvement a random restart is performed.
* the best sequence seen is returned and finally evaluated with the
  provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the active set
    (indices attaining the current maximum).  For each active k the
    contribution to ∂max/∂a_i is a_{k‑i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)  # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score_and_conv(a: np.ndarray, fft_len: int):
    """
    Return the objective value 2·n·max(conv) together with the full
    convolution ``conv = a*a`` (linear, not circular).  The sum of ``a``
    is assumed to be 1, which makes the denominator of the official
    evaluator equal to 1.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    n = a.size
    return 2.0 * n * max_b, conv

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  A simple simulated‑annealing loop is used.  The
    routine respects the 1000 s wall‑clock budget and prints occasional
    progress information.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) initialise – use previously found construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            a /= total
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # start with a modest power‑of‑two length – 4096 works well in practice
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) compute initial score
    # ------------------------------------------------------------------
    cur_score, cur_conv = _score_and_conv(a, fft_len)
    best_a = a.copy()
    best_score = cur_score
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) annealing schedule – exponential cooling
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # safety margin
    # temperature parameters (tuned experimentally)
    T_start = 5e-3
    T_end = 5e-8

    # we keep a few statistics to trigger random restarts
    last_improve = 0
    iter_cnt = 0
    grad = None
    src = tgt = None

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        prog = elapsed / (deadline - start)  # 0 … 1
        T = T_start * ((T_end / T_start) ** prog)   # exponential

        # ------------------------------------------------------------------
        # 3) choose source / target indices
        # ------------------------------------------------------------------
        if iter_cnt % 10 == 0 or grad is None:
            max_b = cur_conv.max()
            eps = max_b * 1e-7
            active = np.where(cur_conv >= max_b - eps)[0]
            grad = _gradient_from_active(a, active)

            # source = index with largest gradient (most responsible for the peak)
            src = int(np.argmax(grad))

            # target = index belonging to the lower 20 % of the gradient values
            thresh = np.percentile(grad, 20)
            low_idx = np.where(grad <= thresh)[0]
            if low_idx.size:
                tgt = int(rng.choice(low_idx))
            else:
                tgt = int(rng.integers(0, n))

        # ------------------------------------------------------------------
        # 4) propose a directed mass‑transfer move
        # ------------------------------------------------------------------
        # log‑uniform fraction in [1 e‑3 , 2 e‑1] – occasionally very aggressive
        frac = np.exp(rng.uniform(np.log(1e-3), np.log(2e-1)))
        delta = a[src] * frac
        # avoid negative source due to rounding
        delta = min(delta, a[src])

        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta

        # --------------------------------------------------------------
        # 5) evaluate candidate
        # --------------------------------------------------------------
        cand_score, cand_conv = _score_and_conv(a_cand, fft_len)
        Δ = cand_score - cur_score

        accept = False
        if Δ <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-Δ / T):
                accept = True

        if accept:
            a = a_cand
            cur_score = cand_score
            cur_conv = cand_conv

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = iter_cnt
                print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

                if best_score <= 1.5025:
                    print("[early stop] target reached")
                    break

        # ------------------------------------------------------------------
        # 6) occasional random restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if iter_cnt - last_improve > 200_000:
            # draw a fresh Dirichlet vector – this respects the simplex
            a = rng.gamma(shape=1.0, scale=1.0, size=n)
            a /= a.sum()
            cur_score, cur_conv = _score_and_conv(a, fft_len)
            grad = None
            last_improve = iter_cnt
            print(f"[{elapsed:6.2f}s] random restart (iter={iter_cnt})")

        # ------------------------------------------------------------------
        # 7) occasional jitter to kick the system out of tiny plateaus
        # ------------------------------------------------------------------
        if iter_cnt % 50_000 == 0 and iter_cnt > 0:
            width = int(rng.integers(1, max(2, n // 100)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.015))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a_jit = np.maximum(a_jit, 0.0)
            a_jit /= a_jit.sum()
            a = a_jit
            cur_score, cur_conv = _score_and_conv(a, fft_len)
            grad = None
            print(f"[{elapsed:6.2f}s] jitter applied")

        iter_cnt += 1

        # occasional log
        if iter_cnt % 30_000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] iter={iter_cnt}, best={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 8) final evaluation with the reference routine
    # ----------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    # return as plain Python list of floats
    return [float(x) for x in best]

            cur_score, cur_conv = _score_and_conv(a, fft_len)
            grad = None
            print(f"[{elapsed:6.2f}s] jitter applied")

        iter_cnt += 1

        if iter_cnt % 30_000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] iter={iter_cnt}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final evaluation with the provided evaluator
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
