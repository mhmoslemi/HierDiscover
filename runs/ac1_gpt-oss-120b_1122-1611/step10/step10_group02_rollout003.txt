The algorithm works on the probability simplex (∑a_i = 1) by optimising

smooth approximation

    f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k]),   α ≥ 0

entirely in the frequency domain (FFT).  α is ramped from a small value

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2·len(a)‑1.
    """
    a_fft = rfft(a, n=fft_len)
    conv = irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:

    Y = rfft(y, n=fft_len)
    conv = irfft(X * Y, n=fft_len)[: x.size + y.size - 1]
    return conv

    # --------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        # ----- 1) current simplex point -------------------------------------------------
        a = _stable_softmax(z)                         # a_i ≥ 0, Σa_i = 1

        # ----- 2) convolution a*a -------------------------------------------------------
        fft_len = _next_pow2(2 * n)
        conv = _conv_fft(a, fft_len)                   # length 2·n‑1

        # ----- 3) smooth maximum (log‑sum‑exp) -----------------------------------------
        # α grows linearly with elapsed time
        elapsed_frac = (time.monotonic() - start_time) / (deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * elapsed_frac

        # stable softmax over α·conv
        conv_max = np.max(conv)
        exp_alpha = np.exp(alpha * (conv - conv_max))  # ≤ 1
        w = exp_alpha / exp_alpha.sum()                # w_k ≥ 0, Σw_k = 1

        # ----- 4) gradient of the smooth max w.r.t a ------------------------------------
        # grad_a_i = 2 * Σ_j w_{i+j} * a_j
        # This is a convolution of w (length 2n‑1) with a_rev (length n)
        a_rev = a[::-1]
        conv_len_w = _next_pow2(len(w) + len(a_rev) - 1)
        conv_w_a_rev = _conv_fft_two(w, a_rev, conv_len_w)          # length 3n‑2
        grad_a = 2.0 * conv_w_a_rev[n - 1 : n - 1 + n]               # slice of length n

        # ----- 5) chain rule: convert gradient w.r.t a into gradient w.r.t z ------------
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)                                 # ∂f/∂z

        # ----- 6) Adam update -----------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr / (np.sqrt(v_hat) + eps)
        z -= step * m_hat

        # occasional jitter to avoid stagnation
        if it % 2000 == 0:
            z += rng.normal(scale=0.08, size=z.shape)

        # ----- 7) periodic evaluation ----------------------------------------------------
        if it % 200 == 0:
            _maybe_update_best(a)
            if best_score <= target_score:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (clip to evaluator limits)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = _stable_softmax(z)   # fallback, should never happen
        best_score = evaluate_sequence(best_a.tolist())

    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best_a.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_a]
