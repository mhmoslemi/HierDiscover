def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5  # safety

    # parameters
    n = 2048
    fft_len = _next_pow2(3*n)

    # state for best
    best_a = None
    best_score = float('inf')

    # define function to compute score from conv
    def score_from_a(a):
        conv = _conv_self(a, fft_len)
        return 2.0 * n * conv.max()

    # main loop for restarts
    while time.monotonic() < deadline - 10:   # keep window for final refine
        # random init
        # Dirichlet concentration param 0.5 (sparse) maybe tune
        concentration = rng.uniform(0.1, 2.0)  # random each restart
        # sample gamma
        x = rng.gamma(concentration, 1.0, size=n)
        a = _simplex_normalise(x.astype(np.float64))

        # Adaptive schedule for gradient descent
        max_iters = 4000
        # Adam params
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        beta1_t = beta1
        beta2_t = beta2
        # smoothing param
        beta = 1.0
        beta_inc = 1.02
        max_beta = 80.0
        # learning rate schedule
        lr0 = 0.1

        for it in range(max_iters):
            if time.monotonic() >= deadline - 5:
                break

            # compute conv (self)
            conv = _conv_self(a, fft_len)
            # score
            cur_score = 2.0 * n * conv.max()
            # record best
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] new global best = {best_score:.6f}")

                # early stop
                if best_score <= 1.5025:
                    print("Target reached")
                    return best_a.tolist()

            # smoothed max weights
            max_c = conv.max()
            # compute weight vector
            exp_vals = np.exp(beta * (conv - max_c))
            w = exp_vals / exp_vals.sum()   # length 2n-1

            # compute gradient
            a_rev = a[::-1]
            # convolution w * a_rev
            conv_full = _conv_weighted(w, a_rev, fft_len)   # length 3n-2
            grad = 2.0 * conv_full[n-1 : n-1 + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            beta1_t *= beta1
            beta2_t *= beta2
            m_hat = m / (1 - beta1_t)
            v_hat = v / (1 - beta2_t)
            lr = lr0 / np.sqrt(it + 1)   # decaying
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto simplex
            a = np.clip(a, 0.0, None)
            a = _simplex_normalise(a)

            # update beta (smoothness)

"""
Propose a low‑max‑convolution step function.

Algorithm
---------
1.  Work on a fixed length `n` (default 1024) probability vector `a`,
    i.e. `a_i ≥ 0` and `sum(a)=1`.  The evaluation score for such a
    vector is `score(a)=2·n·max(a⋆a)`, where `⋆` denotes linear
    convolution.

2.  Perform many *random restarts*:
    * draw a Dirichlet‑type random vector,
    * run a few thousand Adam steps on the *smooth* objective
      `f_β(a)= (1/β)·log(∑_k exp(β·(a⋆a)_k))`.  
      The gradient of `f_β` is obtained analytically as a
      cross‑correlation, computed efficiently with a single FFT per
      iteration.
    * β (smoothing) is increased gradually (β≈1 → 80) so the
      optimisation first follows a smooth landscape and later focuses
      on the true maximum.
    * After each step we project back onto the probability simplex
      (clip negatives and renormalise).

   The best vector over all restarts is kept.

3.  A short *pairwise‑transfer* refinement is applied to the current
    best vector.  The gradient of the active (maximal) convolution
    entry identifies a source index (largest contribution) and a
    target index (smallest contribution); a small fraction of mass
    is moved from source to target if it reduces the score.

4.  Finally a *simulated‑annealing* phase explores the neighbourhood
    of the best vector found so far, accepting uphill moves with a
    temperature that linearly decays to zero.

The whole procedure respects the 1000 s budget, prints progress and
returns the best sequence (as a list of floats) found before the
deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)-1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_weighted(u: np.ndarray, v: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of `u` (length m) with `v` (length n) using FFT.
    Returns the full convolution of length m+n‑1.
    """
    u_fft = np.fft.rfft(u, n=fft_len)
    v_fft = np.fft.rfft(v, n=fft_len)
    full = np.fft.irfft(u_fft * v_fft, n=fft_len)[: u.size + v.size - 1]
    return full


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the convolution entry (a⋆a)[idx] w.r.t. `a`.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)          # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    # each term appears twice except when idx‑i == i (the diagonal);
    # the factor 2 is applied later in the smooth‑max gradient.
    return grad


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Problem size and FFT plan
    # ------------------------------------------------------------------
    n = 1024                     # length of the probability vector (can be changed)
    fft_len = _next_pow2(3 * n)  # enough for all convolutions we require

    # ------------------------------------------------------------------
    # Helper to compute the exact score (sum(a)=1 ⇒ score = 2·n·max(conv))
    # ------------------------------------------------------------------
    def _score_from_a(a_vec: np.ndarray) -> float:
        conv = _conv_self(a_vec, fft_len)
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    best_a = _simplex_normalise(np.full(n, 1.0 / n))
    best_score = _score_from_a(best_a)
    print(f"[0.00s] initial uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Random‑restart Adam optimisation on smoothed max‑convolution
    # ------------------------------------------------------------------
    restart = 0
    while time.monotonic() < deadline - 15.0:  # keep ~15 s for later phases
        restart += 1
        # ---- random start (Dirichlet with random concentration) ----------
        conc = rng.uniform(0.2, 2.0)           # shape varies per restart
        x = rng.gamma(conc, 1.0, size=n)
        a = _simplex_normalise(x.astype(np.float64))

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        beta1_pow = 1.0
        beta2_pow = 1.0

        # smoothing for soft‑max
        beta = 1.0                # start smooth
        beta_inc = 1.02
        max_beta = 80.0

        # learning‑rate schedule
        lr0 = 0.12

        max_iters = 4500
        for it in range(max_iters):
            now = time.monotonic()
            if now >= deadline - 15.0:
                break

            # ---- convolution & current score --------------------------------
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                elapsed = now - start
                print(f"[{elapsed:6.2f}s] new global best (restart {restart}) = {best_score:.6f}")

                if best_score <= 1.5025:
                    print("[early stop] target reached during Adam phase")
                    final = np.clip(best_a, 0.0, 1000.0)
                    print(f"Finished – final score = {evaluate_sequence(final.tolist()):.6f}")
                    return [float(x) for x in final]

            # ---- smooth‑max weights -----------------------------------------
            max_c = conv.max()
            # Stable log‑sum‑exp
            exp_vals = np.exp(beta * (conv - max_c))
            w = exp_vals / exp_vals.sum()                     # length 2n‑1

            # ---- gradient of the smoothed objective -------------------------
            # grad = 2 * cross‑correlation(a, w)
            a_rev = a[::-1]                                   # reversed a
            # full convolution w * a_rev (length 3n‑2)
            conv_full = _conv_weighted(w, a_rev, fft_len)
            grad = 2.0 * conv_full[n - 1 : n - 1 + n]        # slice length n

            # ---- Adam update ------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            lr = lr0 / math.sqrt(it + 1.0)                   # decay

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- simplex projection -----------------------------------------
            a = _simplex_normalise(a)

            # ---- increase smoothness ----------------------------------------
            beta = min(beta * beta_inc, max_beta)

            # ---- occasional jitter (avoid stagnation) -----------------------
            if (it + 1) % 1200 == 0:
                jitter = rng.normal(scale=0.0015, size=n)
                a = _simplex_normalise(np.clip(a + jitter, 0.0, None))

        # end of a single restart
        # ------------------------------------------------------------------
        # 2) Light pairwise‑transfer polishing on the best vector of this run
        # ------------------------------------------------------------------
        a_local = best_a.copy()
        for it2 in range(400):
            if time.monotonic() >= deadline - 15.0:
                break
            conv_loc = _conv_self(a_local, fft_len)
            max_idx = int(np.argmax(conv_loc))
            grad_max = _gradient_for_index(a_local, max_idx)
            src = int(np.argmax(grad_max))
            tgt = int(np.argmin(grad_max))
            if src == tgt:
                continue
            # small fractional transfer
            frac = rng.uniform(0.001, 0.015)
            delta = a_local[src] * frac
            if delta <= 0.0:
                continue
            cand = a_local.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = _simplex_normalise(cand)

            cand_score = _score_from_a(cand)
            cur_score = _score_from_a(a_local)
            # accept improving move
            if cand_score < cur_score - 1e-12:
                a_local = cand
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_a = cand.copy()
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:6.2f}s] new best (pairwise) = {best_score:.6f}")

        # --------------------------------------------------------------
        # go to the next random restart
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # 3) Final simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it3 = 0
    temp_start = 0.03 * best_score
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        fraction = elapsed / (deadline - start)
        temperature = temp_start * (1.0 - fraction)   # linear cooling

        conv = _conv_self(a, fft_len)
        max_idx = int(np.argmax(conv))
        grad_max = _gradient_for_index(a, max_idx)

        # source biased by squared gradient (more aggressive than linear)
        prob_src = grad_max ** 2
        ps = prob_src.sum()
        if ps > 0.0:
            prob_src /= ps
            src = int(rng.choice(n, p=prob_src))
        else:
            src = rng.integers(0, n)

        # uniform target (different from src)
        tgt = rng.integers(0, n)
        if tgt == src:
            tgt = (tgt + 1) % n

        delta = a[src] * rng.uniform(0.001, 0.02)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _simplex_normalise(cand)

        cand_score = _score_from_a(cand)
        cur_score = _score_from_a(a)

        # metropolis acceptance
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f}")

        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / max(1e-12, temperature)):
            a = cand

        it3 += 1
        if it3 % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] SA it={it3}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clip to the evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
