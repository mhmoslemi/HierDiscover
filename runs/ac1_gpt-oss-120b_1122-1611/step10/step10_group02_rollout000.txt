"""
Propose a step‑function sequence that minimizes the max‑convolution objective.

Algorithm
---------
1. **Initialization** – start from the provided `height_sequence_1`
   (if any) or from a random probability vector of length ≈3800.
   The vector is normalised to sum to 1 and lives on the simplex.

2. **Phase 1 – Adam on a smooth approximation of the max**  
   • At iteration *t* we compute the linear convolution `c = a * a`.  
   • A soft‑max weighting `w_k ∝ exp((c_k‑max(c))/τ_t)` is built,
     where the temperature τ_t is exponentially cooled.
   • The weighted gradient of the soft‑max objective w.r.t. the
     coefficients is obtained **exactly** by one extra FFT
     (`grad_i = Σ_j w_{i+j} a_j`).  
   • Adam’s moments are updated on the gradient of `a`; the update is
     applied to the *log‑weights* (mirror‑descent on the simplex) and
     the resulting vector is projected back onto the probability simplex.
   • The best vector seen so far is stored.

3. **Phase 2 – Simulated‑annealing refinement**  
   • Tiny mass‑transfer moves (single index → another index) are proposed,
     accepted with a Metropolis rule based on a linearly‑cooled temperature.
   • Every few thousand iterations a “jitter” move rescales a short block
     of entries, helping the search escape shallow minima.
   • The best vector is again tracked.

Both phases respect the 1000 s time budget (≈5 s safety margin) and
use only top‑level functions, no closures or file I/O.

The returned list is clipped to the limits imposed by `evaluate_sequence`
and its score is printed for verification.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σx = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2·len(a)‑1.
    """
    a_fft = rfft(a, n=fft_len)
    conv = irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_general(u: np.ndarray, v: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution u * v using an FFT of length ``fft_len``.
    Returns the full convolution of length len(u)+len(v)-1.
    """
    uf = rfft(u, n=fft_len)
    vf = rfft(v, n=fft_len)
    conv = irfft(uf * vf, n=fft_len)[: len(u) + len(v) - 1]
    return conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a feasible probability vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a = np.full_like(a0, 1.0 / a0.size)
        else:
            a = a0 / a0.sum()
    else:
        # default length ≈ 3800 (close to the previous best construction)
        n = 3809
        a = rng.random(n)
        a = a / a.sum()

    n = a.size

    # FFT lengths
    fft_len_conv = _next_pow2(2 * n)               # for a*a
    fft_len_grad = _next_pow2(3 * n)               # for w * a_rev

    # --------------------------------------------------------------
    # 1) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    conv = _conv_fft(a, fft_len_conv)
    best_score = 2.0 * n * conv.max()               # same as evaluate_sequence for Σa=1
    best = a.copy()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    target_score = 1.5025

    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 1 – Adam on a soft‑max approximation of the max‑convolution
    # --------------------------------------------------------------
    phase1_end = start_time + 0.60 * (deadline - start_time)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5
    lr_decay = 0.9999

    # Soft‑max temperature schedule
    tau0 = 0.015          # start fairly smooth
    tau_min = 1e-6
    tau_decay = 0.9995

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    last_improve_it = 0

    while time.monotonic() < phase1_end:
        # --- convolution a*a -------------------------------------------------
        conv = _conv_fft(a, fft_len_conv)
        max_conv = conv.max()

        # --- soft‑max weights -------------------------------------------------
        tau = max(tau0 * (tau_decay ** it), tau_min)
        # all entries are ≤ max_conv, so (conv‑max)/tau ≤ 0
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- weighted gradient via convolution --------------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft_general(w, a_rev, fft_len_grad)
        grad = grad_full[n - 1 : n - 1 + n]          # size n

        # --- Adam update on log‑weights --------------------------------------
        g = grad
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 * (lr_decay ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        log_a = np.log(a + 1e-15) - step
        a = np.exp(log_a)
        a = _project_onto_simplex(a)

        # --- evaluate new score ------------------------------------------------
        conv = _conv_fft(a, fft_len_conv)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            last_improve_it = it
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f} (it={it})")
            if best_score <= target_score:
                print("[early stop] target reached in phase 1")
                break

        it += 1
        # occasional status line
        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] phase 1 it={it}, best={best_score:.6f}")

        # if we have gone a long time without improvement, inject a jitter
        if it - last_improve_it > 20000:
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            last_improve_it = it

    # --------------------------------------------------------------
    # 3) Phase 2 – Simulated‑annealing local refinement
    # --------------------------------------------------------------
    a = best.copy()
    curr_score = best_score

    T = 0.02                     # starting temperature
    decay = 0.9999               # exponential cooling
    sa_it = 0
    last_sa_improve = 0

    while time.monotonic() < deadline:
        # ---- tiny directed mass transfer ------------------------------------
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-15:
            continue
        # random fraction in a log‑uniform window
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.04)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        conv_cand = _conv_fft(cand, fft_len_conv)
        cand_score = 2.0 * n * conv_cand.max()
        delta_score = cand_score - curr_score

        # Metropolis acceptance
        if delta_score < 0.0 or rng.random() < np.exp(-delta_score / max(T, 1e-12)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                last_sa_improve = sa_it
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f} (it={sa_it})")
                if best_score <= target_score:
                    print("[early stop] target reached in SA phase")
                    break

        # ---- occasional block jitter (helps escape plateaus) ---------------
        if sa_it % 6000 == 0 and sa_it > 0:
            width = rng.integers(1, max(2, n // 100))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            block = a.copy()
            block[start:start + width] *= factor
            a = _project_onto_simplex(block)

        # ---- cooling ---------------------------------------------------------
        T *= decay
        sa_it += 1

        # ---- periodic status -------------------------------------------------
        if sa_it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA it={sa_it}, T={T:.3e}, best={best_score:.6f}")

        # ---- if stuck for a long stretch, do a global jitter ---------------
        if sa_it - last_sa_improve > 50000:
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            last_sa_improve = sa_it

    # --------------------------------------------------------------
    # 4) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
