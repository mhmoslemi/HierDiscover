def propose_candidate(seed=12345,budget_s=1000.0,**kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    n = ...
    conv_len = 2*n-1
    fft_len = next_pow2(conv_len)
    # define compute_J
    def compute_J(vec):
        A = np.fft.rfft(vec, fft_len)
        conv = np.fft.irfft(A * A, fft_len)[:conv_len]
        return 2.0 * n * conv.max()
    # initial candidates: families
    best_J = np.inf
    best_vec = None
    # define function to test vec
    def evaluate_and_update(vec):
        nonlocal best_J, best_vec
        # ensure nonnegative
        vec = np.maximum(vec,0.0)
        total = vec.sum()
        if total == 0.0:
            return
        vec = vec / total
        J = compute_J(vec)
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
    # generate candidate families
    # uniform
    evaluate_and_update(np.full(n, 1.0/n))
    # geometric decays
    for r in [0.905,0.92,0.94,0.96,0.98,0.99,0.995]:
        v = r ** np.arange(n)
        v = v / v.sum()
        evaluate_and_update(v)
    # power law
    for p in [0.2,0.5,1.0,1.5,2.0]:
        i = np.arange(1, n+1)
        v = 1.0/(i**p)
        v /= v.sum()
        evaluate_and_update(v)
    # exponential edges shape with mixing
    for _ in range(50):
        alpha = 10**rng.uniform(-3, 1)  # between 0.001 and 10
        w = rng.random()
        exp_vec = np.exp(-alpha * np.arange(n)) + np.exp(-alpha * (n-1 - np.arange(n)))
        exp_vec = exp_vec / exp_vec.sum()
        v = (1-w)*np.full(n, 1.0/n) + w*exp_vec
        v /= v.sum()
        evaluate_and_update(v)
    # windows
    windows = {}
    windows['hann'] = np.hanning(n)
    windows['hamming'] = np.hamming(n)
    windows['blackman'] = np.blackman(n)
    windows['bartlett'] = np.bartlett(n)
    for name, wv in windows.items():
        if wv.sum() > 0:
            v = wv / wv.sum()
            evaluate_and_update(v)
    # kaiser with several betas
    for beta in [0.5, 2.0, 5.0, 10.0, 15.0]:
        wv = np.kaiser(n, beta)
        if wv.sum() > 0:
            v = wv / wv.sum()
            evaluate_and_update(v)
    # tukey
    for alpha in np.linspace(0.0,1.0,11):
        # implement tukey
        t = np.arange(n) / (n-1)
        # protect alpha=0 => rectangular, alpha=1 => Hann
        if alpha == 0:
            wv = np.ones(n)
        else:
            # compute
            wv = np.ones_like(t)
            # first region
            mask = t < alpha/2
            wv[mask] = 0.5 * (1 + np.cos(np.pi * (2*t[mask]/alpha - 1)))
            # second region
            mask2 = t > 1 - alpha/2
            wv[mask2] = 0.5 * (1 + np.cos(np.pi * (2*t[mask2]/alpha - 2/alpha + 1)))
        if wv.sum()>0:
            v = wv / wv.sum()
            evaluate_and_update(v)
    # random Dirichlet draws
    for alpha in [0.3,0.6,1.0,2.0]:
        for _ in range(10):
            v = rng.dirichlet(np.full(n, alpha))
            evaluate_and_update(v)
    # previous best if exists
    if "height_sequence_1" in globals():
        v = np.array(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0, None)
        if v.sum()>0:
            v = v / v.sum()
        else:
            v = np.full(n, 1.0/n)
        evaluate_and_update(v)

    print(f"Initial after families best J={best_J:.6f}")

    # Keep best as cur
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_time = time.monotonic()
    start_time = best_time
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Subgradient optimiser similar to prior but with tuned parameters
    iteration = 0
    last_improve_time = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # compute temperature schedule: start relatively high, cool gradually
        # we use T_start proportional to cur_J*0.1
        T_start = best_J * 0.1
        T_end = 1e-7
        T = T_start * (1 - elapsed/(deadline - start_time)) + T_end * (elapsed/(deadline - start_time))

        # compute subgradient of max(conv) for cur_vec
        conv = np.fft.irfft(np.fft.rfft(cur_vec, fft_len)**2, fft_len)[:conv_len]
        max_conv = conv.max()
        # recompute cur_J if needed
        cur_J = 2.0 * n * max_conv

        # maintain best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve_time = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # handle stagnation
        if now - last_improve_time > 20.0:
            # random jitter
            noise = np.exp(rng.normal(0.0, 0.03, size=n))
            cur_vec = best_vec * noise
            cur_vec = cur_vec / cur_vec.sum()
            last_improve_time = now
            continue

        # active set
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        grad = np.zeros_like(cur_vec)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1)
            grad[idx] += 2.0 * cur_vec[p - idx]

        # propose using step size with momentum?
        # we'll use multiplicative update as before, but step size adapt:
        step = 0.4 / np.sqrt(iteration + 1.0)  # slightly larger
        factor = np.exp(-step * grad)
        proposal = cur_vec * factor
        # sometimes try additive random jitter candidate
        if rng.random() < 0.1:
            # random Dirichlet perturbation, mix with proposal
            eps = rng.uniform(0.0, 0.2)
            random_vec = rng.dirichlet(np.full(n, 1.0))
            proposal = (1-eps)*proposal + eps*random_vec
        proposal = proposal / proposal.sum()
        prop_conv = np.fft.irfft(np.fft.rfft(proposal, fft_len)**2, fft_len)[:conv_len]
        prop_J = 2.0 * n * prop_conv.max()

        # metropolis
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # final clipping and scaling
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0/n)
    else:
        best_vec = best_vec / total
    return [float(v) for v in best_vec]

"""
Hybrid parameter‑sweep + exponentiated‑gradient annealing optimizer.

* Generates many diverse candidate step‑functions (uniform, geometric,
  power‑law, exponential‑edge blends, classic window functions, Dirichlet
  draws, and the previously found construction if present) and keeps the
  best one.

* Refines the best vector with a sub‑gradient method on the simplex.
  The sub‑gradient is obtained from the active indices of the full
  auto‑convolution.  A multiplicative (exponentiated‑gradient) update
  together with a simple simulated‑annealing acceptance rule allows the
  search to escape shallow plateaus.

* Works on the probability simplex (∑a_i = 1); therefore the objective
  becomes  J = 2·n·max(conv(a,a)).  The routine stops after the supplied
  time budget (≈1000 s) and returns the best sequence it discovered.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum value of the full linear convolution vec * vec."""
    A = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def objective_from_vec(vec: np.ndarray, n: int,
                       fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(vec,vec)) (sum(vec) assumed = 1)."""
    return 2.0 * n * full_conv_max(vec, fft_len, conv_len)


def tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window."""
    if alpha <= 0.0:
        return np.ones(n, dtype=np.float64)
    if alpha >= 1.0:
        # Tukey with α=1 reduces to a Hann window
        return np.hanning(n)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(t)
    first = t < alpha / 2.0
    w[first] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[first] / alpha - 1.0)))
    last = t > 1.0 - alpha / 2.0
    w[last] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t[last] / alpha - 2.0 / alpha + 1.0)))
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size (length of the vector).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Helper to evaluate a candidate and keep the global best.
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    def evaluate_and_update(vec: np.ndarray) -> None:
        nonlocal best_vec, best_J
        # safeguard against negative numbers
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            return
        vec /= s
        J = objective_from_vec(vec, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = vec.copy()

    # ------------------------------------------------------------------
    # 3.  Broad initial portfolio (parameter sweep + random draws).
    # ------------------------------------------------------------------
    # uniform
    evaluate_and_update(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (both monotone and symmetric)
    for r in [0.905, 0.92, 0.94, 0.96, 0.98, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        evaluate_and_update(v)
        # reversed version (mass on the right side)
        evaluate_and_update(v[::-1])

    # power‑law shapes
    for p in [0.2, 0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1, dtype=np.float64)
        v = 1.0 / (i ** p)
        v /= v.sum()
        evaluate_and_update(v)

    # exponential edge blends (mix uniform + symmetric exponential)
    for _ in range(80):
        alpha = 10 ** rng.uniform(-3.0, 1.0)          # 0.001 … 10
        mix = rng.random()
        exp_vec = np.exp(-alpha * np.arange(n, dtype=np.float64))
        exp_vec += np.exp(-alpha * (n - 1 - np.arange(n, dtype=np.float64)))
        exp_vec /= exp_vec.sum()
        v = (1.0 - mix) * np.full(n, 1.0 / n) + mix * exp_vec
        v /= v.sum()
        evaluate_and_update(v)

    # classic window functions
    windows = {
        "hann": np.hanning(n),
        "hamming": np.hamming(n),
        "blackman": np.blackman(n),
        "bartlett": np.bartlett(n),
    }
    for w in windows.values():
        if w.sum() > 0.0:
            evaluate_and_update(w / w.sum())

    # Kaiser windows with several betas
    for beta in [0.5, 2.0, 5.0, 10.0, 15.0]:
        w = np.kaiser(n, beta)
        if w.sum() > 0.0:
            evaluate_and_update(w / w.sum())

    # Tukey windows (alpha from 0 to 1)
    for alpha in np.linspace(0.0, 1.0, 11):
        w = tukey_window(n, alpha)
        if w.sum() > 0.0:
            evaluate_and_update(w / w.sum())

    # random Dirichlet draws (different concentration parameters)
    for conc in [0.3, 0.6, 1.0, 2.0]:
        for _ in range(12):
            evaluate_and_update(rng.dirichlet(np.full(n, conc)))

    # previous construction, if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = 1.0 / n
        else:
            v /= v.sum()
        evaluate_and_update(v)

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4.  Local refinement – exponentiated‑gradient + simulated annealing.
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # temperature schedule (linear cooling)
    T_start = best_J * 0.1          # 10 % of the current best as start temperature
    T_end = 1e-8

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear cooling of the Metropolis temperature
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * elapsed / (deadline - start_time))

        # --------------------------------------------------------------
        # 4.1 Evaluate current state (full convolution)
        # --------------------------------------------------------------
        max_conv = full_conv_max(cur_vec, fft_len, conv_len)
        cur_J = 2.0 * n * max_conv

        # record a new global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 4.2 Stagnation handling – occasional jitter around the best.
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec /= cur_vec.sum()
            last_improve = now
            continue

        # --------------------------------------------------------------
        # 4.3 Sub‑gradient of max(conv) at the active set.
        # --------------------------------------------------------------
        conv_vals = np.fft.irfft(
            np.fft.rfft(cur_vec, n=fft_len) ** 2,
            n=fft_len)[:conv_len]
        max_c = conv_vals.max()
        tol = max(1e-12, max_c * 1e-12)
        active = np.where(conv_vals >= max_c - tol)[0]

        grad = np.zeros_like(cur_vec)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=np.int64)
            grad[idx] += 2.0 * cur_vec[p - idx]

        # --------------------------------------------------------------
        # 4.4 Propose a new point (multiplicative update).
        # --------------------------------------------------------------
        step = 0.5 / np.sqrt(iteration + 1.0)          # 1/√t schedule
        factor = np.exp(-step * grad)
        proposal = cur_vec * factor

        # occasionally blend with a fresh random Dirichlet point
        if rng.random() < 0.08:
            eps = rng.uniform(0.0, 0.15)
            rand_vec = rng.dirichlet(np.ones(n))
            proposal = (1.0 - eps) * proposal + eps * rand_vec

        # re‑project onto the simplex
        proposal /= proposal.sum()

        # --------------------------------------------------------------
        # 4.5 Evaluate the proposal.
        # --------------------------------------------------------------
        prop_max_conv = full_conv_max(proposal, fft_len, conv_len)
        prop_J = 2.0 * n * prop_max_conv

        # --------------------------------------------------------------
        # 4.6 Metropolis acceptance.
        # --------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (clip, renormalise and cast to list).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
