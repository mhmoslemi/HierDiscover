assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the linear convolution of the coefficient vector with itself.
The search follows three stages that fit comfortably in the 1000 s budget.

1. **Warm‑start** – If a previous construction (`height_sequence_1`) exists it is
   truncated / renormalised, otherwise a random simplex vector of length 1024
   is used.

2. **Adam‑accelerated exponentiated‑gradient descent**  
   The exact sub‑gradient of the *max‑convolution* is obtained from the
   (almost) active indices.  The gradient is applied to the log‑variables
   (i.e. a multiplicative update) and the Adam moment estimates provide
   an adaptive step size.  After each update the vector is projected back onto
   the probability simplex.

3. **Smoothed‑max (soft‑max) gradient descent** –  
   The non‑smooth maximum is replaced by a temperature‑controlled soft‑max.
   The gradient of the smooth surrogate is  
   \[
   \nabla\;M_\alpha(a)=2\,\bigl(w*a^{\text{rev}}\bigr)_{\,n-1:\,2n-1},
   \]
   where \(w_k\propto \exp\bigl(\alpha\,(a*a)_k\bigr)\) and `a_rev` is the
   reversed vector.  A decreasing learning‑rate and an exponential schedule for
   the smoothing parameter \(\alpha\) drive the solution to a low‑peak convolution.

Both phases keep track of the best vector seen so far and stop early if the
target score 1.5025 is reached.

The code relies only on `numpy`, `scipy.optimize.linprog` (for the optional
LP‑boost that can be activated by setting `USE_LP=True`) and standard
Python libraries.  All helper functions are top‑level as required.

```python
"""
Propose a step‑function that minimises the “max‑convolution” score.

The search consists of:
1. A warm start (either the supplied height_sequence_1 or a random simplex).
2. Adam‑accelerated exponentiated‑gradient descent on the exact sub‑gradient
   of the max‑convolution.
3. A soft‑max (smoothed max) gradient descent that gradually sharpens the
   temperature to approach the true maximum.

Both phases keep the vector on the probability simplex and record the best
solution found.  The algorithm stops early when the target 1.5025 is reached
or when the 1000 s budget expires.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog  # optional LP boost, harmless if unused


# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2·len(a)‑1.
    """
    a_fft = rfft(a, n=fft_len)
    conv = irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Temperature‑controlled soft‑max on the convolution values.
    Returns a probability vector w (sum = 1).
    """
    maxc = conv.max()
    # shift by maxc for numerical stability
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / exp_term.sum()
    return w


def _smooth_gradient(a: np.ndarray, w: np.ndarray, cross_fft_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate.
    Returns 2 * (w * a_rev)[n-1:2n-1] where * denotes linear convolution.
    """
    a_rev = a[::-1]
    w_fft = rfft(w, n=cross_fft_len)
    a_rev_fft = rfft(a_rev, n=cross_fft_len)
    cross = irfft(w_fft * a_rev_fft, n=cross_fft_len)
    # keep only the central slice of length n
    start = a.size - 1
    slice_ = cross[start:start + a.size]
    grad = 2.0 * slice_
    return grad


def _maybe_build_lp_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1)×n) that encodes the linear constraints
    (f * b)[k] ≤ M.  Each column j of A contains the vector f shifted down by j.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_boost(f: np.ndarray, M: float, max_iter: int = 1):
    """
    One LP‑boost step: maximise Σ b_j subject to (f*b)[k] ≤ M.
    Returns a feasible b (scaled to sum 1) or None on failure.
    """
    A = _maybe_build_lp_matrix(f)
    rows = A.shape[0]
    c = -np.ones(f.size)           # maximise sum(b)  → minimise –Σb
    bounds = [(0, None)] * f.size
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M),
                  bounds=bounds, method="highs")
    if not res.success:
        return None
    b = res.x
    if b.sum() == 0.0:
        return None
    return b / b.sum()            # normalise to the simplex


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Warm start – use the previous construction if it exists.
    # --------------------------------------------------------------
    max_n = 1024                     # modest size – fast FFTs & LPs
    if "height_sequence_1" in globals():
        arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        arr = np.clip(arr, 0.0, None)
        if arr.size > max_n:
            a = arr[:max_n]          # simple truncation
        else:
            a = arr.copy()
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a.sum()
    else:
        a = rng.random(max_n)
        a = a / a.sum()

    n = a.size
    fft_len = _next_pow2(2 * n)
    cross_fft_len = _next_pow2(3 * n)

    # --------------------------------------------------------------
    # 1) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()               # sum(a) == 1 → exact score
    best = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    target_score = 1.5025

    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 1 – Adam‑accelerated exponentiated‑gradient descent
    # --------------------------------------------------------------
    log_a = np.log(a + 1e-15)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    phase1_end = start_time + 0.45 * (deadline - start_time)   # ~45 % of time

    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        # active set – a tight tolerance keeps a few indices
        active = np.where(conv >= max_b * (1.0 - 1e-8))[0]
        grad = _gradient_from_active(a, active)

        # Adam updates of the *log* variables
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.4 / np.sqrt(t)          # diminishing learning rate
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.exp(log_a)
        a = _project_onto_simplex(a)

        # occasional evaluation
        if t % 25 == 0:
            conv = _conv_fft(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (Adam EG) = {best_score:.6f} (it={t})")
                if best_score <= target_score:
                    print("[early stop] target reached in Phase 1")
                    break

    # --------------------------------------------------------------
    # 3) Optional LP‑boost (a single aggressive improvement step)
    # --------------------------------------------------------------
    USE_LP = False          # set to True if you want to spend a little time on it
    if USE_LP:
        conv = _conv_fft(a, fft_len)
        M = conv.max()
        b_lp = _lp_boost(a, M)
        if b_lp is not None:
            # simple line‑search for the best mixing weight
            best_local = best_score
            best_mix = 0.0
            best_vec = a
            for t_mix in np.linspace(0.05, 0.5, 10):
                cand = (1.0 - t_mix) * a + t_mix * b_lp
                conv_cand = _conv_fft(cand, fft_len)
                score_cand = 2.0 * n * conv_cand.max()
                if score_cand < best_local:
                    best_local = score_cand
                    best_mix = t_mix
                    best_vec = cand.copy()
            if best_mix > 0.0:
                a = best_vec
                best = a.copy()
                best_score = best_local
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] after LP‑boost new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – smoothed‑max gradient descent (temperature annealing)
    # --------------------------------------------------------------
    a = best.copy()
    it = 0
    alpha_start, alpha_end = 5.0, 500.0   # smoothing schedule
    while time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len)
        max_c = conv.max()

        # progress‑dependent smoothing
        prog = (time.monotonic() - start_time) / (deadline - start_time)
        alpha = alpha_start + prog * (alpha_end - alpha_start)

        # soft‑max weights
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth surrogate
        grad = _smooth_gradient(a, w, cross_fft_len)

        # step size – diminishing
        lr = 0.2 / np.sqrt(it + 1.0)

        # multiplicative update (mirror descent)
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a)

        # bookkeeping
        if it % 20 == 0:
            conv = _conv_fft(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (smooth) = {best_score:.6f} (it={it})")
                if best_score <= target_score:
                    print("[early stop] target reached in Phase 2")
                    break
        it += 1

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
```
