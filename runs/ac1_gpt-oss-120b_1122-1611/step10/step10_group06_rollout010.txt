def propose_candidate(seed=12345, budget_s=1000.0):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    n = 2048   # or maybe 3000 if time budget allows
    # Initialize x from normal distribution
    x = rng.normal(size=n)
    # converts to a via softmax
    a = np.exp(x - np.max(x))
    a /= a.sum()
    # Precompute FFT length
    fft_len = 1 << ((3*n - 1).bit_length())
    # Adam parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # smoothing param
    beta = 5.0
    best_a = a.copy()
    # compute initial best score
    conv = _conv_self(a, fft_len)
    best_score = 2.0 * n * conv.max()
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    it = 0
    while time.monotonic() < deadline:
        # compute convolution via FFT
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
        max_c = conv.max()
        # compute softmax weights
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # gradient w.r.t a
        w_fft = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[:n + w.size - 1]
        grad = grad_full[:n]    # shape (n,)
        # gradient w.r.t x (softmax parameterization)
        mu = np.dot(grad, a)
        grad_x = a * (grad - mu)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        it_hat = it + 1
        m_hat = m / (1 - beta1 ** it_hat)
        v_hat = v / (1 - beta2 ** it_hat)
        x -= 0.05 * m_hat / (np.sqrt(v_hat) + eps)    # step size
        # recompute a
        # To avoid overflow, subtract max
        x_max = np.max(x)
        a = np.exp(x - x_max)
        a /= a.sum()
        # evaluate score
        conv = _conv_self(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new best (Adam) = {best_score:.6f} iter={it}")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break
        # schedule for beta: increase slowly

                cand_a = a.copy()
                cand_a[i], cand_a[j] = cand_a[j], cand_a[i]
                conv = _conv_self(cand_a, fft_len)
                score_cand = 2.0 * n * conv.max()
                if score_cand < best_score:
                    best_score = score_cand
                    best_a = cand_a.copy()
                    a = cand_a.copy()
                    # recompute x from a using log

                    # renorm x by subtract max
                    x -= np.max(x)
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:.2f}s] new best (swap) = {best_score:.6f}")

assistantfinal**Solution Overview**

We optimise the step‑function heights on the probability simplex (all entries ≥ 0,
sum = 1).  
The objective to minimise is  

\[
\text{score}= \frac{2\,n\;\max_k (a*a)[k]}{\bigl(\sum_i a_i\bigr)^2}
        = 2\,n\;\max_k (a*a)[k] \qquad (\sum a_i = 1)
\]

so the problem is equivalent to minimising the maximal self‑convolution entry.

The algorithm combines three ideas:

1. **Smooth surrogate** – we replace the non‑smooth `max` by a
   log‑sum‑exp with a temperature `β`.  
   For a given `β` the surrogate is  
   \(\displaystyle L_\beta(a)=\log\bigl(\sum_k e^{\beta (a*a)_k}\bigr)\).  
   Its gradient w.r.t. the probability vector `a` is a convolution of `a`
   with the soft‑max weights `w_k ∝ e^{β (a*a)_k}`.

2. **Mirror‑descent on the soft‑max parametrisation** – we store the
   vector `x` with `a_i = \exp(x_i) / \sum_j \exp(x_j)`.  Updating `x`
   with Adam (first‑ and second‑moment adaptation) yields fast, stable
   descent while automatically keeping `a` on the simplex.

3. **Escaping local minima** – every few thousand iterations we
   (i) try a random swap of two positions and (ii) perform a small
   pairwise mass‑transfer move that directly reduces the currently
   maximal convolution entry.

All heavy work (convolutions, gradients) is performed with FFTs, giving
`O(n log n)` per iteration.  The routine works with `n = 2048`, which
fits comfortably into the 1000 s budget and still allows many
iterations.  The search stops a few seconds before the deadline and
returns the best sequence found.

"""Propose a step‑function that lowers the max‑convolution score.

The optimisation works on the simplex by:
1. A smooth log‑sum‑exp surrogate of the max‑convolution.
2. Adam updates on the logits x (a = softmax(x)) – multiplicative
   updates become additive in log‑space.
3. Periodic random swaps and greedy pairwise mass‑transfer moves to
   escape plateaus.

All convolutions use FFTs for speed.  The routine respects the 1000 s
budget and returns the best sequence observed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using FFT (length 2·len(a)‑1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the convolution entry (a*a)[idx].
    grad_i = a[idx-i] for i in the valid range, else 0.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)          # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    return grad


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Renormalise a non‑negative vector to unit sum."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Problem size – a power‑of‑two length is convenient for FFTs.
    # --------------------------------------------------------------
    n = 2048                      # chosen to allow many cheap iterations
    fft_len = _next_pow2(3 * n)   # enough for all convolutions we need

    # --------------------------------------------------------------
    # 1) Initialise logits x → soft‑max distribution a on the simplex.
    # --------------------------------------------------------------
    x = rng.normal(size=n)        # unrestricted real vector
    # soft‑max (numerically stable)
    x -= np.max(x)
    a = np.exp(x)
    a /= a.sum()

    # --------------------------------------------------------------
    # 2) Adam state and smoothing schedule
    # --------------------------------------------------------------
    m = np.zeros_like(x)          # 1st moment
    v = np.zeros_like(x)          # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta = 5.0                    # start temperature for the soft‑max
    beta_max = 200.0

    # --------------------------------------------------------------
    # 3) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    conv = _conv_self(a, fft_len)
    best_a = a.copy()
    best_score = float(2.0 * n * conv.max())
    print(f"[0.00s] initial score = {best_score:.6f}")

    it = 0
    while time.monotonic() < deadline:
        # ---------------------------  gradient of smooth surrogate ----------
        a_fft = np.fft.rfft(a, n=fft_len)

        # exact self‑convolution
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # soft‑max weights over the convolution entries
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()                     # length 2n‑1

        # gradient of the weighted sum w·(a*a)  (first n entries)
        w_fft = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: n + w.size - 1]
        grad_a = grad_full[:n]                            # ∂L/∂a

        # ---------------------------------------------------  Adam on logits
        # gradient w.r.t. logits x (a = softmax(x))
        mu = np.dot(grad_a, a)                            # ⟨∂L/∂a , a⟩
        grad_x = a * (grad_a - mu)                        # Jᵀ∂L/∂a

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        # learning‑rate schedule (constant base, can be tuned)
        eta = 0.05 / (np.sqrt(v_hat) + eps)
        x -= eta * m_hat

        # re‑compute a = softmax(x) – numerically stable
        x -= np.max(x)            # avoid overflow
        a = np.exp(x)
        a /= a.sum()

        # ---------------------------------------------------  evaluate current score
        conv = _conv_self(a, fft_len)
        score = float(2.0 * n * conv.max())

            best_a = a.copy()
            elapsed = time.monotonic() - start

                break

        # ---------------------------------------------------  increase smoothing
        beta = min(beta * 1.0015, beta_max)

        # ---------------------------------------------------  occasional diversification
        if it % 5000 == 0 and it > 0:

            i, j = rng.integers(0, n, size=2)
            if i != j:

                cand[i], cand[j] = cand[j], cand[i]
                conv_s = _conv_self(cand, fft_len)
                score_s = float(2.0 * n * conv_s.max())
                if score_s < best_score:
                    best_score = score_s
                    best_a = cand.copy()
                    a = cand
                    # rebuild logits from the swapped probabilities
                    x = np.log(a + 1e-30)
                    x -= np.max(x)
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:6.2f}s] new best (swap) = {best_score:.6f}")

            # (ii) greedy pairwise mass transfer on the active convolution entry
            max_idx = int(np.argmax(conv))
            grad_max = _gradient_for_index(a, max_idx)
            src = int(np.argmax(grad_max))
            tgt = int(np.argmin(grad_max))
            delta = a[src] * rng.uniform(0.001, 0.025)   # up to ~2 % of source mass
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = _simplex_normalise(cand)
            conv_c = _conv_self(cand, fft_len)
            score_c = float(2.0 * n * conv_c.max())
            if score_c < best_score:
                best_score = score_c
                best_a = cand.copy()
                a = cand
                x = np.log(a + 1e-30)
                x -= np.max(x)
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] new best (pair) = {best_score:.6f}")

        it += 1

    # ----------------------------------------------------------------------
    # 4) Return the best sequence (clip to evaluator limits)
    # ----------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(v) for v in best]
