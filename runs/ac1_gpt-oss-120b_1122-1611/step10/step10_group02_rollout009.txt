def gradient_smooth(a, beta, L_fft=None):
    conv = _conv_fft(a)   # length 2n-1
    max_conv = conv.max()
    # compute p (softmax weights)
    exp_terms = np.exp(beta * (conv - max_conv))  # shift for stability
    w_sum = exp_terms.sum()
    p = exp_terms / w_sum
    # compute gradient via correlation
    if L_fft is None:
        L_fft = _next_pow2(len(a) + len(p) - 1)
    A_fft = rfft(a, n=L_fft)
    P_fft = rfft(p, n=L_fft)
    corr = irfft(A_fft * np.conj(P_fft), n=L_fft)[:len(a)]
    grad = 2 * corr
    smooth_val = max_conv + np.log(w_sum) / beta
    return grad, smooth_val

def propose_candidate(seed=..., budget_s=1000.):

    rng = np.random.default_rng(seed)

    # Determine n: if height_sequence_1 exists, get length n0 else set n = 2048.
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0:
            n = a0.size
            a = np.full(n, 1/n)
        else:
            a = a0 / a0.sum()
    else:
        n = 2048
        a = rng.random(n)
        a = a / a.sum()

    # Determine FFT length L for conv (linear conv length = 2n-1). For smoothing gradient also need len(p) = 2n-1 for correlation length, compute L = next_pow2(3*n)
    fft_len_conv = _next_pow2(2*n - 1)
    fft_len_corr = _next_pow2(3*n)  # for correlation

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # safety

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"[0.0s] initial score = {best_score}")

    # set smoothing schedule
    beta = 1.0
    beta_max = 2000.0
    beta_factor = 1.05  # each iteration multiply

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2    # learning rate

    it = 0
    while time.monotonic() < deadline:
        # compute convolution conv (length 2n-1)
        a_fft = rfft(a, n=fft_len_conv)
        conv = irfft(a_fft * a_fft, n=fft_len_conv)[:2*n-1]

        # smooth max and gradient
        max_conv = conv.max()
        # compute softmax weights
        shift = max_conv  # for numerical stability
        exp_term = np.exp(beta * (conv - shift))
        w_sum = exp_term.sum()
        p = exp_term / w_sum

        # compute gradient via correlation
        # correlation of a and p
        # use real FFT
        # compute FFTs for a and p using same L
        a_fft_corr = rfft(a, n=fft_len_corr)
        p_fft = rfft(p, n=fft_len_corr)
        corr = irfft(a_fft_corr * np.conj(p_fft), n=fft_len_corr)[:n]   # length n
        grad = 2.0 * corr

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a_new = _project_onto_simplex(a_new)

        # evaluate for true objective
        score = evaluate_sequence(a_new.tolist())

        # update best
        if score < best_score:
            best_score = score
            best = a_new.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new best = {best_score:.6f} (it={it})")
            if best_score <= 1.5025:
                break

        # accept new a anyway (or keep a if not better? In gradient descent we always accept).
        a = a_new

        # increase beta gradually

        # occasional random jitter to escape

    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final best score = {final_score:.6f}")

"""
Improved optimiser for the max‑convolution objective.

The algorithm has two phases.

1. **Smooth exponential‑gradient descent** (≈70 % of the time budget).
   * The true objective  max(conv) is non‑smooth.  We replace it by the
     log‑sum‑exp (soft‑max) with an increasing temperature β.
   * The gradient of the soft‑max is a weighted sum of the exact
     sub‑gradients, where the weights are the soft‑max probabilities.
     This gives a smooth direction early on and approaches the exact
     sub‑gradient as β grows.
   * An exponentiated‑gradient (mirror) step on the simplex is used:
     `log a ← log a – η·∇`, followed by normalisation.
   * All convolutions and correlations are performed with FFTs
     (single FFT of the current vector per iteration, re‑used for both
     the forward convolution and the correlation needed for the gradient).

2. **Simulated‑annealing local refinement** (remaining time).
   Small directed mass‑transfer moves are accepted with a Metropolis
   rule, allowing the search to escape shallow local minima and to
   polish the solution found by the gradient phase.

Both phases keep the best vector seen so far and stop early once the
target score (1.5025) is reached.  The returned sequence is clipped to
the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, ∑x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only works on 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length L (L ≥ 2·len(a)‑1)."""
    A = rfft(a, n=L)
    conv = irfft(A * A, n=L)[: 2 * a.size - 1]
    return conv


def _corr_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation (a ⋆ b) where b is longer than a.
    Returns the first len(a) entries:
    (a ⋆ b)[i] = Σ_j a_j·b_{i+j}.
    """
    A = rfft(a, n=L)
    B = rfft(b, n=L)
    corr = irfft(A * np.conj(B), n=L)[: a.size]
    return corr

# Main optimisation routine
# ----------------------------------------------------------------------

            a = a / a.sum()

    # ------------------------------------------------------------------

    decay = 0.9997                # geometric cooling

    it_sa = 0
    while time.monotonic() < deadline:
        # ---- propose a tiny directed mass transfer ----------------------------
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-15:
            continue
        # multiplicative random fraction in [0.001, 0.03]
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.03)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()                     # stays on simplex

        conv_cand = _conv_fft(cand, L_conv)
        cand_score = 2.0 * n * conv_cand.max()
        Δ = cand_score - curr_score

        # Metropolis acceptance
        if Δ < 0.0 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f} (it={it_sa})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in SA phase")
                    break

        # occasional block‑jitter (helps to reshuffle local structure)
        if it_sa % 6000 == 0 and it_sa > 0:
            width = rng.integers(1, max(2, n // 80))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jitter = a.copy()
            a_jitter[start:start + width] *= factor
            a_jitter /= a_jitter.sum()
            a = a_jitter

        T *= decay
        it_sa += 1
        if it_sa % 4000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA it={it_sa}, T={T:.3e}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to evaluator limits).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
