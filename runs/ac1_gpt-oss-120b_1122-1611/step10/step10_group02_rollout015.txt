def local_refine(seq, n, time_deadline):
    best_seq = seq.copy()
    best_score = compute_score(best_seq)
    rng = np.random.default_rng()
    T = 0.01
    decay = 0.9999
    while time.monotonic() < time_deadline:
        # propose small move: pick src, tgt, delta fraction
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or best_seq[src] < 1e-12:
            continue
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = best_seq[src] * frac
        cand = best_seq.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)  # ensure nonnegative
        cand = cand / cand.sum()  # project onto simplex (renormalize)
        cand_score = compute_score(cand)
        if cand_score < best_score or rng.random() < np.exp(-(cand_score - best_score) / max(T,1e-12)):
            best_seq = cand
            best_score = cand_score
        T *= decay
    return best_seq, best_score
```

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise if rounding killed the mass
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2·len(a)‑1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv

    n = seq.size
    # FFT length – next power of two for speed and exactness
    fft_len = _next_pow2(2 * n)
    conv = _conv_fft(seq, fft_len)
    max_b = conv.max()
    # sum is forced to 1 in the optimisation (simplex), but we keep it
    # for completeness
    sum_a = seq.sum()
    if sum_a < 1e-12:
        # should never happen – protect against division by zero
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Differential‑Evolution search for a non‑negative step function that
    minimises ``evaluate_sequence``.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Problem size & start‑points
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        n = base.size
        # normalise – the objective is scale‑invariant
        base = base / base.sum() if base.sum() > 0 else np.full(n, 1.0 / n)
    else:
        n = 3809               # a moderate size that performed well in the past
        base = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 1) Initialise a small population on the simplex
    # --------------------------------------------------------------
    pop_size = 30
    population = np.empty((pop_size, n), dtype=np.float64)

    # slot 0 – the historic sequence (if any)
    population[0] = base.copy()

    # slot 1 – perfectly uniform
    population[1] = np.full(n, 1.0 / n)

    # remaining slots – Dirichlet draws (α_i = 1)
    for i in range(2, pop_size):
        population[i] = rng.dirichlet(alpha=np.ones(n))

    # pre‑compute the scores
    scores = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        scores[i] = _score_of_sequence(population[i])

    best_idx = int(np.argmin(scores))
    best = population[best_idx].copy()
    best_score = float(scores[best_idx])

    start_time = time.monotonic()
    # leave a few seconds margin for finalisation / printing
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    target_score = 1.5025

    print(f"[{0.00:.2f}s] initial best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Differential Evolution loop
    # --------------------------------------------------------------
    # DE hyper‑parameters
    F = 0.8                     # differential weight
    CR = 0.9                    # crossover probability

    generation = 0
    while time.monotonic() < deadline:
        for i in range(pop_size):
            # pick three distinct donors different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

            mutant = population[r1] + F * (population[r2] - population[r3])
            # enforce non‑negativity before projection
            mutant = np.maximum(mutant, 0.0)
            mutant = _project_onto_simplex(mutant)

            # binomial crossover
            cross_mask = rng.random(n) < CR
            # guarantee at least one coordinate from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = _project_onto_simplex(trial)

            trial_score = _score_of_sequence(trial)

            # greedy selection
            if trial_score < scores[i] - 1e-12:
                population[i] = trial
                scores[i] = trial_score

                # global update
                if trial_score < best_score - 1e-12:
                    best_score = trial_score
                    best = trial.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] new best (DE) = {best_score:.6f} (gen={generation})")
                    if best_score <= target_score:
                        print("[early stop] target reached")
                        # final sanity check with the official evaluator
                        final = evaluate_sequence(best.tolist())
                        print(f"  final evaluator score = {final:.6f}")
                        # clip to evaluator limits before returning
                        best = np.clip(best, 0.0, 1000.0)
                        return [float(x) for x in best]

        generation += 1
        # occasional log
        if generation % 10 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] generation {generation}, best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Return the best sequence found (apply evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
