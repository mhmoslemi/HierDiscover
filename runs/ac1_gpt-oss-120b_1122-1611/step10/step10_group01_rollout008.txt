    n = len(a)

def solve_lp_max_sum(a, M):
    n = a.size
    col = np.r_[a, np.zeros(n-1)]
    row = np.r_[a[0], np.zeros(n-1)]
    A = scipy.linalg.toeplitz(col, row)   # shape (2n-1, n)
    # limit each row
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    # Use linprog
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    return res.x
```

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Candidate families** – generate a few parametrised sequences
   (uniform, Gaussian, arcsine‑law, power‑law, sparse Dirichlet draws and,
   if available, the previous construction ``height_sequence_1``).

2. **Iterative LP + mixing** – for each candidate we repeatedly solve the
   linear program

        maximise   Σ b_i
        subject to  a * b  ≤  M   (elementwise)
                    b_i ≥ 0

   where ``a`` is the current sequence, ``M = max(a*a)`` and the
   convolution ``a*b`` is linear in ``b``.  The LP returns a shape ``b``
   that respects the same maximal self‑convolution bound as ``a`` while
   allowing the total mass to grow.  The normalized vector
   ``g = b / Σb`` is blended with ``a``:  ``a ← (1‑t)·a + t·g`` with a
   small step size ``t`` (≈0.1–0.2).  The blend is accepted only if the
   new maximal convolution does not increase.  Repeating this step drives
   the maximal convolution down.

3. **Local stochastic refinement** – after the LP phase we perform a
   cheap hill‑climbing routine that transfers a tiny amount of mass from
   the index contributing most to the current maximum to a low‑gradient
   index.  Moves that improve the exact objective are kept.

All the work is bounded by the supplied time budget (default 1000 s).  The
best sequence encountered is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy import optimize, linalg

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a via FFT (returns length 2·len(a)‑1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _smooth_max_gradient(a: np.ndarray, p: int, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth p‑norm surrogate of max_k (a∗a)[k].
    """
    conv = _conv_fft(a, fft_len)
    eps = 1e-12
    w = np.maximum(conv, eps) ** (p - 1)            # weights of the lag‑wise terms
    L = fft_len
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    offset = len(w) - 1
    grad = 2.0 * corr[offset: offset + a.size]
    return grad


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value 2·n·max(a∗a) for a already normalised to Σa=1."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Solve  maximise Σ b_i   subject to   a∗b ≤ M (componentwise) ,  b ≥ 0.
    Returns the optimal b as a dense NumPy array, or ``None`` on failure.
    """
    n = a.size
    # Build the (2n‑1) × n convolution matrix:   (a∗b) = A @ b
    col = np.r_[a, np.zeros(n - 1, dtype=a.dtype)]
    row = np.r_[a[0:1], np.zeros(n - 1, dtype=a.dtype)]
    A = linalg.toeplitz(col, row)               # shape (2n‑1, n)

    # Linear program (HiGHS is fast and works with dense matrices)
    c = -np.ones(n, dtype=np.float64)           # maximise Σb  ⇔ minimise –Σb
    bounds = [(0.0, None)] * n
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    res = optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` under a time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # leave a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0) Decide on a few problem sizes (larger n gives more freedom but
    #    also larger LPs).  2500‑3000 is a good compromise for the time budget.
    # ------------------------------------------------------------------
    n_choices = [1800, 2500, 3200]

    # ------------------------------------------------------------------
    # Helper to initialise a candidate from a parametric family
    # ------------------------------------------------------------------
    def _initial_candidates(n: int):
        cand = []

        # 1) uniform
        cand.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) symmetric Gaussian
        mu = (n - 1) / 2.0
        for sigma in (n / 8.0, n / 5.0, n / 4.0):
            idx = np.arange(n, dtype=np.float64)
            g = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
            if g.sum() > 0:
                g /= g.sum()
                cand.append(g)

        # 3) arcsine‑law (higher density near the ends, known to flatten the
        #    self‑convolution)
        x = np.arange(n, dtype=np.float64) + 0.5
        arc = 1.0 / np.sqrt(x * (n + 0.5 - x))
        arc /= arc.sum()
        cand.append(arc)

        # 4) power‑law a_i ∝ (i+1)^{-α}
        for alpha in (0.3, 0.7, 1.2):
            p = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
            p /= p.sum()
            cand.append(p)

        # 5) sparse Dirichlet draws (concentration < 1)
        for _ in range(4):
            s = rng.dirichlet(np.full(n, 0.4))
            cand.append(s)

        # 6) previous construction if available (trim / pad as needed)
        if "height_sequence_1" in globals():
            hs = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if hs.size > n:
                hs = hs[:n]
            elif hs.size < n:
                hs = np.concatenate([hs, np.zeros(n - hs.size)])
            hs = np.clip(hs, 0.0, None)
            if hs.sum() > 0:
                hs = hs / hs.sum()
                cand.append(hs)

        return cand

    # ------------------------------------------------------------------
    # Main search over several problem sizes and random restarts
    # ------------------------------------------------------------------
    for n in n_choices:
        fft_len = _next_pow2(3 * n)       # safe length for all linear convolutions
        init_cands = _initial_candidates(n)

        # Shuffle the initial candidates to diversify the runs
        rng.shuffle(init_cands)

        for start_a in init_cands:
            if time.monotonic() >= deadline:
                break

            a = start_a.copy()
            a = np.clip(a, 0.0, None)
            if a.sum() == 0.0:
                continue
            a = a / a.sum()                # enforce Σa = 1

            # evaluation of the starting point
            cur_score = _objective(a, fft_len)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] n={n} init score={cur_score:.6f}")

            # ------------------------------------------------------------------
            # 1) LP‑based improvement loop
            # ------------------------------------------------------------------
            max_lp_iters = 12
            for lp_it in range(max_lp_iters):
                if time.monotonic() >= deadline:
                    break

                # current bound M = max(a∗a)
                conv = _conv_fft(a, fft_len)
                M = conv.max()

                b = _solve_lp_max_sum(a, M)
                if b is None:
                    break

                sum_b = b.sum()
                if sum_b <= 1.0 + 1e-12:
                    # LP cannot increase the total mass – we are at a (local) optimum
                    break

                g = b / sum_b                    # shape with Σg = 1

                # mixing factor – a modest step towards the LP solution
                t = min(0.18, (sum_b - 1.0) / (sum_b + 1.0))
                a_candidate = (1.0 - t) * a + t * g
                # reconstrain onto the simplex (numerical safety)
                a_candidate = _project_onto_simplex(a_candidate)

                new_score = _objective(a_candidate, fft_len)
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_seq = a_candidate.copy()
                    print(f"[{time.monotonic() - start_time:.2f}s] n={n} LP iter={lp_it} improved to {new_score:.6f}")

                # Accept the move only if it does not increase the exact objective
                if new_score <= cur_score + 1e-12:
                    a = a_candidate
                    cur_score = new_score
                else:
                    # try a smaller step size; if it still does not improve we stop
                    t_small = t * 0.5
                    a_candidate = (1.0 - t_small) * a + t_small * g
                    a_candidate = _project_onto_simplex(a_candidate)
                    new_score = _objective(a_candidate, fft_len)
                    if new_score < cur_score - 1e-12:
                        a = a_candidate
                        cur_score = new_score
                    else:
                        break    # cannot make progress

            # ------------------------------------------------------------------
            # 2) Very cheap stochastic fine‑tuning (mass‑transfer moves)
            # ------------------------------------------------------------------
            # Pre‑compute convolution once and reuse it for a few hundred moves.
            grad_cache = None
            move_iter = 0
            while time.monotonic() < deadline and move_iter < 4000:
                if move_iter % 30 == 0:
                    conv = _conv_fft(a, fft_len)
                    k_max = int(np.argmax(conv))
                    # gradient of the active lag (sub‑gradient of max)
                    n = a.size
                    grad = np.zeros_like(a)
                    i_start = max(0, k_max - (n - 1))
                    i_end = min(n, k_max + 1)
                    i = np.arange(i_start, i_end, dtype=int)
                    grad[i] += a[k_max - i]
                    grad *= 2.0
                    # source = index with largest contribution to the active lag
                    src = int(np.argmax(grad))
                    # choose a low‑gradient target
                    low = np.percentile(grad, 15)
                    low_cands = np.where(grad <= low)[0]
                    tgt = int(rng.choice(low_cands)) if low_cands.size else int(rng.integers(0, n))
                if a[src] <= 0.0:
                    src = int(rng.integers(0, n))
                if src == tgt:
                    tgt = int(rng.integers(0, n))

                # fractional transfer (geometric sampling)
                frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
                delta = a[src] * frac
                cand = a.copy()
                cand[src] -= delta
                cand[tgt] += delta

                cand_score = _objective(cand, fft_len)
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(f"[{time.monotonic() - start_time:.2f}s] local move #{move_iter} improved to {cand_score:.6f}")

                if cand_score <= cur_score + 1e-12:
                    a = cand
                    cur_score = cand_score

                move_iter += 1

    # ------------------------------------------------------------------
    # 5) Final bookkeeping and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform sequence of moderate size
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)   # respects evaluator limits
    final_score = evaluate_sequence(best_seq.tolist())
    total_time = time.monotonic() - start_time
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    return [float(v) for v in best_seq]
