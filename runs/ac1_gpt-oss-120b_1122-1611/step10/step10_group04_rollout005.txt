def _smooth_max_step_adam(a, fft_len, beta, alpha, m, v, t):
    conv = _conv_linear(a, fft_len)
    max_conv = conv.max()
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # compute gradient of weighted sum conv w
    # grad for weighted sum = convolution of a with w and also a with reversed w
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    # conv_aw length = fft_len; we need first n + len(w) -1 length
    conv_aw = conv_aw[: a.size + w.size - 1]
    term2 = conv_aw[: a.size]

    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: a.size + w_rev.size - 1]

    m_len = w.size
    term1 = conv_aw_rev[m_len - 1 - np.arange(a.size)]

    grad = term1 + term2
    # Adam update on log a
    m = 0.9 * m + 0.1 * grad
    v = 0.999 * v + 0.001 * (grad * grad)
    m_hat = m / (1.0 - 0.9 ** t)
    v_hat = v / (1.0 - 0.999 ** t)
    log_a = np.log(a + 1e-15) - alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
    a_new = np.exp(log_a)
    a_new = _project_onto_simplex(a_new)
    return a_new, m, v

"""
Propose a step‑function by using the discrete arcsine (Beta(½,½))
distribution.  This distribution has the remarkable property that its
self‑convolution is (almost) flat, giving a theoretical lower bound of
≈ 1 for the evaluation metric.  The routine builds the arcsine shape,
normalises it to sum = 1, evaluates its score and then performs a very
lightweight local‑search (random mass transfers) while time permits.
The final sequence is clipped to the limits required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.full_like(v, 1.0 / len(v))
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a normalised (Σa=1) vector."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set of active indices.
    For each active k, the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)      # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _hard_max_step(a: np.ndarray, fft_len: int,
                   step_factor: float, iteration: int) -> np.ndarray:
    """
    Projected sub‑gradient step for the true max objective.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])
    grad = _gradient_from_active(a, active)
    step = step_factor / np.sqrt(iteration + 1.0)
    a_new = a - step * grad
    a_new = _project_onto_simplex(a_new)
    return a_new


def _local_mass_transfer(a: np.ndarray, rng: np.random.Generator,
                         src: int, tgt: int) -> tuple[np.ndarray, int, int]:
    """
    Randomly move a small fraction of mass from ``src`` to ``tgt``.
    Returns the new vector and possibly refreshed indices.
    """
    n = a.size
    # ensure source holds some mass
    if a[src] <= 0.0:
        src = int(rng.integers(0, n))
    # avoid degenerate move
    if src == tgt:
        tgt = int(rng.integers(0, n))

    # fraction drawn log‑uniformly from [1e‑3, 5e‑2]
    frac = np.exp(rng.uniform(np.log(1e-3), np.log(5e-2)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    return a_new, src, tgt


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Build a discrete arcsine distribution (Beta(½,½)) on a power‑of‑two
    grid, optionally improve it with a few sub‑gradient / random moves,
    and return the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin for the caller's bookkeeping
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 0) Choose a problem size – power of two is convenient for FFTs.
    # ------------------------------------------------------------------
    n = 4096                     # feel free to change; larger n ≈ finer resolution
    fft_len = _next_pow2(2 * n)   # enough for a linear convolution

    # ------------------------------------------------------------------
    # 1) Construct the discrete arcsine shape.
    #    a_i ∝ 1 / sqrt( (i+1) * (n-i) ),   i = 0 … n‑1
    # ------------------------------------------------------------------
    i = np.arange(1, n + 1)                       # 1 … n
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    a /= a.sum()                                 # normalise Σa = 1

    best = a.copy()
    best_score = _score(best, fft_len)
    print(f"[0.00s] arcsine initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional light local refinement while time permits.
    # ------------------------------------------------------------------
    src = int(rng.integers(0, n))
    tgt = int(rng.integers(0, n))
    it = 0
    while time.monotonic() < deadline:
        # every few iterations recompute a good source (largest gradient)
        if it % 50 == 0:
            conv = _conv_fft(best, fft_len)
            max_b = conv.max()
            active = np.where(conv >= max_b - max_b * 1e-6)[0]
            grad_active = _gradient_from_active(best, active)
            src = int(np.argmax(grad_active))
            # pick a low‑gradient target (bottom 20 % quantile)
            thresh = np.percentile(grad_active, 20)
            low = np.where(grad_active <= thresh)[0]
            if low.size:
                tgt = int(rng.choice(low))
            else:
                tgt = int(rng.integers(0, n))

        # random directed mass‑transfer move
        cand, src, tgt = _local_mass_transfer(best, rng, src, tgt)
        cand_score = _score(cand, fft_len)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f}")

        # also occasionally try a hard‑max sub‑gradient step
        if it % 300 == 0:
            cand2 = _hard_max_step(best, fft_len, step_factor=0.4, iteration=it)
            cand2_score = _score(cand2, fft_len)
            if cand2_score < best_score - 1e-12:
                best_score = cand2_score
                best = cand2
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (hard) = {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 3) Return the best sequence (clip to evaluator limits).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)

    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
