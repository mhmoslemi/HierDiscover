def downsample(arr, n):
    L = len(arr)
    # compute bin edges
    # using integer division: each bin size = L // n, plus remainder.
    # We'll compute indices and sum accordingly.
    result = np.zeros(n, dtype=np.float64)
    bin_sz = L / n
    # Use np.floor for indices:
    for i in range(n):
        start = int(np.floor(i * bin_sz))
        end = int(np.floor((i+1) * bin_sz))
        if end > start:
            result[i] = arr[start:end].sum()
        else:
            # maybe assign zero
            result[i] = 0.0
    # handle any leftover due to rounding by adding to last bin
    remainder = L - int(np.floor(n * bin_sz))
    if remainder > 0:
        result[-1] += arr[-remainder:].sum()
    # normalize
    total = result.sum()
    if total == 0:
        result = np.full(n, 1.0/n)
    else:
        result = result / total
    return result
```

"""
Propose a candidate step‑function using a mixed
Linear‑Program (LP) + mixing strategy.

1.  Initialise a feasible probability vector `a` (sum = 1).  If a
    previous construction `height_sequence_1` is present we down‑sample it;
    otherwise a uniform start of size `n = 1024` is used.

2.  Repeatedly:
    * compute the current auto‑convolution `c = a * a` and its maximum `M`;
    * solve a linear programme
          maximise   sum(g_i)
          subject to  (a * g)_k ≤ M   for all k,
                      g_i ≥ 0
      This yields a direction `g` that does not increase the peak
      convolution.
    * normalise `g` to a probability vector and mix it with the current
      vector: `a ← (1‑t) a + t g`.  Several mixing weights `t` are tried
      (starting from 0.2 downwards); the first one that strictly lowers the
      objective is kept.
    * if no improvement is found a tiny random mass‑transfer move is tried
      (simulated‑annealing style) and, finally, a few exponentiated‑gradient
      steps are applied to fine‑tune the vector.

3.  The loop runs until the time budget (≈ 995 s) expires or no further
    progress is possible.  The best vector found is returned (values are
    clipped to the evaluator limits).

The algorithm stays entirely on the probability simplex, therefore the
evaluation function `evaluate_sequence` simplifies to `2·n·max(a*a)`.
The LP is solved with SciPy’s high‑performance “highs” backend.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise (numerical safety)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Down‑sample a longer construction to a target length `n`
# ----------------------------------------------------------------------
def _downsample(arr: np.ndarray, n: int) -> np.ndarray:
    """Simple block‑averaging down‑sample of `arr` to length `n`."""
    L = len(arr)
    if L == n:
        return arr.copy()
    # compute (possibly non‑integer) block size
    step = L / n
    out = np.zeros(n, dtype=np.float64)
    for i in range(n):
        lo = int(round(i * step))
        hi = int(round((i + 1) * step))
        out[i] = arr[lo:hi].sum()
    # catch any leftover mass due to rounding
    if hi < L:
        out[-1] += arr[hi:].sum()
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (few cheap steps)
# ----------------------------------------------------------------------
def _expgrad_refine(a: np.ndarray, rng: np.random.Generator,
                    steps: int = 6, eps_factor: float = 1e-8) -> np.ndarray:
    """Perform a handful of exponentiated‑gradient steps on `a`."""
    n = a.size
    for it in range(steps):
        conv = np.convolve(a, a)
        M = conv.max()
        eps = M * eps_factor
        active = np.where(conv >= M - eps)[0]

        # exact sub‑gradient of max‑convolution
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n, k + 1)          # exclusive
            i = np.arange(i_start, i_end, dtype=int)
            grad[i] += a[k - i]

        eta = 0.6 / np.sqrt(it + 1.0)      # diminishing step
        log_a = np.log(a + 1e-15)
        log_a -= eta * grad
        a = np.exp(log_a)
        a = _project_onto_simplex(a)
    return a


# ----------------------------------------------------------------------
# Build the linear‑program matrix for the constraint (a * g)_k ≤ M
# ----------------------------------------------------------------------
def _build_lp_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1, n)) such that
    A @ g = conv(a, g).
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes a shifted copy of `a`
    for j in range(n):
        A[j:j + n, j] = a
    return A


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)

    # ---------- 0) initialise ----------
    n = 1024  # moderate size – fast LP solves, still enough resolution
    if "height_sequence_1" in globals():
        full = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        full = np.clip(full, 0.0, None)
        total_full = full.sum()
        if total_full == 0.0:
            full[:] = 1.0 / full.size
            total_full = full.sum()
        full = full / total_full
        a = _downsample(full, n)
    else:
        a = rng.random(n)
        a = a / a.sum()

    # make sure we are on the simplex
    a = _project_onto_simplex(a)

    # bookkeeping
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    iteration = 0

    print(f"[0.00s] initial score = {best_score:.6f}")

    # ---------- main optimisation loop ----------
    while time.monotonic() < deadline and iteration < 200:
        # current convolution peak
        conv = np.convolve(a, a)
        M = conv.max()
        current_score = 2.0 * n * M   # equal to evaluate_sequence for sum=1

        # ----- 1) solve LP to find a promising direction -----
        A = _build_lp_matrix(a)                     # (2n‑1) × n
        rows = A.shape[0]
        lp_res = linprog(
            c=-np.ones(n),                               # maximise sum(g)
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=[(0.0, None)] * n,
            method='highs',
            options={'presolve': True}
        )
        if not lp_res.success:
            print(f"[{time.monotonic() - start_time:.2f}s] LP failed (iter={iteration})")
            break

        g_raw = lp_res.x
        sum_g_raw = g_raw.sum()
        if sum_g_raw <= 0.0:
            print(f"[{time.monotonic() - start_time:.2f}s] LP gave zero mass (iter={iteration})")
            break
        # normalise to a probability vector
        g = g_raw / sum_g_raw

        # ----- 2) try mixing with several step sizes -----
        improved = False
        for t in (0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01):

            conv_cand = np.convolve(a_candidate, a_candidate)
            M_cand = conv_cand.max()
            score_cand = 2.0 * n * M_cand
            if score_cand < best_score - 1e-12:
                best_score = score_cand
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] mix t={t:.3f} → new best = {best_score:.6f}")
                break   # keep the first successful mixing weight

        # ----- 3) if mixing failed, perform a tiny stochastic move -----
        if not improved:
            src = rng.integers(n)
            tgt = rng.integers(n)
            if src != tgt and a[src] > 1e-15:
                frac = np.exp(rng.uniform(np.log(0.001), np.log(0.02)))   # 0.1%–2%
                delta = a[src] * frac
                a_pert = a.copy()
                a_pert[src] -= delta
                a_pert[tgt] += delta
                a_pert = _project_onto_simplex(a_pert)
                conv_pert = np.convolve(a_pert, a_pert)
                score_pert = 2.0 * n * conv_pert.max()
                if score_pert < best_score - 1e-12:
                    best_score = score_pert
                    best_a = a_pert.copy()
                    a = a_pert
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] random move → new best = {best_score:.6f}")
                    improved = True

        # ----- 4) occasional local gradient polishing -----
        if iteration % 5 == 0:
            a = _expgrad_refine(a, rng, steps=4)

        iteration += 1
        # optional early exit if we are already under the target
        if best_score <= 1.5025:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] target reached – stopping")
            break

    # ---------- finalisation ----------
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best_a.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_a]
