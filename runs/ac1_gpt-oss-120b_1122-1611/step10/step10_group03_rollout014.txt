def _smooth_optimize(a0, time_budget, rng):
    a = a0.copy()
    n = a.size
    fft_len_conv = _next_pow2(3*n)
    # Precompute index offset for gradient extraction
    offset = n-1
    best_local = a.copy()
    best_score = _score_from_a(best_local, fft_len_conv)
    start = time.monotonic()
    iteration = 0
    gamma = 2.0  # start
    while time.monotonic() - start < time_budget:
        # compute conv and max
        conv = _conv_fft(a, fft_len_conv)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        # Update gamma gradually: increase slowly
        if iteration % 200 == 0 and iteration > 0:
            gamma = min(gamma * 1.5, 30.0)
        # compute w (weights for smooth max)
        # use scaling to avoid overflow
        scaled = conv - max_conv
        w = np.exp(gamma * scaled)  # note scaled <=0
        w_sum = w.sum()
        w /= w_sum
        # compute gradient
        # conv of w and a_rev
        a_rev = a[::-1]
        # conv length: len_w + len_rev - 1 = (2n-1) + n -1 = 3n-2
        conv_len = _next_pow2(3 * n)
        grad_full = _conv_fft_two(w, a_rev, conv_len)
        # grad_i = 2 * grad_full[i + offset]
        grad = 2.0 * grad_full[offset:offset + n]
        # step size: adapt via L based on gradient norm and distance to optimum.
        # Use simple diminishing step: step = eta / sqrt(iter+1)
        eta0 = 0.5
        step = eta0 / math.sqrt(iteration + 1.0)
        a_new = a - step * grad
        a_new = _project_onto_simplex(a_new)
        # evaluate new score
        conv_new = _conv_fft(a_new, fft_len_conv)
        max_new = conv_new.max()
        new_score = 2.0 * n * max_new
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_local = a_new.copy()
        # accept if improved else maybe accept probabilistically:
        if new_score <= cur_score:
            a = a_new
        else:
            # simple simulated annealing acceptance
            temp = max(1e-4, (time_budget - (time.monotonic() - start)) / time_budget)
            prob = math.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                a = a_new
        iteration += 1
        # maybe occasional jitter to escape plateau
        if iteration % 500 == 0:
            # jitter a bit
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.02))
            a[idx] *= factor
            a = _project_onto_simplex(a)

    return best_local, best_score

def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)

    # Get global best if exists
    best = get_initial_best()
    best = np.clip(best, 0.0, None)
    if best.sum() == 0: best = uniform.

    best = best / best.sum()
    best_score = _score_from_a(best, _next_pow2(3*best.size))
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # length candidates: near best size +- various
    length_pool = [1024, 2048, 3072, 4096, 6000, 8000, 10240]
    min_run_time = 30.0

    while time.monotonic() < deadline:
        # decide length
        if rng.random() < 0.5:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # generate initial point a0
        if n == best.size:
            a0 = best.copy()
            # random jitter to escape local minima
            jitter = rng.normal(scale=0.01, size=n)
            a0 = np.maximum(a0 + jitter, 0.0)
            a0 = _project_onto_simplex(a0)
        else:
            # pick initial type randomly
            init_type = rng.integers(0,3)
            if init_type == 0:
                a0 = rng.dirichlet(np.full(n, 0.6))
            elif init_type == 1:
                # arcsine shape
                i = np.arange(n, dtype=np.float64)
                v = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
                a0 = v / v.sum()
            else:
                # concave shape: a_i ∝ sqrt(i+1)*(n-i)
                i = np.arange(n, dtype=np.float64)
                v = np.sqrt(i + 1) * np.sqrt(n - i)
                a0 = v / v.sum()
        # allocate time for this run:
        remaining = deadline - time.monotonic()
        run_budget = min(min_run_time, remaining)
        if run_budget < 1.0:
            break

        a_opt, score_opt = _smooth_optimize(a0, run_budget, rng)

        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # final processing:
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]

"""
Gradient‑smooth optimisation of a non‑negative step function.
The objective after normalising the sum to 1 is
      2 * n * max(conv(a,a))
where conv is the linear convolution.
We explore a few parametrised families (uniform, power‑law shapes,
Dirichlet draws) and then run a smooth‑max projected‑gradient descent
that uses the full gradient (via FFT) and Euclidean projection onto the
probability simplex.  The routine keeps the best sequence seen and
returns it after the time budget (≈1000 s).  Simple jitter / annealing
steps are added to escape plateaus.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        w = np.full_like(v, 1.0 / len(v))
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_two(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two (possibly different) real vectors."""
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[: a.size + b.size - 1]
    return conv


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum(a)=1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent (core optimiser)
# ----------------------------------------------------------------------


def _smooth_optimize(a0: np.ndarray,
                     time_budget: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Runs a smooth‑max gradient descent on the simplex for at most
    ``time_budget`` seconds.  Returns the best sequence found in this
    run and its score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)          # enough for all convolutions
    offset = n - 1                       # where the gradient slice starts

    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    it = 0
    gamma = 5.0                          # start with a gentle smoothing

    while time.monotonic() - start < time_budget:
        # ---------- convolution of a with itself ----------
        conv = _conv_fft(a, fft_len)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ---------- smooth approximation of max ----------
        # weights w_i = exp(gamma * (conv_i - max_conv)), normalised
        # (all entries ≤ 0 after the shift, safe from overflow)
        scaled = conv - max_conv
        w = np.exp(gamma * scaled)
        w_sum = w.sum()
        if w_sum == 0.0:                 # numerical fallback
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ---------- gradient of the smooth max ----------
        # grad_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = a[::-1]
        grad_full = _conv_fft_two(w, a_rev, fft_len)
        grad = 2.0 * grad_full[offset:offset + n]   # slice to length n

        # ---------- step size (Polyak‑style damped) ----------
        eta = 0.5 / math.sqrt(it + 1.0)      # simple diminishing steps

        # ---------- back‑tracking line search ----------
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - (eta * factor) * grad
            cand = _project_onto_simplex(cand)
            cand_score = _score_from_a(cand, fft_len)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                break

        if not improved:
            # keep current a (no progress this iteration)
            pass

        # record global best seen in this run
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()

        # occasional schedule adaptation
        if it % 400 == 0 and it > 0:
            gamma = min(gamma * 1.3, 40.0)   # sharpen the max approximation

        # occasional jitter to escape plateaus
        if it % 3000 == 0 and it > 0:
            width = max(1, n // 200)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.015))
            jitter = a.copy()
            jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(jitter)

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Minor stochastic local refinement (optional but cheap)
# ----------------------------------------------------------------------


def _local_refine(a: np.ndarray,
                  time_budget: float,
                  rng: np.random.Generator) -> np.ndarray:
    """Tiny mass‑transfer moves + annealing; used after the main run."""
    n = a.size
    fft_len = _next_pow2(3 * n)
    start = time.monotonic()
    cur_score = _score_from_a(a, fft_len)

    while time.monotonic() - start < time_budget:
        conv = _conv_fft(a, fft_len)
        max_idx = int(np.argmax(conv))

        # find a contributing pair with the largest product at the max index
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n, max_idx + 1)
        i_range = np.arange(i_start, i_end, dtype=int)
        contributions = a[i_range] * a[max_idx - i_range]
        src_i = i_range[np.argmax(contributions)]
        src_j = max_idx - src_i
        # choose a low‑mass target
        low_candidates = np.where(a < a.mean())[0]
        if low_candidates.size == 0:
            tgt = int(rng.integers(0, n))
        else:
            tgt = int(rng.choice(low_candidates))

        # move a tiny fraction of the pair mass to the target
        delta = min(a[src_i], a[src_j]) * 0.03
        if delta > 0.0:
            cand = a.copy()
            cand[src_i] -= delta
            cand[src_j] -= delta
            cand[tgt] += 2.0 * delta
            cand = _project_onto_simplex(cand)
            cand_score = _score_from_a(cand, fft_len)
            # accept if improving, otherwise annealed acceptance
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
            else:
                temp = max(1e-4, (time_budget - (time.monotonic() - start)) /
                           time_budget)
                prob = math.exp(-(cand_score - cur_score) / temp)
                if rng.random() < prob:
                    a = cand
                    cur_score = cand_score
    return a

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise global best (use previous result if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        if best.sum() > 0.0:
            best = best / best.sum()
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        best = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_score = _score_from_a(best, _next_pow2(3 * best.size))
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # ------------------------------------------------------------------
    # Brief deterministic grid search over a simple power‑law family.
    # This consumes only a small fraction of the budget and often finds
    # a decent starting point.
    # ------------------------------------------------------------------
    grid_budget = 30.0                       # seconds reserved for grid search
    grid_end = start_time + grid_budget
    alphas = np.linspace(0.0, 0.5, 21)        # includes the uniform case (α=0)

    length_pool = [1024, 2048, 3072, 3809, 4096, 6144, 8000]

    while time.monotonic() < grid_end:
        n = int(rng.choice(length_pool))
        fft_len = _next_pow2(3 * n)

        for alpha in alphas:
            i = np.arange(n, dtype=np.float64)
            # power‑law shape: a_i ∝ ((i+1)*(n-i))^{-α}
            v = ((i + 1.0) * (n - i)) ** (-alpha)
            a = v / v.sum()

            sc = _score_from_a(a, fft_len)
            if sc < best_score - 1e-12:
                best = a
                best_score = sc
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] GRID IMPROVEMENT = {best_score:.6f} (len={n}, α={alpha:.3f})")
                if best_score <= 1.5025:
                    print("[early stop] target reached during grid")
                    final_score = evaluate_sequence(best.tolist())
                    globals()["height_sequence_1"] = best.tolist()
                    return [float(x) for x in best]

        # a few random Dirichlet starts as a backup
        for _ in range(2):
            a = rng.dirichlet(np.full(n, 0.6))
            sc = _score_from_a(a, fft_len)
            if sc < best_score - 1e-12:
                best = a
                best_score = sc
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] GRID DIRICHLET IMPROVEMENT = {best_score:.6f} (len={n})")
                if best_score <= 1.5025:
                    print("[early stop] target reached during grid")
                    final_score = evaluate_sequence(best.tolist())
                    globals()["height_sequence_1"] = best.tolist()
                    return [float(x) for x in best]

    # ------------------------------------------------------------------
    # Main optimisation loop – smooth gradient descent + tiny refinement.
    # ------------------------------------------------------------------
    min_run_time = 20.0  # seconds per restart (adjustable)

    while time.monotonic() < deadline:
        # choose length (biased to current best)
        if rng.random() < 0.6:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # ----------- build an initial point for this run -------------
        if n == best.size:
            # jitter the current best to explore a new neighbourhood
            a0 = best.copy()
            noise = rng.normal(scale=0.015, size=n)
            a0 = np.maximum(a0 + noise, 0.0)
            a0 = _project_onto_simplex(a0)
        else:
            # random family choice for a fresh start
            choice = rng.integers(0, 4)
            if choice == 0:
                a0 = rng.dirichlet(np.full(n, 0.6))
            elif choice == 1:
                # power‑law shape with a random exponent
                alpha = rng.uniform(0.02, 0.48)
                i = np.arange(n, dtype=np.float64)
                v = ((i + 1.0) * (n - i)) ** (-alpha)
                a0 = v / v.sum()
            elif choice == 2:
                # concave “sqrt” shape (more mass in the centre)
                i = np.arange(n, dtype=np.float64)
                v = np.sqrt(i + 1.0) * np.sqrt(n - i)
                a0 = v / v.sum()
            else:
                # plain uniform (unlikely to be best but cheap)
                a0 = np.full(n, 1.0 / n, dtype=np.float64)

        # ----------- run the optimiser for the allotted slice ----------
        remaining = deadline - time.monotonic()
        run_budget = min(min_run_time, remaining)
        if run_budget < 1.0:
            break

        a_opt, score_opt = _smooth_optimize(a0, run_budget, rng)

        # a short post‑processing refinement (few seconds)
        refine_budget = min(5.0, deadline - time.monotonic())
        if refine_budget > 0.5:
            a_opt = _local_refine(a_opt, refine_budget, rng)
            score_opt = _score_from_a(a_opt, _next_pow2(3 * a_opt.size))

        # ----------- update global best if improved --------------------
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # Final processing – evaluate with the provided exact function.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # make the sequence available for a potential next run
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
