"""
Gradient‑smoothed exponentiated‑gradient optimiser.

We minimise   J(a) = 2·n·max(conv(a,a))   with a_i ≥ 0, Σ a_i = 1.
The objective is scale‑invariant, so we work on the probability simplex.

The algorithm differs from the previous state‑of‑the‑art in two ways:

1.  A *smooth* surrogate of the sup‑norm is used.
    w_k ∝ exp(p·(conv_k – max(conv))) gives a soft‑max weighting of all
    convolution positions.  The resulting gradient is a weighted sum of the
    usual sub‑gradients and steers the search away from *all* large peaks,
    not only the active set.

2.  An exponentiated‑gradient (multiplicative) update with a decaying step
    size (∝ 1/√t) is applied, followed by a Euclidean projection onto the
    simplex.  When progress stalls for 30 s a small random jitter is added
    to the current best solution to escape plateaus.

The optimiser runs for the allotted time budget (≈ 1000 s) and finally
returns the best sequence it discovered.  It can start from the previously
found `height_sequence_1` if that variable is present.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of [Wang & Carreira‑Perpiñán 2013].
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def gradient_smooth(a: np.ndarray, conv: np.ndarray,
                    p: float, fft_len_g: int, conv_len: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed max:
        φ(a) = log‑sum‑exp(p·conv(a,a)) / p .
    The gradient is
        g_i = 2 * Σ_k w_k * a_{k-i},
    where w_k = exp(p·conv_k) / Σ_j exp(p·conv_j).
    """
    # soft‑max weights (max weight = 1, all others ≤ 1)
    w = np.exp(p * (conv - conv.max()))

    # reverse a for the cross‑correlation needed in the gradient
    a_rev = a[::-1]

    # convolution w (length 2n‑1) with a_rev (length n)
    # result length = (2n‑1)+n‑1 = 3n‑2; we need the slice [n‑1 : n‑1+n]
    grad_full = np.fft.irfft(
        np.fft.rfft(w, fft_len_g) * np.fft.rfft(a_rev, fft_len_g),
        n=fft_len_g
    )
    grad = 2.0 * grad_full[conv_len - 1: conv_len - 1 + a.size]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  Returns the best sequence found within the
    allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size (use the length of a previous construction
    #    if it exists, otherwise pick a comfortable size).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # for a*a
    fft_len_g = _next_pow2(conv_len + n - 1)  # for gradient (w * a_rev)

    # ------------------------------------------------------------------
    # 2. Build a diverse portfolio of initial probability vectors.
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderate & concentrated)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        # if lengths differ, truncate or pad with zeros
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                pad = np.zeros(n - v.size, dtype=np.float64)
                v = np.concatenate([v, pad])
            v = proj_simplex(v)
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best of the initial candidates.
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = conv_full_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimiser (smoothed gradient + exponentiated update).
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # keep a small safety margin for the host environment
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # parameters for the smooth max
    p_start = 1.0
    p_end = 80.0

    # exponentiated‑gradient base step size
    eta0 = 0.5

    # current state
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)

        # --------------------------------------------------------------
        # 4.1 Update “temperature” for the soft‑max (p) and for jitter.
        # --------------------------------------------------------------
        p = p_start + (p_end - p_start) * frac

        # --------------------------------------------------------------
        # 4.2 Compute convolution and objective for the current vector.
        # --------------------------------------------------------------
        conv = conv_full_fft(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # --------------------------------------------------------------
        # 4.3 Track the global best.
        # --------------------------------------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 4.4 Stagnation handling – jitter after 30 s without improvement.
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = proj_simplex(best_vec * noise)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 4.5 Compute smoothed gradient of the sup‑norm.
        # --------------------------------------------------------------
        grad = gradient_smooth(cur_vec, conv, p, fft_len_g, conv_len)

        # --------------------------------------------------------------
        # 4.6 Adaptive step size (η ∝ 1/√t) with clipping to avoid overflow.
        # --------------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1.0)
        # clip to keep exp(-η·grad) numerically stable
        max_abs = np.max(np.abs(grad))
        if max_abs * eta > 30.0:
            eta = 30.0 / max_abs

        # --------------------------------------------------------------
        # 4.7 Exponentiated‑gradient proposal.
        # --------------------------------------------------------------
        factor = np.exp(-eta * grad)
        proposal = cur_vec * factor
        proposal = proj_simplex(proposal)

        # --------------------------------------------------------------
        # 4.8 Evaluate proposal.
        # --------------------------------------------------------------
        prop_conv = conv_full_fft(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # --------------------------------------------------------------
        # 4.9 Simple Metropolis acceptance (accept improvements,
        #     otherwise accept with a tiny probability to aid escape).
        # --------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # very low temperature – we only accept if the proposal is
            # *significantly* better; otherwise stay put.
            if rng.random() < 1e-7:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random search around the best solution.
    # ------------------------------------------------------------------
    fin_start = time.monotonic()
    fin_deadline = fin_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fin_deadline:
        # small random multiplicative noise around the best vector
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        conv = conv_full_fft(cand, fft_len, conv_len)
        cand_J = float(2.0 * n * conv.max())
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            print(f"[Fine] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence (already on the simplex, non‑negative).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
