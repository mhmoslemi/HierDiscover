def grad_softmax(a, beta):
    conv = _conv_fft(a, fft_len)   # length L = 2n-1
    # compute softmax weights
    max_conv = conv.max()
    # for numerical stability, subtract max
    exp_term = np.exp(beta * (conv - max_conv))
    w = exp_term / np.sum(exp_term)
    # reverse a
    a_rev = a[::-1]
    # compute convolution w * a_rev via FFT (like _conv_fft but with appropriate lengths)
    # compute convolution length = len(w) + len(a_rev) -1 = L + n -1 = 3n - 2
    # we can use FFT length next pow2
    conv_len = _next_pow2(len(w)+len(a_rev)-1)
    w_fft = np.fft.rfft(w, n=conv_len)
    ar_fft = np.fft.rfft(a_rev, n=conv_len)
    conv_wa = np.fft.irfft(w_fft * ar_fft, n=conv_len)[: len(w)+len(a_rev)-1]
    # extract gradient segment
    start = n-1
    grad = 4.0 * n * conv_wa[start:start + n]
    return grad

"""
Optimisation of the step‑function objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

under the constraints a_i ≥ 0, sum(a)=1.

The algorithm uses an adaptive smoothed‑max gradient (p‑norm
approximation) and Adam updates on the probability simplex.
A schedule on the smoothing exponent p gradually turns the smooth
approximation into the true max, allowing the optimiser to escape
poor local minima.  After the main Adam phase a short stochastic
mass‑transfer refinement is performed.

The optimiser is run on a few different lengths (powers of two and
the length of the previous best solution).  The best sequence found
within the allotted time is returned and finally validated with
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two (different) vectors using an FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a probability‑simplex‑normalised vector."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _gradient_pnorm(a: np.ndarray, conv: np.ndarray, p: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the p‑norm (smooth max) approximation
        f(a) = ( Σ conv_k^p )^{1/p}
    w.r.t. a.  The resulting vector has the same length as ``a``.
    """
    n = a.size
    # Normalise by the current maximum to keep numbers O(1)
    maxc = conv.max()
    if maxc <= 0.0:
        # fallback – uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        # w_k ∝ conv_k^{p‑1}
        # use conv / maxc to avoid overflow/underflow
        with np.errstate(divide='ignore', invalid='ignore'):
            tmp = np.power(conv / maxc, p - 1)
        # zero out NaNs that appear when conv == 0 and p < 1
        tmp = np.where(np.isnan(tmp), 0.0, tmp)
        w_sum = tmp.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = tmp / w_sum

    # gradient = 4 n * (w * a_rev)_{slice}
    a_rev = a[::-1]
    conv_wa = _conv_fft_two(w, a_rev, fft_len)
    start = n - 1
    grad = 4.0 * n * conv_wa[start:start + n]
    return grad


def _adam_optimize_one_run(a0: np.ndarray, run_budget: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed‑max objective for a fixed length.
    Returns the best vector seen in this run and its exact score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)          # large enough for all convolutions
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12

    # smoothing exponent schedule
    p = 4.0               # start with a gentle smoothing
    p_max = 200.0
    p_inc = 0.025         # per iteration increment

    start_time = time.monotonic()
    deadline = start_time + run_budget

    # current best
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()
    best = a.copy()

    it = 0
    while time.monotonic() < deadline:
        # 1) evaluate current convolution and exact score
        conv = _conv_fft(a, fft_len)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()

        # 2) gradient w.r.t. the p‑norm smooth max
        grad = _gradient_pnorm(a, conv, p, fft_len)

        # 3) Adam update on the simplex
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a)

        # 4) increase smoothing exponent slowly
        p = min(p_max, p + p_inc)

        # 5) occasional tiny jitter to keep exploration alive
        if it % 1200 == 0:
            jitter = rng.normal(scale=0.005, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a)

    return best, best_score


def _local_refinement(a: np.ndarray, budget: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic mass‑transfer refinement.
    """
    n = a.size
    fft_len = _next_pow2(3 * n)
    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    deadline = start + budget
    it = 0
    while time.monotonic() < deadline:
        src = int(rng.integers(0, n))
        tgt = int(rng.integers(0, n))
        if src == tgt:
            continue
        # move a modest random fraction of the source mass
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.07)))
        delta = best[src] * frac
        cand = best.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        score = _score_from_a(cand, fft_len)
        if score < best_score - 1e-12:
            best_score = score
            best = cand
        it += 1

    return best, best_score


def _run_on_length(base_a: np.ndarray, n: int,
                   total_time: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Prepare an initial vector of length ``n`` (interpolating from ``base_a``
    if possible) and run the optimiser for up to ``total_time`` seconds.
    Returns the best vector and its exact score.
    """
    # initialise a0
    if base_a.size == n:
        a0 = base_a.copy()
        # add a small amount of noise to avoid stagnation
        noise = rng.normal(scale=0.005, size=n)
        a0 = np.maximum(a0 + noise, 0.0)
        a0 = _project_onto_simplex(a0)
    else:
        # linear interpolation from the previous best
        x_old = np.linspace(0.0, 1.0, base_a.size)
        x_new = np.linspace(0.0, 1.0, n)
        a0 = np.interp(x_new, x_old, base_a)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_onto_simplex(a0)

    # main Adam optimisation
    a_opt, score_opt = _adam_optimize_one_run(a0, total_time, rng)

    # short stochastic polish (about 2 % of the run time)
    refinement_time = max(0.0, total_time * 0.02)
    if refinement_time > 0.5:          # only if we have a meaningful budget left
        a_opt, score_opt = _local_refinement(a_opt, refinement_time, rng)

    return a_opt, score_opt

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine runs a few Adam‑optimisation
    stages on different sequence lengths (including the length of the
    best solution discovered so far) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Global time‑budget handling (keep a small safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    safety = 5.0
    deadline = start_time + max(0.0, float(budget_s) - safety)

    # --------------------------------------------------------------
    # Initialise the global best – reuse the previous construction if
    # available, otherwise start from the uniform distribution.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        if best.sum() > 0.0:
            best = best / best.sum()
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        # default start: uniform of length 4096
        n0 = 4096
        best = np.full(n0, 1.0 / n0, dtype=np.float64)

    best_score = _score_from_a(best, _next_pow2(3 * best.size))
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # --------------------------------------------------------------
    # Lengths we are willing to explore – we favour the current best
    # but also try a couple of larger powers of two.
    # --------------------------------------------------------------
    length_pool = [1024, 2048, 4096, 6144, 8192, 12288]

    # --------------------------------------------------------------
    # Main optimisation loop – allocate time slices to each length.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        # decide which length to work on
        if rng.random() < 0.55:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # give each run at most 300 s but also respect the remaining budget
        run_budget = min(300.0, remaining * 0.45)
        if run_budget < 1.5:           # not enough time for a meaningful run
            break

        # run optimiser for the selected length
        a_opt, score_opt = _run_on_length(best, n, run_budget, rng)

        # keep the best solution globally
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            # early stop if we have already beaten the target
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # --------------------------------------------------------------
    # Final post‑processing – enforce evaluator caps, evaluate with the
    # official function and store the result for possible reuse.
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)   # evaluator caps anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
