            conv = _conv_fft(a, fft_len)

def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 2.0)  # safety margin

    # Determine length
    if "height_sequence_1" in globals():
        base_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = base_seq.size
        base_seq = np.clip(base_seq, 0.0, None)
        base_seq_sum = base_seq.sum()
        if base_seq_sum > 0:
            base_seq = base_seq / base_seq_sum
        else:
            base_seq = np.full_like(base_seq, 1.0 / n)
    else:
        n = 4096
        base_seq = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(3 * n)

    best = base_seq.copy()
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()
    print(f"[0.0s] initial score (base) = {best_score:.6f}")

    # parameters
    min_run_time = 1.0  # seconds per restart
    max_restarts = 500  # safety cap

    restart = 0
    while time.monotonic() < deadline - min_run_time and restart < max_restarts:
        restart += 1

        # initialize a new candidate
        if best_score is not None and rng.random() < 0.25:
            # start from current best with small perturbation
            a = best.copy()
            # add log-normal perturbations to keep positivity
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * perturb
        else:
            # random Dirichlet (Gamma) draw; random shape parameter
            shape = rng.uniform(0.2, 2.0)
            a = rng.gamma(shape, 1.0, size=n)
        a = _project_onto_simplex(a)

        # ---- Phase A: smooth‑max mirror descent (short burst) ----
        beta = 1.0
        eta0 = 0.4
        smooth_iters = 400  # enough to move away from start basin
        for it in range(smooth_iters):
            # increase smoothing parameter
            beta = min(2000.0, beta * 1.03)
            eta = eta0 / np.sqrt(it + 1.0)
            a = _smooth_max_step(a, fft_len, beta, eta)

        # ---- Phase B: hard‑max sub‑gradient descent ----
        hard_iters = 150
        for it in range(hard_iters):
            a_candidate = _hard_max_step(a, fft_len, step_factor=0.4, iteration=it)
            # evaluate candidate
            conv_c = _conv_fft(a_candidate, fft_len)
            score_c = 2.0 * n * conv_c.max()
            # keep best ever
            if score_c < best_score - 1e-12:
                best_score = score_c
                best = a_candidate.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (restart {restart}) = {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached")
                    break
            # accept if not higher
            conv_cur = _conv_fft(a, fft_len)
            if score_c <= 2.0 * n * conv_cur.max():
                a = a_candidate

        # Optional tiny random mass‑transfer jitter to escape plateaus
        # (executed only if time remains)
        if time.monotonic() < deadline:
            # apply a few jitter moves
            jitter_moves = 20
            for jm in range(jitter_moves):
                src = rng.integers(0, n)
                if a[src] <= 0.0:
                    continue
                tgt = rng.integers(0, n)
                if src == tgt:
                    continue
                # transfer a small random fraction
                frac = np.exp(rng.uniform(np.log(0.001), np.log(0.02)))
                delta = a[src] * frac
                a[src] -= delta
                a[tgt] += delta
            a = _project_onto_simplex(a)

    # after loop
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total = time.monotonic() - start_time
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]

"""
Multi‑restart optimisation for the max‑convolution objective.

The algorithm repeatedly generates a random probability vector (or a
perturbed copy of the current best), performs a short burst of
exponentiated‑gradient (smooth‑max) updates followed by a short hard‑max
sub‑gradient descent, and keeps the best vector seen.  By allocating the
1000 s budget to many inexpensive restarts we can escape the local
minimum that stalled the previous optimiser and push the score below
the target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Projection only defined for 1‑D vectors.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the full 2*len(a)-1 array.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the active set
    (indices attaining the current maximum).  For each active index k
    the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)      # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Phase‑1 – smooth‑max (exponentiated‑gradient) step
# ----------------------------------------------------------------------


def _smooth_max_step(a: np.ndarray,
                     fft_len: int,
                     beta: float,
                     eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient update on the smoothed max objective.
    The vector stays on the simplex.
    """
    conv = _conv_fft(a, fft_len)            # length 2n‑1
    max_conv = conv.max()
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = (a * w) + (a * rev(w))  (restricted to first n entries)
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)[: a.size + w.size - 1]
    term2 = conv_aw[: a.size]

    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)[: a.size + w_rev.size - 1]
    m = w.size
    term1 = conv_aw_rev[m - 1 - np.arange(a.size)]

    grad = term1 + term2

    # exponentiated‑gradient (mirror descent on log‑probabilities)
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Phase‑2 – hard‑max sub‑gradient descent step
# ----------------------------------------------------------------------


def _hard_max_step(a: np.ndarray,
                   fft_len: int,
                   step_factor: float,
                   iteration: int) -> np.ndarray:
    """
    Projected sub‑gradient step for the true (non‑smoothed) max objective.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    step = step_factor / np.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:

    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin for the outer harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        s = base.sum()
        if s > 0.0:
            base /= s
        else:
            base = np.full_like(base, 1.0 / base.size)
        n = base.size
    else:
        n = 4096
        base = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # 1) Book‑keeping for the overall best solution
    # ------------------------------------------------------------------
    best = base.copy()
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()

    # ------------------------------------------------------------------
    # 2) Main restart loop
    # ------------------------------------------------------------------
    min_run_time = 1.0          # a restart needs at least this much time
    restart_id = 0

        # ------------------------------------------------------------------

            a = best.copy()

            a = a * perturb
        else:

            shape = rng.uniform(0.2, 2.0)
            a = rng.gamma(shape, 1.0, size=n)
        a = _project_onto_simplex(a)

        # ------------------------------------------------------------------
        # 2b) Phase‑1: smooth‑max exponentiated‑gradient (few hundred steps)
        # ------------------------------------------------------------------
        beta = 1.0
        eta0 = 0.4
        smooth_iters = 400

            eta = eta0 / np.sqrt(it + 1.0)
            a = _smooth_max_step(a, fft_len, beta, eta)

        # ------------------------------------------------------------------
        # 2c) Phase‑2: hard‑max sub‑gradient descent (short refinement)
        # ------------------------------------------------------------------
        hard_iters = 150
        for it in range(hard_iters):
            a_candidate = _hard_max_step(a, fft_len,
                                         step_factor=0.4,
                                         iteration=it)

            # evaluate the candidate (cheap – uses the FFT)
            conv_c = _conv_fft(a_candidate, fft_len)
            score_c = 2.0 * n * conv_c.max()

            # keep global best
            if score_c < best_score - 1e-12:
                best_score = score_c
                best = a_candidate.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (restart {restart_id}) = {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached")
                    break

            # accept the move if it does not worsen the objective
            conv_cur = _conv_fft(a, fft_len)
            if score_c <= 2.0 * n * conv_cur.max():
                a = a_candidate

        # ------------------------------------------------------------------
        # 2d) Tiny random jitter to shake out of tiny plateaus
        # ------------------------------------------------------------------
        if time.monotonic() < deadline:
            for _ in range(12):
                src = rng.integers(0, n)
                if a[src] <= 0.0:
                    continue
                tgt = rng.integers(0, n)
                if src == tgt:
                    continue

                frac = np.exp(rng.uniform(np.log(0.001), np.log(0.02)))
                delta = a[src] * frac
                a[src] -= delta
                a[tgt] += delta
            a = _project_onto_simplex(a)

        if best_score <= 1.5025:
