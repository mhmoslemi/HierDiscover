def smoothed_grad(a, tau):
    conv = np.convolve(a, a)
    # for numerical stability, subtract max
    max_c = conv.max()
    # compute weights = exp((conv - max_c)/tau)
    w = np.exp((conv - max_c) / tau)
    w /= w.sum()
    # compute gradient (2 * sum_j a_j * w_{i+j})
    # compute reversed w
    w_rev = w[::-1]
    # conv of a and w_rev
    conv_aw = np.convolve(a, w_rev)
    L = len(conv)  # 2n-1
    # gradient_i = 2 * conv_aw[L-1 - i]  # as derived
    grad = 2.0 * conv_aw[L-1 - np.arange(len(a))]
    return grad
```

Implementation details of gradient descent:

def refine(a, time_limit):
    n = len(a)
    # ensure sum 1
    a = a/ a.sum()
    # precompute FFT length
    fft_len = _next_pow2(2*n)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2=0.999; eps=1e-8
    lr = 0.1   # start
    tau_factor = 0.2
    decay_tau = 0.9995
    iter=0
    start = time.monotonic()
    while time.monotonic() - start < time_limit:
        conv = _conv_fft(a, fft_len)
        max_c = conv.max()
        tau = max_c * tau_factor
        # compute gradient smoothed
        w = np.exp((conv - max_c) / (tau + 1e-12))
        w_sum = w.sum()
        w = w / w_sum
        # compute gradient via convolution
        w_rev = w[::-1]
        conv_aw = np.convolve(a, w_rev)
        L = len(conv)
        grad = 2.0 * conv_aw[L-1 - np.arange(n)]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**(iter+1))
        v_hat = v / (1 - beta2**(iter+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # take a step in negative gradient direction
        a = a - step
        # project onto simplex
        a = _project_onto_simplex(a)
        # evaluate actual occasionally
        if iter % 50 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.monotonic()-start_time:.2f}s] refined score {score:.6f}")
        # schedule
        tau_factor *= decay_tau
        # optional schedule for lr
        lr *= 0.9999
        iter += 1
    return a
```

def refine_with_continuation(a0, time_limit):
    a = a0.copy()
    n = len(a)
    # precompute fft_len for a*a conv
    fft_len = _next_pow2(2*n)
    # For convolution of a with w_rev, we can compute each iteration using full FFT with length L = _next_pow2(n + (2*n-1) - 1) = _next_pow2(3*n - 2) ; compute each iteration.
    # We'll compute conv_a = _conv_fft(a, fft_len) each iteration anyway.
    # Determine iteration limit by time.
    start = time.monotonic()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.1
    tau_factor = 0.5
    decay_tau = 0.9995
    it = 0
    while time.monotonic() - start < time_limit:
        conv = _conv_fft(a, fft_len)  # length 2n-1
        max_c = conv.max()
        tau = max_c * tau_factor
        # compute softmax weights
        # subtract max for stability
        w = np.exp((conv - max_c) / (tau + 1e-12))
        w = w / w.sum()
        # gradient via convolution of a and w_rev
        w_rev = w[::-1]
        # compute full convolution
        size = _next_pow2(n + len(w_rev) - 1)
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(w_rev, n=size)
        conv_aw = np.fft.irfft(A * B, n=size)[:n + len(w_rev) - 1]
        L = len(conv)
        grad = 2.0 * conv_aw[L-1 - np.arange(n)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**(it + 1))
        v_hat = v / (1 - beta2**(it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_onto_simplex(a)
        # optional evaluate periodically
        if it % 100 == 0:
            current_score = 2.0 * n * conv.max() / (a.sum()**2)  # sum==1
            # but conv computed at start of loop may not be after step; recompute
            conv2 = _conv_fft(a, fft_len)
            score = 2.0 * n * conv2.max()
            # update global best via outer scope but can be returned.
        # schedule
        tau_factor *= decay_tau
        lr *= 0.9999
        it += 1
    return a

"""
propose_candidate
=================

A multi‑stage optimizer for the max‑convolution objective.
The algorithm works as follows:

1. **Generate a diverse set of initial shapes** (uniform, triangular,
   Hann, Hamming, exponential, power‑law, random Dirichlet, and the
   warm‑start ``height_sequence_1`` if it exists).  All candidates are
   normalised to sum‑to‑one and scored quickly using an FFT based
   convolution (the exact evaluation is ``2·n·max(conv)`` when the sum is 1).

2. **Pick the best candidate** and improve it by a *continuation*
   gradient descent:
   – The non‑smooth `max` is replaced by a smooth log‑sum‑exp
     approximation with a temperature τ that is gradually reduced.
   – The gradient of the smoothed objective is obtained analytically as
     a convolution of the density with the (soft‑max) weight vector.
   – An Adam‑style update is performed and the iterate is projected back
     onto the probability simplex.
   – This stage runs for the bulk of the time budget.

3. **Optional up‑scaling** – after refining at one size the solution is
   linearly interpolated to a larger length (doubling each time up to a
   configurable maximum) and refined again.  More degrees of freedom
   often give a lower maximum convolution value.

4. **Final simulated‑annealing fine‑tuning** – tiny mass‑transfer moves
   are accepted with a Metropolis rule and an exponential cooling
   schedule, allowing the search to escape shallow local minima.

The final sequence is clipped to the evaluator limits and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length next power of two.
    Returns the convolution of length 2·len(a)‑1.
    """
    n = a.size
    fft_len = _next_pow2(2 * n)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two 1‑D real arrays using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    size = _next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    return np.fft.irfft(A * B, n=size)[: len(a) + len(b) - 1]


def _fast_score(a: np.ndarray) -> float:
    """Score for a probability vector a (sum≈1) using the exact objective."""
    s = a.sum()
    if s == 0.0:
        return float("inf")
    conv = _conv_fft(a)
    return 2.0 * a.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Continuation gradient descent (smooth max)
# ----------------------------------------------------------------------


def _refine_continuation(a0: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Refine a probability vector `a0` using a smoothed max objective.
    The temperature τ is multiplied by a factor <1 at each iteration,
    whilst an Adam‑style update with projection onto the simplex is used.
    """
    a = a0.copy()
    n = a.size
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    # hyper‑parameters
    lr = 0.12                # base learning‑rate
    tau_factor = 0.5         # τ = tau_factor * max(conv)
    decay_tau = 0.9995
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_limit:
        # ----- convolution of a with itself -----
        conv = _conv_fft(a)            # length 2n‑1
        max_c = conv.max()
        # ----- smoothed weights (soft‑max) -----
        tau = max_c * tau_factor + 1e-15
        w = np.exp((conv - max_c) / tau)
        w /= w.sum()
        # ----- gradient = 2 * (a * w_rev) shifted -----
        w_rev = w[::-1]
        aw = _conv_full_fft(a, w_rev)        # length 3n‑2
        L = conv.size                         # = 2n‑1
        grad = 2.0 * aw[L - 1 - np.arange(n)]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- gradient step + projection -----
        a = a - step
        a = _project_onto_simplex(a)

        # ----- schedule -----
        tau_factor *= decay_tau
        lr *= 0.9999
        it += 1
        # occasional early‑exit if the remaining budget is tiny
        if it % 400 == 0 and time.monotonic() - start > time_limit - 0.02:
            break
    return a


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Tiny directed mass‑transfer moves with a Metropolis acceptance rule.
    The temperature decays exponentially.
    """
    rng = np.random.default_rng(424242)
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _fast_score(best)
    curr_score = best_score
    T = 0.012
    decay = 0.99985
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_limit:
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-15:
            continue
        # multiplicative random fraction in [0.001, 0.05]
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)
        cand = _project_onto_simplex(cand)

        cand_score = _fast_score(cand)
        Δ = cand_score - curr_score
        if Δ < 0 or rng.random() < np.exp(-Δ / max(T, 1e-15)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
        # occasional jitter
        if it % 6000 == 0 and it > 0:
            width = rng.integers(1, max(2, n // 80))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            jitter = a.copy()
            jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(jitter)
        T *= decay
        it += 1
    return best


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Build a pool of diverse initial candidates (different lengths)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048]               # try a few modest sizes first
    for n in lengths:
        if time.monotonic() > deadline:
            break
        candidates = []

        # warm‑start if provided
        warm = globals().get("height_sequence_1")
        if warm is not None:
            a0 = np.asarray(warm, dtype=np.float64)
            a0 = np.clip(a0, 0.0, None)
            if a0.shape[0] != n:
                # linear interpolation to the target length
                a0 = np.interp(np.linspace(0.0, 1.0, n),
                               np.linspace(0.0, 1.0, a0.shape[0]),
                               a0)
            if a0.sum() > 0.0:
                a0 = a0 / a0.sum()
                candidates.append(a0)

        # uniform
        candidates.append(np.full(n, 1.0 / n))

        # descending linear (triangular from the left)
        tri_desc = np.arange(n, 0, -1, dtype=float)
        tri_desc = tri_desc / tri_desc.sum()
        candidates.append(tri_desc)

        # symmetric triangular (peak in the centre)
        x = np.arange(n, dtype=float)
        centre = (n - 1) / 2.0
        tri_sym = 1.0 - np.abs(x - centre) / (centre + 1.0)
        tri_sym[tri_sym < 0] = 0.0
        tri_sym = tri_sym / tri_sym.sum()
        candidates.append(tri_sym)

        # Hann window
        hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
        hann = hann / hann.sum()
        candidates.append(hann)

        # Hamming window
        hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
        hamming = hamming / hamming.sum()
        candidates.append(hamming)

        # exponential decays
        for lam in (0.001, 0.005, 0.01, 0.02):
            exp_vec = np.exp(-lam * x)
            exp_vec = exp_vec / exp_vec.sum()
            candidates.append(exp_vec)

        # power‑law decays
        for p in (0.5, 1.0, 1.5, 2.0):
            pow_vec = 1.0 / (x + 1.0) ** p
            pow_vec = pow_vec / pow_vec.sum()
            candidates.append(pow_vec)

        # a few random Dirichlet draws
        for _ in range(6):
            rnd = rng.random(n)
            rnd = rnd / rnd.sum()
            candidates.append(rnd)

        # evaluate all candidates (fast scoring)
        for a in candidates:
            if time.monotonic() > deadline:
                break
            sc = _fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best init (n={n}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Continuation gradient descent on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("Failed to generate any initial candidate.")

    # Main refinement loop – optionally up‑scale after each round
    while time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        if time_left < 20.0:
            break

        # allocate the majority of the remaining budget to refinement
        refine_time = min(time_left * 0.65, 300.0)
        best_seq = _refine_continuation(best_seq, refine_time)

        # update best score after refinement
        cur_score = _fast_score(best_seq)
        if cur_score < best_score:
            best_score = cur_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] refined (n={best_seq.size}) = {best_score:.6f}")

        # decide whether to upscale (double size) and continue
        cur_n = best_seq.size
        target_n = cur_n * 2
        if target_n > 8192:            # upper bound to keep FFTs cheap
            break
        # if we still have enough time after a modest refinement at the larger size
        if deadline - time.monotonic() < 30.0:
            break

        # linear interpolation to the larger size
        old_x = np.linspace(0.0, 1.0, cur_n)
        new_x = np.linspace(0.0, 1.0, target_n)
        upscaled = np.interp(new_x, old_x, best_seq)
        upscaled = upscaled / upscaled.sum()
        best_seq = upscaled
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] up‑scaled to n={target_n}")

    # ------------------------------------------------------------------
    # 3) Final local optimisation with simulated annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        best_seq = _simulated_annealing(best_seq, remaining)

    # ------------------------------------------------------------------
    # 4) Return the best sequence (final official scoring)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_seq]
