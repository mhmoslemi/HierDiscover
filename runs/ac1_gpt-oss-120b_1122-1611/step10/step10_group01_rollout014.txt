"""
search for a non‑negative step function that minimizes ``evaluate_sequence``.

Approach
--------
1. **Broad initialisation** – generate many parametric families
   (uniform, linear, quadratic, exponential, power‑law increasing / decreasing,
   random Dirichlet draws).  The vector is always normalised to sum = 1
   (the objective is homogeneous, so the scale does not matter).

2. **Fast pick of the best seed** – evaluate the exact score
   2·n·max(conv) for every seed (using an FFT based linear convolution).

3. **Refinement**
   * **Projected sub‑gradient descent** on the exact max‑convolution.
     The sub‑gradient for the active lags is computed analytically.
     A simple diminishing step‑size schedule is used and every iterate is
     projected back onto the probability simplex.
   * **Simulated‑annealing local moves** – after the deterministic phase a
     large‑budget stochastic phase runs.  Each move transfers a random
     fraction of mass from one index to another; moves that reduce the
     true objective are always accepted, otherwise they are accepted with
     probability exp(−Δ/T) where the temperature T cools exponentially.

4. **Time budget** – the whole routine stops a few seconds before the
   given budget and returns the best sequence seen so far.

All helper functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level only)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        w = np.full_like(v, 1.0 / len(v))
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise (might be off by 1e‑12)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT of length ``fft_len``.
    Returns the (2*len(a)-1) vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a∗a)[k]`` for the set ``active`` of
    indices attaining the current maximum.
    For each active lag k,
        ∂/∂a_i   = 2·a_{k‑i}   (if 0 ≤ k‑i < n, else 0)
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum = 1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _smooth_max_gradient(a: np.ndarray, p: int, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth approximation
        F_p(a) = ( Σ_k (a∗a)[k]^p )^{1/p}
    (ignoring the constant factor 2 n).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)               # length 2n‑1
    eps = 1e-12
    w = np.maximum(conv, eps) ** (p - 1)       # weight vector
    L = fft_len
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)   # length L
    offset = len(w) - 1                         # 2n‑2
    grad = 2.0 * corr[offset : offset + n]     # slice of length n
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    safety = 5.0                     # seconds left unused as a safety margin
    deadline = start_time + max(0.0, float(budget_s) - safety)

    # --------------------------------------------------------------
    # Choose a length for the vector.
    # Prefer the length of a previous solution (if any) – it is usually
    # already reasonably large.  Fall back to a power‑of‑two for FFT
    # convenience.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        # try a few powers of two and pick the largest that still fits
        # comfortably into the time budget.
        n = 2048
    fft_len = _next_pow2(3 * n)          # enough for all convolutions

    # --------------------------------------------------------------
    # 0) Build a wide collection of seed vectors (all summed to 1).
    # --------------------------------------------------------------
    seeds = []

    # uniform (baseline)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    seeds.append(uniform)

    # linear (increasing) – a_i ∝ i+1
    idx = np.arange(1, n + 1, dtype=np.float64)
    lin_inc = idx.copy()
    lin_inc /= lin_inc.sum()
    seeds.append(lin_inc)

    # linear decreasing – mirror of the above
    lin_dec = lin_inc[::-1]
    seeds.append(lin_dec)

    # quadratic increasing / decreasing
    quad_inc = idx ** 2
    quad_inc /= quad_inc.sum()
    seeds.append(quad_inc)
    seeds.append(quad_inc[::-1])

    # exponential families
    for lam in [0.001, 0.005, 0.01, 0.02]:
        exp_inc = np.exp(lam * idx)
        exp_inc /= exp_inc.sum()
        seeds.append(exp_inc)
        seeds.append(exp_inc[::-1])

    # power‑law families (both directions)
    for alpha in [0.2, 0.5, 0.8, 1.2, 1.5, 2.0]:
        dec = (1.0 / (idx ** alpha))
        dec /= dec.sum()
        seeds.append(dec)
        inc = (idx ** alpha)
        inc /= inc.sum()
        seeds.append(inc)

    # a few random Dirichlet draws (concentration < 1 gives sparsity)
    for _ in range(6):
        d = rng.dirichlet(np.full(n, 0.5))
        seeds.append(d)

    # small random perturbations of the uniform vector – they help
    # escape the perfect symmetry of the uniform seed.
    for _ in range(8):
        perturb = uniform + rng.normal(0.0, 0.02, size=n)
        perturb = np.clip(perturb, 0.0, None)
        perturb /= perturb.sum()
        seeds.append(perturb)

    # -----------------------------------------------------------------
    # 1) Pick the best initial candidate (exact evaluation)
    # -----------------------------------------------------------------
    best = None
    best_score = np.inf
    for cand in seeds:
        cand = np.clip(cand, 0.0, None)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        sc = _score_from_a(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    if best is None:
        best = uniform.copy()
        best_score = _score_from_a(best, fft_len)

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:.2f}s] initial best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2) Projected sub‑gradient descent on the exact max‑convolution.
    #    Simple diminishing step size schedule; stop early if out of time.
    # -----------------------------------------------------------------
    a = best.copy()
    n_iters = 0
    max_iters = 2500
    step0 = 0.5          # initial step size (will be scaled down)
    while n_iters < max_iters and time.monotonic() < deadline:
        n_iters += 1
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        active = np.where(conv == max_val)[0]
        grad = _gradient_from_active(a, active)

        # diminishing step size (1/√t) – ensures convergence of sub‑gradient
        step = step0 / np.sqrt(n_iters)
        a_new = a - step * grad
        a_new = _project_onto_simplex(a_new)

        conv_new = _conv_fft(a_new, fft_len)
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score - 1e-12:
            best_score = new_score
            best = a_new.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] sub‑grad iter={n_iters} new best = {best_score:.6f}")

        # accept the step if it does not increase the true objective
        if new_score <= 2.0 * n * max_val:
            a = a_new
        # otherwise keep the old a (sub‑gradient may oscillate)

    # -----------------------------------------------------------------
    # 3) Simulated‑annealing style stochastic local refinement.
    #    Moves: transfer a random fraction of mass from one index to another.
    # -----------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    iter_sa = 0
    # temperature schedule: start relatively hot, exponential cooling
    T0 = best_score * 0.04
    tau = 0.9995                     # cooling factor per move
    T = T0

    while time.monotonic() < deadline:
        # propose a tiny mass transfer
        src = int(rng.integers(0, n))
        if a[src] <= 0.0:
            continue
        tgt = int(rng.integers(0, n))
        if src == tgt:
            continue
        # draw a fraction from a log‑uniform distribution (0.1 %–5 %)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        conv_cand = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()

        # acceptance rule (Metropolis)
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / T)
            accept = rng.random() < prob

        if accept:
            a = cand
            cur_score = cand_score
            # keep global best if it improves
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] SA move #{iter_sa} new best = {best_score:.6f}")

        # cooling
        T *= tau
        iter_sa += 1

    # -----------------------------------------------------------------
    # 4) Finalisation and return
    # -----------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)   # respect evaluator limits
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # Return plain Python list of floats (as required by the evaluator)
    return [float(x) for x in best]
