"""
propose_candidate – hybrid optimisation

* initialise a short (≈2000) non‑negative step‐function from several
  parametric families (uniform, Gaussian, power‑law, sparse Dirichlet)
  and keep the best (lowest exact objective).

* minimise the smooth ‑max surrogate of the convolution with an Adam
  optimiser.  p (the smoothness exponent) is increased from 4 to 1024,
  each stage performing many iterations while the vector stays on the
  probability simplex (∑a_i=1).

* linear‑programming refinement:  for the current f compute
  M = max(f∗f).  Solve the LP

        maximise   Σ g_j
        subject to  (f∗g)_k ≤ M      for every lag k
                    g_j ≥ 0 , g_j ≤ 1000

  (the constraints are linear because f is fixed).  The optimal g
  increases the total mass while keeping the convolution bound.
  Mix f and g by a line‑search on the mixing factor α∈{0.1,…,1.0};
  accept the best mixture that lowers the true objective.  Repeat the
  LP step several times until the time budget is exhausted.

* occasional random jitter prevents stagnation.

The whole routine respects the 1000 s budget, prints progress and
returns the best list of heights found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(v, 1.0 / len(v))
        else:
            w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact objective 2·n·max(a∗a)/(Σa)² (no clipping)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a ** 2))


def _smooth_max_gradient(a: np.ndarray, p: int, fft_len: int) -> np.ndarray:
    """Gradient of (∑ (a∗a)_k^p)^{1/p} (ignoring the constant factor)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    eps = 1e-12
    w = np.maximum(conv, eps) ** (p - 1)          # shape (2n‑1,)
    L = fft_len
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    offset = len(w) - 1
    grad = 2.0 * corr[offset: offset + n]
    return grad


def _build_conv_constraints(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that
        A @ g  =  (f ∗ g)   (linear in g).
    Row k contains the coefficients f_{k‑j} (zero outside the range).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes f[0…n‑1] to rows j…j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A

    print(f"Finished in {total_elapsed:.2f}s – final exact score = {final_score:.6f}")

    return [float(x) for x in best]
