def build_toeplitz_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # For each b_i coefficient, row index k = i + j where j indexes a_j.
        # So for each j from 0 to n-1:
        # row = i + j
        # coeff = a[j]
        # Append.
        start = i
        rows.extend(range(start, start + n))
        cols.extend([i] * n)
        data.extend(a.tolist())
    # After loops, rows length = n^2, etc.
    A = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A.tocsr()

"""
Mirror‑descent search for a step‑function that minimises `evaluate_sequence`.

The algorithm:
   * start from a random point on the simplex (or from the global best if
     available);
   * repeatedly compute the full convolution a*a (FFT) and identify the
     top‑k largest convolution entries;
   * build a sub‑gradient of the max‑convolution by summing the contributions
     of those active entries;
   * take an additive step against that gradient, project back onto the
     probability simplex;
   * occasionally add a small random jitter to escape plateaus.
The best vector seen is returned (clipped to the evaluator’s limits).
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv

def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max_k (a*a)[k]`` for a set of active indices.
    For each active index k, the contribution to ∂max/∂a_i is a_{k-i}
    wherever the index is defined.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.full_like(v, 1.0 / len(v))
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(v, 1.0 / len(v))
        else:
            w /= w_sum
    return w

def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Proxy objective: 2 * n * max(conv) – the denominator (sum) is 1."""
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Mirror‑descent optimiser for ``evaluate_sequence``.
    Returns a list of non‑negative heights (sum is normalised to 1
    internally; the evaluator later rescales automatically).
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 0) initialise the search vector
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        # choose a moderate size – enough to beat the previous bound
        n = 2048
        a0 = rng.random(n)
        a0 = _project_onto_simplex(a0)

    # if we have a starting vector that is longer than our working size,
    # simply truncate / pad to the chosen length.
    n = a0.size
    fft_len = _next_pow2(2 * n)

    a = a0.copy()

    # -----------------------------------------------------------------
    # bookkeeping
    # -----------------------------------------------------------------
    cur_conv = _conv_fft(a, fft_len)
    cur_score = _objective_from_conv(cur_conv, n)

    best = a.copy()
    best_score = cur_score
    print(f"[0.00s] start score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # optimisation loop
    # -----------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    iteration = 0
    top_k = 5          # number of active convolution entries used for the gradient
    eta0 = 0.7         # initial step length
    jitter_interval = 1200   # introduce jitter every this many iterations
    jitter_scale = 0.02      # relative jitter magnitude

    while time.monotonic() < deadline:
        # compute gradient from the current top‑k peaks
        if top_k >= cur_conv.size:
            active_idx = np.arange(cur_conv.size)
        else:
            # note: np.argpartition is O(n) and returns unsorted indices
            active_idx = np.argpartition(-cur_conv, top_k)[:top_k]

        grad = _gradient_from_active(a, active_idx)

        # step size decays like 1/sqrt(t)
        eta = eta0 / np.sqrt(iteration + 1)

        # additive sub‑gradient step followed by simplex projection
        a_candidate = a - eta * grad
        a_candidate = _project_onto_simplex(a_candidate)

        # evaluate candidate (fast convolution)
        cand_conv = _conv_fft(a_candidate, fft_len)
        cand_score = _objective_from_conv(cand_conv, n)

        # accept only if improvement; otherwise keep the old vector
        if cand_score < cur_score:
            a = a_candidate
            cur_conv = cand_conv
            cur_score = cand_score

        # keep globally best
        if cand_score < best_score - 1e-12:
            best = a_candidate.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f} (iter={iteration})")
            # early exit if already below the target
            if best_score <= 1.5025:
                print("[early stop] reached target bound")
                break

        # -----------------------------------------------------------------
        # occasional random jitter to escape flat regions
        # -----------------------------------------------------------------
        if iteration % jitter_interval == 0 and iteration > 0:
            # multiplicative jitter: a_i <- a_i * (1 + eps_i)
            eps = rng.normal(scale=jitter_scale, size=n)
            a_jitter = a * (1.0 + eps)
            a_jitter = np.clip(a_jitter, 0.0, None)
            a_jitter = _project_onto_simplex(a_jitter)
            jitter_conv = _conv_fft(a_jitter, fft_len)
            jitter_score = _objective_from_conv(jitter_conv, n)

            if jitter_score < best_score - 1e-12:
                best = a_jitter.copy()
                best_score = jitter_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (jitter) = {best_score:.6f}")

            # adopt the jittered vector if it does not worsen the current score
            if jitter_score <= cur_score:
                a = a_jitter
                cur_conv = jitter_conv
                cur_score = jitter_score

        iteration += 1

    # -----------------------------------------------------------------
    # finalise – clip to the evaluator’s admissible range and report score
    # -----------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]

    the simplex;
  * occasionally add a small multiplicative jitter to escape flat regions.

The best vector seen is returned (clipped to the evaluator’s limits).  This
approach differs from the previous simulated‑annealing code by using a
deterministic projected sub‑gradient step instead of single‑pair mass
