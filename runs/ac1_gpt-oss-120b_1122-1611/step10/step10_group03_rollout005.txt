"""
A fresh optimiser for the “max convolution” problem.

The idea
---------
* Work on the probability simplex (∑a_i = 1, a_i ≥ 0).  
* Approximate the L∞‑norm of the convolution a∗a by the
  Lp‑norm   (‖c‖_p = (∑ c_k^p)^{1/p})   with a slowly increasing p.
  The gradient of ‖c‖_p (or of its p‑th power) is a *weighted* sum of the
  exact sub‑gradients ∂(a∗a)_k/∂a_i = 2·a_{k‑i}.  The weights are
  w_k ∝ c_k^{p‑1} and are computed in a numerically‑stable way
  (log‑space exponentiation).
* Perform projected additive gradient descent with a simple back‑tracking
  line‑search.  The step‑size decays like 1/√t and is re‑scaled automatically
  by the line‑search.
* Occasionally inject a tiny Gaussian jitter and increase p (making the
  objective sharper) – this mimics simulated‑annealing / continuation.
* Several short random‑restart runs (different lengths and initialisations)
  are executed until the time budget is exhausted.  The best sequence seen
  is finally evaluated with the supplied ``evaluate_sequence`` routine.

The algorithm is fully vectorised, uses only FFTs (O(n log n)) and works
well for n up to a few ten‑thousands within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a via FFT.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _cross_corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation of w (length 2n‑1) with a (length n).
    Returns a vector of length n where entry i = Σ_k w_k·a_{k‑i}
    (out‑of‑range indices are treated as 0).
    """
    # reverse w, then perform a linear convolution with a
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    a_fft = np.fft.rfft(a, n=fft_len)
    full = np.fft.irfft(w_rev_fft * a_fft, n=fft_len)[: len(w_rev) + len(a) - 1]
    L = len(w_rev)          # = 2·n‑1
    # grad[i] = full[L‑1‑i] for i = 0 … n‑1
    return full[L - 1 - np.arange(len(a))]


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum(a)=1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of the p‑norm surrogate (p≥2, grows during optimisation)
# ----------------------------------------------------------------------


def _grad_p_norm(a: np.ndarray, p: float, fft_len: int):
    """
    Compute the gradient of  (∑_k (a∗a)_k^p)   w.r.t. a.
    Returns (grad, conv) where conv = a∗a (full linear convolution).
    The gradient is   ∂/∂a_i = 2·p·Σ_k w_k·a_{k‑i}
    with weights w_k ∝ (conv_k)^{p‑1}.
    """
    n = a.size
    # convolution a*a
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

    # stable computation of w_k ∝ conv_k^{p‑1}
    eps = 1e-300
    log_conv = np.log(conv + eps)
    max_log = log_conv.max()
    # w_k = exp((p‑1)*(log_conv‑max_log))  = (conv/conv_max)^{p‑1}
    w = np.exp((p - 1.0) * (log_conv - max_log))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation w with a gives Σ_k w_k·a_{k‑i}
    corr = _cross_corr_fft(w, a, fft_len)

    grad = 2.0 * p * corr
    return grad, conv


# ----------------------------------------------------------------------
# One optimisation run (given a starting vector and a time budget)
# ----------------------------------------------------------------------


def _optimize_one_run(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Gradient‑descent with a p‑norm continuation scheme.
    Returns the best vector seen inside the run and its exact score.
    """
    a = a0.copy()
    n = a.size
    # one FFT length works for both convolution (2n‑1) and cross‑corr (3n‑2)
    fft_len = _next_pow2(3 * n)

    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    deadline = start + time_budget

    # continuation parameters
    p = 2.0               # start with a smooth surrogate
    p_max = 48.0
    p_inc_every = 300    # iterations after which p is increased
    step0 = 0.6           # base step size (will be divided by √t)

    it = 0
    while time.monotonic() < deadline:
        # ----- gradient of the current surrogate -----
        grad, conv = _grad_p_norm(a, p, fft_len)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()

        # ----- simple back‑tracking line‑search -----
        step = step0 / math.sqrt(it + 1.0)
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_onto_simplex(cand)
            cand_conv = _conv_fft(cand, fft_len)
            cand_score = 2.0 * n * cand_conv.max()
            if cand_score < cur_score - 1e-12:
                a = cand
                cur_score = cand_score
                improved = True
                break

        if not improved:
            # occasional uphill move (tiny probability) to escape plateaus
            if rng.random() < 0.005:
                a = cand  # accept the last tried candidate anyway

        # ----- continuation & occasional jitter -----
        if it % p_inc_every == 0 and it > 0:
            p = min(p_max, p * 1.22)          # geometric increase

        if it % 1200 == 0 and it > 0:
            # small Gaussian perturbation, then re‑project
            jitter = rng.normal(scale=0.006, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a)

        it += 1

    # final exact score of the run’s best point
    final_score = _score_from_a(best, fft_len)
    return best, final_score


# ----------------------------------------------------------------------
# Public entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine runs many short random‑restart
    optimisations that minimise a smooth Lp‑norm surrogate of the
    convolution peak, gradually sharpening the surrogate (increasing p).
    The best sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time handling (leave a small safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Global best (reuse previous run if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        if best.sum() > 0.0:
            best = best / best.sum()
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        # start from a modest‑size uniform distribution
        best = np.full(3072, 1.0 / 3072, dtype=np.float64)

    best_score = _score_from_a(best, _next_pow2(3 * best.size))
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # ------------------------------------------------------------------
    # Lengths that will be explored (powers of two up to ~8000)
    # ------------------------------------------------------------------
    length_pool = [1024, 2048, 3072, 4096, 6144, 8192]

    # ------------------------------------------------------------------
    # Main random‑restart loop
    # ------------------------------------------------------------------
    min_run_time = 18.0           # seconds per restart (adjusted for budget)
    run_counter = 0
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < min_run_time:
            break

        # ---- pick a length (biased towards the current best) ----
        if rng.random() < 0.55:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # ---- initialise a0 -------------------------------------------------
        if n == best.size:
            # jitter the global best a little
            a0 = best.copy()
            a0 += rng.normal(scale=0.018, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_onto_simplex(a0)
        else:
            # random Dirichlet start – α < 1 favours sparsity (more exploration)
            alpha = np.full(n, 0.45)
            a0 = rng.dirichlet(alpha).astype(np.float64)

        # ---- run the optimiser --------------------------------------------
        run_budget = min(min_run_time, remaining)
        a_opt, score_opt = _optimize_one_run(a0, run_budget, rng)

        # ---- update global best -------------------------------------------
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} "
                  f"(len={best.size})  run#{run_counter}")

        run_counter += 1

        # early stop if we beat the target comfortably
        if best_score <= 1.5025:
            print("[early stop] target reached")
            break

    # ------------------------------------------------------------------
    # Final post‑processing and evaluation
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # evaluator caps values anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # store for a possible subsequent run
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
