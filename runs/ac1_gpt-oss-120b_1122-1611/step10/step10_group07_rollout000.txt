"""
Propose a step‑function sequence that minimises the evaluation function.
The method is a two‑stage optimisation:

1. **Simulated‑annealing (SA) global search** – a mass‑transfer move
   repeatedly shifts a random fraction of probability mass from a
   source index to a target index.  The move is accepted
   according to a temperature schedule, allowing occasional uphill
   steps and thus escaping local minima.  Convolution maxima are
   evaluated by FFT for speed.

2. **Exponential‑gradient refinement** – after the SA budget is exhausted,
   a few passes of exponentiated‑gradient descent (the same sub‑gradient as
   in the reference implementation) are applied to polish the best
   solution.

The algorithm works for any length; it re‑uses the previously best
sequence (`height_sequence_1`) if available, otherwise starts from a
uniform distribution of size 2048.  All helper functions are defined at
module level (no closures).  Progress and timing are printed, and the
best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution(arr: np.ndarray, fft_len: int) -> float:
    """
    Compute max_{k} (arr * arr)[k] using an FFT of length ``fft_len``.
    The convolution length is 2*len(arr)-1.
    """
    af = np.fft.rfft(arr, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * arr.size - 1]
    return float(conv.max())


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] for the set of active indices.
    For each active index k the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    The search consists of a simulated‑annealing phase followed by a
    short exponential‑gradient refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` and normalise to sum = 1
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 = a0 / s0
        else:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # default size – large enough to give a good trade‑off, small
        # enough to keep FFT cheap.
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a0
        n = a.size

    # --------------------------------------------------------------
    # 1) Prepare FFT parameters
    # --------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)   # length for linear convolution

    # --------------------------------------------------------------
    # 2) Helper to compute current objective (sum(a)=1 ⇒ factor 2·n)
    # --------------------------------------------------------------
    def _score(arr: np.ndarray) -> float:
        max_b = _max_convolution(arr, fft_len)
        return 2.0 * n * max_b

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    print(f"[0.00s] initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase A – Simulated annealing (≈70 % of remaining time)
    # --------------------------------------------------------------
    # temperature schedule (geometric cooling)
    init_temp = cur_score * 0.03          # 3 % of the current score
    min_temp = cur_score * 1e-4
    temp = init_temp
    cooling_rate = 0.9995                # 0.05 % decay per iteration

    it = 0
    # track iterations without improvement for optional restart logic
    last_improve_iter = 0

    while time.monotonic() < deadline:
        # ----- propose a mass‑transfer move -----
        # pick source proportional to its current mass (more likely to move from big piles)
        src = int(rng.choice(n, p=a / a.sum()))
        tgt = int(rng.integers(0, n))
        if src == tgt:
            continue

        # random fraction of src's mass (tiny to moderate)
        frac = rng.uniform(0.001, 0.25)   # 0.1 % – 25 %
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta

        # ----- evaluate -----
        cand_score = _score(a_cand)

        # ----- Metropolis acceptance -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability exp( -Δ / T )
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f} (iter={it})")
                last_improve_iter = it
                if best_score <= 1.5025:
                    print("[early stop] target reached during SA")
                    break

        # ----- temperature update -----
        temp = max(min_temp, temp * cooling_rate)

        it += 1
        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA iter={it}, cur={cur_score:.6f}, best={best_score:.6f}")

        # optional simple restart if no improvement for long time
        if it - last_improve_iter > 200_000:
            # re‑initialise a little bit to shake out of a flat region
            a = best.copy()
            # add a tiny random jitter
            jitter = rng.normal(0.0, 0.003, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _score(a)
            last_improve_iter = it
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] restart after stagnation (iter={it})")

    # --------------------------------------------------------------
    # 4) Phase B – Exponentiated‑gradient refinement (remaining time)
    # --------------------------------------------------------------
    # Use the best solution found by SA as the starting point.
    a = best.copy()
    it_exp = 0
    while time.monotonic() < deadline:
        conv_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(conv_fft * conv_fft, n=fft_len)[: 2 * n - 1]
        max_b = float(conv.max())
        eps = max_b * 1e-7
        active = np.where(conv >= max_b - eps)[0]
        # limit active set size for speed
        if active.size > 60:
            active = rng.choice(active, size=60, replace=False)

        grad = _gradient_from_active(a, active)

        eta = 0.5 / np.sqrt(it_exp + 1.0)      # diminishing step
        log_a = np.log(a + 1e-15) - eta * grad
        a = np.exp(log_a)
        a /= a.sum()

        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (exp‑grad) = {best_score:.6f} (iter={it_exp})")
            if best_score <= 1.5025:
                print("[early stop] target reached during exp‑grad")
                break

        it_exp += 1
        if it_exp % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] exp‑grad iter={it_exp}, cur={cur_score:.6f}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
