def conv_full_fft(x, y):
    L = len(x) + len(y) - 1
    fft_len = 1 << ((L - 1).bit_length())
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:L]
    return conv

def propose_candidate(...):
    rng = np.random.default_rng(seed)
    n = 4096
    conv_len = 2 * n - 1
    fft_len1 = 1 << ((conv_len - 1).bit_length())
    fft_len2 = 1 << ((3*n - 2 - 1).bit_length())   # for rev conv
    # initialize a as Dirichlet sample to break symmetry
    a = rng.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_J = 2 * n * np.max(_conv_full_fft(a, a))   # compute via conv_fft with small lengths

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)
    last_improve = start_time

    while time.monotonic() < deadline:
        # compute autocorr conv
        A1 = np.fft.rfft(a, fft_len1)
        conv_full = np.fft.irfft(A1 * A1, n=fft_len1)[:conv_len]
        max_conv = conv_full.max()
        cur_J = 2 * n * max_conv
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_J:.6f}")

        # stagnation restart
        if time.monotonic() - last_improve > 20.0:
            # jitter
            noise = np.exp(rng.normal(0.0, 0.03, size=n))
            a = best_a * noise
            a = proj_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] Restart due to stagnation")
            continue

        # compute softmax weights
        beta = 5.0 + 20.0 * (t_iter / 5000.0)  # increase gradually, cap later
        # optional clamp beta max:
        if beta > 80.0:
            beta = 80.0
        # stable softmax
        maxc = conv_full.max()
        exp_vals = np.exp(beta * (conv_full - maxc))
        w = exp_vals / exp_vals.sum()

        # reverse w
        rev_w = w[::-1]

        # compute gradient via convolution
        A2 = np.fft.rfft(a, fft_len2)
        B = np.fft.rfft(rev_w, fft_len2)
        conv_rev = np.fft.irfft(A2 * B, n=fft_len2)[:3*n-2]
        offset = conv_len - 1  # = 2*n-2
        grad = 4.0 * n * conv_rev[offset:offset + n]

        # Adam update
        t_iter += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1 - 0.9**t_iter)
        v_hat = v / (1 - 0.999**t_iter)
        step = 0.5 * m_hat / (np.sqrt(v_hat) + 1e-8)    # step size
        a = a - step
        # project back onto simplex
        a = proj_simplex(a)
    # end while

    # after loop, return best
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total
    return best_a.tolist()

"""
Gradient‑softmax optimiser for the non‑negative step‑function problem.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) which
removes the scale invariance of the objective.
For a given vector a we compute the full auto‑convolution a*a with a real FFT.
Instead of the non‑smooth max we use a temperature‑controlled soft‑max
(σ(β·conv)) which yields a smooth approximation
    J_β(a) = 2·n·∑_k w_k·conv_k ,   w_k = exp(β·conv_k)/∑_ℓ exp(β·conv_ℓ) .
Its gradient can be written in closed form as a cross‑correlation and is
computed again with FFTs.  An Adam‑style (momentum + adaptive step size)
projected‑gradient descent moves a away from the currently active
convolution indices.  The temperature β is increased linearly so that the
approximation becomes sharper over time.  Stagnation is cured by a small
multiplicative jitter of the current best solution.

The method is deliberately different from the previous sub‑gradient / SA
code and typically reaches lower values of the true objective
    evaluate_sequence(a)  = 2·n·max(a*a) .
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Length of the result is len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    fft_len = _next_pow2(L)
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:L]
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    `evaluate_sequence`.  Returns the best sequence found within the time budget.

    The optimiser works on the simplex, uses a temperature‑controlled soft‑max
    of the auto‑convolution and performs Adam‑style projected gradient descent.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    # A power‑of‑two size gives fast FFTs; 4096 is a good compromise.
    n = 4096
    conv_len = 2 * n - 1                     # length of a*a
    fft_len_conv = _next_pow2(conv_len)      # FFT length for the auto‑convolution
    fft_len_corr = _next_pow2(3 * n - 2)     # length needed for a ⋆ w (correlation)

    # --------------------------------------------------------------
    # 2. Initial candidates – keep the best as the start point
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few Dirichlet draws (both dense and sparse)
    for alpha in (0.3, 0.6, 1.0, 2.0):
        for _ in range(3):
            cand = rng.dirichlet(np.ones(n) * alpha)
            candidates.append(cand)

    # use the previously found construction if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = 1.0 / n
        else:
            v = v / v.sum()
        # pad / truncate to the current length
        if len(v) < n:
            v = np.pad(v, (0, n - len(v)), constant_values=0.0)
        elif len(v) > n:
            v = v[:n]
        candidates.append(v)

    # pick the candidate with the smallest true objective (fast surrogate)
    best_vec = None
    best_J = np.inf
    for cand in candidates:
        conv = conv_fft(cand, cand)[:conv_len]
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best (fast eval) J ≈ {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Optimisation parameters
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(best_vec)
    v = np.zeros_like(best_vec)

    # Soft‑max temperature schedule – start modest, finish aggressive
    beta_start, beta_end = 4.0, 80.0          # β controls how sharp the soft‑max is
    max_iters = 200_000                       # a very generous upper bound

    # Stagnation handling
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    last_improve = start_time
    iteration = 0

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline and iteration < max_iters:
        # ---- 4.1  Compute auto‑convolution and the true objective ----
        A_fft = np.fft.rfft(best_vec, fft_len_conv)
        conv_full = np.fft.irfft(A_fft * A_fft, n=fft_len_conv)[:conv_len]
        max_conv = conv_full.max()
        cur_J = float(2.0 * n * max_conv)

        # ---- 4.2  Keep the global best (using the exact objective) ----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_solution = best_vec.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- 4.3  Stagnation: jitter and continue --------------------
        if time.monotonic() - last_improve > 20.0:
            # Small multiplicative Gaussian perturbation around the best seen point
            noise = np.exp(rng.normal(0.0, 0.025, size=n))
            best_vec = best_solution * noise
            best_vec = proj_simplex(best_vec)
            last_improve = time.monotonic()
            continue

        # ---- 4.4  Soft‑max weights ---------------------------------
        # Linear schedule for β (sharpening)
        beta = beta_start + (beta_end - beta_start) * (iteration / max_iters)
        # Prevent overflow – subtract the max before exponentiation
        shifted = conv_full - conv_full.max()
        exp_vals = np.exp(beta * shifted)
        w = exp_vals / exp_vals.sum()                # ∑ w = 1, w ≥ 0

        # ---- 4.5  Gradient of the smooth surrogate -----------------
        # correlation a ⋆ w = Σ_i a_i w_{i + lag}
        rev_w = w[::-1]                               # reversal turns correlation into convolution
        A2_fft = np.fft.rfft(best_vec, fft_len_corr)
        B_fft = np.fft.rfft(rev_w, fft_len_corr)
        corr_full = np.fft.irfft(A2_fft * B_fft, n=fft_len_corr)[:3 * n - 2]
        offset = conv_len - 1                         # 2·n‑2
        grad = 4.0 * n * corr_full[offset:offset + n]  # ∇J_β

        # ---- 4.6  Adam update (projected onto the simplex) -------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = 0.4 * m_hat / (np.sqrt(v_hat) + eps)   # learning‑rate 0.4 works well empirically
        best_vec = best_vec - step
        # keep on the simplex (non‑negative, sum = 1)
        best_vec = proj_simplex(best_vec)

    # --------------------------------------------------------------
    # 5. Return the best answer found (safeguard clipping & normalisation)
    # --------------------------------------------------------------
    best_solution = np.clip(best_solution, 0.0, 1000.0)
    s = best_solution.sum()
    if s == 0.0:
        best_solution[:] = 1.0 / n
    else:
        best_solution /= s

    return [float(v) for v in best_solution]
