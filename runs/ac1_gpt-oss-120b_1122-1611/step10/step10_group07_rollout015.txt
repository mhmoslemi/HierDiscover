```
def compute_score(seq):
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return np.inf
    # conv using FFT for speed
    # Use next power of two.
    fft_len = 1 << ((2*n - 2).bit_length())
    # Use rfft
    seq_fft = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(seq_fft * seq_fft, n=fft_len)[: 2*n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))
```

Note: convolution of zero-padded sequences length n yields length 2n-1. Using FFT length enough.

def score_for_x(x, n):
    # x = 1 - r
    if x <= 0:
        return np.inf
    if x >= 1:
        # r=0, shape biggest at index 0 => seq = [1,0,0,...]; conv max = 1; score = 2n
        return float(2.0 * n)
    log_r = np.log1p(-x)
    # compute r^n

    c = x / (1.0 - r_n)
    # generate seq
    i = np.arange(n)
    seq = c * np.exp(log_r * i)
    return compute_score(seq)
