def propose_candidate(seed=12345, budget_s=1000.0):
    import time, math, numpy as np, scipy.optimize
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)  # safety

    # Determine length n, try to use existing sequence if provided.
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        init_seq = np.clip(init_seq, 0.0, None)
        n = init_seq.size
        # normalise
        init_seq = init_seq / init_seq.sum()
    else:
        n = 4096  # power of two for FFT speed
        init_seq = np.full(n, 1.0/n, dtype=np.float64)

    # helper to compute score quickly
    def score_of(a):
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            return np.inf
        a = a / s
        fft_len = 1 << ((2*n-1).bit_length())
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
        max_conv = np.max(conv)
        return float(2.0 * n * max_conv)

    # record best
    best_a = init_seq.copy()
    best_score = score_of(best_a)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ---------- Stage 0: random param search ----------
    # explore simple param families
    indices = np.arange(n, dtype=np.float64)

    # power law
    for alpha in np.linspace(0.5, 3.0, 51):
        if time.monotonic() > deadline: break
        a = (indices + 1.0) ** (-alpha)
        a = a / a.sum()
        sc = score_of(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.monotonic()-start:.2f}s] new best (pow {alpha:.3f}) = {best_score:.6f}")

    # exponential decay
    for beta in np.linspace(0.001, 0.02, 40):
        if time.monotonic() > deadline: break
        a = np.exp(-beta * indices)
        a = a / a.sum()
        sc = score_of(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.monotonic()-start:.2f}s] new best (exp {beta:.5f}) = {best_score:.6f}")

    # product of two power laws (decay from both sides)
    for alpha in np.linspace(0.5, 2.0, 21):
        for delta in np.linspace(0.5, 2.0, 21):
            if time.monotonic() > deadline: break
            a = (indices + 1.0) ** (-alpha) * (n - indices) ** (-delta)
            a = a / a.sum()
            sc = score_of(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.monotonic()-start:.2f}s] new best (pow{{a={alpha:.3f},d={delta:.3f}}}) = {best_score:.6f}")

    # random mixtures with reverse
    for _ in range(200):
        if time.monotonic() > deadline: break
        # pick random shape type
        typ = rng.integers(0, 3)
        if typ == 0:
            alpha = rng.uniform(0.5, 3.0)
            a = (indices + 1) ** (-alpha)
        elif typ == 1:
            beta = rng.uniform(0.001, 0.05)
            a = np.exp(-beta * indices)
        else:
            alpha = rng.uniform(0.5, 2.0)
            delta = rng.uniform(0.5, 2.0)
            a = (indices + 1) ** (-alpha) * (n - indices) ** (-delta)
        # optionally blend with reverse

        a = (1 - lam) * a + lam * a[::-1]
        a = np.clip(a, 0.0, None)
        if a.sum() == 0:

        else:

            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

    a = best_a.copy()
    temp_start = 0.02 * best_score
    it = 0

        # pick random source and target, bias by gradient magnitude

        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
        max_idx = int(np.argmax(conv))
        grad_max = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n, max_idx + 1)
        grad_max[i_start:i_end] = a[max_idx - np.arange(i_start, i_end)]

        # source probabilities proportional to grad^2
        probs = grad_max ** 2
        if probs.sum() > 0:
            probs = probs / probs.sum()
            src = int(rng.choice(n, p=probs))
        else:
            src = rng.integers(n)

        tgt = rng.integers(n)
        if tgt == src:
            tgt = (tgt + 1) % n

        # transfer a small fraction
        delta = a[src] * rng.uniform(0.001, 0.02)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)
        cand = cand / cand.sum()

        # evaluate
        cand_fft = np.fft.rfft(cand, n=fft_len)
        conv_cand = np.fft.irfft(cand_fft * cand_fft, n=fft_len)[:2*n-1]
        sc_cand = 2.0 * n * np.max(conv_cand)
        sc_curr = 2.0 * n * np.max(conv)

        if sc_cand < best_score - 1e-12:
            best_score = sc_cand
            best_a = cand.copy()
            print(f"[{time.monotonic()-start:.2f}s] new best (SA it={it}) = {best_score:.6f}")

        # Metropolis acceptance
        if sc_cand < sc_curr or rng.random() < math.exp((sc_curr - sc_cand) / max(1e-15, temperature)):
            a = cand

        it += 1
        if it % 1000 == 0:
            print(f"[{time.monotonic()-start:.2f}s] SA iter={it}, best={best_score:.6f}")

    # final clipping and return
    final_seq = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished – final score = {final_score:.6f}")
    return final_seq.tolist()

"""
Propose a step‑function sequence with a low max‑convolution score.

Algorithm
---------
0. Determine a working length n (reuse the supplied sequence if present,
   otherwise use a power‑of‑two length convenient for FFT).
1. Explore several simple parametric families (power‑law, exponential,
   product of two power‑laws, random mixtures with the reversed vector).
   Each candidate is normalised to the probability simplex and its
   score   2·n·max(a*a)   is evaluated via an FFT‑based convolution.
   The best candidate is kept.
2. Refine the best candidate with a projected sub‑gradient descent on the
   non‑smooth objective  max(a*a).  At each iteration we locate the
   convolution entry attaining the maximum, compute its exact sub‑gradient
   (the reversed slice of `a`), step opposite to the gradient and project
   back onto the simplex by simple clipping+renormalisation.
3. Perform a short simulated‑annealing walk starting from the refined
   vector.  Moves consist of transferring a small fraction of mass from a
   source index (biased by the squared sub‑gradient) to a random target.
   Metropolis acceptance with a linearly decaying temperature helps to
   escape remaining local minima.
4. Return the best sequence found (clipped to the evaluator’s limits).

The whole search respects a 1000 s wall‑clock budget – the code stops a few
seconds early, prints progress, and finally calls ``evaluate_sequence``
once on the best vector before returning it.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT."""
    a_fft = np.fft.rfft(a, n=fft_len)
    return np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to unit sum."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 0) Choose the working length and an initial distribution
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        n = a0.size
        a0 = _simplex_normalise(a0)
    else:
        n = 4096                         # power of two – fast FFT
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(2 * n - 1)      # length for all FFTs we need

    # -----------------------------------------------------------------
    # Helper: compute the exact score (2·n·max(conv)) for a simplex vector
    # -----------------------------------------------------------------
    def _score(a: np.ndarray) -> float:
        conv = _conv_self_fft(a, fft_len)
        return float(2.0 * n * conv.max())

    # -----------------------------------------------------------------
    # Book‑keeping
    # -----------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # safety margin

    best_a = a0.copy()
    best_score = _score(best_a)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 1) Parametric search over simple families
    # -----------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # power‑law a_i ∝ (i+1)^(-α)
    for alpha in np.linspace(0.5, 3.0, 51):
        if time.monotonic() > deadline:
            break
        a = (idx + 1.0) ** (-alpha)
        a = _simplex_normalise(a)
        sc = _score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.monotonic() - start:.2f}s] new best (pow α={alpha:.3f}) = {best_score:.6f}")

    # exponential decay a_i ∝ exp(-β·i)
    for beta in np.linspace(0.001, 0.02, 40):
        if time.monotonic() > deadline:
            break
        a = np.exp(-beta * idx)
        a = _simplex_normalise(a)
        sc = _score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.monotonic() - start:.2f}s] new best (exp β={beta:.5f}) = {best_score:.6f}")

    # product of two power‑laws (decay from left and right)
    for alpha in np.linspace(0.5, 2.0, 21):
        for delta in np.linspace(0.5, 2.0, 21):
            if time.monotonic() > deadline:
                break
            a = (idx + 1.0) ** (-alpha) * (n - idx) ** (-delta)
            a = _simplex_normalise(a)
            sc = _score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[{time.monotonic() - start:.2f}s] new best (pow a={alpha:.2f},d={delta:.2f}) = {best_score:.6f}")

    # random mixtures (shape + its reversal)
    for _ in range(200):
        if time.monotonic() > deadline:
            break
        typ = rng.integers(0, 3)
        if typ == 0:
            alpha = rng.uniform(0.5, 3.0)
            a = (idx + 1.0) ** (-alpha)
        elif typ == 1:
            beta = rng.uniform(0.001, 0.05)
            a = np.exp(-beta * idx)
        else:
            alpha = rng.uniform(0.5, 2.0)
            delta = rng.uniform(0.5, 2.0)
            a = (idx + 1.0) ** (-alpha) * (n - idx) ** (-delta)

        # blend with its reverse
        lam = rng.random()
        a = (1.0 - lam) * a + lam * a[::-1]
        a = _simplex_normalise(a)

        sc = _score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.monotonic() - start:.2f}s] new best (random mix) = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2) Projected sub‑gradient descent on max‑convolution
    # -----------------------------------------------------------------
    a = best_a.copy()
    it = 0
    max_sub_iters = 5000
    while time.monotonic() < deadline and it < max_sub_iters:
        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # sub‑gradient of the active entry
        i_start = max(0, k_max - (n - 1))
        i_end = min(n, k_max + 1)
        grad = np.zeros_like(a)
        grad[i_start:i_end] = a[k_max - np.arange(i_start, i_end)]

        # step size – classic 1/√t schedule
        eta = 0.5 / math.sqrt(it + 1.0)

        a = a - eta * grad
        a = _simplex_normalise(a)

        it += 1

        # keep track of improvement every few steps
        if it % 50 == 0:
            sc = 2.0 * n * max_val
            if sc < best_score - 1e-12:
                best_score, best_a = sc, a.copy()
                print(f"[{time.monotonic() - start:.2f}s] new best (subgrad it={it}) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached in sub‑gradient phase")
                break

    # -----------------------------------------------------------------
    # 3) Simulated‑annealing refinement
    # -----------------------------------------------------------------
    a = best_a.copy()
    it_sa = 0
    temp_start = 0.02 * best_score          # modest initial temperature
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        fraction = elapsed / (deadline - start)
        temperature = temp_start * max(0.0, 1.0 - fraction)

        # recompute convolution and its most‑violated index
        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))

        # sub‑gradient for the maximum entry (used for source bias)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n, k_max + 1)
        grad_max = np.zeros_like(a)
        grad_max[i_start:i_end] = a[k_max - np.arange(i_start, i_end)]

        # source – bias towards large squared gradient entries
        prob = grad_max ** 2
        if prob.sum() > 0.0:
            prob /= prob.sum()
            src = int(rng.choice(n, p=prob))
        else:
            src = rng.integers(0, n)

        # target – uniform random (different from source)
        tgt = rng.integers(0, n)
        if tgt == src:
            tgt = (tgt + 1) % n

        # transfer a small fraction of the source mass
        delta = a[src] * rng.uniform(0.001, 0.025)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _simplex_normalise(cand)

        # evaluate candidate
        conv_cand = _conv_self_fft(cand, fft_len)
        sc_cand = float(2.0 * n * conv_cand.max())
        sc_curr = float(2.0 * n * conv.max())

        if sc_cand < best_score - 1e-12:
            best_score, best_a = sc_cand, cand.copy()
            print(f"[{time.monotonic() - start:.2f}s] new best (SA it={it_sa}) = {best_score:.6f}")

        if (sc_cand < sc_curr) or rng.random() < math.exp((sc_curr - sc_cand) / max(1e-15, temperature)):
            a = cand

        it_sa += 1
        if it_sa % 1000 == 0:
            print(f"[{time.monotonic() - start:.2f}s] SA iter={it_sa}, best={best_score:.6f}")

    # -----------------------------------------------------------------
