"""
Propose a step‐function that (hopefully) improves the current best bound.
The method is a lightweight “LP‑guided” refinement:

1.  Choose a modest length n (≤ 1024) – a smaller problem size makes the
    linear program cheap while still providing a good shape.
2.  Initialise a probability vector f (∑f = 1, f ≥ 0) with a random Dirichlet
    draw or a simple power‑law.
3.  Repeatedly
        • compute M = max_k (f * f)[k];
        • solve a linear program that maximises  Σ g_j  subject to   f * g ≤ M,
          g ≥ 0.  The LP is tiny (≤ 2 n constraints, n variables) and is
          solved with scipy.optimize.linprog (HiGHS).
        • normalise g (∑g = 1) and move a little toward it:
              f ← (1‑β)·f + β·g   (β = 0.3, decaying a bit each iteration).
   The LP step pushes mass into positions that are “inactive’’ for the
   current max‑convolution, thereby shrinking the peak.
4.  Keep the best vector seen over many random restarts and a few
   different lengths.
5.  Perform a brief stochastic local refinement (tiny mass‑transfer moves)
   on the overall best vector.
6.  Return the final vector as a plain list of floats – the evaluator will
   clip negatives / 1000‑limit automatically.

The whole search respects the time budget (≈ 995 s, leaving a 5 s safety
margin).  All heavy work (matrix build, LP solve) stays within modest
memory because n ≤ 1024; larger n would be unnecessary – the objective is
scale‑invariant, so a well‑shaped small vector gives the same bound as a
large one.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_max(a: np.ndarray) -> float:
    """Return the maximum entry of the linear convolution a*a."""
    # small vectors – direct convolution is fine
    conv = np.convolve(a, a)
    return float(conv.max())


def _build_A_ub(f: np.ndarray) -> csr_matrix:
    """
    Build the sparse matrix A_ub for the LP
        A_ub @ g <= M,
    where row k encodes (f * g)[k] = sum_i f_i * g_{k-i}.
    """
    n = f.size
    rows = []
    cols = []
    data = []

    # each column i contributes f[0..n-1] to rows i .. i+n-1
    for i in range(n):
        rows.extend(range(i, i + n))          # length n
        cols.extend([i] * n)
        data.extend(f.tolist())

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_maximize_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        maximise   Σ g_j
        subject to f * g <= M ,  g >= 0.
    Returns the optimal g (unnormalised) or None on failure.
    """
    n = f.size
    A = _build_A_ub(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises c·x, so we minimise -Σg
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # HiGHS works with sparse matrices directly
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method="highs", options={"presolve": True})
    if res.success:
        return res.x
    return None


def _local_refine(a: np.ndarray, rng: np.random.Generator,
                  deadline: float) -> np.ndarray:
    """
    Very cheap stochastic mass‑transfer refinement.
    Moves a tiny fraction of mass from the “most harmful’’ position
    (the one contributing most to the current peak) to a random target.
    """
    n = a.size
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        k_max = int(conv.argmax())

        # gradient of the active convolution index
        i_start = max(0, k_max - (n - 1))
        i_end = min(n, k_max + 1)          # exclusive
        idx = np.arange(i_start, i_end, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = a[k_max - idx]

        src = int(np.argmax(grad))
        tgt = int(rng.integers(0, n))
        if src == tgt:
            continue

        # transfer a modest fraction of the source mass
        frac = np.exp(rng.uniform(np.log(0.0005), np.log(0.02)))   # 0.05‑% … 2 %
        delta = a[src] * frac
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        # accept if we do not worsen the bound
        if _conv_max(cand) <= _conv_max(a) + 1e-12:
            a = cand
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the allocated time budget.  The routine
    combines random restarts, a small linear‑programming sub‑step and a
    final stochastic local refinement.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # 5 s safety margin as requested by the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Random restarts over a few problem sizes
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024]          # keep LP cheap; larger n not needed
    for n in lengths:
        # a couple of restarts per length
        restarts = 4 if n <= 512 else 2
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ----- initial vector -------------------------------------------------
            if rng.random() < 0.6:
                # Dirichlet draw on the full vector
                alpha = rng.uniform(0.3, 2.0)
                f = rng.dirichlet(np.full(n, alpha)).astype(np.float64)
            else:
                # simple power‑law shape
                p = rng.uniform(0.05, 2.5)
                idx = np.arange(1, n + 1, dtype=np.float64)
                f = idx ** (-p)
                f /= f.sum()

            # ensure exact simplex
            f = _project_onto_simplex(f)

            # ----- inner LP‑guided refinement ------------------------------------
            beta = 0.3                      # step size towards the LP‑solution
            max_inner = 12                 # few LP solves per restart
            for it in range(max_inner):
                if time.monotonic() >= deadline:
                    break

                M = _conv_max(f)
                score = 2.0 * n * M / (1.0 ** 2)   # sum(f) == 1

                if score < best_score - 1e-12:
                    best_score = score
                    best_seq = f.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] new best (n={n}, r={r}, it={it}) = {best_score:.6f}")

                # ----- solve LP ---------------------------------------------------
                g = _lp_maximize_sum(f, M)
                if g is None:
                    # fallback: stop the LP‑phase for this restart
                    break
                # normalise the LP‑solution to a probability vector
                if g.sum() == 0.0:
                    break
                g = _project_onto_simplex(g)

                # ----- convex update -----------------------------------------------
                f = (1.0 - beta) * f + beta * g
                f = _project_onto_simplex(f)

                # slowly decay beta (helps fine‑tuning later)
                beta = max(0.05, beta * 0.94)

    # ------------------------------------------------------------------
    # 2) If nothing was found (should not happen), fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3) Final cheap stochastic local refinement on the best candidate.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    a = _local_refine(a, rng, deadline)

    # ------------------------------------------------------------------
    # 4) Return the best sequence (the harness will clip to evaluator limits).
    # ------------------------------------------------------------------
    a = np.clip(a, 0.0, 1000.0)
    final_score = evaluate_sequence(a.tolist())
    total = time.monotonic() - start_time
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in a]
