def propose_candidate(seed=12345, budget_s=1000):
    rng = np.random.default_rng(seed)
    
    # choose size n
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        n = max(256, min(n, 8192))
    else:
        n = 2048

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    # Initialize logits
    if "height_sequence_1" in globals():
        init_a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0., None)
        total = init_a.sum()
        if total == 0.:
            init_a = np.full(n, 1.0/n)
        else:
            # If length mismatch, we need to crop or pad.
            if init_a.size != n:
                # If longer, truncate, if shorter, pad with uniform
                if init_a.size > n:
                    init_a = init_a[:n]
                else:
                    # pad with uniform small part
                    pad = np.full(n - init_a.size, 1.0/n)
                    init_a = np.concatenate([init_a, pad])
            init_a /= init_a.sum()
    else:
        # start uniform
        init_a = np.full(n, 1.0/n)
    
    # convert to logits (softmax inverse)
    epsilon = 1e-12
    logits = np.log(np.maximum(init_a, epsilon))
    
    # Adam parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5 # base learning rate for logits
    # schedule: lr = lr0 * sqrt(1 / t) maybe
    # We'll also use temperature for perturbations
    start_time = time.monotonic()
    deadline = start_time + max(0., float(budget_s) - 5.)
    
    best_a = np.copy(init_a)
    best_J = float('inf')
    
    iteration = 0
    last_improve = start_time
    # need to compute J for init quickly
    conv = _conv_full(best_a, fft_len, conv_len)
    best_J = float(2.0 * n * conv.max())
    
    # main loop
    while time.monotonic() < deadline:
        # compute a from logits (softmax)
        # stable version
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()
        # compute convolution
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        J = float(2.0 * n * max_conv)
        if J < best_J - 1e-14:
            best_J = J
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:.2f}s] New best J={best_J:.6f}")
        
        # compute active set
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        # compute grad_a
        grad_a = np.zeros_like(a)
        # For each active p, add 2 * shift of a
        for p in active:
            # contributions: for k from max(0, p-(n-1)) to min(p,n-1)
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end+1, dtype=int)
            # a_{p - idx} valid
            grad_a[idx] += 2.0 * a[p - idx]
        # compute avg = a dot grad_a
        avg = np.dot(a, grad_a)
        # gradient wrt logits
        grad_z = a * (grad_a - avg)
        # Adam update
        iteration += 1
        lr = lr0 / np.sqrt(iteration)  # decaying learning rate
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        # step

        # occasional perturbation to escape plateaus

            jitter = rng.normal(0.0, sigma, size=n)

"""
Gradient‑descent on the simplex (soft‑max parametrisation) with an
Adam optimiser and occasional stochastic jitter.

* The vector a is kept on the probability simplex via a = softmax(z).
* The objective J(a)=2·n·max(conv(a,a)) is scale‑invariant; with
  ∑a=1 we only need to minimise max(conv).
* The sub‑gradient of max(conv) is obtained from the active set
  (indices where the convolution attains its maximum) and is
  efficiently accumulated by shifting the current a.
* Adam (adaptive moments) provides an aggressive yet stable descent
  direction.  The learning‑rate decays ∝1/√t.
* If no improvement is seen for ~25 s the current logits are jittered
  (Gaussian noise) and the Adam moments are reset – a cheap restart
  that helps escape plateaux.
* The algorithm starts from the best of several heuristic initial
  distributions (uniform, geometric, ramps, Gaussians, Dirichlet draws
  and the previously discovered `height_sequence_1` if available) and
  runs until the supplied time budget (default 1000 s) expires.

The returned list is clipped to the allowed range and normalised so
that `evaluate_sequence` can be called without further processing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (same as the previous optimiser)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (output length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    evaluation function `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds and returns the best sequence discovered.

    The optimisation works on the simplex by, internally, using a
    soft‑max parametrisation `a = softmax(z)`.  Adam‑accelerated gradient
    steps are taken w.r.t. the sub‑gradient of `max(conv(a,a))`.  A simple
    jitter‑restart strategy prevents stagnation.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size (n) – keep it moderate for FFT speed.
    # --------------------------------------------------------------
    global height_sequence_1
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a portfolio of diverse initial vectors (probability vectors).
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – moderately and strongly concentrated
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            # crop or pad to desired length
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (by the raw objective J = 2·n·max(conv)).
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop – Adam on logits + jitter restarts.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # Soft‑max parametrisation
    eps = 1e-12
    logits = np.log(np.maximum(best_a, eps))
    m = np.zeros_like(logits)   # first moment
    v = np.zeros_like(logits)   # second moment

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.5            # base learning rate (will decay)

    iteration = 0
    last_improve = start_time

    # Store the globally best solution seen so far
    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # ---- 4.1 Build current a from logits (soft‑max) ----
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)   # shift for numerical stability
        a = exp_logits / exp_logits.sum()       # this is on the simplex

        # ---- 4.2 Evaluate objective ----
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ---- 4.3 Update global best ----
        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ---- 4.4 Detect stagnation and apply jitter restart ----
        if now - last_improve > 25.0:
            # jitter the logits around the current global best
            jitter_sigma = 0.1                     # fairly large to escape plateaux
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

        # ---- 4.5 Build active set (indices attaining the maximum) ----
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]

        # ---- 4.6 Compute sub‑gradient of max(conv) w.r.t. a ----
        grad_a = np.zeros_like(a)
        # The contribution of each active convolution index is a shifted copy of a
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1, dtype=int)
            grad_a[idx] += 2.0 * a[p - idx]

        # ---- 4.7 Transform gradient to logits space (soft‑max chain rule) ----
        avg = np.dot(a, grad_a)          # scalar = <a, grad_a>
        grad_z = a * (grad_a - avg)      # ∂J/∂z

        # ---- 4.8 Adam step on logits ----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)    # simple 1/√t decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # keep logits in a reasonable numeric range
        logits = np.clip(logits, -500.0, 500.0)

    # --------------------------------------------------------------
    # 5. Return the best sequence (safely clipped and re‑normalised).
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
