def build_convolution_constraints(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for i in range(2*n - 1):
        # j from max(0, i - (n-1)) to min(n-1, i)
        j_start = max(0, i - (n - 1))
        j_end = min(n - 1, i)
        # for each j, column idx = i - j
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(i - j)  # that column index (0 <= i-j < n)
            data.append(a[j])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    return A

def solve_lp_for_b(a, M):
    import scipy.sparse as sp
    import scipy.optimize as opt
    n = a.size
    # Build constraint matrix
    rows = []
    cols = []
    data = []
    for i in range(2*n - 1):
        j_start = max(0, i-(n-1))
        j_end = min(n-1, i)
        # For each j
        # column index = i - j
        for j in range(j_start, j_end+1):
            rows.append(i)
            cols.append(i - j)
            data.append(a[j])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    b_ub = np.full(2*n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    c = -np.ones(n)
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        print("LP solve failed", res.message)
        return None
    return res.x

def _simplex_projection(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Search for a step‑function (non‑negative heights) with a low
`evaluate_sequence` value. The algorithm works in three phases:

    1. Smoothed exponentiated‑gradient descent on a log‑sum‑exp
       approximation of the max‑convolution.  The smoothing parameter
       (beta) is increased gradually while a multiplicative (mirror)
       update keeps the iterate on the probability simplex
       (‖a‖₁ = √(2·n)).

    2. A deterministic linear‑programming improvement step derived from
       the classic “alternating f–g” method: for the current vector *a*
       we solve

            max   Σ b_j
            s.t.  (a * b)_k ≤ M   for all k,
                  b_j ≥ 0,

       where M = max(a*a) and * denotes linear convolution.
       The solution *b* is scaled to the same ℓ₁‑norm as *a* and mixed
       with *a* (convex combination).  A few such LP refinements can
       push the max‑convolution below the value reached by pure
       gradient descent.

    3. A short simulated‑annealing / pairwise‑mass‑transfer phase to
       escape remaining shallow local minima.

All phases respect the time budget (default 1000 s) and continuously
track the best vector found so far. The returned sequence is clipped
to the evaluator’s limits (0 … 1000) and converted to a plain Python
list of floats.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Basic linear‑algebra helpers (no closures, no lambdas)
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (full length 2·len(a)-1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_weighted(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Convolution of a (length n) with a weight vector w (length 2n‑1).
    Returns the full linear convolution (length n+len(w)-1 = 3n‑2).
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(a_fft * w_fft, n=fft_len)
    return full[: a.size + w.size - 1]


def _simplex_project(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s <= 0.0:
        return np.full_like(v, 0.0)

    # make a copy to avoid mutating the argument
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.size + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform distribution
        return np.full_like(v, s / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Linear‑programming refinement (requires SciPy)
# --------------------------------------------------------------

def _lp_improve(a: np.ndarray, M: float, target_sum: float):
    """
    Solve the LP:
        max Σ b_j  s.t.  (a * b)_k ≤ M  ∀k,  b ≥ 0
    and return the scaled vector g (sum = target_sum).  If the LP
    fails or is infeasible, ``None`` is returned.
    """
    try:
        import scipy.sparse as sp
        import scipy.optimize as opt
    except Exception as e:
        # SciPy not available – skip LP phase
        print("[LP] SciPy not available, skipping LP refinement.")
        return None

    n = a.size
    m = 2 * n - 1

    # Build the (m × n) constraint matrix A such that (a * b) = A·b
    rows = []
    cols = []
    data = []
    for i in range(m):
        j_start = max(0, i - (n - 1))
        j_end = min(n - 1, i)
        # column index for a[j] contribution is i - j
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(i - j)
            data.append(a[j])

    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))

    # Upper bounds of the constraints – a tiny epsilon helps feasibility
    b_ub = np.full(m, M + 1e-12, dtype=np.float64)

    # Objective: maximize Σ b_j  →  minimize -Σ b_j
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # Use HiGHS (fast, robust) – set a modest time limit per call
    res = opt.linprog(
        c, A_ub=A, b_ub=b_ub, bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30.0}
    )

    if not res.success:
        print("[LP] linprog failed:", res.message)
        return None

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return None

    # Scale so that Σ g = target_sum
    scale = target_sum / sum_b
    g = b * scale
    return g

            a = _simplex_project(a + jitter, target_sum)
            continue

        # Convex combination a ← (1‑t)·a + t·g
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = _simplex_project(a, target_sum)   # numerical clean‑up

        # Re‑evaluate
        conv = _conv_self(a, fft_len)
        new_M = conv.max()
        if new_M < best_M - 1e-14:
            best_M = new_M
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (LP)  = {best_M:.9f}  (it={lp_iter})")

        # Anneal the mixing factor a little (more aggressive early, gentler later)
        mix_factor = max(0.05, mix_factor * 0.997)

        lp_iter += 1
        if lp_iter % 5 == 0:
            # occasional random perturbation to avoid pathological stagnation
            jitter = rng.normal(scale=0.0015, size=n)
            a = _simplex_project(a + jitter, target_sum)

    # ==================================================================
    # Phase 3 – Simulated‑annealing‑style final polishing (remaining time)
    # ==================================================================
    a = best_a.copy()
    it3 = 0
    temperature_start = 0.025 * best_M   # relative to current max‑conv
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        fraction = elapsed / (deadline - start)
        temperature = temperature_start * (1.0 - fraction)

        conv = _conv_self(a, fft_len)
        max_idx = int(np.argmax(conv))

        # gradient of the active (maximal) convolution entry
        grad_max = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n, max_idx + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad_max[i] = a[max_idx - i]

        # pick source (largest gradient) and target (smallest gradient)
        src = int(np.argmax(grad_max))
        tgt = int(np.argmin(grad_max))
        if src == tgt:
            tgt = (tgt + 1) % n

        # draw a small mass transfer
        delta = a[src] * rng.uniform(0.001, 0.018)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _simplex_project(cand, target_sum)

        # evaluate candidate
        cand_conv = _conv_self(cand, fft_len)
        cand_M = cand_conv.max()

        # metropolis acceptance
        cur_M = conv.max()
        if cand_M < best_M - 1e-14:
            best_M = cand_M
            best_a = cand.copy()
            print(f"[{elapsed:6.2f}s] new best (SA)   = {best_M:.9f}  (it={it3})")
        if cand_M < cur_M or rng.random() < math.exp((cur_M - cand_M) / max(1e-12, temperature)):
            a = cand

        it3 += 1
        if it3 % 4000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] SA iter={it3}, best max‑conv={best_M:.9f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
