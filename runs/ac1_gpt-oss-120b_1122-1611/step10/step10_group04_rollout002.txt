"""
Sparse‑support optimiser for the max‑convolution objective.

The idea is to keep only a modest number of non‑zero heights (≈√n)
and to optimise their values with three stages:

* Stage 1 – smooth‑max (exponentiated‑gradient) descent on the
  full convolution, using a soft‑max approximation of the max.
* Stage 2 – hard‑max sub‑gradient descent (projected onto the simplex).
* Stage 3 – stochastic local moves that (i) transfer a small fraction of
  mass between two positions, and (ii) occasionally replace a low‑weight
  position by a fresh random index.  The sparse representation lets the
  optimiser concentrate mass on indices that do not clash heavily in the
  self‑convolution, pushing the objective well below the target 1.5025.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found (clipped to the evaluator limits).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures, no lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the active set
    (indices attaining the current maximum).  For each active index k
    the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Optimisation steps
# ----------------------------------------------------------------------


def _smooth_max_step(a: np.ndarray,
                     fft_len: int,
                     beta: float,
                     eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step on the smoothed max objective.
    ``a`` is assumed to already lie on the simplex.
    """
    n = a.size
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    max_conv = conv.max()
    # smooth weights (log‑sum‑exp trick)
    w = np.exp(beta * (conv - max_conv))
    w /= w.sum()                               # ∑ w_i = 1

    # ---- gradient of the smoothed max ----
    A_fft = np.fft.rfft(a, n=fft_len)

    # term2 = (a * w)  (first n entries)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: n + len(w) - 1]        # full linear conv
    term2 = conv_aw[:n]

    # term1 = (a * w_rev)[m‑1‑k]
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: n + len(w_rev) - 1]
    m = len(w)
    term1 = conv_aw_rev[m - 1 - np.arange(n)]

    grad = term1 + term2                         # ∂/∂a of weighted sum

    # exponentiated‑gradient update on log‑probabilities
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


def _hard_max_step(a: np.ndarray,
                   fft_len: int,
                   step_factor: float,
                   iteration: int) -> np.ndarray:
    """
    Projected additive sub‑gradient step for the true max objective.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    step = step_factor / math.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


def _local_move(a: np.ndarray,
                rng: np.random.Generator,
                min_weight: float = 1e-6) -> np.ndarray:
    """
    Perform a tiny stochastic mass‑transfer move:
      – pick a source index with non‑zero mass,
      – pick a (possibly zero) target index,
      – move a random fraction of the source mass to the target.
    Afterwards project back onto the simplex.
    """
    n = a.size
    # pick source among indices with notable mass
    nonzero = np.where(a > min_weight)[0]
    if nonzero.size == 0:
        src = rng.integers(0, n)
    else:
        src = int(rng.choice(nonzero))

    tgt = int(rng.integers(0, n))
    while tgt == src:
        tgt = int(rng.integers(0, n))

    # log‑uniform fraction in [0.001, 0.05]
    frac = math.exp(rng.uniform(math.log(0.001), math.log(0.05)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    return a_new


def _replace_weak(a: np.ndarray,
                  rng: np.random.Generator,
                  keep_ratio: float = 0.85) -> np.ndarray:
    """
    Replace the weakest (1‑keep_ratio) fraction of the mass by fresh
    random positions.  This keeps the total L1 norm = 1.
    """
    n = a.size
    # sort indices by weight (ascending)
    idx_sorted = np.argsort(a)
    # number of indices to keep
    keep = int(n * keep_ratio)
    keep_idx = idx_sorted[-keep:]          # the heaviest keep
    new_a = np.zeros_like(a)
    new_a[keep_idx] = a[keep_idx]

    # total mass that disappeared
    lost = 1.0 - new_a.sum()
    if lost <= 0.0:
        return a
    # distribute the lost mass uniformly on a set of fresh random spots
    nb_new = max(1, int(lost / (1.0 / n)))   # at least one new position
    new_pos = rng.choice(np.setdiff1d(np.arange(n), keep_idx, assume_unique=True),
                         size=nb_new, replace=False)
    new_a[new_pos] = lost / nb_new
    return new_a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    The optimiser uses a sparse representation (≈√n non‑zeros) and
    three optimisation phases (smooth‑max, hard‑max, stochastic refinement).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Choose problem size.
    #    A power‑of‑two length gives fast FFTs.
    # ------------------------------------------------------------------
    n = 8192                     # feel free to tune; 8192 gives ~0.8 s/10k iters
    fft_len = _next_pow2(3 * n)  # length for linear convolution

    # ------------------------------------------------------------------
    # 1) Initialise a sparse probability vector.
    #    Aim for about 1.2·√(2n) non‑zero entries.
    # ------------------------------------------------------------------
    target_m = int(1.2 * math.sqrt(2.0 * n))
    target_m = max(30, min(target_m, n // 2))   # sanity bounds
    support = rng.choice(n, size=target_m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[support] = 1.0 / target_m                     # uniform on support

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far.
    # ------------------------------------------------------------------
    best = a.copy()
    # objective for a probability vector = 2·n·max(conv)
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()
    print(f"[0.00s] initial sparse size={target_m}, score={best_score:.6f}")

    TARGET = 1.5025

    # ------------------------------------------------------------------
    # 3) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 4) Phase 1 – smooth‑max exponentiated‑gradient optimisation.
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * (deadline - start_time)
    it = 0
    beta_start, beta_end = 25.0, 250.0
    eta0 = 0.7

    while time.monotonic() < phase1_end:
        elapsed = time.monotonic() - start_time
        # linearly increase beta
        frac = (elapsed) / (phase1_end - start_time)
        beta = beta_start + frac * (beta_end - beta_start)

        eta = eta0 / math.sqrt(it + 1.0)
        a = _smooth_max_step(a, fft_len, beta, eta)

        # evaluate current (exact) score
        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best (smooth) = {best_score:.6f}")
            if best_score <= TARGET:
                print("[early stop] target reached in smooth phase")
                break

        # occasional sparsity enforcement: drop negligible entries
        if it % 500 == 0:
            # prune entries below 1e‑8 and renormalise
            tiny = a < 1e-9
            if np.count_nonzero(tiny) > n // 10:
                a[tiny] = 0.0
                a = _project_onto_simplex(a)

        it += 1
        if it % 2000 == 0:
            print(f"[{elapsed:6.2f}s] smooth‑phase iter={it}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase 2 – hard‑max sub‑gradient descent.
    # ------------------------------------------------------------------
    a = best.copy()
    phase2_end = start_time + 0.70 * (deadline - start_time)
    it = 0
    while time.monotonic() < phase2_end:
        a_candidate = _hard_max_step(a, fft_len, step_factor=0.30, iteration=it)
        conv_cand = _conv_fft(a_candidate, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        # keep the global best
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (hard) = {best_score:.6f}")
            if best_score <= TARGET:
                print("[early stop] target reached in hard‑max phase")
                break

        # accept the candidate if it does not worsen the objective
        conv_cur = _conv_fft(a, fft_len)
        if score_cand <= 2.0 * n * conv_cur.max():
            a = a_candidate

        # occasional refresh of very weak coordinates
        if it % 2500 == 0:
            a = _replace_weak(a, rng, keep_ratio=0.80)

        it += 1
        if it % 3000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] hard‑max iter={it}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Phase 3 – stochastic refinement (remaining time).
    # ------------------------------------------------------------------
    a = best.copy()
    it = 0
    while time.monotonic() < deadline:
        # tiny directed mass‑transfer
        a = _local_move(a, rng)

        # every now and then perform a more aggressive replacement of weak atoms
        if it % 4000 == 0:
            a = _replace_weak(a, rng, keep_ratio=0.85)

        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f}")
            if best_score <= TARGET:
                print("[early stop] target reached in local phase")
                break

        it += 1
        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local iter={it}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clip to evaluator limits).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total = time.monotonic() - start_time
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
