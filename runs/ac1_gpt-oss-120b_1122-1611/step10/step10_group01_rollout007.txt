"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Strategy
--------
1. **Length** – use the length of a previously found solution if it exists,
   otherwise a power‑of‑two (4096) for fast FFT.
2. **Parametric families** – enumerate a few analytically‑defined families:
   uniform, sparse Dirichlet draws, power‑law, Gaussian, *exponential* and
   *symmetric exponential*.  For the exponential families we perform a quick
   log‑space line‑search on the decay parameter (``r``) and keep the best
   candidate.
3. **Projected sub‑gradient descent** – minimise the exact objective
   ``max_k (a*a)[k]`` on the probability simplex.  The sub‑gradient for the
   active (maximum) diagonals is computed analytically; a diminishing step
   size and Euclidean projection keep the iterate feasible.
4. **Stochastic local refinement** – repeatedly move a tiny random fraction
   of mass from the index with the largest sub‑gradient (i.e. contributing
   most to the current peak) to the index with the smallest sub‑gradient,
   accepting moves that do not increase the exact objective.
5. **Book‑keeping** – keep the best vector seen so far, clip it to the
   evaluator limits and return it as a plain list.

The whole routine respects the supplied time‑budget (default 1000 s) and
prints progress along the way.

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set ``active`` of
    maximising indices.
    For each active lag k the contribution to ∂max/∂a_i is a_{k-i}
    (counted twice because the convolution is symmetric).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)  # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad

# ----------------------------------------------------------------------


def _make_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i  (sum‑normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _make_sym_exponential(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i + r**(n‑1‑i)  (sum‑normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _make_powerlaw(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-alpha}  (sum‑normalised)."""
    p = np.arange(1, n + 1, dtype=np.float64) ** (-alpha)
    s = p.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return p / s


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric discrete Gaussian (mean at centre)."""
    mu = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    s = g.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return g / s

def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact evaluation of the objective for a simplex‑normalised vector."""
    conv = _conv_fft(a, fft_len)
    return float(2.0 * a.size * conv.max())

    Fast log‑space search for the best decay parameter ``r`` for the
    supplied ``make_func`` (either exponential or symmetric exponential).
    Returns the best vector and its score.
    """
    best_vec = None
    best_score = np.inf
    # search range for q = 1 - r ; we explore 1e‑6 … 5e‑2 (log‑spaced)
    q_min, q_max = 1e-6, 5e-2
    for _ in range(3):  # three refinement stages
        q_vals = np.logspace(np.log10(q_min), np.log10(q_max), 200)
        for q in q_vals:
            if time.time() >= deadline:
                return best_vec, best_score
            r = 1.0 - q
            a = make_func(n, r)
            score = _score_from_a(a, fft_len)
            if score < best_score:
                best_score = score
                best_vec = a.copy()
        # narrow the interval around the best q for the next stage
        if best_vec is None:
            break
        q_best = 1.0 - (make_func(n, 1.0 - q_min)[0] / best_vec[0])  # not used
        # actually we stored the best q in the loop above – capture it:
        # (we simply recompute by inspecting the last used q that gave best_score)
        # To avoid extra bookkeeping we just shrink by a factor of 4 centred on the
        # previously best q.
        # Find the index of the best q in the current grid:
        idx_best = np.argmin([_score_from_a(make_func(n, 1.0 - q), fft_len) for q in q_vals])
        q_center = q_vals[idx_best]
        q_min = max(q_center / 4.0, 1e-9)
        q_max = min(q_center * 4.0, 0.5)
    if best_vec is None:
        # fallback – uniform
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score_from_a(best_vec, fft_len)
    return best_vec, best_score


def _subgradient_descent(
    a0: np.ndarray,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on ``max_k (a*a)[k]``.
    ``a0`` must already be normalised to sum = 1.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _score_from_a(a, fft_len)
    step0 = 0.4
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        # active set – all indices attaining the maximum (within tiny tolerance)
        active = np.where(np.abs(conv - max_val) <= 1e-14)[0]
        grad = _gradient_from_active(a, active)
        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break
        # diminishing step size, move opposite to the sub‑gradient
        step = step0 / np.sqrt(t) * (grad / norm)
        a_new = a - step
        a_new = _project_onto_simplex(a_new)

        score_new = _score_from_a(a_new, fft_len)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best = a_new.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] subgrad iter={t}   new best = {best_score:.6f}")

        # accept if we did not increase the true objective
        if score_new <= 2.0 * n * max_val:
            a = a_new
        else:
            # occasional jitter to get out of flat regions
            if t % 500 == 0:
                jitter = np.exp(rng.normal(0.0, 0.018))
                a = _project_onto_simplex(a * jitter)
    return best, best_score


def _stochastic_refinement(
    a0: np.ndarray,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Tiny directed mass‑transfer moves: move a random fraction of the donor
    (largest sub‑gradient) to the receiver (smallest sub‑gradient) whilst
    never worsening the exact objective.
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = _score_from_a(a, fft_len)
    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        idx_max = int(np.argmax(conv))
        grad = _gradient_from_active(a, np.array([idx_max]))
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad + (np.arange(n) == src) * np.inf))

        # ensure donor has some mass
        if a[src] <= 0.0:
            src = int(rng.integers(0, n))
            if a[src] <= 0.0:
                it += 1
                continue

        # random fraction between 0.1 % and 5 %
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_score = _score_from_a(cand, fft_len)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] refinement #{it} improved to {best_score:.6f}")

        # keep the move if it does not increase the exact objective
        if cand_score <= max_val:
            a = cand

        # occasional broader jitter
        if it % 4000 == 0 and it > 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)

        it += 1
    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the allocated time budget.
    """
    global start_time
    start_time = time.monotonic()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Determine length of the vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = prev.size
        # make sure it is simplex‑normalised for the initial pool
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0.0:
            prev /= prev.sum()
        else:
            prev = np.full_like(prev, 1.0 / prev.size)
    else:
        # power‑of‑two for cheap FFTs
        n = 4096
        prev = None

    fft_len = _next_pow2(3 * n)         # enough for all linear convolutions

    # ------------------------------------------------------------------
    # 1) Build a pool of parametric candidates
    # ------------------------------------------------------------------
    candidates = []

    # Uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(uniform)

    # If we have a previous construction, include it
    if prev is not None:
        candidates.append(prev.copy())

    # Gaussian family (several sigmas)
    sigmas = [n / 8.0, n / 6.0, n / 5.0, n / 4.0]
    for sigma in sigmas:
        candidates.append(_make_gaussian(n, sigma))

    # Power‑law family
    alphas = [0.2, 0.5, 1.0, 1.5, 2.0]
    for alpha in alphas:
        candidates.append(_make_powerlaw(n, alpha))

    # Sparse Dirichlet draws (few non‑zeros)
    for _ in range(5):
        sparse = rng.dirichlet(np.full(n, 0.3))
        candidates.append(sparse)

    # ------------------------------------------------------------------
    # 2) Exhaustive (but cheap) search over exponential families
    # ------------------------------------------------------------------
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    exp_vec, exp_score = _search_exponential_family(
        n, fft_len, _make_exponential, deadline, rng
    )
    candidates.append(exp_vec)

    symexp_vec, symexp_score = _search_exponential_family(
        n, fft_len, _make_sym_exponential, deadline, rng
    )
    candidates.append(symexp_vec)

    # ------------------------------------------------------------------
    # 3) Pick the best initial candidate (fast scoring)
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    for cand in candidates:
        cand = np.clip(cand, 0.0, None)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        sc = _score_from_a(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    if best_vec is None:
        best_vec = uniform.copy()
        best_score = _score_from_a(best_vec, fft_len)

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:.2f}s] initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected sub‑gradient descent (phase 2)
    # ------------------------------------------------------------------
    best_vec, best_score = _subgradient_descent(best_vec, fft_len, deadline, rng)

    # ------------------------------------------------------------------
    # 5) Stochastic local refinement (phase 3)
    # ------------------------------------------------------------------
    best_vec, best_score = _stochastic_refinement(best_vec, fft_len, deadline, rng)

    # ------------------------------------------------------------------
    # 6) Finalisation
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)   # respect evaluator limits
    final_score = evaluate_sequence(best_vec.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # expose the best sequence for a potential next round
    globals()["height_sequence_1"] = best_vec.tolist()
    return [float(x) for x in best_vec]
