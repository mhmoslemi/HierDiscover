"""
Hybrid soft‑max gradient descent with simulated annealing.

* Works on the probability simplex (∑a_i = 1, a_i ≥ 0) – the evaluation
  function J(a) = 2·n·max(conv(a,a)) is invariant under scaling, so we keep a
  normalised.
* The convolution is computed with a real FFT (zero‑padded to the next power‑
  of two).  This gives O(n log n) per evaluation.
* Instead of using a hard active set we smooth the “max” with a soft‑max
  (temperature β).  The gradient of the smoothed objective can be obtained
  analytically in the Fourier domain:
      ∇_a (∑_k w_k·conv_k) = 2·ifft( conj(FFT(w))·FFT(a) )
  where w_k ∝ exp(β·conv_k).  This captures the influence of *all* large
  convolution entries and gives a more informative descent direction.
* A multiplicative (exponentiated‑gradient) step keeps the iterates on the
  simplex (followed by Euclidean projection).  Step‑size decays like 1/√t.
* Metropolis acceptance with a linearly cooling temperature lets the search
  escape shallow plateaus.
* When no improvement is seen for 30 s a small random jitter is applied to
  the current best solution to restart the walk.
* Several diverse initial vectors (uniform, Dirichlet with different
  concentrations, Gaussian bells and the previous construction if present)
  are evaluated and the best is taken as the start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

# Main search routine
# ----------------------------------------------------------------------

    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
