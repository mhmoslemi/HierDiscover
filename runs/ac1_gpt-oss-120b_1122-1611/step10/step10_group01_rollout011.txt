"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Main ideas
-----------
0. Choose a single length n (use the length of a previously found
   construction if available, otherwise a power‑of‑two ≈ 4000).
1. Generate a diverse set of parametric families (uniform, Gaussian,
   exponential, beta‑shaped, power‑law, arcsine, random Dirichlet).
   Keep the best one w.r.t. the exact objective 2·n·max(a*a).
2. Continuation optimisation:
   minimise a smooth approximation of the max‑convolution
   `softmax_γ(conv) = (1/γ)·log Σ exp(γ·conv)`.
   The variables are the logits `v` of a soft‑max (a_i = exp(v_i)/Σexp(v)).
   For a given γ we minimise the smooth objective using L‑BFGS‑B,
   gradually increasing γ (4 → 16 → 64 → 256 → 512).  The analytic
   gradient uses FFT‑based correlation.
3. Stochastic local refinement:
   repeatedly transfer a tiny fraction of mass from the index with the
   largest sub‑gradient (source) to the index with the smallest
   sub‑gradient (target).  Accept moves that do not worsen the exact
   objective, keep the best ever seen.
4. All steps respect the time budget (≈ 1000 s).  The best sequence found
   is returned after clipping to the evaluator’s limits.
"""

import time
import numpy as np
from math import log, exp

# optional SciPy – if unavailable we fall back to a simple projected GD
try:

    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only works for 1‑D vectors.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a∗a)[k]`` for the set ``active`` of
    indices attaining the current maximum.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n, k + 1)
        idx = np.arange(lo, hi, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact score for a simplex‑normalised vector (sum = 1)."""
    conv = _conv_fft(a, fft_len)
    return float(2.0 * a.size * conv.max())


def _softmax_obj_and_grad(v: np.ndarray, n: int,
                          fft_len: int, gamma: float):
    """
    Smooth max objective and its gradient w.r.t. logits v.
    Returns (obj, grad_v).  The objective is
        2·n·softmax_γ(a*a) ,   a_i = exp(v_i)/Σexp(v).
    """
    # a = soft‑max of v
    w = np.exp(v)
    w_sum = w.sum()
    a = w / w_sum

    # ---------- convolution a*a ----------
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_fft = a_fft * a_fft
    conv = np.fft.irfft(conv_fft, n=fft_len)[: 2 * n - 1]

    # ---------- soft‑max over the convolution ----------
    # use a stable log‑sum‑exp
    gv = gamma * conv
    gv_max = np.max(gv)
    exp_shift = np.exp(gv - gv_max)
    sum_exp = exp_shift.sum()
    s = exp_shift / sum_exp                 # soft‑max weights over lags
    sm = (np.log(sum_exp) + gv_max) / gamma  # smooth approximation to max

    obj = 2.0 * n * sm

    # ---------- gradient w.r.t. a ----------
    # grad_a = Σ_k s_k ∂conv_k/∂a = 2·corr(s,a)
    s_fft = np.fft.rfft(s, n=fft_len)
    corr_full = np.fft.irfft(s_fft * np.conj(a_fft), n=fft_len)
    offset = len(s) - 1                     # 2·n‑2
    grad_a = 2.0 * corr_full[offset: offset + n]

    # ---------- chain rule to logits ----------
    avg = np.dot(a, grad_a)
    grad_v = a * (grad_a - avg)             # ∂obj/∂v_i  (soft‑max Jacobian)

    return obj, grad_v.astype(np.float64)


def _run_continuation_opt(a0: np.ndarray, n: int,
                         fft_len: int, time_deadline: float):
    """
    Run a continuation of smooth‑max minima using L‑BFGS‑B.
    Returns the best vector (sum = 1) found and its exact score.
    """
    best_a = a0.copy()
    best_score = _score_from_a(best_a, fft_len)

    # initialise logits
    v = np.log(best_a + 1e-12)

    # schedule of γ values (increasing sharpness)
    gammas = [4.0, 16.0, 64.0, 256.0, 512.0]

    for gamma in gammas:
        if time.monotonic() >= time_deadline:
            break

        # give each stage a modest iteration budget; adapt to remaining time
        remaining = time_deadline - time.monotonic()
        max_iter = 250
        # using SciPy if available
        if _HAS_SCIPY:
            res = minimize(fun=lambda vv: _softmax_obj_and_grad(vv, n,
                                                                fft_len, gamma),
                           x0=v,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter, 'disp': False})
            v = res.x
        else:
            # fallback: simple projected gradient descent on logits
            step = 0.5
            for _ in range(max_iter):
                obj, g = _softmax_obj_and_grad(v, n, fft_len, gamma)
                v -= step * g
                # re‑normalise via soft‑max (no explicit projection needed)
                # nothing extra required

        # evaluate exact score
        w = np.exp(v)
        a = w / w.sum()
        score = _score_from_a(a, fft_len)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            print(f"[{time.monotonic():.2f}s] γ={gamma:.0f} → improved score {best_score:.6f}")
        else:
            print(f"[{time.monotonic():.2f}s] γ={gamma:.0f} → score {score:.6f}")

        # early stop if already below target
        if best_score < 1.5025:
            break

        # keep the latest a as start for the next γ
        v = np.log(best_a + 1e-12)

    return best_a, best_score


def _local_refinement(a: np.ndarray, n: int,
                      fft_len: int, time_deadline: float):
    """
    Tiny directed mass‑transfer moves guided by the sub‑gradient of the
    current maximum lag.  Returns an improved vector (sum = 1) and its
    exact score.
    """
    best_a = a.copy()
    best_score = _score_from_a(best_a, fft_len)

    conv = _conv_fft(a, fft_len)
    max_conv = conv.max()

    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() < time_deadline:
        # recompute active lag only every few iterations
        if it % 30 == 0:
            conv = _conv_fft(a, fft_len)
            idx_max = int(np.argmax(conv))
            grad_active = _gradient_from_active(a, np.array([idx_max]))
            src = int(np.argmax(grad_active))
            # choose a low‑gradient target (could be many)
            tgt_candidates = np.where(grad_active <= np.percentile(grad_active, 15))[0]
            if tgt_candidates.size:
                tgt = int(rng.choice(tgt_candidates))
            else:
                tgt = int(rng.integers(0, n))

        # ensure source still has mass
        if a[src] <= 0.0:
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero))
        if src == tgt:
            tgt = (tgt + 1) % n

        # transfer a tiny random fraction (0.1 %–5 %)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        conv_cand = _conv_fft(cand, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = cand.copy()
            print(f"[{time.monotonic():.2f}s] refinement #{it} → {best_score:.6f}")

        # accept if we do not increase the true objective
        if score_cand <= max_conv + 1e-12:
            a = cand
            max_conv = conv_cand.max()

        it += 1

        # occasional gentle jitter to escape plateaus
        if it % 4000 == 0:
            width = max(1, n // 80)
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.006))
            a_j = a.copy()
            a_j[start:start + width] *= factor
            a = _project_onto_simplex(a_j)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a time budget (default 1000 s) and returns the
    best sequence found (list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # ------------------------------------------------------------------
    # 0) Choose length n (reuse previous construction if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = prev.size
    else:
        n = 4096                       # power of two → fast FFTs

    fft_len = _next_pow2(3 * n)        # sufficient for all linear convolutions

    # ------------------------------------------------------------------
    # 1) Build a modest pool of parametric seeds
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous solution (if any)
    if "height_sequence_1" in globals():
        prev_norm = prev.copy()
        prev_norm = np.clip(prev_norm, 0.0, None)
        s = prev_norm.sum()
        if s > 0.0:
            prev_norm /= s
            candidates.append(prev_norm)

    idx = np.arange(n, dtype=np.float64)

    # Gaussian family (symmetric)
    mu = (n - 1) / 2.0
    for sigma in [n / 8.0, n / 6.0, n / 4.0, n / 3.0]:
        g = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        g /= g.sum()
        candidates.append(g)

    # Exponential decay
    for tau in [n / 5.0, n / 4.0, n / 3.0]:
        e = np.exp(-idx / tau)
        e /= e.sum()
        candidates.append(e)

    # Beta‑shaped (i+1)^{α‑1}·(n‑i)^{β‑1}
    alphas = [0.5, 1.0, 2.0, 3.0]
    betas = [0.5, 1.0, 2.0, 3.0]
    for a in alphas:
        for b in betas:
            beta = ((idx + 1) ** (a - 1)) * ((n - idx) ** (b - 1))
            beta = np.maximum(beta, 0.0)
            beta_sum = beta.sum()
            if beta_sum > 0:
                beta /= beta_sum
                candidates.append(beta)

    # Power‑law family a_i ∝ (i+1)^{-α}
    for alpha in [0.2, 0.5, 0.8, 1.0, 1.5, 2.0]:
        p = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
        p_sum = p.sum()
        if p_sum > 0:
            p /= p_sum
            candidates.append(p)

    # Arcsine‐law shape (∝ 1/√{x·(1‑x)})
    arc = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
    arc = np.maximum(arc, 0.0)
    arc /= arc.sum()
    candidates.append(arc)

    # Random Dirichlet draws (both dense and sparse)
    for conc in [0.3, 0.7, 1.0, 2.0]:
        for _ in range(5):
            d = rng.dirichlet(np.full(n, conc))
            candidates.append(d)

    # ------------------------------------------------------------------
    # 2) Keep the best raw candidate (exact score)
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        sc = _score_from_a(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[{time.monotonic() - start_time:.2f}s] initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Continuation optimisation (smooth‑max L‑BFGS)
    # ------------------------------------------------------------------
    best_a, best_score = _run_continuation_opt(best_a, n, fft_len, deadline)

    # ------------------------------------------------------------------
    # 4) Fine‑grained stochastic refinement
    # ------------------------------------------------------------------
    best_a, best_score = _local_refinement(best_a, n, fft_len, deadline)

    # ------------------------------------------------------------------
    # 5) Final housekeeping
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)   # respect evaluator clipping limits
    # tiny renormalisation for safety (does not change the objective)
    best_a /= best_a.sum() if best_a.sum() > 0 else 1.0

    final_score = evaluate_sequence(best_a.tolist())
    elapsed = time.monotonic() - start_time
    print(f"Finished in {elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_a]
