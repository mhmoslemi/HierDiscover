"""
Propose a non‑negative step‑function that minimizes

    J(a) = 2·n·max(conv(a,a)) / (sum a)^2

The algorithm is an *alternating linear‑program / convex‑combination* method:

1. Work on the probability simplex (∑a = 1, a_i ≥ 0).  The objective reduces to
   minimising the ℓ_∞–norm of the auto‑convolution.
2. Starting from a diverse set of simple initial vectors (uniform, geometric,
   ramps, Gaussians, Dirichlet draws, and the previously discovered
   `height_sequence_1` if present) the best initial point is chosen.
3. For the current vector `f` we compute `M = max(conv(f,f))`.
4. A linear program is solved:

        maximise   1ᵀ g
        subject to  conv(f, g) ≤ M      (full linear convolution)
                    g ≥ 0

   The LP returns a vector `g₀` with the largest possible ℓ₁‑norm that still
   respects the convolution bound.
5. `g₀` is normalised to a probability vector `g` (sum = 1) and a line search
   over several convex‐combination weights `t ∈ {0.1,0.2,0.5,1.0}` is performed:

        f_new = (1‑t)·f + t·g

   The candidate with the smallest new maximum convolution is kept.
6. If no improvement is found for a while a small multiplicative random jitter
   restarts the walk.
7. The loop runs until the allotted time budget (≈ 1000 s) is exhausted.
8. The best vector found is returned as a plain Python list of floats.

The LP uses SciPy’s sparse‑matrix interface together with the HiGHS solver,
so the constraint matrix (which is just a shifted copy of `f`) can be built
efficiently for the current iterate.
"""

import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution(seq: np.ndarray, fft_len: int) -> float:
    """Return max(conv(seq,seq)) using a real FFT of length `fft_len`."""
    A = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(seq) - 1]
    return float(conv.max())


def _build_constraint_matrix(f: np.ndarray, row_idx: np.ndarray,
                             col_idx: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A where each row implements a convolution constraint:
        (A @ g)[k] = (conv(f, g))[k].

    `row_idx` and `col_idx` encode the fixed sparsity pattern (size n²).
    """
    data = np.tile(f, f.shape[0])          # repeat f for each column
    return sp.csr_matrix((data, (row_idx, col_idx)),
                         shape=(2 * f.shape[0] - 1, f.shape[0]))


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of heights) that minimises `evaluate_sequence`.
    Returns the best sequence found within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Problem size (choose a reasonable n)
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ----------------------------------------------------------------------
    # 2. Generate a portfolio of simple initial probability vectors
    # ----------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderately and strongly concentrated)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction (if any)
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s == 0.0:
            prev = np.full(n, 1.0 / n)
        else:
            prev = prev / s
        if prev.shape[0] == n:
            init_cands.append(prev)

    # ----------------------------------------------------------------------
    # 3. Pick the best initial candidate (based on max convolution)
    # ----------------------------------------------------------------------
    best_f = None
    best_M = np.inf
    for cand in init_cands:
        if cand.shape[0] != n:
            continue
        M = _max_convolution(cand, fft_len)
        if M < best_M:
            best_M = M
            best_f = cand.copy()
    if best_f is None:
        best_f = np.full(n, 1.0 / n, dtype=np.float64)
        best_M = _max_convolution(best_f, fft_len)

    best_J = 2.0 * n * best_M
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # 4. Pre‑compute the fixed sparsity pattern for the LP constraints
    # ----------------------------------------------------------------------
    # Row indices: for each column i, rows i … i+n‑1
    row_idx = np.concatenate(
        [np.arange(i, i + n, dtype=np.int32) for i in range(n)]
    )
    # Column indices: each column repeated n times
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)

    # ----------------------------------------------------------------------
    # 5. Main optimisation loop (LP + convex combination)
    # ----------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    global_best_f = best_f.copy()
    global_best_J = best_J

    last_improve = start_time
    stagnation_limit = 30.0   # seconds without improvement → jitter
    line_steps = [0.1, 0.2, 0.5, 1.0]

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---------- 5.1  Stagnation handling ----------
        if now - last_improve > stagnation_limit:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_f = global_best_f * noise
            cur_f /= cur_f.sum()
            cur_M = _max_convolution(cur_f, fft_len)
            cur_J = 2.0 * n * cur_M
            best_f = cur_f
            best_M = cur_M
            best_J = cur_J
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {global_best_J:.6f}")
            continue

        cur_f = best_f
        cur_M = best_M
        cur_J = best_J

        # ---------- 5.2  Build LP matrix for current f ----------
        A_ub = _build_constraint_matrix(cur_f, row_idx, col_idx)
        b_ub = np.full(conv_len, cur_M, dtype=np.float64)

        # ---------- 5.3  Solve the LP ----------
        lp_res = opt.linprog(
            c=-np.ones(n),                 # maximise sum(g)  ⇔ minimise -sum(g)
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "disp": False},
        )
        if not lp_res.success:
            print(f"[{elapsed:6.2f}s] LP failed ({lp_res.message}), terminating.")
            break

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0.0:
            # fallback to the current iterate
            g0 = cur_f.copy()
            sum_g0 = 1.0

        # Normalise to a probability vector
        g = g0 / sum_g0

        # ---------- 5.4  Line search over convex combinations ----------
        improved = False
        for t in line_steps:
            cand_f = (1.0 - t) * cur_f + t * g
            cand_f /= cand_f.sum()
            cand_M = _max_convolution(cand_f, fft_len)
            cand_J = 2.0 * n * cand_M
            if cand_J < best_J - 1e-12:
                best_f = cand_f
                best_M = cand_M
                best_J = cand_J
                improved = True
                last_improve = now
                print(f"[{elapsed:6.2f}s] Iter {iteration:4d} t={t:.2f}  new best J = {best_J:.6f}")
                if best_J < global_best_J - 1e-12:
                    global_best_f = best_f.copy()
                    global_best_J = best_J
                break  # accept first improvement

        # If none of the preset steps helped, try a tiny step
        if not improved:
            t = 0.01
            cand_f = (1.0 - t) * cur_f + t * g
            cand_f /= cand_f.sum()
            cand_M = _max_convolution(cand_f, fft_len)
            cand_J = 2.0 * n * cand_M
            if cand_J < best_J - 1e-12:
                best_f = cand_f
                best_M = cand_M
                best_J = cand_J
                improved = True
                last_improve = now
                print(f"[{elapsed:6.2f}s] Small‑step improvement J = {best_J:.6f}")
                if best_J < global_best_J - 1e-12:
                    global_best_f = best_f.copy()
                    global_best_J = best_J

        iteration += 1

        # tiny safety break if we are about to run out of time
        if deadline - time.monotonic() < 2.0:
            break

    # ----------------------------------------------------------------------
    # 6. Return the best sequence found (clipped and re‑normalised)
    # ----------------------------------------------------------------------
    result = np.clip(global_best_f, 0.0, 1000.0)
    total = result.sum()
    if total == 0.0:
        result[:] = 1.0 / n
    else:
        result /= total
    return [float(v) for v in result]
