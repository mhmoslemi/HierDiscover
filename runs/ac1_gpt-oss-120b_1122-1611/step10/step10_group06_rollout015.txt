def propose_candidate(seed=12345, budget_s=1000.0):
    import time, numpy as np, math
    start = time.monotonic()
    # choose length
    n = 4096
    rng = np.random.default_rng(seed)

    # try multiple attempts
    best_seq = None
    best_score = float('inf')
    attempts = 0
    while time.monotonic() - start < budget_s * 0.95:
        # random permutation of candidates
        idx_order = rng.permutation(n)
        seq = np.zeros(n, dtype=np.float64)
        sums_used = set()
        selected = []
        for x in idx_order:
            # cannot exceed limit for sums: we break if len(selected) > ??? but fine.
            conflict = False
            # check self sum
            if 2*x in sums_used:
                conflict = True
            else:
                for y in selected:
                    if (x + y) in sums_used:
                        conflict = True
                        break
            if not conflict:
                # add x
                for y in selected:
                    sums_used.add(x + y)
                sums_used.add(2*x)
                selected.append(x)
        k = len(selected)
        if k == 0:
            continue
        weight = 1.0 / k
        seq[selected] = weight
        # compute score via evaluate_sequence
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        attempts += 1
        if attempts >= 5:  # we only need few attempts
            break
    # final output
    print(f"Selected {len(selected)} positions, length {n}, best_score {best_score}")
    return best_seq.tolist()

    # maintain non-negativity and sum to 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

optimisation on a smoothed max objective (log‑sum‑exp surrogate)."""

import time
import numpy as np
import math

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_self(a, fft_len):
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv

def _conv_weighted(a, w, fft_len):
    a_fft = np.fft.rfft(a, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: a.size + w.size - 1]
    return full

def _softmax_weights(conv, beta):
    # conv: 1‑D array length 2n‑1
    max_c = conv.max()
    exp_vals = np.exp(beta * (conv - max_c))
    # prevent overflow by scaling
    w = exp_vals / exp_vals.sum()
    return w

def _project_to_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""Search for a step‑function with a low max‑convolution score.

The algorithm uses a smooth surrogate of the max (log‑sum‑exp) and
optimises it with the Adam optimiser on the probability simplex.
A temperature‑like smoothing parameter β is increased gradually so the
objective becomes increasingly sharp.  The vector is projected onto the
simplex after each update, guaranteeing non‑negativity and unit sum.
Periodic exact evaluations keep track of the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a * a (FFT implementation)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_weighted(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution of a (length n) with a weight vector w
    (length 2·n‑1).  Returns the array of length n+len(w)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: a.size + w.size - 1]
    return full


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a probability vector proportional to exp(beta·conv)."""
    # stabilise the exponentials
    max_c = conv.max()
    exp_vals = np.exp(beta * (conv - max_c))
    return exp_vals / exp_vals.sum()


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # keep a short safety margin for the final printout
    deadline = start + max(0.0, float(budget_s) - 4.0)

    # --------------------------------------------------------------
    # 0) Problem size – a power of two simplifies FFT handling
    # --------------------------------------------------------------
    n = 4096                     # length of the candidate sequence
    fft_len = _next_pow2(3 * n)  # adequate for all convolutions we need

    # --------------------------------------------------------------
    # 1) Initialise a feasible point (uniform probability simplex)
    # --------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Smoothing schedule for the soft‑max surrogate
    beta = 1.0               # start with a very soft approximation
    beta_max = 200.0
    beta_mul = 1.0015

    # bookkeeping of the best exact result seen so far
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Main optimisation loop (Adam on the smoothed max)
    # --------------------------------------------------------------
    it = 0
    steps_without_improve = 0
    while time.monotonic() < deadline:
        # ----- self‑convolution and soft‑max weights ----------------
        conv = _conv_self(a, fft_len)                 # length 2·n‑1
        w = _softmax_weights(conv, beta)              # probability vector

        # ----- gradient of the smoothed objective -----------------
        # grad_i = 2 * Σ_k w_k * a_{k‑i}   (indices out‑of‑range → 0)
        grad_full = _conv_weighted(a, w, fft_len)[: n + w.size - 1]
        grad = 2.0 * grad_full[:n]                    # length n

        # ----- Adam update (gradient descent) ---------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule – 0.5 / √t works well in practice
        lr = 0.5 / math.sqrt(it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- project back onto the simplex (feasibility) ---------
        a = _project_to_simplex(a)

        # ----- gradually sharpen the surrogate --------------------
        if beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        # ----- occasional exact evaluation -------------------------
        if it % 250 == 0 or time.monotonic() >= deadline:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f} (it={it})")
                steps_without_improve = 0
                if best_score <= 1.5025:
                    print("[early stop] target reached")
                    break
            else:
                steps_without_improve += 1

        # ----- random jitter to escape stagnation -----------------
        if steps_without_improve > 50000:
            # re‑initialise with a small random perturbation
            jitter = rng.normal(scale=0.005, size=n)
            a = _project_to_simplex(np.clip(best_a + jitter, 0.0, None))
            m.fill(0.0)
            v.fill(0.0)
            steps_without_improve = 0

        # occasional progress report
        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] iter={it}, beta={beta:.1f}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total = time.monotonic() - start
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
