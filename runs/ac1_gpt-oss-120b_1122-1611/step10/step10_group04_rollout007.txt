"""
Proposed candidate generator for the max‑convolution objective.

The algorithm is deliberately different from the reference three‑phase
optimiser.  It combines three ideas:

1. **Exponentiated‑gradient (smooth‑max) descent** – a mirror‑descent
   step on a log‑softmax surrogate of the max‑convolution.
2. **Frequency‑flattening** – periodic replacement of the current
   spectrum by one with constant magnitude (the “flat‑spectrum”
   heuristic).  After the inverse FFT the result is projected back onto
   the probability simplex.
3. **Simulated‑annealing local moves** – small directed mass‑transfer
   steps that are sometimes accepted even if they temporarily increase
   the true objective; this helps to escape shallow local minima.

All vectors are kept on the simplex  (∑a_i = 1, a_i ≥ 0)  so that the
true objective can be evaluated cheaply as

    score = 2 * n * max(conv(a, a))

instead of calling ``evaluate_sequence`` at every iteration.
The final vector is trimmed of leading / trailing zeros before it is
returned (the length factor ``n`` therefore reflects the true support).

The routine respects the 1000 s time budget, prints a few diagnostics,
and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the active set
    (indices attaining the current maximum).  For each active index k
    the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)  # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Phase‑1 – smooth‑max (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _smooth_max_step(a: np.ndarray,
                     fft_len: int,
                     beta: float,
                     eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step on the smoothed max objective.
    The result stays on the simplex.
    """
    n = a.size
    conv = _conv_fft(a, fft_len)          # length 2n‑1
    max_conv = conv.max()
    # soft‑max weights (log‑sum‑exp trick)
    w = np.exp(beta * (conv - max_conv))
    w /= w.sum()                         # ∑ w_i = 1

    # ---- gradient of the smoothed max (see reference implementation) ----
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)

    # term2 = (a * w)  restricted to first n entries
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    term2 = conv_aw[:n]

    # term1 = (a * w_rev) reflected version
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    m = len(w)
    term1 = conv_aw_rev[m - 1 - np.arange(n)]

    grad = term1 + term2                # ∂/∂a of the smoothed sum

    # exponentiated‑gradient update on log‑probabilities
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Phase‑2 – hard‑max sub‑gradient descent
# ----------------------------------------------------------------------


def _hard_max_step(a: np.ndarray,
                   fft_len: int,
                   step_factor: float,
                   iteration: int) -> np.ndarray:
    """
    Projected sub‑gradient step for the true max objective.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    step = step_factor / np.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


# ----------------------------------------------------------------------
# Phase‑3 – stochastic directed moves (simulated annealing)
# ----------------------------------------------------------------------


def _stochastic_move(a: np.ndarray,
                     rng: np.random.Generator,
                     temperature: float) -> tuple[np.ndarray, bool]:
    """
    Propose a tiny mass‑transfer move and decide acceptance using a
    simple Metropolis rule with the given temperature.
    Returns (new_vector, accepted_flag).
    """
    n = a.size
    # pick source (non‑zero) and target indices
    src = int(rng.choice(np.flatnonzero(a > 0)))  # guarantee a[src] > 0
    tgt = int(rng.integers(0, n))
    while tgt == src:
        tgt = int(rng.integers(0, n))

    # transfer a fraction drawn log‑uniformly from [0.1 %, 5 %]
    frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)

    # evaluate the true objective (no need to call evaluate_sequence)
    return a_new, (src, tgt, delta)


# ----------------------------------------------------------------------
# Additional heuristic – frequency flattening
# ----------------------------------------------------------------------


def _flatten_spectrum_step(a: np.ndarray,
                           fft_len: int,
                           mix: float = 0.2) -> np.ndarray:
    """
    Replace the magnitude of the spectrum by its mean while preserving
    phase, then mix the result with the current vector.
    ``mix`` determines how aggressively the flattening is applied
    (0 → no change, 1 → full replacement).
    """
    A = np.fft.rfft(a, n=fft_len)
    mag = np.abs(A)
    mean_mag = mag.mean()
    # avoid division by zero
    phase = np.where(mag > 0, A / mag, 1.0 + 0j)
    A_flat = mean_mag * phase
    a_flat = np.fft.irfft(A_flat, n=fft_len)[: a.size]
    # the inverse FFT may produce tiny negative values – clip
    a_flat = np.maximum(a_flat, 0.0)
    a_flat = _project_onto_simplex(a_flat)
    # convex combination with the old point
    return (1.0 - mix) * a + mix * a_flat


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    The optimisation consists of
        * smooth‑max exponentiated‑gradient descent,
        * occasional frequency‑flattening,
        * hard‑max sub‑gradient descent,
        * stochastic annealing moves.
    The best sequence found (trimmed of leading/trailing zeros) is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Choose a problem size – a compact support is beneficial.
    #    We start with a moderate length (~2000) for speed.
    # ------------------------------------------------------------------
    n = 2048
    fft_len = _next_pow2(3 * n)          # enough for exact linear convolution

    # ------------------------------------------------------------------
    # 1) Initialise a – use a Dirichlet(1) draw (exponential) for a
    #    more interesting start than flat uniform.
    # ------------------------------------------------------------------
    a = rng.exponential(scale=1.0, size=n)
    a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()    # true objective (sum=1)
    best = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 4) Phase‑1 – smooth‑max exponentiated‑gradient
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * (deadline - start_time)
    beta_start, beta_end = 30.0, 800.0
    eta0 = 0.7
    it = 0
    while time.monotonic() < phase1_end:
        elapsed = time.monotonic() - start_time
        frac = (elapsed) / (phase1_end - start_time)
        beta_cur = beta_start + frac * (beta_end - beta_start)
        eta = eta0 / np.sqrt(it + 1.0)

        a = _smooth_max_step(a, fft_len, beta_cur, eta)

        # occasional spectrum flattening (about every 2000 iters)
        if it % 2000 == 0 and it > 0:
            a = _flatten_spectrum_step(a, fft_len, mix=0.15)

        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best (smooth) = {best_score:.6f}")

        it += 1
        if it % 500 == 0:
            print(f"[{elapsed:6.2f}s] smooth‑phase iter={it}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase‑2 – hard‑max sub‑gradient descent
    # ------------------------------------------------------------------
    a = best.copy()
    phase2_end = start_time + 0.65 * (deadline - start_time)
    it = 0
    while time.monotonic() < phase2_end:
        a_candidate = _hard_max_step(a, fft_len, step_factor=0.35, iteration=it)
        conv_cand = _conv_fft(a_candidate, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (hard) = {best_score:.6f}")

        # accept the move if it does not worsen the objective
        conv_cur = _conv_fft(a, fft_len)
        if score_cand <= 2.0 * n * conv_cur.max():
            a = a_candidate

        it += 1
        if it % 300 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] hard‑max iter={it}, best={best_score:.6f}")

        # periodic extra flattening to re‑inject spectral diversity
        if it % 2500 == 0:
            a = _flatten_spectrum_step(a, fft_len, mix=0.1)

    # ------------------------------------------------------------------
    # 6) Phase‑3 – stochastic annealing refinement (remaining time)
    # ------------------------------------------------------------------
    a = best.copy()
    temperature = 0.02          # initial temperature for Metropolis rule
    temp_decay = 0.9995         # geometric decay per iteration
    it = 0
    while time.monotonic() < deadline:
        # propose a move and decide acceptance
        a_candidate, move_info = _stochastic_move(a, rng, temperature)
        src, tgt, delta = move_info
        conv_cand = _conv_fft(a_candidate, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        # Metropolis acceptance
        if score_cand < best_score:
            accept = True
        else:
            # probability to accept a worsening move
            accept = rng.random() < np.exp(-(score_cand - best_score) / temperature)

        if accept:
            a = a_candidate

        # keep track of the best ever seen
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (anneal) = {best_score:.6f}")

        # temperature schedule
        temperature *= temp_decay
        it += 1

        # occasional jitter (helps to leave tiny plateaus)
        if it % 4000 == 0:
            width = max(2, int(rng.integers(1, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.02))
            a_jitter = a.copy()
            a_jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jitter)

        if it % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] anneal iter={it}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Trim trailing/leading zeros – any zero padding only inflates ``n``.
    # ------------------------------------------------------------------
    tol = 1e-14
    nz = np.flatnonzero(best > tol)
    if nz.size == 0:
        # fallback – return a tiny uniform vector
        final = np.full(1, 1.0)
    else:
        lo, hi = nz[0], nz[-1] + 1
        final = best[lo:hi]

    # ------------------------------------------------------------------
    # 8) Final safety clip (the evaluator caps values to [0, 1000])
    # ------------------------------------------------------------------
    final = np.clip(final, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 9) Return the best sequence (as plain Python floats)
    # ------------------------------------------------------------------
    total_elapsed = time.monotonic() - start_time
    # sanity check with the real evaluator (optional, cheap)
    try:
        real_score = evaluate_sequence(final.tolist())
    except Exception:   # pragma: no cover
        real_score = float('nan')
    print(f"Finished in {total_elapsed:.2f}s – final score = {real_score:.6f}")
    return [float(x) for x in final]
