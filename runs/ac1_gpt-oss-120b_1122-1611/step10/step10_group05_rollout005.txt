def _next_pow2(x):

def _conv_fft_two(x, y, fft_len):
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X*Y, n=fft_len)

def _smooth_obj_and_grad(a, alpha, fft_len):
    # a is 1D numpy array sum to 1
    conv = _conv_fft_two(a, a, fft_len)
    # compute smoothed max via log-sum-exp
    # We will also return gradient.
    # log_w = alpha*conv
    max_log = np.max(alpha*conv)

    n = a.size
    offset = n - 1
    grad = 2.0 * grad_full[offset - np.arange(n)]
    smooth_obj = (max_log + np.log(Z)) / alpha # since log-sum-exp = max_log + log(sum exp)
    return conv, smooth_obj, grad

"""
Propose a non‑negative step‑function (probability vector) that minimizes
``evaluate_sequence``.

Algorithm (single‑threaded):
1. Initialise the vector on the probability simplex.
   * If a previous construction ``height_sequence_1`` exists, it is
     interpolated onto a power‑of‑two length (4096) and normalised.
   * Otherwise a random Dirichlet‑like start is used.

2. Optimisation proceeds with a *smooth‑maximum* surrogate of the
   objective:
      f_α(a) = (1/α)·log∑_k exp(α·(a*a)_k) .
   The gradient of f_α is
      ∇f_α = 2·corr(w,a) ,   w_k ∝ exp(α·(a*a)_k) .
   Using FFTs we obtain the convolution and the correlation efficiently.

3. A projected gradient step (with an exponentially decreasing learning
   rate) moves the point on the simplex.  A tiny simulated‑annealing
   temperature allows occasional uphill moves.

4. Every few thousand iterations a small Gaussian jitter is tried to
   escape plateaus; if it improves the best score it is kept.

5. If no improvement is seen for a long period we mix the current best
   with a fresh random vector (restart).

6. The best vector seen during the run is finally clipped to the limits
   enforced by ``evaluate_sequence`` and returned.

The schedule for the smoothing parameter α, the learning rate and the
temperature are all derived from the elapsed fraction of the supplied
time budget, giving coarse global moves at the start and fine local
refinement near the end.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft_two(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT of length ``fft_len``.
    Returns an array of length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    return conv[: x.size + y.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {u | u_i ≥ 0, Σ u_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # Duchi et al. (2008) – O(n log n) implementation
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise numerically (rarely needed)
    s = w.sum()
    if s <= 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a (probability vector) on a power‑of‑two length.
    # --------------------------------------------------------------
    target_len = 4096  # power of two – good for FFT
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        s = base.sum()
        if s == 0.0:
            base = np.full_like(base, 1.0 / base.size)
        else:
            base = base / s
        # interpolate onto the target length
        x_target = np.linspace(0, base.size - 1, target_len)
        a = np.interp(x_target, np.arange(base.size), base)
        a = np.clip(a, 0.0, None)
        a = a / a.sum()
    else:
        # pure random start (Dirichlet(1) ≈ uniform with noise)
        a = rng.random(target_len)
        a = np.clip(a, 0.0, None)
        a = a / a.sum()

    n = a.size
    fft_len = _next_pow2(3 * n)          # enough for all convolutions

    # --------------------------------------------------------------
    # 1) Compute initial score (exact objective, no smoothing)
    # --------------------------------------------------------------
    conv = _conv_fft_two(a, a, fft_len)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 3) Hyper‑parameters and schedules
    # --------------------------------------------------------------
    alpha_start, alpha_end = 5.0, 150.0              # smoothing parameter
    lr_start, lr_end = 0.5, 0.02                     # learning‑rate schedule
    temp_start = cur_score * 0.02                   # SA temperature at start
    temp_end = 1e-9                                 # near‑zero at the end

    iteration = 0
    last_improve_iter = 0

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time                # ∈[0,1]

        # ---- schedules -------------------------------------------------
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (1.0 - frac) + lr_end * frac
        temperature = temp_start * (1.0 - frac) + temp_end * frac

        # ---- compute convolution and smooth weights --------------------
        conv = _conv_fft_two(a, a, fft_len)               # a*a
        # smooth approximation of the max via log‑sum‑exp
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)                # subtract max for stability
        Z = w.sum()
        w /= Z                                          # w_k ≥ 0 , Σ w_k = 1

        # ---- gradient of the smooth surrogate -------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft_two(w, a_rev, fft_len)   # length 3n‑2
        offset = n - 1
        grad = 2.0 * grad_full[offset - np.arange(n)]   # ∇f_α

        # ---- projected gradient step ---------------------------------
        a_candidate = a - lr * grad
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = _project_onto_simplex(a_candidate)

        # ---- evaluate the candidate (exact objective) -----------------
        conv_cand = _conv_fft_two(a_candidate, a_candidate, fft_len)
        cand_score = 2.0 * n * conv_cand.max()

        # ---- acceptance (simulated annealing) ------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if temperature > 0.0:
                prob = np.exp(-(cand_score - cur_score) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score

        # ---- keep global best -----------------------------------------
        if cand_score < best_score - 1e-12:
            best = a_candidate.copy()
            best_score = cand_score
            last_improve_iter = iteration
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ---- occasional jitter to escape plateaus --------------------
        if iteration % 3000 == 0 and iteration > 0:
            sigma = 0.01 / np.sqrt(n)                 # small relative noise
            jitter = a + rng.normal(0.0, sigma, size=n)
            jitter = np.clip(jitter, 0.0, None)
            jitter = _project_onto_simplex(jitter)
            conv_jit = _conv_fft_two(jitter, jitter, fft_len)
            jitter_score = 2.0 * n * conv_jit.max()
            if jitter_score < best_score - 1e-12:
                best = jitter.copy()
                best_score = jitter_score
                last_improve_iter = iteration
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (jitter) = {best_score:.6f}")
            if jitter_score < cur_score + 1e-6:
                a = jitter
                cur_score = jitter_score

        # ---- restart if stuck for a long time ------------------------
        if iteration - last_improve_iter > 20000:
            # mix the current best with a fresh random vector
            random_vec = rng.random(n)
            random_vec = random_vec / random_vec.sum()
            a = 0.5 * best + 0.5 * random_vec
            a = _project_onto_simplex(a)
            last_improve_iter = iteration
            print(f"[{time.monotonic() - start_time:6.2f}s] restart due to stagnation")

        iteration += 1

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)        # the evaluator enforces these limits
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(v) for v in best]
