"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The search strategy consists of three stages:

1. **Parametric sweep** – generate families of simple analytic shapes
   (uniform, power‑law, exponential, Gaussian‑like).  For each shape we
   normalise the vector to sum‑to‑one, evaluate the exact objective
   (using an FFT based convolution) and keep the best candidate.

2. **Fine tuning of the power‑law exponent** – the power‑law family
   a_i ∝ (i+1)^(-α) is known analytically to give a nearly flat
   self‑convolution.  We perform a tiny line‑search around the best
   exponent found in stage 1 (±0.05, step 0.01) to improve the bound.

3. **Stochastic local refinement** – a cheap hill‑climbing loop that
   repeatedly transfers a tiny random amount of mass from a high‑gradient
   index to a low‑gradient one.  The move is kept if the true
   ``evaluate_sequence`` value improves.  The loop runs until the time
   budget expires.

All heavy work is performed with FFT‑based convolution (``np.fft``) to
keep each iteration O(n log n).  The routine respects the 1000 s budget,
printing intermediate scores and finally returning the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level only)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # all zero – return uniform vector
        w = np.full_like(v, 1.0 / len(v))
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / len(v)
        else:
            w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """
    Exact score for a simplex‑normalised vector (sum = 1).  Because the
    scaling cancels out, the score is simply ``2 * n * max(conv)``.
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a∗a)[k]`` for the given set of
    ``active`` lag indices where the maximum is attained.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time handling
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety

    # ------------------------------------------------------------------
    # 1) Choose a working length.
    #    Larger lengths have a lower theoretical bound (≈1) but
    #    increase per‑iteration cost.  4000–6000 is a good trade‑off.
    # ------------------------------------------------------------------
    n = 4800
    fft_len = _next_pow2(2 * n)      # length sufficient for linear conv

    # ------------------------------------------------------------------
    # 2) Build a collection of candidate families.
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric Gaussian (centered)
    mu = (n - 1) / 2.0
    sigma_vals = [n / 8.0, n / 6.0, n / 5.0, n / 4.0]
    for sigma in sigma_vals:
        idx = np.arange(n, dtype=np.float64)
        g = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        candidates.append(g / g.sum())

    # exponential decay
    gammas = [0.0005, 0.001, 0.002, 0.004]
    for g in gammas:
        e = np.exp(-g * np.arange(n, dtype=np.float64))
        candidates.append(e / e.sum())

    # power‑law family – sweep a coarse grid of exponents
    for alpha in np.linspace(0.3, 0.9, 13):        # 0.30,0.35,…,0.90
        p = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
        candidates.append(p / p.sum())

    # ------------------------------------------------------------------
    # 3) Evaluate all candidates with the cheap FFT‑based score.
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        # ensure non‑negativity & normalisation
        cand = np.clip(cand, 0.0, None)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        sc = _score_from_a(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
        if time.monotonic() >= deadline:
            break

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:.2f}s] best after parametric sweep = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Refine the power‑law exponent locally (tiny line‑search).
    # ------------------------------------------------------------------
    # The best candidate is usually a power‑law; we fine‑tune α around it.
    # Detect the exponent that produced best_a (if any) by a simple regression.
    # If the best vector is not recognisable as power‑law we skip this step.
    def _estimate_alpha(v):
        # fit log‑log line to the tail (ignore first few points)
        x = np.arange(1, len(v) + 1, dtype=np.float64)
        mask = v > 0
        if mask.sum() < 3:
            return None
        coeff = np.polyfit(np.log(x[mask]), np.log(v[mask]), 1)
        return -coeff[0]

    est_alpha = _estimate_alpha(best_a)
    if est_alpha is not None:
        # explore a narrow window around estimated α
        for delta in np.linspace(-0.05, 0.05, 11):
            alpha = max(0.01, est_alpha + delta)
            p = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
            cand = p / p.sum()
            sc = _score_from_a(cand, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"  refined α={alpha:.4f} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Stochastic local refinement (tiny mass transfers).
    # ------------------------------------------------------------------
    a = best_a.copy()
    best = best_a.copy()
    best_score = _score_from_a(best, fft_len)

    # pre‑compute a few useful constants
    n = a.size
    step_counter = 0

    while time.monotonic() < deadline:
        # recompute convolution and the active lag only every few iterations
        if step_counter % 30 == 0:
            conv = _conv_fft(a, fft_len)
            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-15)[0]
            grad = _gradient_from_active(a, active)

            # choose source (largest gradient) and target (smallest gradient)
            src = int(np.argmax(grad))
            tgt = int(np.argmin(grad))
            # if by chance src==tgt pick a random different index
            if src == tgt:
                tgt = int(rng.integers(0, n))

        # ensure the source still has some mass
        if a[src] <= 1e-12:
            src = int(rng.integers(0, n))

        # random fraction of the source mass to move (log‑uniform)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.02)))  # 0.1 %‑2 %
        delta = a[src] * frac

        candidate = a.copy()
        candidate[src] -= delta
        candidate[tgt] += delta

        conv_cand = _conv_fft(candidate, fft_len)
        score_cand = float(2.0 * n * conv_cand.max())

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] local move #{step_counter} improved to {best_score:.6f}")

        # Accept the move if it does not increase the exact max value
        if score_cand <= 2.0 * n * max_val:
            a = candidate

        step_counter += 1

        # occasional random jitter to escape plateaus
        if step_counter % 5000 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02))
            a = _project_onto_simplex(a * jitter)

    # ------------------------------------------------------------------
    # 6) Final evaluation and return.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)           # respect evaluator limits
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
