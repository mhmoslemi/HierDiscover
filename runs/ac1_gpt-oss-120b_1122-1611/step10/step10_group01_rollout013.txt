def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Initial pool** – a handful of parametric families (uniform,
   edge‑biased power‑law, sinusoidal and a few random Dirichlet draws) are
   generated, all normalised to sum = 1.  The best of them (lowest exact
   score) seeds the optimisation.

2. **Simulated‑annealing style stochastic search** – at each iteration a
   tiny amount of mass is transferred from a random donor to a random
   receiver (the *fraction* is drawn from a log‑uniform interval).  The
   new score is computed by an FFT‑based convolution.  Moves that lower
   the score are always accepted; moves that raise it are accepted with
   a temperature‑dependent probability that slowly cools down.

3. **Occasional smoothing** – every few thousand iterations the whole
   vector is averaged with its neighbours and projected back onto the
   probability simplex.  This helps to diffuse isolated spikes and escape
   shallow plateaus.

The routine respects the 1000 s budget (a 5 s safety margin) and returns
the best vector seen so far.  If a previous construction is available
through the global variable ``height_sequence_1`` it is used as a
starting point, otherwise a fresh uniform vector of length 2048 is used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only works for 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)
    # keep only the first 2·len(a)‑1 entries (the linear part)
    return conv_full[: 2 * a.size - 1]


def _score_from_vec(a: np.ndarray, fft_len: int) -> float:
    """Exact score = 2·n·max(conv) for a simplex‑normalised vector."""
    conv = _conv_fft(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Stochastic optimiser that searches for a non‑negative step‑function
    with a low value of the provided ``evaluate_sequence`` metric.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # stop a few seconds before the hard deadline
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Choose problem size (length of the vector)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 2048                     # a comfortable default for the time budget

    fft_len = _next_pow2(3 * n)      # long enough for every linear convolution

    # ------------------------------------------------------------------
    # 0) Assemble a small pool of initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # Uniform (always feasible)
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Edge‑biased power‑law families  a_i ∝ (i+1)^{-α} + (n-i)^{-α}
    for alpha in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=np.float64)
        edge = (i ** (-alpha) + (n - i + 1) ** (-alpha))
        edge /= edge.sum()
        init_candidates.append(edge)

    # Sinusoidal family  a_i ∝ sin(π·(i+0.5)/n)
    i = np.arange(n, dtype=np.float64)
    sin_family = np.sin(np.pi * (i + 0.5) / n)
    sin_family = np.clip(sin_family, 0.0, None)
    sin_family /= sin_family.sum()
    init_candidates.append(sin_family)

    # A few sparse Dirichlet draws (concentration < 1)
    for _ in range(5):
        sparse = rng.dirichlet(np.full(n, 0.3))
        init_candidates.append(sparse)

    # Include the previous construction if it exists
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.sum() > 0.0:
            prev /= prev.sum()
        init_candidates.append(prev)

    # ------------------------------------------------------------------
    # 1) Pick the best initial candidate (exact score)
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    for cand in init_candidates:
        cand = np.clip(cand, 0.0, None)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        sc = _score_from_vec(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    if best is None:                     # should never happen
        best = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score_from_vec(best, fft_len)

    a = best.copy()
    current_score = best_score
    print(f"[{time.monotonic() - start_time:.2f}s] initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing style stochastic search
    # ------------------------------------------------------------------
    iter_no = 0
    # temperature schedule (linear cooling)
    T0 = 0.02
    while time.monotonic() < deadline:
        iter_no += 1

        # ---- pick a random donor / receiver pair --------------------
        src = int(rng.integers(0, n))
        # avoid moving zero mass
        while a[src] <= 0.0:
            src = int(rng.integers(0, n))
        tgt = int(rng.integers(0, n))
        if src == tgt:
            continue

        # ---- fractional transfer (log‑uniform in [1e‑3,5e‑2]) ----
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))   # ~0.1 %–5 %
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # total sum unchanged, non‑negativity already guaranteed

        cand_score = _score_from_vec(cand, fft_len)

        # ---- keep the absolute best seen so far ----
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] new best = {best_score:.6f}")

        # ---- accept / reject (Metropolis) ----
        elapsed = time.monotonic() - start_time
        progress = elapsed / (budget_s - 5.0)
        T = max(T0 * (1.0 - progress), 1e-6)          # linear cooling, never zero

        if cand_score <= current_score:
            a = cand
            current_score = cand_score
        else:
            # accept with probability exp(-(Δ)/T)
            prob = np.exp(-(cand_score - current_score) / T)
            if rng.random() < prob:
                a = cand
                current_score = cand_score

        # ---- occasional smoothing (every few thousand steps) ----
        if iter_no % 5000 == 0:
            # three‑point moving average, then project back onto the simplex
            a_smooth = (a + np.roll(a, 1) + np.roll(a, -1)) / 3.0
            a = _project_onto_simplex(a_smooth)
            current_score = _score_from_vec(a, fft_len)

    # ------------------------------------------------------------------
    # 3) Final bookkeeping and return
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)            # respect evaluator caps
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start_time
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    # Convert to plain Python floats as required by the evaluation harness
    return [float(v) for v in best]
