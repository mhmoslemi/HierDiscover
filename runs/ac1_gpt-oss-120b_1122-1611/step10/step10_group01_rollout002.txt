            if sc < best_local_score:
                best_local_score = sc
                best_local = cand.copy()

        if best_local is None:
            # fallback uniform
            best_local = uniform.copy()
            best_local_score = _score_from_a(best_local, fft_len)

        # subgradient descent for this n
        a = best_local.copy()
        cur_score = best_local_score
        best_local_seq = a.copy()
        best_local_seq_score = cur_score

        iter_count = 0
        # Hyperparameters
        base_eta = 0.5  # step size scaling

        while time.monotonic() < deadline:
            iter_count += 1
            # compute conv and active set
            conv = _conv_fft(a, fft_len)
            max_val = conv.max()
            # active indices (those equal to max within tiny tolerance)
            # tolerance relative to max_val
            tol = 1e-12 * max_val
            active = np.where(conv >= max_val - tol)[0]
            # gradient
            grad = _gradient_from_active(a, active)
            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0:
                break

            # step magnitude
            step_mag = base_eta / np.sqrt(iter_count)
            # direction: negative gradient normalized
            a_new = a - (step_mag / (grad_norm + 1e-12)) * grad
            # project onto simplex
            a_new = _project_onto_simplex(a_new)

            # compute new score
            conv_new = _conv_fft(a_new, fft_len)
            new_score = 2.0 * n * conv_new.max()

            # accept if not significantly worse (allow slight increase for escape)
            if new_score <= cur_score * 1.001:
                a = a_new
                cur_score = new_score

            # update global best
            if new_score < best_local_seq_score - 1e-12:
                best_local_seq_score = new_score
                best_local_seq = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] n={n} iter={iter_count} improved local best to {best_local_seq_score:.6f}")

            # occasional jitter to avoid stalling
            if iter_count % 5000 == 0:
                jitter_factor = np.exp(rng.normal(0, 0.01))
                a = _project_onto_simplex(a * jitter_factor)

            # also exit if iter_count huge (maybe 100k) to move to next length
            if iter_count >= 200000:
                break

        # after descent, local refine moves (tiny transfers)
        a_refine = best_local_seq.copy()
        cur_score_refine = best_local_seq_score
        iter_ref = 0
        while time.monotonic() < deadline:
            iter_ref += 1
            # compute conv and active index
            conv = _conv_fft(a_refine, fft_len)
            max_idx = int(np.argmax(conv))
            # compute gradient for that active index
            grad_active = _gradient_from_active(a_refine, np.array([max_idx]))
            src = int(np.argmax(grad_active))
            # target: smallest grad value
            tgt = int(np.argmin(grad_active))
            if src == tgt:
                # random tiny move
                src = int(rng.integers(0, n))
                tgt = int(rng.integers(0, n))
            if a_refine[src] <= 0:
                src = int(rng.integers(0, n))
            # transfer fraction random log-uniform
            frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
            delta = a_refine[src] * frac
            cand = a_refine.copy()
            cand[src] -= delta
            cand[tgt] += delta
            # projection not needed (sum conserved), just ensure nonneg
            conv_cand = _conv_fft(cand, fft_len)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < best_local_seq_score - 1e-12:
                best_local_seq_score = cand_score
                best_local_seq = cand.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] n={n} local refinement #{iter_ref} improvement to {best_local_seq_score:.6f}")
            if cand_score <= cur_score_refine:
                a_refine = cand
                cur_score_refine = cand_score
            # occasional random jitter
            if iter_ref % 4000 == 0:
                width = int(rng.integers(1, max(2, n//80)))
                start_idx = int(rng.integers(0, n-width+1))
                factor = np.exp(rng.normal(0.0, 0.015))
                a_tmp = a_refine.copy()
                a_tmp[start_idx:start_idx+width] *= factor
                a_refine = _project_onto_simplex(a_tmp)

        # after finishing length n, compare to global best
        if best_local_seq_score < global_best_score:
            global_best_score = best_local_seq_score
            global_best_seq = best_local_seq.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new global best from n={n} score {global_best_score:.6f}")

    # final clamp and return
    if global_best_seq is None:
        # fallback uniform length 512
        n = 512
        global_best_seq = np.full(n, 1.0/n)
    # Clip to bounds (evaluate_sequence does it anyway)
    global_best_seq = np.clip(global_best_seq, 0., 1000.)
    final_score = evaluate_sequence(global_best_seq.tolist())
    total_elapsed = time.monotonic() - start
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in global_best_seq]

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Parametric warm‑starts** – generate a few families (uniform,
   Gaussian, power‑law, Beta, sine) and keep the best one (lowest
   ``2·n·max(conv)``).

2. **Projected sub‑gradient descent on the L∞‑objective** – at each
   iteration compute the exact active set of lags attaining the maximum
   of the auto‑convolution, form the sub‑gradient (sum of the shifted
   copies of the current vector), take a diminishing step in the
   negative gradient direction and project back onto the probability
   simplex.

3. **Stochastic local refinement** – tiny directed mass‑transfer moves
   (source = index with largest gradient entry, target = index with
   smallest) are accepted if they do not increase the exact objective.
   Occasional jitter prevents stagnation.

The routine respects the 1000 s budget, reports intermediate scores and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level only)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact score for a simplex‑normalised vector (sum = 1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a∗a)[k]`` for a set ``active`` of
    indices attaining the current maximum.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0                         # each pair appears twice
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Timing
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # leave a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    global_best_seq = None
    global_best_score = np.inf

    # ------------------------------------------------------------------
    # Length schedule (larger lengths give a lower theoretical bound)
    # ------------------------------------------------------------------
    length_schedule = [512, 1024, 2048, 4096, 8000]

    for n in length_schedule:
        if time.monotonic() >= deadline:
            break

        fft_len = _next_pow2(3 * n)   # enough for all linear convolutions

        # ------------------------------------------------------------------
        # 0) Assemble a collection of warm‑start candidates
        # ------------------------------------------------------------------
        init_candidates = []

        # uniform (fallback)
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        init_candidates.append(uniform)

        # reuse a previously found construction if it exists
        if "height_sequence_1" in globals():
            existing = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if existing.size != n:
                # linear interpolation to the new length
                x_old = np.linspace(0.0, 1.0, existing.size)
                x_new = np.linspace(0.0, 1.0, n)
                interp = np.interp(x_new, x_old, existing)
                interp = np.clip(interp, 0.0, None)
                if interp.sum() > 0.0:
                    interp /= interp.sum()
                init_candidates.append(interp)
            else:
                tmp = np.clip(existing, 0.0, None)
                if tmp.sum() > 0.0:
                    tmp /= tmp.sum()
                init_candidates.append(tmp)

        # symmetric Gaussian family
        mu = (n - 1) / 2.0
        sigma_vals = [n / 8.0, n / 6.0, n / 5.0, n / 4.0]
        idx = np.arange(n, dtype=np.float64)
        for sigma in sigma_vals:
            g = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
            g /= g.sum()
            init_candidates.append(g)

        # power‑law family a_i ∝ (i+1)^{-α}
        alphas = [0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
        for alpha in alphas:
            pvec = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
            pvec /= pvec.sum()
            init_candidates.append(pvec)

        # symmetric Beta family (a = b)
        beta_params = [0.2, 0.5, 1.0, 2.0, 3.0]
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        for a_param in beta_params:
            beta = (x ** (a_param - 1)) * ((1.0 - x) ** (a_param - 1))
            beta /= beta.sum()
            init_candidates.append(beta)

        # half‑sine shape
        sine = np.sin(np.pi * (idx + 0.5) / n)
        sine = np.clip(sine, 0.0, None)
        sine /= sine.sum()
        init_candidates.append(sine)

        # sparse Dirichlet draws (few non‑zeros)
        for _ in range(5):
            sparse = rng.dirichlet(np.full(n, 0.3))
            init_candidates.append(sparse)

        # ------------------------------------------------------------------
        # 1) Pick the best warm‑start according to the exact objective
        # ------------------------------------------------------------------
        best_local = None
        best_local_score = np.inf
        for cand in init_candidates:
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0.0:
                continue
            cand = cand / cand.sum()
            sc = _score_from_a(cand, fft_len)
            if sc < best_local_score:
                best_local_score = sc
                best_local = cand.copy()

        if best_local is None:               # extremely unlikely
            best_local = uniform.copy()
            best_local_score = _score_from_a(best_local, fft_len)

        # ------------------------------------------------------------------
        # 2) Projected sub‑gradient descent on the L∞‑objective
        # ------------------------------------------------------------------
        a = best_local.copy()
        cur_score = best_local_score
        best_seq_n = a.copy()
        best_score_n = cur_score

        max_iters = 5000                # safety cap per length
        base_eta = 0.5                  # step‑size scaling factor
        iter_counter = 0

        while iter_counter < max_iters and time.monotonic() < deadline:
            iter_counter += 1

            conv = _conv_fft(a, fft_len)
            max_val = conv.max()
            # numeric tolerance for the active set
            tol = 1e-12 * max_val
            active = np.where(conv >= max_val - tol)[0]

            grad = _gradient_from_active(a, active)
            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0.0:
                break

            # diminishing step size (scaled to unit direction)
            step = base_eta / np.sqrt(iter_counter)
            a_new = a - (step / (grad_norm + 1e-12)) * grad
            a_new = _project_onto_simplex(a_new)

            # evaluate the new point
            conv_new = _conv_fft(a_new, fft_len)
            new_score = 2.0 * n * conv_new.max()

            # accept if the true objective does not increase appreciably
            if new_score <= cur_score * 1.001:
                a = a_new
                cur_score = new_score

            # record the best ever seen for this length
            if new_score < best_score_n - 1e-12:
                best_score_n = new_score
                best_seq_n = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] n={n} iter={iter_counter} "
                      f"new local best = {best_score_n:.6f}")

            # occasional jitter to shake out of shallow plateaus
            if iter_counter % 1000 == 0:
                jitter = np.exp(rng.normal(0.0, 0.015))
                a = _project_onto_simplex(a * jitter)

        # ------------------------------------------------------------------
        # 3) Fine‑grained stochastic local refinement (tiny mass moves)
        # ------------------------------------------------------------------
        a_ref = best_seq_n.copy()
        cur_score_ref = best_score_n
        iter_ref = 0
        # a modest limit to keep the overall run time bounded
        max_ref_iters = 4000

        while iter_ref < max_ref_iters and time.monotonic() < deadline:
            iter_ref += 1
            conv = _conv_fft(a_ref, fft_len)
            idx_max = int(np.argmax(conv))

            # gradient of the single active lag
            grad_active = _gradient_from_active(a_ref, np.array([idx_max]))
            src = int(np.argmax(grad_active))
            tgt = int(np.argmin(grad_active))

            if src == tgt or a_ref[src] <= 0.0:
                src = int(rng.integers(0, n))
                tgt = int(rng.integers(0, n))
                if src == tgt:
                    tgt = (tgt + 1) % n

            # transfer a tiny random fraction of the source mass
            frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))   # 0.1 %–5 %
            delta = a_ref[src] * frac
            cand = a_ref.copy()
            cand[src] -= delta
            cand[tgt] += delta

            conv_cand = _conv_fft(cand, fft_len)
            cand_score = 2.0 * n * conv_cand.max()

            if cand_score < best_score_n - 1e-12:
                best_score_n = cand_score
                best_seq_n = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] n={n} local refine #{iter_ref} "
                      f"improved to {best_score_n:.6f}")

            if cand_score <= cur_score_ref:
                a_ref = cand
                cur_score_ref = cand_score

            # broader jitter every few thousand iterations
            if iter_ref % 1500 == 0:
                width = int(rng.integers(1, max(2, n // 80)))
                start_idx = int(rng.integers(0, n - width + 1))
                factor = np.exp(rng.normal(0.0, 0.018))
                tmp = a_ref.copy()
                tmp[start_idx:start_idx + width] *= factor
                a_ref = _project_onto_simplex(tmp)

        # ------------------------------------------------------------------
        # 4) Update the global best if this length produced a better result
        # ------------------------------------------------------------------
        if best_score_n < global_best_score - 1e-12:
            global_best_score = best_score_n
            global_best_seq = best_seq_n.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] *** new GLOBAL best (n={n}) "
                  f"score = {global_best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) Final housekeeping and return
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        # ultimate fallback (should never happen)
        n = 512
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence (just for safety)
    global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)

    # Compute the official score once more (this also applies the evaluator's
    # internal clipping and the sum‑check)
    final_score = evaluate_sequence(global_best_seq.tolist())

    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final reported score = {final_score:.6f}")

    return [float(x) for x in global_best_seq]
