def solve_lp_for_b(a, M):
    n = a.size
    rows = 2*n - 1
    # construct sparse matrix
    data = []

    for i in range(n):
        # for offset -i

        rows_i = i + np.arange(n)

        # fallback: return None or zeros

Propose a candidate step‑function (non‑negative sequence) that minimizes
evaluate_sequence.  The algorithm consists of four phases:

0. Initialise – use a supplied best sequence (height_sequence_1) or a uniform
   vector of length 4096, normalised to sum = 1.

1. Random exploration – cheap Dirichlet and power‑law draws provide diverse
   starting points.

2. Linear‑programming refinement – given the current vector a (∑a=1) and its
   maximal self‑convolution M, we solve the LP

        maximise   Σ b_j
        subject to   (a * b)[k] ≤ M   for all k,
                     b_j ≥ 0.

   The LP finds a distribution b that fits under the same bound M but has a
   larger total mass.  After normalising b to sum = 1 we move a a small step
   towards b.  This step never raises the maximal convolution and typically
   lowers it.

3. Stochastic local refinement – directed mass‑transfer moves guided by the
   sub‑gradient of the active (maximal) convolution indices, together with
   occasional jitter.

All phases respect the probability simplex (non‑negative, sum = 1) and the
whole routine stops a few seconds before the 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid tiny rounding errors
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (real‑valued, length 2*len(a)-1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set of active indices.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)  # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# cache for the sparse convolution‑constraint pattern (row/col indices)
_conv_constraint_cache: dict[int, tuple[np.ndarray, np.ndarray]] = {}


def _get_conv_constraint_pattern(n: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (row_idx, col_idx) for the matrix A where (A @ b)[k] = (a*b)[k].
    The pattern does not depend on a, only on n.
    """
    if n in _conv_constraint_cache:
        return _conv_constraint_cache[n]
    # rows = i + j, columns = j, for i,j = 0..n-1
    rows = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(
        np.arange(n, dtype=np.int32), n
    )
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    _conv_constraint_cache[n] = (rows, cols)
    return rows, cols


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   subject to   (a*b)[k] ≤ M  (∀k) , b ≥ 0.
    Returns the optimal b (may be dense) or None on failure.
    """
    n = a.size
    rows, cols = _get_conv_constraint_pattern(n)
    data = np.repeat(a, n)  # a_i repeated n times, aligns with (rows,cols)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    # a tiny tolerance to avoid infeasibility due to rounding
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximise Σb  → minimise -Σb
    # Using HiGHS (default) – robust for medium‑size LPs
    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=(0, None),
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x

            seq = np.repeat(bins, block)
            if seq.size < n:
                seq = np.concatenate([seq, bins[: n - seq.size]])
        else:
            # power‑law shape
            p = rng.uniform(0.05, 2.5)
            idx = np.arange(1, n + 1, dtype=np.float64)
            seq = idx ** (-p)
            seq = seq / seq.sum()

        # Normalise to the simplex (protect against rounding drifts)
        seq = _project_onto_simplex(seq)

        conv = _conv_fft(seq, fft_len)
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best = seq.copy()
            best_M = M
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (rand) = {best_score:.6f}")
        it_rand += 1
        if it_rand % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] random search iter={it_rand}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 1 – linear‑programming refinement (≈45 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    M = best_M
    best_improved = True
    lp_iter = 0
    phase1_end = start_time + 0.60 * (deadline - start_time)

    while time.monotonic() < phase1_end and best_improved:
        # Solve the LP
        b = _solve_lp_max_sum(a, M)
        if b is None:
            print("[LP] solve failed – exiting LP phase")
            break
        sum_b = b.sum()
        # If we cannot increase total mass, stop LP iterations
        if sum_b <= 1.0 + 1e-9:
            print("[LP] no mass increase possible – stopping")
            break
        # Normalise b to sum = 1
        b_norm = b / sum_b

        # Small convex combination – t chosen heuristically
        t = 0.15  # modest step to stay safely within the bound
        a_candidate = (1.0 - t) * a + t * b_norm
        a_candidate = _project_onto_simplex(a_candidate)

        # Evaluate
        conv_c = _conv_fft(a_candidate, fft_len)
        M_c = conv_c.max()
        score_c = 2.0 * n * M_c

        if score_c < best_score - 1e-12:
            best_score = score_c
            best = a_candidate.copy()
            best_M = M_c
            a = a_candidate.copy()
            M = M_c
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] LP improvement (iter={lp_iter}) → {best_score:.6f}")
            # continue – a has been improved
            best_improved = True
        else:
            # No improvement; shrink the step size and try once more
            t = t * 0.5
            if t < 1e-4:
                best_improved = False
                print("[LP] step too small – exiting LP phase")
                break
            a_candidate = (1.0 - t) * a + t * b_norm
            a_candidate = _project_onto_simplex(a_candidate)
            conv_c = _conv_fft(a_candidate, fft_len)
            M_c = conv_c.max()
            score_c = 2.0 * n * M_c
            if score_c < best_score - 1e-12:
                best_score = score_c
                best = a_candidate.copy()
                best_M = M_c
                a = a_candidate.copy()
                M = M_c
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] LP improvement after step‑shrink (iter={lp_iter}) → {best_score:.6f}")
                best_improved = True
            else:
                best_improved = False

        lp_iter += 1

    # --------------------------------------------------------------
    # 5) Phase 2 – stochastic local refinement (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    it_loc = 0
    while time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len)
        M = conv.max()
        idx_max = int(np.argmax(conv))

        # Sub‑gradient at the most‑violated index
        grad = _gradient_from_active(a, np.array([idx_max]))
        src = int(np.argmax(grad))

        # Choose a target with low gradient (preferably a valley)
        low = np.where(grad <= grad.min() + 1e-12)[0]
        if low.size > 1 and rng.random() < 0.8:
            tgt = int(rng.choice(low))
        else:
            tgt = int(rng.integers(0, n))

        if src == tgt:
            tgt = int(rng.integers(0, n))

        # Directed mass transfer – fraction grows slowly with iteration count
        # (start tiny, allow occasional larger moves)
        frac = np.exp(rng.uniform(np.log(5e-4), np.log(2e-2)))  # ~0.05–2 %
        delta = a[src] * frac
        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta
        a_cand = _project_onto_simplex(a_cand)

        conv_c = _conv_fft(a_cand, fft_len)
        M_c = conv_c.max()
        score_c = 2.0 * n * M_c

        if score_c < best_score - 1e-12:
            best_score = score_c
            best = a_cand.copy()
            best_M = M_c
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (move={it_loc})")
        # Accept any non‑worsening move to keep the walk stable
        if score_c <= 2.0 * n * M:
            a = a_cand

        # Occasional jitter to escape plateaus
        if it_loc % 4000 == 0 and it_loc > 0:
            width = max(1, n // 80)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx : start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)

        it_loc += 1
        if it_loc % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local iter={it_loc}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
