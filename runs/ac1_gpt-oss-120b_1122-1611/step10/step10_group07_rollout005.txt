def generate_geometric_seq(n, c):
    r = 1.0 - c / n
    if r <= 0:
        r = 0.0
        raw = np.zeros(n)
        raw[0] = 1.0
    else:
        # compute r**i using exp(i*log(r))
        log_r = np.log(r)
        i = np.arange(n, dtype=np.float64)
        raw = (1 - r) * np.exp(log_r * i)
    seq = raw / raw.sum()
    return seq
```

"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

within a fixed time budget (≈1000 s).  

The search proceeds in four stages:

0.  Initialise from a previously found best sequence (if any) otherwise from a
    uniform distribution.

1.  Global parametric sweep – cheap families that are known to give low
    convolution peaks:
        * geometric (a_i ∝ (1‑c/n)ⁱ, tuned by a single scalar c)
        * power‑law (a_i ∝ (i+1)⁻ᵖ, tuned by p)
        * random Dirichlet draws on a coarse binning.

    The best candidate from this stage is kept.

2.  Exponentiated‑gradient (EG) descent on the simplex using the exact
    sub‑gradient of the current maximal convolution entry.

3.  Additive projected sub‑gradient descent (diminishing step size).

4.  Stochastic local refinement – tiny directed mass‑transfer moves guided
    by the gradient of the most‑violated convolution index together with
    occasional jitter.

All phases respect the simplex constraint (∑a_i = 1) and the non‑negativity
requirement.  The routine stops a few seconds before the time limit so the
harness does not terminate it prematurely.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a 1‑D vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the linear convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _geom_sequence(n: int, c: float) -> np.ndarray:
    """
    Geometric shape a_i ∝ (1‑c/n)ⁱ, normalized to sum = 1.
    ``c`` is a positive scalar; the effective ratio is r = 1‑c/n.
    """
    if c <= 0.0:
        # fall back to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)

    r = 1.0 - c / n
    if r <= 0.0:
        # degenerate case – all mass on the first element
        seq = np.zeros(n, dtype=np.float64)
        seq[0] = 1.0
        return seq

    # use log to avoid loss of precision when r ≈ 1
    log_r = np.log(r)
    i = np.arange(n, dtype=np.float64)
    raw = (1.0 - r) * np.exp(log_r * i)
    # renormalise (the sum of the truncated geometric series is 1‑rⁿ)
    raw_sum = raw.sum()
    if raw_sum == 0.0:
        raw = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        raw /= raw_sum
    return raw


def _powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """
    Power‑law shape a_i ∝ (i+1)⁻ᵖ, normalized to sum = 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= seq_sum
    return seq


def _score_from_seq(seq: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute the objective (2 * n * max(conv)) for a sequence that already
    sums to 1.  The value equals the quantity returned by ``evaluate_sequence``
    up to the trivial scaling (the denominator is 1).
    """
    conv = _conv_fft(seq, fft_len)
    max_b = conv.max()
    return 2.0 * n * max_b

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:

    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    The algorithm mixes cheap parametric sweeps with exponentiated‑gradient,
    additive sub‑gradient and stochastic local refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` – use an existing construction if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        # a reasonably large uniform start (power‑law and geometric work best
        # with a few thousand points)
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------

    for c in np.linspace(0.2, 6.0, 240):          # 240 points gives a fine grid
        if time.monotonic() > phase0_end:
            break
        seq = _geom_sequence(n, c)
        s = _score_from_seq(seq, n, fft_len)
        if s < best_score:
            best_score = s
            best = seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (geom  c={c:.3f}) = {best_score:.6f}")

    # ---- power‑law sweep -------------------------------------------------
    print("[phase 0] power‑law sweep")
    for p in np.linspace(0.05, 2.8, 220):
        if time.monotonic() > phase0_end:
            break
        seq = _powerlaw_sequence(n, p)
        s = _score_from_seq(seq, n, fft_len)
        if s < best_score:
            best_score = s
            best = seq.copy()
            elapsed = time.monotonic() - start_time

    # ---- random Dirichlet / power‑law draws -------------------------------
    print("[phase 0] random exploration")
    K = 128 if n >= 4096 else max(8, n // 32)
    block = n // K
    it_rand = 0
    while time.monotonic() < phase0_end:
        if rng.random() < 0.55:          # Dirichlet on coarse bins
            alpha = rng.uniform(0.3, 2.0)
            bins = rng.dirichlet(np.full(K, alpha))

        else:                            # pure power‑law draw (different exponent)
            p = rng.uniform(0.02, 3.0)
            idx = np.arange(1, n + 1, dtype=np.float64)
            seq = idx ** (-p)
            seq /= seq.sum()

        s = _score_from_seq(seq, n, fft_len)
        if s < best_score:
            best_score = s
            best = seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (rand iter={it_rand}) = {best_score:.6f}")
        it_rand += 1
        if it_rand % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] random search iter={it_rand}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 1 – exponentiated‑gradient descent (≈35 % of time)

    phase1_end = start_time + 0.50 * (deadline - start_time)

    while time.monotonic() < phase1_end:

        max_b = conv.max()
        eps = max_b * 1e-8
        active = np.where(conv >= max_b - eps)[0]
        # limit size of active set to keep work bounded

        eta = 0.9 / np.sqrt(it_exp + 1.0)
        log_a = np.log(a + 1e-15)
        log_a -= eta * grad
        a = np.exp(log_a)
        a /= a.sum()

        # evaluate new objective
        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (exp‑grad iter={it_exp}) = {best_score:.6f}")
            if best_score <= 1.5025:

                break

        # accept non‑worsening step (helps keep progress monotone)
        if score <= 2.0 * n * max_b + 1e-12:
            a = a
        it_exp += 1
        if it_exp % 300 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] EG iter={it_exp}, best={best_score:.6f}")

    # 5) Phase 2 – additive projected sub‑gradient descent (≈25 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    phase2_end = start_time + 0.80 * (deadline - start_time)
    it_sub = 0
    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-6

        a_candidate = _project_onto_simplex(a_candidate)

        score_c = 2.0 * n * conv_c.max()
        if score_c < best_score - 1e-12:
            best_score = score_c
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (sub‑grad iter={it_sub}) = {best_score:.6f}")

        it_sub += 1

        # pick a low‑gradient destination (avoids increasing the peak)

        if score_c < best_score - 1e-12:
            best_score = score_c
            best = a_cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local move={it_loc}) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached in local refinement")
                break

        # keep move if it does not worsen the objective
        if score_c <= 2.0 * n * conv.max() + 1e-12:

    best = np.clip(best, 0.0, 1000.0)

    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
