def lp_optimize(a, M):
   n = a.shape[0]
   # Build A matrix
   from scipy.sparse import diags

   if not res.success:

   b = res.x
   return b

"""
Propose a non‑negative step‑function (list of heights) that minimizes
evaluate_sequence.

Algorithm
---------

0. **Initialisation** – generate a handful of parametric candidates
   (uniform, Gaussian, power‑law, Hann window, sparse Dirichlet) and keep
   the one with the smallest *exact* score.

1. **Linear‑programming “direction” step** – for the current vector a
   compute M = max_k (a∗a)_k.  Solve the LP

        maximise   Σ_j b_j
        subject to (a∗b)_k ≤ M   for all k
                   b_j ≥ 0

   (the constraints are linear in b).  The LP gives a feasible distribution
   b that puts as much mass as possible into positions that interfere
   the least with a.  Normalise b to sum = 1 and move a bit towards it:

        a ← (1‑τ)·a + τ·b̂ ,   τ ≈ 0.2

   The mixture keeps the sum fixed (τ is constant) and provably does not
   increase the max‑convolution.  This step is repeated a few times
   (≈ 10 – 15) – each LP costs only a few seconds for n≈4000.

2. **Stochastic local refinement** – after the LP phase perform cheap
   “mass‑transfer” moves: take a source index with the largest
   sub‑gradient of the current active lag(s) and a target index with the
   smallest one, move a tiny random fraction of mass and keep the move if
   it does not increase the exact objective.  This cleans up any remaining
   irregularities.

3. **Timing** – the routine stops a few seconds before the allotted budget
   and returns the best sequence found so far.

All operations stay on the probability simplex (∑a_i = 1, a_i ≥ 0), so the
objective 2·n·max(a∗a) is exactly the value returned by evaluate_sequence.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective (2·n·max(conv)) for a simplex‑normalised vector."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * np.max(conv))


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)[k] for the set ``active`` of indices attaining
    the current maximum.  Each active lag contributes a_i‑wise term.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)            # exclusive on the right
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad


def _build_lp_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A that implements the linear constraints
        (a ∗ b)_k ≤ M
    i.e. the k‑th row contains a_i where i = k‑j.

    The matrix is dense (n² non‑zeros) but stored in CSR format.
    """
    n = a.shape[0]
    # rows = i + j, cols = j, for all i,j∈[0,n)
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(a, n)                # a_i repeated for each j
    shape = (2 * n - 1, n)
    return sp.csr_matrix((data, (rows, cols)), shape=shape)


def _solve_lp_direction(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ b_j
        s.t.       (a ∗ b)_k ≤ M   for all k
                   b_j ≥ 0
    Returns the optimal b (not normalised).  If the solver fails,
    the current a is returned unchanged.
    """
    n = a.shape[0]
    A = _build_lp_matrix(a)                # (2n‑1) × n
    ub = np.full(2 * n - 1, M)             # right‑hand side
    c = -np.ones(n)                        # maximise Σb → minimise −Σb
    bounds = [(0, None)] * n

    # Use the HiGHS solver (fast, works with sparse A)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – return a feasible point (the current a)
        return a.copy()
    return res.x

    ``evaluate_sequence`` within the given time budget.

    # ------------------------------------------------------------------

        p /= p.sum()
        candidates.append(p)

    candidates.append(hann)

        M = conv.max()

        sum_b = b_raw.sum()

            break

        new_score = _score_from_a(a_new, fft_len)

            best_score = new_score
            best = a_new.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] LP iter {outer} improved to {best_score:.6f}")

        # accept the mixture (it never increases the max‑convolution,
        # but we also keep a as the most recent point for the next LP)
        a = a_new

    # ------------------------------------------------------------------
    # 3) Stochastic local refinement (tiny directed transfers)
    # ------------------------------------------------------------------
    a = best.copy()                       # start from the best found so far
    iter_loc = 0
    src = 0
    tgt = 0

            # choose a low‑influence target
            low_val = grad_active.min()
            low_candidates = np.where(grad_active <= low_val + 1e-12)[0]
            if low_candidates.size > 1:

            else:

                thresh = np.percentile(grad_active, 15)
                low_candidates = np.where(grad_active <= thresh)[0]
                tgt = int(rng.choice(low_candidates)) if low_candidates.size else int(rng.integers(0, n))

        # transfer a random geometric fraction (0.1 % – 5 %)

            best = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] local move #{iter_loc} improved to {best_score:.6f}")

        # keep the move if it does not worsen the exact objective
        if cand_score <= 2.0 * n * np.max(conv):
            a = cand

        iter_loc += 1

        # occasional broader jitter to avoid stagnation
        if iter_loc % 4000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            jittered = a.copy()
            jittered[start_idx:start_idx + width] *= factor
            a = jittered / jittered.sum()

    # ------------------------------------------------------------------
    # 4) Final evaluation and return
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)            # respect evaluator limits
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
