"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm uses a smoothed‐max objective
(soft‑max via a temperature β) and exponentiated‑gradient updates.
A simple simulated‑annealing acceptance rule and periodic jitter
help escape plateaus.  Multiple diversified initialisations (uniform,
geometric, ramps, Gaussians, Dirichlet draws and sparse masks) are
tested and the best is refined.  The whole search runs for the
supplied time budget (default 1000 s) and returns the best vector
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to one).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a real FFT.
    Returns the first ``conv_len`` entries (the full linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max weights for a vector of convolutions."""
    # shift by max to avoid overflow
    shift = conv.max()
    exp_vals = np.exp(beta * (conv - shift))
    return exp_vals / exp_vals.sum()


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float,
                     fft_len_corr: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed max objective
    f(a) = (1/β) log Σ exp(β·conv) with respect to ``a``.
    The gradient is 2 * (w ⋆ a) where w are the soft‑max weights
    and ⋆ denotes the (linear) cross‑correlation.
    """
    w = _softmax_weights(conv, beta)                 # shape (2n‑1,)
    # cross‑correlation w ⋆ a  =  ifft( conj(FFT(a)) * FFT(w) )
    L = fft_len_corr
    A_hat = np.fft.rfft(a, L)
    W_hat = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(A_hat) * W_hat, n=L)   # linear correlation
    n = a.size
    start = n - 1                                      # index where shift = 0
    grad = 2.0 * corr[start:start + n]                 # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine – the function required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size (length of the step function)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # a fairly large size – still cheap to FFT

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)        # for gradient correlation

    # ------------------------------------------------------------------
    # 2. Build a diverse portfolio of initial probability vectors
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells (few widths)
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderately / strongly concentrated)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Sparse masks – a few equally weighted spikes
    for k in [int(np.sqrt(n)), int(np.sqrt(n) * 2), max(2, n // 10)]:
        mask = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=min(k, n), replace=False)
        mask[idx] = 1.0
        mask /= mask.sum()
        init_cands.append(mask)

    # Random binary mask (Bernoulli 0.5)
    bin_mask = rng.binomial(1, 0.5, size=n).astype(np.float64)
    if bin_mask.sum() > 0:
        bin_mask /= bin_mask.sum()
        init_cands.append(bin_mask)

    # previous construction (if present) – normalised to a probability vector
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            # pad with zeros
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        s = v.sum()
        if s == 0.0:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v = v / s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best of the initial candidates (true J, not the
    #    smoothed surrogate)
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len_conv, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop – soft‑maxed gradient + simulated annealing
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time

    # temperature schedule (linear cooling)
    T_start = best_J * 0.05   # 5 % of the current best
    T_end = 1e-8

    # smooth‑max schedule
    beta = 5.0                # start with a gentle soft‑max
    beta_max = 300.0
    beta_inc = 1.02           # multiplicative increase every few iterations

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # recompute exact convolution and true objective
        conv = _conv_full(cur_vec, fft_len_conv, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # keep global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # Stagnation handling – jitter after 30 s without improvement
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.1 Gradient of the smoothed max (soft‑max) objective
        # ------------------------------------------------------------------
        grad = _gradient_softmax(cur_vec, conv, beta, fft_len_corr)

        # ------------------------------------------------------------------
        # 4.2 Propose a new point – exponentiated‑gradient step
        # ------------------------------------------------------------------
        step = 0.5 / np.sqrt(iteration + 1.0)     # decaying step size
        factor = np.exp(-step * grad)            # multiplicative update
        proposal = cur_vec * factor
        proposal = np.clip(proposal, 0.0, None)
        s = proposal.sum()
        if s == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= s

        # ------------------------------------------------------------------
        # 4.3 Evaluate proposal (exact J)
        # ------------------------------------------------------------------
        prop_conv = _conv_full(proposal, fft_len_conv, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ------------------------------------------------------------------
        # 4.4 Metropolis acceptance (linear cooling)
        # ------------------------------------------------------------------
        T = (T_start * (1.0 - elapsed / (deadline - start_time))
             + T_end * (elapsed / (deadline - start_time)))
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        # ------------------------------------------------------------------
        # 4.5 Update smoothing parameter β (slowly increase)
        # ------------------------------------------------------------------
        if iteration % 200 == 0 and beta < beta_max:
            beta = min(beta * beta_inc, beta_max)

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (clip and renormalise for safety)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
