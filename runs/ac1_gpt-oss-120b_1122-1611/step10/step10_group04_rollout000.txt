def _annealing_step(a, fft_len, rng, temperature):
    n = a.size
    # compute gradient contributions
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    grad = _gradient_from_active(a, active)
    # select src with high grad, tgt low
    src = int(np.argmax(grad))
    # pick low gradient candidate
    low_idx = np.where(grad <= np.percentile(grad, 15))[0]
    if low_idx.size:
        tgt = int(rng.choice(low_idx))
    else:
        tgt = int(rng.integers(0, n))
    # propose transfer fraction in [0.001, 0.1] log-uniform maybe bigger for early.
    frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))  # same as before
    delta = a[src] * frac
    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    # evaluate
    conv_new = _conv_fft(a_new, fft_len)
    score_new = 2.0 * a.size * conv_new.max()
    return a_new, score_new

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.  

The optimiser combines several complementary ideas:

* a fast exponentiated‑gradient phase on a smooth approximation of the
  max‑convolution (β increases from 5 to 200);
* a projected sub‑gradient descent on the true max‑convolution;
* directed mass‑transfer refinements that move probability mass from
  high‑gradient to low‑gradient positions;
* a short simulated‑annealing stage that occasionally accepts uphill
  moves to escape shallow local minima;
* occasional symmetrisation (a ← (a+a[::-1])/2) which empirically lowers
  the peak of the self‑convolution;
* a multi‑start strategy (uniform, any previously found sequence and a
  few random Dirichlet starts) and a small schedule of sequence lengths.

All iterates stay on the probability simplex (∑a_i = 1, a_i ≥ 0) so the
objective is simply ``2·n·max(conv)``.  The code respects the 1000 s time
budget, prints progress, and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the active set
    (indices attaining the current maximum). For each active k,
    the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i} and re‑normalise onto the simplex."""
    a = (a + a[::-1]) * 0.5
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase‑1 – smoothed max gradient (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _smooth_max_step(a: np.ndarray,
                     fft_len: int,
                     beta: float,
                     eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step on the smoothed max objective.
    Returns the updated probability vector (still on the simplex).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    max_conv = conv.max()
    # smoothed weights (log‑sum‑exp trick)
    w = np.exp(beta * (conv - max_conv))
    w /= w.sum()                               # now ∑ w_i = 1

    # ----- gradient of the smoothed max -----
    # term2 = (a * w)[k]  for k = 0 … n‑1
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: n + len(w) - 1]        # full linear conv
    term2 = conv_aw[:n]

    # term1 = (a * w_rev)[m‑1‑k] ,  m = len(w)
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: n + len(w_rev) - 1]
    m = len(w)
    term1 = conv_aw_rev[m - 1 - np.arange(n)]

    grad = term1 + term2                         # ∂/∂a of weighted sum

    # exponentiated‑gradient update on log‑probabilities
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Phase‑2 – exact hard‑max sub‑gradient descent
# ----------------------------------------------------------------------


def _hard_max_step(a: np.ndarray,
                   fft_len: int,
                   step_factor: float,
                   iteration: int) -> np.ndarray:
    """
    Projected additive sub‑gradient step for the true max objective.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    step = step_factor / np.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


# ----------------------------------------------------------------------
# Phase‑3 – directed mass‑transfer refinement
# ----------------------------------------------------------------------


def _local_refinement_step(a: np.ndarray,
                           fft_len: int,
                           rng: np.random.Generator,
                           src: int,
                           tgt: int) -> tuple[np.ndarray, int, int]:
    """
    Tiny directed mass‑transfer move.
    Returns (new_sequence, new_src, new_tgt).
    """
    n = a.size
    if a[src] <= 0.0:
        src = int(rng.integers(0, n))
    if src == tgt:
        tgt = int(rng.integers(0, n))

    # transfer fraction in [0.1 % , 5 %] (log‑uniform)
    frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    return a_new, src, tgt


# ----------------------------------------------------------------------
# Phase‑4 – simulated annealing (uphill moves allowed)
# ----------------------------------------------------------------------


def _annealing_step(a: np.ndarray,
                    fft_len: int,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Propose a mass‑transfer move guided by the sub‑gradient of the
    current active set and return (new_a, new_score).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    grad = _gradient_from_active(a, active)

    src = int(np.argmax(grad))
    # pick a target from the low‑gradient tail
    low = np.where(grad <= np.percentile(grad, 15))[0]
    if low.size:
        tgt = int(rng.choice(low))
    else:
        tgt = int(rng.integers(0, n))

    # random transfer fraction (log‑uniform) – a bit larger than in local step
    frac = np.exp(rng.uniform(np.log(0.001), np.log(0.1)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)

    new_conv = _conv_fft(a_new, fft_len)
    new_score = 2.0 * n * new_conv.max()
    return a_new, new_score


# ----------------------------------------------------------------------
# Core optimisation for a single initial sequence
# ----------------------------------------------------------------------


def _optimize_sequence(a_init: np.ndarray,
                       rng: np.random.Generator,
                       deadline: float,
                       target: float = 1.5025) -> tuple[np.ndarray, float]:
    """
    Run the four optimisation phases on a given initial vector.
    Returns the best vector found and its (2·n·max_conv) score.
    """
    a = a_init.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)

    # bookkeeping
    best = a.copy()
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max exponentiated‑gradient
    # ------------------------------------------------------------------
    phase1_time = min(120.0, (deadline - time.monotonic()) * 0.3)
    phase1_end = time.monotonic() + phase1_time
    it = 0
    while time.monotonic() < phase1_end:
        # β ramps from 5 to 200 linearly over the phase
        frac = (time.monotonic() - (phase1_end - phase1_time)) / phase1_time
        beta = 5.0 + frac * (200.0 - 5.0)
        eta = 0.6 / math.sqrt(it + 1.0)
        a = _smooth_max_step(a, fft_len, beta, eta)

        # occasional symmetrisation (every 200 iters)
        if it % 200 == 0:
            a = _symmetrize(a)

        cur_score = 2.0 * n * _conv_fft(a, fft_len).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.monotonic():.2f}s] new best (smooth) = {best_score:.6f}")
            if best_score <= target:
                return best, best_score
        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max sub‑gradient descent
    # ------------------------------------------------------------------
    a = best.copy()
    phase2_time = min(120.0, (deadline - time.monotonic()) * 0.3)
    phase2_end = time.monotonic() + phase2_time
    it = 0
    while time.monotonic() < phase2_end:
        a_candidate = _hard_max_step(a, fft_len, step_factor=0.25, iteration=it)
        cur_score = 2.0 * n * _conv_fft(a, fft_len).max()
        cand_score = 2.0 * n * _conv_fft(a_candidate, fft_len).max()

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = a_candidate.copy()
            print(f"[{time.monotonic():.2f}s] new best (hard) = {best_score:.6f}")
            if best_score <= target:
                return best, best_score

        # accept downhill moves
        if cand_score <= cur_score:
            a = a_candidate

        # symmetrise occasionally
        if it % 300 == 0:
            a = _symmetrize(a)

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – stochastic local mass‑transfer refinement
    # ------------------------------------------------------------------
    a = best.copy()
    src = int(rng.integers(0, n))
    tgt = int(rng.integers(0, n))
    phase3_time = min(120.0, (deadline - time.monotonic()) * 0.2)
    phase3_end = time.monotonic() + phase3_time
    it = 0
    while time.monotonic() < phase3_end:
        if it % 500 == 0:
            a = _symmetrize(a)

        a_candidate, src, tgt = _local_refinement_step(a, fft_len, rng, src, tgt)
        cand_score = 2.0 * n * _conv_fft(a_candidate, fft_len).max()

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = a_candidate.copy()
            print(f"[{time.monotonic():.2f}s] new best (local) = {best_score:.6f}")
            if best_score <= target:
                return best, best_score

        # keep improvement, otherwise stay
        cur_score = 2.0 * n * _conv_fft(a, fft_len).max()
        if cand_score < cur_score:
            a = a_candidate

        it += 1

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (uphill moves allowed)
    # ------------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    T = best_score * 0.01 + 1e-6            # initial temperature
    decay = 0.9995
    it = 0
    while time.monotonic() < deadline:
        a_new, new_score = _annealing_step(a, fft_len, rng)

        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            a = a_new
            cur_score = new_score

        if new_score < best_score - 1e-12:
            best_score = new_score
            best = a_new.copy()
            print(f"[{time.monotonic():.2f}s] new best (anneal) = {best_score:.6f}")
            if best_score <= target:
                return best, best_score

        # temperature schedule
        T *= decay
        # occasional symmetrisation
        if it % 800 == 0:
            a = _symmetrize(a)
            cur_score = 2.0 * n * _conv_fft(a, fft_len).max()
        it += 1

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    target = 1.5025

    best_overall = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0) Gather a few sensible initializations
    # ------------------------------------------------------------------
    init_candidates = []

    # a) any previously discovered construction
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        seq = np.clip(seq, 0.0, None)
        if seq.sum() > 0.0:
            seq = seq / seq.sum()
            init_candidates.append(seq)

    # b) uniform sequences of several lengths
    for n in (2048, 4096, 8000):
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # ------------------------------------------------------------------
    # 1) Run optimisation from each deterministic start
    # ------------------------------------------------------------------
    for a0 in init_candidates:
        if time.monotonic() >= deadline:
            break
        a_opt, score = _optimize_sequence(a0, rng, deadline, target)
        if score < best_score:
            best_score = score
            best_overall = a_opt.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new global best (deterministic) = {best_score:.6f}")
            if best_score <= target:
                break

    # ------------------------------------------------------------------
    # 2) Random Dirichlet restarts while time permits
    # ------------------------------------------------------------------
    lengths = (2048, 4096, 8000)
    while time.monotonic() < deadline and best_score > target:
        n = int(rng.choice(lengths))
        a0 = rng.random(n)
        a0 /= a0.sum()
        a_opt, score = _optimize_sequence(a0, rng, deadline, target)
        if score < best_score:
            best_score = score
            best_overall = a_opt.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new global best (random) = {best_score:.6f}")
            if best_score <= target:
                break

    # ------------------------------------------------------------------
    # 3) Finalise and return
    # ------------------------------------------------------------------
    if best_overall is None:
        # fallback – uniform 4096‑point sequence
        best_overall = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clamp to evaluator's hard limits and evaluate once for reporting
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    final_score = evaluate_sequence(best_overall.tolist())
    total = time.monotonic() - start
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_overall]
