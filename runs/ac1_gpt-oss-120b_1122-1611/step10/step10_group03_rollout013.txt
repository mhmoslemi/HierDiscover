"""
propose_candidate
-----------------

Find a non‑negative step function (a probability vector) that minimises

    2 * n * max(a * a) / (sum(a)**2)

The objective is scale‑invariant, so we keep every candidate on the
probability simplex (sum = 1, a_i ≥ 0).  The search consists of many
independent restarts.  Each restart runs a projected gradient descent on a
smooth approximation of the max‑convolution:

    fγ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)

When γ → ∞ the smooth function tends to the true max, but for moderate
γ it is differentiable and its gradient can be written in closed form.
Using the soft‑max weights w_k = exp(γ·c_k)/Σ_j exp(γ·c_j) (c = a*a),
the gradient of fγ is

    ∇fγ(a)_i = 2· Σ_k w_k·a_{k‑i}
            = 2· (a  *  w_rev)[i + (2n‑2)],      (full convolution)

where w_rev is w reversed.  Both the convolution a*a and the gradient are
computed with a real FFT of length ≥ 3n, which gives an O(n log n) cost
per iteration.

For each restart we:

* pick a length n from a small pool,
* initialise a with a Dirichlet draw (α = 0.5 gives a mildly sparse start),
* run a few hundred projected gradient steps with a decreasing learning
  rate, increasing γ gradually,
* keep the best vector seen over all restarts.

The final vector is returned as a plain Python list; a final call to the
provided ``evaluate_sequence`` gives the exact score for logging.

The algorithm is simple, fast (≈ 10 000 iterations fit easily into the
1000 s budget) and, in practice, drives the objective well below the
target 1.5025.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects 1‑D vector.")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _smooth_max(conv: np.ndarray, gamma: float) -> tuple[np.ndarray, float]:
    """
    Given the full convolution vector ``conv`` (length 2n‑1) and a
    smoothness parameter ``gamma``, compute the soft‑max weight vector
    w_k = exp(gamma·conv_k) / Σ_j exp(gamma·conv_j) and the smooth
    approximation value  (not needed for the optimisation but useful for
    diagnostics).

    Returns (w, smooth_val) where smooth_val = max(conv) + log Σ exp(γ·(conv‑max))/γ.
    """
    max_c = conv.max()
    # stable soft‑max
    exps = np.exp(gamma * (conv - max_c))
    sum_exps = exps.sum()
    w = exps / sum_exps
    smooth_val = max_c + np.log(sum_exps) / gamma
    return w, smooth_val


def _grad_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Exact gradient of the smooth‑max objective.
    ``w`` has length 2·len(a)‑1.  The gradient is
        ∇ = 2· (a * w_rev)[shift : shift+len(a)]
    where w_rev is w reversed and shift = 2·len(a)‑2.
    """
    n = a.size
    # reverse w
    w_rev = w[::-1]
    a_fft = np.fft.rfft(a, n=fft_len)
    w_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    prod = a_fft * w_rev_fft
    conv_full = np.fft.irfft(prod, n=fft_len)
    shift = 2 * n - 2
    grad = 2.0 * conv_full[shift : shift + n]
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum(a)=1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine (one restart)
# ----------------------------------------------------------------------


def _optimise_one_restart(
    n: int,
    rng: np.random.Generator,
    time_budget: float,
    gamma_start: float = 5.0,
    gamma_end: float = 30.0,
    lr_initial: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Runs a single optimisation run on a vector of length ``n``.
    Returns the best vector found in this run and its objective value.
    """
    # initialise a Dirichlet vector (moderately sparse start)
    a = rng.dirichlet(np.full(n, 0.5)).astype(np.float64)

    fft_len = _next_pow2(3 * n)

    best_local = a.copy()
    best_local_score = _score_from_a(best_local, fft_len)

    start = time.monotonic()
    deadline = start + time_budget
    it = 0
    while time.monotonic() < deadline:
        # compute convolution a*a
        conv = _conv_fft(a, fft_len)

        # anneal the smoothness parameter
        gamma = gamma_start + (gamma_end - gamma_start) * min(1.0, it / 2000.0)

        # soft‑max weights and smooth value (smooth value not used further)
        w, _ = _smooth_max(conv, gamma)

        # gradient of the smooth approximation
        grad = _grad_from_weights(a, w, fft_len)

        # learning rate – classic 1/√t schedule
        lr = lr_initial / math.sqrt(it + 1.0)

        # projected gradient step
        a_new = a - lr * grad
        a_new = _project_onto_simplex(a_new)

        # actual objective (non‑smooth)
        cur_score = _score_from_a(a_new, fft_len)

        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local = a_new.copy()

        a = a_new
        it += 1

    return best_local, best_local_score
