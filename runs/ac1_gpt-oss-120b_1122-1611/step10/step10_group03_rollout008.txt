"""
Proposes a non‑negative step function (probability vector) that minimises
    2 * n * max(a * a) / (sum(a)**2)

Algorithm
---------
* Initialise from the previously found best sequence (or a uniform start).
* Run a long simulated‑annealing / stochastic‑gradient search for the whole
  allocated budget.
* At each iteration:
    - Compute the current self‑convolution (FFT) and the active set of indices
      attaining (or nearly attaining) the maximum.
    - Obtain the exact sub‑gradient of the max‑convolution w.r.t. the
      coefficients from the active set.
    - Choose a source index where the gradient is positive (mass should be
      removed) and a destination where the gradient is negative (mass should be
      added).  If no such indices exist, fall back to random picks.
    - Transfer a random *log‑uniform* fraction of the source mass to the
      destination (or split it among two destinations) and project back onto
      the simplex.
    - Accept the move if it improves the score, otherwise accept with the
      usual Metropolis probability exp(-(Δ)/T) where the temperature T
      decreases linearly to zero.
    - Every few thousand steps apply a small multiplicative jitter on a random
      block of coordinates to escape plateaus.
* Keep the best vector seen; stop early if the target score 1.5025 is reached.
* Return the best vector after clipping it to the evaluator limits.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to the centre of the simplex
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set ``active`` of indices
    where the current maximum is attained (or nearly attained).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum(a)=1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())

            cand[dst] += delta

        # keep the vector on the simplex (numerical drift)
        cand = _project_onto_simplex(cand)

        # --------------------------------------------------------------
        # 4.  Evaluate the candidate
        # --------------------------------------------------------------
        cand_score = _score_from_a(cand, fft_len)

        # --------------------------------------------------------------
        # 5.  Metropolis acceptance
        # --------------------------------------------------------------
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        else:
            if T > 0.0:
                prob = math.exp(-(cand_score - cur_score) / T)
                if rng.random() < prob:
                    a = cand
                    cur_score = cand_score

        # --------------------------------------------------------------
        # 6.  Update global best
        # --------------------------------------------------------------
        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f}")

            # early stop if we have already reached the target
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

        # --------------------------------------------------------------
        # 7.  Occasional jitter to escape flat regions
        # --------------------------------------------------------------
        if it % 4000 == 0 and it > 0:
            start_idx = int(rng.integers(0, n - jitter_width + 1))
            factor = math.exp(rng.normal(0.0, jitter_sigma))
            j = a.copy()
            j[start_idx:start_idx + jitter_width] *= factor
            a = _project_onto_simplex(j)
            cur_score = _score_from_a(a, fft_len)

        it += 1

    # ------------------------------------------------------------------
    # Final processing – clip to evaluator limits and compute the exact score
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # evaluator caps values anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # make the sequence available for a potential next run
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
