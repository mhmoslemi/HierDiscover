def _local_optimize(a, deadline):
    n = a.size
    fft_len = _next_pow2(3*n)
    # step 1: projected subgradient descent
    # We'll do limited iterations if time low.
    max_iter = 3000
    it = 0
    best_a = a
    best_score = _score_of_seq(a)
    while it < max_iter and time.monotonic() < deadline:
        # compute gradient
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        active = np.where(conv >= max_b * (1 - 1e-8))[0]
        grad = _gradient_from_active(a, active)
        # step size
        step = 0.5 / np.sqrt(it+1)
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate)
        # evaluate
        score_candidate = _score_of_seq(a_candidate, fft_len)
        if score_candidate < best_score - 1e-12:
            best_score = score_candidate
            best_a = a_candidate.copy()
        # accept if not worse
        if score_candidate <= best_score + 1e-12:
            a = a_candidate
        # else keep old a
        it += 1
    # Phase: small directed moves (hill climbing)
    src = int(np.argmax(best_a))
    tgt = int(np.argmin(best_a))
    while time.monotonic() < deadline:
        # choose src perhaps from high gradient region
        conv = _conv_fft(best_a, fft_len)
        max_b = conv.max()
        active = np.where(conv >= max_b * (1 - 1e-8))[0]
        grad = _gradient_from_active(best_a, active)
        src = int(np.argmax(grad))
        # pick low gradient target
        low_idx = np.where(grad <= np.percentile(grad, 15))[0]
        if low_idx.size:
            tgt = int(np.random.choice(low_idx))
        else:
            tgt = int(np.random.integers(0, n))
        # fraction
        if best_a[src] <= 0:
            continue
        frac = np.exp(np.random.uniform(np.log(1e-5), np.log(1e-2)))
        delta = best_a[src] * frac
        a_new = best_a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        a_new = _project_onto_simplex(a_new)
        score_new = _score_of_seq(a_new, fft_len)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new

"""
Propose a step‑function (non‑negative list of heights) that minimises
``evaluate_sequence``.  The routine combines three ideas:

1.  **Diverse random initialisations** – uniform, Dirichlet, geometric,
    power‑law and truncated linear shapes, each normalised to sum = 1.
2.  **Projected sub‑gradient descent** on the exact max‑convolution
    objective, with a simple back‑tracking line‑search.
3.  **Fine‑grained stochastic mass‑transfer** – repeatedly move a tiny
    fraction of the mass from the index that most contributes to the
    current maximal convolution to an index that contributes little.
    The move is accepted only if it does not increase the objective.
    This local refinement is run until the time budget is exhausted.

The algorithm repeatedly restarts while time permits, keeping the best
sequence seen across all runs.  The final sequence is clipped to
[0, 1000] (the limits of ``evaluate_sequence``) and returned as a list
of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:                     # all entries are zero
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the given active set
    (indices attaining the current maximum).  For each active index k
    the contribution to ∂max/∂a_i is a_{k-i} whenever the index lies
    inside the support.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score_of_seq(a: np.ndarray, fft_len: int) -> float:
    """Return the value of the objective (2·n·max_convolution) for a."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_descent(a: np.ndarray,
                         deadline: float,
                         max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a simple back‑tracking line‑search on the step size.
    Returns the improved vector and its score.
    """
    n = a.size
    fft_len = _next_pow2(3 * n)

    best_a = a.copy()
    best_score = _score_of_seq(best_a, fft_len)

    it = 0
    while it < max_iters and time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        # active set (indices whose value is within 1e-7 relative tolerance)
        eps = max_b * 1e-7
        active = np.where(conv >= max_b - eps)[0]
        grad = _gradient_from_active(a, active)
        # descent direction
        direction = -grad

        # try a handful of step lengths (geometric series)
        improved = False
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            cand = a + alpha * direction
            cand = _project_onto_simplex(cand)
            cand_score = _score_of_seq(cand, fft_len)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = cand.copy()
                improved = True
                break
        if improved:
            a = cand
        else:
            # no improvement – shrink step size aggressively
            a = _project_onto_simplex(a - 0.01 * direction)
        it += 1

    return best_a, best_score


def _micro_refinement(a: np.ndarray,
                      best_score: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Fine‑grained stochastic mass‑transfer refinement.
    Repeatedly moves a tiny fraction of probability mass from an index
    that contributes most to the active convolution peak to a low‑impact
    target.  Accepts only non‑worsening moves.
    """
    n = a.size
    fft_len = _next_pow2(3 * n)

    # initial source / target – will be refreshed during the loop
    src = int(np.argmax(a))
    tgt = int(np.argmin(a))

    # counters for occasional refresh of src/tgt
    refresh_counter = 0

    while time.monotonic() < deadline:
        # recompute gradient only every few iterations to save time
        if refresh_counter % 30 == 0:
            conv = _conv_fft(a, fft_len)
            max_b = conv.max()
            eps = max_b * 1e-6
            active = np.where(conv >= max_b - eps)[0]
            grad = _gradient_from_active(a, active)

            src = int(np.argmax(grad))

            # choose a low‑gradient destination (bottom 15 % quantile)
            thresh = np.percentile(grad, 15)
            low = np.where(grad <= thresh)[0]
            if low.size:
                tgt = int(np.random.choice(low))
            else:
                tgt = int(np.random.integers(0, n))

        # very small fraction (log‑uniform between 1e‑6 and 1e‑3)
        frac = np.exp(np.random.uniform(np.log(1e-6), np.log(1e-3)))
        delta = a[src] * frac
        if delta <= 0.0:
            refresh_counter += 1
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        cand_score = _score_of_seq(cand, fft_len)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            a = cand
        else:
            # keep the move only if it does not increase the objective
            if cand_score <= 2 * n * _conv_fft(a, fft_len).max():
                a = cand

        refresh_counter += 1

    return a, best_score


# ----------------------------------------------------------------------
# Random initialisers (parameterised families)
# ----------------------------------------------------------------------


def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    # random concentration in [0.2, 5.0]  (spiky → flat)
    conc = rng.uniform(0.2, 5.0)
    sample = rng.gamma(conc, 1.0, size=n)
    return sample / sample.sum()


def _make_geometric(n: int, rng: np.random.Generator) -> np.ndarray:
    # r in (0.9, 0.99995)  – log‑uniform sampling of decay factor
    r = np.exp(rng.uniform(np.log(0.9), np.log(0.99995)))
    coeff = (1.0 - r) / (1.0 - r**n)
    a = coeff * (r ** np.arange(n, dtype=np.float64))
    if rng.random() < 0.5:               # random reversal
        a = a[::-1]
    return a


def _make_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 5.0)            # exponent
    w = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return w / w.sum()


def _make_trunc_linear(n: int, rng: np.random.Generator) -> np.ndarray:
    # fraction of indices that carry mass (0.3 – 0.9)
    frac = rng.uniform(0.3, 0.9)
    length = int(np.ceil(frac * n))
    # linearly decreasing on the support, zero afterwards
    vals = np.linspace(1.0, 0.0, length, endpoint=False)
    a = np.concatenate([vals, np.zeros(n - length, dtype=np.float64)])
    return a / a.sum()


def _random_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick one of the families at random."""
    choice = rng.integers(0, 5)
    if choice == 0:
        return _make_uniform(n)
    if choice == 1:
        return _make_dirichlet(n, rng)
    if choice == 2:
        return _make_geometric(n, rng)
    if choice == 3:
        return _make_powerlaw(n, rng)
    return _make_trunc_linear(n, rng)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    # keep a 5‑second safety margin for the final call to evaluate_sequence
    deadline = start + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0)  Use the previously‑found construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() > 0.0:
            a0 /= a0.sum()
        else:
            a0 = _make_uniform(len(a0))
        a0, sc0 = _subgradient_descent(a0, deadline)
        a0, sc0 = _micro_refinement(a0, sc0, deadline)
        best_seq, best_score = a0, sc0
        elapsed = time.monotonic() - start
        print(f"[{elapsed:.2f}s] start from known best → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1)  Repeated random restarts + local optimisation
    # ------------------------------------------------------------------
    n_options = [2048, 4096, 8192]                 # lengths we will try
    while time.monotonic() < deadline:
        # pick a length (bias towards larger, but keep some diversity)
        n = int(rng.choice(n_options))

        # generate a random initial vector
        a = _random_initialiser(n, rng)

        # short sub‑gradient phase (time‑limited)
        a, sc = _subgradient_descent(a, deadline, max_iters=1200)

        # micro‑refinement phase (uses the remaining budget)
        a, sc = _micro_refinement(a, sc, deadline)

        if sc < best_score - 1e-12:
            best_seq, best_score = a.copy(), sc
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new global best = {best_score:.6f} (n={n})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # 2)  Final safety call to the official evaluator
    # ------------------------------------------------------------------
    if best_seq is None:                     # should never happen
        best_seq = _make_uniform(4096)

    # clip to the limits enforced inside ``evaluate_sequence``
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    final_score = evaluate_sequence(best_seq.tolist())
    total = time.monotonic() - start
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")

    # return as plain list of Python floats
    return [float(x) for x in best_seq]
