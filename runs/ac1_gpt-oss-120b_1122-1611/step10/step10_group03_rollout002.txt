1.  Represent a as a probability vector (sum = 1).  The objective depends only
    on the maximum entry of the autocorrelation `c = a * a`.

2.  Replace the non‑smooth `max` by a smooth upper bound
        f_λ(a) = (log ∑_k exp(λ c_k)) / λ,
    which converges to `max(c)` as λ → ∞.  For a fixed λ the gradient is
        ∇f_λ(a) = 2 * corr(a, softmax_λ(c)),
    where `corr` is the (full) cross‑correlation and the soft‑max weights are
    `exp(λ c_k) / Σ exp(λ c)`.  This gradient is inexpensive to evaluate
    with FFTs.

3.  Perform an Adam‑style projected gradient descent while slowly raising λ
    from a modest start (λ≈1) to a larger final value (λ≈12).  After each
    update the iterate is projected back onto the probability simplex.

4.  Run many short restarts with different lengths and random Dirichlet
    starts (different concentration parameters).  The current global best
    is periodically jittered to escape basins.

5.  A cheap stochastic “mass‑transfer” refinement (the same idea as the
    previous code’s Phase 3) is applied for the remaining time.

The whole search stops before the given budget (default 1000 s).  The best
sequence found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise (numerical safety)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_general(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two (potentially different) vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: len(x) + len(y) - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate + gradient (log‑sum‑exp)
# ----------------------------------------------------------------------


def _softmax_obj_and_grad(a: np.ndarray, lam: float, fft_len: int):
    """
    For a probability vector a (sum = 1) compute

        f_λ(a) = (log ∑_k exp(λ·(a*a)_k)) / λ

    and its gradient ∇f_λ(a) = 2 * corr(a, softmax_λ(a*a)).

    Returns
        obj   – the smooth upper bound on max(a*a)
        grad  – same shape as a
        max_val – actual max(a*a) (useful for bookkeeping)
    """
    n = a.size
    # 1) convolution a * a
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    max_val = conv.max()

    # 2) soft‑max weights (stable version)
    shifted = lam * (conv - max_val)            # max becomes 0
    w = np.exp(shifted)
    sum_w = w.sum()
    soft = w / sum_w                            # length 2n‑1, sums to 1

    # 3) gradient = 2 * correlation(a, soft)
    soft_rev = soft[::-1]                       # reversed soft for correlation
    corr = _conv_fft_general(a, soft_rev, fft_len)   # length 3n‑2
    offset = (2 * n - 1) - 1                    # 2n‑2
    grad = 2.0 * corr[offset: offset + n]       # keep only n entries

    # 4) smooth objective value (upper bound on max)
    obj = max_val + np.log(sum_w) / lam

    return obj, grad, max_val


# ----------------------------------------------------------------------
# Annealed Adam optimiser for a single run
# ----------------------------------------------------------------------


def _annealed_adam_optimize(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Perform a deterministic‑annealing Adam optimisation on ``a0``.
    The schedule linearly raises the smoothing parameter λ from λ_start to λ_end.
    The iterate is always projected onto the probability simplex.
    Returns the best vector seen (as a NumPy array) and its exact objective
    value 2·n·max(a*a).
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)

    # best record for this run
    best = a.copy()
    best_score = float(2.0 * n * _conv_fft(a, fft_len).max())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # annealing schedule
    lam_start = 1.0
    lam_end = rng.uniform(10.0, 15.0)   # a bit of randomness per run

    start = time.monotonic()
    deadline = start + time_budget

    it = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        lam = lam_start + (lam_end - lam_start) * (progress ** 0.6)

        # smooth objective + gradient
        obj, grad, _ = _softmax_obj_and_grad(a, lam, fft_len)

        # Adam update (projected)
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step‑size schedule – shrink as we approach the deadline
        base_lr = 0.15
        lr = base_lr * (1.0 - progress) + 0.002

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a)

        # exact score for the current point (cheap: reuse conv)
        conv = _conv_fft(a, fft_len)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()

        # occasional small jitter to shake the iterate (helps escape flat plateaus)
        if it % 3000 == 0:
            width = max(1, n // 150)
            s = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.015))
            jitter = a.copy()
            jitter[s:s + width] *= factor
            a = _project_onto_simplex(jitter)

    return best, best_score


# ----------------------------------------------------------------------
# Very cheap stochastic local refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _local_mass_transfer(a: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Simple stochastic refinement that repeatedly moves a random fraction of
    mass from a source index to a target index.  Improvements are kept;
    occasional uphill moves are allowed with a tiny simulated‑annealing rule.
    Returns the best vector found and its exact score.
    """
    n = a.size
    fft_len = _next_pow2(3 * n)

    best = a.copy()
    best_score = float(2.0 * n * _conv_fft(best, fft_len).max())
    cur = a.copy()
    cur_score = best_score

    start = time.monotonic()
    deadline = start + time_budget

    # pre‑compute a few constants
    min_frac = 0.001
    max_frac = 0.12

    while time.monotonic() < deadline:
        # choose source and target
        src = int(rng.integers(0, n))
        while cur[src] <= 0.0:
            src = int(rng.integers(0, n))
        tgt = int(rng.integers(0, n))
        if tgt == src:
            tgt = (tgt + 1) % n

        # random fraction (log‑uniform)
        frac = np.exp(rng.uniform(np.log(min_frac), np.log(max_frac)))
        delta = cur[src] * frac

        cand = cur.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # projection is cheap, but the move already stays on the simplex
        # (total mass unchanged, negativity prevented by construction)
        cand = _project_onto_simplex(cand)

        cand_score = float(2.0 * n * _conv_fft(cand, fft_len).max())

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand.copy()
            cur = cand
            cur_score = cand_score
        else:
            # simulated‑annealing acceptance
            temperature = max(1e-4, (deadline - time.monotonic()) /
                              (deadline - start))
            prob = math.exp(-(cand_score - cur_score) / temperature) \
                    if cand_score > cur_score else 1.0
            if rng.random() < prob:
                cur = cand
                cur_score = cand_score

        # occasional jitter to keep exploration alive
        if rng.random() < 0.001:
            w = max(1, n // 120)
            start_idx = int(rng.integers(0, n - w + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            jitter = cur.copy()
            jitter[start_idx:start_idx + w] *= factor
            cur = _project_onto_simplex(jitter)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine performs many short random‑restart
    runs of the annealed Adam optimiser (different lengths and initialisations)
    and finishes with a light stochastic refinement.
    The best sequence seen before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise global best – reuse a previously discovered sequence if any.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        if best.sum() > 0.0:
            best = best / best.sum()
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        # start with a modest length uniform vector
        best = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_score = float(2.0 * best.size * _conv_fft(best, _next_pow2(3 * best.size)).max())
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # ------------------------------------------------------------------
    # Lengths we will explore.  Powers of two plus a few odd numbers to add
    # diversity.  The pool is static – we just pick from it.
    # ------------------------------------------------------------------
    length_pool = [1024, 2048, 3072, 3809, 4096, 6144, 8192]

    # time reserved for a single optimisation run (seconds)
    max_run_time = 30.0

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            break

        # ----------------------------------------------------------------
        # Choose a length for the new run (biased towards current best)
        # ----------------------------------------------------------------
        if rng.random() < 0.55:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # ----------------------------------------------------------------
        # Build a starting point
        # ----------------------------------------------------------------
        if n == best.size:
            # jitter the best to escape the present basin of attraction
            a0 = best.copy()
            noise = rng.normal(scale=0.02, size=n)
            a0 = np.maximum(a0 + noise, 0.0)
            a0 = _project_onto_simplex(a0)
        else:
            # Dirichlet start – random concentration α ∈ [0.2, 3.0]
            alpha_val = rng.uniform(0.2, 3.0)
            a0 = rng.dirichlet(np.full(n, alpha_val)).astype(np.float64)

        # ----------------------------------------------------------------
        # Run the annealed Adam optimiser with the allocated budget
        # ----------------------------------------------------------------
        run_budget = min(max_run_time, remaining)
        a_opt, score_opt = _annealed_adam_optimize(a0, run_budget, rng)

        # ----------------------------------------------------------------
        # Update the global best if we improved
        # ----------------------------------------------------------------
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            # early stop if we have already crossed the target threshold
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # Final cheap stochastic refinement (few seconds left)
    # ------------------------------------------------------------------
    time_left = deadline - time.monotonic()
    if time_left > 1.0:
        best, best_score = _local_mass_transfer(best, time_left, rng)

    # ------------------------------------------------------------------
    # Final processing – respect evaluator caps and compute the exact score
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # expose the best sequence for a possible subsequent run
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
