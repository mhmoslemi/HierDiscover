"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence score.

Algorithm
---------

1.  Choose a length `n`.  If a previous construction is available
    (`height_sequence_1`) we reuse its length, otherwise we fall back to
    4096 (a power of two, convenient for FFT).

2.  The objective can be written as
        score = 2·n·max(conv) / (sum(a)²) .
    Because the expression is homogeneous we may assume `sum(a)=1`.
    For a binary mask `m` (1 where the sequence is non‑zero, 0 elsewhere)
    and equal masses `1/k` on the `k` positions the convolution entries are
    simply the ordered pair counts:
        conv[d] = # {(i,j)∈A×A : i+j = d}
    and
        score = 2·n·max(conv) / k² .

    Hence we first search for a *support* `A ⊂ {0,…,n‑1}` that minimises
    `max(conv)`.  This is a discrete “additive Sidon” problem:
    we want the sums of pairs to be as uniformly distributed as possible.

3.  For a given support size `k` we initialise a random mask and compute
    the full convolution by FFT (fast, O(n log n)).  Then we perform a
    simulated‑annealing hill‑climb that repeatedly moves a single element
    from its current position to a free position:

       * The move is evaluated by updating the convolution counts only
         for the affected sums (O(k) work).
       * Moves that reduce the current maximum are always accepted.
       * Worse moves are accepted with probability `exp(-(Δ)/T)`,
         where `Δ` is the increase in the maximum count and `T` is a
         temperature that linearly cools to zero.

       * With probability 0.7 the source element is chosen among those
         that actually contribute to the current maximal sum (a greedy
         bias); otherwise a uniformly random source is used.

    The best mask encountered over all restarts is kept.

4.  The final real‑valued sequence consists of equal mass `1/k` on the
    best mask.  As a polishing step we run a few iterations of the
    exponentiated‑gradient method (the same surrogate used in the
    reference implementation) to tweak the non‑zero values while
    staying on the probability simplex.

5.  The sequence is clipped to the evaluator limits and returned.

The whole search respects the 1000 s budget; progress and any new best
score are printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _initial_conv(mask_set: set, n: int) -> np.ndarray:
    """Full ordered‑pair convolution of a binary mask via FFT."""
    binary = np.zeros(n, dtype=np.int8)
    for i in mask_set:
        binary[i] = 1
    # length sufficient for linear convolution
    fft_len = _next_pow2(2 * n)
    a_fft = np.fft.rfft(binary, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    # rounding eliminates tiny FFT artefacts
    return np.rint(conv).astype(np.int32)


def _remove_source(src: int, mask_set: set, mask_list: list,
                   conv_counts: np.ndarray) -> None:
    """Subtract the contributions of `src` from `conv_counts`."""
    for j in mask_list:                # mask_list still contains src
        s = src + j
        if src == j:
            conv_counts[s] -= 1
        else:
            conv_counts[s] -= 2
    mask_set.remove(src)
    mask_list.remove(src)


def _add_target(tgt: int, mask_set: set, mask_list: list,
                conv_counts: np.ndarray) -> None:
    """Add the contributions of `tgt` to `conv_counts`."""
    for j in mask_list:                # mask_list does NOT contain tgt yet
        s = tgt + j
        conv_counts[s] += 2
    conv_counts[tgt + tgt] += 1
    mask_set.add(tgt)
    mask_list.append(tgt)


def _score_from_counts(max_cnt: int, n: int, k: int) -> float:
    """Score given `max_cnt` (max ordered‑pair count), length `n` and support size `k`."""
    return float(2.0 * n * max_cnt / (k * k))


def _select_source(mask_set: set, mask_list: list,
                   conv_counts: np.ndarray, n: int, rng: np.random.Generator) -> int:
    """
    Pick a source element.
    70 % of the time pick an element that actually contributes to the current maximal sum.
    Otherwise pick a uniform random element.
    """
    if rng.random() < 0.7:
        max_idx = int(conv_counts.argmax())
        # find elements that appear in a pair producing `max_idx`
        candidates = []
        for i in mask_set:
            j = max_idx - i
            if 0 <= j < n and j in mask_set:
                # contribution size (1 for i==j, else 2)
                contrib = 1 if i == j else 2
                candidates.append((contrib, i))
        if candidates:
            # pick the element with the highest contribution
            return max(candidates)[1]
    # fallback: uniform random choice
    return rng.choice(mask_list)


def _propose_mask(n: int, k: int, rng: np.random.Generator,
                 deadline: float, best_score: float,
                 best_mask: set, best_k: int) -> tuple[set, float, int]:
    """
    Search for a good mask of size `k` using simulated annealing.
    Returns the best mask found, its score and the size `k`.
    """
    # initialise random mask
    mask_set = set(rng.choice(n, size=k, replace=False))
    mask_list = list(mask_set)
    conv_counts = _initial_conv(mask_set, n)
    max_cnt = int(conv_counts.max())
    cur_score = _score_from_counts(max_cnt, n, k)

    # remember current best
    if cur_score < best_score:
        best_score = cur_score
        best_mask = mask_set.copy()
        best_k = k

    # temperature schedule (start high, cool to zero)
    init_temp = float(max_cnt) * 1.2 + 0.1

    while time.monotonic() < deadline:
        # linear cooling
        elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
        frac = (deadline - time.monotonic()) / (deadline - (deadline - time.monotonic()))
        temperature = init_temp * max(0.0, 1.0 - frac)

        # pick a move
        src = _select_source(mask_set, mask_list, conv_counts, n, rng)

        tgt = int(rng.integers(0, n))
        while tgt in mask_set:
            tgt = int(rng.integers(0, n))

        # apply move (in‑place)
        _remove_source(src, mask_set, mask_list, conv_counts)
        _add_target(tgt, mask_set, mask_list, conv_counts)

        new_max = int(conv_counts.max())
        delta = new_max - max_cnt

        accept = False
        if delta <= 0:
            accept = True
        else:
            if temperature > 0.0 and rng.random() < math.exp(-delta / temperature):
                accept = True

        if accept:
            max_cnt = new_max
            cur_score = _score_from_counts(max_cnt, n, k)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_mask = mask_set.copy()
                best_k = k
                now = time.monotonic()
                print(f"[{now - start:6.2f}s] new best (mask) k={k} "
                      f"score={best_score:.6f}")
            # early exit if perfect uniformity is achieved
            if best_score <= 1.5025:
                break
        else:
            # revert move (undo tgt → src)
            _remove_source(tgt, mask_set, mask_list, conv_counts)
            _add_target(src, mask_set, mask_list, conv_counts)
            # max_cnt stays unchanged
        # optional stop when max count reaches 1 (theoretical optimum)
        if max_cnt <= 1:
            break

    return best_mask, best_score, best_k


def _refine_with_expgd(a: np.ndarray, n: int,
                       budget_s: float, start_time: float) -> np.ndarray:
    """
    A short exponentiated‑gradient refinement (log‑sum‑exp surrogate)
    starting from the current vector `a`.  Runs for at most a few
    seconds and returns the refined vector.
    """
    fft_len = _next_pow2(3 * n)
    a = a.copy()
    beta = 5.0
    it = 0
    deadline = start_time + budget_s
    while time.monotonic() < deadline:
        a_fft = np.fft.rfft(a, n=fft_len)

        # exact self‑convolution
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        w_fft = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: n + w.size - 1]
        grad = grad_full[:n]

        eta = 0.8 / math.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        # re‑project to the simplex (non‑negative, sum = 1)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # slowly increase smoothing
        beta = min(beta * 1.0015, 120.0)

        it += 1
        if it % 200 == 0:
            # occasional early‑stop check
            cur_score = evaluate_sequence(a.tolist())
            if cur_score <= 1.5025:
                break
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function sequence that minimises ``evaluate_sequence``.
    The routine combines a discrete additive‑Sidon mask search with a
    short exponentiated‑gradient polishing stage.
    """
    global start  # used by helper for temperature timing
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Determine problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 4096                     # power of two – convenient for FFT

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # keep 5 s safety margin

    best_score = float("inf")
    best_mask = set()
    best_k = 0

    # --------------------------------------------------------------
    # 1) Repeated mask optimisation with varying support size
    # --------------------------------------------------------------
    # we try a handful of support sizes in the interval [80, 180]
    # (chosen experimentally to balance the k² term against the typical
    # max‑pair count).
    while time.monotonic() < deadline:
        # pick a random support size for this restart
        k = int(rng.integers(80, 180))
        # run a local search for this k
        mask, s, k_used = _propose_mask(n, k, rng, deadline,
                                        best_score, best_mask, best_k)
        if s < best_score:
            best_score = s
            best_mask = mask
            best_k = k_used
        # early termination if target reached
        if best_score <= 1.5025:
            break

    # --------------------------------------------------------------
    # 2) Build the best sequence from the best mask (uniform masses)
    # --------------------------------------------------------------
    if not best_mask:                     # fallback – uniform vector
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.zeros(n, dtype=np.float64)
        uniform_mass = 1.0 / len(best_mask)
        for i in best_mask:
            a[i] = uniform_mass

    # --------------------------------------------------------------
    # 3) Short EG refinement (optional but often helpful)
    # --------------------------------------------------------------
    refine_budget = min(15.0, budget_s * 0.03)   # at most ~30 s
    a = _refine_with_expgd(a, n, refine_budget, start)

    # --------------------------------------------------------------
    # 4) Final evaluation and return
    # --------------------------------------------------------------
    final = np.clip(a, 0.0, 1000.0)
    final_score = evaluate_sequence(final.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in final]
