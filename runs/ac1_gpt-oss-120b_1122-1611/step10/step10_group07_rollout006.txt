def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)

    # Determine n
    if "height_sequence_1" in globals():
        init_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        init_seq = np.clip(init_seq, 0, None)
        # Ensure sum>0
        s = init_seq.sum()
        if s == 0:
            init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
        else:
            init_seq = init_seq / s
        n = init_seq.size
    else:
        n = 4096
        init_seq = np.full(n, 1.0 / n)

    # We'll use a multistage block representation.
    stages = [64, 128, 256]   # block counts, adjust if >n
    best_seq = None
    best_score = np.inf

    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)

    # Helper functions
    def score(seq):
        conv = _conv_fft(seq, _next_pow2(3*n))
        return float(2.0 * n * conv.max())

    def expand_from_blocks(blocks):
        # blocks shape K, need to repeat each to length n (some leftover)
        K = blocks.size
        block_len = n // K
        seq = np.repeat(blocks, block_len)
        # fill remainder if any
        if seq.size < n:
            seq = np.concatenate([seq, np.full(n - seq.size, blocks[-1])])
        elif seq.size > n:
            seq = seq[:n]
        return seq

    # Stage loop
    current_blocks = None
    for stage_idx, K in enumerate(stages):
        if time.monotonic() >= deadline:
            break
        # If we have a previous finer sequence, summarise to K blocks
        if current_blocks is None:
            # initialise from init_seq using block averaging
            block_len = n // K
            blocks = np.array([init_seq[i*block_len:(i+1)*block_len].sum() for i in range(K)], dtype=np.float64)
            # remainder
            if block_len*K < n:
                remainder = init_seq[block_len*K:].sum()
                blocks = np.append(blocks, remainder)
                K = blocks.size
        else:
            # Upsample previous blocks to new K by splitting each block into two (or proportionally)
            # Simple linear interpolation: replicate each weight proportionally
            # For simplicity, use repeats: each old block weight divided equally among appropriate number of new blocks
            old_K = current_blocks.size
            factor = K // old_K
            if factor * old_K != K:
                # use nearest integer replication then adjust
                factor = K // old_K
                extra = K - factor * old_K
                blocks = np.repeat(current_blocks, factor)
                # distribute extra weight equally from first blocks
                if extra > 0:
                    extra_vec = current_blocks[:extra] / factor
                    blocks = np.concatenate([blocks, extra_vec])
                # ensure sum =1
                blocks = blocks / blocks.sum()
            else:
                blocks = np.repeat(current_blocks, factor)
                blocks = blocks / blocks.sum()
        # Normalize
        blocks = np.clip(blocks, 0, None)
        if blocks.sum() == 0:
            blocks = np.full_like(blocks, 1.0 / blocks.size)
        else:
            blocks = blocks / blocks.sum()

        # Main search within this stage: SA + expgrad
        # Determine time allocation for this stage (remaining time equally divided)
        remaining_time = deadline - time.monotonic()
        # allocate fraction of remaining based on stage index
        stage_budget = remaining_time / (len(stages) - stage_idx)

        stage_start = time.monotonic()
        stage_end = stage_start + stage_budget

        # initialize best for this stage
        seq = expand_from_blocks(blocks)
        cur_score = score(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.monotonic() - start:.2f}s] stage {stage_idx} initial score {best_score:.6f}")

        # Simulated annealing parameters
        T0 = 0.2 * cur_score   # temperature proportional to score
        decay = 0.995
        T = T0
        # adaptation for proposal sigma
        sigma = 0.05

        it = 0
        while time.monotonic() < stage_end:
            # Propose new blocks by adding Gaussian noise to two random indices (conserving sum)
            # Choose source and target index
            i = int(rng.integers(0, K))
            j = int(rng.integers(0, K))
            if i == j:
                continue
            # amount to move
            delta = blocks[i] * rng.uniform(0, sigma)
            # move delta from i to j (keeping nonnegative)
            if delta > blocks[i]:
                delta = blocks[i]
            new_blocks = blocks.copy()
            new_blocks[i] -= delta
            new_blocks[j] += delta
            # ensure positivity via clipping might be unnecessary because we guarantee delta <= blocks[i]
            # project onto simplex (though sum already 1)
            # For numerical stability call projection
            new_blocks = _project_onto_simplex(new_blocks)

            # expand and evaluate
            new_seq = expand_from_blocks(new_blocks)
            new_score = score(new_seq)

            # Acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                blocks = new_blocks
                cur_score = new_score
                # update best
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:.2f}s] new best SA (stage {stage_idx}) = {best_score:.6f} (iter={it})")
            # cooldown
            T *= decay
            it += 1
            if it % 2000 == 0:
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] stage {stage_idx} iter={it}, cur={cur_score:.6f}, best={best_score:.6f}")

        # After SA, run a few exponentiated‑gradient steps on full vector for refinement
        a = best_seq.copy()
        # allocate small budget for gradient steps (maybe 5% of remaining)
        grad_end = min(deadline, time.monotonic() + 0.05 * (deadline - time.monotonic()))
        it_exp = 0
        while time.monotonic() < grad_end:
            conv = _conv_fft(a, _next_pow2(3*n))
            max_b = conv.max()
            eps = max_b * 1e-7
            active = np.where(conv >= max_b - eps)[0]
            if active.size > 50:
                active = rng.choice(active, size=50, replace=False)
            grad = _gradient_from_active(a, active)
            eta = 0.4 / np.sqrt(it_exp + 1.0)
            log_a = np.log(a + 1e-15) - eta * grad
            a = np.exp(log_a)
            a /= a.sum()
            cur_score = score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] new best exp‑grad (stage {stage_idx}) = {best_score:.6f}")
                if best_score <= 1.5025:
                    break
            it_exp += 1
            if it_exp % 500 == 0:
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] exp‑grad iter={it_exp}, cur={cur_score:.6f}")
        # Store refined blocks for next stage
        current_blocks = np.array([best_seq[i* (n // K):(i+1)*(n // K)].sum() for i in range(K)], dtype=np.float64)
        # Normalize
        current_blocks = np.clip(current_blocks, 0, None)
        if current_blocks.sum() > 0:
            current_blocks /= current_blocks.sum()
        else:
            current_blocks = np.full_like(current_blocks, 1.0 / K)

    # Final local refinement on full vector (as in previous code)
    a = best_seq.copy()
    # allocate remaining time
    local_end = deadline
    it_loc = 0
    while time.monotonic() < local_end:
        conv = _conv_fft(a, _next_pow2(3*n))
        idx_max = int(np.argmax(conv))
        grad_max = _gradient_from_active(a, np.array([idx_max]))
        src = int(np.argmax(grad_max))
        low = np.where(grad_max <= grad_max.min() + 1e-12)[0]
        if low.size > 1 and rng.random() < 0.8:
            tgt = int(rng.choice(low))
        else:
            tgt = int(rng.integers(0, n))
        if src == tgt:
            tgt = int(rng.integers(0, n))
        # mass transfer with random fraction
        frac = np.exp(rng.uniform(np.log(0.0005), np.log(0.02)))
        delta = a[src] * frac
        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta
        a_cand = _project_onto_simplex(a_cand)
        cur_score = score(a_cand)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new best (local) = {best_score:.6f} (move={it_loc})")
            if best_score <= 1.5025:
                break
        # accept if not worsening
        cur_max = conv.max()
        if cur_score <= 2.0 * n * cur_max:
            a = a_cand
        # jitter occasionally
        if it_loc % 4000 == 0 and it_loc > 0:
            width = max(1, n // 80)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)
        it_loc += 1
        if it_loc % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] local iter={it_loc}, best={best_score:.6f}")

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished in {time.monotonic() - start:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_seq]

Propose a step‑function (non‑negative list) that minimises
evaluate_sequence().

Algorithm (≈1000 s budget)
---------------------------
0. Initialise – use the previous best sequence if provided, otherwise a
   uniform vector of length n≈4000.

1. Global random exploration (≈25 % of the time)
   • Dirichlet draws on several coarse block granularities.
   • Power‑law shapes a_i ∝ (i+1)⁻ᵖ with p∈[‑0.5, 3.5].
   • Sparse support (≤ 500 non‑zero entries) with Dirichlet weights.
   • Random permutation of the current best.
   Keep the best candidate (lowest evaluation).

2. Simulated‑annealing on a block‑parameterisation (≈45 % of the time)
   • The current best is summarised into K blocks (K starts at 64 and
     is doubled after each SA stage).
   • At each SA step a random amount of mass is transferred between two
     blocks; the new candidate is projected onto the simplex.
   • Acceptance follows the Metropolis rule with an exponential cooling
     schedule.  The temperature is tied to the current objective value,
     allowing large exploratory jumps early on.

3. Full‑vector exponentiated‑gradient descent (≈20 % of the time)
   • Exact sub‑gradient of maxₖ(a∗a)[k] is obtained from the active
     convolution indices.
   • Mirror‑descent on the probability simplex (log‑space updates) with
     a diminishing step size.

4. Stochastic local refinement (the remaining time)
   • Targeted mass‑transfer moves guided by the gradient of the most‑
     violated convolution index, plus occasional jitter.

The routine prints timing/score information, clips the final vector to
the evaluator’s limits and returns it as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only works for 1‑D vectors.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical fix: force exact sum = 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / w.shape[0]
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a via FFT (returns length 2·len(a)‑1)."""
    a_f = np.fft.rfft(a, n=fft_len)
    c = np.fft.irfft(a_f * a_f, n=fft_len)
    return c[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of maxₖ (a∗a)[k] for the active set of indices.
    For each active index k, the contribution to ∂max/∂a_i is a_{k‑i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)            # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad

        blocks = np.empty(K, dtype=np.float64)

        # normalise to sum 1 (probability simplex)

        if s > 0.0:

        else:

        base = n // K
        extra = n % K

        for i, b in enumerate(blocks):
