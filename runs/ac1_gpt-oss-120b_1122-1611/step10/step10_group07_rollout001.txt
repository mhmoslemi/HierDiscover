"""
Propose a step‑function minimizing 2·n·max(conv(a,a))/sum(a)^2.

Algorithm (≈1000 s budget)
---------------------------------
0. Initialise from the previous best sequence if present, otherwise a uniform vector.
1. Random exploration (≈15 % of time) using Dirichlet bins and power‑law shapes.
2. Linear‑programming (LP) lift (≈30 % of time):
       maximise Σ g_j  s.t. (a⋆g)≤M   (M = current max‑convolution)
   Build the Toeplitz convolution matrix on the fly (sparse) and solve
   with SciPy's HiGHS LP solver.  Mix the LP‑solution g with the current a
   using several step‑sizes and keep the best mixture.
3. Exponentiated‑gradient descent on the active set (≈20 %).
4. Additive projected sub‑gradient descent (≈15 %).
5. Stochastic local refinement for the remaining budget.
All updates respect the simplex (non‑negative, Σa=1).  The best sequence
found is returned (clipped to the evaluator limits).
"""

import time
import numpy as np
from scipy import optimize, sparse

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σx = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a⋆a via FFT.  Returns length 2·len(a)‑1 result.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a⋆a)[k] for the active index set.
    For each active k the contribution to ∂max/∂a_i is a_{k‑i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _build_conv_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build the sparse matrix A such that (A @ g) = a⋆g .
    Shape: (2·n‑1, n) with diagonal offsets 0…n‑1.
    """
    n = a.size
    # Each diagonal offset d (0≤d<n) contains the constant a[d].
    diag_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sparse.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the objective value for a normalized (Σ=1) vector."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` and normalise to sum = 1
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        # a moderately large dimension – large enough to give a good bound,
        # but not too huge to make the LP intractable.
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    n = a0.size
    fft_len = _next_pow2(3 * n)          # safe length for linear convolution

    # --------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best = a0.copy()
    best_score = _score_from_conv(_conv_fft(best, fft_len), n)
    print(f"[{0.0:6.2f}s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    phase0_end = start_time + 0.15 * (deadline - start_time)
    it_rand = 0
    while time.monotonic() < phase0_end:

            # Dirichlet on a moderately coarse binning, sometimes monotone
            K = 128 if n >= 4096 else max(8, n // 32)

            bins = rng.dirichlet(np.full(K, alpha))
            if rng.random() < 0.5:
                bins = np.sort(bins)[::-1]           # enforce monotone decreasing
            seq = np.repeat(bins, block)[:n]
        else:
            # Pure power‑law shape, varying exponent
            p = rng.uniform(0.05, 3.0)
            idx = np.arange(1, n + 1, dtype=np.float64)
            seq = idx ** (-p)
            seq = seq / seq.sum()

        conv = _conv_fft(seq, fft_len)
        score = _score_from_conv(conv, n)
        if score < best_score:
            best_score = score
            best = seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (rand) = {best_score:.6f} (iter={it_rand})")
        it_rand += 1
        if it_rand % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] random iter={it_rand}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 1 – LP lift (≈30 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    phase1_end = start_time + 0.45 * (deadline - start_time)
    lp_iter = 0
    t_candidates = [0.1, 0.2, 0.4, 0.6, 0.8]   # mixing coefficients to try
    while time.monotonic() < phase1_end and lp_iter < 12:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        # Safety margin for numerical noise
        M = max_b * (1.0 + 5e-9)

        # Build the Toeplitz convolution matrix for the current a
        A = _build_conv_matrix(a)                     # (2n‑1) × n sparse
        b = np.full(2 * n - 1, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)             # maximise Σg ⇔ minimise –Σg
        try:
            res = optimize.linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True},
            )
        except Exception as exc:
            print(f"[LP] solver exception: {exc}")
            break

        if not res.success:
            print(f"[LP] solver failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            # No improvement possible – stop LP phase
            print("[LP] no sum‑increase, terminating LP phase")
            break

        # Try several mixings of the current a and the LP solution g
        improved = False
        for t in t_candidates:
            a_mix = (1.0 - t) * a + t * g
            a_mix /= a_mix.sum()
            conv_mix = _conv_fft(a_mix, fft_len)
            score_mix = _score_from_conv(conv_mix, n)
            if score_mix < best_score - 1e-12:
                best_score = score_mix
                best = a_mix.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (LP t={t:.2f}) = {best_score:.6f}")
                improved = True
            if score_mix <= 2.0 * n * max_b:          # non‑worsening candidate
                a = a_mix
        if not improved:
            # Even the best mixing did not improve the global best,
            # but still keep the non‑worsening a (if any)
            pass

        lp_iter += 1
        if lp_iter % 3 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] LP iter={lp_iter}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Phase 2 – exponentiated‑gradient descent (≈20 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    phase2_end = start_time + 0.65 * (deadline - start_time)
    it_exp = 0
    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-7
        active = np.where(conv >= max_b - eps)[0]
        # limit the active set for speed
        if active.size > 80:
            active = rng.choice(active, size=80, replace=False)

        grad = _gradient_from_active(a, active)
        eta = 0.6 / np.sqrt(it_exp + 1.0)
        log_a = np.log(a + 1e-15)
        log_a -= eta * grad
        a = np.exp(log_a)
        a /= a.sum()

        score = _score_from_conv(_conv_fft(a, fft_len), n)
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (exp‑grad) = {best_score:.6f} (iter={it_exp})")
            if best_score <= 1.5025:
                print("[early stop] target reached in exponentiated phase")
                break
        # accept non‑worsening move automatically (by construction)
        it_exp += 1
        if it_exp % 250 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] exp‑grad iter={it_exp}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Phase 3 – additive sub‑gradient descent (≈15 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    phase3_end = start_time + 0.80 * (deadline - start_time)
    it_sub = 0
    while time.monotonic() < phase3_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-6
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        grad = _gradient_from_active(a, active)
        step = 0.10 / np.sqrt(it_sub + 1.0)
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate)

        score_c = _score_from_conv(_conv_fft(a_candidate, fft_len), n)
        if score_c < best_score - 1e-12:
            best_score = score_c
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (sub‑grad) = {best_score:.6f} (iter={it_sub})")
            if best_score <= 1.5025:
                break
        if score_c <= 2.0 * n * max_b:
            a = a_candidate
        it_sub += 1
        if it_sub % 200 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] sub‑grad iter={it_sub}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7) Phase 4 – stochastic local refinement (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    it_loc = 0
    while time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len)
        idx_max = int(np.argmax(conv))
        grad_max = _gradient_from_active(a, np.array([idx_max]))
        src = int(np.argmax(grad_max))

        # pick a low‑gradient target
        low = np.where(grad_max <= grad_max.min() + 1e-12)[0]
        if low.size > 1 and rng.random() < 0.75:
            tgt = int(rng.choice(low))
        else:
            tgt = int(rng.integers(0, n))

        if src == tgt:
            tgt = int(rng.integers(0, n))

        # tiny directed mass transfer
        frac = np.exp(rng.uniform(np.log(0.0005), np.log(0.02)))   # ≈0.05 %–2 %
        delta = a[src] * frac
        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta
        a_cand = _project_onto_simplex(a_cand)

        score_c = _score_from_conv(_conv_fft(a_cand, fft_len), n)
        if score_c < best_score - 1e-12:
            best_score = score_c
            best = a_cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (move={it_loc})")
            if best_score <= 1.5025:
                break
        if score_c <= 2.0 * n * np.max(conv):
            a = a_cand

        # occasional jitter to escape shallow plateaus
        if it_loc % 4000 == 0 and it_loc > 0:
            width = max(1, n // 80)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)

        it_loc += 1
        if it_loc % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local iter={it_loc}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 8) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
