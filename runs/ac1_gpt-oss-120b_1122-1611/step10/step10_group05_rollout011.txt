def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    # Determine n
    target_n = 512
    # initialize a

        if a.size < target_n:
            a = np.pad(a, (0, target_n - a.size), mode='constant')

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm works in two phases inside the allotted time budget.

Phase 1 – Linear‑programming boost:
    * Start from the best known construction (``height_sequence_1``) if it
      exists, otherwise from a uniform distribution.
    * Down‑sample / truncate the seed to a moderate size (≈ 500) so that
      a dense convolution matrix (2 n − 1 × n) fits comfortably in memory.
    * Repeatedly solve the LP
          maximise   Σ g_i
          subject to  (f * g)[k] ≤ M   for all k,
                     g_i ≥ 0,
      where M = max(f*f) and ``*`` denotes the linear convolution.
      The LP finds a vector ``g`` that can carry more total mass than ``f``
      without increasing any convolution entry.
    * Rescale ``g`` to unit ℓ₁‑norm and either replace ``f`` with ``g`` (if
      its own max‑convolution is smaller) or blend the two vectors.
    * Update ``f`` and repeat while the LP can still increase the ℓ₁‑mass.
      Each successful step provably reduces the objective value.

Phase 2 – Fast simulated‑annealing fine‑tuning:
    * With the (now improved) vector as a starting point run a cheap
      simulated‑annealing search that moves a small fraction of mass from
      a “source’’ index (largest contribution to the current peaks) to a
      “target’’ index (smallest contribution).
    * Accept every improving move and occasionally accept worsening
      moves with a temperature that falls linearly to zero.
    * Add occasional jitter on a random block and rare full shuffles to
      escape local minima.

Both phases keep the sequence on the probability simplex (∑a_i = 1);
the final sequence is clipped to the limits imposed by ``evaluate_sequence``
and its score is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _downsample_sequence(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Compress ``seq`` to exactly ``target_len`` entries by block‑summing."""
    if seq.size == target_len:
        return seq.copy()
    if seq.size > target_len:
        # block size = ceil(len/target_len)
        blk = int(math.ceil(seq.size / target_len))
        pad = blk * target_len - seq.size
        seq_pad = np.pad(seq, (0, pad), mode="constant")
        return seq_pad.reshape(target_len, blk).sum(axis=1)
    # seq.size < target_len  → pad with zeros at the end
    return np.pad(seq, (0, target_len - seq.size), mode="constant")


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Return the dense matrix A such that A @ g = np.convolve(a, g)."""
    n = a.size
    first_col = np.r_[a, np.zeros(n - 1)]
    first_row = np.r_[a[0], np.zeros(n - 1)]
    return toeplitz(first_col, first_row)  # shape (2*n-1, n)


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a*a."""
    return np.convolve(a, a).max()


def _objective_from_a(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (no clipping needed)."""
    n = a.size
    max_b = _max_conv(a)
    sum_a = a.sum()
    if sum_a < 0.01:  # safeguard – mirrors evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a seed sequence (prefer the previously found one)
    # --------------------------------------------------------------
    TARGET_LEN = 512                     # moderate size – fast LP + SA
    if "height_sequence_1" in globals():
        raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = _downsample_sequence(raw, TARGET_LEN)
    else:

    # Normalise to the probability simplex and protect against pathologies
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full_like(a, 1.0 / a.size)

    n = a.size
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"[0.00s] initial score = {best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – Linear‑programming improvements
    # --------------------------------------------------------------
    lp_iter = 0
    while time.monotonic() < deadline:
        # current bound M
        M = _max_conv(a)

        # Build convolution matrix A (dense, but small enough for TARGET_LEN)
        A = _build_conv_matrix(a)                 # shape (2*n-1, n)
        b_ub = np.full(A.shape[0], M)

        # Linear program: maximise Σ g_i  ⇔ minimise -Σ g_i
        c = -np.ones(n)                           # objective coefficients
        bounds = [(0, None)] * n

        time_left = deadline - time.monotonic()
        if time_left <= 0:
            break

        # Solve with HiGHS – fast for moderate sizes
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_left},
        )
        if not res.success:
            print(f"[LP {lp_iter}] linprog failed (status={res.status})")
            break

        g0 = res.x
        sum_g0 = g0.sum()

        # If LP cannot increase total mass, we are locally optimal
        if sum_g0 <= 1.0 + 1e-9:
            print(f"[LP {lp_iter}] no mass increase (sum={sum_g0:.6f}) – stop LP")

        # Direct replacement if it already improves the bound

            M = M_g

                best_score = cur_score
                elapsed = time.monotonic() - start_time

            for t in (0.5, 0.25, 0.1, 0.05, 0.01):

                        best_score = cur_score

    # --------------------------------------------------------------
    cur = a.copy()
    cur_conv = np.convolve(cur, cur)

    # temperature schedule (linear decay)
    T0 = best_score * 0.05 if best_score > 0 else 0.01
    Tf = 1e-8 * (best_score if best_score > 0 else 1.0)

    while time.monotonic() < deadline:
        # Determine active convolution indices (largest contributions)
        if top_k >= cur_conv.size:
            active_idx = np.arange(cur_conv.size)
        else:
            active_idx = np.argpartition(-cur_conv, top_k)[:top_k]

        # Sub‑gradient of the max‑convolution (restricted to active set)
        grad = np.zeros_like(cur)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n, k + 1)          # exclusive
            i = np.arange(i_start, i_end, dtype=int)
            grad[i] += cur[k - i]

        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))

        # Transfer a random fraction of the source mass to the target
        frac = math.exp(rng.uniform(math.log(0.001), math.log(0.1)))
        delta = cur[src] * frac
        cand = cur.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # Normalise back to the simplex
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand /= cand_sum

        # Fast objective (identical to evaluate_sequence for valid inputs)
        cand_conv = np.convolve(cand, cand)
        cand_score = 2.0 * n * cand_conv.max() / (cand_sum ** 2)

        elapsed = time.monotonic() - start_time
        frac_time = elapsed / (deadline - start_time)
        temp = T0 * (1.0 - frac_time) + Tf * frac_time

        accept = False
        if cand_score < cur_score - 1e-15:
            accept = True
        elif temp > 0.0:
            prob = math.exp(-(cand_score - cur_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_conv = cand_conv
            cur_score = cand_score

        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] SA new best = {best_score:.6f}")

        if it_sa % 3000 == 0 and it_sa > 0:

            block_len = int(rng.integers(1, max(2, n // 60)))

            jitter = cur.copy()
            jitter[start_idx:start_idx + block_len] *= factor
            jitter = np.clip(jitter, 0.0, None)
            jitter /= jitter.sum()
            jitter_conv = np.convolve(jitter, jitter)
            jitter_score = 2.0 * n * jitter_conv.max() / (jitter.sum() ** 2)

                best_score = jitter_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] jitter new best = {best_score:.6f}")

                cur_conv = jitter_conv
                cur_score = jitter_score

        if rng.random() < 0.0005:

    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
