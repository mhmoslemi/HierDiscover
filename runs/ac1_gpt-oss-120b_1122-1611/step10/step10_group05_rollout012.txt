"""
Propose a step‑function that minimises ``evaluate_sequence``.

Approach
--------
* Work on the scale‑invariant formulation (sum of the coefficients = 1).
* The objective becomes ``score = 2 * n * max(conv)`` where
  ``conv = a * a`` (linear convolution).
* Use a *projected gradient descent* on a smooth surrogate of the max:
      w_k ∝ exp(α·(conv_k – max(conv)))          (soft‑max weighting)
  The gradient of the weighted sum can be obtained by a single FFT
  convolution:  grad_i = Σ_k w_k · a_{k‑i}.
* The step size and the smoothing parameter α are annealed:
  – early iterations use a small α (many entries influence the direction);
  – later iterations increase α so the descent focuses on the true peak.
* After each gradient step the vector is projected onto the probability
  simplex (non‑negative, sum = 1).  The projection is Euclidean and uses
  the Duchi‑et‑al. O(n log n) algorithm.
* To avoid being trapped in very flat regions we occasionally apply a
  small random *jitter* on a short block and a tiny probability of a full
  permutation.  These moves are also projected onto the simplex.
* The best vector seen so far is kept and returned after the time budget.

The whole loop runs for ~1000 s (5 s safety margin), performing only
FFT‑based operations (O(n log n) per iteration).  With n≈4096 this allows
tens of thousands of iterations and usually beats the target bound
≈ 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns ``np.convolve(a, a)`` (length 2*len(a)-1).
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft_general(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two (possibly different‑length) vectors using
    an FFT of length ``fft_len``.  Returns the first len(x)+len(y)-1 entries.
    """
    xf = np.fft.rfft(x, n=fft_len)
    yf = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(xf * yf, n=fft_len)
    return conv[: x.size + y.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w


def _weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i}  (indices outside [0,n) give 0).
    This is the convolution of ``w`` with the reversed ``a``.
    """
    a_rev = a[::-1]
    g_full = _conv_fft_general(w, a_rev, fft_len)   # length = (2n-1)+n-1 = 3n-2
    start = a.size - 1           # index of the first useful entry
    return g_full[start:start + a.size]   # length n


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.

    The algorithm combines a projected gradient descent on a soft‑max
    surrogate of the max‑convolution entry with occasional random
    jitter/shuffle moves.  The best vector found before the deadline
    (projected to the evaluator limits) is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` – use previous best if available, otherwise uniform.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
    else:
        n = 4096                               # power of two – convenient for FFT
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(3 * n)               # enough for all convolutions

    # --------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    cur_conv = _conv_fft(a, fft_len)
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max

    best = a.copy()
    best_score = cur_score
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Optimisation parameters
    # --------------------------------------------------------------
    iteration = 0
    momentum = np.zeros_like(a)

    # annealing parameters for the soft‑max weight
    alpha_start = 1.0
    alpha_end = 80.0          # large enough to focus on the true peak

    # learning‑rate schedule (relative to the current magnitude of the gradient)
    lr_start = 0.2
    lr_end = 0.005

    # temperature for occasional acceptance of worsening moves
    T0 = best_score * 0.05
    Tf = 1e-8 * best_score

    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # a) Current convolution and score
        # ----------------------------------------------------------
        cur_conv = _conv_fft(a, fft_len)
        cur_max = cur_conv.max()
        cur_score = 2.0 * n * cur_max

        # keep global best
        if cur_score < best_score - 1e-12:
            best = a.copy()
            best_score = cur_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ----------------------------------------------------------
        # b) Build the soft‑max weights (smooth approximation of max)
        # ----------------------------------------------------------
        conv_min = cur_conv.min()
        diff = cur_max - conv_min
        if diff <= 0:
            scale = np.zeros_like(cur_conv)
        else:
            scale = (cur_conv - cur_max) / diff    # ≤ 0, =0 at the peak

        elapsed = time.monotonic() - start_time
        time_frac = (elapsed) / (deadline - start_time)   # in [0,1]
        alpha = alpha_start * (1.0 - time_frac) + alpha_end * time_frac

        w = np.exp(alpha * scale)        # w_max = 1, others < 1
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                     # normalise – now a probability vector

        # ----------------------------------------------------------
        # c) Gradient of the weighted sum (no explicit factor 2 needed)
        # ----------------------------------------------------------
        grad = _weighted_gradient(a, w, fft_len)

        # momentum (Adam‑style without bias correction)
        beta = 0.9
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----------------------------------------------------------
        # d) Step size (linear cooling)
        # ----------------------------------------------------------
        lr = lr_start * (1.0 - time_frac) + lr_end * time_frac
        # scale step roughly to the average magnitude of a
        step = lr * a.mean() / (np.mean(np.abs(momentum)) + 1e-12)

        cand = a - step * momentum
        cand = _project_onto_simplex(cand)

        # ----------------------------------------------------------
        # e) Acceptance test (simulated‑annealing style)
        # ----------------------------------------------------------
        cand_conv = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * cand_conv.max()

        if cand_score < cur_score - 1e-15:
            accept = True
        else:
            # accept worse move with a temperature that decays linearly
            temp = T0 * (1.0 - time_frac) + Tf * time_frac
            if temp > 0.0:
                prob = np.exp(-(cand_score - cur_score) / temp)
                accept = rng.random() < prob
            else:
                accept = False

        if accept:
            a = cand

        # ----------------------------------------------------------
        # f) Occasional diversification moves
        # ----------------------------------------------------------
        # 1) Local jitter (every few thousand iterations)
        if iteration % 3000 == 0 and iteration > 0:
            block_len = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - block_len + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + block_len] *= factor
            a_jit = _project_onto_simplex(a_jit)
            jit_score = 2.0 * n * _conv_fft(a_jit, fft_len).max()
            if jit_score < best_score - 1e-12:
                best, best_score = a_jit.copy(), jit_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (jitter) = {best_score:.6f}")
            if jit_score < cur_score + 1e-8:
                a = a_jit

        # 2) Global shuffle with a tiny probability (exploratory)
        if rng.random() < 0.0005:
            perm = rng.permutation(n)
            a_perm = a[perm]
            perm_score = 2.0 * n * _conv_fft(a_perm, fft_len).max()
            if perm_score < best_score - 1e-12:
                best, best_score = a_perm.copy(), perm_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (shuffle) = {best_score:.6f}")
            if perm_score < cur_score:
                a = a_perm

        iteration += 1
        # early exit if we already beat the target comfortably
        if best_score <= 1.5025:
            print("[early stop] target reached")
            break

    # --------------------------------------------------------------
    # 4) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # evaluator safety clamp
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
