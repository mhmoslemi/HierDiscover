def candidate_scores(src,tgt,a,conv):

    fractions = [0.5, 0.2, 0.05, 0.01, 0.001]  # relative to a[src]

    best_local_score = inf; best_delta = None; best_a_new = None; best_conv_new = None

    for f in fractions:

        delta = a[src] * f

        if delta <= 0: continue

        # compute new a
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        # compute new conv via incremental update
        conv_new = conv.copy()

        # apply shift contributions
        # we need to apply 2*delta*(shifted_tgt - shifted_src)
        # we can add segment contributions via slices
        conv_new[tgt:tgt+n] += 2*delta*a

        conv_new[2*tgt] += delta*delta

"""
Step‑function optimizer.

The algorithm starts from the best known sequence (if the global variable
``height_sequence_1`` exists) or from a uniform distribution.
It keeps the vector on the probability simplex (∑a_i = 1) and repeatedly
performs a *mass‑transfer* move: a small amount of weight is moved from the
index that contributes most to the current maximal auto‑correlation towards the
index that contributes least.  The effect of a move on the convolution is
updated **incrementally** (no FFT needed) which makes a single iteration
extremely cheap and allows millions of iterations within the 1000 s budget.
A simple simulated‑annealing schedule accepts occasional worsening moves.
Every few thousand iterations a random jitter is applied and very rarely a
global permutation is tried.  The best sequence seen so far is returned
(and finally evaluated with ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] for the set ``active`` of indices attaining
    the current maximum (or a selected top‑k set).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        idx = np.arange(i_start, i_end, dtype=int)
        grad[idx] += a[k - idx]
    return grad


def _apply_mass_move(conv: np.ndarray,
                     a: np.ndarray,
                     src: int,
                     tgt: int,
                     delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving ``delta`` mass from ``src`` to
    ``tgt`` (a[src] -= delta, a[tgt] += delta).  The update is done
    analytically without FFT.
    """
    n = a.size
    # make a copy to avoid mutating the original
    new_conv = conv.copy()

    # linear part: 2*delta*(shift(tgt) - shift(src))
    new_conv[tgt:tgt + n] += 2.0 * delta * a
    new_conv[src:src + n] -= 2.0 * delta * a

    # quadratic part: delta^2 * (δ_{2*tgt} + δ_{2*src} - 2*δ_{src+tgt})
    idx_2src = 2 * src
    idx_2tgt = 2 * tgt
    idx_sum = src + tgt
    new_conv[idx_2src] += delta * delta
    new_conv[idx_2tgt] += delta * delta
    new_conv[idx_sum] -= 2.0 * delta * delta
    return new_conv


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective = 2 * n * max(conv) (the sum of the vector is assumed = 1)."""
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------

        grad = _gradient_from_active(a, active_idx)

        src = int(np.argmax(grad))       # source of mass (largest contribution)
        tgt = int(np.argmin(grad))       # target of mass (smallest contribution)

        if src == tgt:
            # fallback: random distinct pair
            src, tgt = rng.choice(n, size=2, replace=False)

        # ---- propose a move ------------------------------------------------
        # draw a log‑uniform fraction in [1e‑4, 0.2] of the source weight
        log_min, log_max = np.log(1e-4), np.log(0.2)
        frac = np.exp(rng.uniform(log_min, log_max))
        delta = a[src] * frac
        # guard against numerical round‑off
        if delta <= 0.0:
            delta = 0.0
        elif delta > a[src]:
            delta = a[src]

        # compute the new convolution analytically
        cand_conv = _apply_mass_move(conv, a, src, tgt, delta)
        cand_score = _objective_from_conv(cand_conv, n)

        # ---- acceptance (simulated annealing) ----------------------------
        elapsed = time.monotonic() - start_time
        time_frac = (elapsed) / (deadline - start_time)   # ∈[0,1]
        temp = T0 * (1.0 - time_frac) + Tf * time_frac

        accept = False
        if cand_score < cur_score - 1e-15:
            accept = True
        else:
            if temp > 0.0:
                prob = np.exp(-(cand_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            # update the state (exact arithmetic thanks to the incremental formula)
            a[src] -= delta
            a[tgt] += delta
            conv = cand_conv
            cur_score = cand_score

        # ---- keep the best found -----------------------------------------
        if cand_score < best_score - 1e-12:
            best = a.copy()
            best_score = cand_score
            print(f"[{elapsed:6.2f}s] new best (anneal) = {best_score:.6f}")
            if best_score <= target_score:
                print("[early stop] target reached")
                break

        # ---- occasional jitter (helps escape shallow minima) -------------
        if iteration % 5000 == 0 and iteration > 0:
            block_len = int(rng.integers(1, max(2, n // 50)))
            start_idx = int(rng.integers(0, n - block_len + 1))
            factor = np.exp(rng.normal(0.0, 0.025))
            a_j = a.copy()
            a_j[start_idx:start_idx + block_len] *= factor
            a_j = np.clip(a_j, 0.0, None)
            a_j /= a_j.sum()

            # recompute convolution for the jittered vector (FFT is cheap here)
            a_j_fft = np.fft.rfft(a_j, n=fft_len)
            conv_j = np.fft.irfft(a_j_fft * a_j_fft, n=fft_len)[:conv_len]
            conv_j = np.maximum(conv_j, 0.0)
            score_j = _objective_from_conv(conv_j, n)

            if score_j < best_score - 1e-12:
                best = a_j.copy()
                best_score = score_j
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (jitter) = {best_score:.6f}")

            if score_j < cur_score:
                a = a_j
                conv = conv_j
                cur_score = score_j

        # ---- very rare global shuffle (exploratory) -----------------------
        if rng.random() < 0.0003:
            perm = rng.permutation(n)
            a_perm = a[perm]
            a_perm_fft = np.fft.rfft(a_perm, n=fft_len)
            conv_perm = np.fft.irfft(a_perm_fft * a_perm_fft, n=fft_len)[:conv_len]
            conv_perm = np.maximum(conv_perm, 0.0)
            score_perm = _objective_from_conv(conv_perm, n)

            if score_perm < best_score - 1e-12:
                best = a_perm.copy()
                best_score = score_perm
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (shuffle) = {best_score:.6f}")

            if score_perm < cur_score:
                a = a_perm
                conv = conv_perm
                cur_score = score_perm

        # ---- periodic re‑normalisation & full recompute (drift control) ---
        if iteration % 20000 == 0 and iteration > 0:
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            a_fft = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:conv_len]
            conv = np.maximum(conv, 0.0)
            cur_score = _objective_from_conv(conv, n)

        iteration += 1

    # --------------------------------------------------------------
    # 5) Final evaluation and return
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # obey evaluator limits
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start_time
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
