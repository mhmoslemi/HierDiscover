"""
Improved optimiser for the max‑self‑convolution problem.

The algorithm mixes three ideas:
1.  A small LP‑based “mass‑pushing’’ step that moves probability mass towards
    the edges while keeping all convolution entries ≤ the current maximum.
    This dramatically flattens the autocorrelation.
2.  Exponentiated‑gradient (mirror descent) on a smooth approximation of the
    max‑convolution (soft‑max) with momentum (Adam‑style) to polish the shape.
3.  Simple stochastic mass‑transfer moves (local refinement) with an acceptance
    rule that keeps any improvement.

All steps stay on the probability simplex (∑a_i = 1, a_i ≥ 0) so the objective
2·n·max(conv) is exactly the value returned by ``evaluate_sequence``.
The routine respects a 1000 s budget, returns the best sequence seen and
prints progress information.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.full_like(v, 1.0 / len(v))
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _smooth_max_gradient(a: np.ndarray, fft_len: int, beta: float) -> np.ndarray:
    """
    Gradient of the smooth (log‑sum‑exp) approximation of max(conv(a,a)).
    Returns ∂/∂a [ softmax_beta(conv) ].
    """
    n = a.size
    conv = _conv_fft(a, fft_len)               # length 2n‑1
    max_conv = conv.max()
    # soft‑max weights
    w = np.exp(beta * (conv - max_conv))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)

    # term2 = (a * w)[:n]
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: n + len(w) - 1]
    term2 = conv_aw[:n]

    # term1 = (a * reverse(w))[m‑1‑k]  (m = len(w))
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: n + len(w_rev) - 1]
    m = len(w)
    term1 = conv_aw_rev[m - 1 - np.arange(n)]

    grad = term1 + term2
    return grad


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the active set.
    For each active index k, the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _hard_max_step(a: np.ndarray,
                   fft_len: int,
                   step_factor: float,
                   iteration: int) -> np.ndarray:
    """Projected sub‑gradient step for the true max objective."""
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    step = step_factor / np.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


def _local_refinement_step(a: np.ndarray,
                           rng: np.random.Generator,
                           src: int,
                           tgt: int) -> tuple[np.ndarray, int, int]:
    """Tiny directed mass‑transfer move."""
    n = a.size
    if a[src] <= 0.0:
        src = int(rng.integers(0, n))
    if src == tgt:
        tgt = int(rng.integers(0, n))

    # fraction ∈ [0.001,0.05] (log‑uniform)
    frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    return a_new, src, tgt


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  Returns the best sequence found within the given
    time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) initialise a – use previous construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # start with a modest size – larger sizes give more freedom but cost more
        a = np.full(1024, 1.0 / 1024, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(3 * n)   # safe length for all linear convolutions

    # ------------------------------------------------------------------
    # 1) time‑budget handling
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # ------------------------------------------------------------------
    # 2) bookkeeping of the best solution
    # ------------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()
    best = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – LP‑based edge‑pushing (few iterations, heavy work)
    # ------------------------------------------------------------------
    # pre‑compute the index pattern for the convolution constraints
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()          # length n*n
    cols = np.tile(np.arange(n), n)                                 # length n*n
    max_lp_steps = 12
    t_mix = 0.15           # how strongly the new direction is mixed in

    for lp_it in range(max_lp_steps):
        if time.monotonic() >= deadline:
            break
        # current max‑convolution value
        conv = _conv_fft(a, fft_len)
        M = conv.max()

        # build the linear‑program matrix with the *current* a as coefficients
        data = np.repeat(a, n)                # a_i appears n times, column‑wise
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # LP: maximise sum(b)  s.t.  (a * b)[k] ≤ M  for all k,  b ≥ 0
        c = -np.ones(n)                        # minimise –sum(b)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'disp': False})

        if not res.success:
            print(f"[LP] iteration {lp_it} failed: {res.message}")
            break

        b = np.maximum(res.x, 0.0)            # numerical safety
        s_b = b.sum()
        if s_b == 0.0:
            g = a.copy()
        else:
            g = b / s_b                       # normalise to sum‑1

        # convex combination with current iterate
        a = (1.0 - t_mix) * a + t_mix * g
        a = _project_onto_simplex(a)          # keep numerical errors in check

        # evaluate improvement
        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best after LP step {lp_it} = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – smoothed‑max mirror descent with momentum
    # ------------------------------------------------------------------
    # parameters for the smooth‑max stage
    beta_start, beta_end = 40.0, 800.0
    eta0 = 0.6
    momentum = 0.9
    v = np.zeros_like(a)                     # momentum buffer
    it_smooth = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        # linear schedule for the smoothing factor β
        frac = min(1.0, (elapsed) / (deadline - start))
        beta = beta_start + frac * (beta_end - beta_start)
        eta = eta0 / np.sqrt(it_smooth + 1.0)

        # gradient of the soft‑max of the convolution
        grad = _smooth_max_gradient(a, fft_len, beta)
        # momentum update (Nesterov style)
        v = momentum * v + (1.0 - momentum) * grad
        # exponentiated‑gradient step on log‑probabilities
        log_a = np.log(a + 1e-15) - eta * v
        a = np.exp(log_a)
        a /= a.sum()

        # scoring
        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (smooth) = {best_score:.6f}")

        it_smooth += 1
        if it_smooth % 500 == 0:
            # periodic checkpoint – give a chance to exit cleanly
            if time.monotonic() >= deadline:
                break

    # ------------------------------------------------------------------
    # 5) Phase 3 – stochastic local refinement (directed mass transfer)
    # ------------------------------------------------------------------
    a = best.copy()
    src = int(rng.integers(0, n))
    tgt = int(rng.integers(0, n))
    it_local = 0
    while time.monotonic() < deadline:
        if it_local % 30 == 0:
            # re‑evaluate the active set so that we move mass from “high‑gradient”
            conv = _conv_fft(a, fft_len)
            max_b = conv.max()
            active = np.where(conv >= max_b - max_b * 1e-6)[0]
            grad_active = _gradient_from_active(a, active)
            src = int(np.argmax(grad_active))
            # pick a low‑gradient target (bottom 15 % quantile)
            thresh = np.percentile(grad_active, 15)
            low = np.where(grad_active <= thresh)[0]
            if low.size:
                tgt = int(rng.choice(low))
            else:
                tgt = int(rng.integers(0, n))

        a_candidate, src, tgt = _local_refinement_step(a, rng, src, tgt)
        conv_cand = _conv_fft(a_candidate, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_candidate.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f}")

        # accept any move that does not increase the max‑convolution
        conv_cur = _conv_fft(a, fft_len)
        if score_cand <= 2.0 * n * conv_cur.max():
            a = a_candidate

        it_local += 1

        # occasional jitter to escape tiny plateaus
        if it_local % 4000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)

        if it_local % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] local iter={it_local}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
