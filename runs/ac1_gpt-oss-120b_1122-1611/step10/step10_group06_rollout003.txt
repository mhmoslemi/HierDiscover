"""
Propose a step‑function (list of non‑negative heights) that minimises
evaluate_sequence().  The method consists of two complementary optimisation
phases:

1️⃣  A smoothed exponentiated‑gradient descent (soft‑max surrogate of the
    max‑convolution).  A temperature parameter β is increased slowly so that
    the surrogate converges to the true max.  The update is multiplicative
    (mirror‑descent on the probability simplex) and is performed with a
    decreasing step‑size η.

2️⃣  A deterministic pairwise mass‑transfer stage.  For the current
    convolution the index giving the maximal entry is identified; the
    gradient of that entry tells which source position contributes most to it.
    The algorithm moves mass from the *most* contributing source to the
    *least* contributing target.  The amount of mass is chosen analytically:
    moving a fraction δ changes every convolution entry linearly,
    ``conv_i → conv_i + δ·d_i`` where ``d_i = contrib_tgt[i]−contrib_src[i]``.
    The optimal δ that minimises the new maximum can be computed in closed
    form (intersection of lines).  The move is then applied and the new
    score is kept if it does not increase the objective.

Both phases keep the distribution on the simplex (non‑negative and unit
sum) and use FFT‑based convolutions for O(n log n) speed.  The search runs
until a 3 s safety margin before the supplied budget expires; the best
sequence seen is returned (clipped to the evaluator limits).  An early exit
occurs as soon as the target value 1.5025 is reached.

The routine works for any length that fits into memory; a power‑of‑two length
(8192) is used by default because it gives a tighter theoretical lower bound
while still being cheap to FFT.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (result length 2*len(a)-1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact gradient of the convolution entry (a*a)[idx] w.r.t. a.
    Entry k receives a contribution a[idx‑k] whenever the index is
    in range; zero outside.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)          # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    return grad


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _pairwise_opt_move(a: np.ndarray,
                       conv: np.ndarray,
                       src: int,
                       tgt: int,
                       max_idx: int) -> tuple[np.ndarray, float]:
    """
    Perform a deterministic mass transfer from *src* to *tgt*.
    The step size δ is chosen analytically to minimise the new maximum
    of the self‑convolution after the move.
    Returns the candidate vector and the corresponding score (2·n·max(conv)).
    """
    n = a.size
    # contribution vectors for a source / target (length 2n‑1)
    src_contrib = np.zeros(2 * n - 1, dtype=np.float64)
    tgt_contrib = np.zeros(2 * n - 1, dtype=np.float64)
    src_contrib[src:src + n] = a
    tgt_contrib[tgt:tgt + n] = a

    d = tgt_contrib - src_contrib                 # linear change per unit δ
    d_max = d[max_idx]                            # slope of the active entry
    conv_max = conv[max_idx]

    # Identify indices whose line will overtake the current maximum
    mask = d > d_max + 1e-18                       # strict > to avoid div‑by‑0
    if np.any(mask):
        numer = conv_max - conv[mask]               # ≥ 0
        denom = d[mask] - d_max                     # > 0
        # smallest positive intersection gives the optimal δ
        delta_star = np.min(numer / denom)
    else:
        # No crossing line → we can move the whole remaining source mass
        delta_star = a[src]

    # Clip to the amount actually available at the source
    delta = min(delta_star * 0.9999, a[src])        # a tiny safety factor
    if delta <= 0.0:
        # fallback to a tiny move to keep the algorithm alive
        delta = a[src] * 1e-6

    cand = a.copy()
    cand[src] -= delta
    cand[tgt] += delta
    # Numerical drift may break the simplex; re‑normalise (should be tiny)
    cand = _simplex_normalise(cand)

    # Exact new convolution and score
    fft_len = _next_pow2(3 * n)
    conv_cand = _conv_self(cand, fft_len)
    score = float(2.0 * n * conv_cand.max())
    return cand, score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.

    The algorithm consists of a smoothed exponentiated‑gradient phase followed
    by an analytically‑optimal pairwise mass‑transfer phase.  The best vector
    encountered is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a probability vector.
    #    Prefer the provided construction (height_sequence_1); otherwise start
    #    from a uniform distribution of length 8192 (power of two → fast FFT).
    # --------------------------------------------------------------
    target_len = 8192
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.size < target_len:
            a = np.concatenate([base, np.zeros(target_len - base.size, dtype=np.float64)])
        else:
            a = base[:target_len].copy()
    else:
        a = np.full(target_len, 1.0 / target_len, dtype=np.float64)

    a = _simplex_normalise(a)
    n = a.size
    fft_len = _next_pow2(3 * n)        # sufficient for all convolutions we need

    # --------------------------------------------------------------
    # Helper that converts a convolution array to the evaluation score.
    # (Because sum(a)=1, evaluate_sequence returns 2·n·max(conv).)
    # --------------------------------------------------------------
    def _score_from_conv(conv_arr: np.ndarray) -> float:
        return float(2.0 * n * conv_arr.max())

    # --------------------------------------------------------------
    # Book‑keeping for the best solution seen so far.
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 3.0)   # 3 s safety margin

    conv = _conv_self(a, fft_len)
    best_a = a.copy()
    best_score = _score_from_conv(conv)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Smoothed exponentiated‑gradient descent.
    # --------------------------------------------------------------
    phase1_end = start + 0.45 * (deadline - start)   # ~45 % of the budget
    it = 0
    beta = 5.0          # initial smoothing temperature (soft‑max sharpness)
    while time.monotonic() < phase1_end:
        # FFT of the current distribution (re‑used for both convolutions)
        a_fft = np.fft.rfft(a, n=fft_len)

        # Exact self‑convolution
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # Soft‑max over the convolution entries
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))          # stability by subtracting max
        w = exp_vals / exp_vals.sum()                     # probability vector, len 2n‑1

        # Gradient of the smoothed objective (w·conv) w.r.t a
        w_fft = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: n + w.size - 1]
        grad = grad_full[:n]

        # Multiplicative (exponentiated) update on the simplex
        eta = 0.6 / math.sqrt(it + 1.0)                   # decreasing step‑size
        a = a * np.exp(-eta * grad)
        a = _simplex_normalise(a)

        # Evaluate current score
        conv = _conv_self(a, fft_len)
        score = _score_from_conv(conv)

        # Keep the best solution
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] EG improvement → {best_score:.6f} (it={it})")
            if best_score <= 1.5025:
                print("[early stop] target reached during EG phase")
                return [float(x) for x in best_a]

        # Gradually sharpen the soft‑max (increase β)
        beta = min(beta * 1.002, 800.0)

        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] EG iter={it}, β={beta:.1f}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Deterministic optimal pairwise mass transfers.
    # --------------------------------------------------------------
    a = best_a.copy()
    phase2_end = deadline
    it2 = 0
    while time.monotonic() < phase2_end:
        conv = _conv_self(a, fft_len)
        max_idx = int(np.argmax(conv))

        # Gradient of the active (maximal) convolution entry
        grad_max = _gradient_for_index(a, max_idx)

        # Choose source (largest gradient) and target (smallest gradient)
        src = int(np.argmax(grad_max))
        tgt = int(np.argmin(grad_max))
        if src == tgt:
            # Degenerate case – pick a random partner
            tgt = (src + 1) % n

        # Deterministic optimal move
        cand, cand_score = _pairwise_opt_move(a, conv, src, tgt, max_idx)

        # Update bookkeeping
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] pairwise improvement → {best_score:.6f} (it={it2})")
            if best_score <= 1.5025:
                print("[early stop] target reached during pairwise phase")
                return [float(x) for x in best_a]

        # Accept non‑worsening moves to keep a downhill walk
        curr_score = _score_from_conv(conv)
        if cand_score <= curr_score:
            a = cand
        else:
            # Occasionally inject a tiny random jitter to escape plateaux
            if it2 % 1500 == 0:
                jitter = rng.normal(scale=0.0015, size=n)
                a = _simplex_normalise(np.clip(a + jitter, 0.0, None))

        it2 += 1
        if it2 % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] pairwise iter={it2}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits for safety)
    # --------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
