"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(a * a) / (sum(a)**2)

The algorithm is a stochastic multi‑restart Adam optimisation on the
soft‑max approximation of the max‑convolution.  For each restart we

* initialise a probability vector on the simplex (either a jittered copy of
  the current global best or a Dirichlet sample),
* iteratively minimise  f(a)=log‑sum‑exp(α·(a∗a)) by computing the exact
  gradient via FFTs,
* use Adam with Euclidean projection onto the simplex,
* slowly increase the soft‑max temperature α so that the optimisation first
  sees a smooth landscape and then focuses on the true ℓ∞‑norm,
* occasionally add small random jitter to escape plateaus.

Multiple restarts of different lengths are performed until the time budget
(1000 s, 5 s safety margin) is exhausted.  The best sequence found is
returned and stored in the global variable ``height_sequence_1`` for possible
future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the first (2*len(a)-1) entries.
    """
    fft_a = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights approximating the max of ``conv``.
    ``alpha`` controls the sharpness (large → closer to true max).
    """
    max_val = conv.max()
    shifted = conv - max_val          # for numerical stability
    # exponentiation; underflow yields 0 which is fine
    exp_vals = np.exp(alpha * shifted, dtype=np.float64)
    s = exp_vals.sum()
    if s == 0.0:
        # All values under‑flowed – fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  g(a)=∑_k w_k (a*a)[k]  w.r.t. a.
    Uses the convolution identity
        ∂g/∂a_i = 2 * ∑_k w_k a_{k-i}
    which we compute via FFT.
    """
    n = a.size
    # reverse the weight vector
    w_rev = w[::-1]
    # FFT of a and of the reversed weight vector
    fft_a = np.fft.rfft(a, n=fft_len)
    fft_w = np.fft.rfft(w_rev, n=fft_len)
    # convolution w_rev * a
    conv_rev = np.fft.irfft(fft_a * fft_w, n=fft_len)
    offset = w.size - 1               # = 2*n-2
    # grad_i = 2 * conv_rev[offset - i]
    idx = offset - np.arange(n)
    grad_vals = conv_rev[idx]
    return 2.0 * grad_vals


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum(a)=1)."""
    conv = _conv_self_fft(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Core optimiser for a single run (one start, one length)
# ----------------------------------------------------------------------


def _optimize_one_run(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Adam‑style optimisation of the soft‑max surrogate of the ℓ∞‑norm
    of the self‑convolution, projected onto the simplex.
    Returns the best vector found during the run and its exact score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)            # enough for all needed convolutions

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.08                              # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max schedule
    alpha_start = 0.2
    alpha_end = 40.0

    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    t = 0
    last_improve = start

    while True:
        now = time.monotonic()
        if now - start >= time_budget:
            break

        # ----- convolution and current score -----
        conv = _conv_self_fft(a, fft_len)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best = a.copy()
            last_improve = now

        # ----- soft‑max temperature (linear schedule) -----
        frac = (now - start) / time_budget
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # ----- soft‑max weights -----
        w = _softmax_weights(conv, alpha)

        # ----- gradient of the surrogate -----
        grad = _gradient_from_weights(a, w, fft_len)

        # ----- Adam update with Euclidean projection -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_onto_simplex(a)

        # ----- occasional random jitter to escape plateaus -----
        if t % 2000 == 0:
            jitter = rng.normal(scale=0.004, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a)

        # ----- simple stagnation check -----
        if now - last_improve > 4.5 and (time_budget - (now - start)) > 5.0:
            # re‑jitter to provoke movement
            a = a + rng.normal(scale=0.01, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a)
            last_improve = now

    return best, best_score

            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # ---- build an initial point for this run ----
        if n == best.size:
            # jitter the current best a little
            a0 = best.copy()
            noise = rng.normal(scale=0.02, size=n)
            a0 = np.maximum(a0 + noise, 0.0)
            a0 = _project_onto_simplex(a0)
        else:
            # random Dirichlet start – α=0.7 gives a mildly sparse start
            alpha = np.full(n, 0.7)
            a0 = rng.dirichlet(alpha).astype(np.float64)

        # ---- allocate run budget ----
        remaining = deadline - time.monotonic()
        run_budget = min(min_run_time, remaining)
        if run_budget < 1.0:
            break

        # ---- optimise this start ----
        a_opt, score_opt = _optimize_one_run(a0, run_budget, rng)

        # ---- keep the best overall ----
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # Final processing – evaluate with the original evaluator and store
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)       # evaluator caps values anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
```

This implementation introduces a completely different optimisation paradigm (Adam on a soft‑max surrogate) combined with a calibrated temperature schedule, random jitter and multi‑restart handling.  It stays fully within the allowed 1000 s budget, projects onto the probability simplex after every update, and returns a feasible non‑negative sequence suitable for the `evaluate_sequence` evaluator.
