def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)

    # load initial a
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
    else:
        n0 = 2048
        a0 = np.full(n0, 1.0/n0, dtype=np.float64)
    n = a0.size

    # initialize x via log of a0+eps
    eps = 1e-12
    x = np.log(a0 + eps)
    # optionally subtract mean to reduce magnitude
    x = x - np.max(x) # not necessary

    fft_len = _next_pow2(3*n)

    # Helper to compute actual score
    def compute_score(a):
        conv = np.fft.irfft(np.fft.rfft(a, n=fft_len) * np.fft.rfft(a, n=fft_len), n=fft_len)[:2*n-1]
        return float(2.0 * n * conv.max())

    # Phase1: Adam on log-sum-exp approximation
    best_a = None
    best_score = np.inf
    start = time.monotonic()
    deadline = start + budget_s - 5.0
    # hyperparameters
    total_iters_phase1 = 20000
    # schedule of beta
    beta_start = 0.5
    beta_end = 300.0

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    it = 0
    phase1_end = start + 0.5 * (deadline - start)   # allocate 50% time
    while time.monotonic() < phase1_end:
        # compute softmax of x to get a
        # stable softmax
        x_shift = x - np.max(x)
        a = np.exp(x_shift)
        a = a / a.sum()
        # conv self
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
        # smooth max via log-sum-exp
        # compute beta progression
        progress = (time.monotonic() - start) / (phase1_end - start)
        beta = beta_start * (beta_end / beta_start) ** progress
        max_conv = conv.max()
        # compute softmax weights w_i = exp(beta*(conv_i - max_conv)) normalized
        w_exp = np.exp(beta * (conv - max_conv))
        w = w_exp / w_exp.sum()
        # compute gradient wrt a: grad_a = 2 * conv_rev_part where conv_rev = conv(w, a_rev)
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, n=fft_len)
        a_rev_fft = np.fft.rfft(a_rev, n=fft_len)

        grad_a = 2.0 * conv_rev_full[n-1:n-1+n]   # length n
        # gradient wrt x via chain rule: grad_x = a * (grad_a - dot(grad_a, a))
        dot_ga = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot_ga)
        # Adam update
        it += 1
        lr_t = 0.1 * np.sqrt(1 - beta2**it) / (1 - beta1**it)
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        x = x - lr_t * m / (np.sqrt(v) + eps_adam)

        # periodic evaluate true score
        if it % 50 == 0:

            print(f"[{elapsed:6.2f}s] Phase3 iter={it3}, best={best_score:.6f}")

"""
propose_candidate
~~~~~~~~~~~~~~~~~

A three‑stage local optimiser for the max‑convolution objective.

Stage 1 – Adam on a smooth log‑sum‑exp surrogate of the max entry.
           The surrogate temperature (β) is increased exponentially,
           the full gradient is obtained via FFT‑based convolution and
           the simplex constraint is handled by a soft‑max (mirror‑descent).

Stage 2 – Targeted pairwise mass transfers that explicitly reduce the
           currently maximal convolution entry using its exact sub‑gradient.

Stage 3 – Simulated‑annealing‑style random moves to escape remaining
           local minima.

The routine respects the 1000 s budget, prints progress and returns the
best sequence (list of floats) it discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2*len(a)-1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights proportional to exp(beta * conv).
    Returns a probability vector of length len(conv).
    """
    m = conv.max()
    w = np.exp(beta * (conv - m))
    w /= w.sum()
    return w


def _grad_smooth_max_wrt_a(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) objective w.r.t. a.
    Implements  g = 2 * (w ⋆ a)  where ⋆ denotes cross‑correlation.
    """
    n = a.size
    a_rev = a[::-1]
    # convolution of w with reversed a
    w_fft = np.fft.rfft(w, n=fft_len)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    conv_rev_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)[: 3 * n - 2]
    # slice corresponds to indices n‑1 … 2n‑2 (inclusive) → length n
    grad = 2.0 * conv_rev_full[n - 1 : n - 1 + n]
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact evaluation score 2·n·max(conv) (sum(a)=1)."""
    n = a.size
    conv = _conv_self_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a (sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size == 0:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # reasonable default: power‑of‑two length for fast FFTs
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a0.astype(np.float64)
        n = a.size

    # Normalise to the simplex (protect against round‑off)
    a = a / a.sum()

    # FFT length sufficient for all convolutions we need
    fft_len = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # ------------------------------------------------------------------
    # 1) Phase 1 – Adam on smoothed surrogate (≈50 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start + 0.5 * (deadline - start)
    # Adam state
    eps_adam = 1e-8
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initialise optimisation variables in the soft‑max (log‑it) space
    eps = 1e-12
    x = np.log(a + eps)

    # schedules for the smoothing parameter β
    beta_start = 0.5
    beta_end = 300.0

    best_a = a.copy()
    best_score = _score_from_a(best_a, fft_len)

    it = 0
    while time.monotonic() < phase1_end:
        # ---- compute a from the current log‑variables ----
        a = _softmax(x)

        # ---- self‑convolution ----
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ---- smoothing schedule (β grows exponentially) ----
        progress = (time.monotonic() - start) / (phase1_end - start)
        beta = beta_start * (beta_end / beta_start) ** progress

        # ---- soft‑max weights w_i ∝ exp(β·conv_i) ----
        w = _smooth_weights(conv, beta)

        # ---- gradient of the smoothed objective w.r.t. a ----
        grad_a = _grad_smooth_max_wrt_a(a, w, fft_len)

        # ---- chain rule to obtain gradient w.r.t. x (soft‑max Jacobian) ----
        dot_ga = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot_ga)

        # ---- Adam update (learning‑rate schedule) ----
        it += 1
        lr_t = 0.1 * math.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it)
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        x = x - lr_t * m / (np.sqrt(v) + eps_adam)

        # ---- occasional exact evaluation (every few iterations) ----
        if it % 50 == 0:
            score = _score_from_a(a, fft_len)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] Phase 1 it={it:6d}, β={beta:6.2f}, best={best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached in Phase 1")
                    break

    # ------------------------------------------------------------------
    # 2) Phase 2 – Targeted pairwise mass transfers (≈30 % of budget)
    # ------------------------------------------------------------------
    a = best_a.copy()
    phase2_end = start + 0.8 * (deadline - start)
    it2 = 0
    while time.monotonic() < phase2_end:
        # current self‑convolution and its maximiser
        conv = _conv_self_fft(a, fft_len)
        max_idx = int(np.argmax(conv))

        # exact sub‑gradient of the active (maximal) entry
        grad = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n, max_idx + 1)          # exclusive
        idx_range = np.arange(i_start, i_end, dtype=int)
        grad[idx_range] = a[max_idx - idx_range]

        # donor = index with largest gradient, receiver = smallest
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))

        # fractional move (randomised magnitude)
        frac = rng.uniform(0.001, 0.05)            # up to 5 % of source mass
        delta = a[src] * frac
        delta = min(delta, a[src])                # safety

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand /= cand.sum()                        # renormalise to the simplex

        # evaluate candidate
        score_cand = _score_from_a(cand, fft_len)
        score_curr = _score_from_a(a, fft_len)

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Phase 2 it={it2:6d}, new best={best_score:.6f}")

        # accept non‑worsening moves to keep the walk downhill
        if score_cand <= score_curr:
            a = cand

        it2 += 1

        # occasional jitter to avoid stagnation
        if it2 % 5000 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        if it2 % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Phase 2 iter={it2}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 3 – Simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    a = best_a.copy()
    phase3_end = deadline
    it3 = 0
    temp_start = 0.05 * best_score               # modest initial temperature

    while time.monotonic() < phase3_end:
        elapsed = time.monotonic() - start
        fraction = elapsed / (phase3_end - start)
        temperature = temp_start * (1.0 - fraction)

        # propose a random source (biased by current mass)
        src = rng.choice(n, p=a)
        # random distinct target
        tgt = rng.integers(0, n)
        if tgt == src:
            tgt = (tgt + 1) % n

        # move a small random fraction of the source mass
        delta = a[src] * rng.uniform(0.001, 0.03)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand /= cand.sum()

        # evaluate
        score_cand = _score_from_a(cand, fft_len)
        score_curr = _score_from_a(a, fft_len)

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Phase 3 it={it3:6d}, new best={best_score:.6f}")

        # Metropolis criterion
        if (score_cand < score_curr) or (rng.random() < math.exp((score_curr - score_cand) / max(1e-12, temperature))):
            a = cand

        it3 += 1
        if it3 % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Phase 3 iter={it3}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clip to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(v) for v in best]
