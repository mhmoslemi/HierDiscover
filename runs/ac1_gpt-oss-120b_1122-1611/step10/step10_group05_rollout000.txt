"""
Projected gradient descent on a smooth approximation of max (a*a).

The objective to minimise is

    F(a) = 2·n·max_k (a*a)[k]      (a_i ≥ 0,  Σ a_i = 1)

Because the max is non‑smooth we replace it by the log‑sum‑exp
surrogate

    φ_β(a) = (1/β)·log Σ_k exp( β·(a*a)[k] ).

For a fixed β the surrogate is differentiable and its gradient has a
closed form that can be evaluated with two FFT‑based convolutions.
We perform an Adam‑style projected gradient descent on the simplex,
gradually increasing β (homotopy) so that the surrogate tightens to the
true max.  Random jitter is added every few thousand steps to escape
local minima.  The best vector seen during the whole run is returned.

The routine respects the 1000 s budget, prints progress and aborts as
soon as the target score 1.5025 is reached.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, b: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution of two 1‑D real vectors using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    n = a.size
    m = b.size
    conv_len = n + m - 1
    if fft_len is None:
        fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(A * B, n=fft_len)[:conv_len]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the time budget.

    The algorithm:
      1. Initialise a probability vector (either from the previous best
         ``height_sequence_1`` or uniform).
      2. Perform Adam‑style projected gradient descent on the
         log‑sum‑exp surrogate of the max‑auto‑correlation.
         β (the smoothing parameter) is increased linearly from
         ``β_start`` to ``β_end``.
      3. Every few thousand iterations a small Gaussian jitter is added.
      4. The best vector seen is stored; if its exact score drops below
         the target (1.5025) we stop early.
      5. The final best vector is clipped to the limits required by
         ``evaluate_sequence`` and returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a (probability simplex) and problem size n
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
        a = a0.copy()
    else:

        a = np.full(n, 1.0 / n, dtype=np.float64)

    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 0.2                        # base learning rate
    t = 0                           # iteration counter

    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------

    conv = _conv_full_fft(a, a, fft_len=_next_pow2(3 * n))
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best = a.copy()
    best_score = cur_score
    print(f"[0.00s] initial exact score = {best_score:.6f}")

    # --------------------------------------------------------------
    # surrogate schedule
    beta_start = 0.5
    beta_end = 200.0

    # printing control
    last_print = start_time

    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        #   a) compute convolution a*a (exact, needed for both
        #      surrogate and exact objective)
        # ----------------------------------------------------------
        conv = _conv_full_fft(a, a, fft_len=_next_pow2(3 * n))
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----------------------------------------------------------
        #   b) surrogate parameters (β increasing linearly)
        # ----------------------------------------------------------
        elapsed = time.monotonic() - start_time
        total_time = deadline - start_time
        frac_time = min(1.0, elapsed / total_time)
        beta = beta_start + (beta_end - beta_start) * frac_time

        # ----------------------------------------------------------
        #   c) log‑sum‑exp surrogate and its gradient
        # ----------------------------------------------------------
        shift = conv - cur_max                 # ≤ 0
        exp_shift = np.exp(beta * shift)       # safe, max element = 1
        sum_exp = exp_shift.sum()

        # convolution exp_shift * a_rev
        a_rev = a[::-1]
        fft_len2 = _next_pow2(exp_shift.size + a_rev.size - 1)
        exp_fft = np.fft.rfft(exp_shift, n=fft_len2)
        rev_fft = np.fft.rfft(a_rev, n=fft_len2)
        conv_rev = np.fft.irfft(exp_fft * rev_fft, n=fft_len2)
        conv_rev = conv_rev[: exp_shift.size + a_rev.size - 1]

        # slice that corresponds to indices n‑1 … 2n‑2
        grad = a + conv_rev[n - 1:n - 1 + n] / sum_exp

        # ----------------------------------------------------------
        #   d) Adam update on the simplex (project after step)
        # ----------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)      # enforce non‑negativity and Σa = 1

        # ----------------------------------------------------------
        #   e) Keep the globally best solution
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:

            best_score = cur_score
            print(f"[{elapsed:6.2f}s] new best (grad) = {best_score:.6f}")

            if best_score <= 1.5025:

                break

        # ----------------------------------------------------------
        if t % 3000 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            a_jit = a + jitter
            a_jit = np.clip(a_jit, 0.0, None)
            a_jit = _project_onto_simplex(a_jit)

            conv_jit = _conv_full_fft(a_jit, a_jit, fft_len=_next_pow2(3 * n))
            score_jit = 2.0 * n * conv_jit.max()
            if score_jit < best_score - 1e-12:
                best = a_jit.copy()
                best_score = score_jit
                print(f"[{elapsed:6.2f}s] new best (jitter) = {best_score:.6f}")
            # accept jitter as current state if it is not much worse
            if score_jit <= cur_score + 1e-8:
                a = a_jit
                # (m,v) keep their old values – they will be corrected in the next step

        # ----------------------------------------------------------
        #   g) Periodic status output (every ~5 s)
        # ----------------------------------------------------------
        if elapsed - (last_print - start_time) >= 5.0:
            print(f"[{elapsed:6.2f}s] best so far = {best_score:.6f}")
            last_print = time.monotonic()

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
