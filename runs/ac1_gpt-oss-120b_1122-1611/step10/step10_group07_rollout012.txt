"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

using a three‑stage search:

0. Global random sampling (Dirichlet + power‑law).
1. Smoothed‑max gradient descent with an annealed soft‑max weight
   (β grows from 0.5 to ≈2000).  The update is multiplicative
   (exponentiated gradient) followed by projection onto the simplex.
2. Stochastic local refinement (mass‑transfer moves) driven by the
   exact sub‑gradient of the active constraints, with a simple
   simulated‑annealing acceptance rule.

The routine respects the 1000 s budget (5 s safety margin) and prints
progress information.  It returns the best sequence found as a list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length *fft_len*."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max_k (a*a)[k] for the active indices."""
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Return a sequence of non‑negative heights that (hopefully) improves
    the current best bound.  The function respects the time budget
    (≈budget_s seconds) and prints progress information.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Initialise a – use the previous best if available, else uniform.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a0 = np.clip(a0, 0.0, None)
    a0 = a0 / a0.sum()
    n = a0.size
    fft_len = _next_pow2(3 * n)          # enough for all linear convolutions

    # ------------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the best solution so far
    # ------------------------------------------------------------------
    conv0 = _conv_fft(a0, fft_len)
    best = a0.copy()
    best_score = 2.0 * n * conv0.max()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 0 – random exploration (≈10 % of time)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.10 * (deadline - start_time)
    it_rand = 0
    while time.monotonic() < phase0_end:
        if rng.random() < 0.55:                     # Dirichlet from coarse bins
            K = max(8, n // 64)
            alpha = rng.uniform(0.2, 2.5)
            bins = rng.dirichlet(np.full(K, alpha))
            if rng.random() < 0.5:                 # optional decreasing order
                bins = np.sort(bins)[::-1]
            seq = np.repeat(bins, n // K)
            if seq.size < n:
                seq = np.concatenate([seq, bins[: n - seq.size]])
        else:                                        # power‑law shape
            p = rng.uniform(0.1, 3.0)
            idx = np.arange(1, n + 1, dtype=np.float64)
            seq = idx ** (-p)

        seq = seq / seq.sum()
        conv = _conv_fft(seq, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score, best = score, seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (rand) = {best_score:.6f} (iter={it_rand})")
        it_rand += 1
        if it_rand % 4000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] random search iter={it_rand}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 1 – smoothed‑max (soft‑max) gradient descent (≈50 % of time)
    # ------------------------------------------------------------------
    a = best.copy()
    beta = 0.5                     # start with a gentle soft‑max
    beta_factor = 1.02
    max_beta = 2000.0
    base_eta = 0.7                # base step size for exponentiated update
    it_sgd = 0
    phase1_end = start_time + 0.60 * (deadline - start_time)    # 50 % dedicated

    while time.monotonic() < phase1_end:
        # FFT of the current vector
        A_fft = np.fft.rfft(a, n=fft_len)

        # convolution a*a (linear) and its maximum
        conv = np.fft.irfft(A_fft * A_fft, n=fft_len)[: 2 * n - 1]
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # ---- soft‑max weights over the convolution entries ----------
        # shift by max_b for numerical stability
        w_exp = np.exp(beta * (conv - max_b))
        w = w_exp / w_exp.sum()                     # shape (2n‑1,)

        # ---- gradient of the smoothed objective --------------------
        # gradient_i = Σ_k w_k * a_{k-i}  == correlation(a, w)
        w_rev = w[::-1]
        W_rev_fft = np.fft.rfft(w_rev, n=fft_len)

        grad_full = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
        # keep only the linear part (length 3n‑2)
        grad_full = grad_full[: 3 * n - 2]

        # extract the n entries we need (see analysis)
        grad = grad_full[n - 1: 2 * n - 1][::-1]

        # ---- multiplicative (exponentiated) update -----------------
        eta = base_eta / np.sqrt(it_sgd + 1.0)
        log_a = np.log(a + 1e-16) - eta * grad
        a_new = np.exp(log_a)
        a_new = _project_onto_simplex(a_new)

        # ---- evaluate candidate ------------------------------------
        A_fft_new = np.fft.rfft(a_new, n=fft_len)
        conv_new = np.fft.irfft(A_fft_new * A_fft_new, n=fft_len)[: 2 * n - 1]
        score_new = 2.0 * n * conv_new.max()

        # Accept if improvement; otherwise probabilistic uphill move
        if score_new < cur_score:
            a = a_new
        else:
            # temperature for uphill acceptance – decays with iterations
            T = 0.02 * (0.98 ** it_sgd)
            if rng.random() < np.exp(-(score_new - cur_score) / (T + 1e-12)):
                a = a_new

        # keep the best ever seen
        if score_new < best_score - 1e-12:
            best_score, best = score_new, a_new.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (sgd) = {best_score:.6f} (iter={it_sgd})")
            if best_score <= 1.5025:
                print("[early stop] target reached in SGD phase")
                break

        # anneal the soft‑max parameter
        beta = min(beta * beta_factor, max_beta)

        it_sgd += 1
        if it_sgd % 250 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] sgd iter={it_sgd}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase 2 – stochastic local refinement with simulated annealing
    # ------------------------------------------------------------------
    a = best.copy()
    conv = _conv_fft(a, fft_len)
    cur_score = 2.0 * n * conv.max()
    it_loc = 0
    # simple exponential cooling schedule
    T_start = 0.02
    T_end = 1e-5
    total_loc_time = deadline - time.monotonic()
    # we will drive temperature by elapsed fraction of the remaining time
    while time.monotonic() < deadline:
        # recompute convolution and active set (tiny tolerance)
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-8
        active = np.where(conv >= max_b - eps)[0]

        # exact sub‑gradient of the max‑convolution
        grad_act = _gradient_from_active(a, active)

        # source = index with largest gradient, target = smallest gradient
        src = int(np.argmax(grad_act))
        tgt = int(np.argmin(grad_act))
        if src == tgt:
            tgt = int(rng.integers(0, n))

        # random transfer amount (log‑uniform)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.03)))   # 0.1 %–3 %
        delta = a[src] * frac
        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta
        a_cand = _project_onto_simplex(a_cand)

        # evaluate candidate
        conv_c = _conv_fft(a_cand, fft_len)
        score_c = 2.0 * n * conv_c.max()

        # temperature based on how much time we have spent in this phase
        elapsed = time.monotonic() - start_time
        frac_elapsed = (elapsed - (deadline - total_loc_time)) / total_loc_time
        T = T_start * (T_end / T_start) ** frac_elapsed

        # accept if better, otherwise with Metropolis probability
        if score_c < cur_score or rng.random() < np.exp(-(score_c - cur_score) / (T + 1e-12)):
            a = a_cand
            cur_score = score_c
            if score_c < best_score - 1e-12:
                best_score, best = score_c, a_cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (iter={it_loc})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in local phase")
                    break

        # occasional jitter to escape flat regions
        if it_loc % 5000 == 0 and it_loc > 0:
            width = max(1, n // 80)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.019))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a_jit = _project_onto_simplex(a_jit)
            conv_j = _conv_fft(a_jit, fft_len)
            score_j = 2.0 * n * conv_j.max()
            if score_j < cur_score:
                a = a_jit
                cur_score = score_j
                if score_j < best_score - 1e-12:
                    best_score, best = score_j, a_jit.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] new best (jitter) = {best_score:.6f} (iter={it_loc})")

        it_loc += 1
        if it_loc % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local iter={it_loc}, best={best_score:.6f}")
