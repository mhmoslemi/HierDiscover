"""
Hybrid stochastic optimizer for the step‑function problem.

Key ideas
---------

* Work on the probability simplex (∑a_i = 1, a_i ≥ 0).  The objective 
  J(a) = 2·n·max(conv(a,a)) is invariant under scaling, so we keep the
  vector normalized after each update.

* Use an *exponentiated‑gradient* (multiplicative) update driven by a
  sub‑gradient of the max‑convolution.  The active set (diagonals that
  are within a small relative tolerance of the current maximum) is
  enlarged (≈ 1 % of the peak) so that the gradient contains more
  information than the single‑peak version used previously.

* Occasionally perform a cheap *mass‑transfer* move: move a small
  fraction of probability from a high‑gradient index to a low‑gradient
  one.  This gives the walk extra ability to escape shallow plateaus.

* A simple linear cooling schedule is used for Metropolis acceptance.
  If no improvement has been observed for a while a random *jitter*
  (multiplicative Gaussian noise) is applied to the current best
  solution – this restarts the walk without destroying the useful
  structure that has been found.

The routine returns the best vector (as a list of floats) found before
the supplied time budget expires.  All numbers are clipped to the
range accepted by ``evaluate_sequence`` before returning.
"""

import time
import numpy as np

# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution of a with itself using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

    """

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for convolutions
    grad_fft_len = _next_pow2(3 * n - 2)          # for the (optional) gradient step

    # ------------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors.
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (moderate and strong concentration)
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # reuse the previous construction if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best of the initial candidates.
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        max_conv = conv.max()
        J = 2.0 * n * max_conv
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best  J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # small safety margin

    # temperature schedule (linear cooling)
    T_start = best_J * 0.10      # start at 10 % of the current best value
    T_end = 1e-9

    cur_vec = best_vec.copy()
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------- evaluate current state -----------------
        cur_conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = cur_conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ---- update global best ---------------------------------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- stagnation handling --------------------------------
        if now - last_improve > 20.0:
            # multiplicative Gaussian jitter around the best solution
            noise = np.exp(rng.normal(0.0, 0.2, size=n))
            cur_vec = best_vec * noise
            cur_vec = np.clip(cur_vec, 0.0, None)
            cur_vec /= cur_vec.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (jitter) – best J = {best_J:.6f}")

        # ---- compute an enlarged active set --------------------
        # a relative tolerance that loosens the active set early on
        if iteration < 5000:
            rel_tol = 0.01          # 1 % of the peak
        elif iteration < 20000:
            rel_tol = 0.005
        else:
            rel_tol = 0.001
        tol = max(1e-14, max_conv * rel_tol)
        active = np.where(cur_conv >= max_conv - tol)[0]

        # ---- sub‑gradient of max(conv) on the active set ----------
        grad = np.zeros_like(cur_vec)
        for p in active:
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1)
            grad[idx] += 2.0 * cur_vec[p - idx]

        # ---- propose a new point ---------------------------------
        # with probability 0.7 use the exponentiated‑gradient step,
        # otherwise perform a random mass‑transfer move.
        if rng.random() < 0.7:
            # exponentiated‑gradient (multiplicative) update
            step = 0.8 / np.sqrt(iteration + 1.0)   # relatively aggressive early on
            factor = np.exp(-step * grad)
            proposal = cur_vec * factor
            s = proposal.sum()
            if s == 0.0:
                proposal[:] = 1.0 / n
            else:
                proposal /= s
        else:
            # cheap random transfer from a high‑gradient index to a low‑gradient one
            if grad.sum() > 0.0:
                donor = rng.choice(n, p=grad / grad.sum())
            else:
                donor = rng.integers(n)
            # favour low‑gradient indices as receivers
            inv = grad.max() - grad + 1e-12
            receiver = rng.choice(n, p=inv / inv.sum())
            # move a modest fraction of the donor mass
            delta = min(cur_vec[donor], 0.05 * cur_vec[donor])  # up to 5 %
            proposal = cur_vec.copy()
            proposal[donor] -= delta
            proposal[receiver] += delta
            # renormalisation is not needed – the sum is unchanged

        # ---- evaluate the proposal -------------------------------
        prop_conv = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ---- Metropolis acceptance --------------------------------
        # linear temperature schedule
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = proposal

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution (safe clipping and normalisation).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # optional: report final objective (use evaluate_sequence if desired)
    # print(f"Final best J = {best_J:.6f}")

    return [float(v) for v in best_vec]
