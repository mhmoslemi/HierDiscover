"""
Hybrid parametric search + projected sub‑gradient refinement.

The goal is to minimise
    J(a) = 2·n·max(conv(a,a))       (sum(a)=1 is enforced)
which is exactly the value returned by `evaluate_sequence`.

Approach
--------
1.  Choose a few problem sizes (n) – larger n gives a finer
    discretisation of the continuous optimisation problem and often
    yields a lower bound.
2.  For each n generate a diverse portfolio of *parametric* candidates:
        • power‑law    a_i ∝ ((i+0.5)*(n‑0.5‑i))^{-β}
        • exponential  a_i ∝ exp(-λ·|i‑centre|)
        • Gaussian     a_i ∝ exp(-½·((i‑centre)/σ)²)
        • sinusoid     a_i ∝ sin(π·(i+1)/(n+1))^p
        • uniform and a few random Dirichlet draws
   The parameters are swept on coarse grids; each candidate is
   projected onto the probability simplex before its J‑value is computed.

   The best shape found in this exhaustive sweep becomes the starting
   point for the local optimiser.

3.  Local optimiser – projected sub‑gradient descent with a
    Metropolis acceptance rule (annealing).  The sub‑gradient of
    `max(conv)` is the sum of shifted copies of the current vector, taken
    over all (almost) active convolution indices.  The step size
    decays like 1/√t, and a small amount of random jitter is injected
    after long periods without improvement.

4.  The algorithm respects the overall time budget (≈1000 s).  It tracks
    the best‑ever solution and returns it as a list of floats.

The implementation uses only NumPy, the built‑in `time` module and the
helpers defined below.  No file I/O or external processes are used.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _max_conv(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Return max(conv(a,a)) using an FFT‑based convolution."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0.  Global timing control
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0   # 5 s safety margin

    # ------------------------------------------------------------------
    # 1.  Problem sizes to explore
    # ------------------------------------------------------------------
    # Smaller sizes are cheap, larger sizes give a finer approximation.
    # We start with a moderate size and, if time permits, explore a larger one.
    size_candidates = [1024, 2048, 4096, 8192]

    best_vec = None
    best_J = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # 2.  Exhaustive parametric sweep (fast, deterministic)
    # ------------------------------------------------------------------
    for n in size_candidates:
        now = time.monotonic()
        if now >= deadline:
            break

        conv_len = 2 * n - 1
        fft_len = _next_pow2(conv_len)

        # Helper to evaluate a candidate (already normalised)
        def eval_candidate(v: np.ndarray) -> float:
            """Return J = 2·n·max(conv) for a probability vector v."""
            m = _max_conv(v, fft_len, conv_len)
            return float(2.0 * n * m)

        # ------------------------------------------------------------------
        # Uniform baseline
        # ------------------------------------------------------------------
        cand = np.full(n, 1.0 / n, dtype=np.float64)
        J = eval_candidate(cand)
        if J < best_J:
            best_J, best_vec, best_n = J, cand.copy(), n

        # ------------------------------------------------------------------
        # Power‑law family: a_i ∝ ((i+½)*(n‑½‑i))^{-β}
        # ------------------------------------------------------------------
        idx = np.arange(n, dtype=np.float64)
        x = (idx + 0.5) * (n - 0.5 - idx)          # product term, ≈ distance from centre
        eps = 1e-12
        for beta in np.linspace(-2.0, 2.0, 41):    # 41 points ⇒ step 0.1
            w = (x + eps) ** (-beta)
            # Numerical safety – clip extreme values
            w = np.clip(w, 0.0, 1e12)
            s = w.sum()
            if s == 0.0:
                continue
            cand = w / s
            J = eval_candidate(cand)
            if J < best_J:
                best_J, best_vec, best_n = J, cand.copy(), n

        # ------------------------------------------------------------------
        # Symmetric exponential decay: a_i ∝ exp(-λ·|i‑centre|)
        # ------------------------------------------------------------------
        centre = (n - 1) / 2.0
        d = np.abs(idx - centre)
        for lam in np.logspace(-4, -1, 20):       # λ from 1e‑4 to 0.1
            cand = np.exp(-lam * d)
            cand /= cand.sum()
            J = eval_candidate(cand)
            if J < best_J:
                best_J, best_vec, best_n = J, cand.copy(), n

        # ------------------------------------------------------------------
        # Gaussian family: a_i ∝ exp(-½·((i‑centre)/σ)²)
        # ------------------------------------------------------------------
        for sigma in np.linspace(n / 16.0, n / 2.0, 12):
            cand = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
            cand /= cand.sum()
            J = eval_candidate(cand)
            if J < best_J:

        # Sinusoid family: a_i ∝ sin(π·(i+1)/(n+1))^p
        # ------------------------------------------------------------------
        for p in np.linspace(0.1, 5.0, 25):
            # Avoid zero at the ends by starting at i=1 … n‑2 when p large
            sin_vals = np.sin(np.pi * (idx + 1) / (n + 1))
            # sin is zero at the ends; to keep positivity we add a tiny epsilon
            cand = (sin_vals + 1e-12) ** p
            cand /= cand.sum()
            J = eval_candidate(cand)
            if J < best_J:
                best_J, best_vec, best_n = J, cand.copy(), n

        # ------------------------------------------------------------------
        # Random Dirichlet draws – moderate concentration
        # ------------------------------------------------------------------
        for alpha in [0.2, 0.5, 1.0, 2.0]:
            for _ in range(5):
                cand = rng.dirichlet(np.full(n, alpha))
                J = eval_candidate(cand)
                if J < best_J:
                    best_J, best_vec, best_n = J, cand.copy(), n

        # ------------------------------------------------------------------
        # Log progress for this problem size
        # ------------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] Size {n:5d} – best J so far = {best_J:.6f}")

    # If nothing was found (should not happen) fall back to a uniform vector
    if best_vec is None:
        best_n = 2048
        best_vec = np.full(best_n, 1.0 / best_n)
        best_J = 2.0 * best_n * _max_conv(best_vec,
                                          _next_pow2(2 * best_n - 1),
                                          2 * best_n - 1)

    # ------------------------------------------------------------------
    # 3.  Local refinement on the best size discovered
    # ------------------------------------------------------------------
    n = best_n
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    cur_vec = best_vec.copy()
    cur_J = best_J
    iteration = 0
    last_improve = time.monotonic()

    # Temperature schedule for Metropolis acceptance (linear cooling)
    # Initial temperature is set to ~5 % of the current best objective.
    T_start = cur_J * 0.05
    T_end = 1e-8

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear cooling of temperature
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))

        # --------------------------------------------------------------
        # 3.1  Compute sub‑gradient of max(conv) at the current point
        # --------------------------------------------------------------
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        # Update global best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 3.2  Stagnation handling – jitter after 30 s w/o improvement
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            # Refresh the convolution after the jitter
            conv = _conv_full(cur_vec, fft_len, conv_len)
            cur_J = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 3.3  Determine active set (indices where conv is (almost) maximal)
        # --------------------------------------------------------------
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]

        # --------------------------------------------------------------
        # 3.4  Sub‑gradient of max(conv) – sum of shifted copies of a
        # --------------------------------------------------------------
        grad = np.zeros_like(cur_vec)
        # For every active convolution index k, add the contribution
        for k in active:
            start = k - (n - 1)
            if start < 0:
                start = 0
            end = k if k < n else n - 1
            idx = np.arange(start, end + 1, dtype=np.int64)
            grad[idx] += 2.0 * cur_vec[k - idx]

        # --------------------------------------------------------------
        # 3.5  Projected sub‑gradient descent step (with a diminishing step)
        # --------------------------------------------------------------
        step = 0.5 / np.sqrt(iteration + 1.0)      # 1/√t decay
        proposal = cur_vec - step * grad
        proposal = np.maximum(proposal, 0.0)       # stay non‑negative
        proposal = proj_simplex(proposal)         # back onto the simplex

        # --------------------------------------------------------------
        # 3.6  Evaluate the proposal
        # --------------------------------------------------------------
        prop_max = _max_conv(proposal, fft_len, conv_len)
        prop_J = 2.0 * n * prop_max

        # --------------------------------------------------------------
        # 3.7  Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # Accept with probability exp(-(Δ)/T)
            if T > 0.0:
                prob = np.exp(-(prop_J - cur_J) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # ------------------------------------------------------------------
    # 4.  Final sanitisation and return
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"[{time.monotonic() - start_time:6.2f}s] Finished – final best J = {best_J:.6f}")
    return [float(v) for v in best_vec]
