def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    # Choose length n
    n = 4096
    # Initialize a uniform distribution
    a = np.full(n, 1.0 / n)
    # Precompute target uniform convolution
    target_val = 1.0 / (2*n - 1)
    target = np.full(2*n - 1, target_val, dtype=np.float64)
    # FFT lengths
    fft_len_conv = _next_pow2(3*n)
    # Precompute p_rev placeholder; we will compute each iteration.
    best = a.copy()
    cur_conv = _conv_fft(a, fft_len_conv)
    cur_score = 2.0 * n * cur_conv.max()
    best_score = cur_score
    print(f"[0.00s] init score {best_score:.6f}")
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # Stage 1: L2 flattening
    iter1 = 0
    eta0 = 0.5
    while time.monotonic() < deadline * 0.6 + start:  # 60% of time
        conv = _conv_fft(a, fft_len_conv)
        diff = conv - target
        # gradient = 4 * conv(diff, a_rev)
        a_rev = a[::-1]
        # compute convolution diff * a_rev
        # Determine FFT length for diff conv
        L = diff.size + a_rev.size - 1
        fft_len_grad = _next_pow2(L)
        grad_full = _conv_fft_general(diff, a_rev, fft_len_grad)
        grad = 4.0 * grad_full[n-1 : n-1 + n]
        # step size schedule
        eta = eta0 / np.sqrt(1.0 + iter1)
        # gradient descent step
        a_new = a - eta * grad
        # project onto simplex
        a_new = _project_onto_simplex(a_new)
        a = a_new
        # update best
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new best after L2 iter {iter1} = {best_score:.6f}")
        iter1 += 1
        # occasional jitter
        if iter1 % 5000 == 0:
            # add small random noise
            noise = rng.normal(0.0, 0.01, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_onto_simplex(a)
    # Stage 2: subgradient max reduction
    iter2 = 0
    eta0_sub = best_score * 0.05   # initial temperature 5% of current
    while time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len_conv)
        max_val = conv.max()

        active_idx = np.where(conv >= max_val - tol)[0]

        grad = np.zeros_like(a)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n, k + 1)

        a = a * np.exp(-temp * grad)  # use temp as step
        a = _project_onto_simplex(a)
        # Evaluate score

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new best after subgrad iter {iter2} = {best_score:.6f}")
        # optional jitter

            a[start_idx:start_idx+block_len] *= factor
            a = _project_onto_simplex(a)
        iter2 += 1

    final_score = evaluate_sequence(best.tolist())
    total = time.monotonic() - start
    print(f"Finished in {total:.2f}s – final best score = {final_score:.6f}")
    return [float(x) for x in best]

"""
Gradient‑descent refinement for the max‑auto‑correlation problem.

Stage 1 (majority of the time) minimises the L2 distance between the
auto‑correlation a⋆a and the uniform distribution.  This flattens the
whole convolution and usually reduces the peak well below the uniform
baseline (score ≈ 2).

Stage 2 switches to a true L∞ objective: a sub‑gradient of the current
maximum convolution entry is computed and a multiplicative (mirror‑descent)
update is performed, followed by a Euclidean simplex projection.  The
temperature (step size) is linearly cooled.

Both stages keep the vector on the probability simplex (∑a_i = 1,
a_i ≥ 0) and use fast FFT‑based convolutions.  The routine respects the
1000 s budget, prints progress, and finally returns the best sequence
found (clipped to the evaluator limits).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a⋆a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_general(a: np.ndarray, b: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution of two arbitrary 1‑D arrays using FFT.
    The result length is len(a)+len(b)-1.
    """
    if fft_len is None:
        fft_len = _next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=fft_len)
    B = np.fft.rfft(b, n=fft_len)
    C = A * B
    conv = np.fft.irfft(C, n=fft_len)[: len(a) + len(b) - 1]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= s
    return w


def _subgradient_from_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max_k (a⋆a)[k]`` for the given active indices.
    Returns the average sub‑gradient of the active entries.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)            # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    if len(active_idx) > 0:
        grad *= 2.0 / len(active_idx)
    return grad


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Choose problem size and initialise a feasible vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
    else:
        # a power‑of‑two length works nicely with the FFTs
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    # FFT lengths (chosen once – large enough for all required convolutions)
    fft_len_conv = _next_pow2(3 * n)          # for a⋆a
    fft_len_grad = _next_pow2(3 * n)          # for (a⋆a − uniform)⋆a_rev

    # target uniform convolution (average value)
    target_val = 1.0 / (2 * n - 1)
    target = np.full(2 * n - 1, target_val, dtype=np.float64)

    # --------------------------------------------------------------
    # 1) Book‑keeping
    # --------------------------------------------------------------
    cur_conv = _conv_fft(a, fft_len_conv)
    cur_score = 2.0 * n * cur_conv.max()
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Stage 1 – L2 flattening of the convolution
    # --------------------------------------------------------------
    eta0 = 0.6                 # initial step size for L2 descent
    iter1 = 0
    stage1_deadline = start_time + 0.60 * (deadline - start_time)

    while time.monotonic() < stage1_deadline:
        # convolution a⋆a
        conv = _conv_fft(a, fft_len_conv)

        # L2 error w.r.t. the uniform target
        diff = conv - target

        # gradient of ½‖diff‖²  =  (diff ⋆ a_rev) * 4
        a_rev = a[::-1]
        grad_full = _conv_fft_general(diff, a_rev, fft_len_grad)
        grad = 4.0 * grad_full[n - 1: n - 1 + n]   # extract the n‑sized slice

        # simple diminishing step size
        eta = eta0 / np.sqrt(1.0 + iter1)
        a = a - eta * grad
        a = _project_onto_simplex(a)

        # keep best so far
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (L2)  = {best_score:.6f}")

        # occasional tiny jitter to escape plateaus
        if iter1 % 5000 == 0 and iter1 > 0:
            noise = rng.normal(0.0, 0.008, size=n)
            a = np.clip(a + noise, 0.0, None)
            a = _project_onto_simplex(a)

        iter1 += 1

    # --------------------------------------------------------------
    # 3) Stage 2 – sub‑gradient descent on the L∞ objective
    # --------------------------------------------------------------
    # Temperature schedule for the multiplicative mirror‑descent update
    T0 = best_score * 0.05
    Tf = 1e-8 * best_score

    iter2 = 0
    while time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len_conv)
        max_val = conv.max()

        # active set (indices attaining the current maximum, within tolerance)
        tol = 1e-14
        active_idx = np.where(conv >= max_val - tol)[0]

        # sub‑gradient of the maximum entry
        grad = _subgradient_from_active(a, active_idx)

        # linearly cooled temperature
        elapsed = time.monotonic() - start_time
        frac = (elapsed) / (deadline - start_time)
        temp = T0 * (1.0 - frac) + Tf * frac

        # multiplicative (mirror‑descent) update
        a = a * np.exp(-temp * grad)
        a = _project_onto_simplex(a)

        # update best solution
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (L∞) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

        # occasional block‑wise jitter
        if iter2 % 3000 == 0 and iter2 > 0:
            blk = int(max(2, rng.integers(1, n // 80)))
            pos = int(rng.integers(0, n - blk + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a[pos:pos + blk] *= factor
            a = _project_onto_simplex(a)

        iter2 += 1

    # --------------------------------------------------------------
    # 4) Return the best sequence (safeguarded against evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total = time.monotonic() - start_time
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
