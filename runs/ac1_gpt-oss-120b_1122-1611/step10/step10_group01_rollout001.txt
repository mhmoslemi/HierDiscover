"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.  

The algorithm consists of three phases:

1. **Broad initial sampling** – generate a variety of parametric families
   (uniform, Gaussian, power‑law) together with a number of random Dirichlet
   draws for a few candidate lengths.  Keep the best (lowest exact score).

2. **Projected sub‑gradient descent on the true ∞‑norm** – the objective
   `max_k (a∗a)[k]` is convex in each active lag.  For the current active
   set we compute the exact sub‑gradient, take a diminishing step
   `η_t = η₀/√t` and project back onto the probability simplex.
   The step size and a light jitter are tuned so that the objective
   decreases steadily.

3. **Fine stochastic refinement** – repeatedly transfer a tiny fraction of
   mass from the index with the largest sub‑gradient (largest contribution
   to the current maximum) to the index with the smallest sub‑gradient.
   Accept the move if it does not increase the exact objective; keep the
   best sequence seen globally.

All work respects the given time budget (a 5 s safety margin is left for
finalisation).  The routine returns the best sequence found; its length is
chosen automatically from the explored candidates.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # make a copy to avoid modifying the input
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to guard against tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a∗a)[k]`` for the set ``active`` of
    indices attaining the current maximum.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0                         # each pair appears twice
    return grad

            conv = _conv_fft(a, fft_len)
            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-12)[0]
            grad = _gradient_from_active(a, active)

            # source = index that most contributes to the current max
            src = int(np.argmax(grad))
            # target = index with the smallest contribution (preferably zero)
            tgt = int(np.argmin(grad))
            if src == tgt:
                tgt = int(rng.integers(0, n))

        # ensure source still has some mass
        if a[src] <= 0.0:
            src = int(rng.integers(0, n))

        # transfer a modest fraction of the source mass
        frac = rng.uniform(np.log(0.001), np.log(0.04))
        delta = a[src] * (np.exp(frac) - 1.0)   # 0.1 %–4 % of src's mass

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta   # sum unchanged, non‑negativity preserved by construction

        conv_cand = _conv_fft(cand, fft_len)
        obj_cand = 2.0 * n * conv_cand.max()

        if obj_cand < best_obj - 1e-12:
            best_obj = obj_cand
            best = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:.2f}s] local move #{iter_local} improved to {best_obj:.6f}")

        # accept the move if it does not worsen the exact objective
        if obj_cand <= 2.0 * n * max_val + 1e-12:
            a = cand

        iter_local += 1

        # very occasional global jitter (stretch/compress a random segment)
        if iter_local % 8000 == 0:
            seg_len = int(rng.integers(1, max(2, n // 60)))
            start_seg = int(rng.integers(0, n - seg_len + 1))
            mult = np.exp(rng.normal(0.0, 0.015))
            a_jit = a.copy()
            a_jit[start_seg:start_seg + seg_len] *= mult
            a = _project_onto_simplex(a_jit)

    # ------------------------------------------------------------------
    # 4) Final evaluation and return
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # stay inside evaluator's limits
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # Return as plain Python list of floats
    return [float(x) for x in best]
