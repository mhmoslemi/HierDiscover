"""
Propose a non‑negative step‑function that minimises the evaluation function.

Algorithm
---------

* Initialise a candidate from the previous best (if present) or from a uniform
  vector of length *n* (default 4096).  The vector is normalised to have sum = 1.
* **Phase 1 – smoothed‑max gradient descent** (≈ 70 % of the time budget).  
  The objective `max_k (a*a)[k]` is approximated by the
  log‑sum‑exp (soft‑max) with a temperature parameter `α`.  The gradient of the
  smoothed objective is obtained analytically via a second convolution.  After
  each gradient step the iterate is projected back onto the probability simplex.
  `α` is increased slowly so that the optimisation progressively focuses on the
  true maximum.
* **Phase 2 – stochastic local refinement** (remaining time).  
  Small directed mass‑transfer moves (chosen by the current most‑violated
  antidiagonal) are attempted; a move is kept if it does not increase the
  genuine objective.  Occasional jitter helps to escape shallow plateaux.
* The best sequence seen during the whole run is returned (clipped to the
  evaluator limits).  Progress and scores are printed periodically.

The routine obeys the 1000 s budget (5 s safety margin) and uses only NumPy,
the provided helper utilities and the already‑imported `evaluate_sequence`
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_full(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two arbitrary vectors using the same FFT length.
    Returns the full convolution of length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[: x.size + y.size - 1]
    return conv


def _random_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Generate a random feasible candidate (sum = 1, non‑negative)."""
    # coarse Dirichlet on bins or a power‑law shape
    if rng.random() < 0.55:
        K = max(8, n // 32)
        alpha = rng.uniform(0.3, 2.0)
        bins = rng.dirichlet(np.full(K, alpha))
        # optional monotone ordering (helps create skewed shapes)
        if rng.random() < 0.5:
            bins = np.sort(bins)[::-1]
        seq = np.repeat(bins, n // K)
        if seq.size < n:
            seq = np.concatenate([seq, bins[: n - seq.size]])
    else:
        p = rng.uniform(0.05, 2.5)
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = idx ** (-p)
        seq = seq / seq.sum()
    seq = np.maximum(seq, 0.0)
    seq = seq / seq.sum()
    return seq

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a (sum = 1) and problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
    else:
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a = a0.copy()
    fft_len = _next_pow2(3 * n)   # enough for all convolutions we need

    # --------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = float(2.0 * n * conv.max())
    best = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – smoothed‑max gradient descent
    # --------------------------------------------------------------
    phase1_end = start_time + 0.70 * (deadline - start_time)

    # parameters for the smoothed objective
    alpha = 5.0                # start temperature (soft‑max sharpness)
    alpha_max = 80.0
    step0 = 0.25               # base step size for projected gradient
    it = 0
    no_improve = 0
    while time.monotonic() < phase1_end:
        # --- compute convolution a*a (full, length 2n‑1) -----------------
        a_fft = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        # keep track of the global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (grad) = {best_score:.6f} (iter={it})")
        else:
            no_improve += 1

        # --- soft‑max weights over the convolution entries ----------------
        #   w_k ∝ exp(α·(conv_k – max_val))  (numerically stable)
        w = np.exp(alpha * (conv_full - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smoothed max (vector of length n) ----------
        # grad = 2 * (a * w_rev)[n-1 : 2n-1] reversed
        w_rev = w[::-1]                     # length 2n‑1
        conv_rev = _conv_fft_full(a, w_rev, fft_len)   # length 3n‑2
        segment = conv_rev[n - 1: 2 * n - 1]           # length n
        grad = 2.0 * segment[::-1]                    # final gradient

        # --- projected gradient step ------------------------------------
        step = step0 / np.sqrt(it + 1.0)   # diminishing step size
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate)

        # accept the candidate if it does not increase the true objective
        conv_cand = _conv_fft(a_candidate, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score <= cur_score + 1e-12:
            a = a_candidate
        else:
            # keep the current a (the gradient step was too aggressive)
            pass

        # --- schedule updates -------------------------------------------
        it += 1
        # slowly increase the sharpness of the soft‑max
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.007)

        # occasional random restart if we have been stuck for a while
        if no_improve >= 2500:
            a = _random_candidate(rng, n)
            no_improve = 0
            if it % 250 == 0:

        if it % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] grad‑iter={it}, cur={cur_score:.6f}, best={best_score:.6f}")

        if best_score <= 1.5025:
            print("[early stop] target reached in gradient phase")
            break

    # --------------------------------------------------------------
    # 4) Phase 2 – stochastic local refinement (remaining time)
    # --------------------------------------------------------------
    it_loc = 0
    while time.monotonic() < deadline:
        # recompute convolution for the current a
        conv = _conv_fft(a, fft_len)
        idx_max = int(np.argmax(conv))

        # gradient of the (true) active constraint – this points to the
        # indices that contribute most to the current worst antidiagonal
        # (we reuse the original active‑set gradient routine)
        n_a = a.size
        # active set is just the single worst index (for speed)
        grad_max = np.zeros_like(a)
        k = idx_max
        i_start = max(0, k - (n_a - 1))
        i_end = min(n_a, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad_max[i] = a[k - i]

        # source: index with largest gradient contribution
        src = int(np.argmax(grad_max))
        # target: a low‑gradient / low‑value index
        low_candidates = np.where(a <= a.mean() * 0.5)[0]
        if low_candidates.size > 0 and rng.random() < 0.7:
            tgt = int(rng.choice(low_candidates))
        else:
            tgt = int(rng.integers(0, n_a))
        if src == tgt:
            tgt = (src + 1) % n_a

        # tiny directed mass transfer
        frac = np.exp(rng.uniform(np.log(0.0005), np.log(0.02)))  # 0.05 %–2 %
        delta = a[src] * frac
        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta
        a_cand = _project_onto_simplex(a_cand)

        # evaluate candidate
        conv_c = _conv_fft(a_cand, fft_len)
        cand_score = 2.0 * n * conv_c.max()
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = a_cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (move={it_loc})")
            a = a_cand
        else:
            # keep non‑worsening moves
            if cand_score <= 2.0 * n * conv.max():
                a = a_cand

        if it_loc % 4000 == 0 and it_loc > 0:
            width = max(1, n // 80)
            start_idx = int(rng.integers(0, n - width + 1))

        if it_loc % 2500 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local‑iter={it_loc}, best={best_score:.6f}")

        if best_score <= 1.5025:
            print("[early stop] target reached in local phase")
            break

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
