assistantfinal**Solution Overview**

The evaluation function is  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

and the value is invariant under scaling, therefore we can restrict to the probability
simplex \(\sum_i a_i=1\).  
The algorithm below searches the simplex for a vector that minimises  
\(2\,n\;\max (a*a)\).

The search consists of two stages:

1. **Adaptive Adam‑subgradient descent** –  
   the exact sub‑gradient of the current maximal convolution entry is computed,
   then an Adam‑style update (first‑ and second‑moment adaptation) is applied.
   After each step the vector is projected back onto the simplex.  
   This quickly drives the score down and benefits from momentum,
   which the original exponentiated‑gradient / sub‑gradient code lacked.

2. **Finetuning with simulated‑annealing** –  
   once the descent has consumed most of the time, a classic SA loop with tiny
   mass‑transfer moves and occasional “jitter” refines the solution and helps
   escape tiny local minima.

Both phases share the same helper utilities (FFT‑based convolution,
simplex projection, sub‑gradient computation).  
The routine respects the 1000 s budget, prints progress, and finally returns
the best sequence found (clipped to the limits required by `evaluate_sequence`).

```python
"""
propose_candidate
=================

Hybrid optimiser for the max‑convolution objective.

* Phase 1 – Adam‑style projected sub‑gradient descent on the simplex.
  The exact sub‑gradient of the max‑convolution is obtained from the active
  indices and the update uses Adam moments (adaptive learning rate).

* Phase 2 – Simulated‑annealing fine‑tuning with tiny directed mass‑transfer
  moves and occasional block‑wise jitter.

The routine works on the probability simplex (∑a_i = 1), therefore the
evaluation score is simply `2·n·max(conv)`.  The best sequence seen is
returned unchanged – `evaluate_sequence` rescales it automatically.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2·len(a)‑1.
    """
    a_fft = rfft(a, n=fft_len)
    conv = irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a feasible vector on the simplex.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        n = a0.size
        if a0.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a0 / a0.sum()
    else:
        n = 1024
        a = rng.random(n)
        a = a / a.sum()

    fft_len = _next_pow2(2 * n)

    # --------------------------------------------------------------
    # 1) Book‑keeping for the best solution seen so far.
    # --------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    curr_score = 2.0 * n * max_b
    best_score = curr_score
    best = a.copy()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    print(f"[0.00s] initial score = {curr_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 1 – Adam‑style projected sub‑gradient descent.
    # --------------------------------------------------------------
    # Hyper‑parameters
    base_lr = 0.5                # base learning rate (will be scaled by Adam)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    max_step_norm = 0.5          # safety cap on the update norm
    iter_cnt = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    phase1_end = start_time + 0.70 * (deadline - start_time)  # 70 % of time

    while time.monotonic() < phase1_end:
        # active set (all indices that are (almost) tight)
        tol = max_b * 1e-10
        active = np.where(conv >= max_b - tol)[0]

        # exact sub‑gradient of the max‑convolution term
        g = _gradient_from_active(a, active)

        # Adam moments
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        # adaptive step (moving opposite to the sub‑gradient)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # cap the step size – prevents catastrophic jumps
        step_norm = np.linalg.norm(step)
        if step_norm > max_step_norm:
            step = step * (max_step_norm / step_norm)

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate)

        # evaluate candidate
        conv_cand = _conv_fft(a_candidate, fft_len)
        max_b_cand = conv_cand.max()
        cand_score = 2.0 * n * max_b_cand

        # accept only if we do not increase the objective
        if cand_score <= curr_score + 1e-12:
            a = a_candidate
            conv = conv_cand
            max_b = max_b_cand
            curr_score = cand_score

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = a_candidate.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f}")

        # occasional random jitter to shake the system (helps escape flat plateaux)
        if iter_cnt % 5000 == 0:
            width = rng.integers(1, max(2, n // 100))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            a_jit = a.copy()
            a_jit[start:start + width] *= factor
            a = _project_onto_simplex(a_jit)
            conv = _conv_fft(a, fft_len)
            max_b = conv.max()
            curr_score = 2.0 * n * max_b

        if iter_cnt % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] ADAM it={iter_cnt}, score={curr_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 2 – Simulated‑annealing local refinement (remaining time).
    # --------------------------------------------------------------
    a = best.copy()
    curr_score = best_score
    T = 0.008                     # starting temperature
    decay = 0.9999                # exponential cooling
    it = 0
    while time.monotonic() < deadline:
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-15:
            continue

        # transfer a tiny fraction of mass from src to tgt
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        cand = np.clip(cand, 0.0, None)
        cand = _project_onto_simplex(cand)

        conv_cand = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        Δ = cand_score - curr_score

        # Metropolis acceptance
        if Δ < 0 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f}")

        # block‑wise jitter every few thousand moves
        if it % 6000 == 0 and it > 0:
            width = rng.integers(1, max(2, n // 80))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jitter = a.copy()
            a_jitter[start:start + width] *= factor
            a = _project_onto_simplex(a_jitter)

        # cooling
        T *= decay
        it += 1

        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA it={it}, T={T:.3e}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
```
