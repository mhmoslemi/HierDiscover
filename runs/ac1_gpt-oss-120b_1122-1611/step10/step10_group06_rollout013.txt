Propose a step‑function sequence that minimizes ``evaluate_sequence``.

The search proceeds in several independent stages and repeats the whole
pipeline for a few different lengths (4096 and 8192).  For each length we

1. **Random warm‑starts** – many Dirichlet draws (α≈0.5) are screened
   quickly using an FFT‑based convolution, keeping the best seed.

2. **L2‑flattening** – minimize the squared deviation of the
   self‑convolution from the uniform target (1/(2 n‑1)).  The gradient
   of this quadratic surrogate is computed by a single FFT
   convolution; updates are performed with an exponentiated‑gradient
   step and simplex projection.

3. **Smooth max‑convolution descent** – a log‑sum‑exp surrogate for
   the max‑convolution is optimized (the same scheme as the reference
   code) with a slowly increasing temperature β.

4. **Targeted pairwise mass transfer** – a cheap greedy move that
   reduces the currently maximal convolution entry.

5. **Short simulated‑annealing finish** – random swaps are accepted
   with a Metropolis rule to escape remaining plateaus.

All phases respect the overall time budget (default 1000 s).  The
best sequence over all lengths is returned (clipped to the evaluator’s
[0,1000] range).  The implementation uses only NumPy and the provided
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_weighted_fft(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of a (length n) with weight vector w (length 2·n‑1).
    Returns the full result of length n+len(w)‑1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: a.size + w.size - 1]
    return full


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the convolution entry (a*a)[idx].
    The i‑th component equals a[idx-i] when the index is in range.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)  # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    return grad


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

        max_pair_iters = 2000
        for it in range(max_pair_iters):
            if time.monotonic() >= deadline:
                break
            conv = _conv_self_fft(a, fft_len)
            max_idx = int(np.argmax(conv))
            grad_max = _gradient_for_index(a, max_idx)

            src = int(np.argmax(grad_max))
            tgt = int(np.argmin(grad_max))

            # move a modest fraction of source mass
            frac = rng.uniform(0.001, 0.03)
            delta = a[src] * frac
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = _simplex_normalise(cand)

            conv_cand = _conv_self_fft(cand, fft_len)
            score_cand = 2.0 * n * conv_cand.max()

            if score_cand < best_score - 1e-12:
                best_score = score_cand
                best_a = cand.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] n={n} pairwise it={it} new best={best_score:.6f}")

            # accept non‑worsening moves to keep a downhill walk
            if score_cand <= 2.0 * n * conv.max():
                a = cand

        a = best_a.copy()

        # --------------------------------------------------------------
        # 5) Brief simulated‑annealing finish
        # --------------------------------------------------------------
        max_sa_iters = 5000
        temp_start = 0.02 * best_score
        for it in range(max_sa_iters):
            if time.monotonic() >= deadline:
                break
            elapsed = time.monotonic() - start
            fraction = elapsed / (deadline - start)
            temperature = temp_start * (1.0 - fraction)

            conv = _conv_self_fft(a, fft_len)
            max_idx = int(np.argmax(conv))
            grad_max = _gradient_for_index(a, max_idx)

            prob_src = grad_max ** 2
            prob_sum = prob_src.sum()
            if prob_sum > 0.0:
                prob_src /= prob_sum
                src = int(rng.choice(n, p=prob_src))
            else:
                src = rng.integers(0, n)

            tgt = rng.integers(0, n)
            if tgt == src:
                tgt = (tgt + 1) % n

            delta = a[src] * rng.uniform(0.001, 0.025)
            cand = a.copy()
            cand[src] -= delta
            cand[tgt] += delta
            cand = _simplex_normalise(cand)

            conv_cand = _conv_self_fft(cand, fft_len)
            score_cand = 2.0 * n * conv_cand.max()
            curr_score = 2.0 * n * conv.max()

            if score_cand < best_score - 1e-12:
                best_score = score_cand
                best_a = cand.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:.2f}s] n={n} SA it={it} new best={best_score:.6f}")

            # Metropolis acceptance
            if score_cand < curr_score or rng.random() < math.exp(
                (curr_score - score_cand) / max(1e-12, temperature)
            ):
                a = cand

        # --------------------------------------------------------------
        # 6) Final evaluation with the provided evaluator
        # --------------------------------------------------------------
        final_seq = np.clip(best_a, 0.0, 1000.0)
        final_score = evaluate_sequence(final_seq.tolist())
        elapsed = time.monotonic() - start
        print(f"[{elapsed:.2f}s] length={n} final evaluated score = {final_score:.6f}")

        if final_score < best_overall_score:
            best_overall_score = final_score
            best_overall_seq = final_seq.tolist()
            print(f"[{elapsed:.2f}s] *** New overall best = {best_overall_score:.6f}")

    total = time.monotonic() - start
    print(f"Search finished in {total:.2f}s – best score = {best_overall_score:.6f}")
    return best_overall_seq
