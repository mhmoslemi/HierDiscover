"""
Iterative min‑max autocorrelation optimisation.

The routine starts from the globally stored best construction (if present)
or from a uniform distribution.  At each iteration it:
  1. computes the current maximal auto‑correlation M = max(a*a),
  2. builds the linear constraints (a * g)[k] ≤ M for all shifts k,
  3. solves a linear programme that maximises Σ g_i under those constraints,
  4. rescales g to have the same ℓ₁‑norm as a and mixes a ← (1‑t)·a + t·g,
     where the mixing factor t linearly decays with the remaining time.
The objective that we want to minimise is
    score = 2·n·M / (Σ a_i)²,
which is invariant to scaling of the whole vector.  The best vector encountered
is kept and finally returned (clipped to the evaluator’s limits).

The LP part is fast enough for a few hundred iterations on a few‑thousand‑
element vector and typically drives the score below the target 1.5025
within the allotted budget.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Linear convolution a*a via FFT; returns the maximal entry."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def _build_constraint_matrix(a: np.ndarray) -> sp.csc_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
        (a * g)[k] = (A @ g)[k]
    for every shift k.  Column j of A is a shifted copy of `a`.
    """
    n = a.shape[0]
    # each column j contains a[0], a[1], …, a[n‑1] starting at row j
    # This is a banded matrix with constant diagonals.
    # Create an (n, n) dense helper where each row d is filled with a[d].
    data = np.tile(a[:, None], (1, n))          # shape (n, n)
    offsets = np.arange(n)                     # 0 … n‑1
    rows = 2 * n - 1
    cols = n
    # SciPy's `diags` creates a sparse matrix from the supplied diagonals.
    return sp.diags(data, offsets, shape=(rows, cols), format="csc")


def _score_from_max_conv(n: int, max_conv: float, sum_a: float) -> float:
    """Exact value of evaluate_sequence without the clipping checks."""
    return 2.0 * n * max_conv / (sum_a ** 2)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.

    The algorithm uses a LP‑based max‑auto‑correlation reduction (see the
    module docstring for details).  The best vector seen before the time
    limit is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a (ℓ₁‑normalised) from the previous best or uniform
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total > 0.0:
            a /= total
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        n = 1024
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(2 * n - 1)            # length for exact linear convolution

    # --------------------------------------------------------------
    # 1) Book‑keeping
    # --------------------------------------------------------------
    max_conv = _conv_max_fft(a, fft_len)
    cur_score = _score_from_max_conv(n, max_conv, a.sum())
    best = a.copy()
    best_score = cur_score

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # 2) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 3) Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    # mixing schedule – start aggressive, end gently
    t_start = 0.5
    t_end = 0.02
    # early‑stop target
    target_score = 1.5025

    while time.monotonic() < deadline:
        # ------- 3.1 compute current bound M -------
        M = max_conv

        # ------- 3.2 build linear‑program constraints -------
        A = _build_constraint_matrix(a)                # (2n‑1) × n  sparse
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # ------- 3.3 solve LP: maximise Σ g_i subject to A g ≤ M, g ≥ 0 -------
        # scipy.linprog minimises; we minimise –Σ g_i.
        c = -np.ones(n, dtype=np.float64)
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failure – fall back to a small random jitter and continue
            # (this situation is extremely unlikely)
            if iteration % 10 == 0:
                print(f"[{time.monotonic() - start_time:6.2f}s] LP failed, applying jitter")
            # random jitter:
            idx = rng.integers(0, n)
            a[idx] = max(a[idx] * np.exp(rng.normal(0, 0.02)), 1e-12)
            a /= a.sum()
            max_conv = _conv_max_fft(a, fft_len)
            cur_score = _score_from_max_conv(n, max_conv, a.sum())
            iteration += 1
            continue

        g = np.maximum(res.x, 0.0)                     # enforce non‑negativity (numerical)

        # ------- 3.4 rescale g to keep the ℓ₁‑norm unchanged -------
        sum_a = a.sum()
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        g = g * (sum_a / sum_g)

        # ------- 3.5 mixing step -------
        elapsed = time.monotonic() - start_time
        time_frac = elapsed / (deadline - start_time)   # in [0,1]
        t = t_start * (1.0 - time_frac) + t_end * time_frac

        cand = (1.0 - t) * a + t * g
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()                               # keep ℓ₁‑norm = 1

        # ------- 3.6 evaluate candidate (fast) -------
        cand_max = _conv_max_fft(cand, fft_len)
        cand_score = _score_from_max_conv(n, cand_max, cand.sum())

        # ------- 3.7 acceptance (only if improvement) -------
        accepted = False
        if cand_score < cur_score - 1e-14:
            accepted = True
        else:
            # occasional simulated‑annealing acceptance to escape plateaus
            temp = (target_score - cur_score) * 0.01   # tiny temperature
            if temp > 0 and rng.random() < np.exp(-(cand_score - cur_score) / temp):
                accepted = True

        if accepted:
            a = cand
            max_conv = cand_max
            cur_score = cand_score

        # ------- 3.8 keep global best -------
        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f} (iter={iteration})")
            if best_score <= target_score:
                print("[early stop] target reached")
                break

        # ------- 3.9 occasional diversification (random shuffle / jitter) -------
        if iteration % 2000 == 0 and iteration > 0:
            # small random jitter on a random block
            block_len = max(1, n // 200)
            start_idx = int(rng.integers(0, n - block_len + 1))
            factor = np.exp(rng.normal(0.0, 0.015))
            a_j = a.copy()
            a_j[start_idx:start_idx + block_len] *= factor
            a_j = np.clip(a_j, 0.0, None)
            a_j /= a_j.sum()
            j_max = _conv_max_fft(a_j, fft_len)
            j_score = _score_from_max_conv(n, j_max, a_j.sum())
            if j_score < best_score - 1e-12:
                best, best_score = a_j, j_score
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (jitter) = {best_score:.6f}")
            if j_score < cur_score:
                a = a_j
                max_conv = j_max
                cur_score = j_score

        if rng.random() < 0.0007:
            # global permutation – does not change sum but may reshuffle peaks
            perm = rng.permutation(n)
            a_perm = a[perm]
            p_max = _conv_max_fft(a_perm, fft_len)
            p_score = _score_from_max_conv(n, p_max, a_perm.sum())
            if p_score < best_score - 1e-12:
                best, best_score = a_perm.copy(), p_score
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (shuffle) = {best_score:.6f}")
            if p_score < cur_score:
                a, max_conv, cur_score = a_perm, p_max, p_score

        iteration += 1

    # --------------------------------------------------------------
    # 4) Return the best solution (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())   # final verification
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(v) for v in best]
