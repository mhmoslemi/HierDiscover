import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D vector.")
    # shift by minimal value to ensure non‑negativity not required yet
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w

def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv

def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum == 1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())

def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] for a given active set of lags."""
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad

def _exp_smooth_grad(a: np.ndarray, p: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the exponential smooth approximation of ``max(conv)``.
    The loss is (1/p)·log Σ_k exp(p·conv_k).  Its gradient equals
       2 * ( w ⋆ a )
    where w_k = softmax(p·conv_k).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    max_c = conv.max()
    # compute softmax weights in a numerically stable way
    w = np.exp(p * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    L = fft_len
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    offset = len(w) - 1
    grad = 2.0 * corr[offset: offset + n]
    return grad

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Heuristic search for a non‑negative step‑function minimising
    ``evaluate_sequence``.

    1.  Initialise many parametric families (uniform,
        symmetric exponential, symmetric power‑law, edge blocks,
        sparse Dirichlet) on a few problem sizes.
        The best among them serves as a starting point.
    2.  Refine the starter with an Adam optimiser that minimises an
        exponential smooth‑max of the convolution (a tight surrogate of
        the L∞ norm).  The softness parameter `p` grows during the run.
    3.  Finish with cheap stochastic mass‑transfer moves that greedily
        lower the exact objective.

    The routine respects the time budget and returns the best sequence
    found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # 1) Determine a sensible default length and possibly re‑use a prior
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.sum() > 0.0:
            n_default = prev.size
        else:
            n_default = 4096
    else:
        n_default = 4096

    # we will test a handful of lengths – the list is deliberately short
    lengths = [max(256, n_default // 2), n_default, min(8192, n_default * 2)]

    best_a = None
    best_score = np.inf

    # 2) Coarse parametric search (very cheap)
    # ------------------------------------------------------------------

# Helper utilities (top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum = 1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a∗a)[k]`` for the set ``active`` of
    indices attaining the current maximum.  For each active lag k,
    the contribution to ∂max/∂a_i is a_{k-i} (counted twice because the
    convolution is symmetric).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0
    return grad


def _exp_smooth_grad(a: np.ndarray, p: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the exponential smooth‑max surrogate of ``max(conv)``.
    The surrogate loss is (1/p)·log Σ_k exp(p·conv_k).
    Its gradient equals 2·(w ⋆ a) where w is the soft‑max of p·conv.
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(p * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    L = fft_len
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    offset = len(w) - 1            # 2n‑2
    grad = 2.0 * corr[offset: offset + n]
    return grad
