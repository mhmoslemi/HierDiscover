"""
propose_candidate
=================

A two‑phase optimiser for the “max‑convolution” score.

*Phase 1 – Smooth‑max Adam optimisation*  
  The non‑smooth objective `max(conv)` is replaced by a smooth
  approximation
      φ_β(a) = (1/β)·log Σ_i exp(β·(a∗a)_i) .
  For a given β the gradient can be written in closed form as a
  convolution with a weight vector that is proportional to the
  soft‑max of the current convolution.  By parametrising the
  probabilities with a soft‑max (`a = softmax(z)`) we obtain an
  unconstrained Adam update on the logits `z`.  β is increased
  slowly (continuation) so that the optimisation first sees a
  strongly smoothed landscape and then gradually focuses on the true
  maximum.

*Phase 2 – Simulated‑annealing refinement*  
  Starting from the best point found by Adam we perform a classic
  Metropolis‑type local search that repeatedly transfers a small
  fraction of mass between two positions (or a short block).  The
  temperature is exponentially cooled; accepted moves are either
  improving or accepted with probability exp(−Δ/T).  This phase can
  escape shallow local minima that Adam may miss.

Both phases work directly with the linear convolution computed by
FFT, which makes each iteration O(n log n) and fast enough to run
thousands of iterations within the 1000 s budget.  The routine uses
the global variable ``height_sequence_1`` as a warm start when it
exists; otherwise it starts from a random Dirichlet vector.

The returned list is normalised (sum≈1) and therefore the evaluation
score equals `2·n·max(conv)`.  The function prints progress messages
and finally returns the best sequence found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft


# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    e_sum = np.sum(e)
    if e_sum == 0.0:
        # fallback – uniform distribution
        return np.full_like(z, 1.0 / z.size)
    return e / e_sum


def _conv_full_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``x`` and ``y`` using FFT.
    Returns the full convolution of length ``len(x)+len(y)-1``.
    """
    X = rfft(x, n=fft_len)
    Y = rfft(y, n=fft_len)
    conv_full = irfft(X * Y, n=fft_len)
    return conv_full[: x.size + y.size - 1]


def _smooth_max_and_grad(
    a: np.ndarray,
    beta: float,
    fft_len_conv: int,
    fft_len_grad: int,
) -> tuple[float, np.ndarray, float]:
    """
    For a probability vector ``a`` (sum=1) compute

        φ_β(a) = (1/β)·log Σ exp(β·(a∗a)_i)   (smooth max)
        grad_a = ∂φ_β/∂a   (same shape as ``a``)

    Returns (smooth_max, grad_a, max_conv) where ``max_conv`` is the
    exact maximum of the (unsmoothed) convolution and is used for
    reporting the true score ``2·n·max_conv``.
    """
    n = a.size

    # --- convolution a*a -------------------------------------------------
    conv_a = _conv_full_fft(a, a, fft_len_conv)          # length 2·n‑1

    # smooth max (log‑sum‑exp)
    # shift for numerical stability
    conv_shift = conv_a - np.max(conv_a)
    exp_beta = np.exp(beta * conv_shift)
    sum_exp = np.sum(exp_beta)
    smooth_max = (np.log(sum_exp) / beta) + np.max(conv_a)

    # probability weights for the gradient
    w = exp_beta / sum_exp                               # length 2·n‑1, sum=1

    # --- gradient wrt a ---------------------------------------------------
    a_rev = a[::-1]
    conv_w_rev = _conv_full_fft(w, a_rev, fft_len_grad)  # length 3·n‑2

    # slice the middle n entries (indices n‑1 … n‑1+n‑1)
    grad_a = 2.0 * conv_w_rev[n - 1 : n - 1 + n]

    # exact max of a*a (needed for the true evaluation score)
    max_conv = np.max(conv_a)

    return smooth_max, grad_a, max_conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w

    """

    # 1) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------

    _, _, max_conv0 = _smooth_max_and_grad(a, beta=1.0,
                                           fft_len_conv=fft_len_conv,
                                           fft_len_grad=fft_len_grad)
    best_score = 2.0 * n * max_conv0
    best = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    target_score = 1.5025

    print(f"[0.00s] initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 2) Phase 1 – smooth‑max Adam optimisation (≈60 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.60 * (deadline - start_time)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # learning rate for the logits
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    # continuation parameter β
    beta = 1.0
    beta_max = 80.0
    beta_mul = 1.001    # geometric increase per iteration

    it = 0
    while time.monotonic() < phase1_end:
        # ----- smooth max and gradient wrt a -----
        smooth_val, grad_a, max_conv = _smooth_max_and_grad(
            a, beta, fft_len_conv, fft_len_grad
        )
        score = 2.0 * n * max_conv          # true objective (no smoothness)

        # ----- gradient wrt logits -----
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)         # ∂L/∂z = J_softmax·grad_a

        # optional gradient clipping (helps stability)
        g_norm = np.linalg.norm(grad_z)
        clip_thr = 0.5
        if g_norm > clip_thr:
            grad_z = grad_z * (clip_thr / g_norm)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z_update = lr * m_hat / (np.sqrt(v_hat) + eps)
        # Work in log‑space: we maintain logits directly
        # Recover current logits from a (softmax), then apply update
        # Instead of storing logits, we invert the softmax
        # Numerical trick: store logits as log_a (up to an additive constant)
        log_a = np.log(a + 1e-15)            # any constant shift cancels in softmax
        log_a -= z_update
        a = _softmax(log_a)                  # re‑normalise to the simplex

        # ----- bookkeeping -----
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] new best (Adam, it={it}) = {best_score:.6f}"
            )
            if best_score <= target_score:
                print("[early stop] target reached in Adam phase")
                break

        # ----- β continuation -----
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # occasional progress report
        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] Adam it={it}, beta={beta:.2f}, best={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 3) Phase 2 – simulated‑annealing local refinement (remaining time)
    # ------------------------------------------------------------------
    a = best.copy()
    curr_score = best_score

    # SA hyper‑parameters
    T = 0.02                # initial temperature
    decay = 0.9999          # exponential cooling
    it = 0

    while time.monotonic() < deadline:
        # ---- propose a tiny mass transfer between two positions ----
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-15:
            continue
        # random fraction in [0.001, 0.04] (log‑uniform)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.04)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # re‑project onto the simplex (cheap because only two entries changed)
        cand = _project_onto_simplex(cand)

        # evaluate candidate (exact max‑conv)
        _, _, max_conv_cand = _smooth_max_and_grad(
            cand, beta=beta_max, fft_len_conv=fft_len_conv, fft_len_grad=fft_len_grad
        )
        cand_score = 2.0 * n * max_conv_cand
        Δ = cand_score - curr_score

        # Metropolis acceptance
        if Δ < 0 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(
                    f"[{elapsed:6.2f}s] new best (SA, it={it}) = {best_score:.6f}"
                )
                if best_score <= target_score:
                    print("[early stop] target reached in SA phase")
                    break

        # occasional jitter – scale a short block to re‑stir the distribution
        if it % 7000 == 0 and it > 0:
            width = rng.integers(1, max(2, n // 80))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.02))
            block = a.copy()
            block[start:start + width] *= factor
            a = _project_onto_simplex(block)

        # cooling
        T *= decay
        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA it={it}, T={T:.3e}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final output (clip to official limits, compute final score)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # safety for the evaluator
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(
        f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}"
    )
    return [float(x) for x in best]
