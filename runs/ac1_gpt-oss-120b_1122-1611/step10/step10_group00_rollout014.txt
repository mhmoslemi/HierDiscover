"""
Hybrid sub‑gradient + stochastic pair‑exchange optimiser.

* Works directly on the simplex (∑a = 1, a_i ≥ 0) – the objective
  J(a) = 2·n·max(conv(a,a)) is invariant under scaling.
* Two complementary move families are used:
    – *Gradient move*: additive projected sub‑gradient descent with a
      momentum term (the sub‑gradient is the same as in the reference
      implementation, i.e. the sum of shifted copies of the current
      vector over the active set of the convolution).
    – *Exchange move*: pick two indices (i, j) and transfer a random
      amount of mass from i to j.  This explores directions that are
      invisible to the linearised sub‑gradient (large‑scale re‑balancing).
* An exponential cooling schedule controls a Metropolis acceptance
  rule; the temperature starts at the current best value and decays to
  1e‑8.  This lets the search escape shallow plateaus while still
  converging.
* Stagnation (no improvement for 20 s) triggers a short random jitter
  centred on the best solution, helping the walk to leave a local
  minimum.
* The algorithm adapts the step size (∝ 1/√t) and the exchange mass
  (∝ current ‑ average weight) over time.
* Finally the best vector is clipped to the allowed range [0, 1000] and
  re‑normalised before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal (or v is zero); return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Return the objective J = 2·n·max(conv(a,a)) for a vector a that
    already lives on the simplex (sum(a) == 1).
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgradient_active_set(a: np.ndarray,
                           conv: np.ndarray,
                           active_tol: float) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv) at the current active set.
    The result is a vector of the same length as a.
    """
    n = a.size
    max_conv = conv.max()
    active = np.where(conv >= max_conv - active_tol)[0]

    grad = np.zeros_like(a)
    # each active position contributes a shifted copy of a
    for p in active:
        start = max(0, p - (n - 1))
        end = min(p, n - 1)
        idx = np.arange(start, end + 1)
        # contribution 2 * a_{p‑idx}
        grad[idx] += 2.0 * a[p - idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------

def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects the 1000 s time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – use the existing construction if available,
    #    otherwise fall back to a moderate size (2048 ≤ n ≤ 8192).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial points (all lie on the simplex).
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – moderate and strong concentration
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous best (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate (cheap objective evaluation).
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = compute_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best  J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation loop (projected sub‑gradient + exchange moves + SA).
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep 5 s safety margin

    # Temperature schedule (exponential cooling)
    T_start = best_J                # start as high as the current best
    T_end = 1e-8
    total_time = deadline - start_time

    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    last_improve = start_time
    iteration = 0
    momentum = np.zeros_like(cur_vec)   # for the additive sub‑gradient step
    beta = 0.9                         # momentum decay

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # exponential cooling:   T = T_start * (T_end/T_start)^(elapsed/total_time)
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # -----------------------------------------------------------------
        # 4.1  Stagnation handling – small jitter after 20 s without improvement
        # -----------------------------------------------------------------
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = compute_J(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – best J = {best_J:.6f}")

        # -----------------------------------------------------------------
        # 4.2  Decide which move to propose
        # -----------------------------------------------------------------
        if rng.random() < 0.65:          # gradient‑type move (majority)
            # recompute convolution and its active set
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            max_conv = cur_conv.max()
            tol = max(1e-12, max_conv * 1e-12)
            grad = subgradient_active_set(cur_vec, cur_conv, tol)

            # momentum update
            momentum = beta * momentum + (1.0 - beta) * grad

            # step size decays like 1/√t
            step = 0.6 / np.sqrt(iteration + 1.0)

            prop = cur_vec - step * momentum
        else:
            # exchange move – pick two indices and shift a small amount
            i, j = rng.choice(n, size=2, replace=False)
            # transfer a random fraction of the source mass
            delta = min(cur_vec[i], rng.random() * 0.01)   # at most 1 % of mass
            prop = cur_vec.copy()
            prop[i] -= delta
            prop[j] += delta

        # -----------------------------------------------------------------
        # 4.3  Project onto simplex and evaluate
        # -----------------------------------------------------------------
        prop = proj_simplex(prop)
        prop_J = compute_J(prop, n, fft_len, conv_len)

        # -----------------------------------------------------------------
        # 4.4  Metropolis acceptance
        # -----------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # probability of accepting a worse move
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = prop
            cur_J = prop_J
            # record global best
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
