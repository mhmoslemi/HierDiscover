"""
Gradient‑smooth simulated annealing for minimizing the auto‑correlation
peak of a non‑negative step function.
The algorithm performs a projected gradient descent on a smooth
approximation of the max‑convolution (log‑sum‑exp), together with a
temperature‑controlled Metropolis acceptance rule and occasional
random jitter / restart.  It works directly on the probability simplex
so the objective 2·n·max(conv) (sum=1) can be evaluated cheaply with FFT.
"""

import time
import numpy as np
from scipy.special import logsumexp

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= s
    return w

def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv

            jitter = a.copy()
            jitter[start_idx:start_idx + block_len] *= factor
            jitter = _project_onto_simplex(jitter)
            jitter_conv = _conv_fft(jitter, fft_len)
            jitter_obj = 2.0 * n * jitter_conv.max()
            if jitter_obj < best_obj - 1e-12:

    print(f"Finished in {total_elapsed:.2f}s – final evaluated score = {final_score:.6f}")

        simplex,
      * a linear cooling schedule for Metropolis acceptance (simulated
        annealing),
      * occasional random jitter on a small block,
      * a stagnation‑triggered full restart with a fresh Dirichlet sample.

    The best sequence seen before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` (sum = 1) – use the previous best if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
        a = a0.copy()
    else:
        n = 4096                         # power of two – convenient for FFT
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(3 * n)           # length sufficient for all linear convolutions

    # --------------------------------------------------------------
    # 1) Timing and bookkeeping
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    cur_conv = _conv_fft(a, fft_len)
    cur_obj = 2.0 * n * cur_conv.max()   # exact objective for sum(a)=1
    best = a.copy()
    best_obj = cur_obj
    print(f"[0.00s] initial objective = {best_obj:.6f} (n={n})")

    # --------------------------------------------------------------
    # 2) Hyper‑parameters
    # --------------------------------------------------------------
    iteration = 0
    T0 = best_obj * 0.08                 # initial temperature (≈ 8 % of the score)
    Tf = 1e-9 * best_obj                  # final temperature
    kappa_start = 0.5                     # start of log‑sum‑exp smoothing
    kappa_end = 2000.0                    # end of smoothing (approaches true max)
    lr0 = 0.30                            # initial learning‑rate base
    max_no_improve = 15000                # after this many steps we restart
    last_improve_iter = 0

    # --------------------------------------------------------------
    # 3) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # --- a) current FFT of a -------------------------------------------------
        a_fft = np.fft.rfft(a, n=fft_len)

        # --- b) self‑convolution -------------------------------------------------
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # --- c) smoothing schedule ------------------------------------------------
        elapsed = time.monotonic() - start_time
        time_frac = elapsed / (deadline - start_time)   # in [0,1]
        kappa = kappa_start * (kappa_end / kappa_start) ** time_frac

        # --- d) soft‑max weights for the smoothed max ---------------------------
        scaled = kappa * conv
        lse = logsumexp(scaled)        # stable log‑sum‑exp
        w = np.exp(scaled - lse)       # size 2n‑1, sums to 1

        # --- e) gradient of the smoothed objective -------------------------------
        # correlation a ⋆ w = Σ_j a_j w_{j+ℓ}
        w_rev = w[::-1]                # reverse for correlation via convolution
        w_fft = np.fft.rfft(w_rev, n=fft_len)
        corr_full = np.fft.irfft(a_fft * w_fft, n=fft_len)

        N = conv.size                  # = 2n‑1
        corr = corr_full[N - 1 - np.arange(n)]   # correlation values for ℓ = 0 … n‑1
        grad = 2.0 * corr              # ∇ (smooth max)

        # --- f) learning‑rate schedule -----------------------------------------
        lr = lr0 * (1.0 - time_frac) + 0.001 * time_frac

        # --- g) projected gradient step -----------------------------------------
        cand = a - lr * grad
        cand = _project_onto_simplex(cand)

        # --- h) cheap evaluation of the candidate --------------------------------
        cand_conv = _conv_fft(cand, fft_len)
        cand_obj = 2.0 * n * cand_conv.max()

        # --- i) Metropolis acceptance -------------------------------------------
        temp = T0 * (1.0 - time_frac) + Tf * time_frac
        accepted = False
        if cand_obj < cur_obj - 1e-15:
            accepted = True
        elif temp > 0.0:
            prob = np.exp(-(cand_obj - cur_obj) / temp)
            if rng.random() < prob:
                accepted = True

        if accepted:
            a = cand
            cur_obj = cand_obj
            cur_conv = cand_conv

        # --- j) keep the globally best solution ----------------------------------
        if cand_obj < best_obj - 1e-12:
            best = cand.copy()
            best_obj = cand_obj
            last_improve_iter = iteration
            print(f"[{elapsed:6.2f}s] new best = {best_obj:.6f} (iter={iteration})")
            if best_obj <= 1.5025:
                print("[early stop] target reached")
                break

        # --- k) occasional jitter (helps escape shallow basins) -----------------
        if iteration % 2000 == 0 and iteration > 0:
            block_len = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - block_len + 1))
            factor = np.exp(rng.normal(0.0, 0.020))
            jitter = a.copy()
            jitter[start_idx:start_idx + block_len] *= factor
            jitter = _project_onto_simplex(jitter)
            jitter_conv = _conv_fft(jitter, fft_len)
            jitter_obj = 2.0 * n * jitter_conv.max()
            if jitter_obj < best_obj - 1e-12:
                best, best_obj = jitter.copy(), jitter_obj
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (jitter) = {best_obj:.6f}")
            if jitter_obj < cur_obj + 1e-8:
                a, cur_obj, cur_conv = jitter, jitter_obj, jitter_conv

        # --- l) stagnation‑driven restart ---------------------------------------
        if iteration - last_improve_iter > max_no_improve:
            a = rng.dirichlet(np.ones(n)).astype(np.float64)
            cur_obj = 2.0 * n * _conv_fft(a, fft_len).max()
            last_improve_iter = iteration
            print(f"[{time.monotonic() - start_time:6.2f}s] restart after stagnation (iter={iteration})")

        iteration += 1

    # --------------------------------------------------------------
    # 4) Return the best sequence (apply evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final evaluated score = {final_score:.6f}")
    return [float(x) for x in best]
