"""
Hybrid optimizer for the max‑convolution objective.

The algorithm consists of four stages inside the fixed time budget
(≈ 1000 s).  It starts from the last known good construction (if any)
or from a uniform distribution, then

1. **Global random search** – many cheap candidates are drawn from a few
   parametrised families (Dirichlet on bins, power‑law, exponential).  The
   best candidate is kept.

2. **Smooth‑max gradient descent** – we replace the non‑smooth max by a
   log‑sum‑exp with a temperature that slowly decreases.  The exact
   gradient of the smooth surrogate is obtained by one extra FFT
   convolution, giving an O(n log n) update.  An exponentiated‑gradient
   (mirror‑descent) step is performed and the vector is projected back
   onto the probability simplex.  After each step we enforce mirror
   symmetry (a[i]=a[n‑1‑i]) because the optimal solution is symmetric.

3. **Simulated annealing** – small random mass‑transfer moves are
   accepted with a Metropolis probability that gradually cools.  This
   helps to escape shallow local minima that the gradient phase may miss.

4. **Gradient‑guided redistribution** – using the true sub‑gradient of the
   hard max (the active convolution indices) we move a modest fraction of
   mass from the most “dangerous’’ coordinates to the safest ones.
   This often reduces the peak of the convolution further.

All phases keep track of the best sequence seen so far and finally
return it (clipped to the evaluator limits).  The routine respects the
5 s safety margin required by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / len(v)
        else:
            w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT, length 2*len(a)-1."""
    a_f = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_f * a_f, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two (possibly different) 1‑D arrays via FFT."""
    xf = np.fft.rfft(x, n=fft_len)
    yf = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(xf * yf, n=fft_len)
    return conv[: x.size + y.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max_k (a*a)[k] for a set of active indices."""
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)                # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry v[i]=v[n‑1‑i] and renormalise to sum 1."""
    v = (v + v[::-1]) / 2.0
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a non‑negative step‑function that minimises evaluate_sequence."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Starting point and problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() > 0.0:
            a0 /= a0.sum()
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
    else:
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(3 * n)                     # enough for all convolutions
    # --------------------------------------------------------------
    # Helper to evaluate the (hard) objective quickly
    # --------------------------------------------------------------
    def _hard_score(v: np.ndarray) -> float:
        conv = _conv_fft(v, fft_len)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    best = a0.copy()
    best_score = _hard_score(best)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – random exploration (≈15 % of the time)
    # --------------------------------------------------------------
    phase0_end = start + 0.15 * (deadline - start)
    it_rand = 0
    while time.monotonic() < phase0_end:
        scheme = rng.random()
        if scheme < 0.45:                    # Dirichlet on coarse bins
            K = max(8, n // 80)               # ~50 – 500 bins depending on n
            block = n // K
            alpha = rng.uniform(0.2, 2.0)
            bins = rng.dirichlet(np.full(K, alpha))
            if rng.random() < 0.5:
                bins = np.sort(bins)[::-1]    # optional monotone ordering
            seq = np.repeat(bins, block)
            if seq.size < n:
                seq = np.concatenate([seq, bins[: n - seq.size]])
        elif scheme < 0.80:                  # power‑law shapes
            p = rng.uniform(0.05, 3.0)
            idx = np.arange(1, n + 1, dtype=np.float64)
            seq = idx ** (-p)
            seq /= seq.sum()
        else:                                 # exponential decay
            lam = rng.uniform(5e-4, 2e-2)
            idx = np.arange(n, dtype=np.float64)
            seq = np.exp(-lam * idx)
            seq /= seq.sum()

        seq = _symmetrize(seq)                # enforce a[i]=a[n-1-i]
        cur = _hard_score(seq)
        if cur < best_score:
            best_score = cur
            best = seq.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (rand) = {best_score:.6f} (iter={it_rand})")
        it_rand += 1
        if it_rand % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] random search it={it_rand}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (≈40 % of the time)
    # --------------------------------------------------------------
    a = best.copy()
    phase1_end = start + 0.55 * (deadline - start)
    it = 0
    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()

        # temperature for the log‑sum‑exp surrogate (shrinks with √it)
        T = max_b * max(0.02, 0.5 / np.sqrt(it + 1.0))

        # softmax weights (numerically stable)
        w = np.exp((conv - max_b) / T)          # shape: 2n‑1
        sum_w = w.sum()

        # gradient of the smooth max:
        #   grad = 2 * (w * a_rev)  / sum_w   (where * denotes convolution)
        a_rev = a[::-1]
        conv_w = _conv_fft_two(w, a_rev, fft_len)   # length 3n‑2
        grad = 2.0 * conv_w[:n][::-1] / sum_w       # shape (n,)

        # exponentiated‑gradient step
        eta = 0.7 / np.sqrt(it + 1.0)
        log_a = np.log(a + 1e-15) - eta * grad
        a = np.exp(log_a)
        a /= a.sum()
        a = _symmetrize(a)

        # evaluate the hard objective
        cur = _hard_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (smooth‑grad) = {best_score:.6f} (iter={it})")
            if best_score <= 1.5025:
                print("[early stop] target reached in smooth‑grad")
                break
        it += 1
        if it % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] smooth‑grad iter={it}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated annealing (≈25 % of the time)
    # --------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    phase2_end = start + 0.80 * (deadline - start)
    it = 0
    SA_T = cur_score * 0.05                     # initial temperature
    while time.monotonic() < phase2_end:
        # pick a source proportionally to its mass (dense parts move more)
        src = rng.choice(n, p=a)
        tgt = int(rng.integers(0, n - 1))
        if tgt == src:
            tgt = (tgt + 1) % n

        # transfer a small random fraction of the source mass
        max_frac = min(0.2, a[src])            # never move more than 20 % of source
        frac = rng.uniform(0.001, max_frac)
        delta = a[src] * frac

        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta
        a_cand = _symmetrize(a_cand)

        cand_score = _hard_score(a_cand)
        delta_score = cand_score - cur_score
        if delta_score <= 0 or rng.random() < np.exp(-delta_score / SA_T):
            a = a_cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f} (iter={it})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in SA")
                    break

        # cool down slowly
        SA_T = max(SA_T * 0.9999, 1e-8)
        it += 1
        if it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] SA iter={it}, T={SA_T:.2e}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – gradient‑guided redistribution (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    phase3_end = deadline
    it = 0
    while time.monotonic() < phase3_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-8
        active = np.where(conv >= max_b - eps)[0]
        if active.size > 30:
            active = rng.choice(active, size=30, replace=False)

        # hard sub‑gradient for the active set
        grad_h = _gradient_from_active(a, active)

        # indices with largest / smallest gradient components
        n_hi = max(1, n // 200)          # ≈ 0.5 % of the coordinates
        hi_idx = np.argsort(-grad_h)[:n_hi]
        lo_idx = np.argsort(grad_h)[:n_hi]

        mass_hi = a[hi_idx].sum()
        if mass_hi == 0.0:
            break

        # move a modest fraction of the “dangerous’’ mass to the safest places
        transfer = mass_hi * 0.12               # 12 % of the high‑gradient mass
        if transfer > 0.0:
            dec = transfer * (a[hi_idx] / mass_hi)   # take proportionally
            a[hi_idx] -= dec
            inc = transfer / lo_idx.size
            a[lo_idx] += inc

            a = _project_onto_simplex(a)
            a = _symmetrize(a)

            cur = _hard_score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] new best (redistrib) = {best_score:.6f} (iter={it})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in redistribution")
                    break
        it += 1
        if it % 200 == 0:
            elapsed = time.monotonic() - start
            # occasional progress report (no heavy printing)
            pass

    # --------------------------------------------------------------
    # Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
