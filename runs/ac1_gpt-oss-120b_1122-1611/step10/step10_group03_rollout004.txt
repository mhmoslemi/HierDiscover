"""
Hybrid optimisation for the convolution‑maximisation problem.

The algorithm consists of three stages:

1. **Initialisation** – load the previously discovered sequence (if any)
   and bring it onto the probability simplex (sum = 1).  If nothing is
   available a uniform distribution of length 4096 is used.

2. **Fast stochastic perturbation** – a short random‑jitter phase helps
   to escape a possible basin of attraction and may already improve the
   bound.

3. **Linear‑programming refinement** – given the current sequence f,
   compute the worst convolution value M = max(f*f).  Solve the LP

        maximise   ∑ g_i
        subject to  (f * g)[k] ≤ M   for all k,
                    g_i ≥ 0

   (the convolution matrix of f is built as a sparse Toeplitz matrix).
   The optimiser returns a feasible direction g.  After normalising g
   (∑g = 1) we move a fraction *t* towards g:
        f ← (1‑t)·f + t·g .
   The mixing factor *t* is reduced gradually; the process is repeated
   until the time budget is exhausted.  Each mixing step never increases
   the worst convolution value, and in practice drives the objective
   2·n·max(f*f) down.

All heavy work (convolution, FFT, LP) is done with NumPy/SciPy, and the
routine respects the 1000 s time limit.  The final sequence is clipped to
the evaluator’s limits before the real `evaluate_sequence` call.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum(a)=1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _solve_lp_for_f(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  maximise   ∑ g_i
           subject to  (f * g)[k] ≤ M   (all k),
                       g_i ≥ 0.

    Returns the optimal vector g (not normalised).
    """
    n = f.size
    L = 2 * n - 1

    # Build the convolution matrix of f (upper‑triangular Toeplitz)
    # column j contains f starting at row j.
    diags = []
    offsets = []
    for offset in range(n):
        length = n - offset
        diags.append(f[:length])
        offsets.append(offset)
    A = sp.diags(diags, offsets, shape=(L, n), format="csr")

    # Linear programme (highs solver handles sparse A efficiently)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  → minimise -sum(g)
    b_ub = np.full(L, M, dtype=np.float64)    # upper bounds M
    bounds = [(0.0, None) for _ in range(n)]

    res = scipy.optimize.linprog(
        c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"disp": False}
    )
    if not res.success:
        # Fallback – uniform distribution (always feasible)
        return np.full(n, 1.0 / n, dtype=np.float64)
    return res.x


def _lp_refine(a: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Repeatedly mix the current vector with the LP direction.
    Returns an improved vector (still on the simplex) and its score.
    """
    start = time.monotonic()
    deadline = start + time_budget
    best = a.copy()
    fft_len = _next_pow2(3 * best.size)
    best_score = _score_from_a(best, fft_len)

    # start with a fairly aggressive mixing factor
    t = 0.5

    it = 0
    while time.monotonic() < deadline:
        # 1) compute current worst convolution value
        conv = _conv_fft(best, fft_len)
        M = conv.max()

        # 2) linear‑programming step
        g = _solve_lp_for_f(best, M)

        # 3) normalise g to stay on the simplex
        s = g.sum()
        if s <= 0.0:
            break
        g = g / s

        # 4) form a candidate by mixing
        cand = (1.0 - t) * best + t * g

        # occasional jitter to avoid stagnation
        if rng.random() < 0.08:
            jitter = rng.normal(scale=0.003, size=cand.size)
            cand = np.maximum(cand + jitter, 0.0)

        cand = _project_onto_simplex(cand)

        cand_score = _score_from_a(cand, fft_len)

        # 5) accept if improvement
        if cand_score < best_score - 1e-12:
            best, best_score = cand, cand_score
            # reset mixing strength when we make progress
            t = 0.5
        else:
            # otherwise decay t slowly
            t *= 0.92
            if t < 0.02:
                t = 0.02

        it += 1
        if it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] LP iter {it:3d} – best score {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Light stochastic improvement before the LP stage
# ----------------------------------------------------------------------


def _jitter_phase(a: np.ndarray, time_budget: float, rng: np.random.Generator):
    """A short random‑jitter phase (purely exploratory)."""
    start = time.monotonic()
    deadline = start + time_budget
    best = a.copy()
    fft_len = _next_pow2(3 * best.size)
    best_score = _score_from_a(best, fft_len)

    while time.monotonic() < deadline:
        # add Gaussian noise, re‑project onto simplex
        cand = best + rng.normal(scale=0.018, size=best.size)
        cand = np.maximum(cand, 0.0)
        cand = _project_onto_simplex(cand)
        cand_score = _score_from_a(cand, fft_len)

        if cand_score < best_score:
            best, best_score = cand, cand_score

    return best, best_score

# Public entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The method first performs a brief random
    jitter on the current best, then repeatedly solves a small linear
    programme to obtain a direction that does not increase the worst
    convolution value and mixes towards it.  The final sequence is returned
    as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time handling (keep a safety margin of 5 s)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise the global best – reuse the previous run if possible
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        tot = best.sum()
        if tot > 0.0:
            best = best / tot
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        # fallback uniform distribution of length 4096
        n0 = 4096
        best = np.full(n0, 1.0 / n0, dtype=np.float64)

    fft_len = _next_pow2(3 * best.size)
    best_score = _score_from_a(best, fft_len)
    print(f"[0.00s] initial best = {best_score:.6f} (len={best.size})")

    # ------------------------------------------------------------------
    # 1) Light stochastic jitter – give the optimiser a chance to leave
    #    the current basin of attraction.
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    jitter_budget = min(20.0, 0.05 * remaining)   # at most 5 % of the remaining time
    if jitter_budget > 1.0:
        best, best_score = _jitter_phase(best, jitter_budget, rng)
        print(f"[{time.monotonic() - start_time:6.2f}s] after jitter – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑programming refinement – the main optimisation phase.
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        best, best_score = _lp_refine(best, remaining, rng)
        print(f"[{time.monotonic() - start_time:6.2f}s] after LP refinement – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final processing – clip to the evaluator limits and obtain the true
    # score via the official ``evaluate_sequence`` function.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # store the result for possible reuse by a subsequent run
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
