"""
Simulated‑annealing optimizer for the max‑convolution objective.

The evaluator is
    2·n·max(conv(a,a)) / (sum(a)**2)
but because both numerator and denominator scale quadratically with a,
the objective depends only on the *shape* of a (its L1‑normalised form).
The algorithm works directly on the L1‑simplex: we repeatedly transfer a
random fraction of mass from a source index to a target index and accept
the move according to a decreasing temperature schedule (Metropolis rule).
The convolution update is performed incrementally (O(n) per move) so that
hundreds of thousands of proposals fit easily into the 1000 s budget.
Whenever a new best value is found it is printed; the run stops early
as soon as the target value 1.5025 is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns full length 2*len(a)-1 array.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    The optimiser is a simulated annealing loop with cheap incremental
    convolution updates.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Initialise a – Dirichlet(0.5) gives a fairly diverse start.
    # ------------------------------------------------------------------
    n = 4096                       # length of the vector (power of two)
    a = rng.gamma(shape=0.5, size=n)   # heavy‑tailed positive numbers
    a /= a.sum()                       # L1‑normalise → sum == 1

    # full convolution (used only for the initial score)
    fft_len = _next_pow2(3 * n)
    conv = _conv_fft(a, fft_len)        # length 2n‑1
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    print(f"[0.00s] initial score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Simulated‑annealing loop (incremental convolution updates)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # temperature drops linearly from a fraction of the initial score
    init_temp = 0.02 * cur_score
    it = 0
    while time.monotonic() < deadline:
        # ----- pick source and target -------------------------------------------------
        src = int(rng.integers(0, n))
        if a[src] <= 0.0:          # nothing to move – try again next iteration
            continue
        tgt = int(rng.integers(0, n))
        if tgt == src:
            continue

        # ----- how much mass to relocate ------------------------------------------------
        # log‑uniform between 0.001 and 0.2 of the source weight
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.2)))
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        # ----- incremental convolution update -----------------------------------------
        # conv' = conv + 2·(a*Δ) + Δ*Δ  where Δ[i]=‑δ, Δ[j]=+δ
        conv_new = conv.copy()

        # contribution of a*Δ :  -δ·a shifted by src  +δ·a shifted by tgt
        # shifting is done with slice views – indices 0 … 2n‑2 are safe
        conv_new[src: src + n] -= 2.0 * delta * a
        conv_new[tgt: tgt + n] += 2.0 * delta * a

        # Δ*Δ has at most three non‑zero entries
        idx_2src = 2 * src
        idx_2tgt = 2 * tgt
        idx_src_tgt = src + tgt
        conv_new[idx_2src] += delta * delta
        conv_new[idx_2tgt] += delta * delta
        conv_new[idx_src_tgt] -= 2.0 * delta * delta

        # ----- evaluate the proposal ----------------------------------------------------
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # ----- temperature (linear cooling) --------------------------------------------
        elapsed = time.monotonic() - start_time
        t_frac = elapsed / (deadline - start_time)
        temperature = init_temp * (1.0 - t_frac)

        # ----- Metropolis acceptance ----------------------------------------------------
        accept = False
        if new_score < cur_score - 1e-12:
            accept = True
        else:
            if temperature > 1e-12:
                prob = np.exp(-(new_score - cur_score) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            # commit move
            a[src] -= delta
            a[tgt] += delta
            conv = conv_new
            cur_score = new_score

            # keep global best
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = a.copy()
                print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached")
                    break

        it += 1
        # occasional progress report
        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] iter={it}, cur={cur_score:.6f}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Final evaluation (protect against evaluator clipping)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total = time.monotonic() - start_time
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best]
