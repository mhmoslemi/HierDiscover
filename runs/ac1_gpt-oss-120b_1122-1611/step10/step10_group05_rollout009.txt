    rng = np.random.default_rng(seed)
    n = 4096 # or maybe 5000? Use power of two.

    deadline = start + max(0.0, budget_s - 5.0)

    lr0 = 10.0

            best_score = score
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] new best {best_score:.6f} (iter {iter})")
            if best_score <= 1.5025:
                print("target reached")
                break
        
        # schedule
        t = time.monotonic() - start
        progress = (t) / (deadline - start)
        lr = lr0 * (1 - progress) + lr_min * progress
        beta = beta0 + (beta_max - beta0) * progress
        
        # active indices
        if top_k >= cur_conv.size:
            active_idx = np.arange(cur_conv.size)
        else:
            # get top_k highest conv
            active_idx = np.argpartition(-cur_conv, top_k)[:top_k]
        
        # compute gradient (sum contributions)
        grad = np.zeros_like(cur)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # indices i = i_start ... i_end
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = k - idx_i
            # each contribution is 2 * a[idx_j]
            grad[idx_i] += 2.0 * cur[idx_j]
        # Apply multiplicative update

        log_a = np.log(cur + eps) - lr * grad
        cand = np.exp(log_a)
        # project onto simplex

"""
propose_candidate()
-------------------

Search for a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------

1. **Arc‑sine (U‑shaped) start** –  
   The continuous density 1/(π√(x(1‑x))) on [0,1] has the property that the
   sum of two independent copies is uniformly distributed on [0,2].
   Discretising this density yields a sequence whose self‑convolution is
   already close to flat, i.e. its maximal entry is much smaller than the
   uniform (triangular) case.

2. **Projected exponential gradient descent** –  
   Repeatedly smooth‑max the convolution with a temperature β,
   compute the (soft) gradient of the weighted sum of convolution entries,
   perform a multiplicative (exponential) update
   ``a ← a·exp(−η·grad)`` and project back onto the probability simplex.
   The temperature β is increased linearly (more focus on the true maximum)
   while the step size η is decreased linearly (to stabilise near the optimum).

3. **Random jitter** – every few thousand iterations a tiny log‑normal
   jitter is applied to avoid stagnation.

All updates keep the vector on the simplex, guaranteeing that the
objective value ``2·n·max(conv)`` is well defined and comparable with the
reference implementation.

The routine runs for the whole allotted time budget (≤ 1000 s) and returns
the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)‑1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a 1‑D vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) (the objective when the vector is normalised to sum=1)."""
    return 2.0 * n * conv.max()

# Main search routine
# ----------------------------------------------------------------------

    # --------------------------------------------------------------
    n = 4096                     # can be increased if time permits
    fft_len = _next_pow2(3 * n)  # length sufficient for a full linear convolution

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    # x in (0,1) – centre of each bin

    a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-12)          # arcsine shape (U‑shaped)
    a = a / a.sum()                                   # project onto simplex

    # --------------------------------------------------------------

    max_iter = 0

    beta_end = 200.0              # final temperature (focuses on the true max)
