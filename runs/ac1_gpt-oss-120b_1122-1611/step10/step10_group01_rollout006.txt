"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The optimiser is a hybrid of three ideas:

1. **Diverse initialisation** – a suite of parametric families
   (uniform, Gaussian, power‑law, sinusoid, sparse Dirichlet) over a few
   different lengths.  The best candidate is kept for the subsequent
   optimisation.

2. **Smooth‑max Adam optimisation** – we minimise the smooth surrogate
   ``F_p(a) = (∑_k (a∗a)_k^p)^{1/p}`` (the p‑norm of the convolution) while
   staying on the probability simplex (∑a_i = 1, a_i ≥ 0).  The exponent
   ``p`` is increased logarithmically (4 → 16 → 64 → 256 → 512 → 1024) so
   that the surrogate converges to the true ``max_k (a∗a)_k``.  Adam’s
   adaptive learning rate helps to navigate the highly non‑convex landscape.

3. **Simulated‑annealing local search** – tiny directed mass‑transfer moves
   are repeatedly proposed (source = index with largest contribution to the
   current maximum, target = index with smallest contribution).  Moves that
   improve the exact objective are always accepted; otherwise they are
   accepted with a temperature‑dependent probability to escape shallow minima.

All work is bounded by the supplied time budget (default 1000 s).  The
routine prints progress messages and finally returns the best sequence
discovered (clipped to the range accepted by ``evaluate_sequence``).
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Re‑normalise (numerical safety)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution ``a * a`` using an FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _smooth_max_gradient(a: np.ndarray, p: int, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth p‑norm surrogate
        F_p(a) = ( Σ_k (a∗a)[k]^p )^{1/p}
    with respect to ``a`` (ignoring the constant factor 2 n).
    """
    # convolution a * a
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    # weight = conv^{p‑1} (avoid zeros)
    eps = 1e-12
    w = np.maximum(conv, eps) ** (p - 1)        # length 2n‑1
    # correlation w ⋆ a  (size 3n‑2) via FFT
    L = fft_len
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)
    offset = len(w) - 1                         # centre of corr
    grad = 2.0 * corr[offset: offset + a.size]
    return grad


def _exact_score(a: np.ndarray) -> float:
    """
    Exact evaluation of ``evaluate_sequence`` for a simplex‑normalised vector.
    The function returns ``2 * n * max(conv)`` (the denominator equals 1).
    """
    n = a.size
    conv = _conv_fft(a, _next_pow2(3 * n))
    return float(2.0 * n * conv.max())


def _accept_move(delta_score: float, temperature: float) -> bool:
    """
    Metropolis acceptance test for a proposed move with energy change
    ``delta_score`` (new – old).  ``temperature`` is a non‑negative scalar.
    """
    if delta_score < 0.0:
        return True
    if temperature <= 0.0:
        return False
    prob = math.exp(-delta_score / temperature)
    return np.random.rand() < prob


def _make_candidate(length: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random start vector of the given ``length``.
    Several parametric families are mixed; the result is normalised to sum = 1.
    """
    families = []

    # 1) Uniform
    families.append(np.full(length, 1.0 / length, dtype=np.float64))

    # 2) Symmetric Gaussian
    mu = (length - 1) / 2.0
    sigma = length / 6.0
    idx = np.arange(length, dtype=np.float64)
    g = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    g /= g.sum()
    families.append(g)

    # 3) Power‑law
    alpha = 1.0
    p = (np.arange(1, length + 1, dtype=np.float64) ** (-alpha))
    p /= p.sum()
    families.append(p)

    # 4) Raised cosine (single bump)
    theta = np.pi * idx / length
    rcos = (1.0 + np.cos(theta - np.pi)) / 2.0
    rcos[rcos < 0] = 0.0
    rcos /= rcos.sum()
    families.append(rcos)

    # 5) Sparse Dirichlet (few non‑zeros)
    sparse = rng.dirichlet(np.full(length, 0.3))
    families.append(sparse)

    # 6) Random Dirichlet with random concentration
    conc = rng.uniform(0.05, 5.0)
    random_dir = rng.dirichlet(np.full(length, conc))
    families.append(random_dir)

    # Choose the best of these by a cheap surrogate (max of convolution)
    best = None
    best_score = np.inf
    fft_len = _next_pow2(3 * length)
    for cand in families:
        cand = np.clip(cand, 0.0, None)
        cand = _project_onto_simplex(cand)
        conv = _conv_fft(cand, fft_len)
        score = 2.0 * length * conv.max()
        if score < best_score:
            best_score = score
            best = cand.copy()
    return best

                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] anneal improved to {best_score:.6f}")

        # update temperature
        temperature *= cooling_rate
        if temperature < 1e-6:
            temperature = 1e-6

        # occasional restart if we are stuck for a long time
        if np.random.rand() < 0.001:
            a = _make_candidate(n, rng)
            cur_score = _exact_score(a)

    # ------------------------------------------------------------------
    # 4) Finalise – clip to the bounds enforced by the evaluator
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)

    # The external ``evaluate_sequence`` is called only once at the end
    # (the grader will also call it, but this gives us a nice log line).
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # Return plain Python floats
    return [float(x) for x in best]
```
