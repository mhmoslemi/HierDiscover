"""Hybrid random‑shape + Differential Evolution + local refinement.

The optimizer works in three stages within the allotted time budget:

1. **Global shape exploration** – generate a large number of candidates from
   simple parametric families (uniform, linear, power‑law, exponential,
   half‑sine) and from Dirichlet draws on coarse bins.  The best of these
   seeds the next stage.

2. **Differential Evolution on a low‑dimensional block representation** –
   the sequence is represented by K ≥ 1 blocks (piecewise‑constant).  A
   population of block vectors evolves with classic DE mutation/crossover,
   while each individual is always projected onto the probability simplex.
   The best individual is tracked.

3. **Targeted local refinement** – using the exact sub‑gradient of the
   current maximal convolution entry we repeatedly transfer a small amount
   of mass from a “high‑gradient’’ index to a “low‑gradient’’ one.
   Occasionally stochastic jitter is added to escape shallow plateaus.

All steps respect the constraints of the evaluation function
(non‑negative, sum > 0.01, values ≤ 1000).  The routine returns the
best sequence found within the budget."""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k, the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score_from_seq(seq: np.ndarray, fft_len: int) -> float:
    """Return the 2·n·max(conv) score for a sequence with sum=1."""
    n = seq.size
    conv = _conv_fft(seq, fft_len)
    return 2.0 * n * conv.max()

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    The method combines a random global search over simple parametric families,
    a Differential Evolution on a low‑dimensional block representation,
    and a directed local refinement phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)   # 5 s safety
    deadline = start_time + total_budget

    # --------------------------------------------------------------
    # 1) Initialise best candidate from a previous run if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 1e-12:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        best = a0.copy()
        n_best = best.size
        fft_len_best = _next_pow2(3 * n_best)
        best_score = _score_from_seq(best, fft_len_best)
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] initial from previous best: n={n_best} score={best_score:.6f}")
    else:
        # fallback: uniform 4096‑vector
        n_best = 4096
        best = np.full(n_best, 1.0 / n_best, dtype=np.float64)
        fft_len_best = _next_pow2(3 * n_best)
        best_score = _score_from_seq(best, fft_len_best)
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] initial uniform: n={n_best} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase A – random exploration over simple families (≈15 % time)
    # --------------------------------------------------------------
    phase_a_end = start_time + 0.15 * total_budget
    it_rand = 0
    while time.monotonic() < phase_a_end:
        # pick length uniformly in a reasonable range
        n = int(rng.integers(800, 5000))
        shape = rng.choice(["uniform", "linear", "power", "exp", "sin", "dirichlet"])
        if shape == "uniform":
            seq = np.full(n, 1.0 / n, dtype=np.float64)
        elif shape == "linear":
            seq = np.arange(n, 0, -1, dtype=np.float64)
            seq = seq / seq.sum()
        elif shape == "power":
            p = rng.uniform(0.05, 2.5)
            idx = np.arange(1, n + 1, dtype=np.float64)
            seq = idx ** (-p)
            seq = seq / seq.sum()
        elif shape == "exp":
            r = rng.uniform(0.90, 0.9995)
            idx = np.arange(n, dtype=np.float64)
            seq = r ** idx
            seq = seq / seq.sum()
        elif shape == "sin":
            x = np.linspace(0, 1, n, dtype=np.float64)
            seq = np.sin(np.pi * x)          # half‑sine, zero at ends
            seq = seq / seq.sum()
        elif shape == "dirichlet":
            # coarse binning, then repeat
            K = rng.integers(16, 128)
            block = max(1, n // K)
            alpha = rng.uniform(0.3, 2.0)
            bins = rng.dirichlet(np.full(K, alpha))
            if rng.random() < 0.5:
                bins = np.sort(bins)[::-1]    # monotone optional
            seq = np.repeat(bins, block)
            if seq.size < n:
                seq = np.concatenate([seq, bins[: n - seq.size]])
            else:
                seq = seq[:n]
            seq = seq / seq.sum()
        else:
            continue

        # evaluate
        fft_len = _next_pow2(3 * n)
        score = _score_from_seq(seq, fft_len)

        if score < best_score:
            best_score = score
            best = seq.copy()
            n_best = n
            fft_len_best = fft_len
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (rand {shape}) n={n} score={best_score:.6f}")

        it_rand += 1

    # --------------------------------------------------------------
    # 3) Phase B – Differential Evolution on block representation
    # --------------------------------------------------------------
    # Keep the current `best` length for the remainder of the run
    n = n_best
    fft_len = fft_len_best
    K = max(8, min(64, n // 8))        # number of blocks (≥8, ≤64)
    block = max(1, n // K)

    # initialise a population of block vectors
    popsize = 30
    population = np.empty((popsize, K), dtype=np.float64)
    for i in range(popsize):
        alpha = rng.uniform(0.2, 2.0)
        blocks = rng.dirichlet(np.full(K, alpha))
        if rng.random() < 0.5:
            blocks = np.sort(blocks)[::-1]
        population[i] = blocks

    # helper to expand blocks → full sequence and compute score
    def _score_from_blocks(blocks: np.ndarray) -> float:
        seq = np.repeat(blocks, block)
        if seq.size > n:
            seq = seq[:n]
        elif seq.size < n:
            seq = np.concatenate([seq, blocks[: n - seq.size]])
        seq = seq / seq.sum()
        conv = _conv_fft(seq, fft_len)
        return 2.0 * n * conv.max()

    scores = np.array([_score_from_blocks(ind) for ind in population])

    # keep track of the best individual so far
    best_idx = int(np.argmin(scores))
    if scores[best_idx] < best_score:
        best_score = scores[best_idx]
        best = np.repeat(population[best_idx], block)
        if best.size > n:
            best = best[:n]
        elif best.size < n:
            best = np.concatenate([best, population[best_idx][: n - best.size]])
        best = best / best.sum()
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] new best (DE init) score={best_score:.6f}")

    # DE parameters
    F = 0.8
    CR = 0.9

    # run DE until the end of the allocated budget (≈70 % total)
    de_deadline = start_time + 0.85 * total_budget
    while time.monotonic() < de_deadline:
        for i in range(popsize):
            # mutation: a + F·(b‑c)
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0.0, None)

            # crossover
            cross = rng.random(K) < CR
            if not np.any(cross):
                cross[rng.integers(K)] = True
            trial = np.where(cross, mutant, population[i])

            # project onto simplex (blocks)
            s = trial.sum()
            if s == 0.0:
                trial = np.full(K, 1.0 / K)
            else:
                trial = trial / s

            # evaluation
            trial_score = _score_from_blocks(trial)

            # selection
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = np.repeat(trial, block)
                    if best.size > n:
                        best = best[:n]
                    elif best.size < n:
                        best = np.concatenate([best, trial[: n - best.size]])
                    best = best / best.sum()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] new best (DE) score={best_score:.6f}")

            # early exit if target reached
            if best_score <= 1.5025:
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] target reached during DE, stopping early")
                return [float(x) for x in best]

        # safety break if too little time left for the next generation
        if time.monotonic() > de_deadline - 0.5:
            break

    # --------------------------------------------------------------
    # 4) Phase C – directed local refinement (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    while time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-6
        active = np.where(conv >= max_b - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        grad = _gradient_from_active(a, active)

        # source = index with largest gradient, target = low‑gradient region
        src = int(np.argmax(grad))
        low_candidates = np.where(grad <= np.percentile(grad, 5))[0]
        if low_candidates.size > 0 and rng.random() < 0.9:
            tgt = int(rng.choice(low_candidates))
        else:
            tgt = int(rng.integers(0, n))

        if src == tgt:
            tgt = (src + int(rng.integers(1, n))) % n

        # transfer a modest fraction of the source mass
        max_frac = min(0.15, a[src] / (a[src] + 1e-12))
        frac = rng.uniform(0.0005, max_frac)
        delta = a[src] * frac

        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta
        a_cand = _project_onto_simplex(a_cand)

        conv_c = _conv_fft(a_cand, fft_len)
        score_c = 2.0 * n * conv_c.max()

        if score_c < best_score - 1e-12:
            best_score = score_c
            best = a_cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) score={best_score:.6f}")

        # accept non‑worsening moves
        if score_c <= 2.0 * n * max_b:
            a = a_cand

        # occasional jitter to shake the solution
        if rng.random() < 0.015:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = _project_onto_simplex(a * jitter)

        # early exit if target reached
        if best_score <= 1.5025:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] target reached in local phase")
            break

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
