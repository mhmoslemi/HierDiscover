"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises

    2 * n * max(a * a) / (sum(a)**2)

where “*” denotes linear convolution.  

The algorithm is based on a *binary mask* approach:

* Choose a length n and a target number s of active positions.
* Put a uniform weight 1/s on the s active indices (all other entries are 0).
* For a binary mask the convolution values are simply the integer
  representation‑counts of the sums i+j of two active indices.
  The objective becomes 2·n·(max_count / s²).
* By picking the active set at random and then performing a short
  stochastic hill‑climbing (swap an active / inactive pair) we can
  drive the maximal count much below the uniform‑vector bound (≈1/n)
  and therefore obtain scores well under the previous 1.513 … result.

The search runs many independent restarts (different lengths,
different densities, different random seeds) until the allotted
time budget (≈ 1000 s) is exhausted.  The best sequence found is
returned and also stored in the global variable ``height_sequence_1``
for possible reuse by a later run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """
    Exact objective for a simplex‑normalised vector (sum(a)=1).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core search – binary‑mask + hill climbing
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises the
    evaluator ``evaluate_sequence``.  The strategy is:

    1.  Pick a length *n* (several candidates are tried).
    2.  Choose a random subset of *s* indices (the “active set”) and
        assign weight 1/s to those positions – the vector therefore lies
        on the probability simplex.
    3.  Evaluate the objective quickly via FFT.
    4.  Perform a short stochastic hill‑climbing phase that repeatedly
        swaps one active and one inactive index; accept a swap if it
        improves the score (or with a tiny annealing probability).
    5.  Keep the globally best vector across many random restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Global best
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – a mix of powers of two and the length that
    # gave the previous best (3809).  Larger lengths give a finer
    # granularity but also a larger search space.
    # ------------------------------------------------------------------
    length_pool = [1024, 2048, 3072, 3809, 4096, 6144, 8192]

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # Pick a length and a density for the active set
        # --------------------------------------------------------------
        n = int(rng.choice(length_pool))
        frac = rng.uniform(0.30, 0.70)          # active fraction (30‑70 %)

        # --------------------------------------------------------------

            elapsed = time.monotonic() - start_time

        # --------------------------------------------------------------

    total_elapsed = time.monotonic() - start_time

    # make the sequence available for a potential next run

    return [float(x) for x in best]
