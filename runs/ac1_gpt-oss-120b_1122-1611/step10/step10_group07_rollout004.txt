def propose_candidate(seed=12345, budget_s=1000.0):

   rng = np.random.default_rng(seed)

   # Determine length n:
   if 'height_sequence_1' in globals():
        a0 = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0: a0 /= s
   else:
        n = 4096
        a0 = np.full(n, 1.0/n)

   n = a0.shape[0]
   fft_len = _next_pow2(3*n)

   # Helper to compute score:
   def score_of(v):
         conv = _conv_fft(v, fft_len)
         return 2.0 * n * conv.max()

   # Best tracking:
   best = a0.copy()
   best_score = score_of(best)
   print(f"[0.00s] init score {best_score:.6f}")

   start = time.monotonic()
   deadline = start + max(0.0, budget_s - 5.0)

   # Phase 0: random exploration (20% time)
   phase0_end = start + 0.20 * (deadline - start)
   it_rand = 0
   while time.monotonic() < phase0_end:
         choice = rng.random()
         if choice < 0.4:  # power law
               p = rng.uniform(0.01, 5.0)
               seq = np.arange(1, n+1, dtype=np.float64) ** (-p)
               # optionally symmetrize
               if rng.random() < 0.5:
                   seq = (seq[: (n+1)//2])
                   seq = np.concatenate([seq, seq[-2::-1]]) if n%2 else np.concatenate([seq, seq[::-1]])
               seq = seq / seq.sum()
         elif choice < 0.7: # Dirichlet on bins
               K = max(8, n//64)
               block = n // K
               alpha = rng.uniform(0.01, 0.5)
               bins = rng.dirichlet(np.full(K, alpha))
               # optionally monotone decreasing
               if rng.random() < 0.6:
                    bins = np.sort(bins)[::-1]
               seq = np.repeat(bins, block)
               if seq.size < n:
                    seq = np.concatenate([seq, bins[: n - seq.size]])
               seq = seq / seq.sum()
         else:   # sparse support
               M = rng.integers(2, min(200, n//2))
               support = rng.choice(n, size=M, replace=False)
               weights = rng.dirichlet(np.ones(M))
               seq = np.zeros(n)
               seq[support] = weights
               seq /= seq.sum()
         # compute score
         sc = score_of(seq)
         if sc < best_score:
               best_score = sc
               best = seq.copy()
               elapsed = time.monotonic() - start
               print(f"[{elapsed:6.2f}s] new best (rand) {best_score:.6f} iter={it_rand}")
         it_rand += 1
         if it_rand % 2000 == 0:
               elapsed = time.monotonic() - start
               print(f"[{elapsed:6.2f}s] random iter={it_rand} best={best_score:.6f}")

   # Phase 1: gradient descent (exponentiated) with symmetry (35% time)
   a = best.copy()
   phase1_end = start + 0.55 * (deadline - start)
   it_gd = 0
   while time.monotonic() < phase1_end:
         conv = _conv_fft(a, fft_len)
         max_b = conv.max()
         eps = max_b * 1e-7
         active = np.where(conv >= max_b - eps)[0]
         # compute gradient
         grad = _gradient_from_active(a, active)
         # step size schedule
         eta = 0.5 / np.sqrt(it_gd + 1.0)   # moderate
         # multiplicative update using exponentiated gradient
         log_a = np.log(a + 1e-15)
         log_a -= eta * grad
         a = np.exp(log_a)
         a /= a.sum()
         # enforce symmetry
         a = _symmetrize(a)
         # compute new score
         sc = score_of(a)
         if sc < best_score - 1e-12:
               best_score = sc
               best = a.copy()
               elapsed = time.monotonic() - start
               print(f"[{elapsed:6.2f}s] new best (exp) {best_score:.6f} iter={it_gd}")
               if best_score <= 1.5025:

                    break

         it_gd += 1

   # Phase 2: stochastic local refinement (rest)

"""
Multi‑phase stochastic optimiser for the max‑convolution objective.

The optimiser works in three stages:

0. Diverse random seeding (power‑law, Dirichlet on coarse bins, sparse support).
   Every candidate is symmetrised (a_i = a_{n‑1‑i}) and projected onto the
   probability simplex (sum = 1, a_i ≥ 0).

1. Mirror‑enforced exponentiated‑gradient descent.
   The exact sub‑gradient of the active convolution indices is used.
   After each multiplicative update the vector is projected onto the simplex
   and symmetrised again.

2. Simulated‑annealing local refinement.
   Small directed mass‑transfer moves (source → target) are proposed;
   improvements are always accepted, otherwise a Metropolis step with a
   slowly cooling temperature allows occasional uphill moves.  Periodic jitter
   helps escape plateaus.

All operations respect the time budget (budget_s – 5 s safety margin).
The best sequence encountered is returned as a list of floats.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # make a copy to avoid modifying the input
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.full_like(v, 1.0 / len(v))
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise (numerical safety)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _symmetrize_and_project(v: np.ndarray) -> np.ndarray:
    """
    Enforce the mirror symmetry v[i] = v[n‑1‑i] and project onto simplex.
    """
    v_sym = (v + v[::-1]) * 0.5
    return _project_onto_simplex(v_sym)


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set ``active`` (indices attaining
    the current maximum).  For each active index k, the contribution to
    ∂max/∂a_i is a_{k‑i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Stochastic search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0. Initialise -------------------------------------------------
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        # default length – 4096 works well on the reference platform
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    n = a0.size
    fft_len = _next_pow2(3 * n)   # enough for linear convolution

    def _score(vec: np.ndarray) -> float:
        """Internal cheap score = 2·n·max(a*a).  Sum(vec) is assumed = 1."""
        conv = _conv_fft(vec, fft_len)
        return 2.0 * n * conv.max()

    best = _symmetrize_and_project(a0.copy())
    best_score = _score(best)
    print(f"[0.00s] initial score = {best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # --------------------------------------------------------------
    # 1. Random exploration (≈20 % of budget)
    # --------------------------------------------------------------
    phase0_end = start_time + 0.20 * (deadline - start_time)
    it_rand = 0
    while time.monotonic() < phase0_end:
        choice = rng.random()
        if choice < 0.35:                     # power‑law shapes
            p = rng.uniform(0.01, 5.0)
            idx = np.arange(1, n + 1, dtype=np.float64)
            seq = idx ** (-p)
        elif choice < 0.70:                   # Dirichlet on coarse bins
            K = max(8, n // 64)
            block = n // K
            alpha = rng.uniform(0.01, 0.5)
            bins = rng.dirichlet(np.full(K, alpha))
            if rng.random() < 0.5:            # optional monotone ordering
                bins = np.sort(bins)[::-1]
            seq = np.repeat(bins, block)
            if seq.size < n:                  # fill the tail if needed
                seq = np.concatenate([seq, bins[: n - seq.size]])
        else:                                 # sparse support
            M = rng.integers(2, min(200, n // 2))
            supp = rng.choice(n, size=M, replace=False)
            weights = rng.dirichlet(np.ones(M))
            seq = np.zeros(n, dtype=np.float64)
            seq[supp] = weights

        seq = np.maximum(seq, 0.0)
        seq = seq / seq.sum()
        seq = _symmetrize_and_project(seq)

        sc = _score(seq)
        if sc < best_score:
            best_score = sc
            best = seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (rand) = {best_score:.6f} (iter={it_rand})")
        it_rand += 1
        if it_rand % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] random iter={it_rand}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Exponentiated‑gradient descent with symmetry (≈35 % of budget)
    # --------------------------------------------------------------
    a = best.copy()
    phase1_end = start_time + 0.55 * (deadline - start_time)
    it_exp = 0
    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-7
        active = np.where(conv >= max_b - eps)[0]          # usually a handful

        grad = _gradient_from_active(a, active)

        # diminishing multiplicative step
        eta = 0.6 / math.sqrt(it_exp + 1.0)
        log_a = np.log(a + 1e-15)
        log_a -= eta * grad
        a = np.exp(log_a)
        a /= a.sum()
        a = _symmetrize_and_project(a)

        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (exp‑grad) = {best_score:.6f} (iter={it_exp})")
            if best_score <= 1.5025:
                print("[early stop] target reached in exponentiated phase")
                break
        # keep the move if it does not worsen the objective
        if sc <= 2.0 * n * max_b + 1e-12:
            pass
        else:
            # revert to previous best (conservative)
            a = best.copy()
        it_exp += 1
        if it_exp % 300 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] exp‑grad iter={it_exp}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Simulated‑annealing local refinement (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    it_loc = 0
    while time.monotonic() < deadline:
        # gradient at the currently most violated shift
        conv = _conv_fft(a, fft_len)
        idx_max = int(np.argmax(conv))
        g_max = _gradient_from_active(a, np.array([idx_max]))
        src = int(np.argmax(g_max))

        # choose a low‑gradient target
        low = np.where(g_max <= g_max.min() + 1e-12)[0]
        if low.size > 1 and rng.random() < 0.8:
            tgt = int(rng.choice(low))
        else:
            tgt = int(rng.integers(0, n))

        if src == tgt:
            tgt = int((src + rng.integers(1, n)) % n)

        # directed mass transfer
        frac = np.exp(rng.uniform(np.log(0.0005), np.log(0.02)))   # 0.05 %–2 %
        delta = a[src] * frac
        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[tgt] += delta
        a_cand = _symmetrize_and_project(a_cand)

        sc_cand = _score(a_cand)

        # Metropolis acceptance
        if sc_cand < cur_score - 1e-12:
            cur_score = sc_cand
            a = a_cand
            if sc_cand < best_score - 1e-12:
                best_score = sc_cand
                best = a_cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (move={it_loc})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in local phase")
                    break
        else:
            # temperature schedule – slower cooling for long runs
            T = 0.02 * math.exp(-it_loc / 20000.0)
            if rng.random() < math.exp(-(sc_cand - cur_score) / max(T, 1e-9)):
                a = a_cand
                cur_score = sc_cand

        # occasional jitter to shake the solution out of plateaus
        if it_loc % 4000 == 0 and it_loc > 0:
            width = max(1, n // 80)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _symmetrize_and_project(a_jit)

        it_loc += 1
        if it_loc % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local iter={it_loc}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Finalise ---------------------------------------------------
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # respect evaluator limits
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(v) for v in best]
