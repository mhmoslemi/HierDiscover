Simulated‑Annealing + Gradient‑Guided refinement.

The algorithm starts from the best known step‑function (if the global variable
`height_sequence_1` exists) or from a random monotone Dirichlet draw.
At each iteration it:

1. Computes the linear convolution a*a using a cached FFT length.

sequence.  All operations respect the simplex constraint (∑a_i = 1) and the
non‑negativity requirement.

The routine runs for the allotted budget (default 1000 s, with a 5 s safety
margin) and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len``.
    Returns the convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Return max value of the linear convolution a*a (computed via FFT)."""
    conv = _conv_fft(a, fft_len)
    return float(conv.max())


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k the contribution to ∂max/∂a_i is a_{k-i}
    whenever the index exists.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    The method combines random restarts, a gradient‑guided simulated‑annealing
    loop and occasional jitter.  It respects the simplex constraint (∑a=1)
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` – use a known good construction if available,
    #    otherwise start from a random monotone Dirichlet draw.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
    else:
        # default length – a power of two makes the FFT cheap
        n = 4096
        # start from a Dirichlet draw with a decreasing shape
        K = 128
        alpha = rng.uniform(0.3, 2.0)
        bins = rng.dirichlet(np.full(K, alpha))
        bins = np.sort(bins)[::-1]                 # enforce monotone decay
        block = n // K
        a0 = np.repeat(bins, block)[:n]
        a0 /= a0.sum()                             # normalise to sum = 1

    fft_len = _next_pow2(2 * n)   # enough for linear convolution

    # --------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    a = a0.copy()
    cur_score = 2.0 * n * _conv_max_fft(a, fft_len)
    best = a.copy()
    best_score = cur_score
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Helper to compute score quickly
    # --------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        """Return the evaluation score (without the final evaluate_sequence call)."""
        return 2.0 * n * _conv_max_fft(vec, fft_len)

    # --------------------------------------------------------------
    # 4) Simulated‑annealing loop (gradient‑guided moves)
    # --------------------------------------------------------------
    it = 0
    # initial temperature – a fraction of the current gap to the trivial bound (2.0)
    T0 = max(0.02 * (2.0 - cur_score), 0.001)
    T = T0
    while time.monotonic() < deadline:
        # ---- compute convolution and active set ---------------------------------
        conv_full = _conv_fft(a, fft_len)
        max_b = float(conv_full.max())
        eps = max_b * 1e-7                     # tolerance for “active” indices
        active = np.where(conv_full >= max_b - eps)[0]

        # ---- exact sub‑gradient of the max‑convolution -------------------------
        grad = _gradient_from_active(a, active)

        # ---- pick source (largest gradient) and target (smallest gradient) -----
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            # fallback: random distinct pair
            src = int(rng.integers(0, n))
            tgt = int(rng.integers(0, n - 1))
            if tgt >= src:
                tgt += 1

        # ---- propose a directed mass transfer ----------------------------------
        # the transferred amount is a small fraction of the source mass;
        # the fraction itself is annealed (larger when temperature is high)
        frac = float(rng.uniform(0.001, 0.02)) * (T / T0 + 0.1)
        delta = a[src] * frac
        # ensure we never make a component negative
        delta = min(delta, a[src])

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta
        # (sum is preserved, but numerical noise may appear)
        a_candidate = _project_onto_simplex(a_candidate)

        # ---- evaluate the candidate --------------------------------------------
        cand_score = _score(a_candidate)
        delta_obj = cand_score - cur_score

        # ---- Metropolis acceptance ------------------------------------------------
        accept = False
        if delta_obj < 0:
            accept = True
        else:
            prob = np.exp(-delta_obj / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (anneal) = {best_score:.6f}")

        # ---- temperature schedule -------------------------------------------------
        it += 1
        T = T0 * (0.9995 ** it)      # exponential cooling

        # ---- occasional large random restart to avoid being trapped -------------
        if it % 8000 == 0:
            K = 128
            alpha = rng.uniform(0.3, 2.0)
            bins = rng.dirichlet(np.full(K, alpha))
            if rng.random() < 0.7:
                bins = np.sort(bins)[::-1]          # monotone decay (most useful)
            block = n // K
            a_restart = np.repeat(bins, block)[:n]
            a_restart = a_restart / a_restart.sum()
            a = a_restart
            cur_score = _score(a)

        # ---- early exit if we already beat the target ---------------------------
        if best_score <= 1.5025:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] target reached – stopping early")
            break

    # --------------------------------------------------------------
    # 5) Final evaluation using the provided evaluate_sequence (checks)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # Clip to evaluator limits (just in case) and return as plain Python floats
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
