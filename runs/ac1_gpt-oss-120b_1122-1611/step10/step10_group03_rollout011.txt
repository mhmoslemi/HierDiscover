"""
Improved optimiser for the step‑function task.

The algorithm works in three phases for each restart:

1.  *Smooth max* gradient descent – the “max” in the objective is replaced
    by a soft‑max (log‑sum‑exp) with a temperature β that is raised slowly.
    The gradient of this smooth objective is obtained by a single
    FFT‑based convolution, giving a cheap dense update.
    A multiplicative (exponentiated‑gradient) step keeps the vector non‑negative,
    followed by a simple normalisation onto the simplex.

2.  *Exact max* mirror‑descent – a few indices where the true convolution is
    maximal are sampled (top‑k).  The exact sub‑gradient for those indices
    is built analytically and an exponentiated‑gradient step is performed.
    This sharpens the solution once it is already close to a plateau.

3.  *Stochastic refinement* – directed mass‑transfer moves together with a
    tiny simulated‑annealing acceptance rule explore the neighbourhood of the
    best solution found so far.

Restarts are performed with a variety of lengths (powers of two up to 8192)
and with several different initial shapes (uniform, arcsine, sine, beta
distributions, and sparse Dirichlet draws).  The best sequence seen across
all runs is finally passed to ``evaluate_sequence`` and returned.

The implementation avoids expensive per‑iteration projections: after a
multiplicative update the vector is simply renormalised, which for a
non‑negative vector is the Euclidean projection onto the probability simplex.
All convolutions are performed with FFTs (real‑valued) for speed.

The total running time is limited to ``budget_s‑5`` seconds (a 5 s safety
margin) – the optimiser will stop early if the target bound (≈1.5025) is
reached.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog  # not used currently but kept for future extensions

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        v[:] = 1.0 / v.size
    return v


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_general(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two arbitrary real vectors using FFT.
    Returns an array of length len(x) + len(y) - 1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    Z = X * Y
    conv = np.fft.irfft(Z, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set ``active`` of indices
    where the current maximum (or a near‑maximum) is attained.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)               # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a vector with sum(a)=1."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _smooth_step(a: np.ndarray, fft_len: int, beta: float) -> tuple:
    """
    Compute convolution, smooth‑max weight vector, and the gradient of the
    smooth objective 2·n·Sβ(conv) where Sβ is the soft‑max with temperature β.
    Returns (conv, max_conv, grad) where grad has shape (n,).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)               # length 2n‑1
    max_conv = conv.max()
    # stable soft‑max weights
    shifted = conv - max_conv
    exp_vals = np.exp(beta * shifted)
    w = exp_vals / exp_vals.sum()              # length 2n‑1, sums to 1

    # gradient: 4·n·(w ⋆ a_rev)  (cross‑correlation)
    a_rev = a[::-1]
    grad_full = _conv_fft_general(w, a_rev, fft_len)   # length 3n‑2
    grad = grad_full[n - 1: n - 1 + n]                # centre slice, length n
    grad = 4.0 * n * grad
    return conv, max_conv, grad


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------


def _initial_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random non‑negative vector of length n (sum = 1)."""
    choice = rng.choice(
        ["uniform", "arcsine", "sine", "beta", "dirichlet", "sparse_dirichlet"]
    )
    if choice == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)

    elif choice == "arcsine":
        i = np.arange(n, dtype=np.float64)
        # arcsine density ∝ 1/√(x·(1‑x))  – discretised on the interval (0,1)
        raw = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a = raw / raw.sum()

    elif choice == "sine":
        i = np.arange(n, dtype=np.float64)
        raw = np.sin(np.pi * (i + 0.5) / n) + 1e-12
        a = raw / raw.sum()

    elif choice == "beta":
        # random Beta‑like shape
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        x = (np.arange(n) + 0.5) / n
        raw = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1) + 1e-12)
        a = raw / raw.sum()

    elif choice == "dirichlet":
        # moderately sparse Dirichlet
        alpha = rng.uniform(0.1, 1.0)
        a = rng.dirichlet(np.full(n, alpha)).astype(np.float64)

    else:   # sparse_dirichlet
        # very sparse Dirichlet (α < 1) to create few large spikes
        alpha = rng.uniform(0.01, 0.3)
        a = rng.dirichlet(np.full(n, alpha)).astype(np.float64)

    # small jitter to avoid exact zeros that can stall multiplicative updates
    jitter = rng.normal(scale=1e-4, size=n)
    a = np.maximum(a + jitter, 0.0)
    return _normalize(a)


# ----------------------------------------------------------------------
# Core optimisation for a single run (three phases)
# ----------------------------------------------------------------------


def _optimize_one_run(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    optimise a single start vector ``a0`` for at most ``time_budget`` seconds.
    Returns the best vector found and its exact score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)

    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    deadline = start + time_budget

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (≈40 % of the run time)
    # ------------------------------------------------------------------
    phase1_end = start + 0.40 * time_budget
    it = 0
    while time.monotonic() < phase1_end:
        # temperature β rises linearly from 1 to 50 across the phase
        elapsed = time.monotonic() - start
        beta = 1.0 + 49.0 * (elapsed / (phase1_end - start))
        conv, max_conv, grad = _smooth_step(a, fft_len, beta)

        # multiplicative exponentiated‑gradient step
        eta = 0.05 / (np.linalg.norm(grad) + 1e-12)
        factor = np.exp(-eta * grad)
        # protect against under‑/overflow
        factor = np.clip(factor, 1e-15, 1e15)
        a *= factor
        a = _normalize(a)

        cur_score = 2.0 * n * max_conv
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – exact max sub‑gradient (≈30 % of the run time)
    # ------------------------------------------------------------------
    phase2_end = start + 0.70 * time_budget
    it = 0
    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        # take a small active set (the 8 largest indices)
        top_k = np.argpartition(-conv, 8)[:8]
        grad = _gradient_from_active(a, top_k)

        eta = 0.6 / math.sqrt(it + 1.0)          # decaying learning rate
        factor = np.exp(-eta * grad)
        factor = np.clip(factor, 1e-15, 1e15)
        a *= factor
        a = _normalize(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – stochastic local refinement (remaining time)
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        # occasional full convolution for a fresh picture
        if it % 25 == 0:
            conv = _conv_fft(a, fft_len)
            max_idx = int(np.argmax(conv))
            # a heuristic “source” is the index that contributes most to the peak
            grad_peak = _gradient_from_active(a, np.array([max_idx]))
            src = int(np.argmax(grad_peak))
            # a “target” with the smallest influence
            tgt = int(np.argmin(grad_peak))
            if src == tgt:
                tgt = (tgt + 1) % n

        # propose a directed mass transfer (log‑uniform fraction)
        if a[src] <= 0.0:
            src = int(rng.integers(0, n))
        if src == tgt:
            tgt = int(rng.integers(0, n))

        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.12)))   # 0.1 % – 12 %
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _normalize(cand)

        cand_score = _score_from_a(cand, fft_len)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand.copy()
            a = cand
        else:
            # occasional simulated‑annealing accept
            temperature = max(1e-4,
                              (deadline - time.monotonic()) /
                              (deadline - start))
            prob = math.exp(-(cand_score - cur_score) / temperature) \
                if cand_score > cur_score else 1.0
            if rng.random() < prob:
                a = cand

        # gentle jitter to shake the solution
        if it % 3000 == 0 and it > 0:
            width = max(1, n // 140)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.015))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _normalize(a_jit)

        it += 1

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The routine performs many short random‑restart runs, each consisting of
    (i) smooth‑max gradient descent, (ii) exact‑max mirror descent and
    (iii) stochastic local refinement.  The best sequence seen is returned.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a 5 s safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise the global best – use a previous solution if available,
    # otherwise a uniform vector of length 4096.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        best = _normalize(best)
        n_best = best.size
    else:
        n_best = 4096
        best = np.full(n_best, 1.0 / n_best, dtype=np.float64)

    best_score = _score_from_a(best, _next_pow2(3 * n_best))
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # ------------------------------------------------------------------
    # Lengths we will explore (powers of two, plus a few odd values)
    # ------------------------------------------------------------------
    length_pool = [512, 1024, 2048, 3072, 4096, 6144, 8192]

    min_run_time = 18.0   # minimum seconds per restart

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Choose a length – bias towards the current best size
        if rng.random() < 0.55:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # ----------------------------------------------------------------
        # Build an initial point
        # ----------------------------------------------------------------
        if n == best.size:
            # jitter the current best a little
            a0 = best.copy()
            a0 += rng.normal(scale=0.008, size=n)
            a0 = _normalize(a0)
        else:
            a0 = _initial_shape(n, rng)

        # ----------------------------------------------------------------
        # Run a short optimisation on this start
        # ----------------------------------------------------------------
        remaining = deadline - time.monotonic()
        run_budget = min(min_run_time, remaining)
        if run_budget < 3.0:
            break

        a_opt, score_opt = _optimize_one_run(a0, run_budget, rng)

        # ----------------------------------------------------------------
        # Update the global best if improvement occurred
        # ----------------------------------------------------------------
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            # early exit when the target is clearly beaten
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # Final processing – evaluate with the official routine
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)   # evaluator caps values anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # expose the best sequence for possible subsequent runs
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
