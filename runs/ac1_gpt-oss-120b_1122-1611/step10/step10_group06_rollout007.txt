def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 2.0)
    target_score = 1.5025
    best_a = None
    best_score = np.inf
    restart = 0

    # Optionally include a warm start from height_sequence_1
    warm_start = None
    if "height_sequence_1" in globals():
        warm_start = np.clip(np.asarray(globals()["height_sequence_1"], dtype=np.float64), 0, None)
        warm_start = _simplex_normalise(warm_start)

    while time.monotonic() < deadline - 10.0:
        restart += 1
        # choose n based on random selection and remaining time
        # earlier restarts use moderate n; later maybe larger n.
        # Use possible n values:
        n_candidates = [4096, 8192, 16384]
        n = rng.choice(n_candidates)
        # initialize:
        if restart == 1 and warm_start is not None and warm_start.size == n:
            a = warm_start.copy()
        else:
            # Dirichlet with concentration parameter controlling sparsity
            alpha = rng.uniform(0.2, 1.0)  # random concentration
            a = rng.dirichlet(alpha=np.full(n, alpha))
        a = _simplex_normalise(a)
        fft_len = _next_pow2(3 * n)

        # Phase EG
        eg_iters = int(2000 * (1000.0 - (time.monotonic() - start)) / budget_s)  # proportionally allocate
        # Or set time bound: allocate up to ~0.4 of remaining time
        eg_deadline = min(deadline, time.monotonic() + 0.3 * (deadline - start))
        beta = 5.0
        momentum = np.zeros_like(a)
        it = 0
        while time.monotonic() < eg_deadline:
            a_fft = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
            max_c = conv.max()
            # softmax
            exp_vals = np.exp(beta * (conv - max_c))
            w = exp_vals / exp_vals.sum()
            w_fft = np.fft.rfft(w, n=fft_len)
            grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: n + w.size - 1]
            grad = grad_full[:n]

            # momentum update
            momentum = 0.9 * momentum + 0.1 * grad
            eta = 0.6 / math.sqrt(it + 1)
            a = a * np.exp(-eta * momentum)
            a = _simplex_normalise(a)

            # evaluate
            conv = _conv_self(a, fft_len)
            score = 2.0 * n * conv.max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                print(f"[{time.monotonic() - start:6.2f}s] Restart {restart} new best EG {best_score:.6f}")
                if best_score <= target_score:
                    print("[target reached]")
                    break

            # update beta gradually
            beta = min(beta * 1.0015, 250.0)
            it += 1
            if it % 2000 == 0:
                print(f"[{time.monotonic() - start:6.2f}s] Restart {restart} EG iter={it} beta={beta:.1f} best={best_score:.6f}")

        if best_score <= target_score:
            break

        # Phase SA (local refinement)
        sa_deadline = deadline
        temperature_start = 0.02 * best_score
        it_sa = 0
        while time.monotonic() < sa_deadline:
            elapsed = time.monotonic() - start
            fraction = elapsed / (deadline - start)
            temperature = temperature_start * (1.0 - fraction)

            conv = _conv_self(a, fft_len)
            max_idx = int(np.argmax(conv))
            grad_max = _gradient_for_index(a, max_idx)

            # Choose move type: with probability 0.7 continuous mass transfer, 0.3 swap
            if rng.random() < 0.7:
                # continuous mass transfer as before
                src = int(np.argmax(grad_max))
                tgt = int(np.argmin(grad_max))
                # random fraction
                frac = math.exp(rng.uniform(math.log(0.001), math.log(0.04)))
                delta = a[src] * frac
                cand = a.copy()
                cand[src] -= delta
                cand[tgt] += delta
            else:
                # swap two indices
                i, j = rng.integers(0, n, size=2)
                cand = a.copy()
                cand[i], cand[j] = cand[j], cand[i]

            cand = _simplex_normalise(cand)  # ensure tiny rounding errors fixed

            conv_cand = _conv_self(cand, fft_len)
            score_cand = 2.0 * n * conv_cand.max()
            curr_score = 2.0 * n * conv.max()

            # record best
            if score_cand < best_score - 1e-12:
                best_score = score_cand
                best_a = cand.copy()
                print(f"[{time.monotonic() - start:6.2f}s] Restart {restart} new best SA {best_score:.6f}")
                if best_score <= target_score:
                    break

            # metropolis acceptance
            if score_cand < curr_score or rng.random() < math.exp((curr_score - score_cand) / max(1e-12, temperature)):
                a = cand

            it_sa += 1
            if it_sa % 5000 == 0:
                print(f"[{time.monotonic() - start:6.2f}s] Restart {restart} SA iter={it_sa} best={best_score:.6f}")

        if best_score <= target_score:
            break

    # After loop, evaluate final best using evaluate_sequence
    if best_a is None:
        # fallback: uniform
        n = 4096
        best_a = np.full(n, 1.0 / n)
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    print(f"Finished – final score = {final_score:.6f}")
    return [float(x) for x in best]

Multi‑restart stochastic optimizer for the max‑convolution score.

The routine repeatedly creates a random probability vector (sum = 1) of a
chosen length, then improves it by three complementary phases:

1️⃣  Smoothed exponentiated‑gradient descent on a log‑sum‑exp surrogate
   of the max‑convolution.  A momentum term and a slowly increasing
   smoothing parameter β make the updates stable.

2️⃣  Targeted pairwise mass transfers.  For the current worst convolution
   entry we compute its exact gradient, pick a few source indices with
   large gradient and a few target indices with small gradient, and try
   relatively large transfers.  Any improvement updates the global best.

3️⃣  Simulated‑annealing refinement.  The move set contains both
   continuous transfers (as in phase 2) and random swaps of two entries,
   allowing the algorithm to escape shallow minima.  Acceptance follows a
   temperature schedule that cools to zero.

Each restart works on a possibly different length (powers of two up to
16384) and a different Dirichlet concentration, which greatly diversifies
exploration.  The best sequence seen across all restarts is returned.
All vectors are kept on the probability simplex, and the final result is
clipped to the evaluator limits before calling ``evaluate_sequence``.

import math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (full length 2·len(a)‑1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact gradient of the convolution entry (a*a)[idx] w.r.t. a.
    For each i the component equals a[idx‑i] (zero out‑of‑range).
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)          # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    return grad


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    s = v.sum()
    if s <= 0.0:
        # fall back to uniform if numerical issues appear
        return np.full_like(v, 1.0 / v.size)
    return v / s

        # Phase 1 – Smoothed exponentiated‑gradient descent
        # ------------------------------------------------------------------
        phase1_deadline = time.monotonic() + 0.20 * (deadline - start)
        beta = 5.0                         # smoothing parameter
        momentum = np.zeros_like(a)
        it = 0
        while time.monotonic() < phase1_deadline:
            a_fft = np.fft.rfft(a, n=fft_len)

            # self‑convolution
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

            # log‑sum‑exp surrogate -> soft‑max weights
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            w = exp_vals / exp_vals.sum()            # length 2n‑1

            # gradient of the surrogate (first n components)
            w_fft = np.fft.rfft(w, n=fft_len)
            grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: n + w.size - 1]
            grad = grad_full[:n]

            # momentum‑augmented exponentiated update
            momentum = 0.9 * momentum + 0.1 * grad
            eta = 0.6 / math.sqrt(it + 1.0)
            a = a * np.exp(-eta * momentum)
            a = _simplex_normalise(a)

            # evaluate exact score (sum = 1 ⇒ score = 2·n·max(conv))
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] Restart {restart_cnt} – new best (EG) = {best_score:.6f}")
                if best_score <= target_score:
                    break

            # anneal the smoothness
            beta = min(beta * 1.0015, 250.0)
            it += 1
            if it % 2000 == 0:
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] EG iter={it}, beta={beta:.1f}, best={best_score:.6f}")

        if best_score <= target_score:
            break

        # ------------------------------------------------------------------
        # Phase 2 – Focused pairwise mass transfers (larger steps)
        # ------------------------------------------------------------------
        phase2_deadline = time.monotonic() + 0.30 * (deadline - start)
        while time.monotonic() < phase2_deadline:
            conv = _conv_self(a, fft_len)
            max_idx = int(np.argmax(conv))
            grad_max = _gradient_for_index(a, max_idx)

            # select a few promising sources and targets
            top_k = max(3, int(0.003 * n))   # ≈0.3 % of the indices
            src_cand = np.argpartition(-grad_max, top_k - 1)[:top_k]
            tgt_cand = np.argpartition(grad_max, top_k - 1)[:top_k]

            improved = False
            for _ in range(15):            # try a handful of candidate moves
                src = int(rng.choice(src_cand))
                tgt = int(rng.choice(tgt_cand))
                if src == tgt:
                    continue
                # larger transfers than the original script
                frac = rng.uniform(0.02, 0.35)            # 2 % … 35 % of source mass
                delta = a[src] * frac
                if delta <= 0.0:
                    continue
                cand = a.copy()
                cand[src] -= delta
                cand[tgt] += delta
                cand = _simplex_normalise(cand)

                conv_cand = _conv_self(cand, fft_len)
                cand_score = 2.0 * n * conv_cand.max()
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_a = cand.copy()
                    improved = True
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:6.2f}s] Restart {restart_cnt} – new best (pair) = {best_score:.6f}")
                    if best_score <= target_score:
                        break

                # downhill acceptance to keep the walk progressing
                cur_score = 2.0 * n * conv.max()
                if cand_score <= cur_score:
                    a = cand

            if best_score <= target_score:
                break

        if best_score <= target_score:
            break

        # ------------------------------------------------------------------
        # Phase 3 – Simulated‑annealing refinement (continuous moves + swaps)
        # ------------------------------------------------------------------
        phase3_deadline = deadline
        temp_start = 0.02 * best_score
        it3 = 0
        while time.monotonic() < phase3_deadline:
            elapsed = time.monotonic() - start
            frac = elapsed / (deadline - start)
            temperature = temp_start * (1.0 - frac)

            conv = _conv_self(a, fft_len)
            max_idx = int(np.argmax(conv))
            grad_max = _gradient_for_index(a, max_idx)

            # pick move type
            if rng.random() < 0.65:          # continuous transfer
                src = int(np.argmax(grad_max))
                tgt = int(np.argmin(grad_max))
                delta = a[src] * rng.uniform(0.001, 0.04)
                cand = a.copy()
                cand[src] -= delta
                cand[tgt] += delta
            else:                           # random swap
                i, j = rng.integers(0, n, size=2)
                cand = a.copy()
                cand[i], cand[j] = cand[j], cand[i]

            cand = _simplex_normalise(cand)
            conv_cand = _conv_self(cand, fft_len)
            cand_score = 2.0 * n * conv_cand.max()
            cur_score = 2.0 * n * conv.max()

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = cand.copy()
                elapsed2 = time.monotonic() - start
                print(f"[{elapsed2:6.2f}s] Restart {restart_cnt} – new best (SA) = {best_score:.6f}")
                if best_score <= target_score:
                    break

            # Metropolis acceptance
            if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) /
                                                                max(1e-12, temperature)):
                a = cand

            it3 += 1
            if it3 % 5000 == 0:
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] SA iter={it3}, best={best_score:.6f}")

        if best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback: uniform distribution of moderate size
        n = 4096
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total = time.monotonic() - start
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
