"""
Optimization of a non‑negative step function using Adam on a smooth
approximation of the max‑convolution.  The algorithm works on the
probability simplex (the vector is normalised to sum = 1) and uses a
soft‑max (or top‑k) weighting of the convolution values to obtain a
gradient.  Adam‑style moments, a cosine learning‑rate schedule,
periodic random block perturbations and a simple stagnation‑restart
policy help escape local minima.  The best sequence seen before the
budget expires is returned.

The approach differs from the previous method by:
  • Using Adam instead of plain momentum SGD.
  • Switching from a smooth soft‑max to a hard top‑k weighting once
    the temperature grows large, giving a sharper sub‑gradient of the
    max‑norm.
  • A more aggressive learning‑rate decay and a higher final temperature.
  • Starting from the best known construction (if available) but also
    allowing large‑scale jitter and restarts.

The procedure runs for at most ``budget_s`` seconds (default 1000 s) and
produces a list of non‑negative heights that should achieve a score
below the target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold a linear convolution
    of length *n* (i.e. length of the result of np.convolve).
    """
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative vector *a* using FFT.
    Result length is 2*len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of   Σ_i w_i * (a * a)_i   w.r.t. *a*.
    Equivalent to 2 * (w ⊛ rev(a)) restricted to the original length.
    """
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    # slice the middle part: indices n‑1 … n‑1+n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform point.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def _topk_weights(conv: np.ndarray, k: int) -> np.ndarray:
    """
    Return a weight vector of length ``len(conv)`` that places uniform
    weight on the ``k`` largest entries of ``conv`` (ties broken arbitrarily)
    and zero elsewhere.  The weights sum to 1.
    """
    length = conv.size
    if k >= length:
        return np.full_like(conv, 1.0 / length)
    # ``np.argpartition`` gives unsorted top‑k indices.
    idx = np.argpartition(-conv, k - 1)[:k]
    w = np.zeros_like(conv)
    w[idx] = 1.0 / k
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # safety margin

    # --------------------------------------------------------------
    # 1) Choose a starting length and initialise the vector
    # --------------------------------------------------------------
    # Use the best known construction (if present) as a seed.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                base_vec = np.clip(cand, 0.0, None)
            else:
                raise ValueError
        except Exception:
            base_vec = None
    else:
        base_vec = None

    # Target length: we pick a power of two not exceeding 5000.
    # Larger lengths give more degrees of freedom but increase per‑iteration cost.
    target_len = 4096            # a comfortable power of two
    if base_vec is not None:
        # If the known vector is longer, keep it (up to 5000).
        target_len = max(target_len, min(base_vec.size, 5000))
    n = target_len

    # Initialise ``a`` on the simplex.
    if base_vec is not None:
        # Embed the old solution into a larger zero‑padded vector.
        a = np.zeros(n, dtype=np.float64)
        a[: base_vec.size] = base_vec
        a = _project_simplex(a)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    # Pre‑compute FFT lengths needed repeatedly.
    fft_len_conv = _fft_len_for_n(2 * n - 1)          # for self‑convolution
    fft_len_grad = _fft_len_for_n(3 * n - 2)          # for gradient convolution

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len_conv)
    best_score = 2.0 * n * conv_best.max() / (best.sum() ** 2)

    start_time = time.monotonic()
    last_improve = start_time

    # --------------------------------------------------------------
    # 3) Adam hyper‑parameters & schedules
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment

    lr_start = 0.15               # initial learning rate (additive)
    lr_end = 0.005                # final learning rate

    beta_start = 0.5              # temperature for the soft‑max
    beta_end = 5000.0             # final temperature (sharp max)

    block_perturb_prob = 0.03     # probability per iteration
    restart_gap = 20.0            # seconds without improvement before a restart
    topk_frac = 0.001             # fraction of convolution entries used in top‑k mode

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        # ----- current convolution and score -----
        conv = _conv_self(a, fft_len_conv)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)   # a.sum() == 1 after projection

        # ----- keep the best solution seen so far -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule for learning rate and temperature -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # Linear decay for the learning rate.
        lr = lr_start * (1.0 - elapsed / total) + lr_end * (elapsed / total)
        # Exponential schedule for β (soft‑max temperature).
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total)

        # ----- compute weight vector (soft‑max or top‑k) -----
        if beta < 200.0:
            # Soft‑max weighting (log‑sum‑exp approximation).
            shift = max_conv
            w_raw = np.exp(beta * (conv - shift))
            w = w_raw / w_raw.sum()
        else:
            # Hard top‑k weighting – a piecewise linear sub‑gradient of the max.
            k = max(5, int(topk_frac * conv.size))
            w = _topk_weights(conv, k)

        # ----- gradient w.r.t. ``a`` -----
        grad = _gradient_from_weights(a, w, fft_len_grad)

        # ----- Adam update (additive on the simplex) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        # Use bias‑corrected moments (optional but cheap).
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        a_tmp = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection back onto the simplex -----
        a = _project_simplex(a_tmp)

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- restart if we have been stagnant for a while -----
        if now - last_improve > restart_gap:
            # Jitter the current best and resume from there.
            noise = np.exp(rng.normal(0.0, 0.10, size=n))
            a = best * noise
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 5) Final fine‑tuning (local hill‑climbing) – only a few seconds left
    # --------------------------------------------------------------
    final_deadline = deadline
    while time.monotonic() < final_deadline - 0.5:
        # Propose a tiny multiplicative perturbation of a random block.
        candidate = _random_block_perturb(best, rng)
        conv_cand = _conv_self(candidate, fft_len_conv)
        cand_score = 2.0 * n * conv_cand.max() / (candidate.sum() ** 2)
        if cand_score < best_score:
            best_score = cand_score
            best = candidate
            print(f"[{time.monotonic() - start_time:6.2f}s] fine‑tune improvement = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best vector (clip to the allowed range)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
