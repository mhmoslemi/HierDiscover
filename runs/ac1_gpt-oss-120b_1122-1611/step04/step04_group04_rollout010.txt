"""
propose_candidate
-----------------
Hybrid optimizer for the “minimum max self‑convolution” problem.

1.  Starts from the best known construction (if available) or a uniform
    probability vector on a moderately large support (≈4000).  The sum of
    the vector is forced to be 1 – scaling does not affect the objective.

2.  Performs fast exponentiated‑gradient updates on the simplex.
    The gradient is obtained from a soft‑max (or hard‑max) approximation
    of the maximum of the self‑convolution.  Random block jitter,
    cheap pairwise mass‑transfer moves and occasional re‑normalisation
    keep the search lively.

3.  Every few hundred gradient steps a *linear‑program refinement* is
    applied.  For the current vector `a` we compute `M = max(a*a)`.
    Solving

        maximise   1ᵀb
        subject to  a * b ≤ M   (linear convolution)
                    b ≥ 0

    yields a direction `b` that respects the current worst‑case bound.
    After scaling `b` to have ℓ₁‑norm `√(2 n)` we mix a small amount of
    `b` into `a`.  This is exactly the LP step described in the problem
    statement and helps to escape the plateau where pure gradient descent
    stalls.

4.  Restarts with a noisy perturbation of the best vector are performed
    when no improvement has been seen for a while.

The whole routine respects the time budget (default 1000 s) and returns
the best coefficient list discovered.  All numbers are clipped to the
range required by ``evaluate_sequence`` before returning.
"""

import time
import math
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a with itself using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negatives caused by numerical noise
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k  w.r.t. a.
    The formula is   ∂/∂a_j Σ_k w_k·conv_k = 2· Σ_i a_i·w_{i+j}
    which we compute via a single FFT‑based correlation.
    """
    n = a.size
    Lw = w.size                     # = 2*n-1
    wr = w[::-1]                    # reversed for correlation

    # length needed for a (len=n) + wr (len=Lw) - 1 = n+Lw-1 = 3n-2
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep only needed part

    # grad_j = 2 * conv_full[Lw-1 - j]
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)               # ≤20 % of the vector
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 0.0, None)

    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_transfer(a: np.ndarray,
                       conv: np.ndarray,
                       rng: np.random.Generator,
                       max_fraction: float = 0.3):
    """
    Cheap pairwise mass‑transfer: move a small δ from a donor index i
    to a receiver index j if it improves the current objective.
    Returns the new (a, conv, new_score) tuple.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], max_fraction * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Update convolution incrementally (linear in a)
    new_conv = conv.copy()
    lo_i = i
    hi_i = i + n
    lo_j = j
    hi_j = j + n

    new_conv[lo_i:hi_i] -= delta * a
    new_conv[lo_j:hi_j] += delta * a

    new_score = 2.0 * n * np.max(new_conv)
    cur_score = 2.0 * n * np.max(conv)

    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.clip(a2, 0.0, None)
        a2 /= a2.sum()
        return a2, new_conv, new_score
    else:
        return a, conv, None


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1)×n matrix M such that (M @ b) = a * b
    (linear convolution).  Uses scipy.linalg.toeplitz.
    """
    n = a.size
    # First column: a followed by zeros
    c = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    # First row: a[0] followed by zeros
    r = np.concatenate([a[:1], np.zeros(2 * n - 2, dtype=a.dtype)])
    full = toeplitz(c, r)          # shape (2n‑1, 2n‑1)
    return full[:, :n]             # keep only the first n columns


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   subject to   a∗b ≤ M   and   b ≥ 0.
    Returns the optimal b (as a dense vector) or None on failure.
    """
    n = a.size
    # Build the convolution matrix: shape (2n‑1, n)
    A = _build_conv_matrix(a)

    # Right‑hand side for all constraints
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme: minimise -sum(b)  → maximise sum(b)
    c_obj = -np.ones(n, dtype=np.float64)

    # Use HiGHS – fast and robust for medium‑size LPs
    try:
        res = linprog(c=c_obj,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
    except Exception:          # safeguard against unexpected solver crashes
        return None

    if res.success:
        return res.x
    return None


def _lp_refine(a: np.ndarray,
               mix_coeff: float = 0.2,
               rng: np.random.Generator | None = None) -> np.ndarray:
    """
    Perform a single LP‑based refinement step:

        * compute M = max(a∗a)
        * solve the linear programme to obtain b
        * scale b to ℓ₁‑norm √(2 n)
        * mix a ← (1‑mix_coeff)·a + mix_coeff·scaled(b)
        * re‑normalise to sum = 1

    If the LP fails, returns the input unchanged.
    """
    n = a.size
    fft_len = _next_pow2(2 * n - 1)
    conv = _self_conv_fft(a, fft_len)
    M = conv.max()
    b = _solve_lp_max_sum(a, M)
    if b is None:
        return a

    s_b = b.sum()
    if s_b == 0.0:
        return a

    scale = math.sqrt(2.0 * n) / s_b
    g = b * scale
    g = np.clip(g, 0.0, None)

    a_new = (1.0 - mix_coeff) * a + mix_coeff * g
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full_like(a_new, 1.0 / n)

    # Optional tiny jitter to avoid exact ties in later iterations
    if rng is not None and rng.random() < 0.01:
        jitter = np.exp(rng.normal(0.0, 0.05, size=n))
        a_new *= jitter
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

    return a_new


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    Returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine problem size and initialise the vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            # fallback to a modest size
            n = 4000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure the vector lies on the simplex
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a.sum()

    # Pre‑compute an FFT length (power‑of‑two) for the self‑convolution
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float('inf')
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safe margin
    last_improve = start_time

    # hyper‑parameters (chosen for a 1000 s budget)
    eta_start = 0.8          # initial EG learning rate
    eta_end = 0.02           # final learning rate
    beta_start = 0.5         # temperature for soft‑max
    beta_end = 3000.0        # final temperature (hard max)
    hard_max_phase = 0.85    # fraction of time after which we use a hard max
    block_perturb_prob = 0.03
    pairwise_transfer_prob = 0.07
    lp_interval = 400        # perform LP refinement every this many gradient steps
    lp_mix = 0.2             # mixing coefficient used inside LP refinement
    restart_gap = 30.0       # seconds without improvement → restart
    step = 0

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current self‑convolution (FFT) -----
        conv = _self_conv_fft(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- record best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- learning‑rate & temperature schedule -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)

        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- weight vector for a smoothed max -----
        if frac < hard_max_phase:
            w = np.exp(beta * (conv - cur_max))          # soft‑max
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0                     # exact sub‑gradient

        # ----- gradient of the weighted max -----
        grad = _gradient_from_weights(a, w)

        # ----- exponentiated‑gradient update on the simplex -----
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block jitter -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- cheap pairwise mass transfer -----
        if rng.random() < pairwise_transfer_prob:
            a, conv, _ = _pairwise_transfer(a, conv, rng)

        # ----- LP‑based refinement at regular intervals -----
        if step % lp_interval == 0 and step > 0:
            # a small number of refinement passes; each requires an LP solve
            for _ in range(3):
                a = _lp_refine(a, mix_coeff=lp_mix, rng=rng)

        # ----- restart logic (prevent stagnation) -----
        if now - last_improve > restart_gap:
            # restart from the best vector with a noisy perturbation
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.2, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart with noisy best (score {best_score:.6f})")

        step += 1

    # ------------------------------------------------------------------
    # 4) Return best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
