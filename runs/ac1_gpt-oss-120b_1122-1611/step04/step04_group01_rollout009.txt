"""
Hybrid sub‑gradient descent for the max‑convolution problem.

The objective to minimise is
    score = 2 * n * max(conv(a, a))          (sum(a) is fixed to 1)

Because the score is invariant under scaling we restrict the search to the
probability simplex {a ≥ 0, Σ a = 1}.  The algorithm works as follows

1. Initialise a (either the supplied `height_sequence_1` or a uniform vector).
2. Repeatedly
   * compute the self‑convolution via a real FFT,
   * locate the active indices where the convolution attains (or almost attains)
     its maximum,
   * evaluate the exact sub‑gradient of the max‑convolution:
         g_j = 2 * Σ_{k∈active} a_{k‑j}
     (computed efficiently by one additional convolution of the active mask
      with the reversed vector),
   * take a projected sub‑gradient step with a linearly decaying step size,
   * occasionally perform cheap “mass‑transfer” moves that move a small amount
     of probability mass from indices that contribute the most to the current
     peak towards indices that contribute little,
   * occasionally jitter a random block of coordinates.
   The best vector seen so far is kept.
3. The loop stops when the time budget (≈ 1000 s) is exhausted.
4. The best vector is clipped to the limits enforced by `evaluate_sequence`
   and returned as a plain Python list.

The method improves on the previous soft‑max/Adam approach by using the true
sub‑gradient of the ℓ∞‑norm (active‑set gradient) and by a focussed local
mass‑transfer heuristic, both of which tend to reduce the peak of the
self‑convolution more aggressively.

"""

import time
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length able to hold a linear convolution
    of two vectors whose combined length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) using an FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.25))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Choose problem size and an initial feasible vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                raise ValueError
        except Exception:
            n = 3809          # a size that worked well in earlier runs
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 3809
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (once)
    # --------------------------------------------------------------
    fft_len_conv = _fft_len_for_len(2 * n - 1)          # for a*a
    fft_len_aux = _fft_len_for_len(3 * n - 2)          # for auxiliary conv

    # --------------------------------------------------------------
    # 3) Book‑keeping of the best solution
    # --------------------------------------------------------------
    conv = _conv_self(a, fft_len_conv)
    best = a.copy()
    best_score = 2.0 * n * conv.max()               # sum(a)=1 ⇒ denominator = 1
    start_time = time.monotonic()
    last_improve = start_time

    # --------------------------------------------------------------
    # 4) Hyper‑parameters
    # --------------------------------------------------------------
    step_start = 0.8        # initial sub‑gradient step size
    step_end = 0.001        # final step size (linear schedule)
    block_perturb_prob = 0.03
    mass_transfer_prob = 0.12
    restart_gap = 40.0      # seconds without improvement -> restart
    active_tol_factor = 1e-12
    momentum = 0.85
    grad_prev = np.zeros_like(a)

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 5.1) Current convolution and score --------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ---- 5.2) Record a new best --------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- 5.3) Time‑dependent schedules ------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear decay of the step size
        step = step_start * (1.0 - elapsed / total) + step_end * (elapsed / total)

        # ---- 5.4) Sub‑gradient of the max‑convolution ---------------
        # tolerance that determines the active set
        tol = max(cur_max * active_tol_factor, 1e-15)
        active_mask = (conv >= cur_max - tol).astype(float)
        active_cnt = active_mask.sum()
        if active_cnt > 0:
            # convolution of the active mask with the reversed a:
            a_rev = a[::-1]
            conv_active_rev = _conv_fft(active_mask, a_rev, fft_len_aux)
            # extract the central n entries (indices n‑1 … n‑1+n‑1)
            grad = 2.0 * conv_active_rev[n - 1:n - 1 + n] / active_cnt
        else:
            grad = np.zeros_like(a)

        # ---- 5.5) Momentum smoothing --------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- 5.6) Projected sub‑gradient descent step ---------------
        a = a - step * grad
        a = _project_onto_simplex(a)

        # ---- 5.7) Random block perturbation ------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ---- 5.8) Targeted mass‑transfer move -----------------------
        if rng.random() < mass_transfer_prob:
            # locate the index where the convolution is maximal
            k_max = int(np.argmax(conv))
            # compute a shifted copy a_{k_max‑i} (zero outside the range)
            start = max(0, k_max - (n - 1))
            end = min(n - 1, k_max)
            length = end - start + 1
            # obtain the needed slice of the reversed vector
            a_rev = a[::-1]
            slice_src = a_rev[(n - 1 - k_max + start):(n - 1 - k_max + end + 1)]
            a_shift = np.zeros_like(a)
            a_shift[start:end + 1] = slice_src
            # product a_i * a_{k_max‑i} quantifies the contribution of each i
            contrib = a * a_shift
            i_src = int(np.argmax(contrib))
            # choose a destination with relatively small current mass
            low_candidates = np.where(a < np.quantile(a, 0.25))[0]
            if low_candidates.size > 0:
                j_dst = int(rng.choice(low_candidates))
                if i_src != j_dst and a[i_src] > 0.0:
                    # move a modest amount of probability mass
                    delta = rng.random() * min(a[i_src], 0.4 * a[i_src])
                    a[i_src] -= delta
                    a[j_dst] += delta
                    # a stays on the simplex automatically, just enforce non‑negativity
                    a = np.clip(a, 0.0, None)

        # ---- 5.9) Soft restart if no improvement for a while -----
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter the whole vector a little bit
            jitter = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            grad_prev[:] = 0.0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best vector (respect the caps enforced by
    #    ``evaluate_sequence`` – they simply clip to [0, 1000]).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
