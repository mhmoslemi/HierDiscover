Propose‑candidate for the step‑function optimisation problem.

The algorithm is a hybrid of:
* a soft‑max (log‑sum‑exp) surrogate minimisation via exponentiated‑gradient
  (mirror descent) on the probability simplex,
* momentum smoothing,
* several cheap stochastic perturbations (block scaling, pairwise mass
  transfer, swap‑move),
* exponential temperature schedule for the surrogate,
* periodic soft‑restarts with jitter,
* a final greedy polishing stage.

All updates keep the vector on the simplex (non‑negative, sum = 1), which
makes the evaluation (2 n·max(conv)/sum²) scale‑invariant.  The routine
tries to flatten the self‑convolution so that its maximum approaches the
average value 1/(2 n−1), i.e. a score close to 1.0.  The code is fully
self‑contained, uses only NumPy and the supplied ``evaluate_sequence``
function, and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length 2·n‑1).
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate:
        grad_j = 2·∑_k w_k·a_{k−j}
    (a correlation computed via FFT).
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Stochastic perturbations
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by re‑normalisation to the unit simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# ----------------------------------------------------------------------
