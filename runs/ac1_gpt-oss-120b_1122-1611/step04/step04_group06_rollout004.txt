def _project_simplex(v):
    # Projects v onto the probability simplex (v_i >=0, sum=1)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]

"""
Propose a candidate step‑function via variance‑gradient descent.

The objective to minimise is
    F(a) = 2·n·max_k (a∗a)[k] / (∑a)^2 .
Because the function is scale‑invariant we may restrict to ∑a = 1.
If the autocorrelation (linear convolution) is flat, max_k is close to its
average 1/(2n‑1) and the score approaches 1.  The algorithm therefore
optimises the *variance* of the convolution:

    V(a) = Σ_k ( (a∗a)[k] – μ )² ,   μ = 1/(2n‑1).

V(a) is smooth; ∂V/∂a can be written as a correlation between the current
vector a and the centred convolution w = (a∗a) – μ·1.
We perform projected (simplex) gradient descent with momentum on V(a),
adding occasional random block perturbations and pair‑wise mass transfers
to escape plateaux.  The best sequence encountered before the time budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Correlation a * w (full linear correlation) using FFT.
    ``a`` length = n, ``w`` length = 2·n‑1.
    Returns vector of length n."""
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = conv_full[start:start + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    # Implementation of Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.

    The algorithm works on the probability simplex and drives the
    autocorrelation of the vector towards a flat profile by minimising its
    variance.  Projected gradient descent with momentum, random block
    perturbations and occasional pair‑wise mass transfers is employed.
    The best sequence found during the allotted time is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose a starting vector (use the best known construction if
    #    it is available, otherwise fall back to the uniform distribution)
    # ------------------------------------------------------------------
    default_n = 4000
    if "height_sequence_1" in globals():
        try:
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq.ndim == 1 and seq.size > 0 and np.isfinite(seq).all():
                n = seq.size
                a = np.clip(seq, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(n, 1.0 / n)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Pre‑compute the FFT length for the self‑convolution (used every step)
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len)
    best = a.copy()
    best_score = 2.0 * n * conv.max()          # sum(a) == 1 ⇒ score = 2·n·maxconv
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    lr_start = 1.0          # initial step‑size
    lr_end = 0.001          # final step‑size
    momentum = 0.9
    block_perturb_prob = 0.03
    pair_move_prob = 0.08
    restart_gap = 30.0      # seconds without improvement → restart
    v = np.zeros_like(a)    # momentum buffer

    # constant average of the convolution when Σa=1
    avg_conv = 1.0 / (2 * n - 1)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution & score --------------------------------
        conv = _conv_self(a, fft_len)
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val          # sum(a) == 1

        # ----- keep the best solution ------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule ---------------------------------------------------
        now = time.monotonic()
        elapsed_frac = (now - start_time) / (deadline - start_time)
        lr = lr_start * (1.0 - elapsed_frac) + lr_end * elapsed_frac

        # ----- gradient of the convolution variance ----------------------
        #   V(a) = Σ_k (conv_k – avg)^2
        #   ∂V/∂a = 2 * correlation( a , conv - avg )
        w = conv - avg_conv
        grad = 2.0 * _gradient_correlation(a, w)

        # ----- momentum smoothing -----------------------------------------
        v = momentum * v + (1.0 - momentum) * grad

        # ----- projected gradient step ------------------------------------
        a_new = a - lr * v
        a = _project_simplex(a_new)            # enforce a_i ≥ 0, Σa = 1

        # ----- occasional random block perturbation ----------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pair‑wise mass transfer ------------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.2 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = _project_simplex(a_candidate)
                    conv_cand = _conv_self(a_candidate, fft_len)
                    cand_score = 2.0 * n * conv_cand.max()
                    if cand_score < cur_score:
                        a = a_candidate
                        conv = conv_cand
                        cur_score = cand_score
                        max_val = conv_cand.max()

        # ----- soft restart if stuck --------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # inject a mild multiplicative jitter on all coordinates
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = _project_simplex(a)
            v = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clip to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
