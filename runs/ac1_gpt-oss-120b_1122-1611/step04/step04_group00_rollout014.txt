def compute_score_vec(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.size
    # use FFT for convolution
    fft_len = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
    max_b = np.max(conv)
    return 2.0*n*max_b/(s*s)

def perturb(a, rng, sigma):
    n = a.size
    i = rng.integers(n)
    j = rng.integers(n)
    if i==j: return a
    delta = rng.normal(0, sigma) * a[i]
    # ensure non-negative after move
    new_i = a[i] - delta
    new_j = a[j] + delta
    if new_i < 0:
        delta = a[i]  # move all mass
        new_i = 0.0
        new_j = a[j] + delta
    a2 = a.copy()
    a2[i] = new_i
    a2[j] = new_j
    a2 = _project_simplex(a2)  # ensure sum=1
    return a2

propose_candidate
-----------------
Random‑sparse + gradient‑refinement search.

* Phase 1 – generate many random sparse vectors (few hundred non‑zero
  entries, Dirichlet‑distributed weights) and keep the best according to
  the exact evaluation (fast FFT convolution).  Sparse vectors can achieve
  a lower ℓ∞‑norm of the self‑convolution than a dense uniform vector.

* Phase 2 – start from the best sparse candidate and run a projected
  gradient descent on a soft‑max approximation of the max‑convolution.
  This fine‑tunes the weights while keeping the sum fixed (simplex
  projection) and occasionally applies a block perturbation to escape
  shallow minima.

The whole routine respects the 1000 s budget, prints progress and finally
returns the best sequence found.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of length *n*."""
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative vector *a* using FFT.
    Result length is 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Gradient of  Σ_i w_i * (a * a)_i  w.r.t. *a*.
    Equivalent to 2 * (w ⊛ a_rev) restricted to the original length.
    """
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    # slice the middle part: indices n‑1 … n‑1+n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def _compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective (same formula as ``evaluate_sequence``)
    but using an FFT‑based convolution for speed.
    The vector is assumed to be already clipped to [0,1000].
    """
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _random_sparse_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a sparse probability vector:
    - pick a random number k of non‑zero entries,
    - choose k distinct positions,
    - assign Dirichlet(1,…,1) weights (uniform on the simplex).
    """
    k = rng.integers(80, 250)               # typical size that works well
    pos = rng.choice(n, size=k, replace=False)

    w = rng.dirichlet(np.ones(k))
    a = np.zeros(n, dtype=np.float64)
    a[pos] = w
    # ensure exact sum = 1 (numerical safety)
    a = a / a.sum()
    return a

    print(f"[{time.monotonic() - start:5.2f}s] final best score = {final_score:.6f}")
    return [float(x) for x in best]
