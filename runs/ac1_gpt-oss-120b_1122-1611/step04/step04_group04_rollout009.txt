"""
Propose a candidate step‑function by greedy targeted mass transfers.

The algorithm works directly on the simplex (non‑negative entries,
sum = 1).  At each iteration the index `k_max` where the self‑convolution
`a*a` attains its maximum is identified.  The contribution of a component
`a[i]` to this peak is `a[i] * a[k_max-i]`.  To reduce the peak we move a
small amount of mass from a “donor’’ with a large partner
`a[k_max-i]` to a “receiver’’ with a small (ideally zero) partner.
The convolution is updated incrementally in O(n) using slice operations,
so many iterations fit into the 1000 s budget.

Occasionally a random block perturbation is applied and a restart is
performed if no improvement has been seen for a while.  The best vector
found (clipped to the limits of `evaluate_sequence`) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2*n‑1)."""
    return np.convolve(a, a)  # fine for n≈4 k


def _apply_transfer(a: np.ndarray,
                    conv: np.ndarray,
                    i: int,
                    j: int,
                    delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` mass from index i
    to index j (i != j).  The update uses the exact formula derived from
    (a+Δ)*(a+Δ) = a*a + 2*a*Δ + Δ*Δ.

    Parameters
    ----------
    a : ndarray
        Current coefficient vector (length n).
    conv : ndarray
        Current self‑convolution (length 2*n‑1).
    i, j : int
        Source (donor) and target (receiver) indices.
    delta : float
        Amount transferred (positive, ≤ a[i]).

    Returns
    -------
    new_conv : ndarray
        Updated convolution.
    """
    n = a.size
    new_conv = conv.copy()
    # linear part: -2Δ·shift(a,i) + 2Δ·shift(a,j)
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a
    # quadratic part Δ*Δ
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          max_width_frac: float = 0.2,
                          sigma: float = 0.12) -> np.ndarray:
    """
    Apply a multiplicative log‑normal jitter to a random contiguous block
    and renormalise onto the simplex.
    """
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function (a probability vector)
    that minimises ``evaluate_sequence``.  Returns the best sequence
    found within the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Initialise the coefficient vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # a uniform start of moderate length (≈4000)
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    conv = _self_convolution(a)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # small safety margin
    last_improve = start_time

    # --------------------------------------------------------------
    # 2) Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ---- locate the current max of the convolution ----
        k_max = int(np.argmax(conv))

        # ---- compute partner values for every index (a[k_max - i] when defined) ----
        idx = np.arange(n)
        valid = (k_max - idx >= 0) & (k_max - idx < n)
        partner = np.zeros_like(a)
        partner[valid] = a[k_max - idx[valid]]

        # ---- choose donor (large partner, non‑zero mass) ----
        donor_mask = a > 1e-12
        donor_score = partner * donor_mask.astype(float)
        if donor_score.max() == 0.0:
            # no donor with positive partner – fall back to a random move
            # (unlikely but kept for robustness)
            i = rng.integers(0, n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_delta = 0.3 * a[i]
            delta = rng.random() * max_delta
            new_conv = _apply_transfer(a, conv, i, j, delta)
            new_score = 2.0 * n * new_conv.max()
            if new_score < cur_score:
                a[i] -= delta
                a[j] += delta
                conv = new_conv
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] new best {best_score:.6f}")
            continue

        donor = int(np.argmax(donor_score))

        # ---- choose receiver (smallest partner) ----
        # Prefer indices where partner is zero (outside the support of k_max)
        receiver_candidates = np.where(partner == partner.min())[0]
        # Ensure we do not pick the donor itself
        if donor in receiver_candidates:
            if len(receiver_candidates) > 1:
                receiver_candidates = receiver_candidates[receiver_candidates != donor]
            else:
                # pick the second smallest partner
                sorted_idx = np.argsort(partner)
                for cand in sorted_idx:
                    if cand != donor:
                        receiver = cand
                        break
                else:
                    # fallback: random different index
                    receiver = (donor + 1) % n
                # move on to delta step
                max_delta = 0.3 * a[donor]
                delta = rng.random() * max_delta
                new_conv = _apply_transfer(a, conv, donor, receiver, delta)
                new_score = 2.0 * n * new_conv.max()
                if new_score < cur_score:
                    a[donor] -= delta
                    a[receiver] += delta
                    conv = new_conv
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best = a.copy()
                        last_improve = time.monotonic()
                        elapsed = last_improve - start_time
                        print(f"[{elapsed:6.2f}s] new best {best_score:.6f}")
                continue
        receiver = int(rng.choice(receiver_candidates))

        # ---- determine a step size (fraction of donor mass) ----
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = elapsed / total
        # early on we allow larger moves, later we shrink them
        max_frac = 0.3 * (1.0 - frac) + 0.02 * frac
        max_delta = a[donor] * max_frac
        if max_delta < 1e-12:
            continue
        delta = rng.random() * max_delta

        # ---- apply the transfer and evaluate ----
        new_conv = _apply_transfer(a, conv, donor, receiver, delta)
        new_score = 2.0 * n * new_conv.max()

        # ---- accept only if improvement ----
        if new_score < cur_score:
            a[donor] -= delta
            a[receiver] += delta
            conv = new_conv
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] new best {best_score:.6f}")

        # ---- occasional random block perturbation (escape shallow plateaus) ----
        if rng.random() < 0.015:
            a_pert = _random_block_perturb(a, rng)
            conv_pert = _self_convolution(a_pert)
            pert_score = 2.0 * n * conv_pert.max()
            if pert_score < cur_score:
                a = a_pert
                conv = conv_pert
                cur_score = pert_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] block‑perturb improved best {best_score:.6f}")

        # ---- restart if we have been stuck for a while ----
        if time.monotonic() - last_improve > 30.0:
            a = best.copy()
            # jitter the best solution a bit
            noise = np.exp(rng.normal(0.0, 0.25, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _self_convolution(a)
            cur_score = 2.0 * n * conv.max()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – score reset to {cur_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluate_sequence limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
