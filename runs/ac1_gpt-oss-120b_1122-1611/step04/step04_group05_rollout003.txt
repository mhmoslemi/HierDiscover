            best_sum = best.sum()
            if best_sum == 0:

propose_candidate
=================

Random‑parameter search + short exponentiated‑gradient polishing.

The search explores three families of edge‑heavy (U‑shaped) probability
distributions on a fixed length ``n``:

* **Beta**   –  a_i ∝ x^{α‑1} (1‑x)^{β‑1},      x = (i+½)/n,   α,β∈(0,5)
* **Gamma**  –  a_i ∝ ((i+½)(n‑i‑½))^{-γ},    γ∈(0.2,3)
* **Recip**  –  a_i ∝ (1/(i+1)+1/(n‑i))^{γ}, γ∈(0.2,3)

All candidates are normalised to sum 1, the score

    score = 2·n·max(conv(a,a)) / (sum(a)²)

is evaluated with an FFT based convolution (O(n log n)).  Whenever a new
record is found it is printed.  Every few hundred evaluations the current
best point is refined with a handful of exponentiated‑gradient steps that
use a soft‑max weighting of the convolution entries (large β ≈ 600).  If
no improvement occurs for 120 s a small jitter restart is performed.

The routine respects a ``budget_s`` wall‑clock limit (≈ 1000 s) and finally

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Return the evaluation score for a candidate ``a``."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_self(a, fft_len)
    max_conv = conv.max()
    n = a.size
    return float(2.0 * n * max_conv / (s * s))


def _refine_exponential(a: np.ndarray,
                        fft_len: int,
                        steps: int = 30,
                        eta: float = 0.5,
                        beta: float = 500.0,
                        momentum: float = 0.9) -> np.ndarray:
    """A few mirror‑descent (exponentiated‑gradient) steps starting from ``a``.
    The update uses a soft‑max weighting of all convolution entries."""
    n = a.size
    a_cur = a.copy()
    grad_prev = np.zeros_like(a_cur)

    for _ in range(steps):
        conv = _conv_self(a_cur, fft_len)
        max_c = conv.max()
        # soft‑max weights, the entry attaining the max gets weight 1
        w = np.exp(beta * (conv - max_c))

        # gradient of the weighted sum w.r.t. a
        a_rev = a_cur[::-1]
        L = (2 * n - 1) + n - 1          # length needed for the convolution
        fft_len2 = _fft_len_for_len(L)

        fw = np.fft.rfft(w, n=fft_len2)
        fa = np.fft.rfft(a_rev, n=fft_len2)
        grad_full = np.fft.irfft(fw * fa, n=fft_len2)

        # extract the n components that correspond to the gradient entries
        grad = 2.0 * grad_full[n - 1:n - 1 + n] - a_cur * w[2 * np.arange(n)]

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # avoid overflow in the exponent
        grad = np.clip(grad, -50.0, 50.0)
        factor = np.exp(-eta * grad)

        a_cur = a_cur * factor
        a_cur = np.clip(a_cur, 0.0, None)
        s = a_cur.sum()
        if s == 0.0:
            a_cur = np.full_like(a_cur, 1.0 / n)
        else:
            a_cur /= s

    return a_cur

    rng = np.random.default_rng(seed)

    # --------------------------------------------------
    # 1. Determine the working length and an initial point
    # --------------------------------------------------
    default_n = 5000
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a0 = np.clip(cand, 0.0, None)
                s = a0.sum()
                a0 = a0 / s if s > 0.0 else np.full(n, 1.0 / n)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n)

    # deterministic arcsine start (often a good baseline)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = np.clip(arcsine, 0.0, None)
    arcsine /= arcsine.sum()

    fft_len = _fft_len_for_n(n)

    # keep the better of the two cheap starts
    score_a0 = _score_from_a(a0, fft_len)
    score_arc = _score_from_a(arcsine, fft_len)
    if score_arc < score_a0:
        best = arcsine.copy()
        best_score = score_arc
    else:
        best = a0.copy()
        best_score = score_a0

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # Pre‑compute arrays reused for many families
    denom = (i + 0.5) * (n - i - 0.5)          # for the “gamma” family
    uniform = np.full(n, 1.0 / n)              # mixing partner

    # --------------------------------------------------
    # 2. Random‑parameter search loop
    # --------------------------------------------------
    iteration = 0
    REFINE_EVERY = 400                         # polishing frequency
    while time.monotonic() < deadline:
        # ------ pick a random family and draw parameters ------
        r = rng.random()
        if r < 0.35:
            # ---- Beta‑distribution (U‑shaped when α,β<1) ----
            alpha = 0.05 + rng.random() * 4.95   # [0.05,5)
            beta = 0.05 + rng.random() * 4.95
            mix = rng.random()
            x = (np.arange(n) + 0.5) / n
            shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
            shape = np.clip(shape, 0.0, np.finfo(np.float64).max)
            a = (1.0 - mix) * uniform + mix * shape
        elif r < 0.70:
            # ---- Power‑of‑distance family ----
            gamma = 0.2 + rng.random() * 2.8      # [0.2,3.0)
            shape = np.power(denom, -gamma)
            a = shape
        else:
            # ---- Reciprocal‑sum family ----
            gamma = 0.2 + rng.random() * 2.8
            base = 1.0 / (i + 1.0) + 1.0 / (n - i)
            shape = np.power(base, gamma)
            a = shape

        # normalise to the simplex
        s = a.sum()
        if s == 0.0:
            continue
        a /= s

        # ------ evaluate the candidate ------
        cur_score = _score_from_a(a, fft_len)

        # ------ keep the best so far ------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        iteration += 1

        # ------ periodic polishing with a short mirror‑descent run ------
        if iteration % REFINE_EVERY == 0:
            polished = _refine_exponential(best,
                                           fft_len,
                                           steps=25,
                                           eta=0.8,
                                           beta=600.0,
                                           momentum=0.85)
            polished_score = _score_from_a(polished, fft_len)
            if polished_score < best_score:
                best_score = polished_score
                best = polished.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] refined best score = {best_score:.6f}")

        # ------ restart if we have been idle for a while ------
        if time.monotonic() - last_improve > 120.0:
            # small random jitter around the current best
            jitter = np.exp(rng.normal(0.0, 0.10, size=n))
            best = best * jitter
            best = np.clip(best, 0.0, None)
            best /= best.sum()
            best_score = _score_from_a(best, fft_len)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------
    # 3. Final sanity check (optional) and return
    # --------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    result = [float(v) for v in best]

    # If the evaluation function is available, report the definitive score.
    try:
        final_val = evaluate_sequence(result)
        print(f"Final evaluated score (full check) = {final_val:.6f}")
    except Exception:
        # In the rare case ``evaluate_sequence`` is not present, silently ignore.
        pass

    return result
