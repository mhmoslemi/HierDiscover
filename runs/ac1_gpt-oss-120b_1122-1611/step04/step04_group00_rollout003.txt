"""
Propose a candidate step‑function using a mirror‑descent (Adam) optimiser
on the probability simplex.

The evaluation metric

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

is homogeneous, therefore we can restrict the search to vectors with
unit ℓ₁‑norm (the simplex).  The non‑smooth max is replaced by a
soft‑max (log‑sum‑exp) with an annealing temperature β.  The gradient of
the smoothed objective is obtained by a single FFT‑based convolution.
A multiplicative Adam update (mirror descent w.r.t. KL‑divergence)
keeps all coordinates non‑negative; after each step we project back onto
the simplex.  The learning rate η is linearly annealed, β grows
exponentially, occasional log‑normal block perturbations are applied,
and a simple restart on stagnation is performed.  The best vector seen
throughout the run is returned.

The implementation is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` function, and respects the 1000‑second
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures – all top‑level)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    length *n*."""
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative vector *a* using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Gradient of  Σ_i w_i * (a * a)_i  w.r.t. *a*.
    Equivalent to 2 * (w ⊛ a_rev) restricted to the original length."""
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    # slice the middle part: indices n‑1 … n‑1+n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex
    {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    # width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    # safety margin so that we can still return a value at the end
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # 1) Choose starting vector and length
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        # use the previously discovered construction if available
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        if cand.size == 0 or not np.isfinite(cand).all():
            cand = None
    else:
        cand = None

    if cand is not None:
        a = cand.copy()
        # normalise to the simplex
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a.sum()
        n = a.size
    else:
        # fall back to a uniform vector of moderate size
        n = 1024
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Enlarge the vector to a convenient power‑of‑two length (≤ 5000)
    target_len = max(n, 4096)
    target_len = min(target_len, 5000)  # keep FFT cheap
    if target_len != n:
        # pad with a tiny positive constant so that multiplicative updates
        # can later inject mass into the new slots
        pad = np.full(target_len - n, 1e-8, dtype=np.float64)
        a = np.concatenate([a, pad])
        a = _project_simplex(a)      # re‑normalise to sum = 1
        n = target_len

    # FFT lengths required repeatedly
    fft_len_conv = _fft_len_for_n(2 * n - 1)       # for self‑convolution
    fft_len_grad = _fft_len_for_n(3 * n - 2)       # for gradient convolution

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    start_time = time.monotonic()
    last_improve = start_time

    # Adam state
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment

    # hyper‑parameters (tuned for the 1 % improvement regime)
    eta_start = 0.6      # initial learning‑rate
    eta_end = 0.02       # final learning‑rate
    beta_start = 0.5     # initial soft‑max temperature
    beta_end = 5000.0    # final (almost hard) temperature
    block_perturb_prob = 0.03
    restart_gap = 20.0   # seconds of stagnation before a restart
    beta1 = 0.9          # Adam first‑moment decay
    beta2 = 0.999        # Adam second‑moment decay
    eps = 1e-8           # Adam epsilon

    # best solution seen so far (approximate score used for comparison)
    conv = _conv_self(a, fft_len_conv)
    cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_score = cur_score
    best = a.copy()

    # --------------------------------------------------------------
    # 3) Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ------ current convolution and approximate score ------
        conv = _conv_self(a, fft_len_conv)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        # keep the best vector (approximate score)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best approx score = {best_score:.6f}")

        # ------ schedules for η and β ------
        now = time.monotonic()
        elapsed = now - start_time
        progress = min(1.0, elapsed / budget_s)   # in [0,1]

        eta = eta_start * (1.0 - progress) + eta_end * progress
        beta_temp = beta_start * (beta_end / beta_start) ** progress

        # ------ soft‑max weights (stable log‑sum‑exp) ------
        shift = max_conv                       # subtract max for numerical stability
        w_raw = np.exp(beta_temp * (conv - shift))
        w = w_raw / w_raw.sum()               # normalised weights, sum to 1

        # ------ gradient of the smoothed objective ------
        grad = _gradient_from_weights(a, w, fft_len_grad)

        # ------ Adam moment updates ------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # ------ multiplicative (KL‑mirror) update + projection ------
        a = a * np.exp(-step)           # guarantee non‑negativity
        a = _project_simplex(a)         # stay on the probability simplex

        # ------ occasional random block perturbation ------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ------ restart if stuck for a while ------
        if now - last_improve > restart_gap:
            # perturb the current best and start again
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * noise
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best approx {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final evaluation of the best vector (exact)
    # --------------------------------------------------------------
    exact_score = evaluate_sequence(best.tolist())
    print(f"Final exact score: {exact_score:.6f}")

    # Clip to the limits imposed by ``evaluate_sequence`` and return
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
