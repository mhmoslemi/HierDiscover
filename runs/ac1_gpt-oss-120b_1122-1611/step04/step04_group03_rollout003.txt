```
def _project_simplex(v):
    # v: np.ndarray
    # https://arxiv.org/pdf/1101.6081.pdf
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def _project_simplex(v):
    # v: np.ndarray
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
```

"""
Projected‑gradient descent on the simplex with a hard‑max subgradient,
augmented by occasional random block jitter and soft restarts.

The objective that evaluate_sequence() returns (for a non‑zero sum) is

    S(a) = 2·n·max(conv(a,a)) / (∑a)².

Because the ratio is scale‑invariant we may keep the vector on the
probability simplex (∑a = 1).  For the current maximal convolution entry
k* = argmax_k (a∗a)[k] the gradient of convₖ* w.r.t. a is

    ∂convₖ*/∂a_i = 2·a_{k*‑i}   (or 0 if the index is out of range).

Thus a subgradient of the objective is simply 2·(a_rev) shifted by k*.
We perform an additive update a ← a – η·g and project back onto the
simplex (the Euclidean projection described by Duchi et al.).  The steps‑
size η is decreased linearly from η_start to η_end.  To escape shallow
local minima we apply:

* a log‑normal jitter on a random contiguous block (probability
  block_perturb_prob);
* a swap of two random coordinates (swap_prob);
* a soft restart that re‑initialises from the best solution seen so far
  after ``restart_gap`` seconds without improvement.

The algorithm is fully vectorised, each iteration costs only an FFT
(~O(n log n)) for the self‑convolution needed to locate the maximal
entry, and O(n) work for the gradient.  With the typical length
≈ 4 000 it runs a few thousand iterations per second, comfortably
fitting into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implementation follows
    Duchi et al., “Efficient Projections onto the l1‑Ball for Learning
    in High Dimensions”.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:      # all entries are zero or negative
        theta = 0.0
    else:
        rho = rho[-1]      # last true index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:           # numerical safety – fall back to uniform
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and
    re‑project onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                a = a / a.sum()
            else:
                raise ValueError
        except Exception:
            n = 3800
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 3800
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 1.0      # initial additive step size
    eta_end = 0.001      # final step size
    block_perturb_prob = 0.05
    swap_prob = 0.07
    restart_gap = 40.0   # seconds without improvement → restart

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- current convolution and score --------------------------------
        conv = _conv_self(a)
        max_idx = int(conv.argmax())
        cur_max = float(conv[max_idx])
        cur_score = 2.0 * n * cur_max

        # ---- keep the best solution ---------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule (linear interpolation) -------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # ---- compute the (hard‑max) subgradient ----------------------------
        # Gradient of the convolution entry at max_idx.
        grad = np.zeros_like(a)

        L = max(0, max_idx - (n - 1))
        R = min(max_idx, n - 1)
        # The overlapping indices are i in [L, R]; for each such i,
        # the partner index is j = max_idx - i.
        idx = np.arange(L, R + 1, dtype=int)
        grad[idx] = 2.0 * a[max_idx - idx]

        # ---- additive update with projection --------------------------------
        a_candidate = a - eta * grad
        a_candidate = _project_simplex(a_candidate)

        # ---- accept only if we improve (simple backtracking) ---------------
        conv_cand = _conv_self(a_candidate)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < cur_score:
            a = a_candidate
        else:
            # try a smaller step (backtracking) – up to a few reductions
            eta_bt = eta
            for _ in range(5):
                eta_bt *= 0.5
                a_bt = a - eta_bt * grad
                a_bt = _project_simplex(a_bt)
                cand_score = 2.0 * n * _conv_self(a_bt).max()
                if cand_score < cur_score:
                    a = a_bt
                    break
            # otherwise keep the current a (no move)

        # ---- occasional random perturbations --------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # ---- soft restart if progress stalls --------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            a = a * jitter
            a = _project_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best solution (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
