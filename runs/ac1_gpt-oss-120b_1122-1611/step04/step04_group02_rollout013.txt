def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5

    # Choose length
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n0 = base.size
        # maybe adjust n by scaling? We'll keep n = n0
        n = n0
        # maybe use base normalized to sum=1 as initial candidate.
        base = np.clip(base, 0, None)
        if base.sum()>0:
            base = base/base.sum()
        else:
            base = np.full(n, 1/n)
    else:
        # choose default length:
        n = 512  # or 1024 
        base = rng.dirichlet(np.ones(n))
    # We'll also include block of uniform initial.

    # Initialize population
    popsize = max(20, int(0.1*n))   # maybe 50-100
    # Represent individuals as logit vectors: random normal or from base
    pop = []
    scores = []
    for i in range(popsize):
        if i==0:
            # one individual is base (converted to logits)
            x = np.log(base + 1e-12)  # avoid zeros by adding epsilon
        else:
            # random Dirichlet then log
            a = rng.dirichlet(np.ones(n))
            x = np.log(a + 1e-12)
        pop.append(x)
        a = np.exp(x)
        a /= a.sum()
        s = evaluate_score(a)
        scores.append(s)

    best_idx = int(np.argmin(scores))
    best_x = pop[best_idx].copy()
    best_score = scores[best_idx]

    # Precalc FFT length
    fft_len = _fft_len_for_n(n)

    # Main loop
    while time.monotonic() < deadline:
        # DE step: for each individual generate a trial
        for i in range(popsize):
            # Choose distinct r1,r2,r3
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)
            F = rng.uniform(0.5, 0.9)  # mutation factor
            # Mutation
            mutant_x = pop[r1] + F * (pop[r2] - pop[r3])
            # Crossover
            CR = 0.9
            jrand = rng.integers(0, n)
            trial_x = np.empty_like(mutant_x)
            for j in range(n):
                if rng.random() < CR or j == jrand:
                    trial_x[j] = mutant_x[j]
                else:
                    trial_x[j] = pop[i][j]
            # Compute a from logits (softmax)
            # To avoid overflow, subtract max
            max_logit = trial_x.max()
            exp_x = np.exp(trial_x - max_logit)
            a = exp_x / exp_x.sum()

            s = evaluate_score(a)
            # Selection
            if s < scores[i] - 1e-12:
                pop[i] = trial_x
                scores[i] = s
                if s < best_score - 1e-12:

                    best_x = trial_x.copy()

        # We'll also occasionally apply random perturb to best
        if time.monotonic() > deadline - 30:  # last 30 secs for SA
            break

    # Local refinement via simulated annealing on best_x
    best_a = np.exp(best_x - best_x.max())
    best_a = best_a / best_a.sum()
    best_score_local = best_score

    # Simulated annealing
    cur_a = best_a.copy()
    cur_score = best_score_local
    temperature_start = 0.1
    temperature_end = 1e-5
    # schedule based on elapsed time left
    start_sa = time.monotonic()
    for iteration in range(1000000):
        now = time.monotonic()
        if now >= deadline:
            break
        # linear schedule
        t_frac = (now - start_sa) / (deadline - start_sa)
        temperature = temperature_start * (1 - t_frac) + temperature_end * t_frac
        # propose move: pick i, j, move delta
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        max_delta = 0.2 * cur_a[i]
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        s_new = evaluate_score(a_new)
        # accept if better or with probability
        if s_new < cur_score or rng.random() < np.exp(-(s_new - cur_score) / temperature):
            cur_a = a_new
            cur_score = s_new
            if cur_score < best_score_local:
                best_score_local = cur_score
                best_a = cur_a.copy()
                print(f"[{now - (deadline - budget_s + 5):.2f}s] New best SA score {best_score_local:.6f}")
        # optionally occasional block perturb
        if rng.random() < 0.01:
            # apply multiplicative noise to random block
            width = rng.integers(1, max(2, n//20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0, 0.1))

            a_block = np.clip(a_block, 0, None)
            total = a_block.sum()
            if total == 0:
                a_block = np.full(n, 1.0/n)
            else:
                a_block /= total
            s_block = evaluate_score(a_block)
            if s_block < cur_score or rng.random() < np.exp(-(s_block - cur_score) / temperature):
                cur_a = a_block

                    best_score_local = cur_score
                    best_a = cur_a.copy()
                    print(f"[{now - (deadline - budget_s + 5):.2f}s] New best block SA score {best_score_local:.6f}")

    # After finishing, return best_a as list
    best_a = np.clip(best_a, 0.0, 1000.0)  # safety

"""
Improved search for a step‑function minimizing the bound

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of two phases.

Phase 1 – Smoothed ‑max descent on the simplex.
    • The vector is stored as logits x (a = softmax(x)).
    • At each iteration the full self‑convolution is obtained with an FFT.
    • A soft‑max over the convolution entries (temperature β) yields a smooth
      approximation of the max‑norm.
    • Using the weights wₖ = exp(β·(cₖ–c_max)) we form a sub‑gradient of the
      smoothed max w.r.t a and map it to logits via the Jacobian of soft‑max.
    • Adam (or a simple momentum scheme) updates the logits.
    • β is increased from 0.5 to 500 and the learning rate is decreased
      linearly; occasional random block perturbations keep diversity.
    • If no improvement is seen for 30 s the current best is jittered
      (multiplicative log‑normal noise) to escape stagnation.

Phase 2 – Stochastic local refinement (simulated annealing).
    • Pairwise mass transfers together with occasional block perturbations
      are proposed.
    • An exponentially decreasing temperature accepts uphill moves.
    • The best solution seen overall is remembered.

If a previous construction is present in the global variable
`height_sequence_1` it is used as a warm start (its length can be several
thousand).  Otherwise a modest length (≈ 512) is chosen.

The routine respects the time budget (≈ 1000 s) and returns the best
sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length able to hold a linear convolution of
    two vectors of length *n* (full length = 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector `a` using FFT.
    Returned array has length 2*len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(logits)
    exp_logits = np.exp(logits - m)
    return exp_logits / exp_logits.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # safety margin

    # ------------------------------------------------------------------
    # 1) Choose a starting length and initialise logits
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        s = base.sum()
        if s > 0.0:
            base /= s
        else:
            base = np.full_like(base, 1.0 / base.size)
        n = base.size
        # logits = log of the probabilities (add tiny epsilon to avoid log(0))
        logits = np.log(base + 1e-12)
    else:
        n = 512
        base = rng.dirichlet(np.ones(n))
        logits = np.log(base + 1e-12)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution (used many times)
    # ------------------------------------------------------------------
    fft_len_self = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 3) Adam state and hyper‑parameter schedules
    # ------------------------------------------------------------------
    m = np.zeros_like(logits)           # first moment
    v = np.zeros_like(logits)           # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    lr_start, lr_end = 0.30, 0.008
    beta_start, beta_end = 0.5, 500.0

    # ------------------------------------------------------------------
    # 4) Book‑keeping of the best solution
    # ------------------------------------------------------------------
    def state_to_score(logits_vec):
        """Convert logits to a probability vector, evaluate the bound and
        return (score, prob_vec, full_convolution)."""
        a = _softmax(logits_vec)
        conv = _conv_self_fft(a, fft_len_self)
        max_c = conv.max()
        score = 2.0 * n * max_c
        return score, a, conv

    best_score, best_a, best_conv = state_to_score(logits)
    best_logits = logits.copy()
    last_improve = time.monotonic()

    # ------------------------------------------------------------------
    # 5) Phase 1 – Smoothed‑max descent (≈70 % of the budget)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    total_budget = deadline - start_time
    phase1_end = start_time + 0.70 * total_budget
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        if now >= phase1_end:
            break

        # fraction of phase 1 elapsed (0 → 1)
        frac = (now - start_time) / (phase1_end - start_time)

        # schedules
        lr = lr_start * (1.0 - frac) + lr_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # probability vector from current logits
        a = _softmax(logits)

        # self‑convolution
        conv = _conv_self_fft(a, fft_len_self)
        max_conv = conv.max()

        # ----- choose active convolution entries (top K) -----
        K_active = max(200, int(0.12 * conv.size))   # a modest active set
        if K_active >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            kth = conv.size - K_active
            active_idx = np.argpartition(conv, kth)[kth:]

        # ----- compute smoothed sub‑gradient w.r.t. a -----
        grad_a = np.zeros_like(a)
        for idx in active_idx:
            w = np.exp(beta * (conv[idx] - max_conv))   # weight ∈ (0,1]
            if w == 0.0:
                continue
            shift = idx - (n - 1)      # shift d = idx‑(n‑1)
            if shift >= 0:
                # left part
                grad_a[: n - shift] += w * a[shift:]
                # right part
                grad_a[shift:] += w * a[: n - shift]
            else:
                d = -shift
                grad_a[d:] += w * a[: n - d]
                grad_a[: n - d] += w * a[d:]

        # ----- map gradient to logits (Jacobian of soft‑max) -----
        dot = np.dot(a, grad_a)
        grad_logits = a * (grad_a - dot)

        # ----- Adam update on logits -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional random block perturbation (helps avoid plateaus)
        if rng.random() < 0.015:
            a_pert = _random_block_perturb(a, rng)
            logits = np.log(a_pert + 1e-12)

        # current score
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            best_logits = logits.copy()
            best_conv = conv.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] New best (phase 1) score = {best_score:.6f}")

        # soft restart if stuck for a while
        if now - last_improve > 30.0:
            # jitter the best logits
            jitter = rng.normal(0.0, 0.07, size=n)
            logits = best_logits + jitter
            m.fill(0.0)
            v.fill(0.0)
            iteration = 0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart (no improvement) – current best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Phase 2 – Simulated‑annealing refinement (remaining budget)
    # ------------------------------------------------------------------
    # start from the best solution found in phase 1
    cur_a = best_a.copy()
    cur_score = best_score
    conv_len = 2 * n - 1
    fft_len_self = _fft_len_for_n(n)   # re‑usable
    start_sa = time.monotonic()
    last_improve_sa = start_sa

    while time.monotonic() < deadline:
        now = time.monotonic()
        # linear temperature schedule: start 0.09 → end 1e‑7
        t_frac = (now - start_sa) / (deadline - start_sa)
        temperature = max(1e-7, 0.09 * (1.0 - t_frac))

        # ----- pairwise mass transfer -----
        if rng.random() < 0.85:
            i = rng.integers(0, n)
            if cur_a[i] < 1e-12:
                continue
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            max_delta = 0.30 * cur_a[i]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            a_new = cur_a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            # ----- block perturbation (global jitter) -----
            a_new = _random_block_perturb(cur_a, rng)

        # evaluate new candidate
        conv_new = _conv_self_fft(a_new, fft_len_self)
        new_score = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            cur_a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = cur_a.copy()
                last_improve_sa = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New best (phase 2) score = {best_score:.6f}")

        # occasional hard restart if stuck
        if now - last_improve_sa > 45.0:
            # jitter best solution and continue
            cur_a = _random_block_perturb(best_a, rng)
            cur_score = 2.0 * n * _conv_self_fft(cur_a, fft_len_self).max()
            last_improve_sa = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart (phase 2) – retaining best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best vector (clipped for safety)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
