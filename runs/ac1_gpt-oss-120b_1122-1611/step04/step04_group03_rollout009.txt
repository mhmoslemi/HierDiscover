"""
Proposes a non‑negative step‑function heights vector that (approximately) minimises
``evaluate_sequence``.  The routine combines a long‑run mirrored‑gradient (Adam‑style)
optimisation on the soft‑max approximation of the max‑convolution, occasional
log‑normal jitter, and a final simulated‑annealing refinement.  It starts from
the provided ``height_sequence_1`` (if any) and works on a probability simplex
so that the evaluation score equals ``2·n·max(conv(a,a))``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT (real)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.15) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Return a list of heights that (approximately) minimises ``evaluate_sequence``.

    The method runs two phases:
    1.  Adam‑style mirror‑descent on the soft‑max approximation of the
        max–convolution.
    2.  A short simulated‑annealing local search that perturbs the best
        solution found in phase 1.

    The whole routine respects the time budget; if it finishes early it
    returns the best vector discovered so far.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) choose the length and initialise the probability vector a
    # ------------------------------------------------------------------
    default_n = 4096
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a0 = np.clip(cand, 0.0, None)
                s0 = a0.sum()
                if s0 > 0.0:
                    a0 = a0 / s0
                else:
                    raise ValueError
            else:
                raise ValueError
        except Exception:
            n = default_n
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = a0.size
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # add a mild random jitter so that the optimiser does not start from the exact uniform point
    jitter = np.exp(rng.normal(0.0, 0.10, size=n))
    a = a0 * jitter
    a = np.clip(a, 0.0, None)
    a = a / a.sum()

    # pre‑compute FFT lengths (independent of a)
    fft_len_self = _next_pow2(2 * n - 1)          # for a∗a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w∗a_rev

    # ------------------------------------------------------------------
    # 2) bookkeeping for the best solution seen so far
    # ------------------------------------------------------------------
    # evaluate the current vector with the official routine (it performs the same clipping)
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    last_improve = start_time
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase I – Adam mirror‑descent on the smoothed max‑convolution
    # ------------------------------------------------------------------
    # hyper‑parameters
    eta_start = 1.0
    eta_end = 0.001
    beta_start = 0.2
    beta_end = 800.0
    block_perturb_prob = 0.03
    swap_prob = 0.02
    restart_gap = 45.0               # seconds without improvement → restart
    jitter_sigma_restart = 0.12

    # Adam statistics
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    iteration = 0

    # reserve about 200 s for the second phase
    phase1_deadline = deadline - 200.0

    while time.monotonic() < phase1_deadline:
        # --- current objective ------------------------------------------------
        conv = _conv_self(a)                     # length 2n‑1
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max            # sum(a)=1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best (phase I)  = {best_score:.6f}")

        # --- schedule ---------------------------------------------------------
        now = time.monotonic()
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress
        beta = beta_start + (beta_end - beta_start) * progress

        # --- soft‑max weights -------------------------------------------------
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smoothed objective (full soft‑max) ---------------
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: (len(w) + len(a_rev) - 1)]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

        # --- Adam update in the log‑probability space -------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        log_factor = -eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_factor = np.clip(log_factor, -50.0, 50.0)
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # --- occasional random perturbations ---------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng, sigma=0.15)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # --- restart if we have been idle for a while ------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma_restart, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m[:] = 0.0
            v[:] = 0.0
            iteration = 0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart (phase I) – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase II – Simulated annealing local refinement
    # ------------------------------------------------------------------
    # start from the best vector discovered in phase I
    cur = best.copy()
    cur_score = best_score
    phase2_start = time.monotonic()
    # temperature schedule (linear in log‑space)
    T_start = 0.02
    T_end = 0.0002
    # standard deviation for log‑normal block jitter (will be scaled with temperature)
    sigma_start = 0.20
    sigma_end = 0.02

    while time.monotonic() < deadline:
        now = time.monotonic()
        phase2_elapsed = now - phase2_start
        phase2_total = deadline - phase2_start
        if phase2_total <= 0:
            break
        # linear schedule in [0,1]
        phase2_frac = phase2_elapsed / phase2_total
        T = T_start * (1.0 - phase2_frac) + T_end * phase2_frac
        sigma = sigma_start * (1.0 - phase2_frac) + sigma_end * phase2_frac

        # propose a neighbour
        if rng.random() < 0.6:
            # block‑wise log‑normal jitter
            width = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, sigma))
            a_candidate = cur.copy()
            a_candidate[start_idx:start_idx + width] *= factor
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate /= a_candidate.sum()
        else:
            # transfer a small amount of mass from i to j
            i, j = rng.integers(0, n, size=2)
            if cur[i] > 0.0:
                delta = rng.uniform(0.0, min(cur[i], 0.01))
                a_candidate = cur.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                # renormalisation is not needed because total mass unchanged,
                # but we keep the numerical safety net
                a_candidate = np.clip(a_candidate, 0.0, None)
                a_candidate /= a_candidate.sum()
            else:
                a_candidate = cur.copy()

        # evaluate candidate (fast, using internal FFT)
        conv_cand = _conv_self(a_candidate)
        cand_score = 2.0 * n * float(conv_cand.max())

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = a_candidate.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best (phase II) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best solution (clipped to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # convert to plain Python floats (list) as required by the harness
    return [float(x) for x in best]

"""
Proposes a non‑negative step‑function height vector that (approximately) minimizes
the objective defined by ``evaluate_sequence``.

The routine works in two phases:

1.  An Adam‑style mirror‑descent on the soft‑max approximation of the
    max‑convolution (the same idea as the reference implementation but with
    per‑parameter adaptation and a sturdier restart strategy).

2.  A short simulated‑annealing local search that perturbs the best vector
    obtained in phase 1.  The neighbour generation consists of a log‑normal
    block jitter and a tiny mass‑transfer move; acceptance follows the
    Metropolis rule.

The algorithm always keeps the vector on the probability simplex (sum=1) so
that the evaluation score equals ``2·n·max(conv(a,a))``.  The final result is
clipped to the limits forced by ``evaluate_sequence`` before being returned.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT (real)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          sigma: float = 0.15) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Return a list of heights that (approximately) minimizes ``evaluate_sequence``.

    Two optimisation phases are used:
    1. Adam‑style mirror‑descent on a smoothed max‑convolution objective.
    2. Simulated‑annealing refinement around the best solution from phase 1.

    The routine respects the supplied time budget and returns the best vector
    discovered when the time expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # 1) choose length and initialise the probability vector `a`
    # ------------------------------------------------------------------
    default_n = 4096
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a0 = np.clip(cand, 0.0, None)
                s0 = a0.sum()
                if s0 > 0.0:
                    a0 = a0 / s0
                else:
                    raise ValueError
            else:
                raise ValueError
        except Exception:
            n = default_n
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = a0.size
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # a small log‑normal jitter to break perfect symmetry
    jitter = np.exp(rng.normal(0.0, 0.10, size=n))
    a = a0 * jitter
    a = np.clip(a, 0.0, None)
    a = a / a.sum()

    # pre‑compute FFT lengths (independent of `a`)
    fft_len_self = _next_pow2(2 * n - 1)    # for a∗a
    fft_len_grad = _next_pow2(3 * n - 2)    # for w∗a_rev

    # ------------------------------------------------------------------
    # 2) bookkeeping for the best solution seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    last_improve = start_time
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase I – Adam mirror‑descent on the smoothed max‑convolution
    # ------------------------------------------------------------------
    # hyper‑parameters
    eta_start = 1.0
    eta_end = 0.001
    beta_start = 0.2
    beta_end = 800.0
    block_perturb_prob = 0.03
    swap_prob = 0.02
    restart_gap = 45.0               # seconds without improvement → restart
    jitter_sigma_restart = 0.12

    # Adam statistics
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    iteration = 0

    # reserve roughly 200 s for the second phase
    phase1_deadline = deadline - 200.0

    while time.monotonic() < phase1_deadline:
        # current objective
        conv = _conv_self(a)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max      # sum(a)=1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best (phase I) = {best_score:.6f}")

        # schedule
        now = time.monotonic()
        progress = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress
        beta = beta_start + (beta_end - beta_start) * progress

        # soft‑max weights approximating the max‑operator
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective (soft‑max)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: (len(w) + len(a_rev) - 1)]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

        # Adam update in log‑probability space
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        log_factor = -eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_factor = np.clip(log_factor, -50.0, 50.0)   # avoid overflow
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional random perturbations
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng, sigma=0.15)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # restart if stagnating
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma_restart, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m[:] = 0.0
            v[:] = 0.0
            iteration = 0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart (phase I) – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase II – Simulated annealing refinement
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score
    phase2_start = time.monotonic()
    T_start = 0.02
    T_end = 0.0002
    sigma_start = 0.20
    sigma_end = 0.02

    while time.monotonic() < deadline:
        now = time.monotonic()
        phase2_elapsed = now - phase2_start
        phase2_total = deadline - phase2_start
        if phase2_total <= 0.0:
            break
        frac = phase2_elapsed / phase2_total
        T = T_start * (1.0 - frac) + T_end * frac
        sigma = sigma_start * (1.0 - frac) + sigma_end * frac

        # propose a neighbour
        if rng.random() < 0.6:
            # block‑wise log‑normal jitter
            width = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, sigma))
            a_cand = cur.copy()
            a_cand[start_idx:start_idx + width] *= factor
            a_cand = np.clip(a_cand, 0.0, None)
            a_cand /= a_cand.sum()
        else:
            # tiny mass transfer between two entries
            i, j = rng.integers(0, n, size=2)
            if cur[i] > 0.0:
                delta = rng.uniform(0.0, min(cur[i], 0.01))
                a_cand = cur.copy()
                a_cand[i] -= delta
                a_cand[j] += delta
                a_cand = np.clip(a_cand, 0.0, None)
                a_cand /= a_cand.sum()
            else:
                a_cand = cur.copy()

        # fast evaluation of the candidate
        conv_cand = _conv_self(a_cand)
        cand_score = 2.0 * n * float(conv_cand.max())

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = a_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = a_cand.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best (phase II) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best solution (clipped to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
