def max_convolution_gradient(a):
    n = a.size
    conv = _conv_self(a, ff_len)
    max_val = conv.max()
    # find indices where conv close to max, within tolerance
    eps = max(1e-12, max_val * 1e-12)
    active = np.where(conv >= max_val - eps)[0]  # array of indices
    # For each active k, compute shifted gradient: g_k[j] = 2*a[k - j] if 0 <= k-j < n else 0.
    # This yields matrix shape len(active) x n; compute average
    grad = np.zeros_like(a)
    for k in active:
        # slice a to assign: for j where 0 <= k-j < n => j in [k-n+1, k]
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # indices:
        # for j in [j_start, j_end], index in a for a[k - j]
        # So grad[j_start:j_end+1] += 2 * a[k - np.arange(j_start, j_end+1)]
        idx = np.arange(j_start, j_end + 1)
        grad[idx] += 2 * a[k - idx]
    grad /= len(active)
    return grad, conv.max()

        maxc = conv.max()

"""
Optimise a non‑negative step‑function (a list of heights) for the evaluation
criterion

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a two‑stage strategy:

1.  Fast parametric seeding – we scan a few simple families of shapes
    (power‑law / “arcsine”, mixtures with the uniform distribution,
    the previously found global variable ``height_sequence_1``) and keep the
    best candidate.

2.  Refined mirror‑descent on the probability simplex.  The true
    sub‑gradient of the max‑norm of the self‑convolution is computed
    analytically (no soft‑max approximation), and an exponentiated‑gradient
    (KL‑mirror) update is applied.  A few stochastic perturbations,
    pair‑wise mass transfers and soft restarts keep the search from stalling.

The algorithm stays well within the 1000 s time budget, prints progress,
and finally returns the best list of heights found (clipped to the limits
required by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global seed candidate (if any)
    # ------------------------------------------------------------------
    default_n = 4000
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose a working length and create a list of seed candidates
    # ------------------------------------------------------------------
    n = default_n
    seed_candidates = []

    # –‑ uniform (always a useful baseline)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    seed_candidates.append(uniform)

    # –‑ arcsine‑type power‑law family, parameterised by exponent p
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)                     # >0 for all i
    # we will scan p later (see stage 1)

    # –‑ optional previously discovered sequence
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                # use it only if its length matches our working n;
                # otherwise resample it linearly to length n.
                if cand.size != n:
                    # simple linear interpolation / repacking
                    x_old = np.linspace(0, 1, cand.size)
                    x_new = np.linspace(0, 1, n)
                    cand = np.interp(x_new, x_old, cand)
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                else:
                    cand = np.full(n, 1.0 / n, dtype=np.float64)
                seed_candidates.append(cand.copy())
        except Exception:
            pass  # ignore malformed globals

    # ------------------------------------------------------------------
    # 2) Helper to evaluate a normalised vector (sum = 1) quickly
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    def _score_of(a_vec: np.ndarray) -> float:
        """Return the raw score 2*n*max(conv) for a vector that already sums to 1."""
        conv = _conv_self(a_vec, fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 3) Stage 1 – cheap parametric sweep (p∈[0,0.7] and mixtures)
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf

    # scan p‑values (the shape a_i ∝ denom^{-p})
    for p in np.linspace(0.0, 0.7, 36):        # 36 ≈ step 0.02
        shape = denom ** (-p)                # 1 / denom^{p}
        shape = np.clip(shape, 0.0, None)
        a = shape / shape.sum()
        score = _score_of(a)
        if score < best_score:
            best_score = score
            best_vec = a.copy()

    # mixture with uniform – convex combinations improve robustness
    if best_vec is not None:
        for w in np.linspace(0.0, 1.0, 21):   # 21 points from 0 to 1
            a = w * best_vec + (1.0 - w) * uniform
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            score = _score_of(a)
            if score < best_score:
                best_score = score
                best_vec = a.copy()

    # include any global candidate
    for cand in seed_candidates:
        score = _score_of(cand)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()

    # fall back to uniform if something went wrong
    if best_vec is None:
        best_vec = uniform.copy()
        best_score = _score_of(best_vec)

    # ------------------------------------------------------------------
    # 4) Stage 2 – mirror‑descent with exact max‑norm sub‑gradient
    # ------------------------------------------------------------------
    a = best_vec.copy()                # current iterate (always normalised)
    grad_prev = np.zeros_like(a)

    # hyper‑parameters (linear schedules)
    eta_start = 0.5
    eta_end = 0.01
    momentum = 0.9

    # restart / perturbation schedule
    restart_gap = 25.0                 # seconds without improvement → restart
    block_perturb_prob = 0.04
    pair_move_prob = 0.08
    last_improve = start_time

    overall_best = a.copy()
    overall_best_score = best_score

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution and score -----
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # sum(a) == 1 → denominator = 1

        # ----- keep global best -----
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {overall_best_score:.6f}")

        # ----- schedule -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # ----- exact sub‑gradient of max(conv) -----
        # Find all indices where the convolution attains (or nearly attains) its max.
        tol = max(cur_max * 1e-12, 1e-15)      # guard against numerical noise
        active = np.where(conv >= cur_max - tol)[0]

        grad = np.zeros_like(a)
        for k in active:
            # For the given shift k, the contribution to the gradient is
            #   g_j = 2 * a_{k - j}   (if 0 ≤ k‑j < n)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1, dtype=np.intp)
            grad[idx] += 2.0 * a[k - idx]
        if len(active) > 1:
            grad /= len(active)               # average over active constraints

        # ----- momentum smoothing (optional) -----
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad.copy()

        # ----- exponentiated‑gradient (mirror descent) update -----
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pairwise mass transfer -----
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta = rng.random() * min(a[i], 0.3 * a[i])
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = np.clip(a_new, 0.0, None)
                a_new /= a_new.sum()
                conv_new = _conv_self(a_new, fft_len)
                new_score = 2.0 * n * conv_new.max()
                if new_score < cur_score:
                    a = a_new
                    cur_score = new_score          # for the surrounding loop
                    conv = conv_new

        # ----- soft restart when progress stalls -----
        if now - last_improve > restart_gap:
            a = overall_best.copy()
            # gentle multiplicative log‑normal jitter on every coordinate
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev[:] = 0.0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector found (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    overall_best = np.clip(overall_best, 0.0, 1000.0)
    return [float(x) for x in overall_best]
