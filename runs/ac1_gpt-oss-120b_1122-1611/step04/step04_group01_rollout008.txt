A two‑phase gradient based optimiser on the probability simplex.

Phase 1 (first half of the time) minimises a simple L2 loss that
flattens the full convolution **a*a** towards its theoretical
average 1/(2n‑1).  The gradient is
    ∇L = 4·( (conv‑avg) * a_rev )
where * denotes linear convolution.
Phase 2 (second half) switches to a smooth approximation of the
max‑norm (log‑sum‑exp) and uses the same Adam‑style update as the
previous baseline.

Both phases keep the vector on the simplex by Euclidean projection.
Occasional random block perturbations and cheap pair‑wise mass‑transfer
moves are performed to escape shallow local minima.  After the main
loop a short greedy local‑search (random pair moves that reduce the real
score) is run.

The routine works for any initial sequence present in the global variable
`height_sequence_1`; otherwise it starts from the uniform distribution.
The length is fixed (default ≈ 5000) – larger lengths give a better
asymptotic bound at little extra cost.

The returned list is clipped to the limits enforced by
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors whose total length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT.
    The returned array has length ``len(x)+len(y)-1``."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Returns a new array."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    # small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a starting vector
    # ------------------------------------------------------------------
    default_n = 5000                     # a bit larger than the previous 3809
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = np.clip(cand, 0.0, None)
                total = a.sum()
                if total > 0.0:
                    a = a / total
                else:
                    a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_conv = _fft_len_for_len(2 * n - 1)          # for a*a
    fft_len_grad = _fft_len_for_len(3 * n - 2)          # for weighted convolutions

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len_conv)
    cur_max = conv.max()
    best = a.copy()
    best_score = 2.0 * n * cur_max           # sum(a)=1 → denominator =1
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Adam hyper‑parameters (shared between both phases)
    # ------------------------------------------------------------------
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # ------------------------------------------------------------------
    # 5) Miscellaneous probabilities and schedules
    # ------------------------------------------------------------------
    block_perturb_prob = 0.02
    pair_move_prob = 0.06
    restart_gap = 40.0          # seconds without improvement → restart
    momentum = 0.85

    # phase split
    total_time = deadline - start_time
    phase_split = 0.5           # 0‑1 fraction of total time spent in phase 1

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # ----- recompute convolution and current score -----
        conv = _conv_self(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- record best -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- decide which phase we are in -----
        if frac < phase_split:
            # Phase 1: minimise L2 distance to the flat target
            target = 1.0 / (2 * n - 1)                      # average of a*a
            residual = conv - target                       # length 2n‑1
            a_rev = a[::-1]
            # gradient of ½‖conv‑target‖² = 4 * (residual * a_rev)
            grad_raw = 4.0 * _conv_fft(residual, a_rev, fft_len_grad)[n - 1 : n - 1 + n]
        else:
            # Phase 2: smooth approximation of the max‑norm (log‑sum‑exp)
            beta_start, beta_end = 0.5, 1500.0
            eta_start, eta_end = 0.8, 0.001
            beta = beta_start + (beta_end - beta_start) * frac
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # soft‑max weights
            w = np.exp(beta * (conv - cur_max))               # max shift avoids overflow
            sum_w = w.sum()
            # gradient of fβ = (1/β) log Σ exp(β·conv) :
            a_rev = a[::-1]
            conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
            grad_raw = a + conv_w_rev[n - 1 : n - 1 + n] / sum_w

        # ----- momentum smoothing of gradient -----
        grad_raw = momentum * m + (1.0 - momentum) * grad_raw
        m = momentum * m + (1.0 - momentum) * grad_raw  # reuse m as smoothed gradient

        # ----- Adam update (additive) -----
        t += 1
        m = b1 * m + (1.0 - b1) * grad_raw
        v = b2 * v + (1.0 - b2) * (grad_raw ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        # step size schedule (linear decay, safe lower bound)
        eta = eta_start * (1.0 - frac) + eta_end * frac
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- cheap pairwise mass‑transfer move (accepted only if improves) -----
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    # already stays on simplex, just need to enforce non‑negativity
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate = _project_onto_simplex(a_candidate)
                    conv_cand = _conv_self(a_candidate, fft_len_conv)
                    score_cand = 2.0 * n * conv_cand.max()
                    if score_cand < cur_score:
                        a = a_candidate
                        conv = conv_cand
                        cur_max = conv_cand.max()
                        cur_score = score_cand

        # ----- restart if stuck -----
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Final greedy local refinement (pair moves only, no gradients)
    # ------------------------------------------------------------------
    refinement_deadline = time.monotonic() + 5.0   # a few seconds extra
    a = best.copy()
    conv = _conv_self(a, fft_len_conv)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    while time.monotonic() < refinement_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        max_delta = min(a[i], 0.3 * a[i])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate = _project_onto_simplex(a_candidate)
        conv_cand = _conv_self(a_candidate, fft_len_conv)
        score_cand = 2.0 * n * conv_cand.max()
        if score_cand < cur_score:
            a = a_candidate
            conv = conv_cand
            cur_max = conv_cand.max()
            cur_score = score_cand
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[refine] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8) Return the best vector (clipping per evaluate_sequence's limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
