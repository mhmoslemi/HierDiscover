"""
Simple LP‑based iterative improvement for minimizing the evaluation score.
The method follows the iterative scheme from the paper: given a current
distribution f (normalised to sum = 1) compute M=max(conv(f,f)),
solve a linear program that maximises the total mass of a vector g
subject to conv(f,g)≤M, rescale g to sum = 1 and mix it with f.
Repeating this step reduces M and therefore the score.
"""
import time
import numpy as np
from scipy.optimize import linprog

def _fft_len_for_n(n: int) -> int:
    size = n - 1

        # ----- schedule the mixing weight ------------------------------------
        t = max(t * t_decay, min_t)

        # ----- occasional stochastic perturbation -----------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        iteration += 1

    # ------------------------------------------------------------------
    # 5) Return the best found sequence (clip to allowed range)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
