"""
Propose a better step‑function using an exact sub‑gradient descent on the
∞‑norm of the self‑convolution.

The algorithm works on the simplex (a_i ≥ 0, Σ a_i = 1).  At each iteration
it
  * evaluates the self‑convolution a*a,
  * builds the set of indices where the convolution attains its maximum,
  * computes the exact sub‑gradient of  max(a*a)  (a correlation with the mask),
  * moves a small amount of probability mass from the coordinate with the
    largest gradient (most responsible for the peak) to the coordinate with
    the smallest gradient,
  * accepts the move only if the peak value decreased.
A decaying step‑size and occasional random block perturbations help to
escape plateaus; if no improvement occurs for a while the current best
candidate is jittered (soft restart).  The best sequence seen during the
budget is returned.

The implementation uses FFT‑based convolution for speed and works for any
vector length.  If a previous construction is available via the global
``height_sequence_1`` it is used as a warm start; otherwise a random Dirichlet
sample of length 1024 is generated.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    n = a.size
    if fft_len is None:
        fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * n - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))

    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the vector a
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                total = a.sum()
                if total > 0.0:
                    a = a / total
                else:
                    raise ValueError
            else:
                raise ValueError
        except Exception:
            # fallback: random Dirichlet of moderate size
            n = 1024
            a = rng.dirichlet(np.ones(n))
    else:
        n = 1024
        a = rng.dirichlet(np.ones(n))

    n = a.size
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len)
    max_val = float(conv.max())
    best_score_est = 2.0 * n * max_val
    best = a.copy()
    # verify exact score once (cheap for the first candidate)
    try:
        best_exact = evaluate_sequence(best.tolist())
        best_score_est = best_exact
    except Exception:
        best_exact = float("inf")

    last_improve = time.monotonic()
    start_time = last_improve

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    step_start = 0.5   # initial fraction of source mass to move
    step_end = 0.01    # final fraction
    eps_rel = 1e-12    # tolerance for determining the max‑set
    block_perturb_prob = 0.02
    restart_gap = 60.0          # seconds without improvement → restart
    print_interval = 30.0
    next_print = start_time + print_interval
    no_improve_counter = 0
    max_no_improve = 250       # after this many failed moves we perturb

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < start_time + max(0.0, float(budget_s) - 5.0):
        now = time.monotonic()
        elapsed = now - start_time
        total = budget_s - 5.0

        # ---- linear schedule for step size ---------------------------------
        frac = elapsed / total
        step = step_start * (1.0 - frac) + step_end * frac

        # ---- compute convolution and current peak --------------------------
        conv = _conv_self(a, fft_len)
        max_val = float(conv.max())
        cur_score_est = 2.0 * n * max_val

        # ---- record best (exact) -------------------------------------------
        if cur_score_est < best_score_est - 1e-12:

            if exact < best_score_est:
                best_score_est = exact
                best = a.copy()
                last_improve = now

        # ---- periodic status output ----------------------------------------
        if now >= next_print:
            print(f"[{elapsed:6.2f}s] cur est. score = {cur_score_est:.6f}  best = {best_score_est:.6f}")
            next_print = now + print_interval

        # ---- occasional random block perturbation ---------------------------
        if rng.random() < block_perturb_prob:

        # ---- build mask for the (approximate) max‑set ----------------------
        eps = max(max_val * eps_rel, 1e-15)
        mask = (conv >= max_val - eps).astype(float)   # shape 2n‑1

        # ---- exact sub‑gradient of max(conv) --------------------------------
        # grad = 2 * corr(mask, a)   (valid part has length n)
        grad = 2.0 * np.correlate(mask, a, mode='valid')
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # ---- propose a mass transfer ----------------------------------------
        delta = step * a[src]
        delta = min(delta, a[src])          # never move more than we own
        if delta <= 0.0:
            # fallback to a random pair if source is empty
            src = rng.integers(0, n)
            dst = rng.integers(0, n - 1)
            while dst == src:
                dst = rng.integers(0, n)
            delta = step * a[src]

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s

        # ---- accept if the peak decreased -----------------------------------
        conv_new = _conv_self(a_new, fft_len)
        max_new = float(conv_new.max())
        if max_new < max_val - 1e-15:
            a = a_new
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        if now - last_improve > restart_gap or no_improve_counter >= max_no_improve:

            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            no_improve_counter = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score_est:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits of ``evaluate_sequence``)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
