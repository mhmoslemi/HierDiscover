"""
Simple simulated annealing / hill‑climbing optimizer for the step‑function
max–convolution objective.

The routine starts from the best known sequence (if the global variable
``height_sequence_1`` is present) or from a uniform distribution.
It then repeatedly performs cheap pairwise mass‑transfer moves that are
exactly reflected in the self‑convolution via an O(n) update.  The move is
accepted if it improves the current score or with a probability that
decreases according to an annealing schedule.  Occasionally a random
block‑wise perturbation and a global jitter (restart) are applied to
escape local minima.  The best sequence seen before the time budget expires
is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Return the smallest power‑of‑two size that can hold a linear
    convolution of total length ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution ``a * a`` using a real FFT.
    Returns an array of length ``2*len(a)-1``."""
    n = a.shape[0]
    fft_len = _fft_len_for_len(2 * n - 1)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * n - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all zero -> uniform
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1] - 1          # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_onto_simplex(out)
    return out


def _apply_pair_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int,
                         delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving ``delta`` mass from index ``i``
    to index ``j``.  ``a`` is the *old* vector (unchanged), ``conv`` is the
    old self‑convolution.
    """
    n = a.shape[0]
    # copy conv to avoid modifying the original
    new_conv = conv.copy()

    # linear part: 2*delta*(a_{k-j} - a_{k-i})
    # This is realised by two slice updates.
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a

    # quadratic correction
    d2 = delta * delta
    idx_2i = 2 * i
    idx_2j = 2 * j
    idx_ij = i + j
    if idx_2i < new_conv.size:
        new_conv[idx_2i] += d2
    if idx_2j < new_conv.size:
        new_conv[idx_2j] += d2
    if idx_ij < new_conv.size:
        new_conv[idx_ij] -= 2.0 * d2

    # Clip tiny negatives that can appear because of floating‑point noise.
    # This does not affect the exactness of the update.
    np.maximum(new_conv, 0.0, out=new_conv)
    return new_conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the evaluation score (2·n·max(conv)) for a convolution vector."""
    return 2.0 * n * float(conv.max())

"""
Simple simulated‑annealing / hill‑climbing optimizer for the step‑function
max‑convolution objective.

The routine starts from the best known sequence (if the global variable
``height_sequence_1`` is present) or from a uniform distribution.
It repeatedly performs cheap pairwise mass‑transfer moves that are
exactly reflected in the self‑convolution via an O(n) update.  Moves are
accepted if they improve the current score or with a probability that
decreases according to an annealing schedule.  Occasional block‑wise
perturbations and a global jitter (restart) are applied to escape local
minima.  The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Return the smallest power‑of‑two size that can hold a linear
    convolution of total length ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution ``a * a`` using a real FFT.
    Returns an array of length ``2*len(a)-1``."""
    n = a.shape[0]
    fft_len = _fft_len_for_len(2 * n - 1)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * n - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all zero -> uniform
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1] - 1          # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_onto_simplex(out)
    return out


def _apply_pair_transfer(a: np.ndarray, conv: np.ndarray,
                         i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving ``delta`` mass from index ``i``
    to index ``j``.  ``a`` is the *old* vector (unchanged), ``conv`` is the
    old self‑convolution.
    """
    n = a.shape[0]
    new_conv = conv.copy()

    # linear part: 2*delta*(a_{k-j} - a_{k-i})
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a

    # quadratic correction (exact)
    d2 = delta * delta
    idx_2i = 2 * i
    idx_2j = 2 * j
    idx_ij = i + j
    if idx_2i < new_conv.size:
        new_conv[idx_2i] += d2
    if idx_2j < new_conv.size:
        new_conv[idx_2j] += d2
    if idx_ij < new_conv.size:
        new_conv[idx_ij] -= 2.0 * d2

    # Clip tiny negatives from floating‑point noise

    return new_conv
