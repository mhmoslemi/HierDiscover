Improved search for a step‑function with a low self‑convolution peak.

Algorithm
---------
* Phase 1 – smooth exponential‑gradient descent (mirror descent) on the
  soft‑max approximation of the convolution peak.
  - Uses the exact soft‑max weight vector (normalised) for the gradient.
  - Gradient is obtained by a cross‑correlation, computed efficiently with one
    FFT of the current sequence and one FFT of the reversed soft‑max vector.
  - A momentum term (exponential moving average) stabilises the updates.
  - Step size η decreases linearly, while the soft‑max temperature β grows
    from 1 to a large value (≈400).  This “deterministic annealing” focuses the
    optimisation gradually on the true maximum.

* Phase 2 – stochastic local refinement (simulated annealing).
  - Pairwise mass‑transfer moves and random block perturbations are applied.
  - Acceptance follows the Metropolis rule with a linearly decreasing
    temperature.

* Restarts – if no improvement occurs for a while, the current best solution
  is jittered with multiplicative log‑normal noise and optimisation continues.

The routine works on any starting sequence found in the global variable

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    # Width at most 10 % of the vector (but at least 1)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

            max_c = conv.max()

            # soft‑max weights (properly normalised)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / L)
            else:
                w /= w_sum

            # gradient = 2 * cross‑correlation(w, a)
            rev_w = w[::-1]
            rev_w_fft = np.fft.rfft(rev_w, n=Nfft)
            grad_fft = fa * rev_w_fft
            grad_full = np.fft.irfft(grad_fft, n=Nfft)
            grad = 2.0 * grad_full[L - 1:L - 1 + n]  # length‑n slice

            # momentum smoothing
            grad_mavg = momentum * grad_mavg + (1.0 - momentum) * grad

            # exponentiated‑gradient update on the simplex
            a = a * np.exp(-eta * grad_mavg)
            a = np.clip(a, 1e-30, None)       # avoid exact zeros -> underflow
            a_sum = a.sum()
            a /= a_sum

            # occasional random block perturbation to inject diversity
            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)

            # reevaluate after the update
            fa = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(fa * fa, n=Nfft)[:L]
            cur_score = 2.0 * n * conv.max()

        # ----------------------------------------------------------
        # Phase 2 – stochastic local refinement (final part of budget)
        # ----------------------------------------------------------
        else:
            # linear temperature schedule for simulated annealing
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = max(1e-8, 0.1 * (1.0 - t_frac))

            if rng.random() < 0.85:                     # pairwise mass transfer
                i = rng.integers(0, n)
                if a[i] < 1e-12:
                    continue
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_delta = 0.35 * a[i]                  # move at most 35 % of donor
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta

                fa_new = np.fft.rfft(a_new, n=Nfft)
                conv_new = np.fft.irfft(fa_new * fa_new, n=Nfft)[:L]
                new_score = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if (new_score < cur_score) or (rng.random() < np.exp(-(new_score - cur_score) / temperature)):
                    a = a_new
                    conv = conv_new
                    cur_score = new_score

            else:                                       # global block jitter
                a = _random_block_perturb(a, rng)
                fa = np.fft.rfft(a, n=Nfft)
                conv = np.fft.irfft(fa * fa, n=Nfft)[:L]
                cur_score = 2.0 * n * conv.max()

        # ----------------------------------------------------------
        # Keep the best solution seen so far
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Soft restart when stuck for a while (no improvement)
        # ----------------------------------------------------------
        if now - last_improve > 40.0:
            # jitter the best solution and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_mavg = np.zeros_like(a)
            fa = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(fa * fa, n=Nfft)[:L]
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best vector (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
