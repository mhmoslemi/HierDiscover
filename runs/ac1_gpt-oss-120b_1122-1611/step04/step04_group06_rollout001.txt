"""
Propose a candidate step‑function by a hybrid of mirror‑descent (warm‑start)
and simulated‑annealing mass‑transfer local search.

* Warm‑start: if a previous construction “height_sequence_1” exists it is
  normalised to sum‑one; otherwise a uniform distribution is used.
* Stage 1 – a short run of exponentiated‑gradient (the code from the previous
  state‑of‑the‑art) quickly brings the score down to ≈1.514.
* Stage 2 – greedy / simulated‑annealing moves that transfer a small amount
  of mass from one index to another.  The linear effect on the convolution
  can be updated in O(n) using slice operations, so thousands of trial moves
  per second are feasible.  Whenever a move is accepted we recompute the
  full convolution via FFT to avoid drift.  The temperature and the maximal
  transfer fraction decay linearly with time, allowing coarse exploration at
  the start and fine‑grained polishing near the budget end.

The algorithm runs for the full time budget (≈1000 s) and returns the best
sequence seen.  All heights are clipped to the limits enforced by
``evaluate_sequence`` (0 ≤ height ≤ 1000).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for FFT‑based convolution
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent stage (borrowed from the previous solution)
# ----------------------------------------------------------------------
def _mirror_descent_stage(a: np.ndarray, budget_s: float, rng: np.random.Generator) -> np.ndarray:
    """A short mirror‑descent optimisation that quickly reaches the
    neighbourhood of the best known score (≈1.514)."""
    n = a.size
    fft_len_conv = _fft_len_for_n(n)

    # hyper‑parameters (more aggressive than the original code)
    eta_start = 0.9
    eta_end = 0.01
    beta_start = 0.5
    beta_end = 800.0
    block_perturb_prob = 0.01
    pair_move_prob = 0.07
    momentum = 0.9

    grad_prev = np.zeros_like(a)

    start_time = time.monotonic()
    deadline = start_time + min(120.0, budget_s * 0.12)   # at most 2 min or 12 % of budget
    while time.monotonic() < deadline:
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        # probability distribution over the 2·n‑1 lags
        w = np.exp(beta_start * (conv - cur_max))
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # correlation = gradient of the soft‑max surrogate
        n = a.size
        L = w_norm.size
        fft_len = 1 << ((L + n - 2 - 1).bit_length())
        fw = np.fft.rfft(w_norm, n=fft_len)
        fa_rev = np.fft.rfft(a[::-1], n=fft_len)
        conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
        start = n - 1
        grad = 2.0 * conv_full[start:start + n]

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # schedule
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # exponentiated‑gradient (mirror descent) update
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # occasional random block perturbation
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 12))
            start_blk = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a[start_blk:start_blk + width] *= factor
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # occasional pair‑wise mass transfer
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.random() * min(a[i], 0.2 * a[i])
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a /= a.sum()

    return a


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer stage
# ----------------------------------------------------------------------
def _local_search_stage(a: np.ndarray, budget_s: float, rng: np.random.Generator) -> np.ndarray:
    """Fine‑grained local search that moves a tiny amount of mass from one
    index to another.  The convolution update is performed in O(n) using
    slice operations.  The temperature and the maximal transfer fraction
    decay linearly with time."""
    n = a.size
    fft_len = _fft_len_for_n(n)
    conv = _conv_self(a, fft_len)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_a = a.copy()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)   # keep a small safety margin

    # schedule parameters
    max_frac_start = 0.45          # up‑front we may move up to 45 % of source mass
    max_frac_end = 0.001           # near the end we only move a few per‑thousand
    temp_start = 0.008             # initial temperature for uphill acceptances
    temp_end = 1e-7                # final temperature (essentially greedy)

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        # linear interpolation of hyper‑parameters
        max_frac = max_frac_start * (1.0 - elapsed / total) + max_frac_end * (elapsed / total)
        T = temp_start * (1.0 - elapsed / total) + temp_end * (elapsed / total)

        # pick a source index with non‑zero mass
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        # pick a distinct target index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # choose how much to transfer
        eps_frac = rng.random() * max_frac
        epsilon = a[i] * eps_frac
        if epsilon <= 0.0:
            continue

        # --- tentative convolution update (O(n)) ---
        conv_tmp = conv.copy()
        conv_tmp[i:i + n] -= 2.0 * epsilon * a
        conv_tmp[j:j + n] += 2.0 * epsilon * a
        new_max = conv_tmp.max()
        new_score = 2.0 * n * new_max

        # acceptance rule (greedy + simulated annealing)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            # apply the move to the true sequence
            a[i] -= epsilon
            a[j] += epsilon
            # numerical clean‑up
            if a[i] < 0.0:
                a[i] = 0.0
            a_sum = a.sum()
            if a_sum != 0.0:
                a /= a_sum
            else:
                a[:] = 1.0 / n

            # recompute the exact convolution (FFT) to keep precision
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                elapsed_best = time.monotonic() - start_time
                print(f"[{elapsed_best:6.2f}s] new best score = {best_score:.6f}")

        # occasional full random jitter to escape deep plateaus
        if rng.random() < 0.001:
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                elapsed_best = time.monotonic() - start_time
                print(f"[{elapsed_best:6.2f}s] jitter improved score = {best_score:.6f}")

    return best_a

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the vector (warm‑start from a previously found construction)
    # ------------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a0 = np.clip(cand, 0.0, None)
                s = a0.sum()
                a0 = a0 / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
                n = a0.size
            else:
                raise ValueError
        except Exception:
            n = default_n
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Stage 1 – mirror descent (quick descent to the known basin)
    # ------------------------------------------------------------------
    time_before_md = time.monotonic()
    a_md = _mirror_descent_stage(a0, budget_s, rng)
    elapsed_md = time.monotonic() - time_before_md
    print(f"Mirror‑descent stage finished in {elapsed_md:.2f}s, score ≈ "
          f"{2 * n * _conv_self(a_md, _fft_len_for_n(n)).max():.6f}")

    # ------------------------------------------------------------------
    # 3) Stage 2 – local simulated‑annealing refinement
    # ------------------------------------------------------------------
    total_budget = float(budget_s)
    time_used = time.monotonic() - time_before_md
    remaining_budget = max(0.0, total_budget - time_used)
    best_a = _local_search_stage(a_md, remaining_budget, rng)

    # ------------------------------------------------------------------
    # 4) Return the best sequence seen (clip to evaluate_sequence limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
