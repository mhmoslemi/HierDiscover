```
    best_score = 2.0*n*_conv_self(a, fft_len_conv).max()
    t = 0
    # parameters
    max_iters = 2000  # will break earlier if time runs out
    while time.monotonic() < deadline and t < max_iters:
        t += 1
        conv = _conv_self(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # Choose method
        if rng.random() < 0.6:
            # hard subgradient
            i_max = conv.argmax()
            # compute gradient from i_max
            grad = np.zeros_like(a)
            lo = max(0, i_max - (n - 1))
            hi = min(i_max, n - 1)
            idx = np.arange(lo, hi+1)
            grad[idx] = 2.0 * a[i_max - idx]
            eta = 0.5 / np.sqrt(t)
            a = a - eta * grad
        else:
            # softmax gradient
            # schedule for beta: start low, increase linearly up to 2000
            beta = 0.5 + (2000.0 - 0.5) * (t / max_iters)
            w = np.exp(beta * (conv - cur_max))
            sum_w = w.sum()
            a_rev = a[::-1]
            conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
            grad_part = conv_w_rev[n-1:n-1+n] / sum_w
            # raw gradient: exclude +a term
            grad = grad_part
            eta = 0.6 / np.sqrt(t)
            a = a - eta * grad
        # projection
        a = _project_onto_simplex(a)
        # random perturbations
        if rng.random() < 0.02:
            # block perturbation
            width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n-width+1)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[start_idx:start_idx+width] *= factor
            a = _project_onto_simplex(a)
        if rng.random() < 0.04:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0:
                max_delta = min(a[i], 0.2 * a[i])
                delta = rng.random() * max_delta
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_onto_simplex(a_candidate)
                # evaluate candidate quickly
                conv_cand = _conv_self(a_candidate, fft_len_conv)
                score_cand = 2.0 * n * conv_cand.max()
                if score_cand < cur_score:
                    a = a_candidate
        # occasionally re-enter random jitter if stuck
        if t % 200 == 0 and cur_score > best_score * 1.001:
            # small jitter
            jitter = np.exp(rng.normal(0.0, 0.1, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
    return best, best_score
```

The evaluation function  

```
score = 2·n·max(conv(a,a)) / (sum(a)²)
```

is *scale invariant* – multiplying the whole vector by a constant does not change the score.  
Therefore we can (and do) keep the vector on the probability simplex  
`{a ≥ 0, Σ a = 1}` and minimise the simpler objective  

```
F(a) = 2·n·max(conv(a,a))            ( Σa = 1 )
```

`max(conv)` is the maximum entry of the self‑convolution.  
It is a piece‑wise‑smooth function:  

```
max(conv) ≈ (1/β)·log Σ_i exp(β·conv_i)      (β → ∞)
```

A soft‑max with a growing temperature `β` gives a differentiable surrogate.
The gradient of the surrogate is

```
∂F/∂a = (conv_w_rev[n‑1 : n‑1+n]) / Σ w      where  w_i = exp(β·conv_i)
```

(no extra “+a” term is required – the original code added it by mistake,
which slows convergence).

The optimiser combines three ideas

1. **Exponentiated gradient / Adam on the simplex** – updates stay non‑negative
   and the projection onto the simplex is exact.
2. **Hard sub‑gradient steps** – occasionally use the true sub‑gradient of the
   current worst convolution index (`i* = argmax conv`).  
   For a fixed index `i*` the gradient is `g_j = 2·a_{i*‑j}` (zero outside the
   support).  This drives the algorithm directly at the current bottleneck.
3. **Random local moves** – block‑wise log‑normal perturbations and cheap
   pairwise mass transfers allow the search to jump out of shallow plateaus.

The search runs many **independent restarts** with different lengths
(`n ∈ {2000,3000,3809,5000,8000}`) and random initial points.
Each restart performs a few hundred cheap iterations; the best vector seen
overall is returned.  A final polishing phase runs the optimiser a little
longer on the current global best.

All heavy work (convolutions) is performed with FFTs, yielding
`O(n log n)` per iteration and comfortably fitting into the 1000 s time budget.

```python
"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().

The algorithm:
* works on the probability simplex (Σa = 1);
* repeatedly approximates max(conv) with a log‑sum‑exp (soft‑max);
* uses an Adam/Exponentiated‑gradient update based on the soft‑max
  gradient, **without the spurious “+a” term** present in the reference code;
* occasionally replaces the soft‑max gradient by the exact sub‑gradient
  of the current worst convolution index;
* injects random block‑perturbations and pairwise mass‑transfer moves;
* performs many short, independent restarts for several sequence lengths;
* finally fine‑tunes the best candidate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors `x` and `y` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return a vector equal to the sub‑gradient of max(conv) w.r.t. `a`
    (i.e. the gradient of the worst convolution entry).
    """
    n = a.size
    i_max = int(conv.argmax())
    # gradient g_j = 2·a_{i_max‑j} for indices that lie inside the array
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    # convolution of w with the reversed a
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    grad_part = conv_w_rev[a.size - 1 : a.size - 1 + a.size] / sum_w
    return grad_part


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    return _project_onto_simplex(out)


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Cheap move: transfer a random amount of mass from i to j."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = _project_onto_simplex(a2)
    # accept only if it improves the score
    conv = _conv_self(a2, fft_len_conv)
    if 2.0 * n * conv.max() < 2.0 * n * _conv_self(a, fft_len_conv).max():
        return a2
    return a


def _local_optimise(a: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a short, intensive optimisation of `a` (which already satisfies Σa=1).
    Returns the best vector found and its score (2·n·max(conv)).
    """
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state (kept simple – second moment is enough)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_vec = a.copy()
    best_score = 2.0 * n * _conv_self(a, fft_len_conv).max()

    # iteration limit – we also respect the global deadline
    max_iters = 500
    while t < max_iters and time.monotonic() < deadline:
        t += 1

        # --------------------------------------------------------------
        # 1) convolution and current score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # 2) choose gradient type
        # --------------------------------------------------------------
        if rng.random() < 0.6:
            # hard sub‑gradient of the current worst entry
            raw_grad = _subgradient_step(a, conv)
        else:
            # smooth max – β grows linearly during the short run
            beta = 0.5 + (2000.0 - 0.5) * (t / max_iters)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # 3) Adam update (additive) – we use a modest step size schedule
        # --------------------------------------------------------------
        lr = 0.7 / np.sqrt(t)          # decreasing learning rate
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                       # descent step
        a = _project_onto_simplex(a)       # stay on simplex

        # --------------------------------------------------------------
        # 4) occasional random moves
        # --------------------------------------------------------------
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.04:
            a = _pairwise_move(a, rng, fft_len_conv)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    evaluate_sequence() before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, evaluate it first
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                # bring to the simplex (scale does not matter for the score)
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cur = 2.0 * cand.size * _conv_self(cand,
                                                   _fft_len_for_len(2 * cand.size - 1)).max()
                if cur < best_score:
                    best_score = cur
                    best_vec = cand.copy()
                print(f"Initial global candidate score = {cur:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths we will explore – a mix of small/medium/large vectors
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 3809, 5000, 8000]

    # ------------------------------------------------------------------
    # Main search loop – many cheap restarts
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (biased towards larger ones, but keep variety)
        n = int(rng.choice(lengths, p=np.array([0.15, 0.2, 0.3, 0.2, 0.15]))
                 )
        # random start on the simplex
        a0 = rng.random(n)
        a0 /= a0.sum()

        # run a short local optimisation
        a_opt, score_opt = _local_optimise(a0, rng, deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})")

        # If time remains, try polishing the current global best
        if best_vec is not None and time.monotonic() < deadline - 0.5:
            # short fine‑tune (more iterations, higher β)
            a_polished, score_polished = _local_optimise(
                best_vec.copy(), rng, deadline)
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polished global best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Final safety clipping – evaluate_sequence will also clip,
    # but we do it here for completeness.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
```
