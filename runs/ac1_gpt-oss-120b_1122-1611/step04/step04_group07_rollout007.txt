"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The strategy is a two‑phase hybrid:

1. **Exploratory sampling** – generate a diverse pool of candidate
   shapes (uniform, linear ramps, triangles, quadratic, Gaussian,
   exponential, etc.) on a modest length (default 1024).  The fast
   FFT‑based convolution is used to obtain a cheap estimate of the
   objective; the best few candidates are kept.

2. **Hard‑max sub‑gradient descent** – starting from each shortlisted
   vector we perform mirror‑descent on the probability simplex using
   the true sub‑gradient of the max‑convolution (the indicator of the
   maximal lag).  The update is an exponentiated‑gradient step,
   automatically preserving non‑negativity and the unit sum.
   A simple schedule for the stepsize together with occasional
   random perturbations helps escape plateaus.  The best exact score
   (evaluated with the provided ``evaluate_sequence``) found before the
   time budget expires is returned.

The code contains only top‑level helper functions (no closures) and
relies on ``numpy`` and the already‑imported ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n``.
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def conv_self(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``.
    """
    if fft_len is None:
        fft_len = _fft_len_for_n(a.size)
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def max_conv(a: np.ndarray, fft_len: int | None = None) -> float:
    """
    Return the maximum entry of the self‑convolution of ``a``.
    """
    return float(conv_self(a, fft_len).max())


def generate_candidate_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a modest pool of structured candidate vectors of length ``n``.
    Each candidate is normalised to unit sum.
    """
    cand = []

    # 1. Uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. Linear ramps (increasing / decreasing)
    inc = np.arange(1, n + 1, dtype=np.float64)
    cand.append(inc / inc.sum())
    cand.append(inc[::-1] / inc.sum())

    # 3. Symmetric triangle (peak in the middle)
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = np.arange(half, 0, -1)
    tri = np.concatenate([left, right])[:n].astype(np.float64)
    cand.append(tri / tri.sum())

    # 4. Quadratic (parabolic) shape – high in the centre
    i = np.arange(n, dtype=np.float64)
    quad = (i + 1) * (n - i)
    cand.append(quad / quad.sum())

    # 5. Gaussian windows with a few width settings
    i_center = (n - 1) / 2.0
    for sigma_factor in (0.10, 0.20, 0.30, 0.50, 0.80):
        sigma = sigma_factor * n
        g = np.exp(-0.5 * ((np.arange(n) - i_center) / sigma) ** 2)
        cand.append(g / g.sum())

    # 6. One‑sided exponential decay (different rates)
    for alpha in (0.05, 0.10, 0.20, 0.40, 0.80):
        e = np.exp(-alpha * np.arange(n))
        cand.append(e / e.sum())

    # 7. Symmetric exponential decay about the centre
    for alpha in (0.05, 0.10, 0.20, 0.40, 0.80):
        e = np.exp(-alpha * np.abs(np.arange(n) - i_center))
        cand.append(e / e.sum())

    # 8. Small random perturbations of the above (adds diversity)
    for base in cand[:8]:               # take a subset to keep pool size modest
        noisy = base * np.exp(rng.normal(0.0, 0.06, size=n))
        noisy = np.clip(noisy, 0.0, None)
        noisy /= noisy.sum()
        cand.append(noisy)

    return cand


def subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of ``max(conv(a, a))`` with respect to ``a``.
    The gradient is 2·corr(delta, a) where delta is the indicator of the
    maximal convolution entries.
    """
    conv = conv_self(a)
    max_val = conv.max()
    delta = (conv == max_val).astype(np.float64)  # shape (2*n-1,)
    # correlation of delta with a in 'valid' mode gives length n
    grad = 2.0 * np.correlate(delta, a, mode="valid")
    return grad


def refine_by_subgradient(
    a0: np.ndarray,
    deadline: float,
    start_eta: float = 1.0,
    end_eta: float = 0.01,
    max_iters: int = 5000,
    rng: np.random.Generator | None = None,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the hard‑max sub‑gradient.
    Returns the best vector seen and its corresponding max‑convolution value.
    """
    if rng is None:
        rng = np.random.default_rng()

    a = a0.copy()
    n = a.size
    fft_len = _fft_len_for_n(n)

    best_a = a.copy()
    best_M = max_conv(a, fft_len)

    for it in range(max_iters):
        now = time.monotonic()
        if now >= deadline - 2.0:           # keep a safety margin
            break

        # linear schedule for stepsize
        eta = start_eta * (1.0 - it / max_iters) + end_eta * (it / max_iters)

        # sub‑gradient of the max‑convolution
        grad = subgradient_max_conv(a)

        # exponentiated‑gradient (mirror‑descent on the simplex)
        factor = np.exp(-eta * grad)
        a = a * factor
        # renormalise, clip tiny negatives that may appear due to round‑off
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional tiny random perturbation to escape shallow plateaus
        if rng.random() < 0.02:
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        cur_M = max_conv(a, fft_len)
        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()

    return best_a, best_M

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # ---- 0) Global time handling -------------------------------------------------
    start_time = time.monotonic()
    # Leave a few seconds for the outer harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ---- 1) Choose the length of the step function -------------------------------
    # If a previous construction exists we reuse its length, otherwise we pick 1024.
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
                n = prev.size
            else:
                raise ValueError
        except Exception:
            n = 1024
    else:
        n = 1024

    # ---- 2) Generate an initial pool of structured candidates --------------------
    pool = generate_candidate_pool(n, rng)

    # ---- 3) Quick evaluation of the pool (FFT based, no expensive calls) ---------
    fft_len = _fft_len_for_n(n)
    scores = []
    for vec in pool:
        cur_max = max_conv(vec, fft_len)
        est_score = 2.0 * n * cur_max          # sum(vec) == 1  -> 2*n*max
        scores.append(est_score)

    # Keep the 5 best vectors according to the cheap estimate
    best_indices = np.argsort(scores)[:5]
    best_candidates = [pool[i] for i in best_indices]

    # ---- 4) Refine each top candidate with hard‑max sub‑gradient descent ---------
    best_overall_vec = None
    best_overall_score = np.inf
    best_overall_exact = np.inf

    for idx, cand in enumerate(best_candidates):
        # local refinement
        refined_vec, refined_M = refine_by_subgradient(
            cand,
            deadline,
            start_eta=1.0,
            end_eta=0.02,
            max_iters=4000,
            rng=rng,
        )
        # cheap score for the refined vector
        refined_score_est = 2.0 * n * refined_M

        # occasional exact evaluation (still cheap compared to full evaluation)
        if refined_score_est < best_overall_score:
            exact = evaluate_sequence(refined_vec.tolist())
            if exact < best_overall_exact:
                best_overall_vec = refined_vec.copy()
                best_overall_score = refined_score_est
                best_overall_exact = exact
                elapsed = time.monotonic() - start_time
                print(
                    f"[{elapsed:6.2f}s] new overall best (exact) = {best_overall_exact:.6f}"
                )

        # also keep a copy of the best refined (even if not yet evaluated exactly)
        if refined_score_est < best_overall_score:
            best_overall_vec = refined_vec.copy()
            best_overall_score = refined_score_est

        # check time budget
        if time.monotonic() >= deadline - 2.0:
            break

    # ---- 5) If time permits, perform a final stochastic polishing run ----------
    # Start from the best vector found so far and run a few extra sub‑gradient steps.
    if best_overall_vec is not None and time.monotonic() < deadline - 1.0:
        extra_vec, _ = refine_by_subgradient(
            best_overall_vec,
            deadline,
            start_eta=0.5,
            end_eta=0.005,
            max_iters=2000,
            rng=rng,
        )
        # final exact check
        final_exact = evaluate_sequence(extra_vec.tolist())
        if final_exact < best_overall_exact:
            best_overall_vec = extra_vec.copy()
            best_overall_exact = final_exact
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] final polishing improved exact score to {best_overall_exact:.6f}"
            )

    # ----------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of ``evaluate_sequence``)
    # ----------------------------------------------------------------------
    if best_overall_vec is None:
        # fallback – return a simple uniform vector
        best_overall_vec = np.full(n, 1.0 / n)

    best_overall_vec = np.clip(best_overall_vec, 0.0, 1000.0)
    return [float(x) for x in best_overall_vec]
