def _project_simplex(v):
    # ensure non-negative after projection onto simplex.
    # Based on algorithm from Duchi et al. (2008).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.size
    Lw = w.size # = 2*n-1
    wr = w[::-1]
    conv_len = _next_pow2(n + Lw - 1) # = pow2(3n-2)
    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)
    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad

"""
propose_candidate
-----------------
A multi‑restart projected‑gradient / simulated‑annealing hybrid that
optimises a non‑negative step‑function a on the simplex
(sum(a)=1) to minimise   2·n·max(a∗a)   (the evaluate_sequence objective).
The method works with an additive gradient (sub‑gradient of a soft‑max
approximation of the max‑convolution) and mirrors the iterate onto the
simplex with a Duchi‑et‑al. projection. Random block jitter,
pairwise mass‑transfer moves and occasional restarts (with log‑normal
noise) help escape local minima.  The schedule sharpens a soft‑max
weight from a smooth average to the exact max, while the learning rate
decays linearly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries, clipped to non‑negative.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip possible tiny negatives caused by rounding errors.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a (full‑gradient).
    Implements   ∂/∂a_j Σ_k w_k·conv_k = 2·Σ_i a_i·w_{i+j}
    via a single FFT‑based correlation.
    """
    n = a.size
    Lw = w.size                         # = 2*n-1
    # reverse w for correlation via convolution
    wr = w[::-1]

    # length needed for a (len=n) + wr (len=Lw) - 1 = n+Lw-1 = 3n-2
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep only needed part

    # gradient_j = 2 * conv_full[Lw-1 - j]
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  O(n) algorithm (Duchi et al.).
    """
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise exactly
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)                # ≤20 % of the vector
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start : start + width] *= factor
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative coefficient list (step function) that
    minimises ``evaluate_sequence``.  The routine respects the time
    budget, prints progress, and returns the best sequence found as a
    plain Python list.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Determine problem size and an initial feasible vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a0 = np.clip(cand0, 0.0, None)
            s = a0.sum()
            if s > 0.0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = 4096
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # Pre‑compute FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    # fft_len_grad is computed inside _gradient_from_weights when needed

    # --------------------------------------------------------------
    # 2) Book‑keeping and hyper‑parameters
    # --------------------------------------------------------------
    # Current candidate (the one we iterate on)
    a_cur = a0.copy()

    # Best solution seen so far
    conv_best = _self_conv_fft(a0, fft_len_self)
    best_score = 2.0 * n * conv_best.max()
    best = a0.copy()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # tiny safety margin

    # Learning‑rate schedule (additive)
    eta_start = 0.70
    eta_end = 0.02

    # Soft‑max temperature schedule (smooth → hard)
    beta_start = 0.5
    beta_end = 1500.0
    smooth_phase = 0.80          # fraction of total time where we use soft‑max

    momentum = 0.85
    grad_prev = np.zeros_like(a_cur)

    # Random‑move probabilities
    prob_block = 0.04
    prob_pairwise = 0.08

    # Restart control
    last_improve = start_time
    restart_gap = 30.0            # seconds without improvement → restart
    jitter_sigma = 0.20           # log‑normal jitter after a restart

    # --------------------------------------------------------------
    # 3) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- compute self‑convolution and current score -----
        conv = _self_conv_fft(a_cur, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep global best -----
        if cur_score < best_score:
            best_score = cur_score
            best = a_cur.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule parameters -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 0.0

        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- construct weight vector w (soft‑max → hard max) -----
        if frac < smooth_phase:
            # Soft‑max; shift by cur_max for numerical stability
            w = np.exp(beta * (conv - cur_max))
        else:
            # Hard max (sub‑gradient). Use a small tolerance to collect
            # all near‑optimal entries.
            eps = max(1e-12, 0.001 * cur_max)
            active = np.where(conv >= cur_max - eps)[0]
            w = np.zeros_like(conv)
            if active.size > 0:
                w[active] = 1.0 / active.size

        # ----- gradient (full weighted sum) -----
        grad = _gradient_from_weights(a_cur, w)

        # Momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- additive projected‑gradient step -----
        a_next = a_cur - eta * grad
        a_next = _project_simplex(a_next)

        # ----- occasional random block jitter -----
        if rng.random() < prob_block:
            a_next = _random_block_perturb(a_next, rng)

        # ----- cheap pairwise mass‑transfer move (always accepted) -----
        if rng.random() < prob_pairwise:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a_next[i], 0.3 * a_next[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_next[i] -= delta
                    a_next[j] += delta
                    a_next = _project_simplex(a_next)

        a_cur = a_next

        # ----- restart when stuck -----
        if now - last_improve > restart_gap:
            # jitter the current best and restart from there
            a_cur = best.copy()
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a_cur = a_cur * noise
            a_cur = _project_simplex(a_cur)
            grad_prev = np.zeros_like(a_cur)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence (respecting evaluate_sequence limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
