    sums = set()

    for i in range(n):
        # check candidate
        conflict = False
        # check i+i
        if i+i in sums:

        # check with each existing j

            if i+j in sums:
                conflict = True
                break
        if conflict:
            continue
        # add i

        sums.add(i+i)

    return indices

"""
Hybrid optimizer for the “step‑function” evaluation problem.

The score to minimise (with Σ a_i = 1) is
        S(a) = 2·n·max( a ∗ a )
where ∗ denotes the ordinary linear convolution.
The algorithm combines several cheap heuristics:

* a fast exponentiated‑gradient (mirror‑descent) update that
  focuses on the K largest convolution entries (K is reduced
  from ≈10 to 1 as the optimisation proceeds);
* occasional random perturbations of a whole block,
  pairwise mass transfers and random permutations of the
  coordinates;
* a “restart” policy – if no improvement is seen for a few
  seconds the current iterate is replaced by a fresh random
  probability vector;
* once every few dozen iterations a tiny linear‑programming
  step (SciPy’s high‑performance “highs” solver) computes a
  vector g that maximises Σ g_i while keeping the convolution
  (a ∗ g) bounded by the current maximum.  A short convex
  mixture a ← (1‑t)·a + t·g (t≈0.2) usually lowers the peak.

All updates keep the vector non‑negative and renormalise it
to unit sum, which makes the evaluation scale‑invariant.
The method is deliberately stochastic and runs many restarts
within the allotted time budget, keeping the best sequence
encountered.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using the FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Correlation needed for the gradient of Σ_k w_k·(a∗a)_k.
    Returns a vector of length len(a)."""
    n = a.size
    L = w.size                      # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _lp_step(a: np.ndarray, M: float, rng: np.random.Generator) -> np.ndarray | None:
    """
    One LP refinement: maximise Σ g_i subject to (a∗g)_k ≤ M,
    g_i ≥ 0.  Returns the normalised optimiser g (sum(g)=1) or None
    if the LP fails or yields a zero vector.
    """
    n = a.size
    # Build the convolution matrix C (shape (2n‑1) × n)
    # C[k, j] = a_{k‑j} if 0 ≤ k‑j < n else 0
    rows = []
    data = []
    col_idx = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            col_idx.append(j)
            data.append(a[i])
    # Convert to dense matrix – small n makes this cheap
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    C[rows, col_idx] = data

    c = -np.ones(n)                     # minimise –Σ g_i  ⇔ maximise Σ g_i
    bounds = [(0.0, None)] * n
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve using the HiGHS interior‑point method (fast for our sizes)
    res = linprog(c, A_ub=C, b_ub=ub, bounds=bounds, method="highs",
                  options={"presolve": True, "time_limit": 5.0})
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s <= 0.0:
        return None
    return g / s


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (a probability vector)
    that (approximately) minimises ``evaluate_sequence`` before the
    time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose an initial length n and vector a (unit‑sum).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(n, 1.0 / n)
            else:
                raise ValueError
        except Exception:
            n = 256
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len_conv = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * np.max(_conv_self(best, fft_len_conv))
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (tuned for this problem)
    # ------------------------------------------------------------------
    eta_start = 0.9          # initial multiplicative‑gradient step size
    eta_end = 0.02           # final step size
    restart_gap = 8.0        # seconds without improvement → fresh random start
    block_perturb_prob = 0.05
    pair_move_prob = 0.12
    permute_prob = 0.03
    momentum = 0.9
    lp_interval = 45          # perform an LP refinement every ~45 iterations
    t_mix = 0.22              # mixing weight for LP step

    grad_prev = np.zeros_like(a)

    iteration = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current self‑convolution & score -----------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max     # Σa = 1 → score = 2·n·max(conv)

        # ----- keep the best -------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule for η and K -----------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        progress = elapsed / total
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # K decays from ~10 to 1 as optimisation proceeds
        K = max(1, int(round(10 * (1.0 - progress))))

        # ----- build weights that concentrate on the K largest conv entries -
        if K >= conv.size:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            idx = np.argpartition(conv, -K)[-K:]
            w = np.zeros_like(conv)
            w[idx] = 1.0 / K

        # ----- (soft) gradient of the surrogate -----------------------------
        grad = _gradient_correlation(a, w)

        # ----- momentum smoothing -------------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror descent) update ---------------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation -------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pair‑wise mass transfer ---------------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                max_delta = min(a[i], 0.3 * a[i])
                delta = rng.random() * max_delta
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = np.clip(a_candidate, 0.0, None)
                a_candidate /= a_candidate.sum()
                conv_cand = _conv_self(a_candidate, fft_len_conv)
                cand_score = 2.0 * n * conv_cand.max()
                if cand_score < cur_score:
                    a = a_candidate
                    cur_score = cand_score
                    cur_max = conv_cand.max()

        # ----- occasional permutation of coordinates ------------------------
        if rng.random() < permute_prob:
            perm = rng.permutation(n)
            a = a[perm]

        # ----- periodic linear‑programming refinement ----------------------
        if iteration % lp_interval == 0:
            M = cur_max
            g = _lp_step(a, M, rng)
            if g is not None:
                a_mix = (1.0 - t_mix) * a + t_mix * g
                a_mix = np.clip(a_mix, 0.0, None)
                a_mix /= a_mix.sum()
                # accept the mixture only if it does not increase the peak
                conv_mix = _conv_self(a_mix, fft_len_conv)
                mix_score = 2.0 * n * conv_mix.max()
                if mix_score <= cur_score:
                    a = a_mix
                    cur_score = mix_score
                    cur_max = conv_mix.max()

        # ----- restart if stuck ---------------------------------------------
        if now - last_improve > restart_gap:
            # fresh random start (Dirichlet with concentration 1)
            a = rng.random(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] random restart – best score {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clip to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
