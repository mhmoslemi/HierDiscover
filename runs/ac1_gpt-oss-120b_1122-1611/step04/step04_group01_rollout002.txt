def propose_candidate(seed=123, budget_s=1000.0):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    best_seq = None
    best_score = np.inf
    # pre-set list of lengths to try
    lengths = [2000, 3000, 4000, 5000]
    # main outer loop restart
    while time.monotonic() < deadline:
        # pick n randomly from lengths maybe with probability
        n = rng.choice(lengths)
        # init a random Dirichlet
        a = rng.exponential(scale=1.0, size=n)
        a = a / a.sum()
        # precompute FFT length for conv (2n-1) power of two maybe.
        fft_len = 1 << ((2*n - 2).bit_length())
        # Adam parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        # inner loop for random steps
        inner_deadline = time.monotonic() + min(30.0, deadline - time.monotonic())
        while time.monotonic() < inner_deadline:
            # compute conv
            fa = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.monotonic():.2f}s] New best {best_score:.6f} n={n}")
            # gradient: subgradient of max
            k = int(conv.argmax())
            # compute shift
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            shift = np.zeros_like(a)
            shift[valid] = a[idx[valid]]
            grad = a + shift  # gradient of conv_k
            # step
            t += 1
            beta1=0.9
            beta2=0.999
            eps=1e-8
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m/(1-beta1**t)
            v_hat = v/(1-beta2**t)
            # schedule step size linearly decreasing from 1.0 to 0.01 over outer deadline
            elapsed = time.monotonic() - (deadline - budget_s)
            total_time = budget_s
            eta = 1.0 * (1 - elapsed/total_time) + 0.01 * (elapsed/total_time)
            step = eta * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            # project onto simplex
            a = _project_onto_simplex(a)
            # occasional random perturbation
            if rng.random() < 0.02:
                width = rng.integers(1, max(2, n // 20))
                start = rng.integers(0, n-width+1)
                factor = np.exp(rng.normal(0.0, 0.3))
                a[start:start+width] *= factor
                a = np.clip(a,0, None)
                a = _project_onto_simplex(a)
            # occasional pair move
            if rng.random() < 0.05:
                i,j = rng.integers(0,n, size=2)

            # early exit if too many iterations or improvement stagnated.

"""
propose_candidate – multi‑restart exponentiated‑gradient search

The objective after fixing Σ a_i = 1 is     score = 2·n·max_k (a∗a)_k .
We minimise the ℓ∞ norm of the self‑convolution on the probability simplex.

Algorithm
---------
* Use many short restarts (different lengths n and random seeds) – the
  longer the sequence the lower the theoretical limit (≈1), but very long
  vectors are expensive for the FFT.
* Initialise each restart with a Dirichlet (exponential) random vector.
* Perform a projected *exponentiated‑gradient* descent on the simplex.
  For the current maximal convolution index k* we have
        ∂(a∗a)_k*/∂a_j = a_j + a_{k*−j}   (out‑of‑range terms are 0).
  The update is
        a ← a ⊙ exp( −η·g )   (⊙ = element‑wise product),
  then renormalise to Σ a = 1 (i.e. Euclidean projection onto the simplex).
* η is decreased slowly (linear schedule) and a few stochastic moves
  (block‑perturbation, pairwise mass transfer) are injected to escape
  plateaux.
* The best sequence seen over the whole budget is returned.

All operations are O(n log n) using a real FFT, so many iterations
fit comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (required for a real FFT)."""
    return 1 << ((length - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgrad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a∗a)_k   with respect to a.
    For the index k* attaining the maximum we have
        ∂(a∗a)_k*/∂a_j = a_j + a_{k*−j}   (zero outside the support).
    """
    n = a.size
    k = int(conv.argmax())
    # build the shifted copy of a: a_shift[j] = a[k−j] if 0≤k−j<n else 0
    idx = k - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    a_shift = np.zeros_like(a)
    a_shift[valid] = a[idx[valid]]
    return a + a_shift  # gradient of the maximal convolution entry


def _random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Sample a random point on the simplex using i.i.d. exponential variables."""
    x = rng.exponential(scale=1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.3))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _pairwise_mass_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small amount of mass from i to j and re‑project."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    return _project_onto_simplex(a2)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf
    best_time = None

    # ------------------------------------------------------------------
    # Optionally import a previously discovered construction.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                cand = cand / cand.sum()
                # evaluate once so we have a baseline
                conv = _conv_self_fft(cand, _fft_len_for_len(2 * cand.size - 1))
                score = 2.0 * cand.size * conv.max()
                best_seq = cand.copy()
                best_score = float(score)
                print(f"[init] Imported previous best score = {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter ranges (tuned experimentally)
    # ------------------------------------------------------------------
    length_candidates = [1800, 2500, 3200, 3809, 4600]   # various n values
    min_eta = 0.015   # final learning rate
    max_eta = 0.85    # initial learning rate (exponentiated‑gradient factor)
    restart_gap = 30.0   # seconds without improvement → force a restart

    # ------------------------------------------------------------------
    # Main search loop – many independent restarts
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (occasionally vary)
        n = int(rng.choice(length_candidates))

        # initialise a random point on the simplex
        a = _random_dirichlet(rng, n)

        # FFT length for the self‑convolution (power‑of‑two)
        fft_len = _fft_len_for_len(2 * n - 1)

        # Adam‑style moments – they stabilise the noisy sub‑gradient
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t = 0

        # schedule bookkeeping
        start_restart = time.monotonic()
        last_improve = start_restart

        # inner optimisation for this restart
        while time.monotonic() < deadline:
            # ---- 1) evaluate current objective ---------------------------------
            conv = _conv_self_fft(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep global best
            if cur_score < best_score:
                best_score = float(cur_score)
                best_seq = a.copy()
                best_time = time.monotonic()
                last_improve = best_time
                elapsed = best_time - start_restart
                print(f"[{elapsed:6.2f}s] new global best score = {best_score:.6f} (n={n})")

            # ---- 2) sub‑gradient of the maximal convolution entry ---------------
            g = _subgrad_max_conv(a, conv)

            # ---- 3) Adam update (applied to the exponentiated‑gradient rule) --
            t += 1
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # linear learning‑rate schedule over the whole budget
            now = time.monotonic()
            used = now - start_restart
            total = deadline - start_restart
            eta = max_eta * (1.0 - used / total) + min_eta * (used / total)

            # exponentiated‑gradient step (mirror descent w.r.t. KL)
            step = eta * m_hat / (np.sqrt(v_hat) + eps)
            # keep the exponent safe
            log_a = np.log(np.maximum(a, 1e-16)) - step
            a = np.exp(log_a)
            a = _project_onto_simplex(a)

            # ---- 4) occasional stochastic moves --------------------------------
            # block perturbation (helps escape flat plateaus)
            if rng.random() < 0.018:
                a = _random_block_perturb(a, rng)

            # cheap pairwise mass transfer
            if rng.random() < 0.06:
                a2 = _pairwise_mass_transfer(a, rng)
                # accept if it improves the objective (cheap check)
                conv2 = _conv_self_fft(a2, fft_len)
                score2 = 2.0 * n * conv2.max()
                if score2 < cur_score:
                    a = a2
                    cur_score = score2   # keep local variable up to date

            # ---- 5) restart if we have been stagnant for a while --------------
            if now - last_improve > restart_gap:
                # jitter the current global best and continue from there
                if best_seq is not None:
                    a = best_seq.copy()
                else:
                    a = _random_dirichlet(rng, n)
                jitter = np.exp(rng.normal(0.0, 0.12, size=n))
                a = a * jitter
                a = _project_onto_simplex(a)

                # reset Adam statistics
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                last_improve = now
                elapsed = now - start_restart
                print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

            # optional early exit from this restart (to give time for new n)
            if now - start_restart > 120.0:   # 2 min per restart is enough
                break

    # ------------------------------------------------------------------
    # Return the best sequence found (clipped to the limits used by
    # evaluate_sequence).  If nothing was found we return an empty list.
    # ------------------------------------------------------------------
    if best_seq is None:
        return []
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
