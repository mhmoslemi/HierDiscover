"""
propose_candidate
-----------------

Exponentiated‑gradient optimisation on the probability simplex.
The goal is to flatten the self‑convolution a*a (the “sum‑distribution”) so
that its maximum is as close as possible to the average value 1/(2n‑1).
The algorithm works directly on the simplex (∑a=1, a≥0) by

* a smooth approximation of the max‑norm  fβ(a)= (1/β)·log∑exp(β·conv(a))
* gradient of fβ obtained from two FFT‑based convolutions
* multiplicative (exponentiated) updates a←a·exp(−η·∇fβ) and renormalisation
* occasional random block‑perturbations and cheap pair‑wise mass transfers
* a soft restart with jitter when no improvement is seen for a while.

All operations are O(n log n) and the routine runs for the whole time budget
(≈1000 s).  The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (suitable FFT size)."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` via real FFT.
    Result length = len(x)+len(y)-1 (truncated to that)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex."""
    n = a.size
    # block width up to 10 % of the length (at least 1)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:          # should never happen, but guard against divide‑by‑0
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _pairwise_mass_move(a: np.ndarray, rng: np.random.Generator,
                        fft_len: int, n: int) -> np.ndarray:
    """Cheap local search: move a random fraction of mass from i to j."""
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] <= 0.0:
        return a
    # at most 30 % of the source mass may be moved
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    # re‑normalise (the total mass is unchanged, but numerical errors may appear)
    cand_sum = cand.sum()
    if cand_sum == 0.0:
        cand[:] = 1.0 / n
    else:
        cand /= cand_sum
    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose problem size and initial point
    # ------------------------------------------------------------------
    # If a previous construction exists, start from it; otherwise use uniform.
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
                raise ValueError
        except Exception:
            init = None
    else:
        init = None

    # Desired length – a few thousand gives a good trade‑off between quality
    # and per‑iteration cost.
    default_n = 4000
    if init is not None:
        a = np.clip(init, 0.0, 1000.0)
        n = a.size
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (once)
    # ------------------------------------------------------------------
    fft_len_conv = _fft_len_for_len(2 * n - 1)          # for a*a
    fft_len_grad = _fft_len_for_len(3 * n - 2)          # for w * a_rev

    # ------------------------------------------------------------------
    # 3) Book‑keeping (best solution seen so far)
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len_conv)
    best = a.copy()
    best_score = 2.0 * n * conv.max()                  # sum(a)=1 ⇒ denominator =1
    best_time = time.monotonic()
    start_time = best_time
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Schedules for the two main hyper‑parameters
    # ------------------------------------------------------------------
    beta_start = 0.5          # smoothing of the max‑norm (soft‑max temperature)
    beta_end = 5000.0
    eta_start = 1.0           # initial multiplicative‑gradient step size
    eta_end = 0.001

    # ------------------------------------------------------------------
    # 5) Misc. probabilities
    # ------------------------------------------------------------------
    block_perturb_prob = 0.03
    pair_move_prob = 0.12
    restart_gap = 25.0        # seconds without improvement → restart

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < (start_time + max(0.0, float(budget_s) - 5.0)):
        # ----- 6.1) Current convolution & score -----------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # sum(a)=1 → denominator=1

        # ----- 6.2) Keep the best found -------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- 6.3) Time‑dependent schedules ---------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = budget_s - 5.0
        # linear interpolation, clamped to [0,1]
        frac = min(max(elapsed / total, 0.0), 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # ----- 6.4) Soft‑max weights for the max‑norm ------------------
        # shift by cur_max to avoid overflow (all exponents ≤ 0)
        w = np.exp(beta * (conv - cur_max))
        sum_w = w.sum()
        if sum_w == 0.0:      # numerical safety
            w[:] = 1.0
            sum_w = w.sum()

        # ----- 6.5) Gradient of the smooth max (two FFT convolutions) -
        a_rev = a[::-1]
        conv1 = _conv_fft(w, a_rev, fft_len_grad)
        conv2 = _conv_fft(w[::-1], a_rev, fft_len_grad)
        # slice the part that corresponds to d fβ / d a
        seg = slice(n - 1, n - 1 + n)
        grad_raw = (conv1[seg] + conv2[seg]) / sum_w

        # Remove the constant component (does not affect simplex after renorm)
        grad_centered = grad_raw - grad_raw.mean()

        # ----- 6.6) Exponentiated‑gradient update ----------------------
        # multiplicative factor = exp(−η·gradient); clip to avoid under/overflow
        delta = -eta * grad_centered
        delta = np.clip(delta, -12.0, 12.0)      # limits exp to ~6·10⁻⁶ … 1.6·10⁵
        a *= np.exp(delta)
        # re‑normalise onto the simplex (sum to 1)
        a_sum = a.sum()
        if a_sum == 0.0:                # safeguard – re‑initialise uniformly
            a[:] = 1.0 / n
        else:
            a /= a_sum

        # ----- 6.7) Occasional random block perturbation --------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- 6.8) Cheap pairwise mass‑transfer move ------------------
        if rng.random() < pair_move_prob:
            cand = _pairwise_mass_move(a, rng, fft_len_conv, n)
            # compute the score of the candidate (cheap, re‑use cached conv?)
            conv_cand = _conv_self(cand, fft_len_conv)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score:
                a = cand
                conv = conv_cand
                cur_max = conv_cand.max()
                cur_score = cand_score

        # ----- 6.9) Soft restart when progress stalls -----------------
        if now - last_improve > restart_gap:
            # jitter the current best to escape a flat region
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            a = best * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – current best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best vector found (clip to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
