def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # shift to ensure sum >0; if all zero produce uniform.
    if np.allclose(v,0):
        return np.full(n, 1.0/n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function using a sub‑gradient method on the
max‑convolution objective.

The algorithm works directly on the probability simplex (a_i ≥ 0,
Σ a_i = 1).  At each iteration we

1. compute the full linear convolution a*a,
2. locate the index k* where the convolution attains its maximum,
3. build a sub‑gradient of the max‑convolution (which is 2·a_{k*‑i}
   for all valid i, with a single‑count on the diagonal),
4. take a momentum‑augmented gradient step,
5. project back onto the simplex.

Occasional random Dirichlet perturbations and timed restarts help
to escape plateaus.  The best sequence seen during the allotted
budget (≈ 1000 s) is returned.  All numbers are clipped to the limits
enforced by ``evaluate_sequence`` before returning.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the standard O(n log n) algorithm.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # If already feasible, just return the clipped version.
    if v.min() >= 0 and np.isclose(v.sum(), 1.0):
        return np.clip(v, 0.0, None)

    # Sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # Find the largest ρ satisfying u_ρ - (cssv_ρ / ρ) > 0.
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would be projected to zero → uniform distribution.
        return np.full(n, 1.0 / n)

    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise to exactly 1.
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return w / s
