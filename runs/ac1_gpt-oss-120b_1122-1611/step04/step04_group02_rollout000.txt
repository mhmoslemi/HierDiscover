def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector `a` using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax_gradient(a: np.ndarray, beta: float, fft_len: int):
    """
    Compute the self‑convolution of `a`, the soft‑max weights and the gradient
    of the soft‑max approximation of max(conv) w.r.t. `a`.

    Returns
    -------
    conv : np.ndarray
        Full convolution a*a (length 2·n‑1).
    grad : np.ndarray
        Gradient vector of length n.
    """
    n = a.size
    # ------- convolution a*a -------
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    # ------- soft‑max weights -------
    max_c = conv_full.max()
    # subtract max to avoid overflow; exponentiate
    w = np.exp(beta * (conv_full - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:                       # numerically impossible, but safe
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ------- gradient via convolution a * w_rev -------
    w_rev = w[::-1]                         # reversed weights
    fw_rev = np.fft.rfft(w_rev, n=fft_len)
    # convolution length = n + (2n‑1) – 1 = 3n‑2
    grad_full = np.fft.irfft(fa * fw_rev, n=fft_len)[: 3 * n - 2]

    # gradient_i = 2 * grad_full[2·n‑2‑i]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * grad_full[idx]

    return conv_full, grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _upsample_to_power_of_two(a: np.ndarray, max_len: int = 4096) -> np.ndarray:
    """
    Upsample `a` by linear interpolation to the next power‑of‑two length,
    but never exceed `max_len`.  Returns a new normalised vector.
    """
    current = a.size
    # target length = smallest power of two ≥ current, capped by max_len
    target = 1 << ((current - 1).bit_length())
    target = min(target, max_len)
    if target == current:
        return a.copy()
    x_old = np.arange(current)
    x_new = np.linspace(0, current - 1, target)
    a_up = np.interp(x_new, x_old, a)
    s = a_up.sum()
    if s == 0.0:
        a_up[:] = 1.0 / target
    else:
        a_up /= s
    return a_up
