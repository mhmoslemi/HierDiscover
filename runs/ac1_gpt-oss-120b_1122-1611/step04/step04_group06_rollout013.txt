def _self_convolution_max(a):
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, n=fft_len)

def _gaussian_seq(n, alpha):
    sigma = alpha * n
    i = np.arange(n)
    mid = (n-1)/2.0
    a = np.exp(-((i - mid)**2) / (2.0 * sigma**2))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a_sum
    return a

def _window(seq_type, n):
    i = np.arange(n)
    if seq_type == 'hann':
        w = 0.5 * (1 - np.cos(2*np.pi*i/(n-1)))
    elif seq_type == 'hamming':
        w = 0.54 - 0.46*np.cos(2*np.pi*i/(n-1))
    elif seq_type == 'blackman':
        w = 0.42 - 0.5*np.cos(2*np.pi*i/(n-1)) + 0.08*np.cos(4*np.pi*i/(n-1))
    elif seq_type == 'semicircle':
        x = (2*i/(n-1) - 1)  # between -1 and 1
        w = np.sqrt(1 - x**2)
    else:
        raise
    w = np.clip(w, 0, None)
    return w / w.sum()
```

"""
Optimising `evaluate_sequence` with a simple analytic family.

The objective can be written as

    score = 2·n·max(conv(a,a)) / (∑a)².

Scaling `a` does not change the score, so we may assume ∑a = 1.
If we take a power‑law step function

    a_i ∝ (i+1)ᵖ,   i = 0,…,n‑1   (p ≥ 1)

then the self‑convolution at the centre (index n‑1) is

    conv[n‑1] = (p+1)²·B(p+1,p+1) / n   (B = beta function).

Consequently

    score = 2·(p+1)²·B(p+1,p+1),

which is independent of `n` and monotonically decreasing with `p`.
For p = 3 the bound is 2·(4)²·B(4,4) = 0.22857…,
far below the target 1.5025.

The implementation below picks a length `n` (the length of the
previous best construction if available, otherwise a safe default),
uses exponent `p = 3`, normalises the vector to sum to one and returns it.
A quick FFT‑based check is printed so that the achieved score can be
verified before the function returns.
"""

import time
import numpy as np

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _max_self_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)) using an FFT of the appropriate length."""
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(conv.max())

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Build a power‑law step function a_i = (i+1)^p (p = 3) of length ``n``.
    The vector is normalised to unit sum, clipped to the limits enforced
    by ``evaluate_sequence`` and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)          # only for possible future use

    # --------------------------------------------------------------
    # 1) Choose a length.
    #    Re‑use the length of the previously found construction if it
    #    exists; otherwise fall back to a safe default.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
        except Exception:
            n = 4000
    else:
        n = 4000                     # a few thousand entries give a stable bound

    # --------------------------------------------------------------
    # 2) Build the power‑law sequence.
    # --------------------------------------------------------------
    p = 3                                            # exponent (>=2 gives huge gain)
    idx = np.arange(1, n + 1, dtype=np.float64)     # 1,…,n
    a = idx ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a_sum                               # normalise to unit sum

    # --------------------------------------------------------------
    # 3) Quick sanity‑check (FFT based, cheap) and logging.
    # --------------------------------------------------------------
    start = time.monotonic()
    max_conv = _max_self_convolution(a)
    score = 2.0 * n * max_conv
    elapsed = time.monotonic() - start
    print(f"[{elapsed:.3f}s] power‑law (p={p}) sequence of length {n}")
    print(f"    max(conv) = {max_conv:.8e}   →   score = {score:.6f}")

    # --------------------------------------------------------------
    # 4) Clamp to the limits enforced by ``evaluate_sequence`` and

    a = np.clip(a, 0.0, 1000.0)                     # protects against over‑flows
