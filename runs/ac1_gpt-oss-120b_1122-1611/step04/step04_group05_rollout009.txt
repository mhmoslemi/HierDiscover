def generate_power_shape(n, e):
    idx = np.arange(n, dtype=np.float64) + 0.5
    denom = idx * (n - idx)
    a = denom ** (-e)
    a_sum = a.sum()
    if a_sum == 0: return np.full(n, 1.0/n)
    return a / a_sum

def generate_exp_shape(n, c):
    idx = np.arange(n)
    dist = np.minimum(idx, n-1-idx)
    a = np.exp(-c * dist)
    a_sum = a.sum()
    return a / a_sum

def mirror_descent_optimize(a, time_budget, n, start_time):

    a = a.copy()
    ff_len = _fft_len_for_n(n)

    best = a.copy()
    best_score = compute_score(a)

    grad_prev = np.zeros_like(a)
    restart_gap=30
    last_improve = time.monotonic()

    while time.monotonic() < start_time + time_budget:
        # compute conv
        conv = _conv_self(a, ff_len)
        cur_max = conv.max()
        cur_score = 2.0*n*cur_max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:.2f}s] new best score = {best_score:.6f}")

        # schedule times
        elapsed = time.monotonic() - start_time
        total = time_budget
        # schedule linear as before, maybe use exponent for beta
        eta = 0.8 * (1 - elapsed/total) + 0.02 * (elapsed/total) # example
        beta = 0.5 + (2000 - 0.5) * (elapsed/total) # increase to high

        # weighted gradient
        max_conv = cur_max
        w = np.exp(beta * (conv - max_conv))
        a_rev = a[::-1]
        L = (2*n-1) + n - 1
        fft_len = _fft_len_for_len(L)
        fw = np.fft.rfft(w, n=fft_len)
        fr = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(fw*fr, n=fft_len)
        grad = 2.0 * grad_full[n-1:n-1+n] - a * w[2*np.arange(n)]

        # momentum
        momentum = 0.95
        grad = momentum * grad_prev + (1 - momentum) * grad
        grad_prev = grad

        # update
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0, None)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0/n)

        # optional random perturbations
        if np.random.rand() < 0.03:
            a = _random_block_perturb(a, np.random.default_rng())
        if np.random.rand() < 0.05:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.01 * min(a[i], a[j])
                if delta>0:
                    a[i] -= delta
                    a[j] += delta
                    a /= a.sum()

        if time.monotonic() - last_improve > restart_gap:
            a = best.copy()
            # jitter
            noise = np.exp(np.random.default_rng().normal(0.0, 0.1, size=n))
            a = a * noise
            a = np.clip(a, 0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic()-start_time:.2f}s] restart – best score {best_score:.6f}")

    return best, best_score

def compute_gradient(a, conv, beta, fft_len):
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    a_rev = a[::-1]
    L = (2*a.size - 1) + a.size - 1

def mirror_descent_optimize(a0, time_budget, n, start_time, rng):
    a = a0.copy()
    ff_len = _fft_len_for_n(n)
    best = a.copy()
    best_score = compute_score(a)   # use compute_score (which recomputes conv)
    grad_prev = np.zeros_like(a)
    last_improve = time.monotonic()
    restart_gap = 30.0
    # hyperparams
    eta_start = 1.0
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 1500.0
    momentum = 0.95
    block_prob = 0.04
    pair_prob = 0.07

    while time.monotonic() < start_time + time_budget:
        # compute conv and score
        conv = _conv_self(a, ff_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max  # since sum=1
        # Update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:.2f}s] new best (refine) score = {best_score:.6f}")

        # schedule parameters
        elapsed = time.monotonic() - start_time
        total = time_budget
        t_frac = min(elapsed / total, 1.0)
        eta = eta_start * (1 - t_frac) + eta_end * t_frac
        beta = beta_start + (beta_end - beta_start) * t_frac

        # compute gradient
        grad = compute_gradient(a, conv, beta, ff_len)  # may need using appropriate fft length

        # momentum
        grad = momentum * grad_prev + (1 - momentum) * grad
        grad_prev = grad

        # exponentiated gradient update
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        # random block perturb
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        # pairwise mass transfer
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.random() * min(a[i], 0.3 * a[i])
                if delta > 0:
                    a[i] -= delta
                    a[j] += delta
                    # renormalize
                    a = np.clip(a, 0.0, None)
                    a = a / a.sum()

        # restart if stuck
        if time.monotonic() - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic()-start_time:.2f}s] restart – best score {best_score:.6f}")

    return best, best_score

"""
Propose a low‑score step‑function.

The program consists of three stages:

1. **Initialisation** – use the arcsine distribution (the current state‑of‑the‑art)
   or a previously‑found sequence (global ``height_sequence_1``) as a starter.

2. **Shape sweep** – enumerate a few simple parametric families
   (power‑law and exponential edge‑concentrated shapes) on the same length.
   The best candidate from this cheap sweep is kept.

3. **Mirror‑descent refinement** – a soft‑max based exponentiated‑gradient
   optimiser (similar to the reference code) is run from the best shape.
   Learning‑rate and soft‑max temperature are linearly scheduled, momentum
   is added, and occasional random block perturbations / pairwise mass
   transfers help escape plateaus.  Restarts are triggered when no improvement
   is seen for a while.

The whole search respects the 1000 s wall‑time limit and prints progress.
The returned list of heights is clipped to the limits enforced by
``evaluate_sequence`` (0 ≤ x ≤ 1000).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)

    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def compute_score(a: np.ndarray) -> float:
    """Score = 2 * n * max(conv(a,a)) / (sum(a)**2).  ``a`` may be un‑normalised."""
    n = a.size
    if n == 0:
        return float("inf")
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_self(a, _fft_len_for_n(n))
    return float(2.0 * n * conv.max() / (s * s))


def generate_power_shape(n: int, e: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))**(-e)   (edges are favoured when e>0.5)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    denom = idx * (n - idx)
    a = denom ** (-e)
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / a_sum


def generate_exp_shape(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c * min(i, n‑1‑i)).  c=0 → uniform."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)
    a = np.exp(-c * dist.astype(np.float64))
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / a_sum


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Approximate sub‑gradient of max(conv) using a soft‑max weighting.
    Returns a vector of size len(a).
    """
    n = a.size
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))                # w_k = exp(beta*(c_k - max))
    a_rev = a[::-1]

    # Convolution w * a_rev (linear) – length = (2n‑1)+(n‑1)-1 = 3n‑2
    L = (2 * n - 1) + n - 1
    fft_len = _fft_len_for_len(L)

    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(a_rev, n=fft_len)
    grad_full = np.fft.irfft(fw * fr, n=fft_len)

    # slice the part that corresponds to the n gradient components
    grad = 2.0 * grad_full[n - 1 : n - 1 + n] - a * w[2 * np.arange(n)]

    return grad


def mirror_descent_optimize(
    a_start: np.ndarray,
    time_budget: float,
    total_start: float,
    rng: np.random.Generator,
    target_score: float = 1.5025,
) -> tuple[np.ndarray, float]:
    """
    Refine ``a_start`` with an exponentiated‑gradient (mirror descent)
    optimiser that uses a soft‑max approximation of the max‑norm.
    Returns the best vector seen and its score.
    """
    n = a_start.size
    ff_len = _fft_len_for_n(n)
    a = a_start.copy()
    best = a.copy()
    best_score = compute_score(a)

    grad_prev = np.zeros_like(a)
    last_improve = time.monotonic()
    restart_gap = 30.0                      # seconds without improvement → restart
    momentum = 0.94

    # schedules (linear)
    eta_start, eta_end = 1.0, 0.02
    beta_start, beta_end = 0.5, 2000.0
    block_prob = 0.04
    pair_prob = 0.07

    while time.monotonic() < total_start + time_budget:
        # ----- current score
        conv = _conv_self(a, ff_len)
        cur_score = 2.0 * n * conv.max()           # sum(a) == 1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(
                f"[{last_improve - total_start:6.2f}s] new best (refine) score = {best_score:.6f}"
            )
            if best_score <= target_score:
                break

        # ----- time‑dependent hyper‑parameters
        elapsed = time.monotonic() - total_start
        frac = min(elapsed / time_budget, 1.0)
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- soft‑max gradient
        grad = _gradient_softmax(a, conv, beta)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror‑descent) update
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # ----- occasional random block perturbation
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pair‑wise mass transfer
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # move a small fraction of mass from i to j
                delta = rng.random() * min(a[i], 0.3 * a[i])
                if delta > 0.0:
                    a[i] -= delta
                    a[j] += delta
                    a = np.clip(a, 0.0, None)
                    a /= a.sum()

        # ----- soft restart when stuck
        if time.monotonic() - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = time.monotonic()
            print(
                f"[{last_improve - total_start:6.2f}s] restart – best score {best_score:.6f}"
            )

    return best, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.
    """
    start_time = time.monotonic()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine length and an initial distribution
    # ------------------------------------------------------------------
    DEFAULT_N = 4000
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a0 = np.clip(cand, 0.0, None)
                s = a0.sum()
                a0 = a0 / s if s > 0.0 else np.full(n, 1.0 / n, dtype=np.float64)
            else:
                raise ValueError
        except Exception:
            n = DEFAULT_N
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = DEFAULT_N
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # arcsine shape (the current SOTA starter)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = arcsine / arcsine.sum()

    # pick the better of {previous candidate, arcsine}
    score_arc = compute_score(arcsine)
    score_cand = compute_score(a0)
    if score_arc < score_cand:
        a = arcsine.copy()
        best_score = score_arc
    else:
        a = a0.copy()
        best_score = score_cand

    best = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Cheap parametric shape sweep (≈ 15 % of the budget)
    # ------------------------------------------------------------------
    shape_budget = min(150.0, budget_s * 0.15)
    shape_deadline = start_time + shape_budget

    # power‑law family  (exponent e controls edge concentration)
    e_vals = np.linspace(0.30, 1.30, 21)          # coarse grid
    for e in e_vals:
        if time.monotonic() > shape_deadline:
            break
        cand = generate_power_shape(n, e)
        sc = compute_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            print(
                f"[{time.monotonic() - start_time:6.2f}s] better power e={e:.3f} → {best_score:.6f}"
            )
    # fine‑grained refinement around the best e (if time permits)
    if time.monotonic() < shape_deadline:
        # locate the best e from the coarse sweep
        best_e = min(e_vals, key=lambda ee: compute_score(generate_power_shape(n, ee)))
        fine_es = np.linspace(max(0.1, best_e - 0.15), best_e + 0.15, 15)
        for e in fine_es:
            if time.monotonic() > shape_deadline:
                break
            cand = generate_power_shape(n, e)
            sc = compute_score(cand)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
                print(
                    f"[{time.monotonic() - start_time:6.2f}s] fine power e={e:.3f} → {best_score:.6f}"
                )

    # exponential edge‑decay family
    c_vals = np.linspace(0.0, 5.0, 21)
    for c in c_vals:
        if time.monotonic() > shape_deadline:
            break
        cand = generate_exp_shape(n, c)
        sc = compute_score(cand)
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            print(
                f"[{time.monotonic() - start_time:6.2f}s] better exp c={c:.3f} → {best_score:.6f}"
            )
    # fine‑grained refinement for exponential parameter
    if time.monotonic() < shape_deadline:
        best_c = min(c_vals, key=lambda cc: compute_score(generate_exp_shape(n, cc)))
        fine_cs = np.linspace(max(0.0, best_c - 0.5), best_c + 0.5, 15)
        for c in fine_cs:
            if time.monotonic() > shape_deadline:
                break
            cand = generate_exp_shape(n, c)
            sc = compute_score(cand)
            if sc < best_score:
                best = cand.copy()
                best_score = sc
                print(
                    f"[{time.monotonic() - start_time:6.2f}s] fine exp c={c:.3f} → {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3) Mirror‑descent refinement (uses the remaining budget)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.monotonic() - start_time) - 5.0  # 5 s safety margin
    if remaining > 5.0:
        best, best_score = mirror_descent_optimize(
            best, remaining, start_time, rng, target_score=1.5025
        )

    # ------------------------------------------------------------------
    # 4) Final post‑processing
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    print(f"[{time.monotonic() - start_time:6.2f}s] final best score = {best_score:.6f}")
    return [float(x) for x in best]
