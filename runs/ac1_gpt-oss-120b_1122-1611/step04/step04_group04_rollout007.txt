"""
propose_candidate
-----------------
Hybrid optimisation for the L∞ self‑convolution problem.

* Phase 1 – fast exponentiated‑gradient minimisation of a soft‑max
  approximation of the convolution peak (softmax temperature is linearly
  increased, learning‑rate is decreased).  Momentum is applied and a
  small block‑jitter is injected to escape plateaus.

* Phase 2 – precise greedy hill‑climbing.  At each step a random pair
  (i, j) of indices is selected, a tiny amount of mass δ is moved from
  i to j and the convolution is updated *exactly* (the linear term with
  factor 2 δ and the quadratic δ² correction are applied via simple
  slice operations).  Moves that improve the current score are kept; a
  simple simulated‑annealing acceptance rule allows occasional
  uphill moves.

The algorithm starts from a mixture of the uniform distribution and an
“arcsine‑shaped’’ distribution (large values near the ends, small in the
middle) – this shape is known to flatten the self‑convolution.  The best
vector discovered during the whole run is returned as a plain Python
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may introduce tiny negative values – clip.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Formula: ∂/∂a_j Σ_k w_k·(a*a)_k = 2· Σ_i a_i·w_{i+j}.
    Implemented by a single cross‑correlation via FFT.
    """
    n = a.size
    Lw = w.size                         # = 2*n-1
    wr = w[::-1]                        # reverse for correlation

    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)

    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _exact_conv_update(conv: np.ndarray, a: np.ndarray,
                       i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a *new* convolution after moving a mass `delta` from index `i`
    to index `j`.  The update follows the exact algebra:
       a' = a - δ·e_i + δ·e_j
       conv' = conv + 2·δ·(shift_j(a) - shift_i(a))
               + δ²·(δ_{2j} + δ_{2i} - 2·δ_{i+j})
    The `shift` operation is a copy of `a` placed at positions i..i+n‑1.
    The routine works in O(n) time by slice operations.
    """
    n = a.size
    new_conv = conv.copy()

    # linear term (factor 2δ)
    lo_i = i
    hi_i = i + n
    lo_j = j
    hi_j = j + n

    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    new_conv[lo_j:hi_j] += 2.0 * delta * a

    # quadratic term
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta

    # Numerical safety
    np.maximum(new_conv, 0.0, out=new_conv)
    return new_conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the evaluation score for a given convolution."""
    return 2.0 * n * conv.max()


def _arcsine_initial(n: int) -> np.ndarray:
    """
    Produce an “arcsine‑shaped’’ probability vector:
    a_i ∝ 1 / sqrt((i+½)*(n‑½‑i)) .
    This shape has large mass near the boundaries and a low centre,
    which tends to flatten the self‑convolution.
    """
    eps = 1e-12
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size & initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = 4096
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Blend with arcsine shape – this gives a better starting point.
    a_arc = _arcsine_initial(n)
    a = 0.5 * a + 0.5 * a_arc
    a /= a.sum()

    # FFT lengths (powers of two) for the two convolutions we need
    fft_len_self = _next_pow2(2 * n - 1)       # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)       # for correlation a⋆w

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _self_conv_fft(best, fft_len_self)
    best_score = _score_from_conv(conv_best, n)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # small safety margin

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters for Phase 1 (soft‑max EG)
    # ------------------------------------------------------------------
    eta_start = 0.8
    eta_end   = 0.03
    beta_start = 0.3
    beta_end   = 1800.0
    momentum   = 0.85
    block_perturb_prob = 0.03
    restart_gap = 25.0          # seconds without improvement → restart
    hard_max_phase = 0.80       # after this fraction we switch to true max

    grad_prev = np.zeros_like(a)
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Phase 1 – exponentiated gradient optimisation
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.25 * (deadline - start_time)  # ~25 % time
    while time.monotonic() < phase1_deadline:
        # current convolution
        conv = _self_conv_fft(a, fft_len_self)
        cur_max = conv.max()
        cur_score = _score_from_conv(conv, n)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Phase 1 new best = {best_score:.6f}")

        # schedule
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # weight vector (smooth max)
        if frac < hard_max_phase:
            w = np.exp(beta * (conv - cur_max))   # max exponent = 0
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # gradient of weighted max
        grad = _gradient_from_weights(a, w)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update (mirror descent on the simplex)
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional random block jitter
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # restart if stuck
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.15, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Phase 1 restart – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase 2 – precise greedy hill‑climbing with exact conv updates
    # ------------------------------------------------------------------
    # Initialise from the best vector found in Phase 1
    a = best.copy()
    conv = _self_conv_fft(a, fft_len_self)
    cur_max = conv.max()
    cur_score = _score_from_conv(conv, n)

    # parameters for hill‑climbing
    move_prob = 0.9               # probability we try a pairwise move each loop
    temp_start = cur_score * 0.005    # initial “temperature’’ for SA
    temp_end   = cur_score * 1e-6     # final temperature
    last_improve = time.monotonic()
    accept_counter = 0
    total_moves = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # simulated‑annealing temperature (linear decay)
        temp = temp_start * (1.0 - frac) + temp_end * frac

        # decide whether to perform a move
        if rng.random() < move_prob:
            # pick donor proportional to its mass (more chance to move big chunks)
            donor = rng.choice(n, p=a)
            if a[donor] < 1e-12:
                continue
            # maximum transferable mass – keep steps modest
            max_delta = min(a[donor] * 0.4, 0.2)   # at most 40 % of donor, globally ≤0.2
            delta = rng.random() * max_delta

            # pick a distinct receiver uniformly
            receiver = rng.integers(0, n)
            if receiver == donor:
                continue

            # exact update of the convolution
            conv_new = _exact_conv_update(conv, a, int(donor), int(receiver), delta)

            new_max = conv_new.max()
            # Metropolis acceptance
            if new_max < cur_max or rng.random() < math.exp(-(new_max - cur_max) / max(temp, 1e-18)):
                # accept the move
                a[donor] -= delta
                a[receiver] += delta
                # renormalise only for numerical drift
                a = np.clip(a, 0.0, None)
                a_sum = a.sum()
                if a_sum != 1.0:
                    a /= a_sum

                conv = conv_new
                cur_max = new_max
                cur_score = _score_from_conv(conv, n)
                total_moves += 1
                accept_counter += 1

                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = now
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] Phase 2 new best = {best_score:.6f}")

        # occasional global re‑normalisation via FFT (helps drift)
        if total_moves % 5000 == 0 and total_moves > 0:
            conv = _self_conv_fft(a, fft_len_self)
            cur_max = conv.max()
            cur_score = _score_from_conv(conv, n)

        # periodic heavy perturbation if we have been idle too long
        if now - last_improve > restart_gap:
            # shake the whole vector a bit and recompute conv from scratch
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _self_conv_fft(a, fft_len_self)
            cur_max = conv.max()
            cur_score = _score_from_conv(conv, n)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Phase 2 restart – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to bounds of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
