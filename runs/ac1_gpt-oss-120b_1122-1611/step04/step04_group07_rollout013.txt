"""
Propose a step‑function that minimises

        score = 2·n·max(conv(a,a)) / (sum(a)**2)

The implementation combines three ideas:

* **Mirror‑descent on the simplex** (exponentiated‑gradient) – a fast
  smooth‑approximation of the max‑norm that quickly yields a decent
  candidate.
* **Active‑set linear programming** – given the current vector `a`,
  we solve a tiny LP that maximises the L1‑norm of a new vector `b`
  under the constraint  conv(a,b) ≤ max(conv(a,a)).  The LP uses only
  those convolution shifts whose value is close to the current maximum,
  so the problem stays tiny (a few dozen constraints) even for large
  `n`.
* **Scheduled mixing & restarts** – we replace `a` by a convex
  combination `(1‑t)·a + t·b` with a stepsize `t` that decays over
  time, apply occasional random perturbations and restart from the
  best‑seen vector if the optimisation stalls.

The whole loop runs until the time budget (≈ 1000 s) expires.  The
best sequence found is returned as a plain list of Python floats and is
guaranteed to satisfy the limits imposed by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length able to hold the full linear convolution
    of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the probability simplex (Dirichlet(1))."""
    v = rng.gamma(1.0, 1.0, size=n)
    v_sum = v.sum()
    return v / v_sum if v_sum > 0 else np.full(n, 1.0 / n)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _solve_lp_max_sum(a: np.ndarray, max_conv: float, tol: float = 0.05):
    """
    Solve the small LP

        maximize   sum(b_i)
        subject to conv(a,b)[k] ≤ max_conv    for all active k
                    b_i ≥ 0

    Only the constraints whose convolution value lies within ``tol``·max_conv
    of the current maximum are kept (active set).  Returns the optimal ``b``
    as a dense 1‑D array or ``None`` if the LP fails.
    """
    n = a.size
    # full self‑convolution (FFT is cheap for the active‑set computation)
    fft_len = _fft_len_for_n(n)
    conv = _conv_self(a, fft_len)
    # active shifts – those that are “almost” tight
    active = np.where(conv >= max_conv - tol * max_conv)[0]

    # build the (usually tiny) constraint matrix
    m = active.size
    if m == 0:
        return None
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        A[row, lo : hi + 1] = a[lo : hi + 1]
    b = np.full(m, max_conv, dtype=np.float64)

    # linear program – maximise sum(b)  == minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # ``highs`` is fast and robust; we turn off its output.
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "disp": False},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.

    The routine starts from the optional global ``height_sequence_1``,
    otherwise from a random Dirichlet point, runs a few hundred
    exponentiated‑gradient iterations (soft‑max objective) and then
    repeatedly solves a tiny linear programme to improve the maximum
    self‑convolution.  All vectors are kept normalised to unit sum, so
    the objective is scale‑invariant.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the vector a
    # ------------------------------------------------------------------
    default_n = 2048                    # a convenient size for the LP step
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
                n = a.size
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = _random_dirichlet(n, rng)
    else:
        n = default_n
        a = _random_dirichlet(n, rng)

    # make sure we are on the probability simplex
    a = np.clip(a, 0.0, None)
    a = a / a.sum()

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping and initial evaluation
    # ------------------------------------------------------------------
    # cheap estimate (sum = 1 → score = 2·n·max(conv))
    conv = _conv_self(a, fft_len)
    cur_max = float(conv.max())
    cur_score_est = 2.0 * n * cur_max
    best_score = float(evaluate_sequence(a.tolist()))   # exact
    best = a.copy()
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # mirror‑descent (soft‑max) parameters – a short warm‑up phase
    eta_start = 0.6
    eta_end = 0.02
    beta_start = 0.8
    beta_end = 400.0

    # LP‑mixing schedule
    t_start = 0.5      # initial mixing weight
    t_end = 0.08       # final mixing weight
    lp_mix_iters = 200  # number of LP iterations we plan to do

    # restarts / perturbations
    stagnation_time = 45.0   # seconds without progress → restart
    block_perturb_prob = 0.04
    print_interval = 30.0   # seconds between status prints
    next_print = start_time + print_interval

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < start_time + budget_s - 5.0:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- soft‑max gradient descent (warm‑up) -----------------------
        # The warm‑up runs for the first 30 s (or until we have done a few
        # hundred iterations).  After that we switch to the LP phase.
        if elapsed < 30.0:
            # linear schedules for η and β
            eta = eta_start * (1.0 - elapsed / 30.0) + eta_end * (elapsed / 30.0)
            beta = beta_start + (beta_end - beta_start) * (elapsed / 30.0)

            conv = _conv_self(a, fft_len)
            max_conv = float(conv.max())
            # soft‑max weighting of *all* convolution entries
            shift = max_conv
            w_raw = np.exp(beta * (conv - shift))
            w = w_raw / w_raw.sum()

            # gradient of the smooth max (2·corr(w,a))
            grad = 2.0 * np.correlate(w, a, mode="valid")
            # exponentiated‑gradient (mirror‑descent on the simplex)
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        else:
            # ----- LP‑based improvement ---------------------------------
            conv = _conv_self(a, fft_len)
            max_conv = float(conv.max())
            # active‑set linear program
            b0 = _solve_lp_max_sum(a, max_conv, tol=0.05)
            if b0 is not None:
                # mixing weight that linearly decays with the number of LP steps
                mix_t = t_start * (1 - it / lp_mix_iters) + t_end * (it / lp_mix_iters)
                mix_t = max(t_end, min(t_start, mix_t))
                a = (1.0 - mix_t) * a + mix_t * b0
                a = np.clip(a, 0.0, None)
                a /= a.sum()
            else:
                # if the LP failed we fall back to a cheap random perturbation
                if rng.random() < block_perturb_prob:
                    a = _random_block_perturb(a, rng)

            it += 1

        # ----- keep the best exact score ---------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score_est = 2.0 * n * cur_max
        if cur_score_est < best_score:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best (exact) score = {best_score:.6f}")

        # ----- occasional status print -----------------------------------
        if now >= next_print:
            print(
                f"[{elapsed:6.2f}s] est. score = {cur_score_est:.6f}  best = {best_score:.6f}"
            )
            next_print = now + print_interval

        # ----- stagnation handling ---------------------------------------
        if now - last_improve > stagnation_time:
            # jitter the current best and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
