def propose_candidate(seed=42, budget_s=1000.0):
    rng = np.random.default_rng(seed)

    # Determine n: if height_sequence_1 exists, use its length else default 4096 (power of two)
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        n = a0.size
        if n == 0:
            n = 4096
            a0 = rng.random(n)
    else:
        n = 4096
        a0 = rng.random(n)

    # Normalize to sum 1
    a0 /= np.maximum(a0.sum(), 1e-12)

    fft_len = _fft_len_for_n(n)
    # compute initial conv and score
    conv = _conv_self(a0, fft_len)
    cur_score = 2.0 * n * conv.max()
    cur_a = a0.copy()

    best_a = cur_a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)
    total = deadline - start_time

    # Simulated annealing parameters
    T0 = best_score * 0.2  # initial temperature
    Tmin = 1e-6
    move_prob = [0.5, 0.2, 0.2, 0.1]  # transfer, swap, block, global
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        ratio = elapsed / total
        # temperature linear cooling
        T = T0 * (1 - ratio) + Tmin

        # propose new candidate
        move_type = rng.choice(4, p=move_prob)
        a_new = cur_a.copy()

        if move_type == 0:
            # transfer mass
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 0:
                max_delta = min(a_new[i], 0.5 * a_new[i])
                delta = rng.random() * max_delta
                a_new[i] -= delta
                a_new[j] += delta
        elif move_type == 1:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            a_new[i], a_new[j] = a_new[j], a_new[i]
        elif move_type == 2:
            # block perturbation (multiplicative)
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            a_new[start:start+width] *= factor
            a_new = np.clip(a_new, 0.0, None)
        else:
            # global noise
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a_new *= noise
            a_new = np.clip(a_new, 0.0, None)

        # renormalize (sum unchanged for transfer, swap, but some moves change sum)
        s = a_new.sum()
        if s <= 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s

        # evaluate candidate
        conv_new = _conv_self(a_new, fft_len)
        new_score = 2.0 * n * conv_new.max()

        # acceptance
        dE = new_score - cur_score
        if dE <= 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            cur_a = a_new
            cur_score = new_score
            # update best
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best {best_score:.6f}")

        # optional restart if no improvement for some time
        if now - last_improve > 30.0:
            # restart from best with jitter
            cur_a = best_a.copy()
            jitter = np.exp(rng.normal(0.0, 0.1, size=n))
            cur_a *= jitter
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a /= cur_a.sum()
            cur_score = best_score
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation")
    
    # Return best after clipping
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]

"""
Propose a candidate step‑function by *flattening* its self‑convolution.

The evaluation score equals  2·n·max(conv)  (the sum of the coefficients is kept
to 1, therefore the denominator of ``evaluate_sequence`` is constant).
To minimise the maximum convolution value we try to make the whole
convolution as close as possible to a constant vector.
The constant that respects the normalisation is 1/(2·n‑1) because
∑_k conv_k = (∑ a_i)² = 1.

Algorithm
---------

* Initialise a probability vector a (sum = 1); use the previously found
  sequence if it exists, otherwise a random vector.
* Repeatedly
    1. Compute the linear self‑convolution `conv = a * a` by FFT.
    2. Form the residual `r = conv – target`, where `target` is the flat
       vector `1/(2·n‑1)`.
    3. The gradient of the squared‑error loss `‖conv‑target‖²` w.r.t. a is
       `g = 2·corr(r, a)`.  The helper ``_gradient_correlation`` returns exactly
       this quantity (the factor 2 is already included).
    4. Mirror‑descent (exponentiated‑gradient) update:
           a ← a · exp(‑η·g)   (element‑wise)
       and re‑normalise to keep a on the probability simplex.
    5. Occasionally apply cheap stochastic perturbations
       (block scaling, global log‑normal jitter) and restart from the best
       solution if no improvement is seen for a while.
* Keep the best vector seen (according to the true score
  `2·n·max(conv)`).  After the time budget is exhausted, clip the values to
  the limits enforced by ``evaluate_sequence`` and return the list.

The method directly attacks the quantity we want to minimise
(max conv) instead of a soft‑max surrogate, and because the target
convolution is flat the gradient points precisely away from the
currently largest entries.  With a simple learning‑rate schedule and
periodic restarts the search explores many basins and typically beats
the previous mirror‑descent baseline within the 1000 s budget.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·corr(w, a) where ``w`` has length 2·n‑1 and ``a`` has length n.

    The result is the gradient of the squared‑error loss
    L(a)=‖(a∗a)‑w_target‖² w.r.t. a.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = conv_full[start:start + n]      # this already is 2·corr(w,a)
    return grad
