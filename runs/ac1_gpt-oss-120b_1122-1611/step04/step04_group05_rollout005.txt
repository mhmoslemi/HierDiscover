def _descent(a, start_iter, max_iters, start_time, deadline, ff_len, rng,
            eta_start, eta_end, beta_start, beta_end,
            momentum, block_prob, pair_prob):
    grad_prev = np.zeros_like(a)
    best_local = a.copy()
    best_local_score = 2.0 * n * _conv_self(a, ff_len).max()
    iter = 0
    while iter < max_iters and time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        # schedule
        eta = eta_start * (1 - elapsed/total) + eta_end * (elapsed/total)
        beta = beta_start + (beta_end - beta_start) * (elapsed/total)
        # compute conv
        conv = _conv_self(a, ff_len)
        max_conv = conv.max()
        # compute w (softmax)
        # To avoid overflow, clip exponent
        exp_arg = beta * (conv - max_conv)
        exp_arg = np.clip(exp_arg, -700, 0)  # avoid underflow/overflow
        w = np.exp(exp_arg)
        # compute gradient via FFT
        a_rev = a[::-1]
        L = (2*n-1) + n - 1
        fft_len = _fft_len_for_len(L)
        grad_full = np.fft.irfft(np.fft.rfft(w, n=fft_len) *
                                np.fft.rfft(a_rev, n=fft_len), n=fft_len)
        grad = 2.0 * grad_full[n-1:n-1+n] - a * w[2*np.arange(n)]
        # momentum smoothing
        grad = momentum * grad_prev + (1-momentum) * grad
        grad_prev = grad
        # EG update
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
        # random perturbations
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])

"""
Hybrid parametric initialization + exponentiated‑gradient descent.

The solver builds a collection of simple analytic “U‑shaped’’ candidates
(arcsine, Beta‑distribution, power‑law) and a few random Dirichlet starts.
Each candidate is refined by a mirror‑descent (exponentiated‑gradient) loop
that minimises  

    score = 2·n·max(conv(a,a)) / (sum(a)**2)   (with sum(a)=1)

using a soft‑max approximation of the max‑norm.  The schedule drives the
temperature β quickly to a very large value (≈1500) so that the gradient
focuses essentially on the maximal convolution entries.  A momentum term,
occasional block‑perturbations, pair‑wise mass transfers and timed restarts
help escape plateaus.  The best vector found during the whole budget is
returned (clipped to the limits enforced by ``evaluate_sequence``).

The implementation is pure NumPy, uses only FFT‑based convolutions and
runs comfortably inside the 1000 s wall‑time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _make_arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_power_shape(n: int, p: float, eps: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + eps) * (n - i - 1 + eps)
    a = 1.0 / (prod ** p + 1e-12)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    # symmetric Beta(α,α) on the discrete grid
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _gradient_from_conv(a: np.ndarray,
                        conv: np.ndarray,
                        beta: float,
                        fft_len_grad: int) -> np.ndarray:
    """Soft‑max weighted sub‑gradient of max(conv) w.r.t. ``a``."""
    n = a.size
    max_conv = conv.max()
    # soft‑max weights (avoid overflow)
    exp_arg = beta * (conv - max_conv)
    exp_arg = np.clip(exp_arg, -800.0, 0.0)
    w = np.exp(exp_arg)

    # convolution w * a_rev (full length)
    a_rev = a[::-1]
    grad_full = np.fft.irfft(np.fft.rfft(w, n=fft_len_grad) *
                            np.fft.rfft(a_rev, n=fft_len_grad), n=fft_len_grad)

    # extract the n components that correspond to ∂/∂a_j
    grad = 2.0 * grad_full[n - 1 : n - 1 + n] - a * w[2 * np.arange(n)]
    return grad


def _optimize(a0: np.ndarray,
              start_time: float,
              deadline: float,
              ff_len: int,
              rng: np.random.Generator,
              eta_start: float,
              eta_end: float,
              beta_start: float,
              beta_end: float,
              momentum: float = 0.9,
              block_prob: float = 0.04,
              pair_prob: float = 0.08,
              restart_gap: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation loop.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.size
    grad_prev = np.zeros_like(a)
    best = a.copy()
    # initial score
    conv = _conv_self(a, ff_len)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = start_time

    # length needed for the gradient convolution (w_len + a_rev_len - 1)
    L = (2 * n - 1) + n - 1
    fft_len_grad = _fft_len_for_len(L)

    while time.monotonic() < deadline:
        now = time.monotonic()
        # schedule parameters (linear annealing)
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # convolution and current score
        conv = _conv_self(a, ff_len)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (np.sum(a) ** 2)

        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] new best = {best_score:.6f}")

        # gradient (soft‑max approximation of the max‑norm)
        grad = _gradient_from_conv(a, conv, beta, fft_len_grad)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient (mirror descent) update
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional random block perturbation
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        # occasional pair‑wise mass transfer
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a[i] -= delta
                    a[j] += delta
                    a = np.clip(a, 0.0, None)
                    a /= a.sum()

        # soft restart if we have stalled for a while
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget expires.

    The routine builds several analytic starting points (arcsine, Beta‑family,
    power‑law, random Dirichlet), runs a short mirror‑descent from each of them,
    keeps the best, and then performs a long refined optimisation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) problem size
    # ------------------------------------------------------------------
    default_n = 4000                     # safe compromise between quality and speed
    n = default_n

    # ------------------------------------------------------------------
    # 2) build a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # arcsine shape (the classic “U‑shape’’)
    candidates.append(_make_arcsine_shape(n))

    # uniform shape (useful as a sanity check)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # if a previous construction exists, reuse it (normalised)
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                arr = np.clip(arr, 0.0, None)
                s = arr.sum()
                if s > 0.0:
                    arr = arr / s
                else:
                    arr = np.full(arr.shape[0], 1.0 / arr.shape[0])
                # if the size differs from the default, just keep it (the
                # optimiser works for any length)
                candidates.append(arr)
                n = arr.size        # adopt the external length if given
        except Exception:
            pass

    # Beta‑family (symmetric Beta(α,α) with α∈[0.25,0.9])
    for alpha in np.linspace(0.25, 0.9, 13):
        candidates.append(_make_beta_shape(n, alpha))

    # Power‑law family  a_i ∝ 1/[(i+½)(n‑i‑½)]^p   with p∈[0.4,1.2]
    for p in np.linspace(0.4, 1.2, 9):
        candidates.append(_make_power_shape(n, p))

    # a few random Dirichlet seeds (concentration 0.5 gives U‑shapes)
    for _ in range(5):
        candidates.append(rng.dirichlet(np.full(n, 0.5)))

    # ------------------------------------------------------------------
    # 3) bookkeeping and time allocation
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    total_deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    scan_deadline = start_time + 0.20 * (total_deadline - start_time)  # 20 % for scans

    fft_len = _fft_len_for_n(n)

    best = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 4) quick scan of the candidate pool (short runs)
    # ------------------------------------------------------------------
    for idx, cand in enumerate(candidates):
        if time.monotonic() > scan_deadline:
            break

        # normalise – safety for any external length changes
        a0 = cand.astype(np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s

        # evaluate the raw candidate
        conv0 = _conv_self(a0, fft_len)
        score0 = 2.0 * a0.size * conv0.max() / (a0.sum() ** 2)
        if score0 < best_score:
            best_score = score0
            best = a0.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] init {idx} score {score0:.6f}")

        # short mirror‑descent run (few hundred iterations)
        short_deadline = min(scan_deadline,
                             time.monotonic() + 0.7)   # about half a second per candidate
        a_opt, sc_opt = _optimize(
            a0,
            start_time=time.monotonic(),
            deadline=short_deadline,
            ff_len=fft_len,
            rng=rng,
            eta_start=0.8,
            eta_end=0.05,
            beta_start=0.5,
            beta_end=400.0,
            momentum=0.9,
            block_prob=0.03,
            pair_prob=0.03,
            restart_gap=25.0,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best = a_opt.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] improved after short run {idx} -> {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 5) long fine‑tuning from the best candidate found so far
    # ------------------------------------------------------------------
    if best is None:
        best = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"[{time.monotonic() - start_time:6.2f}s] start final optimisation (score {best_score:.6f})")

    best, best_score = _optimize(
        best,
        start_time=start_time,
        deadline=total_deadline,
        ff_len=fft_len,
        rng=rng,
        eta_start=0.9,
        eta_end=0.001,
        beta_start=0.2,
        beta_end=1500.0,
        momentum=0.95,
        block_prob=0.04,
        pair_prob=0.08,
        restart_gap=30.0,
    )

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] final best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) final safety clipping and return
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
