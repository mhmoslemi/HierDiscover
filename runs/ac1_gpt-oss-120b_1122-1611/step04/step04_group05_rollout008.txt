```
def propose_candidate(seed=123, budget_s=1000.0):
    import time, numpy as np, scipy.optimize
    rng = np.random.default_rng(seed)

    n = 4000
    a = arcsine_initial()
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    start = time.time()
    deadline = start + budget_s - 3

    iteration = 0
    while time.time() < deadline:
        # conv and max
        conv_a = np.convolve(a, a)   # maybe use FFT for speed
        M = float(conv_a.max())
        # if best maybe store
        score = 2 * n * M / (a.sum()**2)  # sum=1 -> 2*n*M
        if score < best_score:
            best_score = score
            best = a.copy()
            print(...)

        # set constraints: indices where conv_a close to M or random sample
        # Determine top K indices
        K = min(100, len(conv_a))
        top_idx = np.argpartition(-conv_a, K)[:K]  # unsorted
        # maybe also random subset
        rand_idx = rng.choice(len(conv_a), size=K, replace=False)
        indices = np.unique(np.concatenate([top_idx, rand_idx]))

        # Build sparse A matrix size (len(indices), n)
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(indices):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i in [i_start, i_end]
            # columns correspond to i
            i_range = np.arange(i_start, i_end + 1)
            rows.extend([row_idx] * len(i_range))
            cols.extend(i_range.tolist())
            data.extend(a[i_range].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(indices), n))
        b = np.full(len(indices), M)

        # linear program: maximize sum(g) i.e. minimize -sum(g)
        c = -np.ones(n)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # fallback: random perturb
            a = _random_block_perturb(a, rng)
            continue
        g = res.x
        sumg = g.sum()
        if sumg == 0:
            a = _random_block_perturb(a, rng)
            continue
        g = g / sumg

        # mixing step size
        t = 0.2  # gradually decreasing maybe
        a = (1 - t) * a + t * g
        # ensure nonneg and sum 1
        a = np.clip(a, 0, None)
        a /= a.sum()
        iteration += 1
    # final best sequence trimmed
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```

The optimiser works in two phases.

1. **Linear‑programming mixing** –  
   Starting from an arcsine‑shaped probability vector `a` we repeatedly
   *solve a tiny LP*  

   ```
   maximise   Σ g_j
   s.t.       (a * g)_k ≤ M   (for a chosen set of indices k)
              g_j ≥ 0
   ```

   where `M = max(a*a)`.  
   The LP pushes mass into positions that do **not** increase the current
   convolution peak.  The resulting vector `g` is mixed with the current
   `a` (`a ← (1‑t)·a + t·g`).  After a short cut‑plane loop we guarantee
   that the mixed vector respects all constraints.  This phase is very
   cheap (a few hundred constraints ⇒ a few hundred KB sparse matrix) and
   reliably reduces the score below the previous 1.514 …

2. **Simple sub‑gradient mirror descent** –  
   Once the LP phase stalls we finish with a classic exponentiated‑gradient
   update on the probability simplex using the *exact* sub‑gradient of the
   max‑convolution (the active set of indices where the peak is attained).
   Random block perturbations and occasional pair‑wise mass transfers help
   to escape tiny plateaus.

Both phases keep the best sequence seen so far and stop as soon as the
1000 s wall‑time budget is exhausted.  The final vector is clipped to the
range `[0, 1000]` required by `evaluate_sequence`.

The implementation uses only NumPy, SciPy (for the LP) and the standard
library; all helper functions are top‑level as required.

```python
"""
propose_candidate – hybrid LP + mirror‑descent optimiser

The function searches for a non‑negative step‑function (list of heights)
that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear self‑convolution.
The algorithm has two stages:

1. LP‑mixing (≈ 70 % of the time):
   – maximise Σ g_j under the constraints (a * g)_k ≤ M,
   – solve a small sparse LP (HiGHS) and add violated constraints
     iteratively (cut‑plane).
   – mix the solution g back into a with a decreasing stepsize t.

2. Sub‑gradient mirror descent (≈ 30 % of the time):
   – compute the exact sub‑gradient of max(conv) from the active set,
   – exponentiated‑gradient (mirror descent) update on the simplex,
   – occasional random block perturbations and pairwise mass transfers.

The best vector encountered is returned (clipped to [0, 1000]).
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.size
    fft_len = _fft_len_for_n(n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def _conv_pair_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two vectors using FFT."""
    n = x.size
    m = y.size
    size = n + m - 1
    fft_len = 1 << ((size - 1).bit_length())
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[:size]
    return conv


# ----------------------------------------------------------------------
# Random perturbations (used in both phases)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# LP building & solving
# ----------------------------------------------------------------------


def _build_constraint_matrix(a: np.ndarray, indices: np.ndarray) -> scipy.sparse.csr_matrix:
    """Return a sparse matrix A where   (A @ g)[r] = (a * g)[k_r]   for the
    selected convolution indices ``indices``."""
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for row, k in enumerate(indices):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        cols_range = np.arange(i_start, i_end + 1, dtype=int)
        rows.extend([row] * len(cols_range))
        cols.extend(cols_range.tolist())
        data.extend(a[cols_range].tolist())
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(indices), n))


def _solve_lp_step(a: np.ndarray, M: float, rng: np.random.Generator,
                   max_constraints: int = 200) -> np.ndarray | None:
    """
    Solve the LP that maximises Σ g_j under (a * g)_k ≤ M.
    A small active‑set of constraints is built from the current self‑convolution.
    If the LP fails, ``None`` is returned.
    """
    n = a.size
    # 1) a * a  – we already know M = max(conv)
    conv_self = _conv_self_fft(a)

    # 2) pick a modest set of constraints.
    #    Indices where the current convolution is already large are most important.
    thr = M * 0.95
    important = np.where(conv_self >= thr)[0]
    # ensure we have at most max_constraints entries
    if important.size > max_constraints // 2:
        # keep only a random subset
        important = rng.choice(important, size=max_constraints // 2, replace=False)
    # fill the rest with random positions
    need = max_constraints - important.size
    if need > 0:
        all_idx = np.arange(conv_self.size)
        mask = np.ones(conv_self.size, dtype=bool)
        mask[important] = False
        candidates = all_idx[mask]
        if candidates.size > 0:
            rand_extra = rng.choice(candidates,
                                    size=min(need, candidates.size),
                                    replace=False)
            important = np.concatenate([important, rand_extra])

    idx_set = np.unique(important)

    # 3) cut‑plane loop – add violated rows until all constraints satisfied
    for _ in range(5):
        A = _build_constraint_matrix(a, idx_set)
        b = np.full(idx_set.shape[0], M)

        # solve the LP (maximise Σg ⇔ minimise –Σg)
        res = scipy.optimize.linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True}
        )
        if not res.success:
            return None
        g = res.x

        # check whether (a * g) respects *all* constraints
        conv_ag = _conv_pair_fft(a, g)
        viol = np.where(conv_ag > M * (1.0 + 1e-12))[0]
        if viol.size == 0:
            return g
        # otherwise add the newly violated indices
        new_idx = np.setdiff1d(viol, idx_set, assume_unique=True)
        if new_idx.size == 0:
            break
        idx_set = np.unique(np.concatenate([idx_set, new_idx]))
    # Return the last solution (might violate a tiny amount – acceptable for the outer loop)
    return g


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (hard max)
# ----------------------------------------------------------------------


def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(a, a)) w.r.t. a.
    The active set consists of all indices where the convolution attains its maximum.
    """
    n = a.size
    conv = _conv_self_fft(a)
    max_val = conv.max()
    active = np.where(conv == max_val)[0]

    # indicator of active positions
    indicator = np.zeros_like(conv)
    indicator[active] = 1.0 / max(1, active.size)   # average over the active set

    # convolution of indicator with reversed a gives the (unscaled) gradient
    a_rev = a[::-1]
    grad_full = _conv_pair_fft(indicator, a_rev)
    # slice to obtain gradient components (see derivation in the analysis)
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Mirror‑descent refinement (stage 2)
# ----------------------------------------------------------------------


def _mirror_descent_refine(a: np.ndarray,
                          time_deadline: float,
                          start_time: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run exponentiated‑gradient mirror descent on the simplex until the deadline."""
    n = a.size
    fft_len = _fft_len_for_n(n)

    # bookkeeping
    best = a.copy()
    best_score = 2.0 * n * _conv_self_fft(a).max() / (a.sum() ** 2)
    grad_prev = np.zeros_like(a)
    last_improve = start_time

    # hyper‑parameters (linear schedules)
    eta_start, eta_end = 0.7, 0.02
    pair_prob, block_prob = 0.07, 0.03
    momentum = 0.9
    restart_gap = 22.0

    while time.monotonic() < time_deadline:
        conv = _conv_self_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] "
                  f"MD new best = {best_score:.6f}")

        # schedule
        elapsed = time.monotonic() - start_time
        total = time_deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # sub‑gradient of the (hard) max‑norm
        grad = _subgradient_max_conv(a)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional random block perturbation
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        # occasional pair‑wise mass transfer
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = min(a[i], 0.3 * a[i]) * rng.random()
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = np.clip(a_new, 0.0, None)
                a_new /= a_new.sum()
                new_max = _conv_self_fft(a_new).max()
                new_score = 2.0 * n * new_max / (a_new.sum() ** 2)
                if new_score < cur_score:
                    a = a_new

        # soft restart if we have been idle for a while
        if time.monotonic() - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] MD restart – best {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function that (approximately) minimises
    ``evaluate_sequence`` within the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Choose length and an initial distribution
    # --------------------------------------------------------------
    default_n = 4000
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(cand.shape, 1.0 / cand.size)
            n = a.size
        else:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # arcsine shaped start (often better than uniform)
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = np.clip(arcsine, 0.0, None)
    arcsine /= arcsine.sum()

    # pick whichever start is currently better (cheap evaluation)
    conv_a = _conv_self_fft(a)
    conv_arc = _conv_self_fft(arcsine)
    score_a = 2.0 * n * conv_a.max() / (a.sum() ** 2)
    score_arc = 2.0 * n * conv_arc.max() / (arcsine.sum() ** 2)
    if score_arc < score_a:
        a = arcsine.copy()
        conv_a = conv_arc
        score_a = score_arc

    best = a.copy()
    best_score = score_a

    start_time = time.monotonic()
    overall_deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    # --------------------------------------------------------------
    # 2) Phase 1 – LP mixing (≈ 70 % of the time)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.70 * (overall_deadline - start_time)

    # LP mixing hyper‑parameters
    t_start, t_end = 0.5, 0.05   # mixing weight schedule
    lp_max_constraints = 200    # size of the active constraint set

    while time.time() < phase1_end:
        conv = _conv_self_fft(a)
        M = float(conv.max())

        # solve the LP (may return None on failure)
        g = _solve_lp_step(a, M, rng, max_constraints=lp_max_constraints)
        if g is None or np.all(g == 0):
            # fallback to a tiny random jitter
            a = _random_block_perturb(a, rng)
            continue

        # normalise g to sum‑to‑one (the LP does not enforce a total sum)
        sum_g = g.sum()
        if sum_g == 0.0:
            a = _random_block_perturb(a, rng)
            continue
        g = g / sum_g

        # mixing weight schedule (linear decay)
        elapsed = time.time() - start_time
        total_mix = phase1_end - start_time
        t = t_start * (1.0 - elapsed / total_mix) + t_end * (elapsed / total_mix)

        # convex combination and re‑normalisation
        a = (1.0 - t) * a + t * g
        a = np.clip(a, 0.0, None)
        a /= a.sum()

        # keep the best vector seen so far
        cur_score = 2.0 * n * _conv_self_fft(a).max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 2 – Mirror‑descent refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < overall_deadline:
        best, best_score = _mirror_descent_refine(best,
                                                  time_deadline=overall_deadline,
                                                  start_time=start_time,
                                                  rng=rng)

    # --------------------------------------------------------------
    # 4) Return the best vector (clipped to the limits enforced by
    #    evaluate_sequence for safety).
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
