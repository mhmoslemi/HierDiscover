    # evaluate initial conv and score
    conv = conv_self_fft(a)
    score = 2*n*conv.max()
    # precompute n? Should be passed
    for it in range(n_iter):
        i = rng.integers(0, a.size)

"""
Propose a step‑function sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a two‑stage optimisation:

* Stage 1 – Adam optimiser on a smooth (log‑sum‑exp) approximation of the
  infinity‑norm of the self‑convolution.  The temperature β is
  linearly increased from a small start (≈0.5) to a large final value
  (≈2000) so the objective gradually focuses on the true peak.  The
  gradient of the smoothed max is computed exactly via FFT:  
  ∇L = 2·corr(w, a) where w are the soft‑max weights of the
  convolution entries.

* Stage 2 – a short stochastic‑annealing local search (pairwise mass
  transfers) that polishes the solution found by Adam.

Multiple independent runs with different lengths (from a few thousand up
to ~8000) are performed within the 1000 s budget, each run starting from
either a random Dirichlet draw or a resampled version of the best
sequence discovered so far.  The best vector seen overall is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    size = x.size + y.size - 1
    fft_len = _next_pow2(size)
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)[:size]
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full linear) via FFT."""
    n = a.size
    size = 2 * n - 1
    fft_len = _next_pow2(size)
    F = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(F * F, n=fft_len)[:size]
    return conv


# ----------------------------------------------------------------------
# Resampling helper – interpolate an existing distribution to a new length
# ----------------------------------------------------------------------


def _resample_distribution(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative distribution to a new length."""
    old_n = seq.size
    if old_n == new_n:
        out = seq.copy()
    else:
        old_x = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
        new_x = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
        out = np.interp(new_x, old_x, seq)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / new_n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑norm of the self‑convolution
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    beta_start: float,
    beta_end: float,
    max_iters: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of `max(conv(a,a))` using Adam.
    Returns the best vector found and its (exact) score `2*n*max(conv)`.
    """
    n = a0.size
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_start, lr_end = 0.20, 0.005

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_score = 2.0 * n * conv.max()

    for it in range(1, max_iters + 1):
        if time.monotonic() > deadline:
            break

        # linear schedules
        frac = it / max_iters
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # self‑convolution and smooth‑max weights
        conv = _conv_self_fft(a)
        maxc = conv.max()
        shift = conv - maxc                     # non‑positive
        exp_shift = np.exp(beta * shift)        # stable because maxc subtracted
        w = exp_shift / exp_shift.sum()         # soft‑max weights, sum = 1

        # exact gradient of the smooth max: 2 * corr(w, a)
        grad_full = _conv_fft(w, a[::-1])       # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : 2 * n - 1]   # valid part, size n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # keep on the probability simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s

        # occasional multiplicative jitter to avoid stagnation
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            a_new *= jitter
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()

        a = a_new
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.monotonic():6.2f}s] Adam iter {it:5d}  new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Local stochastic refinement (pairwise mass transfers)
# ----------------------------------------------------------------------


def _local_refine(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_moves: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing‑style hill climbing.  Proposes
    pairwise mass transfers; accepts only if they improve the exact
    `2*n*max(conv)` score.
    """
    n = a.size
    cur_a = a.copy()
    conv = _conv_self_fft(cur_a)
    cur_score = 2.0 * n * conv.max()

    for move in range(max_moves):
        if time.monotonic() > deadline:
            break
        # pick donor with non‑trivial mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue
        # pick distinct recipient
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # transfer up to 30 % of donor mass
        delta = rng.random() * 0.30 * cur_a[i]
        if delta <= 0.0:
            continue
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # recompute full convolution (FFT, cheap for moderate n)
        conv_new = _conv_self_fft(a_new)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            cur_a = a_new
            cur_score = new_score
            # keep the best solution inside the refinement as‑well
            # (no need for extra bookkeeping – we just keep cur_a)
    return cur_a, cur_score
