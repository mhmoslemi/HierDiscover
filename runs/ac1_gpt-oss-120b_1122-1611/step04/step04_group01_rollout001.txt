"""
Propose a non‑negative height sequence that minimizes
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The optimizer uses a two‑phase strategy:

1. Adam‑style gradient descent on a smooth approximation of the
   max‑norm (soft‑max with a temperature β that grows during the run).
   After each step the vector is projected onto the probability simplex
   (non‑negative, sum = 1) so the denominator stays constant.

2. A cheap local‑search layer that repeatedly tries small “mass‑transfer”
   moves – moving a tiny amount of probability from indices that contribute
   heavily to the current maximal convolution value to indices that
   contribute little.  Only moves that strictly decrease the objective are
   accepted.

Stagnation (no improvement for ``restart_gap`` seconds) triggers a soft
restart: the best vector found so far is jittered with a log‑normal
perturbation and the Adam statistics are cleared.  Random block
perturbations and occasional pair‑wise swaps are also performed to keep
the search diverse.

The routine runs for the allotted time (default 1000 s) and returns the
best sequence encountered, clipped to the limits enforced by
``evaluate_sequence`` (0 ≤ x ≤ 1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors whose total length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using real FFT.
    The result length is ``len(x)+len(y)-1`` (truncated to that)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Returns a new array."""
    # reference: https://doi.org/10.1016/j.csda.2009.06.014
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all zero – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **_):
    """
    Search for a sequence of non‑negative heights that (approximately)
    minimizes ``evaluate_sequence``.  The function respects the 1000 s
    budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the candidate (use the provided sequence if available)
    # ------------------------------------------------------------------
    default_n = 3809                     # length of the known best construction
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim != 1 or cand.size == 0 or not np.isfinite(cand).all():
                raise ValueError
            n = cand.size
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_conv = _fft_len_for_len(2 * n - 1)          # for a*a
    fft_len_grad = _fft_len_for_len(3 * n - 2)          # for w * a_rev

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv = _conv_self_fft(a, fft_len_conv)
    best_score = 2.0 * n * conv.max()                   # sum(a)=1 → denominator = 1
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Adam hyper‑parameters
    # ------------------------------------------------------------------
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # ------------------------------------------------------------------
    # 5) Schedules and probabilities
    # ------------------------------------------------------------------
    eta_start, eta_end = 0.8, 0.003          # learning‑rate schedule
    beta_start, beta_end = 0.5, 2000.0      # soft‑max temperature schedule
    block_perturb_prob = 0.04
    pair_move_prob = 0.12
    restart_gap = 45.0                       # seconds without improvement → restart
    target_phase_frac = 0.70                 # proportion of time spent in gradient phase

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # safety margin

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        phase_grad = (elapsed < target_phase_frac * total)

        # ----- 6.1) Convolution and current score -----------------------
        conv = _conv_self_fft(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- 6.2) Record best -----------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- 6.3) Update schedules ------------------------------------
        # Linear interpolation, clamped to the interval [0, 1]
        progress = min(1.0, elapsed / total)
        eta = eta_start * (1.0 - progress) + eta_end * progress
        beta = beta_start + (beta_end - beta_start) * progress

        # ----------------------------------------------------------------
        # 7) Gradient‑descent phase (dominates early part of the run)
        # ----------------------------------------------------------------
        if phase_grad:
            # ----- 7.1) Soft‑max weighting of the whole convolution -------
            # shift by cur_max for numerical stability
            w = np.exp(beta * (conv - cur_max))
            sum_w = w.sum()

            # ----- 7.2) Gradient of the smooth max (≈ 2 * Σ w_i a_{i‑j}) --
            a_rev = a[::-1]
            conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)    # length = 3n‑2
            grad_part = conv_w_rev[n - 1: n - 1 + n] / sum_w
            raw_grad = 2.0 * grad_part                         # ∂/∂a of smooth max

            # ----- 7.3) Adam update ------------------------------------
            t += 1
            m = b1 * m + (1.0 - b1) * raw_grad
            v = b2 * v + (1.0 - b2) * (raw_grad ** 2)
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)
            step = eta * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_onto_simplex(a)

            # ----- 7.4) Occasional random block perturbation ------------
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # ----- 7.5) Pair‑wise mass transfer (cheap local move) -------
            if rng.random() < pair_move_prob:
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 0.0:
                    max_delta = min(a[i], 0.3 * a[i])
                    if max_delta > 0.0:
                        delta = rng.random() * max_delta
                        cand = a.copy()
                        cand[i] -= delta
                        cand[j] += delta
                        cand = _project_onto_simplex(cand)
                        conv_cand = _conv_self_fft(cand, fft_len_conv)
                        cand_score = 2.0 * n * conv_cand.max()
                        if cand_score < cur_score:
                            a = cand
                            conv = conv_cand
                            cur_max = conv_cand.max()
                            cur_score = cand_score
        else:
            # -------------------------------------------------------------
            # 8) Fine‑grained local search (post‑gradient phase)
            # -------------------------------------------------------------
            # Try a few random mass‑transfer attempts per loop iteration.
            # Each attempt is accepted only if it strictly improves the score.
            for _ in range(3):
                i, j = rng.integers(0, n, size=2)
                if i == j or a[i] <= 0.0:
                    continue
                # move a small fraction of mass from i to j
                delta = min(a[i], 0.02 * a[i]) * rng.random()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # no need to re‑project – sum stays 1 and all entries stay ≥0
                conv_cand = _conv_self_fft(cand, fft_len_conv)
                cand_score = 2.0 * n * conv_cand.max()
                if cand_score < cur_score:
                    a = cand
                    conv = conv_cand
                    cur_max = conv_cand.max()
                    cur_score = cand_score
                    break  # accept first improvement

            # Targeted mass‑transfer based on the current active set.
            # We move probability away from indices that contribute heavily
            # to the maximal convolution value.
            if rng.random() < 0.25:
                # mask of positions that are (almost) maximal
                tol = 1e-12 * cur_max
                mask = (conv >= cur_max - tol).astype(np.float64)
                if mask.sum() > 0:
                    contrib = _conv_fft(mask, a_rev, fft_len_grad)[n - 1: n - 1 + n]
                    # Choose a source index with large contribution and non‑zero mass
                    src_candidates = np.where((contrib > np.median(contrib)) & (a > 1e-12))[0]
                    if src_candidates.size:
                        src = rng.choice(src_candidates)
                        # Choose a destination with small contribution
                        dst_candidates = np.where(contrib < np.median(contrib))[0]
                        dst = rng.choice(dst_candidates) if dst_candidates.size else rng.integers(0, n)
                        if src != dst:
                            delta = min(a[src], 0.02 * a[src]) * rng.random()
                            cand = a.copy()
                            cand[src] -= delta
                            cand[dst] += delta
                            conv_cand = _conv_self_fft(cand, fft_len_conv)
                            cand_score = 2.0 * n * conv_cand.max()
                            if cand_score < cur_score:
                                a = cand
                                conv = conv_cand
                                cur_max = conv_cand.max()
                                cur_score = cand_score

        # ----- 6.6) Restart on stagnation --------------------------------
        if now - last_improve > restart_gap:
            # Soft restart: jitter the best solution and reset Adam state
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 9) Return the best vector (apply evaluate_sequence's clipping limits)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
