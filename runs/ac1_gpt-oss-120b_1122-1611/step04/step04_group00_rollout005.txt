"""
Propose a step‑function sequence that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a projected exponentiated‑gradient descent on the probability
simplex together with a temperature schedule that smoothly turns the
soft‑max approximation of the max into the exact max.  The algorithm
starts from an existing solution (``height_sequence_1``) if present,
otherwise from a random Dirichlet vector.  It periodically applies
log‑normal block perturbations and a simple restart on stagnation.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, may be used for occasional LP steps


def _next_pow2(x: int) -> int:
    """
    Smallest power of two that is >= x (x >= 1).
    """
    if x <= 1:
        return 1
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative vector *a* using FFT.
    The result has length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a*a)_i w.r.t. *a*.
    Equivalent to 2·(w ⊛ rev(a)) restricted to the original length.
    """
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator,
                         min_width: int = 1, max_ratio: float = 0.2,
                         sigma: float = 0.15) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex.
    """
    n = a.size
    width = rng.integers(min_width, max(2, int(n * max_ratio)) + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    # keep strictly positive to stay off the simplex boundary
    out = np.maximum(out, 1e-12)
    out /= out.sum()
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that
    minimises ``evaluate_sequence`` before the time budget elapses.
    The routine runs a temperature‑controlled exponentiated‑gradient
    descent on the simplex with occasional random block perturbations
    and a stagnation‑restart mechanism.
    """
    rng = np.random.default_rng(seed)
    # safety margin – leave a few seconds for clean‑up / printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a feasible starting vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                if a.sum() == 0.0:
                    a = np.full_like(a, 1.0 / a.size)
                else:
                    a = a / a.sum()                     # normalise to a simplex
            else:
                raise ValueError
        except Exception:
            a = rng.dirichlet(np.ones(1024)).astype(np.float64)   # fallback
    else:
        # random Dirichlet of length 1024 (power‑of‑two is convenient)
        a = rng.dirichlet(np.ones(1024)).astype(np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) FFT‑related pre‑computations
    # ------------------------------------------------------------------
    fft_len_conv = _next_pow2(2 * n - 1)      # for self‑convolution
    fft_len_grad = _next_pow2(3 * n - 2)      # for gradient convolution

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_conv = _conv_self(best, fft_len_conv)

    start_time = time.monotonic()

    eta_end = 0.001          # final learning‑rate

    momentum = 0.85

    # ------------------------------------------------------------------

    while time.monotonic() < deadline:

        conv = _conv_self(a, fft_len_conv)

        grad = _gradient_from_weights(a, w, fft_len_grad)

        a = a * np.exp(-eta * grad_mom)

        if rng.random() < block_perturb_prob:

        # ----- soft restart on stagnation -----

            a = best * noise
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            grad_mom.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        print(f"Final evaluate_sequence score: {final_eval:.6f}")
    except Exception:
        pass
    return [float(x) for x in best]
