    L = len(x) + len(y) - 1

"""
Propose a step function that improves the current bound (< 1.5025).

Algorithm Overview
------------------
1. **Initialisation** – start from a Dirichlet draw (or the previous best
   sequence if available) and normalise the vector to a probability
   distribution (∑a = 1).

2. **Phase 1 – Adam‑based soft‑max mirror descent**  
   * Objective: minimise a smooth approximation of the
     max‑convolution ‑ `softmax_β(conv) = log∑exp(β·conv)`.  
   * For a given temperature β we compute the weights
     `w_k = exp(β·(conv_k‑max(conv)))`.  
   * The gradient of the smoothed objective is
     `∂/∂a_i = 2·∑_k w_k·a_{k‑i}` – efficiently obtained by an FFT‑based
     convolution of `w` with the reversed `a`.  
   * An Adam optimiser (with exponential‑moving‑average of the gradient
     and its square) performs an *exponentiated‑gradient* (multiplicative)
     update, keeping the iterates on the simplex.
   * β grows linearly from 0.5 to 800, while the stepsize η shrinks from 1.0
     to 0.001.

3. **Phase 2 – Stochastic local refinement** (simulated annealing)  
   Pairwise mass transfers and block‑perturbations are applied;
   acceptance follows a decreasing temperature schedule.

4. **Phase 3 – Length refinement & final polishing**  
   Occasionally the current best distribution is *up‑sampled* (linear
   interpolation) to a longer length (1.5 × or 2 ×) and the whole
   optimisation restarts on this finer grid.  Random jitter on the best
   solution is also tried.

5. **Restarts** – if no improvement occurs for 30 s the algorithm restarts
   from the best known solution with a small multiplicative noise.

The routine respects the 1000 s budget, prints progress, and finally
returns the best non‑negative step heights found (already clipped to the
`evaluate_sequence` limits)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution
    of two vectors of length *n* (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_sizes(s1: int, s2: int) -> int:
    """Power‑of‑two length for the convolution of lengths *s1* and *s2*."""
    size = s1 + s2 - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector `a` using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of `x` and `y` using an FFT of length `fft_len`."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Soft restart if no improvement for a while
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            t_iter = 0
            conv = _conv_self(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by
    #    `evaluate_sequence` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
