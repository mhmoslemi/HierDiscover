Propose a candidate step‑function (non‑negative heights) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation combines three ideas that are not used together in the
reference code:

1. **Coarse parametric sweep** – a handful of simple analytic families

   This yields a much sharper sub‑gradient of the max‑norm and drives the
   optimisation faster towards a flat convolution.

3. **Exponentiated‑gradient with Adam‑style per‑coordinate adaptivity**.
   The update `a_i ← a_i·exp(-η·g_i)` automatically keeps the vector on the
   probability simplex.  Adam‑style momentum (`m`) and second‑moment (`v`)
   are kept for every coordinate, allowing larger steps early on and fine
   tuning later.

The optimiser runs until the time budget expires (leaving a few seconds
margin for the harness).  It periodically evaluates the *exact* objective
via `evaluate_sequence` and keeps the best vector seen.  Random block
perturbations and occasional large Gaussian noise prevent the search from
getting stuck in a local minimum.

The function returns the best sequence found, clipped to the limits imposed
by ``evaluate_sequence`` (0 ≤ x ≤ 1000).

# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _topk_softmax_weights(conv: np.ndarray,
                          beta: float,
                          k_frac: float = 0.01) -> np.ndarray:
    """
    Returns a probability vector w of the same shape as ``conv`` that
    concentrates its mass on the largest ``k`` entries (k = max(2,
    int(k_frac*len(conv)))) using a soft‑max with temperature ``beta``.
    """
    n_conv = conv.size
    k = max(2, int(k_frac * n_conv))
    # indices of the k largest entries (unsorted)
    idx = np.argpartition(conv, -k)[-k:]
    # shift for numerical stability
    shift = conv[idx].max()
    raw = np.exp(beta * (conv[idx] - shift))
    s = raw.sum()
    w = np.zeros_like(conv)
    if s > 0.0:
        w[idx] = raw / s
    else:
        # fall back to a uniform distribution over the selected indices
        w[idx] = 1.0 / k
    return w


def _analytic_family(a_type: str,
                     p: float,
                     n: int) -> np.ndarray:
    """
    Return a normalised distribution of length ``n`` for a simple analytic
    family.
      * ``'decr'`` – monotone decreasing power law  (1‑x)^p
      * ``'U'``    – symmetric U‑shape          x^p + (1‑x)^p
      * ``'bump'`` – centred bump               (x·(1‑x))^p
    ``x`` runs from 0 to 1 (inclusive at both ends).
    """
    idx = np.arange(n, dtype=np.float64)
    x = idx / (n - 1)

    if a_type == "decr":
        vals = (1.0 - x) ** p
    elif a_type == "U":
        vals = (x ** p) + ((1.0 - x) ** p)
    elif a_type == "bump":
        vals = (x * (1.0 - x)) ** p
    else:
        raise ValueError("unknown family")
    # avoid an all‑zero vector when p is huge
    if np.all(vals == 0):
        vals[:] = 1.0
    a = vals / vals.sum()
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine first scans a few analytic families, then refines the best
    candidate using a top‑K soft‑max exponentiated‑gradient optimiser with
    Adam‑style adaptive steps and occasional stochastic perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine problem size -------------------------------------------------
    # ------------------------------------------------------------------
    default_n = 4096
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a0 = np.clip(cand, 0.0, None)
                s0 = a0.sum()
                if s0 > 0.0:
                    a0 = a0 / s0
                else:
                    a0 = np.full_like(cand, 1.0 / cand.size, dtype=np.float64)
                n = a0.size
                a = a0.copy()
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Quick sweep over a few analytic families
    # ------------------------------------------------------------------
    best_score = float("inf")
    best = a.copy()          # fallback – will certainly be overwritten
    families = [("decr", np.linspace(0.0, 4.0, 17)),
                ("U",    np.linspace(0.0, 4.0, 17)),
                ("bump", np.linspace(0.0, 4.0, 17))]
    print("[0.00s] Starting analytic sweep (n = {})".format(n))
    sweep_start = time.monotonic()

    for ftype, plist in families:
        for p in plist:
            a_try = _analytic_family(ftype, p, n)
            # exact evaluation (the very cheap check – the harness clamps the
            # vector internally, we do the same.)
            score = evaluate_sequence(a_try.tolist())
            if score < best_score:
                best_score = score
                best = a_try.copy()
                print("[{:.2f}s] New best from {}(p={:.3f}) score = {:.6f}"
                      .format(time.monotonic() - sweep_start,
                              ftype, p, best_score))

    # make the current iterate the best found so far
    a = best.copy()

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    cur_conv = _conv_self(a, fft_len)
    cur_max = float(cur_conv.max())
    cur_score_est = 2.0 * n * cur_max / (np.sum(a) ** 2)          # sum(a)==1 → 2*n*max
    best_score = cur_score_est if cur_score_est < best_score else best_score
    best = a.copy() if cur_score_est <= best_score else best
    last_improve = time.monotonic()
    start_time = last_improve

    # ------------------------------------------------------------------
    # 4) Optimiser hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 0.9
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 2000.0          # sharper max‑approximation
    momentum = 0.95            # exponential smoothing of the gradient
    grad_prev = np.zeros_like(a)

    # Adam‑style accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9                # 1st‑moment decay
    beta2 = 0.999              # 2nd‑moment decay
    eps_adam = 1e-8

    block_perturb_prob = 0.025
    large_perturb_prob = 0.005
    sigma_large = 0.5          # log‑normal factor for large jumps

    restart_gap = 45.0         # seconds without improvement → soft restart
    print_interval = 20.0
    next_print = start_time + print_interval

    k_frac = 0.01              # fraction of entries used in the top‑k soft‑max

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ---- schedules for η and β (linear ramps) ------------------------
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- current convolution and estimated score ----------------------
        conv = _conv_self(a, fft_len)
        max_conv = float(conv.max())
        cur_score_est = 2.0 * n * max_conv / (np.sum(a) ** 2)    # Σa = 1 → 2*n*max

        # ---- exact evaluation when we see an improvement -----------------
        if cur_score_est < best_score:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best = a.copy()
                last_improve = now
                print("[{:.2f}s] NEW BEST (exact) score = {:.6f}"
                      .format(elapsed, best_score))

        # ---- periodic status output ---------------------------------------
        if now >= next_print:
            print("[{:.2f}s] est. score = {:.6f}   best = {:.6f}"
                  .format(elapsed, cur_score_est, best_score))
            next_print = now + print_interval

        # ---- top‑k soft‑max weighting ------------------------------------
        w = _topk_softmax_weights(conv, beta, k_frac)

        # ---- gradient of the soft‑max objective (mirror‑descent) ----------
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # momentum smoothing (helps with the noisy β‑approximation)
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- Adam‑style adaptive scaling ---------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        t = now - start_time + 1.0                     # iteration count approx.
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        adap_step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- exponentiated‑gradient update (mirror descent on simplex) --
        factor = np.exp(-adap_step)
        a *= factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- occasional stochastic perturbations ---------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < large_perturb_prob:
            noise = np.exp(rng.normal(0.0, sigma_large, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # ---- soft restart when stagnating ---------------------------------
        if now - last_improve > restart_gap:
            # jitter the best known point and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            last_improve = now
            print("[{:.2f}s] restart – best score {:.6f}".format(elapsed,
                                                                 best_score))

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
