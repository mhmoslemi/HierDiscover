def build_conv_matrix(f):
    n = len(f)
    m = 2*n -1
    data = []
    rows = []
    cols = []
    for j in range(n):
        # shift f by j: for k from j to j+n-1 inclusive (within 0..m-1)
        # For k in range(j, j+n):
        #   i = k - j => f[i]
        #   column = j
        #   row = k
        # So for i in range(n):
        #   row = i + j
        #   col = j
        #   data = f[i]
        i_vals = np.arange(n)
        rows.extend(i_vals + j)   # will be list of ints
        cols.extend([j] * n)
        data.extend(f.tolist())
    # Build csr
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

"""
Propose a high‑quality step function.

Algorithm (two phases)

Phase‑1 – Linear‑programming boost
  • Keep the total mass fixed to √(2·n) (this makes the
    evaluation score equal to the maximal self‑convolution).
  • For the current sequence f compute M = max(conv(f,f)).
  • Solve a linear program:
        maximise   Σ b_j
        subject to conv(f,b) ≤ M   (all 2·n‑1 convolutions)
                  b ≥ 0
    The LP tries to increase the total mass without enlarging the
    worst convolution value.
  • Scale the LP solution to the prescribed total mass and mix it
    with f (mixing weight t is gradually reduced).  Whenever the
    maximal convolution decreases we keep the change and record a
    new best sequence.

Phase‑2 – Sub‑gradient mirror‑descent refinement
  • After the LP phase, switch to a fast sub‑gradient scheme that
    directly minimises the maximal convolution.
  • At each step locate the index p where conv(f,f) is maximal.
    The sub‑gradient of max(conv) w.r.t. f is
        g_i = 2·f_{p‑i}  (if 0≤p‑i<n, else 0).
    An exponentiated‑gradient (mirror‑descent on the simplex)
    update with a decreasing stepsize η drives mass away from the
    entries that contribute most to the peak.
  • Occasional log‑normal block perturbations and soft restarts
    keep the search from stagnating.

Both phases respect the non‑negativity and per‑element clipping
required by ``evaluate_sequence``.  The routine stops just before
the time budget expires and returns the best vector found.
"""

import time
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using FFT (fast for n up to a few 10⁴)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the required total mass."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and combines a
    linear‑programming phase with a sub‑gradient mirror‑descent phase.
    """
    rng = np.random.default_rng(seed)

    # ----------------------- 0) bookkeeping ----------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------- 1) initialise ---------------------------
    # Prefer an existing construction if present.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
            else:
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # Fallback: a power‑of‑two length is convenient for FFT.
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # Target total mass (makes the score equal to the maximal convolution)
    target_sum = np.sqrt(2.0 * n)

    # Normalise to the target total mass.
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full(n, target_sum / n, dtype=np.float64)

    a = np.clip(a, 0.0, None)

    # Pre‑compute FFT length for fast convolutions.
    fft_len = _fft_len_for_n(n)

    # ------------------- 2) bookkeeping for best solution -----------
    def exact_score(vec):
        """Exact evaluation via the supplied ``evaluate_sequence``."""
        return evaluate_sequence(vec.tolist())

    cur_conv = _conv_self(a, fft_len)
    M = float(cur_conv.max())
    best = a.copy()
    best_score = exact_score(a)
    print(f"[0.00s] initial exact score = {best_score:.6f}")

    start_time = time.monotonic()
    last_improve = start_time

    # ------------------- 3) Phase‑1 : LP boost -----------------------
    # Build index arrays for the convolution matrix once – the coefficient
    # pattern is always the same (each column is a shifted copy of f).
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    idx = 0
    for j in range(n):
        rows[idx : idx + n] = np.arange(n) + j
        cols[idx : idx + n] = j
        idx += n
    # Shape of the linear‑convolution matrix (2n‑1) × n
    conv_shape = (2 * n - 1, n)

    # LP hyper‑parameters
    t_mix = 0.5            # initial mixing weight with the LP solution
    min_tmix = 0.05
    t_decay = 0.93

    lp_iterations = 0
    max_lp_iterations = 30   # safety guard – we stop earlier if time runs out

    while (time.monotonic() < deadline) and (lp_iterations < max_lp_iterations):
        # Stop the LP phase when only a few seconds remain.
        if deadline - time.monotonic() < 10.0:
            break

        # -------- build the linear constraints (A·b ≤ M) ----------
        # Data of the matrix is simply the current vector a repeated n times.
        data = np.tile(a, n)                 # shape (n*n,)
        A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=conv_shape)

        # -------- solve the LP ------------------------------------
        c = -np.ones(n)                      # maximise Σb   → minimise -Σb
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Using HiGHS (fast interior‑point) solver.
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # LP occasionally fails (numerical issues).  Recover by a tiny
            # random perturbation and continue.
            print(f"[LP] iteration {lp_iterations}: linprog failed – {res.message}")
            a = _random_block_perturb(a, rng)
            continue

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            # Degenerate solution – fall back to a perturbation.
            a = _random_block_perturb(a, rng)
            continue

        # -------- scale the LP solution to the prescribed mass ----
        g = b * (target_sum / sum_b)

        # -------- mix the old and new vectors --------------------
        f_new = (1.0 - t_mix) * a + t_mix * g
        # Renormalise explicitly (numerical drift may appear)
        f_new = np.clip(f_new, 0.0, None)
        f_new_sum = f_new.sum()
        if f_new_sum == 0.0:
            f_new = np.full_like(f_new, target_sum / n)
        else:
            f_new = f_new * (target_sum / f_new_sum)

        # -------- evaluate improvement ----------------------------
        new_conv = _conv_self(f_new, fft_len)
        M_new = float(new_conv.max())

        if M_new < M - 1e-12:                # genuine improvement
            a = f_new
            M = M_new
            cur_score = exact_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(
                    f"[{elapsed:6.2f}s] LP improvement: exact score = {best_score:.6f} (M={M:.6f})"
                )
            # gradually reduce the mixing rate – we want finer steps later.
            t_mix = max(min_tmix, t_mix * t_decay)
            lp_iterations += 1
        else:
            # No improvement – try a small random jitter.
            a = _random_block_perturb(a, rng)

        # occasional status line
        now = time.monotonic()
        if now - last_improve > 30.0:
            elapsed = now - start_time
            print(
                f"[{elapsed:6.2f}s] LP phase ongoing – M={M:.6f}, best score={best_score:.6f}"
            )
            last_improve = now

    # ------------------- 4) Phase‑2 : Sub‑gradient descent ----------
    # Parameters for the mirror‑descent phase
    eta_start = 2.0
    eta_end = 0.02
    momentum = 0.9
    grad_prev = np.zeros_like(a)

    block_perturb_prob = 0.02
    restart_gap = 40.0     # seconds without improvement → restart
    print_interval = 30.0

    next_print = time.monotonic() + print_interval

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- linear schedules for stepsize ---------------------------------
        total_time = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total_time) + eta_end * (elapsed / total_time)

        # ---- compute full self‑convolution ---------------------------------
        conv = _conv_self(a, fft_len)
        max_conv = float(conv.max())
        cur_score_est = max_conv  # because total mass = √(2n) → score = max_conv

        # ---- update best record (exact evaluation) -------------------------
        if cur_score_est < best_score:
            exact = exact_score(a)
            if exact < best_score:
                best_score = exact
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best (exact) score = {best_score:.6f}")

        # ---- periodic status print -----------------------------------------
        if now >= next_print:
            print(
                f"[{elapsed:6.2f}s] sub‑grad est. max‑conv = {cur_score_est:.6f} | best = {best_score:.6f}"
            )
            next_print = now + print_interval

        # ---- locate the peak index p ---------------------------------------
        p = int(np.argmax(conv))

        # ---- sub‑gradient of max‑convolution (2·a_{p‑i}) -------------------
        subgrad = np.zeros_like(a)
        lo = max(0, p - (n - 1))
        hi = min(p, n - 1)
        idxs = np.arange(lo, hi + 1, dtype=int)
        subgrad[idxs] = 2.0 * a[p - idxs]       # g_i = 2·a_{p‑i}

        # ---- momentum smoothing --------------------------------------------
        subgrad = momentum * grad_prev + (1.0 - momentum) * subgrad
        grad_prev = subgrad

        # ---- exponentiated‑gradient (mirror descent on the simplex) --------
        # To avoid overflow we clip the exponent.
        exponent = -eta * subgrad
        exponent = np.clip(exponent, -40, 40)    # safe numerical range
        a = a * np.exp(exponent)

        # ---- renormalise to the fixed total mass ---------------------------
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, target_sum / n)
        else:
            a = a * (target_sum / s)

        # ---- occasional block perturbation ----------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ---- soft restart if stagnating ------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # small log‑normal jitter around the best point
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
