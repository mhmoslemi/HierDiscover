"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a))

where conv(a,a) is the linear self‑convolution of *a*.
The method consists of two phases:

1. **Full‑gradient smooth‑max mirror descent** – the true smooth
   max (soft‑max) of the convolution is minimised by an exponentiated‑gradient
   (mirror‑descent) update.  The gradient w.r.t. the soft‑max is computed
   efficiently by a single FFT‑based convolution (≈ 3 FFTs per iteration).
   Temperature β grows from 2 to ~800, while the stepsize η decays from 0.8
   to 0.001.  Momentum (0.85) stabilises the updates.  Random block
   perturbations keep the search from stagnating.

2. **Stochastic local refinement** – in the last ~30 % of the time budget a
   simulated‑annealing style search performs pairwise mass transfers and
   block‑perturbations, accepting worsening moves with a linearly decreasing
   temperature.  This polishes the solution found by phase 1.

The routine respects the 1000 s time budget, restarts when no improvement
is seen for 25 s, and returns the best vector found (clipped to the limits
checked by ``evaluate_sequence``).  If a previous construction is available
via the global variable ``height_sequence_1`` it is used as the initial
vector; otherwise a Dirichlet draw of length 512 is taken.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two length‑n vectors."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector `a` using FFT.
    Returns an array of length 2*len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    # block width: at most 10 % of the sequence, at least 1
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose an initial probability vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0:
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full_like(a, 1.0 / a.size)
        else:
            a = np.full(512, 1.0 / 512, dtype=np.float64)
    else:
        n0 = 512
        a = rng.dirichlet(np.full(n0, 0.5)).astype(np.float64)  # mildly sparse start

    n = a.size
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.70 * total_time   # 70 % for smooth‑max descent

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # schedules (exponential for better control)
    eta_start = 0.80
    eta_end = 0.001
    beta_start = 2.0
    beta_end = 800.0

    momentum = 0.85
    grad_prev = np.zeros_like(a)

    restart_gap = 25.0      # seconds without improvement before a soft restart

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_total = (now - start_time) / (deadline - start_time)

        # --------------------------------------------------------------
        # Phase 1 – smooth‑max mirror descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # exponential schedules
            eta = eta_start * (eta_end / eta_start) ** frac_total
            beta = beta_start * (beta_end / beta_start) ** frac_total

            # FFT of the current vector
            a_fft = np.fft.rfft(a, n=fft_len)

            # self‑convolution (objective component)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
            max_conv = conv.max()

            # ----- smooth‑max weights (softmax) -----
            # use max_conv for numerical stability
            w_exp = np.exp(beta * (conv - max_conv))
            w = w_exp / w_exp.sum()

            # ----- gradient of smooth‑max -----
            # convolution w * a_rev (a_rev = reversed a)
            a_rev = a[::-1]
            a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
            w_fft = np.fft.rfft(w, n=fft_len)

            grad_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)[: 2 * n - 1]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # length n

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient (mirror‑descent) update on simplex
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # occasional random block perturbation to maintain diversity
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)

            # re‑evaluate score
            cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # Phase 2 – stochastic local refinement (simulated annealing)
        # --------------------------------------------------------------
        else:
            # linear temperature schedule
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = max(1e-8, 0.08 * (1.0 - t_frac))

            # ----- pairwise mass transfer -----
            if rng.random() < 0.85:
                # donor with non‑trivial mass
                i = rng.integers(0, n)
                if a[i] < 1e-12:
                    # fallback: random block jitter
                    a = _random_block_perturb(a, rng)
                    conv = _conv_self(a, fft_len)
                    cur_score = 2.0 * n * conv.max()
                    continue

                # distinct recipient
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1

                # move at most 30 % of donor's mass
                max_delta = 0.30 * a[i]
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta

                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta

                conv_new = _conv_self(a_new, fft_len)
                new_score = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                    conv = conv_new
                    cur_score = new_score

            # ----- block perturbation (global jitter) -----
            else:
                a = _random_block_perturb(a, rng)
                conv = _conv_self(a, fft_len)
                cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart when stuck
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final polishing – a few greedy hill‑climbs on the best vector
    # ------------------------------------------------------------------
    polish_deadline = time.monotonic() + min(5.0, deadline - time.monotonic())
    while time.monotonic() < polish_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue
        delta = rng.random() * 0.15 * best[i]
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        conv_cand = _conv_self(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_score:
            best_score = cand_score
            best = cand

    # ------------------------------------------------------------------
    # 6) Return the best vector (clipped to the limits enforced by
    #    `evaluate_sequence` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
