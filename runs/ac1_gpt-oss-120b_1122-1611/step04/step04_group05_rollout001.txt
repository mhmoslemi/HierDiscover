Search for a non‑negative step‑function (list of heights) that minimises

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_linear(v: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    L = v.size + w.size - 1
    fft_len = _fft_len_for_len(L)
    fv = np.fft.rfft(v, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(fv * fw, n=fft_len)[:L]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution entries."""
    max_c = conv.max()
    # shift by max for numerical stability
    w = np.exp(beta * (conv - max_c))
    s = w.sum()
    if s == 0.0:
        # fallback to uniform weights (should not happen)
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w / s
    return w

    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:

    else:
        out /= s
    return out

            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final local improvement (pairwise hill‑climbing)
    # ------------------------------------------------------------------
    # A short additional refinement using cheap pairwise moves.
    # Limited to the remaining few seconds (if any).
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        max_delta = min(best[i], 0.25 * best[i])
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()
        conv_cand = _conv_self(cand, fft_len_a)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] local improve → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
