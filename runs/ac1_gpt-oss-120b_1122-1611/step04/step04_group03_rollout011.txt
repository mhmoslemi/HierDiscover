```
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    sequence = [max(0, x) for x in sequence]
```

The evaluation function is  

\[
S(a)=\frac{2\,n\;\max\bigl(a\star a\bigr)}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge0 .
\]

Because the denominator is homogeneous, we can keep the vector on the
probability simplex (`∑a = 1`).  
The term `max(a∗a)` is the infinity‑norm of the *self‑convolution* and
is a non‑smooth function of the coefficients.  
We minimise a *smooth* approximation of the maximum
by a soft‑max (log‑sum‑exp) and apply exponentiated‑gradient (mirror
descent on the simplex).  To avoid getting stuck in a local minimum we
add a small set of stochastic operators (block jitter, swaps,
full permutation, occasional restart from the best solution).  

The whole procedure runs inside the supplied time budget; the best
vector seen so far is returned.

```python
"""
Hybrid mirror‑descent + evolutionary perturbations.

* keep the candidate on the probability simplex (∑a = 1)
* smooth the max‑convolution with a soft‑max whose temperature β grows
* exponentiated‑gradient (mirror descent with negative entropy) updates
  a in the log‑domain
* occasionally jitter a random block, swap two coordinates,
  apply a full random permutation and perform soft‑restarts.
* a simple hill‑climbing step (mass transfer) is also tried.
* the best vector seen before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT (real‑valued)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:n]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator,
                         sigma: float = 0.15) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renorm."""
    n = a.size
    # block width between 1 and max(2, n//8)
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _mass_transfer(a: np.ndarray, rng: np.random.Generator,
                   eps: float = 0.005) -> np.ndarray:
    """Move a tiny amount of mass from i to j (i≠j) and renormalise."""
    n = a.size
    i, j = rng.choice(n, size=2, replace=False)
    # amount to move is a fraction of the source entry, bounded by eps
    delta = min(a[i] * eps, eps * 0.1)
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    # numerical safety
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _score_simple(a: np.ndarray) -> float:
    """
    Exact evaluation for a vector that is already on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = np.convolve(a, a)                # full linear convolution
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine stays inside the given time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget – leave a small safety margin.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initial vector – use the supplied construction if present,
    #    otherwise start from a random Dirichlet point.
    # ------------------------------------------------------------------
    default_n = 4096                         # power of two – good for FFT
    if "height_sequence_1" in globals():
        try:
            a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a.ndim != 1 or a.size == 0:
                raise ValueError
        except Exception:
            a = rng.dirichlet(np.ones(default_n))
    else:
        a = rng.dirichlet(np.ones(default_n))

    # keep everything non‑negative and on the simplex
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s

    n = a.size

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = _score_simple(best)
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (tuned empirically)
    # ------------------------------------------------------------------
    # mirror‑descent schedule
    eta_start = 0.8          # initial stepsize
    eta_end   = 0.0004       # final stepsize
    beta_start = 0.05        # start temperature
    beta_end   = 2500.0      # final temperature (very sharp)

    momentum = 0.9           # exponential smoothing of the gradient
    block_perturb_prob = 0.12
    swap_prob = 0.12
    perm_prob = 0.03         # full random permutation
    transfer_prob = 0.08     # tiny mass transfer
    restart_gap = 12.0       # seconds without improvement → restart
    jitter_sigma = 0.20      # log‑normal jitter during restart

    grad_prev = np.zeros_like(a)

    # FFT lengths (fixed for this run)
    fft_len_self = _next_pow2(2 * n - 1)          # for a∗a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w∗a_rev

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- current self‑convolution and score -----------------------
        conv = _conv_self_fft(a)                 # length 2n‑1, FFT based
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max            # sum(a)=1 by construction

        # ---- keep the best solution ----------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule (linear interpolation) -------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- soft‑max weights (smooth max of the convolution) --------
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                 # under‑flow – fall back to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective -----------------------
        a_rev = a[::-1]                 # reversed copy
        # convolution w * a_rev (length 3n‑2)
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)
        # slice the part that corresponds to the gradient (indices n‑1 … n‑1+n‑1)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ---- momentum smoothing ---------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- exponentiated‑gradient (mirror descent) update -----------
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -50.0, 50.0)   # numerical safety
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ----------------------------------------------------------------
        # 5) Stochastic perturbations to escape plateaus
        # ----------------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng, sigma=0.15)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        if rng.random() < perm_prob:
            a = a[rng.permutation(n)]

        if rng.random() < transfer_prob:
            a = _mass_transfer(a, rng, eps=0.005)

        # ----------------------------------------------------------------
        # 6) Soft restart when progress stalls
        # ----------------------------------------------------------------
        if now - last_improve > restart_gap:
            # restart from the best solution plus a gentle log‑normal jitter
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a[:] = 1.0 / n
            else:
                a /= a_sum
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best solution (safety clipping for the external
    #    ``evaluate_sequence`` routine)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
```
