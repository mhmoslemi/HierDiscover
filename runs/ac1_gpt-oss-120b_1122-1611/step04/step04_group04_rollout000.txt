"""
Propose a candidate step‑function minimizing `evaluate_sequence`.
The algorithm is a three‑phase Adam‑style optimizer on the simplex
with a smooth‑max (log‑sum‑exp) surrogate for the L∞ self‑convolution.
A symmetric‐enforcement step reduces the search space, and occasional
random block perturbations help escape local minima.  Restarts are
triggered after a period without improvement.  The method works for
any length `n` (default ≈ 8000) and respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries and clips tiny negatives.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # eliminate numerical noise
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k·(a*a)_k.
    The formula is   ∂/∂a_j Σ_k w_k·conv_k = 2· Σ_i a_i·w_{i+j}
    which is a cross‑correlation between a and w.
    """
    n = a.size
    Lw = w.size                         # = 2*n-1
    wr = w[::-1]                        # reverse for correlation

    conv_len = _next_pow2(n + Lw - 1)    # length needed for convolution
    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)[: n + Lw - 1]
    idx = Lw - 1 - np.arange(n)         # map correlation to gradient indices
    grad = 2.0 * conv_full[idx]
    return grad


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x >= 0, Σ x = 1}.
    Implements the algorithm of Duchi et al. (2008) in O(n log n).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index of rho
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safeguard: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a multiplicative log‑normal perturbation to a random contiguous
    block of the vector, then renormalise onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _proj_simplex(out)


def _pairwise_transfer(a: np.ndarray, conv: np.ndarray, rng: np.random.Generator):
    """
    Cheap stochastic mass transfer between two indices.
    Returns (new_a, new_conv, new_score) if improvement is found,
    otherwise the original tuple (a, conv, None).
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    # Transfer at most 30 % of donor's mass
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Update convolution incrementally
    new_conv = conv.copy()
    lo_i, hi_i = i, i + n
    lo_j, hi_j = j, j + n
    new_conv[lo_i:hi_i] -= delta * a
    new_conv[lo_j:hi_j] += delta * a

    new_score = 2.0 * n * new_conv.max()
    cur_score = 2.0 * n * conv.max()
    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.clip(a2, 0.0, None)
        a2 = _proj_simplex(a2)
        return a2, new_conv, new_score
    else:
        return a, conv, None


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function minimizing `evaluate_sequence`
    within the given time budget.  Returns the best sequence (list of floats)
    discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size & initial vector
    # ------------------------------------------------------------------
    # Use an existing good construction if present, else start fresh.
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            a = a / max(a.sum(), 1e-12)
        else:
            n = 8192
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # choose a somewhat larger size than the previous best to give
        # the optimizer more freedom.
        n = 8192
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # FFT length for self‑convolution (next power of two)
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _self_conv_fft(best, fft_len_self)
    best_score = 2.0 * n * conv_best.max()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (tuned for a 1000 s run)
    # ------------------------------------------------------------------
    eta_start = 0.5          # initial learning rate (additive)
    eta_end = 0.001          # final learning rate
    beta_start = 0.01        # start temperature (soft‑max is very soft)
    beta_end = 4000.0        # final temperature – close to hard max
    beta_hard_frac = 0.85    # fraction of total time after which we use hard max
    beta_half = 0.5          # for smoothing softmax (optional)
    momentum = 0.0           # not used – Adam replaces it
    block_perturb_prob = 0.03
    pairwise_prob = 0.07
    restart_gap = 30.0       # seconds without improvement → restart

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        # ----- compute self‑convolution -----
        conv = _self_conv_fft(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule for learning rate and temperature -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- construct weight vector w -----
        if frac < beta_hard_frac:
            # soft‑max (stable: subtract cur_max)
            w = np.exp(beta * (conv - cur_max))
            w_sum = w.sum()
            if w_sum > 0.0:
                w /= w_sum
            else:
                # numerical safeguard: uniform weights
                w = np.full_like(conv, 1.0 / conv.size)
        else:
            # hard max – sub‑gradient from the max entry only
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ----- gradient of the surrogate -----
        grad = _gradient_from_weights(a, w)

        # ----- Adam update (additive on the simplex) -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = -eta * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a + step

        # ----- project onto probability simplex -----
        a = _proj_simplex(a_candidate)

        # ----- enforce symmetry (reduces dimensionality & often helps) -----
        a = (a + a[::-1]) * 0.5

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- cheap stochastic pairwise mass transfer -----
        if rng.random() < pairwise_prob:
            a, conv, _ = _pairwise_transfer(a, conv, rng)

        # ----- restart if stuck for too long -----
        if now - last_improve > restart_gap:
            # re‑initialize around the best known point with small noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.14, size=n))
            a = a * noise
            a = _proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best sequence (clipped to evaluate_sequence limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
