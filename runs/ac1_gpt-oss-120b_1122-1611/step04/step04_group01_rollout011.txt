```
def _conv_self_fft(a):
    n = a.size
    fft_len = 1 << ((2*n - 2).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[:2*n-1]

def _score(a):
    n = a.size
    conv = _conv_self_fft(a)
    m = conv.max()
    return 2.0 * n * m
```

```
def _beta_distribution(n, alpha, beta):
    xs = (np.arange(n) + 0.5) / n
    pdf = np.power(xs, alpha-1) * np.power(1 - xs, beta-1)
    pdf[np.isnan(pdf)] = 0.
    pdf = np.maximum(pdf, 0.)
    total = pdf.sum()
    if total == 0:
        return np.full(n, 1.0/n)
    return pdf/ total
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 ,
\]

where `a*a` is the linear self‑convolution of the coefficient vector `a`
and `n = len(a)`.  
Because the objective is homogeneous we may keep the coefficients on the
probability simplex (`∑a = 1`). With this normalisation  

\[
\text{score}(a)=2\,n\;\max (a*a) .
\]

The optimisation problem is non‑convex, but its landscape is smooth
enough for a combination of **gradient descent on a soft‑max
approximation of the max‑norm** and **stochastic annealing** to escape
local minima.  The algorithm also uses a few cheap local moves
(pair‑wise mass transfer and block‑wise log‑normal perturbations) and
periodic restarts with jitter.

The implementation follows these ideas:

* **Simplex projection** – guarantees feasibility (`a_i ≥ 0`, `∑a = 1`).
* **FFT‑based convolutions** – cheap `O(n log n)` evaluation of the
  self‑convolution and of the gradient of the smooth‑max.
* **Soft‑max smoothing** – replaces `max` by  

  \[
  f_\beta(a)=\frac{1}{\beta}\log\!\sum_k \exp\!\bigl(\beta\,(a*a)_k\bigr) .
  \]

  Its gradient can be obtained with one extra convolution.
* **Adam‑style adaptive step size** – stabilises the descent on the
  simplex.
* **Simulated‑annealing acceptance** – occasional uphill moves are
  accepted with probability `exp(-Δ/ T)` where the temperature `T`
  decays during the run.
* **Exploration moves** –  
  * multiplicative log‑normal perturbation of a random block,  
  * pairwise transfer of a small amount of mass,  
  * full‑restart with jitter when the search stalls.
* **Multiple restarts** – the outer loop creates a fresh initial vector
  (either a random exponential shape or a jittered copy of the current
  best) and runs a limited inner optimisation.  
  The best vector ever seen is kept as the final answer.

All helper functions are defined at top level, there are no closures,
no file I/O and the code respects the 1000 s budget.  The routine
returns a plain list of non‑negative floats that can be fed directly to
`evaluate_sequence`.

```python
"""
Hybrid stochastic‑annealing + Adam optimisation on the probability simplex.

The algorithm keeps the coefficient vector on the simplex (∑a = 1, a_i ≥ 0)
and minimises   score = 2·n·max(conv(a,a)).

* a soft‑max approximation of the max‑norm provides a smooth gradient;
* Adam supplies an adaptive step;
* a temperature schedule enables occasional uphill moves (simulated annealing);
* cheap local moves (block perturbation, pairwise mass transfer) add exploration;
* periodic restarts with jitter prevent stagnation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# linear‑algebra helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """
    Smallest power‑of‑two length that can contain a linear convolution of
    two vectors whose total length is ``length``.
    """
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution (full length 2·len(a)‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``x`` and ``y`` via FFT.
    Returned length is len(x)+len(y)-1 (truncated to that size).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# random perturbations
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and project back
    onto the simplex.
    """
    n = a.size
    # block length between 1% and 10% of the vector
    min_len = max(1, n // 100)
    max_len = max(2, n // 10)
    width = rng.integers(min_len, max_len + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


# ----------------------------------------------------------------------
# score and gradient utilities
# ----------------------------------------------------------------------


def _score_simplex(a: np.ndarray, fft_len: int) -> float:
    """
    Score for a vector that already satisfies ∑a = 1:
    2·n·max(conv(a,a)).
    """
    n = a.size
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


def _softmax_gradient(a: np.ndarray, beta: float, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max   fβ(a) = (1/β) log Σ exp(β·conv_i(a))
    w.r.t. a.  The formula is

        ∂fβ/∂a = a + (conv(w, a_rev) / Σ w),

    where w_i = exp(β·conv_i) and a_rev is a reversed copy of a.
    """
    conv = _conv_self(a, fft_len)
    cur_max = conv.max()
    # avoid overflow by subtracting the maximum value
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()

    a_rev = a[::-1]
    # convolution of w with reversed a (length = 3·n‑2)
    conv_len = 3 * a.size - 2
    fft_len_grad = _fft_len_for_len(conv_len)
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)

    # extract the part that corresponds to Σ_i w_i a_{i‑j}
    start = a.size - 1
    grad_part = conv_w_rev[start : start + a.size] / sum_w
    return grad_part + a  # the “+ a” comes from the derivative of the linear term


# ----------------------------------------------------------------------
# initial‑vector factories
# ----------------------------------------------------------------------


def _exponential_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Decreasing exponential shape a_i ∝ r^i  (0 < r < 1)  normalised to sum 1.
    The rate r is sampled uniformly in a range that yields a noticeable
    asymmetry but keeps the vector far from the uniform extreme.
    """
    r = rng.uniform(0.80, 0.97)          # moderate decay
    geom = r ** np.arange(n, dtype=np.float64)
    # avoid the pathological case r≈1 where the vector would be almost uniform
    if geom.sum() == 0.0:
        geom = np.full(n, 1.0 / n)
    else:
        geom = geom / geom.sum()
    return geom


def _beta_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Beta‑distribution shape on the discrete grid.
    Parameters α,β are drawn from (0.3, 5) – wide enough to explore left‑skewed,
    right‑skewed and near‑uniform densities.
    """
    a = rng.uniform(0.3, 5.0)
    b = rng.uniform(0.3, 5.0)
    xs = (np.arange(n) + 0.5) / n
    pdf = np.power(xs, a - 1.0) * np.power(1.0 - xs, b - 1.0)
    pdf = np.nan_to_num(pdf, nan=0.0, posinf=0.0, neginf=0.0)
    if pdf.sum() == 0.0:
        pdf = np.full(n, 1.0 / n)
    else:
        pdf = pdf / pdf.sum()
    return pdf


def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Uniform draw from the probability simplex (Dirichlet(1,…,1)).
    """
    x = rng.exponential(1.0, size=n)
    x /= x.sum()
    return x


def _initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Pick an initial vector at random from a small family of sensible families.
    """
    choice = rng.integers(0, 4)
    if choice == 0:
        return _exponential_init(n, rng)
    if choice == 1:
        return _beta_init(n, rng)
    if choice == 2:
        return _random_simplex(n, rng)
    # fallback – uniform
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Core optimisation loop (inner optimisation)
# ----------------------------------------------------------------------


def _optimise_inner(a: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a bounded number of Adam‑gradient + annealing steps starting
    from ``a`` (already on the simplex).  Returns the best vector found
    during this inner run and its score.
    """
    n = a.size
    fft_len = _fft_len_for_len(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    t = 0

    # schedule parameters
    beta_start, beta_end = 5.0, 500.0         # soft‑max temperature
    eta_start, eta_end = 0.8, 0.01           # learning‑rate envelope
    T_start, T_end = 0.12, 1e-5             # simulated‑annealing temperature

    # bookkeeping
    cur_score = _score_simplex(a, fft_len)
    best = a.copy()
    best_score = cur_score
    start = time.monotonic()
    last_improve = start

    while time.monotonic() < deadline:
        # -----------------------------------------------------------------
        # 1) compute smooth gradient
        # -----------------------------------------------------------------
        elapsed = time.monotonic() - start
        frac = elapsed / (deadline - start)
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start + (eta_end - eta_start) * frac
        T = T_start + (T_end - T_start) * frac

        grad = _softmax_gradient(a, beta, fft_len)

        # -----------------------------------------------------------------
        # 2) Adam update (additive) and projection onto simplex
        # -----------------------------------------------------------------
        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_proposed = a - step
        a_proposed = _project_onto_simplex(a_proposed)

        # -----------------------------------------------------------------
        # 3) stochastic‑annealing acceptance
        # -----------------------------------------------------------------
        new_score = _score_simplex(a_proposed, fft_len)
        delta = new_score - cur_score
        accept = (delta < 0) or (rng.random() < np.exp(-delta / max(T, 1e-12)))
        if accept:
            a = a_proposed
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{elapsed:6.2f}s] inner best = {best_score:.6f}")
                last_improve = time.monotonic()

        # -----------------------------------------------------------------
        # 4) cheap exploration moves
        # -----------------------------------------------------------------
        # (a) block log‑normal perturbation
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)
            cur_score = _score_simplex(a, fft_len)

        # (b) pairwise mass transfer
        if rng.random() < 0.07:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                delta_max = min(a[i], 0.3 * a[i])
                delta_move = rng.random() * delta_max
                cand = a.copy()
                cand[i] -= delta_move
                cand[j] += delta_move
                cand = _project_onto_simplex(cand)
                cand_score = _score_simplex(cand, fft_len)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best = a.copy()
                        print(f"[{elapsed:6.2f}s] pair‑move improvement = {best_score:.6f}")

        # -----------------------------------------------------------------
        # 5) restart if stagnated for a while (timeout ≈ 30 s)
        # -----------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > 30.0:
            # jittered copy of the current best
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * jitter
            a = _project_onto_simplex(a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            cur_score = _score_simplex(a, fft_len)
            last_improve = now
            print(f"[{now - start:6.2f}s] inner restart – score {cur_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.
    The routine mixes gradient descent on a soft‑max approximation of the
    max‑convolution with simulated‑annealing style acceptance and a handful of
    cheap random moves.  Multiple restarts are performed until the time
    budget expires; the best sequence ever observed is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Decide on the problem dimension
    # ------------------------------------------------------------------
    default_n = 2500                     # a good compromise between quality and speed
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a0 = np.clip(cand, 0.0, None)
                if a0.sum() > 0.0:
                    a0 = a0 / a0.sum()
                else:
                    a0 = np.full(cand.size, 1.0 / cand.size)
                n = a0.size
                a = a0
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Global bookkeeping
    # ------------------------------------------------------------------
    overall_best = a.copy()
    overall_best_score = _score_simplex(overall_best,
                                        _fft_len_for_len(2 * n - 1))
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # tiny safety margin

    print(f"starting optimisation, n = {n}, initial score = {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Main outer loop – many independent restarts
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # a fresh start (either the current best with jitter or a brand‑new shape)
        if rng.random() < 0.4:
            a0 = _initial_vector(n, rng)
        else:
            jitter = np.exp(rng.normal(0.0, 0.22, size=n))
            a0 = overall_best * jitter
            a0 = _project_onto_simplex(a0)

        # run a bounded inner optimisation (≈ 20–30 s or until deadline)
        inner_deadline = min(deadline,
                             time.monotonic() + 25.0)    # 25 s per inner run
        a_opt, opt_score = _optimise_inner(a0, rng, inner_deadline)

        if opt_score < overall_best_score:
            overall_best_score = opt_score
            overall_best = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST  = {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final clipping to respect the limits imposed by evaluate_sequence
    # ------------------------------------------------------------------
    overall_best = np.clip(overall_best, 0.0, 1000.0)

    # Return a plain Python list of floats
    return [float(x) for x in overall_best]
```
