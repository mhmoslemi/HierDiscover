```
def beta_bin_mass(n, a, b):
    i = np.arange(n)
    x0 = i / n
    x1 = (i+1) / n
    # Beta CDF: I_x(a,b)
    from scipy.special import betainc
    cdf1 = betainc(a, b, x1)
    cdf0 = betainc(a, b, x0)
    mass = cdf1 - cdf0
    return mass
```

-----------------

Search for a non‑negative step‑function (list of heights) that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear self‑convolution.
The method combines three ideas:

   discretisations (including the exact arcsine law) is scanned and the

   (large η, small β) and becomes very fine‑grained near the end
   (η≈0.005, β≈3000).  Symmetry (a = (a+a[::-1])/2) is enforced after every

import scipy.sparse as sp

# Helper functions (top‑level, no closures)

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _beta_bin_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mass of a symmetric Beta(α,β) distribution on the interval
    [i/n,(i+1)/n).  The result sums to 1."""
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    cdf1 = sps.betainc(alpha, beta, x1)
    cdf0 = sps.betainc(alpha, beta, x0)
    mass = cdf1 - cdf0
    # numerical issues can produce tiny negatives; clip them
    mass = np.clip(mass, 0.0, None)
    total = mass.sum()
    if total == 0.0:
        mass[:] = 1.0 / n
    else:
        mass /= total
    return mass


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force exact symmetry a[i]=a[n-1-i]."""
    return (a + a[::-1]) * 0.5


def _score_of(a: np.ndarray) -> float:
    """Exact score (no clipping) for a normalised vector (sum=1)."""
    n = a.size
    conv = _conv_self(a, _fft_len_for_n(n))
    return 2.0 * n * conv.max() / (np.sum(a) ** 2)


def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """Sparse Toeplitz matrix that computes linear convolution f * b.
    Shape = (2*len(f)-1, len(f))."""
    n = f.size
    # each diagonal d (0≤d≤n‑1) contains the constant value f[d] repeated n times
    diagonals = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)                     # positive offsets
    return sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format="csc")
