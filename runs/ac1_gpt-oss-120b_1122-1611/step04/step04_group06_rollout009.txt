def random_move(seq):
    n = len(seq)
    i, j = rng.integers(0, n, size=2)
    if i == j: return seq
    # Move a small fraction from i to j
    delta = seq[i] * rng.random() * 0.2   # up to 20% of mass.
    new = seq.copy()
    new[i] -= delta
    new[j] += delta
    return new
```

def propose_candidate(seed=42, budget_s=1000.0):
    """
    Generate step-function heights minimizing evaluate_sequence.
    Uses an analytical arcsine‑distribution shape (self‑convolution close to flat)
    as a strong baseline and performs cheap local refinement.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_seq = None
    best_score = float('inf')

            best_score = score
            best_seq = seq
            print(f"[{time.monotonic() - start:6.2f}s] new best {score:.6f} len={len(seq)}")

    if "height_sequence_1" in globals():
        try:

        except Exception:
            pass

"""
propose_candidate
=================

Generate a non‑negative step‑function whose self‑convolution is as flat as
possible.  The main idea is to start from the *discrete arcsine*
distribution  

    a_i ∝ 1 / √((i+½)·(n‑½‑i))

which (in the continuous limit) has a perfectly flat self‑convolution.
For several lengths `n` and a few exponents we evaluate the score
directly with the supplied `evaluate_sequence`.  The best of those
candidates is then refined by a cheap exponentiated‑gradient descent that
acts only on the index where the current convolution attains its maximum.
All work is stopped after the allotted time budget (default 1000 s).  The
returned list is guaranteed to contain only finite numbers in the range
[0, 1000] as required by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must live at top level – no closures)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # a small safety margin – we stop a little before the deadline
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0)  Keep the existing solution (if any) as a baseline
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = list(map(float, globals()["height_sequence_1"]))
            sc = evaluate_sequence(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] baseline score = {sc:.6f} "
                      f"(length {len(cand)})")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1)  Deterministic arcsine‑family candidates
    # --------------------------------------------------------------
    # a few lengths and a few powers around the true arcsine (α = 0.5)
    lengths = [128, 256, 512, 1024, 2048, 4096]
    alphas = [0.45, 0.5, 0.55]          # exponent in the denominator

    for n in lengths:
        if time.monotonic() > deadline:
            break
        i = np.arange(n, dtype=np.float64)
        for alpha in alphas:
            if time.monotonic() > deadline:
                break
            # raw (unnormalised) shape
            raw = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) ** alpha)
            # clip to the limits imposed by ``evaluate_sequence``
            raw = np.clip(raw, 0.0, 1000.0)
            # normalise – the score is scale‑invariant, but a non‑zero sum
            # avoids the “sum too close to zero” guard in ``evaluate_sequence``.
            raw /= raw.sum()
            seq = raw.tolist()
            sc = evaluate_sequence(seq)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best score = {sc:.6f} "
                      f"(n={n}, α={alpha})")

    # --------------------------------------------------------------
    # 2)  Light local refinement on the current best (if we have one)
    # --------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, dtype=np.float64)
        n = a.size
        fft_len = _fft_len_for_n(n)

        # start with a modest step size – exponentiated‑gradient on the
        # active (max‑conv) index.
        eta = 0.3
        # refinement loop runs until time runs out
        while time.monotonic() < deadline:
            # compute the full self‑convolution and the index of the maximum
            conv = _conv_self(a, fft_len)
            k_max = int(np.argmax(conv))

            # gradient of the (non‑smooth) max‑convolution:
            #   ∂max/∂a_i = 2·a_{k_max−i}   (if the index is in range)
            grad = np.zeros_like(a)
            idx = np.arange(n, dtype=np.int64)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)
            grad[mask] = 2.0 * a[partner[mask]]

            # exponentiated‑gradient update (mirror descent on the simplex)
            a = a * np.exp(-eta * grad)

            # occasional tiny random jitter to escape flat plateaus
            if rng.random() < 0.01:
                jitter = np.exp(rng.normal(0.0, 0.08, size=n))
                a *= jitter

            # renormalise to the simplex (scale does not matter for the score)
            a = np.clip(a, 0.0, None)
            total = a.sum()
            if total == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= total

            # evaluate every few iterations (≈ every 0.2 s on average)
            if rng.random() < 0.05:
                seq = a.tolist()
                sc = evaluate_sequence(seq)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] refined best = {sc:.6f}")

            # slowly decay the step size
            eta *= 0.9999

            # safety: stop if we have exhausted the whole budget
            if time.monotonic() - start_time > budget_s - 0.5:
                break

    # --------------------------------------------------------------
    # 3)  Fallback – if nothing succeeded, return a tiny uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)

    # Clip to the limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] final best score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq
