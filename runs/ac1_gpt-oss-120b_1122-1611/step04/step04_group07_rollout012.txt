"""
Propose a candidate step‑function by alternating a linear‐programming
expansion and a simple convex combination.  The idea follows the
theoretical scheme from the paper: given a current non‑negative vector
f (∑f = √(2n)) we solve

    max   1ᵀb
    s.t.  f * b ≤ M   (all 2n‑1 convolution entries),
          b ≥ 0,

where M = max(f*f).  The LP pushes mass into coordinates where f is
small, thereby reducing the peak of the self‑convolution when we mix
f and the LP solution g.  Repeating the update shrinks M and thus the
objective 2·n·max(conv)/∑a² = max(conv) (because ∑a = √(2n)).
The routine tries a few problem sizes (the length of the step
function) within the allotted time and returns the best sequence it
found.
"""

import time
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (length needed for a linear convolution)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using FFT (fast, O(n log n))."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    best_score = float("inf")
    best_seq = None
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # Helper to evaluate the exact score (max of the self‑convolution)
    # ------------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        # the evaluation function returns max(conv) when the sum is √(2n)
        return float(_conv_self(vec, _fft_len_for_n(vec.size)).max())

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore.  If a previous construction
    # exists we start from its length first.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]          # reasonable sizes for the budget
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
                lengths.insert(0, int(seq0.size))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Main loop over chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() > deadline:
            break

        sqrt_sum = np.sqrt(2.0 * n)            # target ℓ₁‑norm
        fft_len = _fft_len_for_n(n)

        # --------------------------------------------------------------
        # Initialise f : use previous construction if length matches,
        # otherwise a random Dirichlet draw (ensures positivity).
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            f = best_seq.copy()
        elif "height_sequence_1" in globals():
            try:
                cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
                cand = np.clip(cand, 0.0, None)
                if cand.size == n:
                    f = cand / cand.sum() * sqrt_sum
                else:
                    raise ValueError
            except Exception:
                f = np.full(n, sqrt_sum / n, dtype=np.float64)
        else:
            # random Dirichlet – multiplicative log‑normal perturbation of uniform
            perturb = np.exp(rng.normal(0.0, 0.2, size=n))
            f = np.full(n, 1.0 / n, dtype=np.float64) * perturb
            f = f / f.sum() * sqrt_sum

        # --------------------------------------------------------------
        # Pre‑compute the row / column index pattern for the convolution
        # matrix A_ub where (A_ub @ b) = f * b.  This pattern depends only on n.
        # --------------------------------------------------------------
        idx = np.arange(n, dtype=np.int32)
        rows = (idx[:, None] + idx[None, :]).ravel()          # shape (n*n,)
        cols = np.repeat(idx, n)
        # rows ranges from 0 … 2n‑2, exactly the indices of the linear conv

        # --------------------------------------------------------------
        # Iterative improvement: LP → convex blend → repeat.
        # --------------------------------------------------------------
        max_iter = 30                      # enough for a noticeable drop
        t_mix = 0.5                        # convex‑combination weight
        for it in range(max_iter):
            now = time.monotonic()
            if now > deadline:
                break

            # current peak of self‑convolution
            M_cur = _score(f)

            # record best overall solution
            if M_cur < best_score:
                best_score = M_cur
                best_seq = f.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] length={n}, new best M={M_cur:.6f}")

            # ----------------------------------------------------------
            # Build the (2n‑1) × n constraint matrix A_ub for f * b ≤ M_cur.
            # ----------------------------------------------------------
            data = np.repeat(f, n)                     # each column i repeats f[i] n times
            A_ub = _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

            # ----------------------------------------------------------
            # Linear programme: maximise 1ᵀb  (i.e. minimise -1ᵀb)
            # ----------------------------------------------------------
            c = -np.ones(n, dtype=np.float64)           # –sum(b)
            ub_vec = np.full(2 * n - 1, M_cur, dtype=np.float64)
            res = _opt.linprog(
                c, A_ub=A_ub, b_ub=ub_vec, bounds=(0, None), method="highs"
            )
            if res.status != 0:
                # solver failed – abort further LP attempts for this n
                print("LP failed (status {}).  Stopping iterations for n={}."
                      .format(res.status, n))
                break

            b_opt = res.x
            sum_b = b_opt.sum()
            if sum_b <= 0.0:
                break

            # scale to the required ℓ₁‑norm
            scale = sqrt_sum / sum_b
            g = b_opt * scale

            # ----------------------------------------------------------
            # Convex blend and accept if it reduces the peak.
            # ----------------------------------------------------------
            f_candidate = (1.0 - t_mix) * f + t_mix * g
            f_candidate = np.clip(f_candidate, 0.0, None)
            f_candidate = f_candidate / f_candidate.sum() * sqrt_sum

            M_new = _score(f_candidate)

            if M_new < M_cur - 1e-12:          # improvement (tiny tolerance)
                f = f_candidate
                # keep the same mixing weight for the next iteration
            else:
                # no improvement – shrink the step size and retry a few times
                t_mix *= 0.5
                if t_mix < 1e-4:
                    # step became too small, abort the LP loop for this length
                    break

        # --------------------------------------------------------------
        # After the LP‑driven phase we run a few cheap soft‑max mirror‑descent
        # iterations (the method used in the original script) to polish the
        # current vector.  This step is optional but often yields a small
        # extra drop.
        # --------------------------------------------------------------
        # Re‑use the helper utilities from the original code (they do not
        # depend on any closure).  Only a few iterations are performed
        # because they are cheap and the remaining time is small.
        a = f / f.sum()                     # normalise to unit sum for the mirror step
        n_a = a.size
        fft_len_a = _fft_len_for_n(n_a)

        eta_start, eta_end = 0.8, 0.02
        beta_start, beta_end = 0.5, 200.0
        momentum = 0.9
        grad_prev = np.zeros_like(a)
        block_perturb_prob = 0.05
        last_improve = time.monotonic()
        restart_gap = 20.0
        next_print = time.monotonic() + 15.0

        while time.monotonic() < deadline:
            now = time.monotonic()
            elapsed = now - start_time
            total = deadline - start_time
            if now >= next_print:
                cur_est = 2.0 * n_a * float(_conv_self(a, fft_len_a).max())
                print(f"[{elapsed:6.2f}s] polish est. score = {cur_est:.6f}")
                next_print = now + 15.0

            # linear schedules
            eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
            beta = beta_start + (beta_end - beta_start) * (elapsed / total)

            conv = _conv_self(a, fft_len_a)
            max_conv = float(conv.max())
            # soft‑max over all entries
            shift = max_conv
            w_raw = np.exp(beta * (conv - shift))
            w = w_raw / w_raw.sum()
            grad = 2.0 * np.correlate(w, a, mode='valid')
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            if a.sum() == 0.0:
                a = np.full_like(a, 1.0 / n_a)
            else:
                a /= a.sum()

            # occasional block perturbation
            if rng.random() < block_perturb_prob:
                # multiplicative log‑normal block perturbation
                width = rng.integers(1, max(2, n_a // 10))
                start = rng.integers(0, n_a - width + 1)
                factor = np.exp(rng.normal(0.0, 0.15))
                a[start:start + width] *= factor
                a = np.clip(a, 0.0, None)
                a /= a.sum()

            # soft restart if stuck
            if now - last_improve > restart_gap:
                a = best_seq / best_seq.sum() if best_seq is not None else a
                noise = np.exp(rng.normal(0.0, 0.12, size=n_a))
                a = a * noise
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                grad_prev = np.zeros_like(a)
                last_improve = now

            # update best record (exact evaluation)
            cur_score = 2.0 * n_a * float(_conv_self(a, fft_len_a).max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a * sqrt_sum   # rescale to the √(2n) norm used above
                last_improve = now
                print(f"[{elapsed:6.2f}s] polish new best M={cur_score:.6f}")

        # end of polishing for this length

    # ------------------------------------------------------------------
    # Return the best sequence found (clipped to the limits of the
    # official evaluator).  The result already satisfies Σa = √(2n),
    # which makes the evaluation equal to the maximal convolution entry.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector of a modest size
        n = 256
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq.tolist()]
