def propose_candidate(seed=42, budget_s=1000.0):
    import time, numpy as np, scipy.optimize as opt

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)

    # Helper functions defined outside.

    best_score = np.inf
    best_seq = None

    # define some lengths
    lengths = [256, 512, 1024, 1500, 2000]  # small-ish to keep LP fast

    for n in lengths:
        if time.monotonic() >= deadline: break

        # initial candidate: random Dirichlet (alpha=0.5)
        a = rng.dirichlet(np.full(n, 0.5))
        # optionally symmetrize
        a = (a + a[::-1])/2
        a = a / a.sum()

        # optionally run a few gradient descent steps to improve
        # Use similar loop as previous code but with fewer iterations.

        # compute initial score
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.monotonic() - start:.2f}s] init n={n} score={cur_score:.6f}")

        # start LP refinement loop
        lp_iters = 5
        for it in range(lp_iters):
            if time.monotonic() >= deadline: break

            # Compute current max conv
            conv = _conv_self(a)
            M = float(conv.max())
            # Build A matrix
            ccol = np.concatenate([a, np.zeros(n-1, dtype=np.float64)])
            crow = np.concatenate([a[:1], np.zeros(n-1, dtype=np.float64)])
            # Using toeplitz
            from scipy.linalg import toeplitz
            A = toeplitz(ccol, crow)  # shape (2n-1, n)

            # b vector
            b = np.full(2*n-1, M, dtype=np.float64)

            # solve LP
            res = opt.linprog(-np.ones(n), A_ub=A, b_ub=b,
                              bounds=[(0, None)]*n,
                              method='highs',
                              options={'presolve': True, 'time_limit': max(0.1, deadline - time.monotonic())})
            if not res.success:
                # maybe due to infeasibility; break
                break
            g = res.x
            sum_g = g.sum()
            if sum_g <= 1 + 1e-12:
                # cannot improve further
                break

            # normalize
            a_new = g / sum_g
            # symmetrize
            a_new = (a_new + a_new[::-1]) / 2
            a_new = a_new / a_new.sum()

            # evaluate
            conv_new = _conv_self(a_new)
            new_score = 2.0 * n * conv_new.max()
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.monotonic() - start:.2f}s] LP improve n={n} iter={it} score={new_score:.6f}")

    if best_seq is None:

"""
Propose a step-function (non‑negative vector) with a low value of

    S(a) = 2·n·max(conv(a,a)) / (∑a)²

The algorithm works in two phases:

1.  Random Dirichlet initialisations for a few problem sizes.  For each
    length `n` a short LP‑refinement (see below) is performed.  The LP
    maximises the ℓ₁‑norm of a new vector `g` under the constraint
    `conv(f,g) ≤ M` where `M` is the current maximal self‑convolution of
    the candidate `f`.  After normalising `g` the new candidate often
    enjoys a smaller maximum convolution.

2.  The best vector found in phase‑1 is optionally polished with a few
    mirror‑descent/soft‑max iterations (the same update that the reference
    code uses).  This helps to escape any remaining local minima.

All operations stay on the probability simplex (`∑a = 1`), which makes the
objective scale‑invariant.  The routine respects the 1000 s time budget,
returns a list of floats and never produces negative or infinite values.
"""

import time
import numpy as np
from scipy import optimize
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Small random perturbation – used in the polishing phase
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Mirror‑descent (soft‑max) polishing – optional final phase
# ----------------------------------------------------------------------


def _polish(candidate: np.ndarray,
            time_budget: float,
            rng: np.random.Generator) -> np.ndarray:
    """
    A few iterations of exponentiated‑gradient (entropy mirror descent)
    with a soft‑max approximation of the max‑convolution.
    The routine works on the simplex and returns a possibly improved vector.
    """
    n = candidate.size
    a = candidate.copy()
    start = time.monotonic()
    deadline = start + time_budget

    # hyper‑parameters (tuned for short polishing runs)
    eta_start = 0.5
    eta_end = 0.001
    beta_start = 0.5
    beta_end = 400.0
    momentum = 0.9
    block_perturb_prob = 0.03
    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:
        conv = _conv_self(a)
        cur_max = conv.max()
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # soft‑max weights for the current temperature
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        a_rev = a[::-1]
        conv_rev = np.convolve(w, a_rev)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update (mirror descent on simplex)
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -50.0, 50.0)
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional jitter
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # enforce symmetry (optimal solutions are symmetric)
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_score = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # 1) Phase 1 – a few lengths, random starts, LP refinement
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 1500, 2000]  # moderate sizes – LP is cheap enough
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # a random Dirichlet start (alpha < 1 gives a non‑uniform shape)
        a = rng.dirichlet(np.full(n, 0.5))
        # enforce symmetry – optimal solutions are symmetric
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

        # evaluate initial score
        conv = _conv_self(a)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] init n={n} score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Repeated LP refinement – stop after a few iterations or when no
        # improvement is possible.
        # --------------------------------------------------------------
        max_lp_iters = 12
        for lp_it in range(max_lp_iters):
            if time.monotonic() >= deadline:
                break

            # recompute current bound M
            conv = _conv_self(a)
            M = float(conv.max())

            # Build the full convolution matrix A where (A @ g)_k = (a * g)_k
            # A has shape (2n‑1, n).  For speed we use scipy.linalg.toeplitz.
            ccol = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
            crow = np.concatenate([a[:1], np.zeros(n - 1, dtype=np.float64)])
            A = toeplitz(ccol, crow)  # dense matrix, still modest for chosen n

            b = np.full(2 * n - 1, M, dtype=np.float64)

            # Linear program: maximise sum(g)  ⇔ minimise –sum(g)
            lp_res = optimize.linprog(
                c=-np.ones(n, dtype=np.float64),
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True},
            )

            if not lp_res.success:
                # numerical difficulty – abort this length
                print(f"[{time.monotonic() - start_time:.2f}s] LP failed (n={n}, it={lp_it})")
                break

            g = lp_res.x
            sum_g = g.sum()
            # if the LP cannot increase the ℓ₁‑norm we have reached a fixed point
            if sum_g <= 1.0 + 1e-12:
                break

            # normalise, enforce symmetry and the simplex
            a_new = g / sum_g
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = a_new / a_new.sum()

            # evaluate new candidate
            conv_new = _conv_self(a_new)
            new_score = 2.0 * n * float(conv_new.max())

            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(
                    f"[{time.monotonic() - start_time:.2f}s] LP improve n={n} it={lp_it} score={new_score:.6f}"
                )

            # stop early if improvement is negligible
            if new_score >= cur_score * (1.0 - 1e-5):
                a = a_new
                break

            a = a_new
            cur_score = new_score

    # ------------------------------------------------------------------
    # 2) Phase 2 – optional polishing with mirror‑descent (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0 and best_seq is not None:
        # allocate at most half of the remaining budget to polishing
        polish_budget = min(remaining * 0.5, 30.0)
        polished = _polish(best_seq, polish_budget, rng)
        # evaluate polished version
        n = polished.size
        conv = _conv_self(polished)
        polished_score = 2.0 * n * float(conv.max())
        if polished_score < best_score:
            best_score = polished_score
            best_seq = polished
            print(
                f"[{time.monotonic() - start_time:.2f}s] polishing improved score={polished_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 3) Fallback – if nothing succeeded, return a modest uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 512
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by ``evaluate_sequence`` (just for safety)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store the result in a global variable so that subsequent runs can reuse it
    # (the harness may read ``height_sequence_1``).  This is optional but handy.
    globals()["height_sequence_1"] = [float(x) for x in best_seq]

    return [float(x) for x in best_seq]
