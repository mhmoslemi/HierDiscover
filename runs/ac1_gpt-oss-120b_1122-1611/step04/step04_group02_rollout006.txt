"""
Propose a high‑quality step‑function (non‑negative vector whose entries sum to 1)
that minimises ``evaluate_sequence``.

Algorithm
---------
* **Phase 1 – smooth exponential‑gradient descent**  
  The objective  max(conv) is approximated by a soft‑max:
      Lβ(a) = log Σk exp(β·(convk – max(conv))) .
  Its gradient w.r.t. a can be written in closed form as a correlation of
  the current vector with the soft‑max weights.  Using the FFT we obtain the
  full gradient (no “active set” truncation) in O(n log n) time.
  An exponentiated‑gradient (mirror‑descent on the simplex) update
  `a_i ← a_i·exp(‑η·∇i)` is performed, followed by a normalisation.
  The temperature β grows from 1 to 500 and the stepsize η shrinks from 0.6
  to 0.02, concentrating the optimisation on the true peak.

* **Phase 2 – fine‑grained stochastic refinement**  
  After the bulk of the budget we keep η at its minimum and β at its maximum.
  Occasionally we apply random block‑perturbations (log‑normal jitter on a
  contiguous sub‑array) and pairwise mass‑transfer moves accepted by a tiny
  simulated‑annealing schedule.  Restarts are triggered if no improvement is
  observed for 30 s.

All computations are performed with real FFTs; the convolution
``b = a * a`` and the correlation needed for the gradient share the same
FFT plan, so each iteration costs only a few FFTs.  The routine respects the
time budget (a 5 s safety margin) and returns the best vector seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT of length ``fft_len``."""
    Fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Fa * Fa, n=fft_len)
    # keep only the first 2·len(a)‑1 entries (linear convolution)
    return conv_full[: 2 * a.size - 1]


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a normalised weight vector w_i ∝ exp(beta·(conv_i‑max(conv)))."""
    max_c = conv.max()
    # exponent is ≤ 0, so overflow is not a problem
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        # numerical underflow – fall back to uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w / w_sum
    return w


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the gradient of  Σ_k w_k·(a*a)_k  w.r.t. a.
    The formula is  ∇ = 2·corr(a, w)  (full linear correlation).
    Only the n entries corresponding to the original a are returned.
    """
    # full linear correlation a ⋆ w  of length n+len(w)-1 = 3n‑2
    Fa = np.fft.rfft(a, n=fft_len)
    Fw = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(Fa * np.conj(Fw), n=fft_len)
    corr_full = corr_full[: a.size + w.size - 1]            # length = 3n‑2
    # the zero‑lag position is at index len(w)-1 = 2n‑2
    start = w.size - 1
    grad = 2.0 * corr_full[start : start + a.size]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    # block width between 1 and max(2, n//20)
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.18))                # slightly stronger jitter
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).  Used only for
    safety when additive updates are applied.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    current best bound.  The routine respects the time budget and keeps
    the best vector seen so far.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose problem size and starting point
    # ------------------------------------------------------------------
    # Reasonable length – large enough to give degrees of freedom,
    # but not so large that each iteration becomes costly.
    # We sample a length in a modest band around the previously known
    # construction (3800–4200).  If a previous solution exists we reuse its
    # length, otherwise we draw a fresh length.
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
            # reuse length, but also allow occasional length changes
            base_n = prev.size
        else:
            base_n = rng.integers(3000, 4500)
    else:
        base_n = rng.integers(3000, 4500)

    # Add a small random perturbation to the length to escape possible
    # length‑dependent local minima.
    n = int(base_n + rng.integers(-100, 100))
    if n < 2000:
        n = 2000
    if n > 6000:
        n = 6000

    # Initialise a probability vector.
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            # Resample the old vector onto the new size (linear interpolation)
            x_old = np.linspace(0, 1, cand.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, cand)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n)
        else:
            a = rng.dirichlet(np.ones(n)).astype(np.float64)
    else:
        a = rng.dirichlet(np.ones(n)).astype(np.float64)

    # ------------------------------------------------------------------
    # 2) FFT set‑up (single length large enough for both convolution
    #    (2·n‑1) and correlation (3·n‑2)).
    # ------------------------------------------------------------------
    fft_len = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max                # sum(a)=1 ⇒ score = 2·n·max(conv)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # Phase split: 80 % of time for smooth mirror descent, remainder for
    # fine‑grained stochastic refinement.
    phase1_end = start_time + 0.80 * total_time

    # Hyper‑parameters (smooth schedules)
    eta_start = 0.60
    eta_end = 0.02
    beta_start = 1.0
    beta_end = 500.0
    restart_gap = 30.0       # seconds without improvement → restart
    block_perturb_prob = 0.015   # probability per iteration

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase‑1 : smooth exponential‑gradient descent
        # --------------------------------------------------------------
        if now < phase1_end:
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac          # linear decay
            beta = beta_start + (beta_end - beta_start) * frac      # linear increase
        else:
            # Fine‑grained phase – keep parameters near their terminal values
            eta = eta_end
            beta = beta_end

        # ----- compute self‑convolution and soft‑max weights -------------
        Fa = np.fft.rfft(a, n=fft_len)                    # reuse for both ops
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
        cur_max = conv_full.max()
        # Soft‑max over the whole convolution array
        w = _softmax_weights(conv_full, beta)

        # ----- gradient (full correlation) ------------------------------
        # Correlation a ⋆ w : use conj(Fw) in the frequency domain
        Fw = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(Fa * np.conj(Fw), n=fft_len)
        # keep only the needed prefix (length = n + len(w) - 1 = 3n‑2)
        corr_full = corr_full[: n + w.size - 1]
        start_idx = w.size - 1                # = 2·n‑2, zero‑lag position
        grad = 2.0 * corr_full[start_idx : start_idx + n]

        # ----- exponentiated‑gradient update on the simplex -------------
        factor = np.exp(-eta * grad)          # element‑wise
        a = a * factor
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ----- occasional random block jitter ---------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- recompute score for the new point -----------------------
        conv = _conv_self_fft(a, fft_len)
        cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Phase‑2 : occasional pairwise mass transfers (simulated annealing)
        # --------------------------------------------------------------
        if now >= phase1_end:
            # linear temperature schedule descending from 0.03 to 1e‑6
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = max(1e-6, 0.03 * (1.0 - t_frac))

            # pairwise transfer with probability 0.85
            if rng.random() < 0.85:
                i = rng.integers(0, n)
                if a[i] < 1e-14:
                    # donor empty – skip
                    pass
                else:
                    j = rng.integers(0, n - 1)
                    if j >= i:
                        j += 1
                    # move up to 25 % of the donor's mass
                    delta = rng.random() * 0.25 * a[i]
                    a_new = a.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    # evaluate new candidate
                    conv_new = _conv_self_fft(a_new, fft_len)
                    new_score = 2.0 * n * conv_new.max()
                    # Metropolis acceptance
                    if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                        a = a_new
                        conv = conv_new
                        cur_score = new_score
            else:
                # block perturbation (global jitter)
                a = _random_block_perturb(a, rng)
                conv = _conv_self_fft(a, fft_len)
                cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart when stuck for a while
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # multiplicative jitter around the best
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            a /= a_sum if a_sum > 0 else 1.0
            # recompute convolution for the restarted point
            conv = _conv_self_fft(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped for safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
