import time
import numpy as np

def _next_pow2(length: int) -> int:
    """Return the smallest power‑of‑two >= length (for FFT padding)."""
    if length <= 0:
        return 1
    return 1 << ((length - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT."""
    n = a.size
    needed = 2 * n - 1
    if fft_len is None:
        fft_len = _next_pow2(needed)
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[:needed]
    return conv

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    # width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10) + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _next_pow2(length: int) -> int:

    if length <= 0:

def _conv_self(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """Linear self‑convolution a∗a using FFT."""
