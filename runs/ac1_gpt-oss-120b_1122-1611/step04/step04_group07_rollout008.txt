"""
Propose a step‑function (non‑negative vector) that minimizes

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ideas:

1.  Mirror‑descent (exponentiated‑gradient) on the probability simplex.
    The objective is approximated by a soft‑max of the convolution entries,
    whose temperature β is increased exponentially so the optimisation
    focuses quickly on the true max.

2.  Efficient FFT‑based convolution and gradient computation.
    The gradient of the soft‑max objective is 2 · corr(w,a) where
    w are the soft‑max weights; both the self‑convolution and this
    correlation are evaluated with a single FFT each iteration.

3.  A simple stochastic “global‑search’’ wrapper:
    - occasional log‑normal block perturbations,
    - occasional smoothing by a small 3‑point kernel,
    - a restart strategy that jitter‑perturbs the current best when
      no improvement is seen for a while.
    The best vector seen (exact score computed from the fast formula) is
    returned at the end.

The code runs for up to the supplied time budget (default 1000 s) and
automatically adapts its step‑size and temperature schedules.
"""

import time
import numpy as np

# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_len(L: int) -> int:
    """Smallest power‑of‑two ≥ L (length needed for an FFT)."""
    return 1 << ((L - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_corr_w_a(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute 2 · corr(w,a) (valid part) using FFT.
    Equivalent to 2 * np.correlate(w, a, mode='valid'), but O(n log n).
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(fw * fa_rev, n=fft_len)
    n = a.size
    start = n - 1
    # slice length n gives the “valid’’ correlation
    grad = 2.0 * conv[start:start + n]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _smooth_step(a: np.ndarray) -> np.ndarray:
    """Apply a tiny 3‑point moving‑average and renormalise."""
    kernel = np.array([0.2, 0.6, 0.2], dtype=np.float64)
    sm = np.convolve(a, kernel, mode='same')
    sm = np.clip(sm, 0.0, None)
    sm /= sm.sum()
    return sm

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # linear schedules for η, exponential schedule for β
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        # exponential growth of β gives a sharper focus early on
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total)

        # ---- self‑convolution and fast score estimate ------------------
        conv = _conv_self(a, fft_len_self)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        # ---- update the best record (exact score) ----------------------
        if cur_score < best_score:
            best_score = float(cur_score)
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- periodic status -------------------------------------------
        if now >= next_print:
            print(f"[{elapsed:6.2f}s] current score = {cur_score:.6f}   best = {best_score:.6f}")
            next_print = now + print_interval

        # ---- soft‑max weighting of convolution entries -----------------
        shift = max_conv
        w_raw = np.exp(beta * (conv - shift))   # max weight = 1
        w_sum = w_raw.sum()
        if w_sum == 0.0:   # guard against numerical underflow
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ---- gradient via FFT‑based correlation ------------------------
        grad = _grad_corr_w_a(w, a, fft_len_grad)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- exponentiated‑gradient (mirror‑descent) update -----------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = a / s if s > 0.0 else np.full_like(a, 1.0 / n)

        # ---- occasional smoothing step ---------------------------------
        if rng.random() < smooth_prob:
            a = _smooth_step(a)

        # ---- occasional random block perturbation ----------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ---- soft restart when stuck -----------------------------------
        if now - last_improve > restart_gap:
            # jitter the current best and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – jittered around best ({best_score:.6f})")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clip to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
