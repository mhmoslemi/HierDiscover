"""
Propose a candidate sequence that (approximately) minimises
    S(a) = 2·n·max(conv(a,a)) / (∑a)² .

The algorithm combines three ingredients

* a fast FFT based evaluation of the objective,
* a simulated‑annealing / hill‑climbing loop that repeatedly
  mutates the probability vector (mass transfer, block scaling,
  random swaps) and accepts uphill moves with a temperature schedule,
* occasional “soft‑restart” from the best vector seen so far with a
  log‑normal jitter.

The search works directly on the probability simplex (∑a = 1) so the
denominator of the evaluation cancels out – the only quantity that
matters is 2·n·max(conv).  With the modest dimensionalities (≈4000)
the FFT based convolution is only a few milliseconds, allowing many
thousands of iterations within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT (real input)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _score(a: np.ndarray) -> float:
    """
    Compute the surrogate score 2·n·max(conv(a,a))   (∑a = 1).
    The vector `a` is assumed to be non‑negative and sum≈1.
    """
    n = a.size
    conv = _conv_self_fft(a)
    return 2.0 * n * float(conv.max())


def _normalise(a: np.ndarray) -> np.ndarray:
    """Clip to non‑negative, re‑normalise to sum 1 (avoid division by zero)."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s
    return a


def _transfer_move(a: np.ndarray, rng: np.random.Generator, eps: float) -> np.ndarray:
    """
    Move a small amount of probability mass from i → j.
    eps controls the maximal fraction of the source weight that can be moved.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a
    src = a[i]
    if src == 0.0:
        return a
    delta = eps * src * rng.random()
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    return _normalise(out)


def _block_scale_move(a: np.ndarray, rng: np.random.Generator,
                     max_width_frac: float, sigma: float) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor.
    `max_width_frac` is the maximum block width as a fraction of the length.
    `sigma` is the log‑normal standard deviation.
    """
    n = a.size
    max_width = max(1, int(max_width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    return _normalise(out)


def _swap_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two random coordinates."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    out = a.copy()
    out[i], out[j] = out[j], out[i]
    return out


def _jitter_restart(best: np.ndarray,
                    rng: np.random.Generator,
                    sigma: float) -> np.ndarray:
    """Create a new start by jittering the best solution."""
    out = best.copy()
    out *= np.exp(rng.normal(0.0, sigma, size=out.size))
    return _normalise(out)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose length and initial distribution
    # ------------------------------------------------------------------
    default_n = 4000                               # convenient power of two neighbourhood
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = _normalise(cand.copy())
                n = a.size
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = _score(best)
    cur_score = best_score
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    T_start = 0.02          # initial temperature for SA
    T_end = 1e-6           # final temperature
    eps_transfer = 0.2      # fraction of source mass that can be moved
    block_prob = 0.10       # probability of applying a block scaling move
    swap_prob = 0.07        # probability of swapping two entries
    restart_gap = 45.0      # seconds without improvement before a restart
    jitter_sigma = 0.12     # log‑normal jitter strength on restart

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear temperature schedule
        T = T_start * (1.0 - elapsed / total) + T_end * (elapsed / total)

        # ------------------------------------------------------------------
        # 4.1) Propose a new candidate
        # ------------------------------------------------------------------
        proposal = a.copy()
        # always try a cheap transfer move
        proposal = _transfer_move(proposal, rng, eps_transfer)

        # occasional extra perturbations
        if rng.random() < block_prob:
            proposal = _block_scale_move(proposal, rng,
                                         max_width_frac=0.15, sigma=0.18)
        if rng.random() < swap_prob:
            proposal = _swap_move(proposal, rng)

        # ------------------------------------------------------------------
        # 4.2) Evaluate the proposal (fast surrogate)
        # ------------------------------------------------------------------
        new_score = _score(proposal)

        # ------------------------------------------------------------------
        # 4.3) Acceptance rule (Metropolis)
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = proposal
            cur_score = new_score

            # ----------------------------------------------------------------
            # 4.4) Record a new best solution
            # ----------------------------------------------------------------
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                last_improve = now
                elapsed_since_start = now - start_time
                print(f"[{elapsed_since_start:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4.5) Restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if now - last_improve > restart_gap:
            a = _jitter_restart(best, rng, jitter_sigma)
            cur_score = _score(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
