assistantfinal**Solution Overview**

The score  

\[
S(a)=\frac{2\,n\; \max (a*a)}{(\sum a)^2}
\]

is invariant to scaling, therefore we can keep the vector on the probability
simplex (∑ a = 1).  
The only thing that matters is the *max* entry of the self‑convolution.
For a given index \(k\) the gradient of the exact term
\(\max(a*a)\) is

\[
\frac{\partial}{\partial a_i}\,(a*a)_k
=
\begin{cases}
2\,a_{k-i} & \text{if }0\le k-i<n,\\[2mm]
0          & \text{otherwise.}
\end{cases}
\]

If several indices tie for the maximum we simply average their gradients.
The algorithm is therefore a **projected (mirror) gradient descent on the
simplex** that always uses the *exact* sub‑gradient of the ‑∞‑norm,
instead of the soft‑max approximation used before.

To keep the search robust we add occasional random jitter on a block,
swap two coordinates, and perform a soft‑restart when no improvement has
been seen for a while.  All operations are vectorised and the dominant
cost per iteration is a single FFT (self‑convolution).

The method works for any length; if a good solution already exists in
`height_sequence_1` it is used as a starting point, otherwise a uniform
distribution is taken.

```python
"""
Propose a sequence of non‑negative heights that (approximately) minimises

    S(a) = 2·n·max(conv(a,a)) / (∑a)²

using exact L∞‑sub‑gradients, exponentiated‑gradient (mirror) descent on the
probability simplex, and occasional random perturbations / restarts.
The routine respects the 1000 s time budget enforced by the evaluation
harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using a single FFT."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _max_conv_and_score(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Return (conv, 2·n·max(conv)) for a simplex‑normalised a."""
    conv = _conv_self(a)
    score = 2.0 * a.size * float(conv.max())
    return conv, score


def _gradient_at_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)) at the (possibly multiple) maximal
    indices `max_idx`.  For a single index k the gradient is

        g[i] = 2·a[k-i]    if 0 ≤ k-i < n
               0           otherwise.

    The returned gradient is the average over all maximal indices.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        grad[i] += 2.0 * a[k - i]
    grad /= max(1, max_idx.size)  # average if several maxima
    return grad


def _exp_gradient_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) step on the simplex:
        a ← a ⊙ exp(−η·grad)  , then re‑normalise.
    The exponent is clipped to avoid overflow.
    """
    log_factor = -eta * grad
    log_factor = np.clip(log_factor, -50.0, 50.0)
    a = a * np.exp(log_factor)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s
    return a


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (simplex‑normalised) that (approximately)
    minimise ``evaluate_sequence`` within the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initial vector – use the best-known construction if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            a = np.full(1024, 1.0 / 1024, dtype=np.float64)
    else:
        a = np.full(1024, 1.0 / 1024, dtype=np.float64)

    n = a.size
    a = a.astype(np.float64)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # a tiny safety margin

    conv, best_score = _max_conv_and_score(a)
    best = a.copy()
    last_improve = start_time
    print(f"[{0.0:6.2f}s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 2.0      # initial stepsize (large)
    eta_end = 0.001      # final stepsize (fine‑tuning)
    momentum = 0.85      # exponential smoothing of the gradient
    block_perturb_prob = 0.04
    swap_prob = 0.05
    restart_gap = 30.0   # seconds without improvement → restart
    jitter_sigma = 0.2   # log‑normal jitter for a restart

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution & score -----
        conv = _conv_self(a)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----- keep the best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule (linear interpolation) -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # ----- exact sub‑gradient of the L∞ term -----
        max_idx = np.flatnonzero(conv == cur_max)          # all maximal positions
        grad = _gradient_at_max(a, max_idx)

        # ----- momentum smoothing -----
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- mirror‑descent step -----
        a = _exp_gradient_update(a, grad, eta)

        # ----- occasional random perturbations -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]   # simple coordinate swap

        # ----- soft restart when progress stalls -----
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best solution (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best.tolist()]
```
