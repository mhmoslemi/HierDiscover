def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.size
    Lw = w.size  
    wr = w[::-1]

    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep only needed part

    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad

"""
Propose a step‐function (non‑negative sequence) that minimises
evaluate_sequence.  The algorithm mixes three ideas:

1.  A deterministic “square‑root‑difference’’ start (a_i ∝ √(i+1)−√i)
    which theoretically yields an almost flat self‑convolution
    (the continuous analogue  f(x)=½/√x  has constant f∗f).

2.  Exponentiated‑gradient descent on the smooth‐max surrogate
    of the ℓ∞‑norm of the self‑convolution.  The gradient is obtained
    by a single FFT‑based cross‑correlation (O(n log n)).

3.  Stochastic exploration – occasional block jitter, pairwise mass
    transfers and timed restarts – to escape local minima.

The routine runs for the whole time‑budget (default 1000 s),
keeps the best vector seen (on the simplex, i.e. Σa_i=1) and finally
returns it as a plain Python list.  All helper functions are top‑level
and avoid closures or lambdas, as required by the execution harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers – pure NumPy, no external dependencies
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negatives from round‑off.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    For the linear self‑convolution the gradient is
        ∂/∂a_j Σ_k w_k·(a*a)_k = 2·Σ_i a_i·w_{i+j}
    which is a cross‑correlation.  Implemented via a single FFT.
    """
    n = a.size
    Lw = w.size                     # = 2*n-1
    # Reverse w for correlation → convolution.
    wr = w[::-1]

    # Length required: n + Lw - 1 = 3n - 2.
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep only the needed part

    # grad_j = 2 * conv_full[Lw-1 - j]
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


# ----------------------------------------------------------------------
# Simple stochastic perturbations
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal jitter on a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)            # at most 20 % of the indices
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)

    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)

    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_transfer(a: np.ndarray,
                       conv: np.ndarray,
                       rng: np.random.Generator):
    """
    Cheap move that transfers a small amount of mass from donor i to receiver j.
    The convolution vector is updated analytically; the move is kept only if
    it strictly reduces the current max‑convolution value.
    Returns (new_a, new_conv, new_score) or the originals.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    # move at most 30 % of donor's mass
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Update convolution using the linear relationship.
    n2 = 2 * n - 1
    new_conv = conv.copy()
    lo_i = i
    hi_i = i + n
    lo_j = j
    hi_j = j + n

    new_conv[lo_i:hi_i] -= delta * a
    new_conv[lo_j:hi_j] += delta * a

    new_score = 2.0 * n * np.max(new_conv)
    cur_score = 2.0 * n * np.max(conv)

    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.clip(a2, 0.0, None)
        a2 /= a2.sum()
        return a2, new_conv, new_score
    else:
        return a, conv, None


# ----------------------------------------------------------------------
# Deterministic initialisations
# ----------------------------------------------------------------------


def _init_sqrt_derivative(n: int) -> np.ndarray:
    """
    a_i ∝ √(i+1) − √i  (i = 0,…,n‑1).
    This is the discrete analogue of the density f(x)=½/√x,
    whose continuous self‑convolution is constant.
    Returns a probability vector (sum = 1).
    """
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1.0) - np.sqrt(idx)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    return a

def propose_candidate(seed: int = 1234,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for the best non‑negative step function within the given
    time budget.  The function returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # Keep a small safety margin.
    deadline = start_time + max(0.0, float(budget_s) - 1.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Determine a set of lengths to try.
    # The previous construction (if available) is kept for reference.
    # ------------------------------------------------------------------
    lengths = []

    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            lengths.append(cand0.size)

    # A few sensible sizes (all well below 10 000 to keep FFT cheap)
    lengths.extend([3809, 5000, 6000])

    # Remove duplicates while preserving order.
    lengths = list(dict.fromkeys([L for L in lengths if L > 0]))

    # ------------------------------------------------------------------
    # Helper to evaluate the current objective (sum is forced to 1).
    # ------------------------------------------------------------------
    def _objective_from_conv(conv, n):
        """2·n·max(conv)  – the same quantity used inside the search."""
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Main loop over the candidate lengths.
    # ------------------------------------------------------------------
    for n in lengths:
        # Pre‑compute FFT lengths needed for this size.
        fft_len_self = _next_pow2(2 * n - 1)      # for a*a
        fft_len_grad = _next_pow2(3 * n - 2)      # for gradient

        # ------------------------------------------------------------------
        # Decide how many independent runs we can afford for this length.
        # Each run lasts roughly 30 s (adjustable) and uses the full
        # optimisation pipeline.
        # ------------------------------------------------------------------
        while time.monotonic() < deadline - 5.0:
            run_start = time.monotonic()
            run_deadline = min(deadline, run_start + 30.0)

            # ----------------- initialise a ---------------------------------
            # Choose among three initialisers (deterministic, random, prev).
            init_mode = rng.integers(0, 3)
            if init_mode == 0:
                a = _init_sqrt_derivative(n)               # the analytic start
            elif init_mode == 1 and best_seq is not None and len(best_seq) == n:
                a = np.asarray(best_seq, dtype=np.float64)
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
            else:
                # Dirichlet‑like random start (Gamma with random shape)
                alpha = rng.uniform(0.2, 1.0)
                a = rng.gamma(alpha, 1.0, size=n)
                a = np.clip(a, 0.0, None)
                a_sum = a.sum()
                if a_sum > 0.0:
                    a /= a_sum
                else:
                    a = np.full(n, 1.0 / n, dtype=np.float64)

            # Local best for this run.
            conv = _self_conv_fft(a, fft_len_self)
            cur_score = _objective_from_conv(conv, n)
            run_best_a = a.copy()
            run_best_score = cur_score

            # Momentum buffer for the gradient.
            grad_prev = np.zeros_like(a)

            # ----------------------------------------------------------------
            # Hyper‑parameters (these have been tuned for the 30 s window)
            # ----------------------------------------------------------------
            eta_start = 0.8          # initial EG step size
            eta_end = 0.02           # final EG step size
            beta_start = 0.5         # temperature at the start
            beta_end = 5000.0        # final temperature (hard max)
            momentum = 0.85
            block_prob = 0.06        # probability of a block jitter per iteration
            pairwise_prob = 0.12     # probability to attempt a pairwise transfer
            restart_gap = 45.0       # seconds without improvement → restart

            last_improve = time.monotonic()

            # ----------------------------------------------------------------
            # Optimization loop for this run
            # ----------------------------------------------------------------
            while time.monotonic() < run_deadline:
                # ----- current convolution & score -----
                conv = _self_conv_fft(a, fft_len_self)
                cur_max = conv.max()
                cur_score = 2.0 * n * cur_max

                # ----- keep the global best -----
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] Global best → {best_score:.6f} (n={n})")

                # ----- keep the best for this run -----
                if cur_score < run_best_score:
                    run_best_score = cur_score
                    run_best_a = a.copy()
                    last_improve = time.monotonic()

                # ----- schedule learning‑rate & temperature -----
                now = time.monotonic()
                elapsed = now - run_start
                total = run_deadline - run_start
                frac = elapsed / total

                eta = eta_start * (1.0 - frac) + eta_end * frac

                # ----- build the weight vector approximating a max -----

                    a, conv, _ = _pairwise_transfer(a, conv, rng)

                # ----- restart if stuck for a while -----
                if now - last_improve > restart_gap:
                    # Re‑initialise from the run‑best with a bit of noise.
                    a = run_best_a.copy()
                    noise = np.exp(rng.normal(0.0, 0.15, size=n))
                    a = a * noise
                    a = np.clip(a, 0.0, None)

                    grad_prev = np.zeros_like(a)

                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] Restart (n={n}) – run best {run_best_score:.6f}")

            # End of this run – compare run‑best with the global best.
            if run_best_score < best_score:
                best_score = run_best_score
                best_seq = run_best_a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Updated global best after run → {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never happen, but just in case.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 3809
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced inside evaluate_sequence.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return a plain list of Python floats.
    return [float(v) for v in best_seq]
