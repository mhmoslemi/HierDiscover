def propose_candidate(seed=42, budget_s=1000.0):
    import time, numpy as np
    rng = np.random.default_rng(seed)

    # Starting vector
    default_n = 256
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim==1 and cand.size>0 and np.isfinite(cand).all():
            a = np.clip(cand, 0, None)
            s = a.sum()
            if s>0:
                a /= s
            else:
                a = np.full(cand.size, 1.0/cand.size)
        else:
            n = default_n
            a = np.full(n, 1.0/n)
    else:
        n = default_n
        a = np.full(n, 1.0/n)

    n = a.size

    # Precompute FFT length for conv
    fft_len = 1 << ((2*n-2).bit_length())  # smallest power-of-two >= 2*n-1

    def conv_self(v):
        # compute full linear convolution via FFT
        fa = np.fft.rfft(v, n=fft_len)
        fc = fa * fa
        conv = np.fft.irfft(fc, n=fft_len)[:2*n-1]
        return conv

    # First evaluate
    conv = conv_self(a)
    max_conv = conv.max()
    cur_score = 2.0 * n * max_conv
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)
    last_improve = start_time

    # Algorithm hyperparameters
    eta_start = 1.0
    eta_end = 0.001
    momentum = 0.9
    grad_prev = np.zeros_like(a)
    block_prob = 0.01
    pair_prob = 0.05
    restart_gap = 30.0   # secs

    # Also keep random temperature for occasional acceptance of worse moves
    temp_start = 0.5
    temp_end = 0.001

    while time.monotonic() < deadline:
        # compute conv and max
        conv = conv_self(a)
        max_idx = int(np.argmax(conv))
        max_conv = conv[max_idx]
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve-start_time:.2f}s] new best {best_score:.6f}")

        # schedule parameters
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear schedule
        eta = eta_start * (1 - elapsed/total) + eta_end * (elapsed/total)
        temp = temp_start * (1 - elapsed/total) + temp_end * (elapsed/total)

        # gradient for max shift
        # compute grad = 2 * a shifted by max_idx
        grad = np.zeros_like(a)
        # Index range i where j = max_idx - i in [0,n-1]
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        # i vector
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end+1, dtype=np.int64)
            j = max_idx - i_range
            grad[i_range] = 2.0 * a[j]

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated update
        factor = np.exp(-eta * grad)
        a = a * factor
        # ensure non-neg and renorm
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional block perturb
        if rng.random() < block_prob:
            # same as previous _random_block_perturb code
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.15))
            a[start:start+width] *= factor
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # occasional pairwise transfer
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate /= a_candidate.sum()
                    # compute new score quickly maybe only evaluate conv at old max shift? but safe compute full conv
                    conv_cand = conv_self(a_candidate)
                    cand_score = 2.0 * n * conv_cand.max()
                    # accept if better, else maybe accept with probability
                    if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score)/temp):
                        a = a_candidate
                        # don't recompute conv now, loop will re-evaluate.

        # restart if stagnated
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now-start_time:.2f}s] restart – current best {best_score:.6f}")

    # End loop
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

"""
Propose a step‑function by minimizing the worst self‑convolution value.
The algorithm works on the probability simplex (a_i ≥ 0, Σ a_i = 1) and
uses a *max‑gradient* (sub‑gradient of the max‑convolution) with an
exponentiated‑gradient (mirror‑descent) update.  A linear schedule for the
stepsize η slowly cools the update, while occasional random block
perturbations, pairwise mass transfers and soft restarts avoid stagnation.
All work is done with FFT‑based convolutions for speed.  The best vector
found before the time budget expires is returned.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n`` (i.e. length ``2·n‑1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length ``2·len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the max‑convolution term at index ``k_max``.
    For a given shift k, the gradient is 2·a_{k‑i} (zero outside the support)."""
    n = a.size
    grad = np.zeros_like(a)
    # i ∈ [max(0, k_max‑(n‑1)), min(n‑1, k_max)]
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_start <= i_end:
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k_max - i_range
        grad[i_range] = 2.0 * a[j]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector (unit sum)
    # ------------------------------------------------------------------
    default_n = 256
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(cand.size, 1.0 / cand.size)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    max_conv = float(conv.max())
    cur_score = 2.0 * n * max_conv          # sum(a)=1 ⇒ score = 2·n·max(conv)
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 1.0          # initial stepsize for exponentiated gradient
    eta_end = 0.001          # final stepsize
    momentum = 0.9           # EMA on gradients
    block_perturb_prob = 0.01
    pair_move_prob = 0.05
    restart_gap = 30.0       # seconds without improvement → restart
    temp_start = 0.4         # temperature for occasional uphill moves
    temp_end = 0.001

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- evaluate current state -----------------------------------
        conv = _conv_self_fft(a, fft_len)
        k_max = int(np.argmax(conv))
        max_conv = float(conv[k_max])
        cur_score = 2.0 * n * max_conv

        # ----- keep the best solution ----------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule -------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        temp = temp_start * (1.0 - elapsed / total) + temp_end * (elapsed / total)

        # ----- gradient of the active (maximum) shift -----------------
        grad = _gradient_max(a, k_max)

        # ----- momentum smoothing --------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror descent) update ----------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation --------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pair‑wise mass transfer -----------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate /= a_candidate.sum()
                    # evaluate candidate (full conv – cheap enough)
                    conv_cand = _conv_self_fft(a_candidate, fft_len)
                    cand_score = 2.0 * n * conv_cand.max()
                    # accept if improvement or with small uphill probability
                    if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(temp, 1e-12)):
                        a = a_candidate
                        # cur_score will be recomputed at top of loop

        # ----- soft restart when stuck ---------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter all coordinates multiplicatively
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (respect the clipping performed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
