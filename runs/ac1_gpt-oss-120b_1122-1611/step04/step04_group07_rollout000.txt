"""
Propose a candidate step‑function sequence.

Algorithm
---------
1.  Initialise a probability vector `a` (sum = 1) – either from the
    provided `height_sequence_1` or a random Dirichlet draw.
2.  **Linear‑programming refinement** – repeatedly solve a small LP that
    maximises the total mass of a partner vector `b` while keeping the
    convolution `a*b` below the current peak `M = max(conv(a,a))`.  The
    solution is normalised back to sum‑one and mixed with `a` (mixing
    factor `t`).  This step quickly reduces the worst‑case autocorrelation
    (the `max` in the score).
3.  **Soft‑max mirror‑descent (Adam on logits)** – after the LP phase the
    sequence is further polished with a smooth approximation of the
    `max` (soft‑max) and an Adam optimiser on the log‑odds (logits) of the
    probabilities.  This refines the shape without leaving the simplex.
4.  Throughout the search the exact `evaluate_sequence` score is computed
    whenever a better estimate is found.  The best sequence seen before the
    time limit is returned.

The combination of an LP “peak‑flattening’’ step and a modern
gradient‑based fine‑tuner gives a good chance of improving the previous
bound (≈ 1.514) within the 1000 s budget.
"""

import time
import numpy as np
import math

# try to import cvxpy for the linear programme; fall back to scipy if unavailable
try:
    import cvxpy as cp
    _HAVE_CVX = True
except Exception:   # pragma: no cover
    from scipy.optimize import linprog
    _HAVE_CVX = False

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that holds the full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using FFT (full length 2*n‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# LP refinement
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Perform a few iterations of the LP “peak‑flattening’’ step.
    The vector `a` is assumed to sum to 1.
    """
    n = a.shape[0]
    # mixing weight (how aggressively we move towards the LP solution)
    mix = 0.2
    # max number of LP iterations per call (small, each is cheap)
    max_iter = 8

    # pre‑allocate the convolution matrix shape
    rows = np.arange(2 * n - 1)[:, None]   # (2n‑1,1)
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # current peak M
        fft_len = _fft_len_for_n(n)
        conv = _conv_self(a, fft_len)
        M = float(conv.max())

        # Build dense convolution matrix A where (A @ b)[k] = sum_i a_i * b_{k-i}
        # Column j (corresponding to b_j) contains a shifted downwards by j.
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        # Solve LP: maximise sum(b)  s.t.  A @ b <= M,  b >= 0
        if _HAVE_CVX:
            b_var = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)),
                              [A @ b_var <= M + 1e-12])
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
            except Exception:  # pragma: no cover
                prob.solve(solver=cp.SCS, warm_start=True, max_iters=2500, verbose=False)
            if b_var.value is None:
                break
            b_opt = np.maximum(0.0, np.asarray(b_var.value, dtype=np.float64))
        else:   # fallback to scipy.linprog (slower but works)
            c = -np.ones(n)                     # maximise sum(b)  -> minimise -sum(b)
            bounds = [(0, None) for _ in range(n)]
            # linprog expects A_ub @ x <= b_ub
            res = linprog(c, A_ub=A, b_ub=M + 1e-12, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                break
            b_opt = np.maximum(0.0, res.x)

        sum_b = b_opt.sum()
        if sum_b == 0.0:
            break

        # Normalise the LP solution to sum‑one and mix it back into a
        g = b_opt / sum_b
        a = (1.0 - mix) * a + mix * g
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a

# Gradient (soft‑max + Adam) refinement
# ----------------------------------------------------------------------


def _gradient_refine(a: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Mirror‑descent on the probability simplex using Adam on logits.
    Uses a soft‑max approximation of the max‑norm.
    """
    n = a.shape[0]
    fft_len = _fft_len_for_n(n)

    # initialise logits u = log(a) (add tiny epsilon to avoid log(0))
    eps = 1e-12
    u = np.log(np.maximum(a, eps))

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    step = 0
    start = time.monotonic()

    # hyper‑parameter schedules
    eta_start = 0.8
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 500.0

    block_perturb_prob = 0.01
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        if total <= 0.0:
            break

        # schedules (linear)
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # compute full convolution of current a
        conv = _conv_self(a, fft_len)
        max_conv = float(conv.max())

        # soft‑max weights (numerically stable)
        shift = max_conv
        w_raw = np.exp(beta * (conv - shift))        # max weight = 1
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # gradient w.r.t. a (valid part only)
        grad_a = 2.0 * np.correlate(w, a, mode='valid')   # length n

        # gradient w.r.t. logits u (softmax Jacobian)
        grad_u = a * (grad_a - np.dot(a, grad_a))

        # Adam update on logits
        step += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        u -= eta * m_hat / (np.sqrt(v_hat) + epsilon)

        # back to probabilities (softmax)
        u_shift = u - np.max(u)          # for numerical stability
        a = np.exp(u_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # occasional block perturbation to escape shallow minima
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

    return a
