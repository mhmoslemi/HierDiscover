"""
propose_candidate
-----------------
Hybrid gradient‑descent + simulated‑annealing search for the step‑function
that minimises ``evaluate_sequence``.
The algorithm
* starts from a uniform simplex vector (or the supplied ``height_sequence_1``)
* repeatedly computes the self‑convolution via FFT,
* builds a soft‑max approximation of the infinity‑norm,
* takes an Adam‑style step on the probability simplex,
* interleaves cheap pairwise mass‑transfer moves and random block
  perturbations,
* performs occasional restarts with log‑normal jitter when progress stalls.
All operations are O(n log n) and the loop runs until the 1000 s time budget
is exhausted.
The best vector seen is returned (clipped to the limits enforced by
``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length sufficient for a linear convolution
    of two vectors whose combined length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT.  The result length is
    ``2*len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of ``x`` and ``y`` using FFT (full length)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.  A new array is returned."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u > cssv / ind
    if not np.any(mask):
        # corner case – return uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[mask][-1] - 1
    theta = cssv[mask][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)

# ----------------------------------------------------------------------

"""
propose_candidate
-----------------
Hybrid gradient‑descent + simulated‑annealing search for the step‑function
that minimises ``evaluate_sequence``.  

The algorithm
* starts from a uniform simplex vector (or from the optional
  ``height_sequence_1`` if it is present),
* repeatedly computes the self‑convolution via FFT,
* builds a soft‑max approximation of the infinity‑norm,
* takes an Adam‑style step while staying on the probability simplex,
* interleaves cheap pair‑wise mass‑transfer moves and random block
  perturbations,
* performs a soft restart with log‑normal jitter when progress stalls.

All operations are ``O(n log n)`` and the loop runs until the 1000 s
budget is exhausted.  The best vector seen is returned (clipped to the
limits enforced by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length sufficient for a linear convolution
    of two vectors whose combined length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT.  The result length is
    ``2*len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of ``x`` and ``y`` using FFT (full length)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Returns a new array."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    mask = u > cssv / ind
    if not np.any(mask):
        # corner case – return uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[mask][-1] - 1
    theta = cssv[mask][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **_):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided ``evaluate_sequence`` function.  The routine respects the
    1000 s time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the vector (use the global hint if available)
    # ------------------------------------------------------------------
    default_n = 2400
    if "height_sequence_1" in globals():
        try:
            cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
                a = np.clip(cand0, 0.0, None)
                if a.sum() > 0.0:
                    a /= a.sum()
                else:
                    raise ValueError
                n = a.size
            else:
                raise ValueError
        except Exception:
            n = default_n

    else:

    # ------------------------------------------------------------------

    fft_len_grad = _fft_len_for_len(3 * n - 2)   # for gradient term

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len_conv)
    best = a.copy()
    best_score = 2.0 * n * conv.max()            # sum(a) == 1 → denominator = 1
    start_time = time.monotonic()

    eps = 1e-8

    v = np.zeros_like(a)
    t = 0

    # ------------------------------------------------------------------
    block_prob = 0.03
    pair_prob = 0.1
    restart_gap = 25.0            # seconds without improvement → restart
    momentum = 0.8                # smoothed gradient term

    while time.monotonic() < deadline:

        conv = _conv_self(a, fft_len_conv)

            best = a.copy()

        # ----- 6.3) Linear schedules for η and β ------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta_start, eta_end = 0.9, 0.003
        beta_start, beta_end = 0.5, 800.0
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ----- 6.4) Soft‑max weighting (smooth approximation of max) -----
        w = np.exp(beta * (conv - cur_max))          # shifted for stability
        sum_w = w.sum()

        # ----- 6.5) Gradient of the smooth max -------------------------
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
        grad_piece = conv_w_rev[n - 1:n - 1 + n] / sum_w
        raw_grad = a + grad_piece                     # ∂f_β/∂a

        # momentum smoothing (helps stability)
        raw_grad = momentum * raw_grad + (1.0 - momentum) * raw_grad

        # ----- 6.6) Adam update (additive) ----------------------------
        t += 1
        m = b1 * m + (1.0 - b1) * raw_grad
        v = b2 * v + (1.0 - b2) * (raw_grad ** 2)

        v_hat = v / (1.0 - b2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

                delta = rng.random() * min(a[i], 0.3 * a[i])
                a_candidate = a.copy()
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _project_onto_simplex(a_candidate)
                conv_cand = _conv_self(a_candidate, fft_len_conv)
                cand_score = 2.0 * n * conv_cand.max()
                if cand_score < cur_score:

                    cur_score = cand_score
                    cur_max = conv_cand.max()

            last_improve = now
            elapsed = now - start_time

    # ------------------------------------------------------------------
    # 7) Return the best vector (respect clipping limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
