def conv_max(a):
    n = len(a)
    fft_len = 1 << ((2*n - 2).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)

def _project_simplex(v):
    """ Euclidean projection of v onto probability simplex (non-negative, sum=1). """
    # v is 1D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # Find rho
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

"""
propose_candidate
-----------------
Search for a non‑negative step function (probability vector) that
minimises ``evaluate_sequence``.  The algorithm starts from the
previous best construction (if available) and then runs a
mirror‑descent / Adam optimisation on a smooth approximation of the
max‑convolution functional.  A temperature schedule turns the smooth
soft‑max into a hard max, while an additive Adam update with projection
onto the simplex refines the vector.  Random block jitter and cheap
pair‑wise mass‐transfer moves break symmetry and occasional restarts
avoid stagnation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negatives introduced by numerical noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted max (or smooth max) of the convolution
    ``a*a`` with respect to ``a``.
    The formula is  ∂/∂a_j Σ_k w_k·(a*a)_k = 2· Σ_i a_i·w_{i+j}
    which is a cross‑correlation between ``a`` and the weight vector.
    This implementation uses a single FFT‑based convolution.
    """
    n = a.size
    Lw = w.size                         # = 2*n-1
    wr = w[::-1]                        # reversed for correlation

    # length needed for a (n) + wr (2n-1) - 1 = 3n-2
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep needed part only

    # gradient_j = 2 * conv_full[Lw-1 - j]
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a multiplicative log‑normal perturbation to a random contiguous
    block of the vector, then renormalise onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)               # ≤20 % of the vector
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)

    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_transfer(a: np.ndarray, conv: np.ndarray, rng: np.random.Generator):
    """
    Cheap pairwise mass transfer move.
    Returns a new (a, conv, score) tuple if the move improves the
    current objective, otherwise the original tuple unchanged.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Update the convolution in O(n) using the linearity of convolution.
    new_conv = conv.copy()
    lo_i = i
    hi_i = i + n
    lo_j = j
    hi_j = j + n
    new_conv[lo_i:hi_i] -= delta * a
    new_conv[lo_j:hi_j] += delta * a

    new_score = 2.0 * n * np.max(new_conv)
    cur_score = 2.0 * n * np.max(conv)

    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.clip(a2, 0.0, None)
        a2 /= a2.sum()
        return a2, new_conv, new_score
    else:
        return a, conv, None

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Determine problem size and initialise the probability vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = 4096
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # FFT lengths for the two convolutions we need
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for correlation a ⋆ w

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _self_conv_fft(best, fft_len_self)
    best_score = 2.0 * n * conv_best.max()      # sum(best) == 1

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # small safety margin
    last_improve = start_time

    # --------------------------------------------------------------
    # 3) Hyper‑parameters (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # Learning‑rate schedule (multiplicative factor on the gradient)
    lr_start = 0.15
    lr_end = 0.001

    # Temperature schedule for the soft‑max weighting of the convolution entries
    beta_start = 0.5
    beta_end = 1800.0

    block_perturb_prob = 0.025
    restart_gap = 30.0                     # seconds without improvement → restart
    pairwise_prob = 0.07

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current self‑convolution (FFT) -----
        conv = _self_conv_fft(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule for learning‑rate and temperature -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        lr = lr_start * (1.0 - frac) + lr_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- weight vector w (soft‑max or hard max) -----
        if beta < 10.0:
            # soft‑max (stable: max exponent = 0)
            w = np.exp(beta * (conv - cur_max))
        else:
            # exact sub‑gradient for the hard max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ----- gradient of the weighted max (full‑gradient) -----
        grad = _gradient_from_weights(a, w)

        # ----- Adam update -----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # additive update and projection onto the simplex
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- cheap pairwise mass transfer -----
        if rng.random() < pairwise_prob:
            a, conv, _ = _pairwise_transfer(a, conv, rng)

        # ----- restart when stuck -----
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.14, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            t_iter = 0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return best sequence (clipped to limits of evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
