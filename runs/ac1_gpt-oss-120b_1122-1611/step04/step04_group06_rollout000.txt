"""
Propose a candidate step‑function via a two‑phase mirror‑descent /
sub‑gradient descent with aggressive random local search.

* Phase 1 (soft‑max)   –  smooth surrogate of the ∞‑norm, i.e.
  w_k ∝ exp(β·(conv_k – max(conv))) with β rising to a very large
  value.  This quickly lowers the peak of the self‑convolution.

* Phase 2 (hard‑max)   –  switch to the exact sub‑gradient of the
  max‑operator (mask of indices attaining the current maximum).  This
  gives a sharper descent direction.

* Random perturbations –  pairwise mass transfers, block‑wise log‑normal
  jitter and a targeted move that steals a little mass from the
  coordinates that contribute most to the current peak lag.

* Restarts –  if no improvement for ``restart_gap`` seconds, jitter the
  current best solution and continue.

The algorithm works on the probability simplex (a_i ≥ 0, Σa_i = 1),
so the objective 2·n·max(conv) is exactly the score returned by
``evaluate_sequence`` (the sum‑normalisation cancels out).  The code
uses FFT‑based convolution/correlation for efficiency and respects the
1000 s time budget.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution
    of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the soft‑max / masked surrogate:
       grad_j = 2·∑_k w_k·a_{k−j}
       (linear correlation via FFT)."""
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _targeted_move(a: np.ndarray, conv: np.ndarray, rng: np.random.Generator,
                   min_dist_frac: float = 0.15, max_delta_frac: float = 0.15) -> np.ndarray:
    """Steal a small amount of mass from the index that contributes most
    to the current maximal lag of the convolution and give it to a far
    away index."""
    n = a.size
    # lag with maximal convolution value
    L = int(np.argmax(conv))
    # admissible i where both i and L-i lie in [0,n)
    i_min = max(0, L - (n - 1))
    i_max = min(n - 1, L)
    i_vals = np.arange(i_min, i_max + 1)
    j_vals = L - i_vals
    contrib = a[i_vals] * a[j_vals]
    if contrib.size == 0:
        return a
    idx = int(np.argmax(contrib))
    i = i_vals[idx]

    # choose a destination far enough from i
    min_dist = int(min_dist_frac * n)
    far = np.abs(np.arange(n) - i) > min_dist
    if not np.any(far):
        return a
    j = rng.choice(np.where(far)[0])

    delta = min(a[i] * max_delta_frac, a[i]) * rng.random()
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """

    # ------------------------------------------------------------------

                s = a.sum()

            else:

        except Exception:

    else:

    n = a.size

    while time.monotonic() < deadline:
