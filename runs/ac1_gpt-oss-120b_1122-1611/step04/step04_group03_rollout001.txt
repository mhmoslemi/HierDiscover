    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0., budget_s - 5.)

    # initial a:
    if 'height_sequence_1' in globals():
        cand = np.asarray(height_sequence_1, dtype=float)
        a = np.clip(cand, 0, None)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = uniform.
    else:
        n = 2048 # maybe choose power-of-two
        a = np.full(n, 1/n)

    n = a.shape[0]

    best = a.copy()
    best_score = 2 * n * _conv_self(a).max()
    last_improve = time.monotonic()
    start = time.monotonic()

    # hyperparameters
    eta0 = 0.8
    eta1 = 0.001
    beta0 = 0.2
    beta1 = 2000.0
    momentum = 0.9
    grad_prev = np.zeros_like(a)
    # perturbation
    block_prob = 0.07
    swap_prob = 0.07
    restart_gap = 15.0
    jitter_sigma = 0.2

    # FFT lens
    fft_len_self = _next_pow2(2*n-1)
    fft_len_grad = _next_pow2(3*n-2)

    while time.monotonic() < deadline:

        # compute conv and score
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:.2f}s] new best {best_score:.6f}")

        # schedule
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        eta = eta0 * (1 - elapsed/total) + eta1 * (elapsed/total)
        beta = beta0 + (beta1 - beta0) * (elapsed/total)

        # compute softmax weights
        shift = conv.max()   # largest
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # compute gradient: conv of w with a_rev
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: len(w) + n - 1]
        grad = conv_rev[n-1:n-1+n] + a

        # add momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated gradient update
        log_factor = -eta * grad
        # Clip
        log_factor = np.clip(log_factor, -50, 50)
        factor = np.exp(log_factor)
        a = a * factor
        # clip and renorm
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # random perturb block
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        # random swap
        if rng.random() < swap_prob:
            i,j = rng.integers(0,n,2)
            a[i], a[j] = a[j], a[i]

        # restart if stagnated
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now-start:.2f}s] restart – best {best_score:.6f}")

    # Return best as list
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

The objective (with the sum of the sequence fixed to 1) is

    F(a) = 2·n·max(conv(a,a))

i.e. a convolution of the softmax weights with the reversed vector plus the
probability vector itself.  We perform exponentiated‑gradient (mirror‑descent
on the simplex) using this gradient, with a linearly growing temperature β
and a linearly decreasing stepsize η.  To escape plateaus we occasionally
apply a multiplicative log‑normal jitter to a random block, swap two random
coordinates and perform soft restarts when no improvement occurs for a
while.  The algorithm is fully vectorised and runs comfortably within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT (length 2·len(a)‑1)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renorm."""
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative sequence that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector (sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    raise ValueError
            else:
                raise ValueError
        except Exception:
            default_n = 4096
            a = np.full(default_n, 1.0 / default_n, dtype=np.float64)
    else:
        default_n = 4096
        a = np.full(default_n, 1.0 / default_n, dtype=np.float64)

    n = a.size                     # length of the candidate
    a_rev = a[::-1]                # will be reused after updates

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 1.5               # stepsize at the beginning
    eta_end = 0.001                # final stepsize
    beta_start = 0.2               # low temperature – very smooth
    beta_end = 5000.0              # high temperature – near hard max
    momentum = 0.8                 # exponential smoothing of the gradient
    block_perturb_prob = 0.1       # how often we jitter a block
    swap_prob = 0.1                # how often we swap two coordinates
    restart_gap = 12.0             # seconds without improvement → restart
    jitter_sigma = 0.22            # log‑normal jitter during restart

    grad_prev = np.zeros_like(a)

    # Pre‑compute FFT lengths (depend only on n)
    fft_len_self = _next_pow2(2 * n - 1)      # for a∗a
    fft_len_grad = _next_pow2(3 * n - 2)      # for w∗a_rev

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- current self‑convolution and score -----------------------
        conv = _conv_self(a)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- keep the best solution ----------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule (linear interpolation) -------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- smoothed max via soft‑max weights -----------------------
        shift = conv.max()                # avoid overflow
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- exact gradient of the smoothed objective -----------------
        # convolution w * a_rev (length 3n‑2)
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a[::-1], n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: (len(w) + n - 1)]
        # gradient = (w * a_rev)[n‑1 : n‑1+n] + a
        grad = conv_rev[n - 1 : n - 1 + n] + a

        # ---- momentum smoothing ---------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- exponentiated‑gradient (mirror descent) update ----------
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -50.0, 50.0)   # avoid overflow
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- occasional random perturbations ---------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # ---- soft restart when progress stalls -------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best solution (clip to limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
