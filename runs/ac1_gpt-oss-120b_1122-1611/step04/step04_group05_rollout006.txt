"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a fast simulated‑annealing / record‑to‑record‑travel strategy.
The algorithm works on the probability simplex (∑a = 1) and keeps the
convolution up‑to‑date with cheap O(n) updates for elementary
pairwise mass‑transfer moves.  Large “block‑perturb” moves are evaluated
with a full FFT convolution only rarely.  A simple temperature schedule
allows occasional uphill steps early on and converges to a greedy
improvement phase later.  The best sequence seen is returned at the end
of the time budget.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _initial_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a good starting distribution of length ``n``.
    We compare the stored ``height_sequence_1`` (if any) with the
    arcsine shape and keep the better one (by the objective value)."""
    # default uniform
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # arcsine shape (≈ 1/√(i·(n‑1‑i)))
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = np.clip(arcsine, 0.0, None)
    arcsine /= arcsine.sum()
    # pick better of uniform, arcsine and possible external candidate
    candidates = [a, arcsine]

    # use external candidate if it exists and is valid
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size == n and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                else:
                    cand = np.full(n, 1.0 / n, dtype=np.float64)
                candidates.append(cand)
        except Exception:
            pass

    fft_len = _fft_len_for_n(n)
    best = candidates[0]
    best_score = np.inf
    for cand in candidates:
        conv = _conv_self(cand, fft_len)
        cur = 2.0 * n * conv.max()          # sum(cand)=1 → denominator =1
        if cur < best_score:
            best_score = cur
            best = cand.copy()
    return best


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.

    The optimizer uses a simulated‑annealing style search with cheap O(n)
    convolution updates for pairwise mass moves and occasional full‑FFT
    block perturbations.  The best sequence encountered is printed and
    finally returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose length and initialise a probability vector (∑a = 1)
    # ------------------------------------------------------------------
    default_n = 4000                     # a good trade‑off between quality and speed
    n = default_n
    a = _initial_distribution(n, rng)    # already normalised to sum = 1

    # ------------------------------------------------------------------
    # 2) Pre‑compute the initial convolution and bookkeeping
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)
    conv = _conv_self(a, fft_len)        # length 2*n‑1
    cur_max = float(conv.max())
    cur_score = 2.0 * n * cur_max        # sum(a) == 1

    best = a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a small safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (linear schedules)
    # ------------------------------------------------------------------
    # temperature controls how much uphill movement we allow (in absolute score units)
    T_start = best_score * 0.025          # ≈ 2.5 % of the current score
    T_end = 1e-6

    # maximal fraction of a[i] that can be moved in a single pairwise step
    frac_start = 0.5
    frac_end = 0.02

    # probability of a “big” block perturbation (evaluated with a full FFT)
    block_prob = 0.02

    # restart if we have been stuck for this many seconds
    restart_gap = 30.0

    M = conv.size                         # == 2*n‑1

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # linear interpolation of schedules
        T = T_start * (1.0 - elapsed / total) + T_end * (elapsed / total)
        move_frac = frac_start * (1.0 - elapsed / total) + frac_end * (elapsed / total)

        # ------------------------------------------------------------------
        # 4a) Decide on the move type
        # ------------------------------------------------------------------
        if rng.random() < block_prob:        # ***** block perturbation *****
            # Choose a random contiguous block and multiply it by a log‑normal factor
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.18))

            a_new = a.copy()
            a_new[start:start + width] *= factor
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s == 0.0:
                a_new[:] = 1.0 / n
            else:
                a_new /= s

            # Full convolution (rare, so FFT cost is acceptable)
            conv_new = _conv_self(a_new, fft_len)
            new_max = float(conv_new.max())
            new_score = 2.0 * n * new_max

            if new_score <= cur_score + T:
                a = a_new
                conv = conv_new
                cur_score = new_score
                cur_max = new_max
                if new_score < best_score:
                    best_score = new_score
                    best = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

            continue        # go to next loop iteration

        # ***** pairwise mass transfer *****
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # choose transfer amount (cannot exceed a[i])
        max_delta = a[i] * move_frac
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta      # 0 ≤ delta ≤ max_delta

        # --------------------------------------------------------------
        # cheap O(n) update of the convolution for the proposal
        # --------------------------------------------------------------
        conv_new = conv.copy()

        # contribution from a*d (and d*a) → factor -2·delta for i, +2·delta for j
        # slice size can be truncated at the right boundary
        Li = min(n, M - i)
        conv_new[i:i + Li] += -2.0 * delta * a[:Li]

        Lj = min(n, M - j)
        conv_new[j:j + Lj] += 2.0 * delta * a[:Lj]

        # d*d term (three isolated indices)
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = float(conv_new.max())
        new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # acceptance test (record‑to‑record travel)
        # --------------------------------------------------------------
        if new_score <= cur_score + T:
            # accept the move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_score = new_score
            cur_max = new_max

            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                last_improve = time.monotonic()
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4b) Restart / jitter if we have been stagnant for a while
        # ------------------------------------------------------------------
        if now - last_improve > restart_gap:
            # jitter the current best a little and resume from there
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            cur_max = conv.max()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart with jitter – best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) Return the best vector (clipped to the hard limits of evaluate_sequence)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
