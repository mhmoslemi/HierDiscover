def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    m = a.size + b.size - 1
    L = _next_pow2(m)
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)[:m]
    return conv


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) – FFT length must be ≥ 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    # block width at most 10 % of the vector, at least 1
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

                    # transfer up to 30 % of donor's mass
                    max_delta = 0.30 * a[i]
                    if max_delta > 0.0:
                        delta = rng.random() * max_delta

                        a_new = a.copy()
                        a_new[i] -= delta
                        a_new[j] += delta

                        conv_new = _self_conv(a_new, fft_len_self)
                        new_score = 2.0 * n * conv_new.max()

                        # Metropolis acceptance
                        if new_score < cur_score or rng.random() < np.exp(
                            -(new_score - cur_score) / temperature
                        ):
                            a = a_new
                            conv = conv_new
                            cur_score = new_score

            # ----- block‑level perturbation (global jitter) ----------
            else:
                a = _random_block_perturb(a, rng)
                conv = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # Re‑initialise from the current best plus moderate noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            conv = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
