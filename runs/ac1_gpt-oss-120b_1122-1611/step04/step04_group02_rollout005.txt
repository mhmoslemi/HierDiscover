def _conv_fft(x, y, fft_len=None):

    conv = np.fft.irfft(fx * fy, n=fft_len)[:L]
    return conv

def _project_simplex(v):
    # v may be any real numbers
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    # find rho
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that reduces
evaluate_sequence().

Algorithm – ADAM + smooth max surrogate + occasional stochastic
pairwise moves + adaptive dimensionality
--------------------------------------------------------------------
* The objective f(a)=2·n·max(conv(a))/ (∑a)² is invariant under scaling,
  therefore we fix the ℓ¹‑norm to 1 (simplex).

* We minimise a smooth approximation of max(conv) based on the
  log‑sum‑exp (soft‑max) with a temperature β that grows from 1 to a
  few thousand. The gradient of
      g(a)=log‑sum‑exp(β·conv(a))
  can be written as a cross‑correlation of a and the soft‑max weights,
  which we compute efficiently with FFTs.

* An Adam optimiser (first‑order momentum) updates a on the simplex.
  After each update we project back onto the probability simplex.

* To avoid being trapped in a smooth‑approximation local minimum we
  interleave cheap stochastic moves: pick two indices (i,j) and transfer
  a random fraction of mass from i to j. Acceptance follows a simple
  Metropolis rule with a linearly decreasing temperature.

* The search starts with a modest dimension (n≈256). After roughly
  half of the time budget the vector is “refined” by splitting each
  entry into two equal halves (n doubles, up to max_n). This adds
  degrees of freedom while keeping the current shape as a good
  initial guess.

* Restarts: if no improvement is seen for ~30 s we jitter the current
  best solution and continue.

The routine respects the 1000 s budget (leaving a 5 s safety margin),
keeps the best vector seen and finally returns it as a plain list of
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    if fft_len is None:
        fft_len = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[: len(x) + len(y) - 1]
    return conv


def _conv_self_fft(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT."""
    return _conv_fft(a, a, fft_len)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, ∑x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and re‑normalise.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_simplex(out)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.
    The implementation follows the description in the module docstring.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling – leave a small safety margin.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose an initial vector on the simplex.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        init = np.clip(init, 0.0, None)
        if init.size == 0 or not np.isfinite(init).all():
            init = rng.dirichlet(np.ones(256)).astype(np.float64)
    else:
        init = rng.dirichlet(np.ones(256)).astype(np.float64)

    if init.sum() == 0.0:
        init = np.full_like(init, 1.0 / init.size)
    else:
        init = init / init.sum()

    a = init.copy()
    n = a.size

    # ------------------------------------------------------------------
    # 2) Adaptive dimensionality – we will double once after ~50 % of the time.
    # ------------------------------------------------------------------
    max_n = 1024                # upper bound for the dimension
    expanded = False
    expand_frac = 0.5           # fraction of the total budget after which we expand

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT lengths for the current dimension.
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_cross = _next_pow2((2 * n - 1) + n - 1)   # length needed for cross‑correlation

    # ------------------------------------------------------------------
    # 4) Adam state and hyper‑parameters.
    # ------------------------------------------------------------------
    m = np.zeros_like(a)        # 1st moment
    v = np.zeros_like(a)        # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_start = 0.07             # initial learning rate (for Adam)
    iter_cnt = 0

    # ------------------------------------------------------------------
    # 5) Soft‑max surrogate schedule.
    # ------------------------------------------------------------------
    beta_start = 1.0
    beta_end = 2000.0

    # ------------------------------------------------------------------
    # 6) Book‑keeping.
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len_self)
    cur_score = 2.0 * n * conv.max()          # sum(a)=1 ⇒ evaluate_sequence == this
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    restart_gap = 30.0       # seconds without improvement → restart

    # ------------------------------------------------------------------
    # 7) Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = min(1.0, elapsed / total_time)   # 0 → 1 over the whole run

        # --------------------------------------------------------------
        # Dimension expansion (once).
        # --------------------------------------------------------------
        if not expanded and frac > expand_frac and n < max_n:
            new_n = min(max_n, n * 2)
            # split each entry equally to keep the shape
            a = np.repeat(a, new_n // n) / (new_n // n)
            n = new_n
            # reset Adam buffers for the new size
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            # recompute FFT lengths
            fft_len_self = _next_pow2(2 * n - 1)
            fft_len_cross = _next_pow2((2 * n - 1) + n - 1)
            # recompute convolution & score
            conv = _conv_self_fft(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            expanded = True
            print(f"[{elapsed:6.2f}s] expanded dimension to {n}")

        # --------------------------------------------------------------
        # Choose an update mode.
        # --------------------------------------------------------------
        if rng.random() < 0.70:               # 70 % gradient‑based step
            # ----- soft‑max temperature & learning‑rate schedules -----
            beta = beta_start + (beta_end - beta_start) * frac
            lr = lr_start * (0.5 ** frac)   # exponential decay

            # ----- current convolution and soft‑max weights ------------
            conv = _conv_self_fft(a, fft_len_self)
            max_conv = conv.max()
            # shift for numerical stability (max_conv ≤ conv ≤ max_conv)
            shifted = conv - max_conv
            # w_i ∝ exp(beta·(conv_i – max_conv)); all w sum to 1
            w = np.exp(beta * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient of the smooth surrogate --------------------
            # grad = 2 * (w ⋆ a)   (cross‑correlation)
            a_rev = a[::-1]
            cross = _conv_fft(w, a_rev, fft_len_cross)       # length = 3n‑2
            grad = 2.0 * cross[(n - 1) : (n - 1 + n)]        # extract n entries

            # ----- Adam update on the simplex -------------------------
            iter_cnt += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** iter_cnt)
            v_hat = v / (1.0 - beta2 ** iter_cnt)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _project_simplex(a)

            # ----- occasional random block perturbation ---------------
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)

            # ----- recompute convolution & score ----------------------
            conv = _conv_self_fft(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()

        else:                                 # 30 % stochastic local move
            # ----- pick donor (i) and recipient (j) -----------------
            i = rng.integers(0, n)
            if a[i] > 1e-12:
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_delta = 0.30 * a[i]
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_new = a.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    conv_new = _conv_self_fft(a_new, fft_len_self)
                    new_score = 2.0 * n * conv_new.max()
                    # simple annealing temperature (linear decay)
                    temperature = max(1e-8, 0.02 * (1.0 - frac))
                    if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                        a = a_new
                        conv = conv_new
                        cur_score = new_score

        # --------------------------------------------------------------
        # Keep the best solution seen so far.
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart if we have been stuck for a while.
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # multiplicative jitter around the current best
            jitter = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * jitter
            a = _project_simplex(a)
            # reset Adam moments
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            conv = _conv_self_fft(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8) Return the best vector (clipped to the limits enforced by evaluate_sequence).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
