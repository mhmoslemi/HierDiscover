### Proposed code structure

```python
"""
Mirror‑descent + hard‑max refinement for minimizing the max
self‑convolution under a simplex constraint.

The algorithm runs three phases:
  1. Soft‑max smoothing of the max‑convolution (wide temperature).
  2. Sharper soft‑max (higher temperature) with momentum.
  3. Exact max gradient (sub‑gradient) for fine‑tuning.
Random block perturbations, swaps and occasional jitter restarts
avoid stagnation. The best sequence seen is returned.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT."""
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv

def _weighted_gradient(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """Gradient of 2·n·∑_k w_k·(a*a)[k] with respect to a."""
    n = a_rev.shape[0]
    N = _next_pow2(w.size + a_rev.size - 1)
    fw = np.fft.rfft(w, n=N)
    fa = np.fft.rfft(a_rev, n=N)
    conv = np.fft.irfft(fw * fa, n=N)[: w.size + a_rev.size - 1]
    start = n - 1               # index that aligns with the centre of the product
    grad = 2.0 * conv[start : start + n]
    return grad

def _grad_at_max(k: int, a: np.ndarray) -> np.ndarray:
    """Exact gradient of (a*a)[k] (the max entry) with respect to a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _swap_two_coords(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two entries chosen uniformly at random."""
    n = a.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i != j:
        a[i], a[j] = a[j], a[i]
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimizes
    ``evaluate_sequence`` within the time budget.

    The routine uses an exponentiated‑gradient (mirror‑descent) loop with a
    soft‑max approximation of the max‑convolution, followed by a hard‑max
    refinement phase. Random block jitter, coordinate swaps and occasional
    jitter‑restart help to escape local plateaus.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a working length. Re‑use a previously discovered sequence if
    # available, otherwise start from a uniform vector of size ≈4000.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    raise ValueError
            else:
                raise ValueError
        except Exception:
            n = 4096
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size        # final length

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best)
    best_score = 2.0 * n * conv_best.max()
    start_time = time.monotonic()
    last_improve = start_time

    # hyper‑parameters
    momentum = 0.85
    grad_prev = np.zeros_like(a)

    # Block‑perturb / swap probabilities – they will be slowly annealed.
    block_perturb_prob = 0.05
    swap_prob = 0.07

    # Restart parameters
    restart_gap = 20.0      # seconds without improvement → jitter restart
    jitter_sigma = 0.12

    # Pre‑compute FFT size for weighted‑gradient (soft‑max) phase
    fft_len_grad = _next_pow2(3 * n - 2)   # size needed for w (2n‑1) * a_rev (n)

    # ------------------------------------------------------------------
    # Main optimisation loop – three phases
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Phase boundaries (as fractions of the total runtime)
    total_runtime = deadline - start_time
    phase1_end = start_time + 0.35 * total_runtime
    phase2_end = start_time + 0.75 * total_runtime

    while time.monotonic() < deadline:
        # ---- current convolution and score --------------------------------
        conv = _conv_self(a)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max          # sum(a) == 1 by construction

        # ---- keep the best -------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule ------------------------------------------------------
        now = time.monotonic()
        if now < phase1_end:
            # Phase 1 – gentle soft‑max, relatively large step‑size
            # Linear schedules for temperature β and stepsize η
            frac = (now - start_time) / (phase1_end - start_time)
            beta = 0.1 + (30.0 - 0.1) * frac
            eta = 1.0 - 0.9 * frac          # 1.0 → 0.1
            use_hard_max = False
        elif now < phase2_end:
            # Phase 2 – sharper soft‑max, slower steps
            frac = (now - phase1_end) / (phase2_end - phase1_end)
            beta = 30.0 + (200.0 - 30.0) * frac
            eta = 0.1 - 0.08 * frac         # 0.1 → 0.02
            use_hard_max = False
        else:
            # Phase 3 – exact max gradient (hard sub‑gradient)
            beta = None
            # η continues to decay to a very small value for fine‑tuning
            frac = (now - phase2_end) / (deadline - phase2_end)
            eta = 0.02 - 0.019 * frac       # 0.02 → 0.001
            use_hard_max = True

        # ---- gradient computation -----------------------------------------
        a_rev = a[::-1]

        if use_hard_max:
            # Exact max gradient (sub‑gradient)
            max_idx = int(conv.argmax())
            grad = _grad_at_max(max_idx, a)
        else:
            # Soft‑max weights – stable exponentiation
            shift = conv.max()
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum
            # Weighted gradient via a single FFT
            grad = _weighted_gradient(w, a_rev)

        # ---- momentum smoothing -------------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- mirror‑descent update (exponentiated gradient) ---------------
        # Clip the exponent to avoid overflow/underflow.
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -50.0, 50.0)
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- occasional stochastic moves -----------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            a = _swap_two_coords(a, rng)

        # ---- stagnation restart --------------------------------------------
        if now - last_improve > restart_gap:
            # Jitter the best solution and resume from there
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

        # ---- slowly anneal the perturbation probabilities --------------------
        # Reduce randomness as we approach the final phase.
        if now > phase2_end:
            # linear decay to a small floor
            decay_frac = (now - phase2_end) / (deadline - phase2_end)
            block_perturb_prob = 0.05 * (1.0 - decay_frac) + 0.005
            swap_prob = 0.07 * (1.0 - decay_frac) + 0.003

    # ------------------------------------------------------------------
    # After the main loop we perform a short focused hill‑climb that
    # moves tiny amounts of mass between two coordinates.
    # ------------------------------------------------------------------
    # The amount of time left is tiny – just a few seconds – but we
    # attempt a few thousand cheap proposals.
    remaining = deadline - time.monotonic()
    if remaining > 0.5:
        # Convert to plain Python list for cheap copies.
        a_local = best.copy()
        # Fixed number of attempts (bounded to stay within the budget)
        attempts = int(min(5000, remaining / 0.0004))  # rough 0.4 ms per attempt
        for _ in range(attempts):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # Transfer a small fraction of the larger coordinate to the smaller one.
            if a_local[i] > a_local[j]:
                donor, recv = i, j
            else:
                donor, recv = j, i
            delta = a_local[donor] * rng.uniform(0.0, 0.03)   # up to 3 %
            if delta <= 0.0:
                continue
            a_new = a_local.copy()
            a_new[donor] -= delta
            a_new[recv] += delta
            # evaluate
            conv_new = _conv_self(a_new)
            new_score = 2.0 * n * conv_new.max()
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                a_local = a_new
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] hill‑climb improve = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best solution, clipped to the limits imposed by evaluate_sequence
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

"""
Mirror‑descent + hard‑max refinement for minimizing the maximum
self‑convolution under a simplex (∑a = 1, a_i ≥ 0).

The optimiser runs three phases:
  1. Soft‑max smoothing of the max‑convolution (large temperature,
     multiplicative (exponentiated) update).
  2. Sharper soft‑max (higher temperature) with lower step‑size.
  3. Exact max‑gradient (sub‑gradient) together with an additive
     projected‑gradient step (mirrors Euclidean projection onto the
     probability simplex).

Random block‑jitter, coordinate swaps and occasional jitter‑restarts
prevent premature stagnation. A short focused hill‑climb is performed
after the main loop to fine‑tune the best solution found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT."""
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _weighted_gradient(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """Gradient of 2·n·∑_k w_k·(a*a)[k] w.r.t. a (soft‑max case)."""
    n = a_rev.shape[0]
    N = _next_pow2(w.size + a_rev.size - 1)
    fw = np.fft.rfft(w, n=N)
    fa = np.fft.rfft(a_rev, n=N)
    conv = np.fft.irfft(fw * fa, n=N)[: w.size + a_rev.size - 1]
    start = n - 1                     # centre index of the product
    grad = 2.0 * conv[start : start + n]
    return grad


def _grad_at_max(k: int, a: np.ndarray) -> np.ndarray:
    """Exact gradient of the k‑th convolution entry (the max) w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        # all entries ≤ 0 → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor, then renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _swap_two_coords(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two randomly chosen coordinates."""
    n = a.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i != j:
        a[i], a[j] = a[j], a[i]
    return a

        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

        # ---- stage‑wise annealing of stochastic‑move rates ------------
        if now > phase2_end:
            decay = (now - phase2_end) / (deadline - phase2_end)
            block_perturb_prob = 0.05 * (1.0 - decay) + 0.005
            swap_prob = 0.07 * (1.0 - decay) + 0.003

    # ------------------------------------------------------------------
    # 6) Short focused hill‑climb (tiny mass transfers) after time‑out
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 0.5:
        a_local = best.copy()
        # Rough estimate of how many cheap proposals we can still afford
        max_attempts = int(min(2000, remaining / 0.001))
        for _ in range(max_attempts):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # Move a small fraction of the larger coordinate toward the smaller one
            if a_local[i] > a_local[j]:
                donor, recv = i, j
            else:
                donor, recv = j, i
            delta = a_local[donor] * rng.uniform(0.0, 0.03)   # ≤ 3 %
            if delta <= 0.0:
                continue
            a_new = a_local.copy()
            a_new[donor] -= delta
            a_new[recv] += delta
            conv_new = _conv_self(a_new)
            new_score = 2.0 * n * conv_new.max()
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                a_local = a_new
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] hill‑climb improve = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
```

This implementation follows the described multi‑phase optimisation strategy, includes adaptive schedules, momentum smoothing, stochastic perturbations, jitter‑restarts, and a final hill‑climb. It respects the time budget, returns a non‑negative sequence, and is ready to be called by the evaluation harness.
