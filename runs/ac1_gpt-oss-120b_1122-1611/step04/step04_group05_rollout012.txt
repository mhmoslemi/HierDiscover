"""

2. **Occasional linear‑programming boost** – given the current
   distribution `a` with maximal convolution value `M`,
   we solve the LP

       maximise   1·b
       subject to  conv(a, b) ≤ M ,  b ≥ 0

   The solution `b` has a larger ℓ₁‑norm while respecting the same
   bound `M`.  After normalising `b` the maximal convolution strictly
   decreases, so we replace the current iterate by this `b` whenever it
   yields an improvement.

3. **Random local perturbations** (block scaling, pair‑wise mass
   transfer) and a soft restart when progress stalls.

The blend of a smooth optimisation phase and a global LP “jump’’

All helper functions are top‑level, there are no closures, and the

# ----------------------------------------------------------------------
# FFT helpers

def _fft_len_for_n(n: int) -> int:
    """Smallest power of two able to hold the full linear convolution of two
    vectors of length ``n`` (i.e. length ``2*n-1``)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_lin(v: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution v * w (both real, non‑negative)."""
    fv = np.fft.rfft(v, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(fv * fw, n=fft_len)[: v.size + w.size - 1]
    return conv

# Random perturbations

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random amount of mass from i → j."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.4 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    b = np.clip(b, 0.0, None)
    b /= b.sum()
    return b

def _lp_boost(a: np.ndarray, M: float, rng: np.random.Generator):
    """
    Solve the LP  maximise 1·b  subject to  conv(a, b) ≤ M  and  b ≥ 0.
    Return a new probability vector (sum==1) if it improves the score,
    otherwise return ``None``.
    """
    if not _HAS_SCIENTIFIC_LP:
        return None

    n = a.size
    L = 2 * n - 1

    # Ensure strict positivity to keep the LP bounded.
    eps = 1e-12
    a_pos = np.maximum(a, eps)
    a_pos /= a_pos.sum()

    # Build the (2n‑1) × n constraint matrix.
    # Column i contains a_pos placed from row i to i+n‑1.
    # Sparse representation keeps memory modest.
    data = np.repeat(a_pos, n)
    rows = np.concatenate([np.arange(i, i + n) for i in range(n)])
    cols = np.repeat(np.arange(n), n)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(L, n))

    # Objective: minimise -∑ b   (i.e. maximise ∑ b)
    c = -np.ones(n, dtype=np.float64)

    # Right‑hand side of the inequalities
    b_ub = np.full(L, M, dtype=np.float64)

    # Solve with HiGHS (fast interior‑point)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:  # pragma: no cover
        return None

    if not res.success:
        return None

    b = res.x
    if b.sum() == 0.0:
        return None

    # Normalise and test improvement
    b_norm = b / b.sum()
    fft_len = _fft_len_for_n(n)
    conv_b = _conv_self(b_norm, fft_len)
    new_M = conv_b.max()
    new_score = 2.0 * n * new_M  # sum(b_norm)=1 → denominator =1

    old_score = 2.0 * n * M
    if new_score < old_score * (1.0 - 1e-6):
        return b_norm
    return None

    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = score_a
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # margin
    last_improve = start_time
    last_lp = start_time
    iter_cnt = 0

    # Hyper‑parameters
    eta_start = 0.7
    eta_end = 0.015
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    block_prob = 0.03
    pair_prob = 0.07
    stagnation_secs = 20.0          # restart if no improvement
    lp_cooldown = 30.0              # seconds between LP attempts
    t_lp_mix = 1.0                  # we replace a by the LP solution when it improves

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iter_cnt += 1
        # ----- current convolution & score (sum(a)=1) -----
        fft_len = _fft_len_for_n(n)
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----- track best -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear decay for learning rate
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # ------------------------------------------------------------------
        # Decide whether to attempt a linear‑programming jump
        # ------------------------------------------------------------------
        if (_HAS_SCIENTIFIC_LP and (now - last_lp) > lp_cooldown
                and (now - last_improve) > stagnation_secs):
            new_a = _lp_boost(a, cur_max, rng)
            if new_a is not None:
                # successful LP step – replace the current iterate
                a = new_a
                w = np.log(a + 1e-30)
                m.fill(0.0)
                v.fill(0.0)
                last_lp = now
                # recompute score after the jump
                continue  # skip the gradient step this iteration

        # ------------------------------------------------------------------
        # Exact sub‑gradient of the max‑norm (using argmax mask)
        # ------------------------------------------------------------------
        # mask of active constraints
        mask = (conv >= cur_max - 1e-12).astype(np.float64)

        # gradient w.r.t. a:  g = 2 * (mask * a_rev) convolved, slice n elements
        a_rev = a[::-1]
        L_conv = mask.size + a_rev.size - 1           # = 3n‑2
        fft_len_grad = _fft_len_for_len(L_conv)
        fm = np.fft.rfft(mask, n=fft_len_grad)
        fr = np.fft.rfft(a_rev, n=fft_len_grad)
        full = np.fft.irfft(fm * fr, n=fft_len_grad)[:L_conv]
        grad_a = 2.0 * full[n - 1:n - 1 + n]          # length n

        # gradient w.r.t. softmax logits w
        dot = np.dot(a, grad_a)
        grad_w = a * (grad_a - dot)

        # ------------------------------------------------------------------
        # Adam update on w
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        w -= step

        # recover a from softmax
        w_max = np.max(w)
        a = np.exp(w - w_max)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ------------------------------------------------------------------
        # Occasional random moves (to escape plateaus)
        # ------------------------------------------------------------------
        if rng.random() < block_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < pair_prob:
            a = _pairwise_transfer(a, rng)

        # ------------------------------------------------------------------
        # Soft restart when stuck for a while
        # ------------------------------------------------------------------
        if now - last_improve > stagnation_secs:
            # re‑initialise around the current best with modest jitter
            a = best_a.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            w = np.log(a + 1e-30)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) Return the best vector found (clipped to evaluation limits)
    # ----------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
