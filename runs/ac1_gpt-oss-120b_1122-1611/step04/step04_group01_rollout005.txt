"""Hybrid greedy/multiplicative simplex optimisation for minimising the
self‑convolution L∞‑norm. ...

"""

import time
import numpy as np

def _fft_len_for_len(length: int) -> int:
    """Return the smallest power‑of‑two ≥ length."""
    return 1 << ((length - 1).bit_length())

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT. Result length 2*len(a)-1."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Implementation same as earlier script.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # ------------------------------------------------------------------
    # 5) Return the best vector (apply evaluate_sequence limits) ----------
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Hybrid multiplicative‑gradient + greedy mass‑transfer optimisation.

We minimise   score = 2·n·max(conv(a,a)) / (sum(a)**2)

under the constraints a_i ≥ 0, Σ a_i = 1.  The algorithm works on the
probability simplex:

* a is kept symmetric (a_i = a_{n‑1‑i}) – this reduces the search space
  and matches the symmetry of the objective.
* At each iteration we compute the exact sub‑gradient of the max‑convolution
  (the derivative of the active convolution entry).  A mirror‑descent
  multiplicative update  a ← a·exp(−η·g)  is performed and the vector is
  re‑normalised onto the simplex.
* A cheap greedy move transfers a small amount of mass from the index with
  the largest gradient (most responsible for the peak) to the index with
  the smallest gradient.

  local minima.

                a[src] -= delta
                a[tgt] += delta
                a = (a + a[::-1]) * 0.5
                a = a / a.sum()

        # ----- g) soft restart if stuck for a while -----------------
        if now - last_improve > restart_gap:
            a = rng.gamma(0.5, size=n)
            a = (a + a[::-1]) * 0.5
            a = a / a.sum()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best vector (apply evaluate_sequence limits).
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
