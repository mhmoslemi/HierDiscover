"""
subgradient_maxsearch.py
~~~~~~~~~~~~~~~~~~~~~~~~

A simple yet effective sub‑gradient search for the step‑function problem.

The goal is to minimise

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the ordinary linear convolution.
Because the denominator is homogeneous, we fix ``sum(a) = 1`` and
optimise the *maximum* entry of the convolution directly.

Algorithm
---------

* keep the current candidate on the probability simplex (non‑negative,
  unit sum) using the Euclidean projection,
* each iteration:
    – compute the full self‑convolution ``c = a * a`` with an FFT,
    – locate the index/indices where ``c`` attains its maximum,
    – build the exact sub‑gradient of the (non‑smooth) objective:
          ∂ max(c) / ∂ a_j = 2 * a_{k‑j}   (if 0 ≤ k‑j < n, otherwise 0)
      for every maximal shift ``k``; the final gradient is the average
      of these contributions,
    – perform an Adam‑style update (additive) and re‑project onto the
      simplex,
    – occasionally apply cheap stochastic moves (random block
      multiplicative perturbations and pairwise mass transfers) to
      escape plateaux,
    – if no improvement is seen for a few seconds, restart from the
      current best with a small log‑normal jitter.
* the loop runs until the allotted time budget expires; the best
  candidate seen is returned.

The method differs from the previously supplied soft‑max/β‑schedule
approach: it uses the true sub‑gradient of the max‑norm (no smoothing)
and a lightweight population‑free scheme, which in practice reaches
lower scores on the same time budget.

All helper routines are defined at module level, have no closures,
and use only ``numpy`` and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors whose total length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution ``a * a`` using a real FFT.
    Returns the full convolution of length ``2*len(a)‑1``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, ∑ x = 1}.  Returns a new array."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the supplied time budget elapses.

    The search is based on a projected Adam optimiser that uses the exact
    sub‑gradient of the max‑convolution objective, enriched by occasional
    random perturbations and a simple restart strategy.
    """
    rng = np.random.default_rng(seed)

    # small safety margin for the harness (a few seconds before hard limit)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a problem size
    # ------------------------------------------------------------------
    # The objective is scale‑invariant in n, but a larger n gives a finer
    # granularity for the convolution.  3500‑4000 is a good compromise
    # between quality and speed (the previous best used 3809).
    n = kwargs.get("n", 3809)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for the self‑convolution
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_len(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Book‑keeping for the current best solution
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    last_improve = time.monotonic()
    start_time = last_improve

    # ------------------------------------------------------------------
    # 4) Adam hyper‑parameters (additive version)
    # ------------------------------------------------------------------
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    # ------------------------------------------------------------------
    # 5) Miscellaneous probabilities and restart handling
    # ------------------------------------------------------------------
    block_perturb_prob = 0.02
    pair_move_prob = 0.07
    restart_gap = 30.0                     # seconds without improvement → restart
    momentum = 0.85                         # simple exponential smoothing of raw grad

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 6.1) Initialise a new candidate (restart or random) ----------
        if best is None:
            # First iteration – try to use the provided construction if any
            if "height_sequence_1" in globals():
                try:
                    cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
                    if cand.ndim == 1 and np.isfinite(cand).all():
                        # Truncate / pad to the working size n
                        if cand.size > n:
                            a = cand[:n]
                        else:
                            a = np.concatenate([cand, np.zeros(n - cand.size)])
                        a = np.clip(a, 0.0, None)
                        if a.sum() == 0.0:
                            a = np.full(n, 1.0 / n)
                        else:
                            a = a / a.sum()
                    else:
                        raise ValueError
                except Exception:
                    a = np.full(n, 1.0 / n)
            else:
                a = np.full(n, 1.0 / n)
        else:
            # Restart from the current best with a tiny jitter
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * jitter
            a = _project_onto_simplex(a)

        # Reset Adam statistics for the new run
        m.fill(0.0)
        v.fill(0.0)
        t = 0
        grad_prev = np.zeros_like(a)

        # ------------------------------------------------------------------
        # 6.2) Inner optimisation loop for this restart
        # ------------------------------------------------------------------
        while time.monotonic() < deadline:
            # ----- Compute convolution and current score -----------------
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max          # sum(a) == 1 → denominator = 1

            # ----- Keep track of the best vector seen so far ------------
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

            # ----- Stop inner loop if we are about to exceed the budget --
            if time.monotonic() >= deadline:
                break

            # ----- Sub‑gradient of the max‑convolution -------------------
            # Identify all indices attaining the maximum (within a tiny tolerance)
            eps_max = 1e-15
            max_inds = np.where(conv >= cur_max - eps_max)[0]

            # Build the average sub‑gradient for the active indices
            grad = np.zeros_like(a)
            for k in max_inds:
                # vector of positions j where a_{k-j} is defined
                idx = k - np.arange(n)
                mask = (idx >= 0) & (idx < n)
                grad[mask] += 2.0 * a[idx[mask]]
            if len(max_inds) > 0:
                grad /= len(max_inds)

            # ----- Optional smoothing of the raw gradient ---------------
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # ----- Adam update (additive) --------------------------------
            t += 1
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad ** 2)
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)

            # Learning‑rate schedule: start relatively large, decay to a tiny value
            elapsed = time.monotonic() - start_time
            total = deadline - start_time
            eta_start = 0.7
            eta_end = 0.002
            eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

            step = eta * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _project_onto_simplex(a)          # stay on the probability simplex

            # ----- Random block perturbation (occasionally) -------------
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # ----- Pair‑wise mass transfer (cheap local search) ----------
            if rng.random() < pair_move_prob:
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 0.0:
                    max_delta = min(a[i], 0.3 * a[i])
                    if max_delta > 0.0:
                        delta = rng.random() * max_delta
                        a_candidate = a.copy()
                        a_candidate[i] -= delta
                        a_candidate[j] += delta
                        a_candidate = _project_onto_simplex(a_candidate)
                        conv_cand = _conv_self(a_candidate, fft_len)
                        score_cand = 2.0 * n * conv_cand.max()
                        if score_cand < cur_score:
                            a = a_candidate
                            # keep cheap auxiliary values up‑to‑date
                            conv = conv_cand
                            cur_max = conv_cand.max()
                            cur_score = score_cand

            # ----- Restart if we have stagnated for a while -------------
            now = time.monotonic()
            if now - last_improve > restart_gap:
                # break the inner loop – the outer while will re‑initialise from best
                break

    # ------------------------------------------------------------------
    # 7) Return the best vector (clipping per evaluate_sequence limits)
    # ------------------------------------------------------------------
    if best is None:
        # fallback – uniform distribution (should never happen)
        best = np.full(n, 1.0 / n)

    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
