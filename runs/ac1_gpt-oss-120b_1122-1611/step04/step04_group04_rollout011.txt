A fast Adam‑based optimiser on the soft‑max (logits) parametrisation of the
step‑function.  The objective 2·n·max(conv)/(∑a)² is invariant to scaling,
so we keep the coefficients on the probability simplex.
The algorithm uses a smooth‑max (soft‑max) approximation that slowly hardens
to the true max, computes the exact gradient of the weighted convolution
via a single FFT, and updates the logits with Adam.  Random “jitter”
perturbations, cheap pairwise mass‑transfer moves and an adaptive restart
strategy help to escape plateaux.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # remove tiny negatives
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. a.
    Formula:   ∂/∂a_j Σ_k w_k·conv_k = 2· Σ_i a_i·w_{i+j}
    Implemented via a single FFT‑based correlation.
    """
    n = a.size
    Lw = w.size                     # = 2·n‑1
    wr = w[::-1]                    # reverse for correlation

    # Length needed for a (len=n) + wr (len=Lw) – 1 = n+Lw‑1 = 3·n‑2
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep necessary part

    idx = Lw - 1 - np.arange(n)          # index mapping from derivation
    grad = 2.0 * conv_full[idx]
    return grad


def _pairwise_transfer(a: np.ndarray, conv: np.ndarray,
                       rng: np.random.Generator):
    """
    Cheap mass‑transfer move.  Pick two indices i≠j, move a small amount
    δ from i to j, update the convolution linearly (ignoring O(δ²)),
    and accept the move if it lowers the objective.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # linear update of the convolution (first‑order)
    new_conv = conv.copy()
    lo_i, hi_i = i, i + n
    lo_j, hi_j = j, j + n
    new_conv[lo_i:hi_i] -= delta * a
    new_conv[lo_j:hi_j] += delta * a

    new_score = 2.0 * n * np.max(new_conv)
    cur_score = 2.0 * n * np.max(conv)

    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.clip(a2, 0.0, None)
        a2 /= a2.sum()
        return a2, new_conv, new_score
    else:
        return a, conv, None


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(logits)
    ex = np.exp(logits - m)
    s = ex.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return ex / s


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  Returns the best list found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size & initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full(n, 1.0 / n)
        else:
            n = 4096
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # initialize logits (log‑softmax parameters)
    logits = np.log(a + 1e-12)

    # FFT length for self‑convolution
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Adam bookkeeping
    # ------------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iter_cnt = 0

    # best solution seen so far
    conv_best = _self_conv_fft(a, fft_len_self)
    best_score = 2.0 * n * conv_best.max()
    best = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (tuned for the 1000 s budget)
    # ------------------------------------------------------------------
    lr_start = 0.20          # initial learning rate for logits
    lr_end = 0.001           # final learning rate
    beta_start = 0.5         # start temperature for soft‑max
    beta_end = 2000.0        # final temperature – almost hard max
    hard_max_phase = 0.85    # switch to exact max near the end
    jitter_prob = 0.025      # chance to add gaussian jitter to logits
    restart_gap = 30.0       # seconds without progress → restart
    transfer_prob = 0.07     # probability of trying a pairwise transfer

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution & score -----
        conv = _self_conv_fft(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- record best -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ----- schedule -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        lr = lr_start * (1.0 - frac) + lr_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- weight vector (smooth max) -----
        if frac < hard_max_phase:
            # soft‑max on the whole convolution vector
            w = np.exp(beta * (conv - cur_max))          # max exponent = 0
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0                     # exact sub‑gradient

        # ----- gradient w.r.t. a -----
        grad_a = _gradient_from_weights(a, w)

        # ----- transform gradient to logits (softmax Jacobian) -----
        mean_grad = np.dot(grad_a, a)                   # ⟨grad_a , a⟩
        grad_log = a * (grad_a - mean_grad)             # ∂L/∂z_i

        # ----- Adam update for logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        iter_cnt += 1
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- optionally add jitter ----- 
        if rng.random() < jitter_prob:
            logits += rng.normal(0.0, 0.02, size=n)

        # ----- map back to probability simplex -----
        a = _softmax(logits)

        # ----- cheap pairwise mass transfer move -----
        if rng.random() < transfer_prob:
            a, conv, _ = _pairwise_transfer(a, conv, rng)
            # after the move, keep logits consistent with the new a
            logits = np.log(a + 1e-12)

        # ----- restart if stagnated -----
        if now - last_improve > restart_gap:
            # re‑initialise around the current best with a moderate noise burst
            noise = rng.normal(0.0, 0.5, size=n)
            logits = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
