def propose_candidate(seed=42, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq = None
    best_score = np.inf

    # Param search
    n_options = [800, 1000, 1200, 1500, 2000]
    p_options = [0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]

    for n in n_options:
        i = np.arange(n, dtype=np.float64)
        for p in p_options:
            a = ((i+0.5)*(n - i - 0.5))**(-p)
            a = np.clip(a, 0, None)
            s = a.sum()
            if s == 0:
                continue
            a = a / s
            conv = _conv_self(a, _fft_len_for_n(n))
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max / (a.sum()**2)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f'Init n={n} p={p:.3f} score={cur_score:.6f}')

            # early break if time near deadline
            if time.monotonic() > deadline:
                break
        if time.monotonic() > deadline:
            break

    # If no best found due to early exit, fallback uniform:
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0/n)

    # Optimization refinement: Mirror descent with subgradient
    a = best_seq.copy()
    n = len(a)
    fft_len = _fft_len_for_n(n)
    # parameters
    eta_start = 1.0
    eta_end = 0.01
    momentum = 0.9
    grad_prev = np.zeros_like(a)

    # main loop until deadline
    while time.monotonic() < deadline:
        conv = _conv_self(a, fft_len)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] new best {best_score:.6f}")
        # compute gradient for active maxima
        tol = 1e-12 * max_val
        active = np.where(conv >= max_val - tol)[0]
        # binary mask
        w = np.zeros_like(conv)
        w[active] = 1.0
        # convolve w with reversed a to get gradient contributions
        a_rev = a[::-1]
        L = (2*n-1) + n - 1
        fft_len_g = _fft_len_for_len(L)
        fw = np.fft.rfft(w, n=fft_len_g)
        fa = np.fft.rfft(a_rev, n=fft_len_g)
        grad_full = np.fft.irfft(fw * fa, n=fft_len_g)
        grad = 2.0 * grad_full[n-1 : n-1 + n] / len(active)  # average over active indices
        # momentum smoothing
        grad = momentum * grad_prev + (1 - momentum) * grad
        grad_prev = grad

        # schedule step size
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # exponentiated gradient update
        factor = np.exp(-eta * grad)
        a = a * factor
        # re-normalize
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional random perturbation
        if rng.random() < 0.02:
            # select random block and perturb
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            factor_block = np.exp(rng.normal(0.0, 0.2))
            a[start:start+width] *= factor_block
            a = np.clip(a, 0, None)
            a /= a.sum()

        # check time
        if now - start_time > deadline - start_time - 0.1:
            break

    # final clip
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

"""
Propose a step‑function (list of non‑negative heights) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear self‑convolution.
The algorithm consists of three phases:

1. **Broad parametric sweep** – we generate a variety of symmetric
   “beta‑type” shapes a_i ∝ ((i+½)*(n‑i‑½))^(−p) for several lengths *n*
   and exponents *p*.  Classic windows (triangular, Hann, Hamming,
   Blackman) are also tried.  The best candidate from this cheap
   exploration is kept.

2. **Linear‑programming improvement** – for the current best vector *f*
   we solve a small linear programme that maximises ∑ g_j subject to
   g≥0 and (f∗g) ≤ M where M = max(f∗f).  The solution *g* is mixed with
   *f* (convex combination) which provably does not increase the
   ``max(conv)`` value.  A few such LP steps are performed while time
   permits.

3. **Mirror‑descent (exponentiated‑gradient) refinement** – we treat
   the objective as 2·n·max(conv) (the denominator is 1 after normalising
   the vector).  The sub‑gradient of the max‑norm is computed exactly by
   convolving the indicator of the active indices with the reversed
   vector.  An exponentiated‑gradient update with a linearly decreasing
   step‑size, momentum smoothing and occasional random perturbations
   pushes the solution towards a lower peak.  The best vector seen is
   returned.

All steps respect the 1000 s wall‑time limit; intermediate progress is
printed and the final sequence is clipped to the limits used by
``evaluate_sequence`` (0 ≤ x ≤ 1000).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT helper utilities
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Shape generators (all symmetric and sum‑normalised)
# ----------------------------------------------------------------------


def _beta_shape(n: int, p: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ ((i+½)*(n‑i‑½))^(−p).  p=0.5 reproduces the
    classic arcsine law."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - center) / center
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


def _hann_shape(n: int) -> np.ndarray:
    """Hann (sin²) window."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * i / (n - 1)) ** 2
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


def _hamming_shape(n: int) -> np.ndarray:
    """Hamming window."""
    i = np.arange(n, dtype=np.float64)
    vals = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


def _blackman_shape(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    vals = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Scoring helper (the same quantity that ``evaluate_sequence`` returns)
# ----------------------------------------------------------------------


def _score_of(a: np.ndarray) -> float:
    """Return the evaluation score of a normalised (sum>0) vector."""
    n = a.size
    conv = _conv_self(a, _fft_len_for_n(n))
    cur_max = conv.max()
    return float(2.0 * n * cur_max / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def _lp_improve(f: np.ndarray, mix: float = 0.2) -> np.ndarray:
    """
    Perform a single LP improvement of the current vector ``f``.
    We maximise ``∑ g_j`` subject to g≥0 and (f∗g) ≤ M,
    where M = max(f∗f).  The returned vector is the convex mixture
    (1‑mix)·f + mix·g_norm (g_norm has unit sum).
    """
    n = f.size
    # current maximal convolution value
    conv_f = _conv_self(f, _fft_len_for_n(n))
    M = conv_f.max()

    # Build the dense convolution matrix A (size (2n‑1) × n) such that
    # (A @ g)[k] = (f ∗ g)_k.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for col in range(n):
        A[col : col + n, col] = f  # shifted copy of f

    # Solve  max 1ᵀ g   ⇔   min -1ᵀ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # ``highs`` works with dense matrices and is reasonably fast for our sizes.
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return f  # fall back to the original vector

    g = res.x
    if g.sum() == 0.0:
        return f

    g_norm = g / g.sum()
    new_f = (1.0 - mix) * f + mix * g_norm
    new_f = np.clip(new_f, 0.0, None)
    new_f /= new_f.sum()
    return new_f


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.

    The algorithm explores a handful of parametric families, refines the
    best candidate with a linear‑programming step, and finally applies a
    mirror‑descent (exponentiated‑gradient) optimisation on the simplex.
    Progress is printed to stdout.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1)  Initial sweep over several analytic families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    def consider(candidate: np.ndarray):
        """Normalise, evaluate and keep the best candidate seen so far."""
        nonlocal best_seq, best_score
        if candidate.size == 0:
            return
        s = candidate.sum()
        if s == 0.0:
            return
        candidate = candidate / s
        sc = _score_of(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # --- beta‑type family (arcsine is p=0.5) -------------------------
    n_candidates = [800, 1000, 1200, 1500, 1800, 2000]
    p_vals = np.linspace(0.35, 0.70, 8)                # 0.5 is the classic case
    for n in n_candidates:
        if time.monotonic() >= deadline:
            break
        for p in p_vals:
            a = _beta_shape(n, p)
            consider(a)
            if time.monotonic() >= deadline:
                break

    # --- classic windows ------------------------------------------------
    window_funcs = [_triangular_shape, _hann_shape, _hamming_shape, _blackman_shape]
    for func in window_funcs:
        for n in n_candidates:
            if time.monotonic() >= deadline:
                break
            a = func(n)
            consider(a)

    # If nothing was found (should never happen) fall back to uniform.
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score_of(best_seq)

    # ------------------------------------------------------------------
    # 2) Linear‑programming improvement (few iterations, quick)
    # ------------------------------------------------------------------
    f = best_seq.copy()
    lp_iters = 0
    while time.monotonic() < deadline and lp_iters < 5:
        new_f = _lp_improve(f, mix=0.2)
        new_score = _score_of(new_f)
        if new_score < best_score - 1e-6:
            best_score = new_score
            best_seq = new_f.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] LP improvement, new best = {best_score:.6f}")
        # accept the improvement for the next LP step
        f = new_f
        lp_iters += 1

    # ------------------------------------------------------------------
    # 3) Mirror‑descent (exponentiated‑gradient) refinement
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.size
    fft_len = _fft_len_for_n(n)

    # hyper‑parameters
    eta_start = 1.0           # initial stepsize (aggressive)
    eta_end = 0.02            # final stepsize
    momentum = 0.9
    grad_prev = np.zeros_like(a)
    last_improve = time.monotonic()
    restart_gap = 25.0        # seconds without improvement → soft restart

    while time.monotonic() < deadline:
        # ----- current convolution and score --------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)

        # ----- keep the best solution ---------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] refinement best = {best_score:.6f}")

        # ----- schedule for the stepsize ------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # ----- sub‑gradient of the max‑norm (exact, no smoothing) ----------
        max_val = conv.max()
        # active set (indices where conv is within a tiny tolerance of the max)
        tol = max(1e-12, 1e-14 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        # binary mask for active rows
        w = np.zeros_like(conv)
        w[active] = 1.0

        # gradient = average of 2 * (w * a_rev) convolution
        a_rev = a[::-1]
        L = (2 * n - 1) + n - 1
        fft_len_g = _fft_len_for_len(L)
        fw = np.fft.rfft(w, n=fft_len_g)
        fa = np.fft.rfft(a_rev, n=fft_len_g)
        grad_full = np.fft.irfft(fw * fa, n=fft_len_g)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n] / max(1, len(active))

        # ----- momentum smoothing -------------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient update ----------------------------------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation -------------------------
        if rng.random() < 0.02:
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a[start : start + width] *= factor
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # ----- soft restart if stuck ----------------------------------------
        if now - last_improve > restart_gap:
            a = best_seq.copy()
            jitter = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev[:] = 0.0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
