    rng = np.random.default_rng(seed)

    # determine n
    default_n = 1024 # smaller for LP speed

    # load previous sequence if available; if longer, truncate or if shorter, pad with uniform?
    if "height_sequence_1" in globals():

        if cand.ndim==1 and cand.size>0:
            if cand.size != default_n:
                # downsample to default_n by averaging blocks
                # simple resizing: compute using interpolation.
                # We'll use linear interpolation with np.interp at equally spaced points.
                x_old = np.linspace(0,1,cand.size,endpoint=False)
                x_new = np.linspace(0,1,default_n,endpoint=False)
                a = np.interp(x_new, x_old, cand)
                a = np.clip(a,0,None)
            else:
                a = np.clip(cand,0,None)
            # Normalize
            s = a.sum()
            if s>0:
                a = a / s
            else:
                a = np.full_like(a, 1.0/a.size)
        else:
            a = np.full(default_n, 1.0/default_n)
    else:
        a = np.full(default_n, 1.0/default_n)

    n = a.size
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"Initial score {best_score}")

    deadline = time.monotonic() + max(0.0, budget_s - 5.0)

    # parameters
    alpha = 0.5   # mixing coefficient after LP improvement
    min_alpha = 0.05
    decay = 0.99  # decay alpha over iterations
    iteration = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:

"""propose_candidate
-------------------

An alternative to pure mirror‑descent.  The algorithm repeatedly solves a
tiny linear programme

    minimise   t
    subject to  (f * g)[k] ≤ t   for all k
                Σ g_j = 1
                g_j ≥ 0

for the current step‑function *f*.  The optimal *g* is the best
candidate that can be paired with *f* without exceeding the current
maximum convolution value.  If *g* improves the self‑convolution of
*f* we replace *f* by *g* (or a convex combination) and continue.
The LP is tiny (≈2 n constraints, n variables) and solved with
SciPy’s HiGHS interior‑point method – a few hundred solves easily fit
into the 1000 s budget.  Periodic random jitter prevents stagnation.
The best sequence seen is returned.

The routine works for any length; if a previous solution
`height_sequence_1` is present we down‑sample it to a manageable size
(≈1024) and continue the optimisation.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution of a real vector using FFT (fast for
    moderate lengths).  Returns an array of length 2*len(a)-1.
    """
    n = a.size
    # Use power‑of‑two size for speed
    size = 2 * n - 1
    fft_len = 1 << ((size - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[:size]
    return conv


def _downsample(seq: np.ndarray, target_len: int) -> np.ndarray:
    """
    Simple linear interpolation down‑sampling from `len(seq)` to
    `target_len`.  The result is non‑negative and sums to 1.
    """
    if seq.size == target_len:
        out = seq.copy()
    else:
        x_old = np.linspace(0.0, 1.0, seq.size, endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_len, endpoint=False)
        out = np.interp(x_new, x_old, seq)
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / target_len
    else:
        out /= s
    return out


def _build_A_ub(a: np.ndarray) -> np.ndarray:
    """
    Build the inequality matrix for the LP:

        (f * g)[k] - t ≤ 0   for k = 0 … 2n‑2

    Variables are [g_0 … g_{n‑1} , t] (t is the last column).
    """
    n = a.size
    # Toeplitz matrix with first column a padded with zeros, first row [a0,0,…,0]
    c = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    r = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    T = toeplitz(c, r)               # shape (2n‑1, n)
    A = np.hstack([T, -np.ones((2 * n - 1, 1), dtype=a.dtype)])
    return A


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Find a non‑negative step‑function (list of heights) that minimises

        score = 2 * n * max(conv(a, a)) / (sum(a)**2)

    using an alternating linear‑programming / convex‑combination scheme.
    The function respects the 1000 s time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise a candidate
    # ------------------------------------------------------------------
    DEFAULT_N = 1024                     # comfortable size for the LP
    if "height_sequence_1" in globals():
        try:
            raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if raw.ndim == 1 and raw.size > 0 and np.isfinite(raw).all():
                a = _downsample(raw, DEFAULT_N)
            else:
                raise ValueError
        except Exception:
            a = np.full(DEFAULT_N, 1.0 / DEFAULT_N, dtype=np.float64)
    else:
        a = np.full(DEFAULT_N, 1.0 / DEFAULT_N, dtype=np.float64)

    n = a.size
    best = a.copy()
    best_score = float(2 * n * _conv_self(best).max())   # exact since sum=1
    cur_score = best_score
    print(f"[init] length={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Timing
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start = time.monotonic()
    last_improve = start
    iter_no = 0

    # also keep a small mixing schedule for when replacement does not help
    mix_alphas = [0.8, 0.5, 0.3, 0.1]

    while time.monotonic() < deadline:
        iter_no += 1
        # --------------------------------------------------------------
        # 2a) Build the LP that tries to push the max‐convolution down
        # --------------------------------------------------------------
        A_ub = _build_A_ub(a)                     # (2n‑1) × (n+1)
        b_ub = np.zeros(2 * n - 1, dtype=np.float64)

        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0                         # Σ g_j = 1
        b_eq = np.array([1.0], dtype=np.float64)

        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[-1] = 1.0                           # minimise t

        # bounds for all variables (g_j ≥ 0, t ≥ 0)
        bounds = [(0.0, None)] * (n + 1)

        # --------------------------------------------------------------
        # 2b) Solve the LP
        # --------------------------------------------------------------
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})

        if not res.success:
            # LP failed – fall back to a tiny random perturbation
            noise = np.exp(rng.normal(0.0, 0.15, size=n))
            a = a * noise
            a /= a.sum()
            continue

        g = res.x[:n]          # optimal partner
        t_opt = res.x[-1]      # the achieved upper bound on (a * g)

        # --------------------------------------------------------------
        # 2c) Evaluate the candidate g on its *own* self‑convolution
        # --------------------------------------------------------------
        conv_g = _conv_self(g)
        g_score = float(2 * n * conv_g.max())    # sum(g) = 1 by construction

        # --------------------------------------------------------------
        # 2d) Accept improvements (direct replacement or convex combo)
        # --------------------------------------------------------------
        improved = False

        if g_score < cur_score - 1e-9:
            # pure replacement works
            a = g
            cur_score = g_score
            improved = True
        else:
            # try convex combinations with a few preset alphas
            for alpha in mix_alphas:
                cand = (1.0 - alpha) * a + alpha * g
                cand = np.clip(cand, 0.0, None)
                cand /= cand.sum()
                cand_score = float(2 * n * _conv_self(cand).max())
                if cand_score < cur_score - 1e-9:
                    a = cand
                    cur_score = cand_score
                    improved = True
                    break

        # --------------------------------------------------------------
        # 2e) Record the best sequence seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s]  new best (est.) score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2f) Stagnation handling – occasional jitter around the best
        # --------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > 30.0:
            # jitter the current best to escape a flat region
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            print(f"[{now - start:6.2f}s]  restart after idle period (score={best_score:.6f})")

        # periodic status
        if iter_no % 20 == 0:
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] iter={iter_no}  cur_score={cur_score:.6f}  best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Return the best sequence (safeguard clipping)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # final exact evaluation for reporting (not needed for return)
    final_score = evaluate_sequence(best.tolist())
    print(f"[final] best estimated score = {best_score:.6f}, exact = {final_score:.6f}")
    return [float(x) for x in best]
