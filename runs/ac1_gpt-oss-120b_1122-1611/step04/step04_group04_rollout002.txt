"""
Hybrid Adam‑gradient + periodic LP refinement.

The objective of evaluate_sequence() for a normalized vector (sum = 1) is
    score = 2·n·max(a*a) .
We minimise the L∞‑norm of the self‑convolution.  The algorithm works on the
simplex by exponentiated‑gradient (Adam style) updates of a smooth
approximation of the max (soft‑max with a temperature β).  Every few seconds
a tiny linear programme is solved:

    maximise   ∑ g_j
    subject to a⋆g ≤ M   (M = current max of a*a)
               g ≥ 0

Only the most‑active convolution constraints (the K largest entries of a*a)
are kept, keeping the LP tiny (≈K·n variables).  The optimiser mixes the LP
solution back into the current iterate (mixing factor t).  Random block
perturbations and occasional restarts avoid stagnation.

The routine respects the time budget (default 1000 s) and returns the best
sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT. Returns length 2·len(a)-1 array."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)      # clip very small negatives
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Uses a single FFT to compute the required cross‑correlation.
    """
    n = a.size
    Lw = w.size                         # = 2·n−1
    wr = w[::-1]                       # reverse for correlation

    conv_len = _next_pow2(n + Lw - 1)   # ≥ n+Lw−1 = 3·n‑2

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep needed part

    # gradient_j = 2 * conv_full[Lw‑1‑j]
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


# ----------------------------------------------------------------------
# Random block perturbation (multiplicative log‑normal jitter)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = a.size
    max_width = max(1, n // 3)          # allow up to ~33 % of the vector
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.2))   # log‑normal factor

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Tiny LP that tries to increase total mass while keeping the convolution
# bound M (current max) satisfied.  Only the K largest entries of the
# self‑convolution are kept as constraints.
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray,
               conv: np.ndarray,
               M: float,
               max_constraints: int,
               rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve
        maximise   ∑ g_j
        subject to a⋆g ≤ M   (using only the K most active constraints)
                   g ≥ 0
    Returns a normalized vector g (sum = 1) or None if the LP fails.
    """
    n = a.size
    # pick the K constraints with largest convolution values
    idx_sorted = np.argsort(-conv)          # descending order
    K = min(max_constraints, len(idx_sorted))
    active_idx = idx_sorted[:K]

    # Build sparse matrix A (K × n) for a⋆g ≤ M
    rows = []
    cols = []
    data = []
    for row_id, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # column indices i = lo … hi
        cols.extend(range(lo, hi + 1))
        rows.extend([row_id] * (hi - lo + 1))
        data.extend([a[i] for i in range(lo, hi + 1)])

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
    b = np.full(K, M)

    # Linear programme (maximise sum(g) <=> minimise -sum(g))
    c = -np.ones(n)                         # objective vector
    res = scipy.optimize.linprog(c,
                                 A_ub=A,
                                 b_ub=b,
                                 bounds=[(0, None)] * n,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s == 0.0:
        return None
    g /= s                                   # normalise to sum = 1
    return g

            last_improve = time.monotonic()
            elapsed = last_improve - start_time

        # ----- time‑dependent schedules -----
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)   # fraction of total budget

        eta = eta_start * (1.0 - frac) + eta_end * frac          # learning‑rate schedule
        beta = beta_start + (beta_end - beta_start) * frac      # temperature

                # evaluate candidate (fast via conv)
                conv_cand = _self_conv_fft(a_candidate, fft_len_self)
                cand_score = 2.0 * n * conv_cand.max()
                if cand_score < best_score:
                    best_score = cand_score
                    best = a_candidate.copy()
                    last_improve = now
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] LP improve → {best_score:.6f}")
                # accept candidate if it improves the current point
                if cand_score < cur_score:
                    a = a_candidate

        # ----- restart if stuck too long -----
        if now - last_improve > restart_gap:
            # restart from the best seen, add a small random perturbation
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.15, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart → keep best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
