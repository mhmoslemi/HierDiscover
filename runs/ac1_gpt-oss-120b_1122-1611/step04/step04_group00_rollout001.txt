"""
Proposes a step‑function (non‑negative vector) that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1.   Quick baselines (global solution, uniform vectors).
2.   Exploration of a parametric “arcsine” family
     a_i ∝ (x_i·(1−x_i))^{−p}  with  x_i = (i+0.5)/n   and p∈[0.15,0.5].
     This family is known to give an almost flat self‑convolution,
     i.e. a near‑uniform distribution of the sum of two independent copies.
3.   A short Adam‑style projected gradient descent that refines the
     best candidate found in (2).  The objective is smoothed with a
     soft‑max (log‑sum‑exp) whose temperature β grows from 0.5 to ≈5 000.

All vectors are projected onto the probability simplex after each update.
The routine respects the 1000 s time budget and returns the best sequence
discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ (n‑1).  Used for linear convolutions."""
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative vector *a* using FFT.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                          fft_len: int) -> np.ndarray:
    """
    Gradient of  Σ_i w_i * (a * a)_i  w.r.t. *a*.
    Equivalent to 2·(w ⊛ rev(a)) restricted to the original length.
    """
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    # Linear convolution of w (length 2n‑1) with rev_a (length n)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point
# ----------------------------------------------------------------------

            # ----- schedules -----
