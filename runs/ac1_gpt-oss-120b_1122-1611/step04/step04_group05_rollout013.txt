"""
propose_candidate
-----------------

Search for a non‑negative step function (list of heights) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear self‑convolution.
The method combines

* a cheap parametric “arcsine‑like’’ start,
* a continuous‑relaxation of the max‑norm via a soft‑max (log‑sum‑exp) of
  all convolution entries,
* an Adam‑style exponentiated‑gradient update on the probability simplex,
* enforced mirror symmetry (the optimal shape is symmetric),
* occasional random block perturbations and pair‑wise mass transfers,
* a simple restart‑when‑stalled strategy.

All operations are performed with FFTs, so each iteration is O(n log n) and
hundreds of thousands of iterations fit comfortably into the 1000 s budget.
The best vector seen is returned (clipped to the limits of ``evaluate_sequence``).

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``.
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _symmetrize(x: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry and renormalise to unit sum."""
    y = (x + x[::-1]) * 0.5
    s = y.sum()
    if s == 0.0:
        y[:] = 1.0 / y.size
    else:
        y /= s
    return y


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (non‑negative, sum=1)."""
    # implementation of the algorithm by Wang & Carreira‑Perpiñán (2013)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    # block width between 1 and 10 % of the vector length (at least 1)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_to_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.

    Parameters
    ----------
    seed : int
        Seed for the RNG (deterministic runs).
    budget_s : float
        Wall‑time budget in seconds (the function will stop a few seconds
        before the limit to avoid hard termination).

    Returns
    -------
    list[float]
        The best step‑function found (already normalised to unit sum; the
        ``evaluate_sequence`` utility will clip values to the allowed range).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose length and an initial distribution
    # ------------------------------------------------------------------
    default_n = 4000  # good trade‑off between quality and speed

    # If a previous construction is available, use its length.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = np.clip(cand, 0.0, None)
                a = _project_to_simplex(a)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Build a few simple parametric starts and keep the best of them
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    def score_of(vec):
        conv = _conv_self(vec, fft_len)
        return 2.0 * n * conv.max()  # sum(vec) == 1 → denominator = 1

    best_local = a.copy()
    best_score = score_of(best_local)

    # arcsine‑type start (a_i ∝ 1 / sqrt((i+δ)(n‑i‑δ)))
    i = np.arange(n, dtype=np.float64)
    delta = 0.5
    arcsine = 1.0 / np.sqrt((i + delta) * (n - i - delta))
    arcsine = _project_to_simplex(arcsine)
    arcsine = _symmetrize(arcsine)
    s_arc = score_of(arcsine)
    if s_arc < best_score:
        best_local, best_score = arcsine.copy(), s_arc

    # simple power‑law families a_i ∝ ((i+δ)*(n‑i‑δ))^{-p}
    # we sweep a modest grid around the arcsine exponent p=0.5
    p_vals = np.linspace(0.35, 0.65, 7)          # 0.35, 0.40, … 0.65
    delta_vals = np.linspace(0.2, 1.5, 7)        # 0.2 … 1.5
    for p in p_vals:
        for d in delta_vals:
            shape = ((i + d) * (n - i - d)) ** (-p)
            shape = _project_to_simplex(shape)
            shape = _symmetrize(shape)
            sc = score_of(shape)
            if sc < best_score:
                best_local, best_score = shape.copy(), sc

    # mixture of uniform and arcsine (λ·arcsine + (1‑λ)·uniform)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * arcsine + (1.0 - lam) * uniform
        mix = _project_to_simplex(mix)
        mix = _symmetrize(mix)
        sc = score_of(mix)
        if sc < best_score:
            best_local, best_score = mix.copy(), sc

    # start from the best of the above
    a = best_local.copy()
    conv_a = _conv_self(a, fft_len)
    cur_score = 2.0 * n * conv_a.max()

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    # stop a little early to guarantee clean exit
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 0.5      # initial step‑size (log‑space)
    eta_end = 0.02       # final step‑size
    beta_start = 0.5     # soft‑max temperature start
    beta_end = 5000.0    # very sharp max approximation at the end
    block_perturb_prob = 0.02
    pair_move_prob = 0.06
    restart_gap = 60.0   # seconds without improvement → soft restart
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)          # first moment (Adam)
    v = np.zeros_like(a)          # second moment (Adam)
    it = 0                        # iteration counter

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # -------------------- evaluate current state --------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max      # sum(a) == 1

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # -------------------- schedule hyper‑parameters -----------------
        now = time.monotonic()
        elapsed_total = now - start_time
        total = deadline - start_time
        # linear schedules for learning‑rate and temperature
        eta = eta_start * (1.0 - elapsed_total / total) + eta_end * (elapsed_total / total)
        beta = beta_start * (beta_end / beta_start) ** (elapsed_total / total)

        # -------------------- soft‑max weighting ------------------------
        max_conv = conv.max()
        # w_k = exp(beta * (conv_k - max_conv)) , max weight = 1
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:  # numerically impossible, but guard against it
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                       # normalised soft‑max

        # -------------------- gradient (∂/∂a of soft‑max objective) -----
        a_rev = a[::-1]
        L = (2 * n - 1) + n - 1               # full length of w * a_rev
        fft_len_g = _fft_len_for_len(L)
        fw = np.fft.rfft(w, n=fft_len_g)
        fa = np.fft.rfft(a_rev, n=fft_len_g)
        grad_full = np.fft.irfft(fw * fa, n=fft_len_g)
        # slice the n components that correspond to ∂/∂a_j
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # -------------------- Adam‑style mirror‑gradient update -------
        it += 1
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad ** 2)
        m_hat = m / (1.0 - adam_beta1 ** it)
        v_hat = v / (1.0 - adam_beta2 ** it)
        # exponentiated‑gradient step on the simplex (log‑space Adam)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(-step)
        a = np.clip(a, 0.0, None)
        a = _project_to_simplex(a)

        # -------------------- enforce symmetry -------------------------
        a = _symmetrize(a)

        # -------------------- occasional random perturbations ----------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # -------------------- occasional pairwise mass transfer -------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_new = a.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    a_new = _project_to_simplex(a_new)
                    # quick check – only accept if it improves the score
                    conv_new = _conv_self(a_new, fft_len)
                    new_score = 2.0 * n * conv_new.max()
                    if new_score < cur_score:
                        a = a_new

        # -------------------- soft restart when stuck ------------------
        if now - last_improve > restart_gap:
            # restart from the best solution seen so far, add a mild jitter
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = _project_to_simplex(a)
            a = _symmetrize(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best vector (safely clipped for evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
