Implementation of simplex projection:

def proj_simplex(v):
    # Duchi algorithm
    # ensure nonnegative sum=1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho = max j such that u_j > cssv_j / j
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose candidate sequence for the “minimum L∞ self‑convolution’’ problem.

Algorithm
---------
* Work with a probability vector (sum = 1) – the objective is scale‑invariant.
* Use a projected Adam optimiser on the simplex:
    – Gradient = ∂/∂a Σ_k w_k·(a*a)_k .
    – w_k is a soft‑max of the convolution (temperature β) which is
      gradually sharpened; after ~70 % of the time a hard max (δ–weight)
      is used.
    – After each Adam step a Euclidean projection onto the simplex
      (non‑negative, sum = 1) is performed.
* Three cheap diversification mechanisms are mixed in:
    – Random multiplicative jitter on a short contiguous block.
    – Pairwise “mass‑transfer’’ moves that try to lower the current
      max‑convolution directly.
    – Restart when no improvement has been seen for 30 s: start from the
      best solution seen so far and add log‑normal noise.
* The routine runs as many Adam iterations as fit into the allotted
  time (default 1000 s).  Multiple problem sizes (the length of the
  initial vector and a few preset lengths) are tried sequentially; the
  best sequence over all runs is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # clip tiny negatives
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements 2·(a ⋆ w) where ⋆ is cross‑correlation.
    """
    n = a.size
    Lw = w.size                 # = 2*n-1
    wr = w[::-1]                # reverse for correlation
    conv_len = _next_pow2(n + Lw - 1)   # needed length for convolution

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep needed part

    idx = Lw - 1 - np.arange(n)          # mapping to correlation result
    grad = 2.0 * conv_full[idx]
    return grad


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation on a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _proj_simplex(out)


def _pairwise_transfer(a: np.ndarray, conv: np.ndarray,
                       rng: np.random.Generator):
    """
    Cheap mass‑transfer between two random indices.
    Returns (a_new, conv_new, new_score) if the move improves the
    max‑convolution; otherwise returns the original tuple and None.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    # move up to 30 % of donor's mass
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # update convolution efficiently
    new_conv = conv.copy()
    lo_i, hi_i = i, i + n
    lo_j, hi_j = j, j + n
    new_conv[lo_i:hi_i] -= delta * a    # donor contribution removed
    new_conv[lo_j:hi_j] += delta * a    # receiver contribution added

    new_score = 2.0 * n * np.max(new_conv)
    cur_score = 2.0 * n * np.max(conv)

    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = _proj_simplex(a2)
        return a2, new_conv, new_score
    else:
        return a, conv, None


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` (i.e. 2·n·max(a*a) / (sum a)^2 ).

    The routine respects the supplied time budget and returns the best
    coefficient list found (as plain Python floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine a few problem sizes and starting vectors
    # ------------------------------------------------------------------
    runs = []

    # Primary run – reuse the previously found construction if present
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand = cand / s
        else:
            cand = np.full_like(cand, 1.0 / cand.size)
        runs.append({"n": cand.size, "a0": cand})
    else:
        # fallback uniform of size ~4000
        n0 = 4096
        runs.append({"n": n0, "a0": np.full(n0, 1.0 / n0)})

    # Additional runs with other lengths (to diversify the search)
    for n in (1024, 2048, 4096, 8192):
        if n != runs[0]["n"]:
            a0 = rng.random(n)
            a0 = _proj_simplex(a0)
            runs.append({"n": n, "a0": a0})

    # ------------------------------------------------------------------
    # 2) Global bookkeeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety margin
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 3) Iterate over the selected runs
    # ------------------------------------------------------------------
    for run_id, run in enumerate(runs, start=1):
        n = run["n"]
        a = run["a0"].copy()

        # FFT lengths (powers of two)
        fft_self = _next_pow2(2 * n - 1)
        fft_grad = _next_pow2(3 * n - 2)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t_iter = 0

        # restart timer
        last_improve = time.monotonic()

        # ------------------------------------------------------------------
        # 4) Main optimisation loop for this run
        # ------------------------------------------------------------------
        while time.monotonic() < deadline:
            # ----- convolution & current score -----
            conv = _self_conv_fft(a, fft_self)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep the best overall solution
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] run {run_id}/{len(runs)} "
                      f"new best score = {best_score:.6f}")

            # ----- schedule (time‑based) -----
            now = time.monotonic()
            elapsed = now - start_time
            total = deadline - start_time
            frac = elapsed / total if total > 0 else 0.0

            # learning rate – from 0.5 down to 0.01
            lr = 0.5 * (1.0 - frac) + 0.01 * frac
            # temperature – from 0.5 to 2000 (soft‑max → hard)
            beta = 0.5 + (2000.0 - 0.5) * frac

            # ----- weight vector w (soft‑max or hard) -----
            if frac < 0.70:                     # soft‑max phase
                # stable soft‑max: subtract cur_max before exponentiation
                w = np.exp(beta * (conv - cur_max))
            else:                               # hard‑max sub‑gradient
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

            # ----- gradient of the weighted max -----
            grad = _gradient_from_weights(a, w)

            # ----- Adam update -----
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _proj_simplex(a)

            # ----- occasional exploration moves -----
            if rng.random() < 0.025:        # block jitter
                a = _random_block_perturb(a, rng)

            if rng.random() < 0.07:         # cheap pairwise move
                a, conv, _ = _pairwise_transfer(a, conv, rng)

            # ----- restart if stuck -----
            if now - last_improve > 30.0:
                # start from the best solution seen so far, add log‑normal noise
                a = best_seq.copy()
                noise = np.exp(rng.normal(0.0, 0.12, size=n))
                a = a * noise
                a = _proj_simplex(a)
                m.fill(0.0)
                v.fill(0.0)
                t_iter = 0
                last_improve = now
                print(f"[{now - start_time:6.2f}s] run {run_id} restart")

        # End of this run – loop proceeds to the next length
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    if best_seq is None:
        # This should never happen, but fallback to a uniform vector
        n = runs[0]["n"]
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
