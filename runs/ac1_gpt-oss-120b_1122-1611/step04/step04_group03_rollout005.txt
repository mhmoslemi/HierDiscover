def proj_simplex(v):
    # v numpy array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000.0, **kwargs):

    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # init a
    if 'height_sequence_1' in globals():
        seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        seq = np.clip(seq,0,None)
        # normalise to sum=1
        total = seq.sum()
        if total > 0:
            a = seq/ total
        else:
            a = np.full(seq.shape, 1.0/seq.size)
    else:
        n = 1024
        a = np.full(n, 1.0/n)

    n = a.size # final length

    best = a.copy()
    conv_best = _conv_self(best)
    best_score = float(2*n*conv_best.max())
    last_improve = time.monotonic()
    start_time = last_improve

    # hyperparameters
    eta_start = 1.0
    eta_end = 0.001
    restart_gap = 30.0
    block_perturb_prob = 0.04
    swap_prob = 0.03
    jitter_sigma = 0.15

    grad_prev = np.zeros_like(a) # maybe optional momentum

    # precompute FFT lengths for conv self and grad conv
    fft_len_self = _next_pow2(2*n - 1)
    fft_len_grad = _next_pow2(3*n - 2)

    while time.monotonic() < deadline:
        # evaluate conv and current
        conv = _conv_self(a) # length 2n-1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        # update best
        if cur_score < best_score - 1e-12: # improvement
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # schedule eta
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1 - elapsed/total) + eta_end * (elapsed/total)

        # compute subgradient with hardmax
        # tolerance for max indices
        # Use relative tolerance: within 1e-12 of max
        tol = 1e-12
        max_idxs = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if max_idxs.size > 0:
            w[max_idxs] = 1.0 / max_idxs.size
        # compute gradient via FFT
        a_rev = a[::-1]
        # compute conv_rev = w * a_rev
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: (len(w) + len(a_rev) - 1)]
        grad = 2.0 * conv_rev[n-1 : n-1 + n]

        # momentum optional: incorporate previous grad
        # set momentum factor
        momentum = 0.8
        grad = momentum*grad_prev + (1-momentum)*grad
        grad_prev = grad

        # exponentiated gradient update
        log_factor = -eta*grad
        # clipping
        log_factor = np.clip(log_factor, -45.0, 45.0)
        factor = np.exp(log_factor)
        a = a * factor
        # ensure positivity
        a = np.clip(a, 0.0, None)
        # renormalize
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0/n
        else:
            a /= s

        # occasional perturbations
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # restart if no improvement
        if now - last_improve > restart_gap:
            # reset to best with jitter
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a[:] = 1.0/n
            else:
                a /= a_sum
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # after loop, return best clipped to constraints
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Propose a candidate step‐function sequence via hard‑max sub‑gradient
exponentiated‑gradient descent on the simplex.

The evaluation metric (with the sequence normalised to sum = 1) is

    S(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full linear self‑convolution.
If we keep a on the probability simplex then minimising S(a) is
equivalent to minimising the L∞‑norm of the autocorrelation.
The algorithm repeatedly

* evaluates the current self‑convolution,
* builds the exact sub‑gradient by placing a unit weight on a *random*
  index attaining the current maximum (hard‑max, i.e. the true
  sub‑gradient of max‑conv),
* updates the probabilities with an exponentiated‑gradient (mirror‑descent)
  step,
* re‑normalises the vector,
* occasionally perturbs a random contiguous block, swaps two
  coordinates, or restarts from the best solution seen so far (with a
  log‑normal jitter and a random permutation).

A simple linear schedule drives the stepsize η from a large initial value
to a tiny final value, allowing aggressive early moves and fine‑tuning
later.  The method is cheap (two FFTs per iteration) and easily fits
the 1000 s budget while exploring many thousands of iterations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using FFT (returns length 2·n‑1)."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:n]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renorm."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # Leave a small safety margin for the harness.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initial vector – use the supplied best construction if present.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    raise ValueError
            else:
                raise ValueError
        except Exception:
            n0 = 512
            a = np.full(n0, 1.0 / n0, dtype=np.float64)
    else:
        n0 = 512
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size                     # final length (kept constant)

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best)
    best_score = float(2.0 * n * conv_best.max())
    start_time = time.monotonic()
    last_improve = start_time

    # --------------------------------------------------------------
    # 3) Hyper‑parameters
    # --------------------------------------------------------------
    eta_start = 2.0                # aggressive initial step size
    eta_end = 0.001                # fine‑tuning step size
    restart_gap = 20.0             # seconds without improvement → restart
    block_perturb_prob = 0.05
    swap_prob = 0.05
    jitter_sigma = 0.20
    momentum = 0.85                # exponential smoothing of the gradient
    grad_prev = np.zeros_like(a)

    # Pre‑compute the FFT lengths that never change.
    fft_len_self = _next_pow2(2 * n - 1)          # for a∗a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w∗a_rev

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- current self‑convolution and score --------------------
        conv = _conv_self(a)                      # length 2·n‑1
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max              # sum(a)=1 by construction

        # ---- record the best solution so far ----------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule the stepsize (linear decay) -----------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # ---- identify a *random* index attaining the maximum ------- 
        # (hard‑max gives the exact sub‑gradient of max‑conv)
        tol = 1e-12
        max_idxs = np.where(conv >= cur_max - tol)[0]
        # pick one uniformly at random when there are ties
        k = rng.choice(max_idxs)
        w = np.zeros_like(conv)
        w[k] = 1.0

        # ---- sub‑gradient of max‑conv via a single FFT -------------
        a_rev = a[::-1]                           # reversed copy
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: (len(w) + len(a_rev) - 1)]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

        # ---- momentum smoothing ------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- exponentiated‑gradient (mirror descent) update -------
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -45.0, 45.0)   # avoid overflow
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- occasional random perturbations ------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # ---- soft restart when progress stalls ----------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # random permutation before jitter to explore new orderings
            perm = rng.permutation(n)
            a = a[perm]
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best solution (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
