   The gradient of the soft‑max is computed exactly via an FFT‑based
   correlation, so every iteration uses the full convolution information.
   β is increased linearly from 0.5 to 500, while the learning rate is
   annealed from 0.2 to 0.001.

2. **Stochastic local refinement (≈30 % of the budget)** –  
   A simulated‑annealing style search makes pairwise mass transfers
   and block‑wise log‑normal perturbations.  Acceptance follows the
   Metropolis rule with a linearly decreasing temperature.

Both phases keep the best solution seen so far and perform a soft
restart (small jitter around the best) if no improvement occurs for
30 seconds.  The final sequence is projected onto the feasible domain
(non‑negative, sum>0) and clipped to the limits enforced by
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution of a via FFT (returns length 2*len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _correlate_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear correlation r = correlate(a, b) (mode='full')
    using FFT. Result length = len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    corr = np.fft.irfft(fa * np.conj(fb), n=fft_len)
    return corr[: a.size + b.size - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of
    Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety – renormalise if rounding error accumulated
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection.
    """
    n = a.size
    # width at most 10 % of the vector, at least 1
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_simplex(out)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling
    # --------------------------------------------------------------
    # keep a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Starting vector (simplex‑normalised)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(cand, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # default length – a power‑of‑two work well with FFT
        n = 1024
        a = rng.dirichlet(np.ones(n)).astype(np.float64)

    n = a.size

    # --------------------------------------------------------------
    # 2) FFT length – must accommodate a (len n) and a*a (len 2n‑1)
    #    for the correlation we need length ≥ 3n‑2.
    # --------------------------------------------------------------
    fft_len = _next_pow_two(3 * n - 2)

    # --------------------------------------------------------------
    # 3) Adam state and bookkeeping
    # --------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0

    # evaluate the initial candidate
    conv = _conv_self_fft(a, fft_len)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (a.sum() ** 2)   # sum≈1 after projection
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.70 * total_time   # 70 % for gradient phase

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ==============================================================
        # Phase 1 – soft‑max Adam descent (smoothly tightening β)
        # ==============================================================
        if now < phase1_end:
            # linear schedules for learning‑rate η and temperature β
            frac = (now - start_time) / (phase1_end - start_time)
            eta = 0.20 * (1.0 - frac) + 0.001 * frac          # step size
            beta = 0.5 + (500.0 - 0.5) * frac                 # soft‑max sharpness

            # ----- compute convolution, soft‑max weights, and gradient -----
            Fa = np.fft.rfft(a, n=fft_len)
            conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]

            max_c = conv_full.max()
            # stable soft‑max (subtract max)
            exp_w = np.exp(beta * (conv_full - max_c))
            Z = exp_w.sum()
            if Z == 0.0:                     # numerical safeguard
                p = np.full_like(conv_full, 1.0 / conv_full.size)
            else:
                p = exp_w / Z                # probabilities, length 2n‑1

            # correlation a * p  → gradient (2 · ⟨a,p⟩)
            Fp = np.fft.rfft(p, n=fft_len)
            corr_full = np.fft.irfft(Fa * np.conj(Fp), n=fft_len)
            start_idx = p.size - 1           # = 2n‑2, start of the “central” slice
            grad = 2.0 * corr_full[start_idx : start_idx + n]

            # ----- Adam update on the simplex -----
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            step = m_hat / (np.sqrt(v_hat) + eps)

            a = a - eta * step               # gradient descent
            a = _project_simplex(a)          # stay on the simplex

            # recompute score after the step
            Fa = np.fft.rfft(a, n=fft_len)
            conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
            cur_max = conv_full.max()
            cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

        # ==============================================================
        # Phase 2 – stochastic local refinement (simulated annealing)
        # ==============================================================
        else:
            # linear temperature schedule
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = max(1e-7, 0.05 * (1.0 - t_frac))

            if rng.random() < 0.85:          # pairwise mass transfer
                i = rng.integers(0, n)
                if a[i] < 1e-12:
                    continue
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_delta = 0.30 * a[i]
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # evaluate new candidate
                Fa_new = np.fft.rfft(a_new, n=fft_len)
                conv_new = np.fft.irfft(Fa_new * Fa_new, n=fft_len)[: 2 * n - 1]
                new_score = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

                # Metropolis acceptance
                if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                    cur_score = new_score
            else:                             # global block jitter
                a = _random_block_perturb(a, rng)
                Fa = np.fft.rfft(a, n=fft_len)
                conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
                cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart when the search stalls for a while
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # jitter around the best and reset Adam state
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
