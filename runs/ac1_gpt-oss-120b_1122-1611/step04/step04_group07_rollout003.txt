"""
Propose a candidate step‑function using Adam on the simplex (logits).

The algorithm works on a fixed length `n` (default 4096) and optimises
the soft‑max approximation of the max‑convolution:

    Fβ(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

Gradients are obtained analytically (via FFT‑based correlation) and the
updates are performed on the log‑probabilities (logits) so that the
variables automatically stay on the probability simplex.  Adam’s
adaptive step‑size together with a linear temperature schedule (β) and
periodic random block perturbations/soft restarts give a good balance
between exploration and exploitation.

The best sequence seen (exactly evaluated with `evaluate_sequence`) is
returned at the end of the allotted time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Return a power‑of‑two length that can accommodate both the
    self‑convolution (length 2·n‑1) and the correlation
    (length 3·n‑2).  Using the larger of the two guarantees that a single
    FFT length works for all needed convolutions.
    """
    size = max(2 * n - 1, 3 * n - 2)
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a real vector via FFT.
    Returned length is ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    return conv_full[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using the supplied FFT length.
    Result length is ``len(x)+len(y)-1``.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv_full = np.fft.irfft(fx * fy, n=fft_len)
    return conv_full[: x.size + y.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 20))          # smaller blocks than before
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.20))            # a bit stronger noise

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.

    The search runs an Adam optimiser on logits with a soft‑max
    approximation of the max‑norm of the self‑convolution, plus occasional
    random block perturbations and soft restarts.
    """
    rng = np.random.default_rng(seed)

    # leave a few seconds for the harness to finish cleanly
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    default_n = 4096                     # a power‑of‑two that worked well before
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a0 = np.clip(cand, 0.0, None)
                s = a0.sum()
                if s > 0.0:
                    a0 = a0 / s
                else:
                    a0 = np.full_like(cand, 1.0 / cand.size)
                n = a0.size
                # if the provided sequence is far from default_n, fall back to default_n
                if n != default_n:
                    n = default_n
                    a0 = None
            else:
                a0 = None
        except Exception:
            a0 = None
    else:
        a0 = None

    n = default_n
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Initialise logits (log‑probability vector)
    # ------------------------------------------------------------------
    eps = 1e-12
    if a0 is not None and a0.size == n:
        a = a0
    else:
        a = rng.dirichlet(np.ones(n))      # uniform Dirichlet -> random simplex point
    logits = np.log(a + eps)               # guarantees positivity & sum=1

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters & schedules
    # ------------------------------------------------------------------
    eta_start = 1.0      # initial Adam step‑size (acts as learning rate)
    eta_end = 0.01       # final step‑size
    beta_start = 0.5     # start temperature for soft‑max
    beta_end = 1500.0    # final temperature (high → sharper max)

    block_perturb_prob = 0.07   # fairly aggressive early on
    restart_gap = 25.0          # seconds without improvement → restart
    print_interval = 30.0
    next_print = start_time + print_interval

    # ------------------------------------------------------------------
    # 4) Best‑so‑far bookkeeping
    # ------------------------------------------------------------------
    # initial (fast) estimate
    conv_self = _conv_self_fft(a, fft_len)
    cur_max = float(conv_self.max())
    best_est_score = 2.0 * n * cur_max          # sum(a)=1 → denominator =1
    best_exact_score = float('inf')
    best_seq = a.copy()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        progress = (now - start_time) / total_time
        # ---- linear schedules for η and β ---------------------------------
        eta = eta_start * (1.0 - progress) + eta_end * progress
        beta = beta_start + (beta_end - beta_start) * progress

        # ---- obtain current probability vector ----------------------------
        # stable soft‑max on logits
        logit_shift = np.max(logits)
        exp_shifted = np.exp(logits - logit_shift)
        a = exp_shifted / exp_shifted.sum()

        # ---- self‑convolution (FFT) ---------------------------------------
        conv_self = _conv_self_fft(a, fft_len)
        max_conv = float(conv_self.max())
        est_score = 2.0 * n * max_conv                 # sum(a)=1 ⇒ denominator =1

        # ---- keep exact best ------------------------------------------------
        if est_score + 1e-9 < best_est_score:           # small tolerance to avoid noise
            # exact evaluation (a bit slower)
            exact = evaluate_sequence(a.tolist())
            if exact < best_exact_score:
                best_exact_score = exact
                best_est_score = est_score
                best_seq = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best exact score = {best_exact_score:.6f}")

        # ---- periodic status printing --------------------------------------
        if now >= next_print:
            print(f"[{now - start_time:6.2f}s] est. score = {est_score: .6f}  "
                  f"best = {best_exact_score: .6f}")
            next_print = now + print_interval

        # ---- soft‑max weighting of all convolution entries -----------------
        # shift by the maximum for numerical stability
        shift = max_conv
        w_raw = np.exp(beta * (conv_self - shift))
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ---- gradient of the soft‑max objective (FFT‑based correlation) -----
        a_rev = a[::-1]
        # FFTs for correlation
        fw = np.fft.rfft(w, n=fft_len)
        fra = np.fft.rfft(a_rev, n=fft_len)
        corr_full = np.fft.irfft(fw * fra, n=fft_len)
        # extract the “valid” part of length n (indices n‑1 … n‑1+n‑1)
        start = n - 1
        grad_a = 2.0 * corr_full[start:start + n]

        # ---- map gradient to logits (mirror‑descent on simplex) -------------
        dot_a_grad = np.dot(a, grad_a)
        grad_logits = a * (grad_a - dot_a_grad)

        # ---- Adam update ----------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- occasional random block perturbation --------------------------
        if rng.random() < block_perturb_prob:
            a_pert = _random_block_perturb(a, rng)
            logits = np.log(a_pert + eps)

        # ---- soft restart when stuck ----------------------------------------
        if now - last_improve > restart_gap:
            # jitter around the current best
            if best_seq is not None:
                a_restart = best_seq.copy()
                noise = np.exp(rng.normal(0.0, 0.12, size=n))
                a_restart = a_restart * noise
                a_restart = np.clip(a_restart, 0.0, None)
                a_restart /= a_restart.sum()
                logits = np.log(a_restart + eps)
                m.fill(0.0)
                v.fill(0.0)
                t_iter = 0
                last_improve = now
                print(f"[{now - start_time:6.2f}s] restart – best score {best_exact_score:.6f}")

    # ----------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ----------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
