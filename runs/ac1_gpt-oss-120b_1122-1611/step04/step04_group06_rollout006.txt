def lp_optimize(f, M):
    n = len(f)
    conv_mat = scipy.linalg.toeplitz(np.r_[f, np.zeros(n-1)], np.r_[f[0], np.zeros(n-1)])
    b = cp.Variable(n)
    constraints = [conv_mat @ b <= M, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    b_val = b.value
    # may get None if unsolved
    return b_val
```

def propose_candidate(seed=0, budget_s=1000.0):
    """
    LP + mirror descent hybrid search for step‑function sequence minimizing
    evaluate_sequence(). The algorithm fixes the sum of the heights to
    sqrt(2·n) so that the evaluation reduces to the maximal entry of the
    self‑convolution. Starting from a random or provided vector it repeatedly
    solves a linear program that, for the current sequence f, finds a vector b
    with the largest possible ℓ₁‑norm under the same convolution bound.
    The new candidate g = scaling·b is mixed with f. The mixing factor t is
    gradually decreased. The best vector seen during the budget is returned.
    """
    import time, numpy as np, cvxpy as cp, scipy.linalg

    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)  # leave a safety margin

    best_f = None
    best_M = float('inf')
    best_n = None

    # Choose a set of lengths to explore – small enough to keep LP cheap.
    lengths = [128, 192, 256, 320, 384]
    # optionally include length of global if exists
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0:
            lengths.append(cand.size)

    # Remove duplicates
    lengths = sorted(set(lengths))

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # create initial f
        if "height_sequence_1" in globals():
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if init.size == n:
                f = np.clip(init, 0.0, None)
            else:
                # random Dirichlet
                f = rng.random(n)
        else:
            f = rng.random(n)
        # Normalise to sum sqrt(2*n)
        target_sum = np.sqrt(2.0 * n)
        s = f.sum()
        if s == 0:
            f = np.full(n, target_sum / n)
        else:
            f = f * (target_sum / s)

        # hyper‑parameters
        # mixing factor t – start relatively high, decrease slowly
        t = 0.5
        t_decay = 0.998  # each iteration
        min_t = 0.02
        # slack for LP (tiny epsilon to avoid numerical infeasibility)
        eps = 1e-9

        # iteration counter
        it = 0
        # record last improvement time
        last_imp = time.monotonic()

        while time.monotonic() < deadline:
            it += 1

            # current convolution and its maximum
            conv_f = np.convolve(f, f)
            M = float(conv_f.max())

            # keep best
            if M < best_M:
                best_M = M
                best_f = f.copy()
                best_n = n
                last_imp = time.monotonic()
                elapsed = last_imp - start
                print(f"[{elapsed:6.2f}s] New best M = {best_M:.6f} (n={n})")

            # build convolution matrix
            c = np.concatenate([f, np.zeros(n - 1)])
            r = np.concatenate(([f[0]], np.zeros(n - 1)))
            conv_mat = scipy.linalg.toeplitz(c, r)  # (2n-1) × n

            # linear program
            b = cp.Variable(n, nonneg=True)
            constraints = [conv_mat @ b <= M + eps]
            objective = cp.Maximize(cp.sum(b))
            prob = cp.Problem(objective, constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
            except Exception as e:
                print("LP solve failed:", e)
                break

            if b.value is None:
                # infeasible or numerical issue – abort this n
                print("LP returned no solution, breaking.")
                break

            b_val = np.asarray(b.value).flatten()
            sum_b = b_val.sum()
            if sum_b <= 0:
                # degenerate solution, break
                break

            # scale to maintain target sum
            g = b_val * (target_sum / sum_b)

            # convex combination update
            f = (1 - t) * f + t * g

            # re‑normalise to exact sum (numerical drift)
            s = f.sum()
            if s == 0:
                f = np.full(n, target_sum / n)
            else:
                f = f * (target_sum / s)

            # schedule t
            t = max(min_t, t * t_decay)

            # occasional random perturbation to escape tiny plateaus
            if rng.random() < 0.03:
                # random Dirichlet jitter
                jitter = rng.random(n)
                jitter = jitter / jitter.sum() * target_sum
                f = 0.9 * f + 0.1 * jitter
                f = np.clip(f, 0.0, None)
                s = f.sum()
                if s > 0:
                    f = f * (target_sum / s)

            # restart if stuck too long
            if time.monotonic() - last_imp > 60.0:
                # re‑initialize from the current best of this length
                if best_f is not None and best_n == n:
                    f = best_f.copy()
                else:
                    f = rng.random(n)
                s = f.sum()
                f = f * (target_sum / s)
                t = 0.5
                last_imp = time.monotonic()
                print(f"[{time.monotonic() - start:6.2f}s] Restart on n={n}")

    # After search, return the best found sequence (scaled to original sum=1)
    if best_f is None:
        # fallback to uniform
        n = 256
        best_f = np.full(n, 1.0 / n)
    # Scale to sum 1 (evaluate_sequence is scale invariant)
    best_f = best_f.astype(np.float64)
    total = best_f.sum()
    if total == 0:
        best_f = np.full_like(best_f, 1.0 / best_f.size)
    else:
        best_f = best_f / total

    # Clip for safety
    best_f = np.clip(best_f, 0.0, 1000.0)

    return [float(x) for x in best_f]

"""
LP‑plus‑mirror‑descent search for a step‑function with minimal
evaluate_sequence() value.

The evaluation for a non‑negative vector a with Σa = 1 reduces to
   score = 2·n·max(a∗a) .
The algorithm works on the slice ∑a = √(2·n) (so that the score equals
the maximal convolution entry M).  For a given f we solve a linear program

    maximise   Σ b_j
    subject to  (f * b)_k ≤ M   for all k,
                b_j ≥ 0.

The optimal b is an extreme point that maximises the ℓ¹‑norm while keeping
the same convolution bound.  Mixing f←(1‑t)f+t·g (with g the scaled b) reduces
M monotonically.  After a budget of LP steps we finish with a short
exponentiated‑gradient (mirror‑descent) refinement using a smooth soft‑max
surrogate of the max‑convolution.

The best sequence found (scaled back to Σ=1) is returned.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.linalg


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (used for FFT‑based convolutions)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a∗a using FFT (returns length 2·n‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the soft‑max surrogate.
    grad_j = 2·∑_k w_k·a_{k‑j}.
    Computed by a linear correlation via FFT.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _lp_update(f: np.ndarray, target_sum: float, eps: float = 1e-9) -> np.ndarray | None:
    """
    Given the current vector `f` (Σf = target_sum) and its current maximal
    self‑convolution M, solve the linear program described in the docstring.
    Returns the new candidate `g` (also Σg = target_sum) or ``None`` if the
    LP fails.
    """
    n = f.size
    # maximal entry of the self‑convolution of f
    M = np.convolve(f, f).max()

    # Toeplitz matrix that implements the linear convolution f * b
    c = np.concatenate([f, np.zeros(n - 1)])
    r = np.concatenate(([f[0]], np.zeros(n - 1)))
    conv_mat = scipy.linalg.toeplitz(c, r)          # shape (2n‑1, n)

    b = cp.Variable(n, nonneg=True)
    constraints = [conv_mat @ b <= M + eps]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except Exception:        # pragma: no cover – safeguard
        return None

    if b.value is None:
        return None

    b_val = np.asarray(b.value).flatten()
    sum_b = b_val.sum()
    if sum_b <= 0.0:
        return None

    # scale back to the required sum
    g = b_val * (target_sum / sum_b)
    return g


def _mirror_descent_refine(a: np.ndarray,
                           remaining_sec: float,
                           rng: np.random.Generator) -> np.ndarray:
    """
    Short mirror‑descent (exponentiated‑gradient) refinement.
    Works on the probability simplex (Σa = 1).
    """
    n = a.size
    fft_len = _fft_len_for_n(n)

    best_a = a.copy()
    best_score = np.convolve(best_a, best_a).max()

    start = time.monotonic()
    deadline = start + remaining_sec

    # schedules (same as the reference implementation)
    eta_start, eta_end = 0.7, 0.01
    beta_start, beta_end = 0.5, 600.0
    block_perturb_prob = 0.02
    pair_move_prob = 0.09
    momentum = 0.95
    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # soft‑max weights over the convolution entries
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        w_norm = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad = _gradient_correlation(a, w_norm)
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update (mirror descent on the simplex)
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional local perturbations
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate /= a_candidate.sum()
                    conv_cand = _conv_self(a_candidate, fft_len)
                    cand_max = conv_cand.max()
                    if cand_max < cur_max:
                        a = a_candidate

    return best_a

        t_decay = 0.9995
        eps = 1e-9

        while time.monotonic() < sub_deadline:
            cur_conv = np.convolve(f, f)
            cur_M = float(cur_conv.max())

            # keep the best for this length
            if cur_M < best_M_len:
                best_M_len = cur_M
                best_f_len = f.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] n={n} new best M={best_M_len:.6f}")

            # ----------------------------------------------------------
            # LP step
            # ----------------------------------------------------------
            g = _lp_update(f, target_sum, eps=eps)
            if g is None:
                # LP failed – break out of this length
                break

            # convex combination
            f = (1.0 - t) * f + t * g
            # re‑normalise exactly to target_sum (numerical drift)
            s = f.sum()
            if s == 0.0:
                f = np.full(n, target_sum / n)
            else:
                f = f * (target_sum / s)

            # update mixing weight
            t = max(t_min, t * t_decay)

            # occasional random jitter to escape plateaus
            if rng.random() < 0.02:
                jitter = np.exp(rng.normal(0.0, 0.12, size=n))
                f = f * jitter
                f = np.clip(f, 0.0, None)
                f = f * (target_sum / f.sum())

            # restart if no improvement for ~90 s
            if time.monotonic() - last_improve > 90.0:
                f = best_f_len.copy()
                t = 0.4
                last_improve = time.monotonic()
                print(f"[{time.monotonic() - start_time:6.2f}s] restart on n={n}")

        # keep the best across all n‑sizes
        if best_M_len < best_score_overall:
            best_score_overall = best_M_len
            best_overall = best_f_len.copy()
            best_n_overall = n
            print(f"[{time.monotonic() - start_time:6.2f}s] GLOBAL best M={best_score_overall:.6f} (n={n})")

        # stop early if overall budget is exhausted
        if time.monotonic() >= overall_deadline:
            break

    # ------------------------------------------------------------------
    # 3) Final mirror‑descent refinement (uses Σ=1 normalisation)
    # ------------------------------------------------------------------
    time_left = overall_deadline - time.monotonic()
    if best_overall is None:
        # fallback – uniform vector of length 256
        best_n_overall = 256
        best_overall = np.full(best_n_overall, np.sqrt(2.0 * best_n_overall) / best_n_overall)

    if time_left > 5.0:
        # scale to probability simplex for the refinement routine
        prob_vec = best_overall / best_overall.sum()
        refined = _mirror_descent_refine(prob_vec, time_left, rng)
        # bring back to the original target sum (√(2·n))
        refined = refined * (np.sqrt(2.0 * best_n_overall) / refined.sum())
        # update global best if improvement observed
        final_M = np.convolve(refined, refined).max()
        if final_M < best_score_overall:
            best_score_overall = final_M
            best_overall = refined
            print(f"[{time.monotonic() - start_time:6.2f}s] post‑refine M={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # 4) Return a sequence with Σ=1 (the scaling does not affect the score)
    # ------------------------------------------------------------------
    out = best_overall.astype(np.float64)
    total = out.sum()
    if total == 0.0:
        out = np.full_like(out, 1.0 / out.size)
    else:
        out = out / total
    out = np.clip(out, 0.0, 1000.0)          # safety for evaluate_sequence
    return [float(x) for x in out]
