def _gradient_from_weights(a, w, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: 3*n -2]
    grad = 2.0 * full[n-1 : n-1 + n]
    return grad
```

Algorithm: hybrid softmax + hard max subgradient descent on the
infinity norm of the self‑convolution, with Euclidean projection onto
the probability simplex.  Starts from the provided global solution
(if any), pads to a moderate size, then runs a momentum gradient
descent with a temperature schedule.  After the temperature reaches
a high value we switch to a true subgradient of the max.  Stagnation
triggers a random jitter restart.  The best vector seen is returned.
"""

import time
import numpy as np

def _fft_len_for_n(n):
    return 1 << (n - 1).bit_length()

def _conv_self(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _gradient_from_weights(a, w, fft_len):
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad

def _project_simplex(v):
    # Euclidean projection onto the simplex {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _random_block_perturb(a, rng):
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out

def _hard_max_gradient(a, conv):
    n = a.size
    k = int(conv.argmax())
    grad = np.zeros_like(a)
    start_i = max(0, k - (n - 1))
    end_i = min(n, k + 1)
    idx = np.arange(start_i, end_i)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid gradient / subgradient optimisation of a non‑negative step
    function.  Returns a list of heights with the smallest value of
    evaluate_sequence() found before the time limit.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # 1) Initialise a feasible vector (sum = 1, non‑negative)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            a = np.full(cand.shape, 1.0 / cand.size, dtype=np.float64)
        else:
            a = cand / s
        n = a.size
    else:
        n = 1024
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # If the vector is unusually short, pad to a convenient size.
    target_len = max(n, 1024)
    if target_len != n:
        a = np.concatenate([a, np.zeros(target_len - n, dtype=np.float64)])
        n = target_len
        a = _project_simplex(a)

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # --------------------------------------------------------------
    fft_len_conv = _fft_len_for_n(2 * n - 1)          # for a*a
    fft_len_grad = _fft_len_for_n(3 * n - 2)          # for weighted gradient

    # --------------------------------------------------------------
    # 3) Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len_conv)
    best_score = 2.0 * n * np.max(conv_best) / (np.sum(best) ** 2)

    start_time = time.monotonic()
    last_improve = start_time

    # --------------------------------------------------------------
    # 4) Hyper‑parameters
    # --------------------------------------------------------------
    eta_start = 0.7   # step size at the beginning
    eta_end   = 0.02  # step size at the end
    beta_start = 0.5
    beta_end   = 2500.0
    block_perturb_prob = 0.03
    restart_gap = 20.0       # seconds without improvement before a restart
    momentum = 0.9
    grad_mom = np.zeros_like(a)

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # current convolution and score
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)

        # keep the best
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # schedule
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start * (beta_end / beta_start) ** frac

        # compute gradient
        if beta < 500.0:          # smooth phase
            shift = conv.max()
            # avoid overflow / underflow
            w_raw = np.exp(beta * (conv - shift))
            w = w_raw / w_raw.sum()
            grad = _gradient_from_weights(a, w, fft_len_grad)
        else:                     # hard max phase (subgradient)
            grad = _hard_max_gradient(a, conv)

        # momentum
        grad_mom = momentum * grad_mom + (1.0 - momentum) * grad

        # projected update
        a_tmp = a - eta * grad_mom
        a = _project_simplex(a_tmp)

        # occasional random block perturbation to escape shallow wells
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # restart if we have been stagnant for a while
        if now - last_improve > restart_gap:
            # jittered copy of the best solution
            noise = np.exp(rng.normal(0.0, 0.18, size=n))
            a = best * noise
            a = _project_simplex(a)
            grad_mom.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best vector found (clipped to problem limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Hybrid soft‑max / hard‑max subgradient optimisation of a non‑negative step
function.  The objective is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

which is scale‑invariant; we therefore keep the iterate on the probability
simplex (sum = 1, a_i ≥ 0).  The algorithm starts from the global solution
(if the variable ``height_sequence_1`` is present), pads it to a moderate
length, and runs a momentum projected gradient descent.  A temperature β
controls a smooth (log‑sum‑exp) approximation of the max‑norm; once β is
large enough we switch to the true sub‑gradient of the max term.  Stagnation
triggers a jittered restart, and a short stochastic refinement is performed
until the time budget expires.  The best vector seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold a linear convolution of
    size *n* (i.e. 2^{ceil(log2(n-1))}).
    """
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length *fft_len*.
    Result length is 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a*a)_i w.r.t. a.
    Equivalent to 2·(w ⊛ a_rev) restricted to the original length.
    """
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive — return the uniform point.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def _hard_max_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a, a)) w.r.t. a.
    If k = argmax(conv), then ∂conv_k/∂a_i = 2·a_{k‑i} (zero outside).
    """
    n = a.size
    k = int(conv.argmax())
    grad = np.zeros_like(a)
    start_i = max(0, k - (n - 1))
    end_i = min(n, k + 1)
    idx = np.arange(start_i, end_i)
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid gradient / subgradient optimiser for the evaluation function.
    Returns a list of non‑negative heights (the best vector found before the
    time budget expires).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # 1) Initialise a feasible vector (sum = 1, a_i ≥ 0)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            a = np.full(cand.shape, 1.0 / cand.size, dtype=np.float64)
        else:
            a = cand / s
        n = a.size
    else:
        n = 1024
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Pad a short vector to a comfortable size (still ≤ 5000 for speed).
    target_len = max(n, 1024)
    if target_len != n and target_len <= 5000:
        a = np.concatenate([a, np.zeros(target_len - n, dtype=np.float64)])
        n = target_len
        a = _project_simplex(a)

    # --------------------------------------------------------------
    # 2) FFT lengths (stable throughout the run)
    # --------------------------------------------------------------
    fft_len_conv = _fft_len_for_n(2 * n - 1)      # for a*a
    fft_len_grad = _fft_len_for_n(3 * n - 2)      # for the weighted gradient

    # --------------------------------------------------------------
    # 3) Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len_conv)
    best_score = 2.0 * n * np.max(conv_best) / (np.sum(best) ** 2)

    start_time = time.monotonic()
    last_improve = start_time

    # --------------------------------------------------------------
    # 4) Hyper‑parameters
    # --------------------------------------------------------------
    eta_start = 0.7          # initial step size
    eta_end   = 0.02         # final step size
    beta_start = 0.5         # start temperature (smooth max)
    beta_end   = 2500.0      # final temperature (sharp max)
    block_perturb_prob = 0.03
    restart_gap = 20.0       # seconds without improvement → restart
    momentum = 0.9
    grad_mom = np.zeros_like(a)

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # Current convolution and score
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)

        # Record the best vector seen so far
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # Time‑dependent schedules
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start * (beta_end / beta_start) ** frac

        # Gradient (soft‑max while beta is moderate, hard max afterwards)
        if beta < 500.0:
            shift = conv.max()
            w_raw = np.exp(beta * (conv - shift))
            w = w_raw / w_raw.sum()
            grad = _gradient_from_weights(a, w, fft_len_grad)
        else:
            grad = _hard_max_gradient(a, conv)

        # Momentum smoothing
        grad_mom = momentum * grad_mom + (1.0 - momentum) * grad

        # Projected gradient step on the simplex
        a = _project_simplex(a - eta * grad_mom)

        # Occasional random block perturbation to escape shallow minima
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # Restart if we have stalled for a while
        if now - last_improve > restart_gap:
            noise = np.exp(rng.normal(0.0, 0.18, size=n))
            a = best * noise
            a = _project_simplex(a)
            grad_mom.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Short stochastic refinement (uses remaining time)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # Propose a tiny mass exchange between two random indices
        cand = best.copy()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        move = rng.uniform(0.0, 0.005) * cand[i]   # up to 0.5 % of a_i
        cand[i] -= move
        cand[j] += move
        cand = _project_simplex(cand)
        conv = _conv_self(cand, fft_len_conv)
        score = 2.0 * n * np.max(conv) / (np.sum(cand) ** 2)
        if score < best_score - 1e-9:
            best_score = score
            best = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] refinement improvement = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7) Return the best sequence (clipped to the problem limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
