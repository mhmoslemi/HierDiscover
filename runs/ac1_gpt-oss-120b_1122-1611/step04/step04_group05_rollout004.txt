"""
Propose a step‑function that (approximately) minimises

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three phases

1. **Parametric warm‑start** –  a dense grid of symmetric Beta
   distributions  a_i ∝ (i+½)^{α‑1} (n‑i‑½)^{β‑1}  is scanned.
   The best vector (usually close to the arcsine shape) is kept.
   A few cheap “mixture” candidates (arcsine ↔ uniform) are also tested.

2. **Hard‑max gradient descent** –  starting from the best vector we
   repeatedly compute the self‑convolution, locate the index `k_max`
   where it attains its maximum and apply an *additive* projected
   gradient step on the simplex using the exact sub‑gradient of the
   max‑norm:
       ∂/∂a_j max(conv) = 2·a_{k_max‑j}  (if the index is in range, else 0)
   The step size η is linearly annealed from a relatively aggressive
   value (≈0.8) down to 0.001.  After every update the vector is
   clipped to non‑negative values and renormalised.

3. **Exploration & restarts** –  occasional multiplicative log‑normal
   jitter on a random block, random pairwise mass transfers and a
   soft restart when no improvement has been seen for a few seconds
   keep the search from stagnating.

All heavy work (convolution) is performed with a single FFT per
iteration; the whole routine respects the 1000 s wall‑time limit and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta‑type shape on a discrete grid.
    a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1} ,  i = 0,…,n‑1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    # Use logs for numerical stability
    log_a = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i)
    a = np.exp(log_a)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


def _score_of(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact score (without the safety‑checks of evaluate_sequence)."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))  # a.sum() == 1 → 2·n·max

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.

    The routine first searches a small parametric family (Beta‑type
    densities) and then performs a hard‑max projected gradient descent
    on the probability simplex.  Random block perturbations and a
    soft restart keep the procedure from getting trapped in a local
    minimum.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Determine length and an initial candidate (uniform fallback)
    # --------------------------------------------------------------
    default_n = 4000
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT length for this n
    # --------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 3) Warm‑start: search a parametric Beta family (symm & mixtures)
    # --------------------------------------------------------------
    best_a = a.copy()
    best_score = _score_of(best_a, fft_len, n)

    # symmetric Beta shapes (α = β) – coarse grid first
    alphas = np.linspace(0.30, 0.80, 26)               # step ≈0.02
    for alpha in alphas:
        a_beta = _beta_shape(n, alpha, alpha)
        sc = _score_of(a_beta, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_beta
            print(f"[init] α={alpha:.3f} symmetric → score={sc:.6f}")

    # finer search around the current best α
    # (extract the α that gave the best symmetric score)
    # If the best came from a non‑symmetric source keep the current best.
    # Otherwise we can refine.
    # ------------------------------------------------------------------
    # Mixtures of arcsine (α=β=0.5) and uniform
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _beta_shape(n, 0.5, 0.5)
    for lam in np.linspace(0.0, 1.0, 21):              # λ = weight on arcsine
        mixed = lam * arcsine + (1.0 - lam) * uniform
        mixed /= mixed.sum()
        sc = _score_of(mixed, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = mixed
            print(f"[init] mix λ={lam:.2f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Small asymmetric refinement around the current best parameters
    # ------------------------------------------------------------------
    # Estimate parameters of the current best (if it is a beta shape)
    # – we simply reuse the symmetric α found above as a centre.
    # Conduct a tiny grid search for (α,β) close to that centre.
    # ------------------------------------------------------------------
    # If the current best came from the mixture step, we still try a
    # tiny beta‑shape neighbourhood because it is cheap.
    # ------------------------------------------------------------------
    centre_alpha = 0.5  # default centre if we have no clue
    # try to infer a centre from the best score if it is a beta shape
    # (the symmetric grid that produced the best score is stored above)
    # For simplicity we just reuse 0.5 which is close to the optimum.
    for da in np.linspace(-0.06, 0.06, 13):
        for db in np.linspace(-0.06, 0.06, 13):
            α = 0.5 + da
            β = 0.5 + db
            if α <= 0.05 or β <= 0.05:
                continue
            a_beta = _beta_shape(n, α, β)
            sc = _score_of(a_beta, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a_beta
                print(f"[init] asymmetric α={α:.3f}, β={β:.3f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 4) Hard‑max projected gradient descent (adds fine adjustments)
    # --------------------------------------------------------------
    a = best_a.copy()
    best = a.copy()
    best_score = _score_of(best, fft_len, n)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    last_improve = start_time

    # schedule parameters
    eta_start = 0.80    # aggressive early steps
    eta_end = 0.001     # fine‑grained final steps
    block_perturb_prob = 0.02
    pair_move_prob = 0.03
    restart_gap = 30.0   # seconds without improvement → restart
    momentum = 0.0       # no momentum for pure additive update
    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:
        # ----- convolution, score & max‑index ---------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        k_max = int(conv.argmax())
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)   # a.sum()==1 → 2·n·cur_max

        # ----- keep the best --------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ----- annealing of the stepsize --------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # ----- hard‑max sub‑gradient (only the active entry) ------------------
        # gradient_j = 2 * a_{k_max - j}  iff 0 ≤ k_max - j < n
        grad = np.zeros_like(a)
        j_start = max(0, k_max - (n - 1))
        j_end = min(n - 1, k_max)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1, dtype=np.int64)
            grad[idx] = 2.0 * a[k_max - idx]

        # ----- (optional) momentum smoothing ----------------------------------
        if momentum > 0.0:
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad.copy()

        # ----- additive projected gradient step (simple) -----------------------
        a = a - eta * grad
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ----- occasional random block perturbation ---------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pair‑wise mass transfer -------------------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # transfer up to 30 % of the mass at i
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = np.clip(a_candidate, 0.0, None)
                    a_candidate /= a_candidate.sum()
                    sc = _score_of(a_candidate, fft_len, n)
                    if sc < cur_score:
                        a = a_candidate
                        cur_score = sc
                        # (best will be recorded at the top of the loop)

        # ----- soft restart if we are stuck ------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter each coordinate a little to try a new direction
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best solution (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    print(f"Final best score (internal) = {best_score:.6f}")
    # optional sanity check using the provided evaluate_sequence (if available)
    try:
        final_check = evaluate_sequence(best.tolist())
        print(f"evaluate_sequence reports score = {final_check:.6f}")
    except Exception:
        pass
    return [float(x) for x in best]
