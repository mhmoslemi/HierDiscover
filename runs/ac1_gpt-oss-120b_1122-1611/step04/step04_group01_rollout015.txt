"""
search_subgradient
------------------

A multi‑restart sub‑gradient optimiser for the max‑auto‑convolution objective.

The evaluation to be minimised is

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

and the optimiser works on the probability simplex (sum(a)=1).  At each
iteration the gradient of the active (max) convolution entry is computed
exactly (the sub‑gradient of the L∞ norm) via a single FFT, then a
projected sub‑gradient step (Adam‐style) is taken.  Random perturbations,
pairwise mass transfers and occasional restarts keep the search diverse.
Multiple random restarts with different lengths and initialisations are
performed until the time budget expires, and the best sequence found is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length able to hold a linear convolution of
    two vectors whose total length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all zero → uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random start vector on the simplex.

    Several families are mixed:
        – Dirichlet with random concentration,
        – geometric (exponential) decay,
        – sinusoidal power,
        – simple jittered uniform.
    """
    choice = rng.integers(0, 4)
    if choice == 0:                         # Dirichlet
        alpha = rng.uniform(0.3, 3.0)
        a = rng.gamma(alpha, 1.0, size=n)
        a = a / a.sum()
    elif choice == 1:                       # exponential (geometric)
        # pick a θ ∈ (0.5, 3) and set q = 1 – θ/n
        theta = rng.uniform(0.5, 3.0)
        q = 1.0 - theta / n
        q = max(min(q, 0.999999), 0.0)       # safety clamp
        a = ((1.0 - q) / (1.0 - q ** n)) * (q ** np.arange(n))
    elif choice == 2:                       # sin^p window
        p = rng.uniform(1.0, 5.0)
        a = np.sin(np.pi * np.arange(n) / (n - 1)) ** p
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
    else:                                   # uniform + tiny jitter
        a = np.full(n, 1.0 / n)
        jitter = np.exp(rng.normal(0.0, 0.05, size=n))
        a = a * jitter
        a = _project_onto_simplex(a)

    # final safety
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    return a


def _maxconv_and_score(a: np.ndarray, fft_len_conv: int) -> tuple[float, float]:
    """Return (max_conv, score) for a vector a (assumed sum(a)=1)."""
    conv = _conv_self(a, fft_len_conv)
    m = conv.max()
    score = 2.0 * a.size * m
    return m, score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # safety margin – leave a few seconds for the final return
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Parameters controlling the inner optimisation
    # ------------------------------------------------------------------
    max_local_iters = 3000          # iterations per restart
    eta_start = 0.8                 # initial learning rate
    eta_end = 0.005                 # final learning rate
    b1 = 0.9                        # Adam first‑moment decay
    b2 = 0.999                      # Adam second‑moment decay
    eps = 1e-8

    # probabilities for occasional moves
    block_perturb_prob = 0.02
    pair_move_prob = 0.08

    # ------------------------------------------------------------------
    # Main loop – keep launching new restarts until time runs out
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- pick a problem size ------------------------------------
        # length grows with time – start small, then try larger.
        elapsed = time.monotonic() - start_time
        # linearly increase the allowed length from 250 up to 4000
        max_len = int(250 + (3800 - 250) * (elapsed / budget_s))
        n = rng.integers(200, max_len + 1)

        # ---- initialise a candidate ----------------------------------
        a = _make_initial_sequence(n, rng)

        # ---- pre‑compute FFT lengths ---------------------------------
        fft_len_conv = _fft_len_for_len(2 * n - 1)          # for a*a
        fft_len_grad = _fft_len_for_len(3 * n - 2)          # for gradient

        # Adam bookkeeping
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0

        # ---- local optimisation --------------------------------------
        for it in range(max_local_iters):
            now = time.monotonic()
            if now >= deadline:
                break

            # convolution and current score
            conv = _conv_self(a, fft_len_conv)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                best_n = n
                print(f"[{now - start_time:6.2f}s] new best score = "
                      f"{best_score:.6f} (n={best_n})")
                # early exit when we already beat the target
                if best_score < 1.5025:
                    deadline = now   # stop outer loop early

            # ---- Adam step based on the exact sub‑gradient -------------
            # Determine the active (maximal) index
            k_max = int(conv.argmax())

            # sub‑gradient of max(conv) : for each j, grad_j = 2 * a_{k_max-j}
            # when the index is inside the support
            # implement via a single convolution with a one‑hot weight
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[k_max] = 1.0
            a_rev = a[::-1]
            grad_part = _conv_fft(w, a_rev, fft_len_grad)[n - 1:n - 1 + n]
            raw_grad = 2.0 * grad_part                     # sub‑gradient

            # Adam update (additive)
            t += 1
            m = b1 * m + (1.0 - b1) * raw_grad
            v = b2 * v + (1.0 - b2) * (raw_grad ** 2)
            m_hat = m / (1.0 - b1 ** t)
            v_hat = v / (1.0 - b2 ** t)

            # linear schedule for the learning rate
            progress = it / max_local_iters
            eta = eta_start * (1.0 - progress) + eta_end * progress

            step = eta * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step
            a = _project_onto_simplex(a)

            # ---- occasional random block perturbation ------------------
            if rng.random() < block_perturb_prob:
                width = rng.integers(1, max(2, n // 10))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.2))
                a[start:start + width] *= factor
                a = _project_onto_simplex(a)

            # ---- cheap pair‑wise mass transfer -------------------------
            if rng.random() < pair_move_prob:
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 1e-12:
                    max_delta = min(a[i], 0.4 * a[i])
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = _project_onto_simplex(a_candidate)

                    conv_cand = _conv_self(a_candidate, fft_len_conv)
                    cand_score = 2.0 * n * conv_cand.max()
                    if cand_score < cur_score:
                        a = a_candidate

        # after a restart we keep the best local solution automatically
        # the outer loop will launch a new one unless the budget is exhausted.

    # ------------------------------------------------------------------
    # Return the best sequence (clip to evaluate_sequence limits)
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely – fallback to a uniform vector
        best_seq = np.full(256, 1.0 / 256)
        best_n = 256

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
