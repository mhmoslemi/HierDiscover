def simplex_proj(v):
    # ensures sum(v) = 1, v_i >= 0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function via a two‑phase projected gradient / simulated annealing.
Phase 1 (coarse) – mirror‑descent on a soft‑max surrogate of the convolution
               with a decreasing stepsize and increasing temperature.
Phase 2 (fine)   – projected gradient on the true maximal convolution entry,
               plus occasional small random perturbations and pairwise mass
               transfers.  The current best solution is kept and used for
               occasional restarts.  The algorithm works on the probability
               simplex (non‑negative entries, sum = 1) and returns the best
               sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two length that can hold a full linear
    convolution of two vectors of length *n* (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Correlation used for the soft‑max surrogate.
    Computes  grad_j = 2·∑_k w_k·a_{k−j}."""
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project *v* onto the probability simplex {x ≥ 0, Σx = 1}."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise a starting vector (use previously found construction if
    #    available, otherwise fall back to the uniform distribution).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full(cand.size, 1.0 / cand.size)
                n = a.size
            else:
                raise ValueError
        except Exception:
            n = 3809               # a reasonable default length
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 3809
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Pre‑compute the FFT length for the self‑convolution (constant throughout)
    fft_len_conv = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_conv = _conv_self(best_a, fft_len_conv)
    best_score = 2.0 * n * best_conv.max()

    start_time = time.monotonic()
    # reserve a tiny safety margin so that we always return something
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters for the two phases
    # ------------------------------------------------------------------
    # Phase split (fraction of total budget spent in the coarse stage)
    coarse_frac = 0.6

    # ----- coarse phase -----
    eta_start_coarse = 0.8
    eta_end_coarse = 0.05
    beta_start_coarse = 0.5
    beta_end_coarse = 200.0
    block_prob_coarse = 0.05
    pair_move_prob_coarse = 0.10

    # ----- fine phase -----
    eta_start_fine = 0.05
    eta_end_fine = 0.001
    beta_start_fine = 200.0
    beta_end_fine = 2000.0
    block_prob_fine = 0.01
    pair_move_prob_fine = 0.05

    # Momentum used for smoothing the gradient estimate
    momentum = 0.8
    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # ----- current convolution and score -----
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----- keep the best solution seen so far -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- choose hyper‑parameters according to the current phase -----
        if frac < coarse_frac:
            eta_start, eta_end = eta_start_coarse, eta_end_coarse
            beta_start, beta_end = beta_start_coarse, beta_end_coarse
            block_prob = block_prob_coarse
            pair_prob = pair_move_prob_coarse
        else:
            eta_start, eta_end = eta_start_fine, eta_end_fine
            beta_start, beta_end = beta_start_fine, beta_end_fine
            block_prob = block_prob_fine
            pair_prob = pair_move_prob_fine

        # linear schedules
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- compute gradient (soft‑max surrogate early, hard max later) -----
        if beta < 500:                     # use smooth surrogate
            shift = cur_max
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum > 0.0:
                w_norm = w / w_sum
            else:
                w_norm = np.full_like(w, 1.0 / w.size)
            grad = _gradient_correlation(a, w_norm)
        else:                              # hard max gradient
            k_max = int(conv.argmax())
            j = k_max - np.arange(n)
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[j[mask]]

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad.copy()

        # ----- projected gradient descent step -----
        a_candidate = a - eta * grad
        a_candidate = _simplex_projection(a_candidate)

        # evaluate candidate (full convolution)
        conv_cand = _conv_self(a_candidate, fft_len_conv)
        cand_max = float(conv_cand.max())
        cand_score = 2.0 * n * cand_max

        # accept if improvement, otherwise possibly accept uphill (simulated annealing)
        if cand_score < cur_score:
            a = a_candidate
            conv = conv_cand
            cur_score = cand_score
            cur_max = cand_max
        else:
            # temperature decays linearly; we add a tiny constant to avoid division by zero
            temperature = max(0.01, (1.0 - frac)) * (best_score - cur_score + 1e-12)
            prob = np.exp(-(cand_score - cur_score) / temperature)
            if rng.random() < prob:
                a = a_candidate
                conv = conv_cand
                cur_score = cand_score
                cur_max = cand_max

        # ----- occasional random block perturbation -----
        if rng.random() < block_prob:
            # perturb a small contiguous block multiplicatively
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            # log‑normal factor – smaller sigma in the fine phase
            sigma = 0.15 if frac < coarse_frac else 0.05
            factor = np.exp(rng.normal(0.0, sigma))
            a_tmp = a.copy()
            a_tmp[start:start + width] *= factor
            a_tmp = np.clip(a_tmp, 0.0, None)
            a_tmp = _simplex_projection(a_tmp)
            a = a_tmp

        # ----- occasional pairwise mass transfer -----
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-14:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_tmp = a.copy()
                    a_tmp[i] -= delta
                    a_tmp[j] += delta
                    a_tmp = _simplex_projection(a_tmp)
                    conv_tmp = _conv_self(a_tmp, fft_len_conv)
                    score_tmp = 2.0 * n * conv_tmp.max()
                    if score_tmp < cur_score:
                        a = a_tmp
                        conv = conv_tmp
                        cur_score = score_tmp
                        cur_max = conv_tmp.max()

        # ----- soft restart if we have been stuck for a while -----
        if now - last_improve > 180.0:      # 3 minutes of stagnation
            a = best_a.copy()
            # add a tiny multiplicative jitter to escape flat regions
            jitter = np.exp(rng.normal(0.0, 0.01, size=n))
            a *= jitter
            a = _simplex_projection(a)
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score still {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clip to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
