def _project_onto_simplex(v):
    """Projects v onto the probability simplex."""
    # Ensure v shape (n,)
    v = v - np.max(v)  # shift for numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------
Search for a non‑negative step function that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a *mirror‑descent* on the probability simplex driven by
the **active‑set sub‑gradient** of the max‑convolution.
Only the convolution entries that are (almost) maximal are used to form
the gradient, so the update focuses on the constraints that actually
matter.  The iterates stay non‑negative and are re‑normalised to unit sum,
hence they respect the domain of ``evaluate_sequence``.
A small amount of random block perturbations and occasional restarts are
added to avoid getting stuck in poor local minima.
The routine runs for the allotted time (default 1000 s) and returns the
best sequence seen (clipped to the limits imposed by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _active_gradient(
    a: np.ndarray,
    conv: np.ndarray,
    n: int,
    max_active: int = 12,
    rel_eps: float = 1e-5,
) -> np.ndarray:
    """
    Gradient of the (non‑smooth) max‑convolution objective,
    restricted to the active set of indices where the convolution value
    is within ``rel_eps`` of the current maximum.
    The gradient is
        g_i = 2 * Σ_{k∈A} a_{k‑i}
    where A is the active index set.
    """
    max_val = conv.max()
    # keep indices whose value is close enough to the maximum
    thresh = max_val * (1.0 - rel_eps)
    active = np.where(conv >= thresh)[0]

    # If many entries are active, keep only the top few (they dominate)
    if active.size > max_active:
        # argpartition gives the largest `max_active` entries (unsorted)
        top_idxs = np.argpartition(-conv, max_active - 1)[:max_active]
        active = top_idxs

    grad = np.zeros_like(a)

    for k in active:
        if k < n:
            # contribution on positions 0 … k
            grad[: k + 1] += a[k::-1]
        else:
            # contribution on positions (k‑n+1) … (n‑1)
            offset = k - n + 1
            length = 2 * n - 1 - k
            grad[offset:] += a[:length][::-1]

    grad *= 2.0
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.18))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # reference: https://doi.org/10.1.1.269.6484
    v_shifted = v - np.max(v)  # improve numerical stability
    u = np.sort(v_shifted)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v_shifted - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 43, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights (the best found) that
    minimises ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine the length and initialise the vector a
    # ------------------------------------------------------------------
    default_n = 4096  # a convenient power‑of‑two size
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full_like(cand, 1.0 / cand.size, dtype=np.float64)
                n = a.size
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    # initial exact evaluation (cheap, because a is uniform)
    cur_est = 2.0 * n * _conv_self(a, fft_len).max()
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    last_improve = time.monotonic()
    start_time = last_improve

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 1.2   # initial step‑size (exponentiated‑gradient)
    eta_end = 0.02    # final step‑size
    momentum = 0.85   # smoothing of gradient
    grad_prev = np.zeros_like(a)

    block_perturb_prob = 0.02
    restart_gap = 25.0          # seconds without improvement → restart
    print_interval = 20.0       # seconds between status prints
    next_print = start_time + print_interval

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < start_time + max(0.0, float(budget_s) - 5.0):
        now = time.monotonic()
        elapsed = now - start_time
        total = (start_time + max(0.0, float(budget_s) - 5.0)) - start_time

        # linear schedule for η
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # --------------------------------------------------------------
        # convolution and current estimate
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        max_conv = float(conv.max())
        cur_est = 2.0 * n * max_conv / (np.sum(a) ** 2)   # sum(a)==1 → 2·n·max

        # --------------------------------------------------------------
        # keep the best record (exact evaluation only when promising)
        # --------------------------------------------------------------
        if cur_est < best_score - 1e-4:  # small margin before costly call
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best = a.copy()
                last_improve = now
                print(
                    f"[{elapsed:6.2f}s] new best (exact) score = {best_score:.6f}"
                )

        # --------------------------------------------------------------
        # periodic status output
        # --------------------------------------------------------------
        if now >= next_print:
            print(
                f"[{elapsed:6.2f}s] est. score = {cur_est:.6f}  best = {best_score:.6f}"
            )
            next_print = now + print_interval

        # --------------------------------------------------------------
        # compute gradient from the active set of convolution entries
        # --------------------------------------------------------------
        grad = _active_gradient(a, conv, n, max_active=12, rel_eps=1e-5)
        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # --------------------------------------------------------------
        # mirror‑descent (exponentiated‑gradient) update on the simplex
        # --------------------------------------------------------------
        factor = np.exp(-eta * grad)
        a = a * factor
        # numerical safety – clip negatives that could appear due to round‑off
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --------------------------------------------------------------
        # occasional random block perturbation
        # --------------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # restart if no improvement for a while
        # --------------------------------------------------------------
        if now - last_improve > restart_gap:
            # jitter the current best and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.13, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
