"""Propose a step function using a full‑softmax mirror‑descent
gradient (computed with two FFTs per iteration) followed by a
simulated‑annealing refinement.  The gradient is the exact derivative
of the soft‑max approximation of the ∞‑norm of the self‑convolution,
so every update respects the simplex constraints via an
exponentiated‑gradient (KL‑mirror) step.  The method runs within the
1000 s budget, keeps the best vector seen and returns it ready for
`evaluate_sequence`.
"""

import time
import numpy as np

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length suitable for a full linear convolution
    of two vectors of length *n*."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative vector *a* using FFT.
    Returns an array of length ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _grad_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Gradient of the weighted sum  ``∑_i w_i (a*a)_i`` with respect to *a*.
    Uses two FFTs (a and the reversed weight vector).  Returns a vector of
    length ``len(a)``."""
    n = a.size
    w_rev = w[::-1]
    # The full convolution length is n + len(w_rev) - 1 = 3*n-2.
    conv_len = n + w_rev.size - 1
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    conv_full = np.fft.irfft(fa * fw, n=fft_len)[:conv_len]
    # Slice the central block of length n, then reverse.
    block = conv_full[n - 1 : n - 1 + n]
    grad = 2.0 * block[::-1]
    return grad

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    # width between 1% and 10 % of the total length
    min_w = max(1, n // 100)
    max_w = max(2, n // 10)
    width = rng.integers(min_w, max_w + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

    current best bound.  The algorithm consists of:

    1. a smooth mirror‑descent phase that minimises a

    2. a short stochastic‑annealing phase that performs pairwise mass

    """

    # --------------------------------------------------------------

        # ----------------------------------------------------------

            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

    #    `evaluate_sequence` for safety).
    # ------------------------------------------------------------------
