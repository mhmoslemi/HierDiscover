def propose_candidate(seed=1234, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)

    # 1. Determine n and initial a.
    if 'height_sequence_1' in globals():
        a0 = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        a0 = np.clip(a0, 0, None)
        if a0.size > 0:
            n = a0.size
            a0 = a0 / a0.sum()
        else:
            n = 4096; a0 = np.full(n, 1/n)
    else:
        n = 4096; a0 = np.full(n, 1/n)

    # optionally increase length if n < 5000
    target_n = max(4000, n)
    if n < target_n:
        # pad zeros and renormalize
        a0 = np.concatenate([a0, np.zeros(target_n - n)])
        n = target_n

    fft_len_self = _next_pow2(2*n-1)

    # compute up-to-date best from a0
    conv = _self_conv_fft(a0, fft_len_self)
    best = a0.copy()
    best_score = 2.0*n*conv.max()
    cur = a0.copy()
    cur_score = best_score
    cur_conv = conv

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5) # small margin

    # Hardcoded hyperparameters
    T0 = 0.02
    T_end = 1e-6
    max_no_improve = 30.0  # seconds
    last_improve = start

    # stats
    iter_count = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        remaining = deadline - start
        frac = elapsed / remaining
        T = T0 * (T_end / T0) ** frac   # exponential decay

        # choose move type
        r = rng.random()
        if r < 0.7:
            # pairwise transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # donor i can give at most a_i * 0.3
            max_delta = a0[i] * 0.3
            if max_delta <= 0:
                continue
            delta = rng.random() * max_delta
            # create new candidate
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        elif r < 0.9:
            # block perturb
            max_width = max(1, n // 5)
            width = rng.integers(1, max_width+1)
            start_idx = rng.integers(0, n-width+1)
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start_idx:start_idx+width] *= factor
        else:
            # random noise injection (global)
            noise = np.exp(rng.normal(0.0, 0.14, size=n))
            cand = cur * noise

        # ensure non-negativity and renormalize
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand = np.full(n, 1.0 / n)
        else:

"""
Hybrid Simulated‑Annealing + Targeted Mass‑Transfer optimiser.

The routine starts from the best known step‑function (the global variable
``height_sequence_1`` if it exists, otherwise a uniform vector) and then
searches the simplex of non‑negative vectors with unit sum.  The objective
to be minimised is

    f(a) = 2·n·max(conv(a,a))          (sum(a)=1)

where ``conv`` denotes the linear self‑convolution.
The optimiser uses a mixture of moves:

  * **Targeted pairwise transfer** – the donor is chosen among the indices
    that contribute most to the current maximal convolution entry.
  * **Random pairwise transfer** – a classic donor/receiver move.
  * **Log‑normal block perturbation** – multiplicative jitter on a random
    contiguous block.
  * **Global log‑normal noise** – occasional reshuffling of the whole vector.

A simple exponential‑temperature schedule (simulated‑annealing) is used to
accept uphill moves early on while converging to a pure descent near the
budget end.  The best sequence seen so far is kept and periodically used
as a restart centre if no improvement occurs for a while.

All heavy linear algebra (self‑convolution) is performed with a single FFT,
so each iteration costs only O(n log n).  The implementation respects the
time‑budget (default 1000 s) and returns a plain Python list of floats.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)    # eliminate tiny negatives
    return conv_full


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects the given time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size == 0 or not np.isfinite(a0).all():
            a0 = None
    else:
        a0 = None

    if a0 is None:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a0.size
        a = a0.copy()
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length (power of two) for speed.
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Book‑keeping.
    # ------------------------------------------------------------------
    # current state
    cur = a.copy()
    cur_conv = _self_conv_fft(cur, fft_len_self)
    cur_score = 2.0 * n * cur_conv.max()

    # best state seen so far
    best = cur.copy()
    best_score = cur_score

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # annealing schedule (linear from T_start to T_end)
    T_start = 0.5          # temperature at the beginning
    T_end = 1e-5           # temperature close to zero at the end

    # restart parameters
    max_no_improve = 30.0          # seconds without improvement → restart
    last_improve = start

    # move‑type probabilities
    prob_targeted = 0.55
    prob_random_pair = 0.25
    prob_block = 0.15
    prob_global_noise = 0.05

    # maximum fraction of a donor’s mass that may be moved in a single step
    max_move_frac = 0.30

    iteration = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop (simulated annealing with targeted moves).
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / (deadline - start)          # 0 … 1
        T = T_start * (1.0 - frac) + T_end * frac    # linear cooling

        # ----- decide which move to perform -----
        r = rng.random()
        if r < prob_targeted:
            # ----- Targeted pairwise transfer (donor from the peak contribution) -----
            # Identify the index of the current maximal convolution entry
            k_max = int(np.argmax(cur_conv))
            # Valid i such that j = k_max - i lies in [0, n)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k_max - i_range

            # Contribution of each i to the peak (product a_i * a_j)
            contrib = cur[i_range] * cur[j_range]
            if contrib.size == 0:
                # fallback to a random pair
                i = rng.integers(0, n)
            else:
                donor = i_range[int(np.argmax(contrib))]
                i = int(donor)

            # choose a recipient different from donor
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)

            max_delta = cur[i] * max_move_frac
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        elif r < prob_targeted + prob_random_pair:
            # ----- Random pairwise transfer -----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            max_delta = cur[i] * max_move_frac
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        elif r < prob_targeted + prob_random_pair + prob_block:
            # ----- Log‑normal block perturbation -----
            max_width = max(1, n // 5)
            width = rng.integers(1, max_width + 1)
            start_idx = rng.integers(0, n - width + 1)
            factor = math.exp(rng.normal(0.0, 0.12))
            cand = cur.copy()
            cand[start_idx:start_idx + width] *= factor

        else:
            # ----- Global log‑normal noise (reshuffle) -----
            noise = np.exp(rng.normal(0.0, 0.14, size=n))
            cand = cur * noise

        # ----- Normalise and keep non‑negative -----
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand[:] = 1.0 / n
        else:
            cand /= s

        # ----- Evaluate candidate -----
        cand_conv = _self_conv_fft(cand, fft_len_self)
        cand_score = 2.0 * n * cand_conv.max()

        # ----- Accept / reject (simulated annealing) -----
        improve = cand_score < cur_score
        if improve or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur = cand
            cur_score = cand_score
            cur_conv = cand_conv

            # keep global best
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                last_improve = now
                print(f"[{now - start:6.2f}s] new best score = {best_score:.6f}")

        # ----- Restart if stuck for a while -----
        if now - last_improve > max_no_improve:
            # re‑initialise around the best solution with some noise
            cur = best.copy()
            noise = np.exp(rng.normal(0.0, 0.15, size=n))
            cur = cur * noise
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()
            cur_conv = _self_conv_fft(cur, fft_len_self)
            cur_score = 2.0 * n * cur_conv.max()
            last_improve = now
            print(f"[{now - start:6.2f}s] restart – best score {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence``).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
```

This implementation respects all the constraints, runs within the allotted 1000 s, and employs a different search strategy (targeted simulated‑annealing moves) that should explore the solution space more effectively than the previous pure exponentiated‑gradient approach, giving a good chance of beating the previous bound of 1.514166.
