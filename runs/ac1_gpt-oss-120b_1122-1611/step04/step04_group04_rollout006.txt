"""
propose_candidate
-----------------
Simple stochastic construction for the L∞ self‑convolution problem.
The algorithm draws many random sparse step functions (zeroes everywhere
except on a randomly chosen support) with uniform weight on the support.
For a random support the self‑convolution is close to uniform, so the
objective 2·n·max(a*a) is typically close to the theoretical lower bound 1.
The search keeps the best vector found within the allotted time budget.
A few occasional Dirichlet‑weighted trials are added to diversify the
search space.  All operations are performed with FFT‑based convolution
(≈ O(n log n) per trial) so thousands of candidates can be examined in the
~1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)‑1) entries.
    """
    # FFT of the zero‑padded vector
    Fa = np.fft.rfft(a, n=fft_len)
    # Element‑wise product (â·â) and inverse FFT
    conv_full = np.fft.irfft(Fa * Fa, n=fft_len)
    # Keep the linear convolution part
    conv = conv_full[: 2 * a.size - 1]
    # Numerical noise may give tiny negatives – clip.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    The method repeatedly samples sparse random supports, assigns uniform
    weights on the support (and occasionally Dirichlet weights), evaluates

    Returns the best sequence as a plain Python list of floats.
    """

            n = int(len(cand0))
        except Exception:
            n = 4096
    else:

    # Ensure the length is not trivially small – larger n gives a tighter bound.
    if n < 2000:
        n = 2000

    # Pre‑compute an FFT length that is a power of two (speed‑friendly)
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_seq = None

    start_time = time.monotonic()

    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:

        density = rng.uniform(0.20, 0.55)          # 20 % … 55 % non‑zeros
