Now define beta_shape:

def beta_shape(n, alpha):
    x = np.linspace(0, 1, n)
    # avoid endpoints zero - use eps
    eps = 1e-12
    vals = np.maximum(x * (1-x), eps) ** (alpha - 1)  # for alpha<1 yields high at edges
    # if alpha==1 produce uniform
    if np.all(vals == 0):
        vals = np.ones_like(x)
    vals = vals / np.sum(vals)
    return vals

But exponent for alpha < 1 leads to (x(1-x))^(α-1) diverging at edges. Use eps to clamp small values.

def exp_shape(n, r):
    idx = np.arange(n)
    # compute sym factor: min(i, n-1-i)
    dist = np.minimum(idx, n-1-idx)
    vals = r**dist
    vals = vals / np.sum(vals)
    return vals

def run_optimize(a0, time_budget, rng):
    a = a0.astype(np.float64)
    # ensure sum = 1
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s>0:
        a = a / s
    else:
        a = np.full_like(a, 1.0/a.shape[0])
    n = a.size
    fft_len_conv = _fft_len_for_len(2*n-1)
    fft_len_grad = _fft_len_for_len(3*n-2)
    # Adam hyperparameters
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    # compute initial
    conv = _conv_self_fft(a, fft_len_conv)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best_time = time.monotonic()
    start = best_time
    # schedule
    deadline = start + time_budget
    while time.monotonic() < deadline:
        # compute conv and gradient
        conv = _conv_self_fft(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # print improvement
            print(f"    [{time.monotonic() - start:.2f}s] improve {best_score:.6f}")
        # schedule parameters based on progress
        elapsed = time.monotonic() - start
        progress = elapsed / time_budget
        eta = (1.0 - progress) * 0.8 + progress * 0.005  # similar schedule but starting bigger
        beta = 0.1 + progress * (2000 - 0.1)
        # compute softmax weights
        w = np.exp(beta * (conv - cur_max))
        sum_w = w.sum()
        # compute gradient via convolution
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
        grad_part = conv_w_rev[n-1:n-1+n] / sum_w
        raw_grad = a + grad_part
        # momentum smoothing optional (not needed)
        # Adam update
        t += 1
        m = b1 * m + (1 - b1) * raw_grad
        v = b2 * v + (1 - b2) * (raw_grad ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # ensure nonnegative & project onto simplex
        a = _project_onto_simplex(a)
        # symmetrize
        a = (a + a[::-1]) / 2.0
        # occasional random perturb
        if rng.random() < 0.03:
            # random block multiply
            width = rng.integers(1, max(2, n // 8))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[start_idx:start_idx+width] *= factor
            a = _project_onto_simplex(a)
            a = (a + a[::-1]) / 2.0
        # occasional pair transfer larger delta
        if rng.random() < 0.1:
            i, j = rng.integers(0, n, size=2)
            if i!=j and a[i] > 0:
                max_delta = min(a[i], 0.5 * a[i])  # up to 50% move

                    a[i] -= delta
                    a[j] += delta
                    a = _project_onto_simplex(a)
                    a = (a + a[::-1]) / 2.0
        # occasional restart if stagnant
        # (optional)
        # break if too many iterations?
    return best_a, best_score

"""
Search for a step‑function (non‑negative heights) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv(a, a) is the linear convolution of the sequence with itself.
The algorithm works as follows:

1. Choose a set of diverse initial vectors (uniform, Dirichlet samples,
   symmetric beta‑shaped, symmetric exponential‑decay and, if present,
   the previously discovered best sequence).
2. For each initial vector run a short “Adam on the simplex’’ optimisation.
   The objective is smoothed with a soft‑max approximation of the max‑norm.
   The temperature β grows from a small value to a large one, so the
   optimiser eventually attacks the true peak.
3. After every Adam step the vector is projected onto the probability
   simplex and forced to be symmetric (a[i] = a[n‑1‑i]), which is natural
   for the convolution problem.
4. Random block perturbations, pairwise mass‑transfer moves and occasional
   restarts (jitter) are added to escape plateaux.
5. The best sequence seen over the whole budget is returned.

All operations use FFT‑based convolutions (O(n log n)) and run well within
the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT of length fft_len."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    # keep the exact linear‑convolution length
    return conv[: len(x) + len(y) - 1]


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] = a[n‑1‑i] (exact symmetry)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Simple parametric families (used to generate diverse starting points)
# ----------------------------------------------------------------------


def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape on n points, normalised to sum‑1."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    # density ∝ (x·(1‑x))^{α‑1}; add eps to avoid 0⁰
    vals = np.maximum(x * (1.0 - x), eps) ** (alpha - 1.0)
    # make it symmetric (already symmetric) and normalise
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals = vals / s
    return vals


def _exp_sequence(n: int, r: float) -> np.ndarray:
    """Symmetric exponential decay: a[i] ∝ r^{min(i, n‑1‑i)}."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)
    vals = r ** dist
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals = vals / s
    return vals


# ----------------------------------------------------------------------
# Core optimiser (Adam on the simplex with soft‑max smoothing)
# ----------------------------------------------------------------------


def _run_optimize(initial: np.ndarray,
                  time_budget: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Optimise a single start vector for at most *time_budget* seconds.
    Returns the best vector found and its associated score.
    """
    a = initial.astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a.fill(1.0 / a.size)

    n = a.size
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # Adam state
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # bookkeeping of the best point
    conv = _conv_self_fft(a, fft_len_conv)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best_a = a.copy()
    start = time.monotonic()
    deadline = start + time_budget

    # schedule parameters (tuned a little more aggressively)
    beta_start, beta_end = 0.2, 4000.0
    eta_start, eta_end = 1.0, 0.001

    while time.monotonic() < deadline:
        # ----- convolution & score ---------------------------------
        conv = _conv_self_fft(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # record improvement
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"    [{elapsed:6.2f}s] new local best = {best_score:.6f}")

        # ----- schedule --------------------------------------------
        now = time.monotonic()
        prog = (now - start) / time_budget
        prog = min(1.0, max(0.0, prog))
        eta = eta_start * (1.0 - prog) + eta_end * prog
        beta = beta_start + (beta_end - beta_start) * prog

        # ----- soft‑max weights ------------------------------------
        # shift by cur_max to avoid overflow
        w = np.exp(beta * (conv - cur_max))
        sum_w = w.sum()
        # ----- gradient of the smooth max -------------------------
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
        # slice the part that corresponds to Σ_i w_i a_{i‑j}
        grad_part = conv_w_rev[n - 1 : n - 1 + n] / sum_w
        raw_grad = a + grad_part                     # ∂fβ/∂a

        # ----- Adam update ----------------------------------------
        t += 1
        m = b1 * m + (1.0 - b1) * raw_grad
        v = b2 * v + (1.0 - b2) * (raw_grad ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)
        a = _symmetrize(a)

        # ----- occasional random block perturbation ---------------
        if rng.random() < 0.03:
            width = rng.integers(1, max(2, n // 8))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            a[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a)
            a = _symmetrize(a)

        # ----- occasional pair‑wise mass transfer -----------------
        if rng.random() < 0.12:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                max_delta = min(a[i], 0.5 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a[i] -= delta
                    a[j] += delta
                    a = _project_onto_simplex(a)
                    a = _symmetrize(a)

        # ----- simple restart if stagnation -----------------------
        if now - start > 180.0 and best_score < cur_score * 0.9999:
            # we have improved recently – keep going
            pass
        elif now - start > 240.0:
            # no decent progress, jitter around the current best
            a = best_a.copy()
            jitter = np.exp(rng.normal(0.0, 0.14, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            a = _symmetrize(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            start = time.monotonic()  # reset the “age’’ of this run
            print(f"    [{time.monotonic() - start:.2f}s] restart with jitter")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that yields the smallest possible
    evaluation score before the time budget expires.
    The routine samples several sensible initialisations and runs a
    projected Adam optimiser on each of them, keeping the best result.
    """
    rng = np.random.default_rng(seed)
    # give a safety margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Choose vector length – prefer the length of a previously found
    # sequence if it exists, otherwise fall back to 2500.
    # ------------------------------------------------------------------
    default_n = 2500
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a0 = np.clip(cand, 0.0, None)
                s = a0.sum()
                if s > 0.0:
                    a0 = a0 / s
                else:
                    a0 = np.full(n, 1.0 / n)
                initial_vectors = [a0]
            else:
                raise ValueError
        except Exception:
            n = default_n
            initial_vectors = []
    else:
        n = default_n
        initial_vectors = []

    # ------------------------------------------------------------------
    # Additional diverse starting points
    # ------------------------------------------------------------------
    initial_vectors.append(np.full(n, 1.0 / n))                     # uniform

    # symmetric beta shapes (α≈0.3 … 3)
    for _ in range(4):
        alpha = rng.uniform(0.2, 3.0)
        initial_vectors.append(_beta_sequence(n, alpha))

    # symmetric exponential shapes (r≈0.85 … 0.99)
    for _ in range(4):
        r = rng.uniform(0.85, 0.99)
        initial_vectors.append(_exp_sequence(n, r))

    # random Dirichlet samples (concentration 0.5 gives sparse vectors)
    for _ in range(6):
        a = rng.dirichlet(np.ones(n) * 0.5)
        initial_vectors.append(a)

    # ------------------------------------------------------------------
    # Run optimisation from each start, respecting the global deadline
    # ------------------------------------------------------------------
    best_overall = None
    best_score = float("inf")
    start_total = time.monotonic()

    for start_vec in initial_vectors:
        if time.monotonic() >= deadline:
            break
        # allocate a slice of the remaining budget (at least 8 s per run)
        remaining = deadline - time.monotonic()
        run_time = min(200.0, max(8.0, remaining / 2.0))
        a_opt, sc = _run_optimize(start_vec, run_time, rng)
        if sc < best_score:
            best_score = sc
            best_overall = a_opt.copy()
            elapsed = time.monotonic() - start_total
            print(f"[{elapsed:6.2f}s] *** global best score = {best_score:.6f}")

    # If everything failed, fall back to a simple uniform vector
    if best_overall is None:
        best_overall = np.full(n, 1.0 / n)

    # final safety clipping (evaluate_sequence caps at 0–1000)
    best_overall = np.clip(best_overall, 0.0, 1000.0)

    return [float(x) for x in best_overall]
