def _project_simplex(v):
    # v may be any real vector
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _pairwise_transfer_exact(a, conv, rng, max_delta_fraction=0.3):
    n = a.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n-1)
    if i == j:
        return a, conv, None
    # Choose delta up to max_delta_fraction of a_i, but also ensure not moving all mass.
    max_delta = min(a[i], max_delta_fraction * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    new_conv = conv.copy()

    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    new_conv[2*i] += delta * delta   # add delta^2

    lo_j = j

    new_conv[2*j] += delta * delta

    a2[i] -= delta

        return a2, new_conv, new_score

"""
Propose a candidate step function minimizing the evaluation metric.

The algorithm consists of two phases:

1. Smooth‑max gradient descent on the simplex.
   * Uses an exponentiated‑gradient (mirror descent) update.
   * The max‑convolution is approximated by a soft‑max (log‑sum‑exp) whose
     temperature β is increased from a small value to a very large one,
     so the algorithm first follows a smooth landscape and then focuses on
     the true L∞‑norm.
   * Momentum (exponential smoothing) stabilises the updates.
   * Occasional multiplicative block perturbations inject global noise.
   * If the solution has not improved for a while we add a larger random
     jitter and reset the momentum (a cheap “restart”).

2. Local refinement by exact pairwise mass transfers (simulated‑annealing style).
   * Maintains the full self‑convolution vector and updates it in O(n)
     time when moving a small amount of mass from index i to j.
   * Accepts improving moves always; uphill moves are accepted with a
     probability decreasing with a linear temperature schedule.
   * Complemented by occasional block perturbations.

Both phases respect the 1000 s budget, keep the coefficients non‑negative,
and normalise them to sum = 1 (the evaluation function is homogeneous, so
this does not affect optimality).  The best sequence found is returned as a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negative values caused by floating‑point noise.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Derivation: grad_j = 2·Σ_i a_i·w_{i+j}.
    Implemented as a correlation a ⋆ w using a single FFT.
    """
    n = a.size
    Lw = w.size                     # = 2*n‑1
    wr = w[::-1]                    # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)   # length for convolution a ⋆ wr

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)         # mapping to obtain correlation result
    grad = 2.0 * conv_full[idx]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to counter floating‑point drift.
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)

    return _project_onto_simplex(out)


def _pairwise_transfer_exact(a: np.ndarray, conv: np.ndarray,
                             rng: np.random.Generator,
                             max_delta_frac: float = 0.3):
    """
    Exact O(n) update for moving a random amount of mass from a donor index i
    to a receiver index j.  Returns (new_a, new_conv, new_score) if the move
    reduces the objective, otherwise the original tuple (a, conv, None).
    """
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if i == j:
        return a, conv, None

    # maximal transferable mass – keep a[i] ≥ 0 after the move
    max_delta = min(a[i], max_delta_frac * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Exact convolution update (see analysis for derivation)
    new_conv = conv.copy()
    # donor contribution removed
    lo_i = i
    hi_i = i + n
    new_conv[lo_i:hi_i] -= 2.0 * delta * a
    # receiver contribution added
    lo_j = j
    hi_j = j + n
    new_conv[lo_j:hi_j] += 2.0 * delta * a
    # diagonal corrections (+delta^2)
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    # cross‑term second‑order correction
    if i != j:
        new_conv[i + j] -= 2.0 * delta * delta

    # update the coefficient vector
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 = np.clip(a2, 0.0, None)
    a2 = _project_onto_simplex(a2)   # protects against rounding drift

    new_score = 2.0 * n * new_conv.max()
    cur_score = 2.0 * n * conv.max()
    if new_score < cur_score - 1e-15:      # strict improvement
        return a2, new_conv, new_score
    else:
        return a, conv, None


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best coefficient list found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size and initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = 4000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len_self = _next_pow2(2 * n - 1)          # for a*a

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _self_conv_fft(best, fft_len_self)
    best_score = 2.0 * n * conv_best.max()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (tuned for this budget)
    # ------------------------------------------------------------------
    # Phase division: 0.65 of time → smooth‑max descent,
    # the rest → exact pairwise refinement.
    phase_split = 0.65

    # Gradient‑descent schedules
    eta_start = 1.0
    eta_end = 0.01
    beta_start = 0.2
    beta_end = 4000.0
    momentum = 0.9
    block_perturb_prob = 0.03

    # Local‑search parameters
    temp_start = 0.015 * best_score   # initial temperature for SA
    temp_end = 1e-5 * best_score
    pairwise_prob = 0.4              # frequency of exact pairwise attempts

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # --------------------------------------------------------------
        # Phase decision
        # --------------------------------------------------------------
        if frac < phase_split:
            # ------------------- Smooth‑max Gradient Descent ---------------
            conv = _self_conv_fft(a, fft_len_self)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # keep best
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

            # schedules
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # soft‑max weights (stable)
            exp_terms = np.exp(beta * (conv - cur_max))
            sum_exp = exp_terms.sum()
            if sum_exp > 0:
                w = exp_terms / sum_exp          # proper soft‑max
            else:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

            # gradient (with momentum)
            grad = _gradient_from_weights(a, w)
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient update
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 1e-15, None)           # keep tiny positive mass
            a = _project_onto_simplex(a)

            # occasional random block perturbation
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # simple stagnation escape – add a bigger jitter if stuck
            if now - last_improve > 60.0:
                noise = np.exp(rng.normal(0.0, 0.2, size=n))
                a = a * noise
                a = _project_onto_simplex(a)
                grad_prev = np.zeros_like(a)
                last_improve = now
                print(f"[{elapsed:6.2f}s] jitter reset – best score {best_score:.6f}")

        else:
            # ------------------- Local refinement (SA) --------------------
            # Use the current best as the starting point for refinement.
            if 'ref_a' not in locals():
                # initialise refinement structures only once
                ref_a = best.copy()
                ref_conv = _self_conv_fft(ref_a, fft_len_self)
                ref_score = best_score
                # linear temperature schedule
                ref_start = now
                ref_deadline = deadline
                ref_last_improve = now

            # temperature based on linear schedule
            ref_elapsed = now - ref_start
            ref_total = ref_deadline - ref_start
            temp_frac = 1.0 - ref_elapsed / ref_total
            T = temp_start * temp_frac + temp_end * (1.0 - temp_frac)

            # attempt a pairwise transfer with probability `pairwise_prob`
            if rng.random() < pairwise_prob:
                ref_a, ref_conv, new_score = _pairwise_transfer_exact(
                    ref_a, ref_conv, rng, max_delta_frac=0.3)

                if new_score is not None:
                    # accepted improvement (deterministic)
                    ref_score = new_score
                    if ref_score < best_score:
                        best_score = ref_score
                        best = ref_a.copy()
                        last_improve = now
                        print(f"[{elapsed:6.2f}s] refinement improved to {best_score:.6f}")

                else:
                    # uphill move – accept with Metropolis probability
                    cur_score = 2.0 * n * ref_conv.max()
                    if cur_score > ref_score:
                        # (should not happen, sanity check)
                        ref_score = cur_score
                    delta = cur_score - ref_score
                    if delta > 0 and rng.random() < math.exp(-delta / max(T, 1e-12)):
                        # accept the uphill move
                        ref_score = cur_score
                        # keep current a and conv (already updated)
                    # else reject – keep previous state (nothing to do)

            # occasional block perturbation in refinement phase
            if rng.random() < 0.02:
                ref_a = _random_block_perturb(ref_a, rng)
                ref_conv = _self_conv_fft(ref_a, fft_len_self)
                ref_score = 2.0 * n * ref_conv.max()
                if ref_score < best_score:
                    best_score = ref_score
                    best = ref_a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] refinement perturb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
