def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # Choose n adaptively: if height_sequence_1 exists, use its length else default
    if "height_sequence_1" in globals():
        # Use length from existing perhaps but we can try larger
        n0 = len(globals()["height_sequence_1"])
        n = max(512, n0) # maybe increase
    else:
        n = 512

    # initial distribution: Dirichlet with concentration <1 to produce sparsity.
    a = rng.dirichlet(np.full(n, 0.3))
    # maybe scale for clipping: ensure values not >1000 (should be okay)
    a = np.clip(a, 0.0, None)  # all non-negative
    a = a / a.sum()

    # compute current best
    best_a = a.copy()
    best_score = compute_score(best_a)
    cur_score = best_score

    # annealing parameters:
    # initial temperature based on best_score magnitude
    T0 = best_score * 0.5
    Tmin = 1e-6
    # number of total iterations approximated by budget time // time per iteration? We'll just while loop.

    iteration = 0
    while time.monotonic() < deadline:
        # schedule temperature
        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)
        T = max(Tmin, T0 * (1 - frac))

        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # step size scaling decreasing
        base_step = 0.05  # step fraction of a[i] magnitude maybe
        eps = base_step * (1 - frac) * rng.random() * a[i]  # mass to move
        # ensure not move more than a[i]
        eps = min(eps, a[i])
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # renormalize to sum=1 (should still sum)
        # Because we moved mass, sum unchanged.
        # Compute score
        new_score = compute_score(a_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{elapsed:.2f}s] new best score {best_score:.6f}")

        iteration += 1

    # Return best_a as list
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]

def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = base.size
    else:
        n = 1024
    # parameters
    # Initialize logits from Dirichlet small concentration for diversity
    a0 = rng.dirichlet(np.full(n, 0.2))
    eps = 1e-12
    x = np.log(a0 + eps)
    # Precompute FFT length
    fft_len = _fft_len_for_n(n)
    # Adam variables
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t_iter = 0

    # schedule for beta (softmax temperature)
    beta_start = 0.1
    beta_end = 5000.0

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # best tracking
    # Compute initial a, conv, score
    a = np.exp(x)
    a = a / a.sum()
    conv = _conv_self(a, fft_len)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * cur_max
    best_time = start_time

    # restart handling
    last_improve = start_time
    restart_gap = 30.0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # update schedules:
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        # compute a
        a = np.exp(x)
        a = a / a.sum()
        # compute conv
        conv = _conv_self(a, fft_len)
        max_idx = int(conv.argmax())
        max_val = float(conv[max_idx])
        # weight w
        # use softmax smoothing, but also we might want almost indicator
        w_raw = np.exp(beta * (conv - max_val))
        w_sum = w_raw.sum()
        if w_sum > 0.0:
            w = w_raw / w_sum
        else:
            w = np.zeros_like(conv)
            w[max_idx] = 1.0
        # gradient wrt a (size n)
        grad_a = _gradient_correlation(a, w)  # this returns 2*... already

        # gradient wrt logits
        mu = np.dot(a, grad_a)
        grad_x = a * (grad_a - mu)

        # Adam step
        t_iter += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        # learning rate schedule
        eta0 = 0.5
        eta = eta0 * (1 - elapsed / total) + 0.01 * (elapsed / total)  # linearly down to 0.01
        step = eta * m_hat / (np.sqrt(v_hat) + epsilon)
        x = x - step

        # Occasionally add jitter to escape flat regions
        if rng.random() < 0.001:
            jitter = 0.1 * rng.normal(size=n)
            x = x + jitter

        # Compute new score
        a = np.exp(x)
        a = a / a.sum()
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score {best_score:.6f}")

        # restart if stuck
        if now - last_improve > restart_gap:
            # reinitialize with small noise around best
            x = np.log(best_a + eps)
            noise = np.exp(rng.normal(0.0, 0.2, size=n))
            x = np.log(best_a * noise + eps)
            # reset Adam moments
            m = np.zeros(n)
            v = np.zeros(n)
            t_iter = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart")

    # Return best vector clipped
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]

"""
Project‑Adam optimizer for the step‑function problem.

The objective to minimise is  

    F(a) = 2·n·max(conv(a,a)) / (sum(a))² ,

with a_i ≥ 0 and sum(a) > 0.01 (the scaling of a does not matter,
so we keep sum(a)=1).

The code works in the log‑softmax (log‑it) space:
    a_i = exp(x_i) / Σ_j exp(x_j) .
For the current maximum‑convolution index k* we have an exact
sub‑gradient

    ∂max(conv)/∂a_i = 2·a_{k*‑i}   (0‑based, out‑of‑range → 0),

which we obtain efficiently by a single FFT‑based correlation.
The gradient w.r.t. the logits x is

    g_x = a ∘ (g_a – μ) ,   μ = Σ_i a_i·g_a_i ,

where “∘” denotes element‑wise multiplication.
We apply Adam on x, periodically add small random jitter and restart
if the best value has not improved for a while.  The routine stops
shortly before the time budget expires and returns the best sequence
found.

The algorithm is self‑contained, uses only NumPy and the provided
`evaluate_sequence` for occasional verification, and respects all
security checks of that function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (taken from the reference implementation)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ w_k·conv_k w.r.t. a for a fixed weight vector w.
    Implements the correlation needed for ∂conv/∂a (see docstring)."""
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.

    The optimisation runs an Adam update on the log‑softmax parameters
    of the sequence, using the exact gradient of the maximal
    convolution entry.  Random jitter and restarts help to escape local
    minima.
    """
    rng = np.random.default_rng(seed)

    # ------------------- 1) problem size & initial vector -----------------
    # Use the length of the previously found construction if available,
    # otherwise fall back to a moderate size.
    height_seq = globals().get("height_sequence_1")
    if isinstance(height_seq, list) and len(height_seq) > 0:
        n = len(height_seq)
        # initialise from the previous construction (clamped & normalised)
        a0 = np.clip(np.asarray(height_seq, dtype=np.float64), 0.0, None)
        s0 = a0.sum()
        a0 = a0 / s0 if s0 > 0 else np.full(n, 1.0 / n)
    else:
        # a sparse random Dirichlet start – encourages non‑uniformity
        n = 1024
        a0 = rng.dirichlet(np.full(n, 0.2))

    eps = 1e-16                     # for safe log / division
    x = np.log(a0 + eps)            # log‑softmax parameters

    # ------------------- 2) pre‑computations ------------------------------
    fft_len_conv = _fft_len_for_n(n)

    # Adam moments
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8

    # bookkeeping
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin
    best_a = np.exp(x)
    best_a /= best_a.sum()
    conv = _conv_self(best_a, fft_len_conv)
    best_score = 2.0 * n * float(conv.max())
    best_time = start_time
    last_improve = start_time

    # schedule / hyper‑parameters
    iter_cnt = 0
    restart_gap = 30.0          # seconds without improvement → restart
    jitter_prob = 0.001         # probability of a random log‑jitter

    # ------------------- 3) optimisation loop ---------------------------
    while time.monotonic() < deadline:
        # ----- recompute the current probability vector -----------------
        a = np.exp(x)
        a /= a.sum()          # stay on the simplex

        # ----- convolution and index of the current maximum ------------
        conv = _conv_self(a, fft_len_conv)
        max_idx = int(conv.argmax())
        max_val = float(conv[max_idx])

        # ----- exact sub‑gradient of max(conv) w.r.t. a ----------------
        w = np.zeros_like(conv)
        w[max_idx] = 1.0               # one‑hot weight vector
        grad_a = _gradient_correlation(a, w)   # 2·a_{max_idx‑i}

        # ----- transform gradient to log‑softmax (logit) space ---------
        mu = np.dot(a, grad_a)
        grad_x = a * (grad_a - mu)     # g_x = a ∘ (g_a – μ)

        # ----- Adam update --------------------------------------------
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        # learning‑rate schedule: start larger, end small
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta_start, eta_end = 0.6, 0.03
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        step = eta * m_hat / (np.sqrt(v_hat) + adam_eps)
        x -= step

        # ----- occasional random jitter in log‑space --------------------
        if rng.random() < jitter_prob:
            jitter = 0.12 * rng.normal(size=n)
            x += jitter

        # ----- evaluate the new candidate -------------------------------
        a = np.exp(x)
        a /= a.sum()
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- restart if we have been stuck for a while ---------------
        if now - last_improve > restart_gap:
            # jump back to the best solution with a small multiplicative noise
            x = np.log(best_a + eps)
            noise = np.exp(rng.normal(0.0, 0.15, size=n))
            x = np.log(best_a * noise + eps)
            m.fill(0.0)
            v.fill(0.0)
            iter_cnt = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------- 4) finalise and return -------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)   # obey evaluate_sequence limits
    return [float(v) for v in best_a]
