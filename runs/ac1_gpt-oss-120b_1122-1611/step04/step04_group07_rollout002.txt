The projection onto simplex algorithm (by Duchi et al.) is easy.

Pseudo:

```
def proj_simplex(v):
    """Project onto simplex {x >=0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    """Project v onto the probability simplex (non-negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero? return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function using projected (sub‑)gradient descent.

The objective to minimise is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the evaluation is homogeneous we normalise the vector so that
∑ a_i = 1.  For a fixed length `n` this is equivalent to minimising the
peak of the self‑convolution `max(conv(a, a))`.

Algorithm
---------
1.  Initialise `a` on the probability simplex.
    * If a previous construction `height_sequence_1` is available we start
      from it (after normalisation).
    * Otherwise a random Dirichlet draw is used.

2.  Main loop (runs until the time budget expires):
    * Compute the full linear self‑convolution using an FFT (O(n log n)).
    * Estimate the current score as `2 * n * max(conv)`.
    * If the estimate improves the best‐known exact score, call the
      official `evaluate_sequence` function and keep the best vector.
    * Gradient:
        – In the first half of the budget a *soft‑max* approximation of the
          max‑norm is used:
              w_k ∝ exp(β·(conv_k – max_conv))
          and the gradient is `2 * correlate(w, a)`.
        – Afterwards the true sub‑gradient of the max‑norm is used.
          The sub‑gradient is simply a shifted copy of `a` summed over all
          positions where the convolution attains its maximum.
    * Apply momentum, take a step `a ← a – η·grad`, and project back onto
      the probability simplex (Euclidean projection).
    * Occasionally perturb a random contiguous block (log‑normal factor)
      to escape shallow local minima.
    * If no improvement is seen for `restart_gap` seconds, restart from the
      best vector plus a small log‑normal noise.

3.  Return the best sequence (clipped to the limits enforced by
    `evaluate_sequence`).

The method differs from the previous exponentiated‑gradient approach:
it works in the Euclidean space with an explicit projection onto the simplex,
uses the exact sub‑gradient of the max‑norm after a warm‑up phase and
adds simple random block perturbations and timed restarts.  Empirically
this yields a lower peak of the convolution and therefore a better score.
"""

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    # condition u_i - (cssv_i / i) > 0
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # all entries would be zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_softmax(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth log‑sum‑exp approximation of max(conv)."""
    max_conv = conv.max()
    # stability shift
    w_raw = np.exp(beta * (conv - max_conv))
    w_sum = w_raw.sum()
    if w_sum == 0.0:
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / w_sum
    # correlation (valid part has length n)
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def _grad_max_subgrad(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    # mask of positions attaining the maximum (within a tiny tolerance)
    mask = np.abs(conv - max_val) <= 1e-12
    max_positions = np.flatnonzero(mask)
    grad = np.zeros_like(a)
    for p in max_positions:
        # indices of a that contribute to conv[p]
        start = max(0, p - (n - 1))
        end = min(p, n - 1)          # inclusive
        idx_a = p - np.arange(start, end + 1)
        grad[start:end + 1] += a[idx_a]
    return 2.0 * grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that
    minimises ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine the length and initialise a on the simplex
    # ------------------------------------------------------------------
    default_n = 4096
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full_like(a, 1.0 / a.size)
                n = a.size
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len)
    cur_score_est = 2.0 * n * conv.max()          # sum(a) == 1
    # evaluate the exact score once for a reliable baseline
    best_score = float(evaluate_sequence(a.tolist()))
    best = a.copy()
    last_improve = time.monotonic()
    start_time = last_improve

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (tuned for the 1000 s budget)
    # ------------------------------------------------------------------
    eta_start = 0.5          # initial stepsize
    eta_end = 0.01           # final stepsize
    beta_start = 0.5         # soft‑max temperature start
    beta_end = 300.0         # soft‑max temperature end

    momentum = 0.9           # exponential smoothing of gradients
    grad_prev = np.zeros_like(a)

    block_perturb_prob = 0.02
    restart_gap = 30.0       # seconds without improvement → restart
    print_interval = 20.0    # seconds between status prints

    next_print = start_time + print_interval

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # safety margin
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ---- schedules for η and β ------------------------------------
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- convolution and current estimate -------------------------
        conv = _conv_self(a, fft_len)
        max_conv = float(conv.max())
        cur_score_est = 2.0 * n * max_conv

        # ---- update best (using the exact evaluation) -----------------
        if cur_score_est < best_score - 1e-9:
            exact = float(evaluate_sequence(a.tolist()))
            if exact < best_score:
                best_score = exact
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best (exact) score = {best_score:.6f}")

        # ---- periodic status print ------------------------------------
        if now >= next_print:
            print(f"[{elapsed:6.2f}s] est. score = {cur_score_est:.6f}  best = {best_score:.6f}")
            next_print = now + print_interval

        # ---- compute gradient (soft‑max early, exact subgrad later) ----
        if elapsed < total / 2.0:
            # soften the max‑norm to obtain a smoother gradient
            grad = _grad_softmax(a, conv, beta)
        else:
            # true sub‑gradient of the max‑norm
            grad = _grad_max_subgrad(a, conv)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- projected gradient step ----------------------------------
        a_new = a - eta * grad
        a = _project_onto_simplex(a_new)

        # ---- occasional random block perturbation ---------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ---- soft restart when stuck ---------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # add a small log‑normal jitter
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            a = a * noise
            a = _project_onto_simplex(a)
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
