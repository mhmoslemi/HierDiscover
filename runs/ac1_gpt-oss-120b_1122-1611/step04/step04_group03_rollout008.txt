"""
Propose candidate using an accelerated exponentiated‑gradient (mirror‑descent)
optimiser on the probability simplex.  The objective
    S(a) = 2·n·max(conv(a,a)) / (∑a)²
is scale‑invariant, so we keep ∑a = 1 and minimise 2·n·max(conv).
The optimiser works in three stages:

  1. A soft‑max approximation of the max operator with a temperature β that
     grows linearly from 0.1 to a large value (≈3000).  The resulting
     “soft‑max weights’’ give a smooth gradient that is cheap to compute
     via a single FFT.

  2. When β is large we switch to the exact sub‑gradient of the ‖·‖∞‑norm
     by selecting all shifts that are (numerically) maximal.  This sharpens
     the updates in the final phase.

  3. Randomised “jitter’’ moves (multiplicative log‑normal perturbations of
     a contiguous block), coordinate swaps and occasional soft‑restarts
     (re‑initialising from the current best with extra noise) help to
     escape local minima.

The routine continuously monitors the best solution and, when time allows,
re‑runs the whole optimiser on a larger support (up to ≈12000 entries);
larger supports give a tighter theoretical lower bound
2n/(2n−1) → 1, so expanding the dimension is the main source of improvement.

The function returns a list of non‑negative heights that (approximately) minimises
``evaluate_sequence`` within the allotted time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a∗a using a single FFT."""
    return _conv_fft(a, a)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    # Width is at most 10 % of the vector, at least 1.
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Core optimiser for a single length

def _optimise_length(
    n: int,
    rng: np.random.Generator,
    deadline: float,
    overall_start: float,
    total_budget: float,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Run the exponentiated‑gradient optimiser for a fixed support size ``n``.
    Returns the best sequence found (arranged on the simplex) and its score.
    """
    # initialise a on the simplex
    if init_seq is not None and init_seq.size > 0:
        # embed the initial sequence into a vector of length n
        a = np.full(n, 1.0 / n, dtype=np.float64)
        m = min(init_seq.size, n)
        a[:m] = init_seq[:m]
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a[:] = 1.0 / n
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # bookkeeping
    best = a.copy()
    cur_conv = _conv_self(a)
    best_score = 2.0 * n * cur_conv.max()
    best_time = time.monotonic()
    last_improve = best_time

    grad_prev = np.zeros_like(a)

    # hyper‑parameters (chosen to give a stronger focus on the max)
    eta_start = 1.0
    eta_end = 0.001
    beta_start = 0.1
    beta_end = 3000.0          # large → almost hard max
    momentum = 0.7
    block_perturb_prob = 0.08
    swap_prob = 0.07
    restart_gap = 35.0
    jitter_sigma = 0.25
    # after beta reaches this value we use the exact max‑subgradient
    beta_hard_threshold = 2500.0
    tol_max = 1e-12

    # pre‑compute FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)          # for a∗a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w∗a_rev

    # main loop
    while time.monotonic() < deadline:
        # ---- compute self‑convolution and current score -----------------
        conv = _conv_self(a)                      # length 2n‑1
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- keep the best --------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - overall_start
            print(f"[{elapsed:6.2f}s] n={n} new global best = {best_score:.6f}")

        # ---- schedule (linear interpolation) -------------------------
        now = time.monotonic()
        elapsed_total = now - overall_start
        progress = min(1.0, elapsed_total / total_budget)

        eta = eta_start * (1.0 - progress) + eta_end * progress
        beta = beta_start + (beta_end - beta_start) * progress

        # ---- compute soft‑max (or hard) weights -----------------------
        shift = cur_max
        if beta < beta_hard_threshold:
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # exact sub‑gradient: equal weight on all maximal entries
            mask = conv >= cur_max - tol_max * max(1.0, cur_max)
            w = mask.astype(float)
            cnt = w.sum()
            if cnt == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= cnt

        # ---- gradient of the smoothed (or hard) objective -------------
        a_rev = a[::-1]                           # reversed copy
        # convolution w * a_rev (full length 3n‑2)
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: (len(w) + len(a_rev) - 1)]
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # slice the relevant segment

        # ---- momentum smoothing ---------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- exponentiated‑gradient (mirror descent) update -----------
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -50.0, 50.0)  # protect against overflow
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- occasional random perturbations ---------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # ---- soft restart when progress stalls -------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - overall_start
            print(f"[{elapsed:6.2f}s] n={n} restart – best score {best_score:.6f}")

    return best, best_score

        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)

        # ---- occasional random perturbations ---------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # ---- soft restart when progress stalls -------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - overall_start
            print(f"[{elapsed:6.2f}s] n={n} restart – best score {best_score:.6f}")

    return best, best_score

# Entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # give the harness a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    overall_start = time.monotonic()
    total_budget = deadline - overall_start

    best_global_seq = None
    best_global_score = float("inf")
    best_global_n = 0

    # --------------------------------------------------------------
    # Choose a schedule of support sizes.
    # --------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    # Use the previously discovered sequence (if any) as a warm start for the
    # first run – it is often already stored in the global variable
    # ``height_sequence_1``.
    first_init = None
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0:
            first_init = seq0

    for idx, n in enumerate(lengths):
        if time.monotonic() >= deadline - 30.0:       # keep a safety buffer
            break

        # Decide which initial vector to use:
        if idx == 0 and first_init is not None:
            init_seq = first_init
        elif best_global_seq is not None and best_global_n <= n:
            init_seq = best_global_seq
        else:
            init_seq = None

        seq, score = _optimise_length(
            n=n,
            rng=rng,
            deadline=deadline,
            overall_start=overall_start,
            total_budget=total_budget,
            init_seq=init_seq,
        )

        if score < best_global_score:
            best_global_score = score
            best_global_seq = seq.copy()
            best_global_n = n
            elapsed = time.monotonic() - overall_start
            print(
                f"[{elapsed:6.2f}s] >>> NEW OVERALL BEST (n={n}) = {best_global_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final safety clipping – evaluate_sequence caps values to [0, 1000].
    # ------------------------------------------------------------------
    if best_global_seq is None:
        best_global_seq = np.full(512, 1.0 / 512, dtype=np.float64)

    best_global_seq = np.clip(best_global_seq, 0.0, 1000.0)
    return [float(x) for x in best_global_seq]
