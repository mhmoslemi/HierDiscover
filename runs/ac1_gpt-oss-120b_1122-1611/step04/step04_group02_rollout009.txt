"""
Propose a step‐function that (hopefully) beats the current bound ≈1.514.

Algorithm – two phases (≈80 % gradient descent, ≈20 % stochastic refinement)

Phase 1 – full‑softmax mirror descent.
    * The objective  f(a)=max_k (a*a)[k]  is approximated by a smooth
      log‑sum‑exp:  φβ(a)=log(∑_k exp(β·conv_k))/β .
    * The gradient of φβ is
          ∇φβ(a)=2·B(w)·a ,   w_k ∝ exp(β·(conv_k−max(conv))),
      where B(w) is the Hankel matrix with entries w_{i+j}.
      The product B(w)·a is a weighted convolution and is computed
      efficiently with FFTs.
    * An exponentiated‑gradient (mirror‑descent) update
          a←a·exp(−η·∇φβ) ,   a←a/∑a
      keeps the vector non‑negative and on the probability simplex.
    * β grows linearly from 1 → 800, η shrinks from 5 → 1e‑3.
      Momentum (0.9) smooths the noisy gradient.  Small log‑normal
      jitter is injected occasionally to avoid premature stagnation.

Phase 2 – fine‐grained stochastic local search.
    * Pairwise mass transfers (move a fraction of a donor entry to a recipient)
      are accepted with a Metropolis rule using a linearly decreasing temperature.
    * Occasionally a whole random block is jittered.

The routine respects the 1000 s budget, records the best vector seen and
returns it (clipped to the limits enforced by `evaluate_sequence`).
If a previous construction is available (global `height_sequence_1`) it is
used as the initial point, otherwise a Dirichlet draw of length 4000 is taken.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _pow2_ceil(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n(n: int) -> int:
    """FFT length for a full linear self‑convolution of a length‑n vector."""
    return _pow2_ceil(2 * n - 1)


def _fft_len_for_grad(n: int) -> int:
    """FFT length for the weighted‑gradient convolution (size 3·n‑2)."""
    return _pow2_ceil(3 * n - 2)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (full length 2·n‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max approximation.
    Implements   ∇φβ(a) = 2·∑_k w_k·∂conv_k/∂a  .
    The formula reduces to a weighted convolution:
          g_i = 2·∑_j w_{i+j}·a_j .
    Computed via FFT for O(n log n) cost.
    """
    n = a.size
    # reverse the weight vector once (w_rev[k] = w[2n‑2‑k])
    w_rev = w[::-1]
    L = _fft_len_for_grad(n)
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(fa * fw, n=L)[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)          # slice that yields g_i
    grad = 2.0 * conv_rev[idx]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Main search routine
# ----------------------------------------------------------------------
