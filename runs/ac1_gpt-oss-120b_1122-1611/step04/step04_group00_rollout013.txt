"""
propose_candidate
-----------------

A two‑phase optimiser for the “max‑convolution” score.

Phase 1 – fast first‑order optimisation.
    * The step heights are kept on the probability simplex
      (non‑negative and sum = 1) – the objective is homogeneous.
    * An Adam‑style optimiser is used.
    * The non‑smooth max is approximated by a soft‑max
      (log‑sum‑exp) with a temperature β that is annealed
      from a low to a high value.
    * A cheap FFT based convolution provides both the objective
      and the gradient of the smoothed loss.

Phase 2 – simulated‑annealing fine tuning.
    * Small mass transfers between two random positions are
      proposed and accepted either if they improve the true
      (hard) objective or with a Metropolis probability that
      slowly cools.
    * This helps escape the plateau where the soft‑max gradient
      provides no signal.

The routine keeps the best sequence seen so far and returns it
as a plain list of floats.  All numbers are non‑negative,
clipped to the 0‑1000 limits imposed by ``evaluate_sequence``.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of length *n*."""
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative vector *a* using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Gradient of Σ_i w_i·(a*a)_i w.r.t. *a*.
    Equivalent to 2·(w ⊛ a_rev) restricted to the original length."""
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (a list of non‑negative heights) that minimises
    ``evaluate_sequence`` under the given time budget.

    Returns
    -------
    list[float]
        The best sequence found (already clipped to the bounds required by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)
    # safety margin – leave a couple of seconds for finalisation
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1) Choose a working length.
    # ------------------------------------------------------------------
    # Use a length similar to the previous best, but allow a small
    # random variation to explore the trade‑off between n and the possible
    # structure of the solution.
    base_n = 3809
    n = int(base_n * rng.uniform(0.9, 1.1))
    n = max(1024, min(5000, n))          # keep it in a comfortable range

    # ------------------------------------------------------------------
    # 2) Initialise a – start from the previous best if it exists,
    #    otherwise from a random Dirichlet vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        if cand.size < n:
            cand = np.concatenate([cand, np.zeros(n - cand.size)])
        else:
            cand = cand[:n]
        a = _project_simplex(cand)
    else:
        a = rng.random(n)
        a = _project_simplex(a)

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT lengths that will be reused.
    # ------------------------------------------------------------------
    fft_len_conv = _fft_len_for_n(2 * n - 1)          # for the forward convolution
    fft_len_grad = _fft_len_for_n(3 * n - 2)          # for the gradient convolution

    # ------------------------------------------------------------------
    # 4) Book‑keeping of the best solution.
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len_conv)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)   # sum(a) == 1 --> 2·n·max(conv)
    best = a.copy()
    start_all = time.monotonic()

    # ------------------------------------------------------------------
    # 5) Phase 1 – Adam + soft‑max.
    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_it = 0

    # hyper‑parameters
    eta_start, eta_end = 0.8, 0.02                # learning‑rate schedule
    beta_start, beta_end = 0.5, 5000.0           # soft‑max temperature schedule
    block_perturb_prob = 0.03
    phase1_end = deadline - 300.0                # keep roughly the last 5 min for SA

    while time.monotonic() < phase1_end:
        # ----- current convolution & score -----
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)   # a.sum() stays = 1

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_all
            print(f"[{elapsed:6.2f}s] new best (phase 1) = {best_score:.6f}")

        # ----- schedules -----
        elapsed_phase = time.monotonic() - start_all
        total_phase = phase1_end - start_all
        frac = min(1.0, elapsed_phase / total_phase)
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start * (beta_end / beta_start) ** frac

        # ----- smooth‑max weights -----
        shift = conv.max()
        w_raw = np.exp(beta * (conv - shift))
        w = w_raw / w_raw.sum()            # normalised, sum to 1

        # ----- gradient of the smoothed objective -----
        grad = _gradient_from_weights(a, w, fft_len_grad)

        # ----- Adam momentum update -----
        t_it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_it)
        v_hat = v / (1.0 - beta2 ** t_it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projected additive step -----
        a_tmp = a - step
        a = _project_simplex(a_tmp)

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

    # ------------------------------------------------------------------
    # 6) Phase 2 – Simulated annealing on the true hard objective.
    # ------------------------------------------------------------------
    # The current a becomes the starting point for the SA phase.
    cur_a = a.copy()
    conv = _conv_self(cur_a, fft_len_conv)
    cur_score = 2.0 * n * conv.max() / (cur_a.sum() ** 2)

    # temperature schedule (linear decay)
    sa_start = time.monotonic()
    temp_start, temp_end = 0.5, 1e-4
    while time.monotonic() < deadline:
        elapsed_sa = time.monotonic() - sa_start
        total_sa = deadline - sa_start
        frac_sa = min(1.0, elapsed_sa / total_sa)
        temperature = temp_start * (1.0 - frac_sa) + temp_end * frac_sa

        # propose a small transfer of mass from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if cur_a[i] <= 0.0:
            continue
        # transfer up to 5 % of the source mass
        delta = cur_a[i] * rng.uniform(0.001, 0.05)
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # compute new hard score (no soft‑max)
        conv_new = _conv_self(new_a, fft_len_conv)
        new_score = 2.0 * n * conv_new.max() / (new_a.sum() ** 2)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur_a.copy()
                elapsed = time.monotonic() - start_all
                print(f"[{elapsed:6.2f}s] new best (SA)   = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (final safety clipping).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
