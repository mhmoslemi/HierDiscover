"""
Hybrid gradient+simulated-annealing optimizer for step‑function heights.

The algorithm maintains a probability vector `a` (non‑negative, sum = 1)
and repeatedly:
  * computes the self‑convolution via FFT,
  * builds a soft‑max approximation of the infinity‑norm,
  * takes an Adam‑style gradient step on the simplex,
  * proposes stochastic local moves (block scaling, pairwise mass transfer)
    which are accepted according to a temperature schedule (simulated annealing).
When no improvement is seen for a while the current solution is jittered
around the best found so far.  The routine runs until the supplied time
budget expires and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors whose total length is ``length``."""
    # The convolution length of vectors of lengths n and m is n+m-1.
    # We need the next power of two >= n+m-1.
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT.
    The result length is ``len(x)+len(y)-1`` (truncated to that)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Returns a new array."""
    # Implementation follows the algorithm of
    # Wang & Carreira‑Perpiñán (2013).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random local modifications
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10) + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _pairwise_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random positive amount of mass from one index to another,
    then project onto the simplex."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] <= 0.0:
        return a.copy()
    # Transfer at most 30 % of the source mass.
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a.copy()
    delta = rng.random() * max_delta
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    # The projection step guarantees feasibility (non‑negativity + sum = 1)
    return _project_onto_simplex(out)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- 7.3) Compute schedules ------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # Linear interpolation for learning‑rate and beta
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        temp = temp_start * (1.0 - elapsed / total) + temp_end * (elapsed / total)

        # ---- 7.4) Gradient of smooth max -------------------------------
        # Soft‑max weights (shifted by current maximum for numerical stability)
        w = np.exp(beta * (conv - cur_max))
        sum_w = w.sum()

        # Compute a_rev once
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
        # Extract the segment that corresponds to Σ_i w_i a_{i‑j}
        grad_part = conv_w_rev[n - 1 : n - 1 + n] / sum_w

        # Gradient of the smoothed objective (the +a term stems from the
        # derivative of the denominator 1 / (sum a)^2 when the sum is fixed to 1)
        raw_grad = grad_part + a

        # Momentum smoothing on the raw gradient (helps stability)
        raw_grad = grad_momentum * grad_prev + (1.0 - grad_momentum) * raw_grad
        grad_prev = raw_grad

        # ---- 7.5) Adam update (additive) --------------------------------
        t += 1
        m = b1 * m + (1.0 - b1) * raw_grad
        v = b2 * v + (1.0 - b2) * (raw_grad ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # ---- 7.6) Stochastic local moves with SA acceptance ------------
        # Block perturbation
        if rng.random() < block_prob:
            a_candidate = _random_block_perturb(a, rng)
            conv_cand = _conv_self(a_candidate, fft_len_conv)
            cand_score = 2.0 * n * conv_cand.max()
            delta = cand_score - cur_score
            accept = (delta < 0.0) or (rng.random() < np.exp(-delta / max(temp, 1e-12)))
            if accept:
                a = a_candidate
                conv = conv_cand
                cur_max = conv_cand.max()
                cur_score = cand_score
                # No need to recompute gradient immediately – it will be
                # recomputed at the beginning of the next loop iteration.

        # Pairwise transfer
        if rng.random() < pair_prob:
            a_candidate = _pairwise_transfer(a, rng)
            conv_cand = _conv_self(a_candidate, fft_len_conv)
            cand_score = 2.0 * n * conv_cand.max()
            delta = cand_score - cur_score
            accept = (delta < 0.0) or (rng.random() < np.exp(-delta / max(temp, 1e-12)))
            if accept:
                a = a_candidate
                conv = conv_cand
                cur_max = conv_cand.max()
                cur_score = cand_score

        # ---- 7.7) Restart when stuck ------------------------------------
        if now - last_improve > restart_gap:
            # Jitter around the best solution found so far.
            jitter = np.exp(rng.normal(0.0, 0.13, size=n))
            a = best * jitter
            a = _project_onto_simplex(a)
            # Reset Adam statistics
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            grad_prev.fill(0.0)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8) Return the best vector (apply the clamps that evaluate_sequence does)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Hybrid gradient + simulated‑annealing optimizer for step‑function heights.

The optimizer keeps a probability vector `a` (non‑negative, sum = 1) and
repeatedly:
  * evaluates the self‑convolution via FFT,
  * builds a soft‑max approximation of the ∞‑norm,
  * takes an Adam‑style gradient step on the simplex,
  * proposes stochastic local moves (block scaling, pairwise mass transfer)
    which are accepted according to a temperature schedule (simulated annealing),
  * restarts with a jitter around the best solution when progress stalls.

The routine runs until the supplied time budget expires and returns the best
sequence discovered.  All numbers are clipped to the limits required by
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors whose total length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT.
    The result length is ``len(x)+len(y)-1`` (truncated to that)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Returns a new array."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random local modifications
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    max_width = max(2, n // 10)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)


def _pairwise_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random amount of mass from one index to another,
    then project onto the simplex."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] <= 0.0:
        return a.copy()
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a.copy()
    delta = rng.random() * max_delta
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    return _project_onto_simplex(out)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    `evaluate_sequence` before the time budget elapses.

    The optimiser mixes an Adam‑style gradient descent on a smooth
    approximation of the max‑norm with a simulated‑annealing acceptance
    rule for stochastic local moves.  Periodic restarts (jitter around
    the best solution) help escape plateaus.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------------------
    # 1) Time management
    # ------------------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------------------
    # 2) Initial sequence (prefer a previously found construction if available)
    # ------------------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
                a = np.clip(cand0, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full_like(a, 1.0 / a.size)
            else:
                raise ValueError
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # No previous construction – start from a smooth half‑sine window.
        n = int(rng.integers(2500, 4000))
        x = np.arange(n, dtype=np.float64)
        a = np.sin(np.pi * (x + 0.5) / n) ** 1.5
        a = np.clip(a, 0.0, None)
        a = a / a.sum()

    n = a.size

    # ------------------------------------------------------------------------------
    # 3) Pre‑compute FFT lengths
    # ------------------------------------------------------------------------------
    fft_len_conv = _fft_len_for_len(2 * n - 1)      # for a*a
    fft_len_grad = _fft_len_for_len(3 * n - 2)      # for w * a_rev

    # ------------------------------------------------------------------------------
    # 4) Book‑keeping
    # ------------------------------------------------------------------------------
    conv = _conv_self(a, fft_len_conv)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max                     # sum(a) == 1
    best = a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------------------
    # 5) Adam hyper‑parameters
    # ------------------------------------------------------------------------------
    beta_start = 0.5
    beta_end = 1500.0          # large final β for a sharper max‑approximation
    eta_start = 0.7
    eta_end = 0.002
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # ------------------------------------------------------------------------------
    # 6) Simulated‑annealing schedule
    # ------------------------------------------------------------------------------
    temp_start = 0.5
    temp_end = 1e-4

    # ------------------------------------------------------------------------------
    # 7) Miscellaneous probabilities and parameters
    # ------------------------------------------------------------------------------
    block_prob = 0.05          # probability of trying a block perturbation
    pair_prob = 0.12           # probability of trying a pairwise mass transfer
    restart_gap = 30.0         # seconds without improvement → restart
    grad_prev = np.zeros_like(a)
    grad_momentum = 0.85

    # ------------------------------------------------------------------------------
    # 8) Main optimisation loop
    # ------------------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 8.1) Re‑evaluate current solution ------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ---- 8.2) Record the best found so far ------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- 8.3) Compute schedules -----------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # Linear interpolation for learning‑rate, β and temperature
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        temp = temp_start * (1.0 - elapsed / total) + temp_end * (elapsed / total)

        # ---- 8.4) Gradient of the smooth max -------------------------------
        # Soft‑max weights (shifted by current maximum for numerical stability)
        w = np.exp(beta * (conv - cur_max))
        sum_w = w.sum()
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
        grad_part = conv_w_rev[n - 1 : n - 1 + n] / sum_w
        raw_grad = grad_part + a                     # derivative of the smoothed objective
        raw_grad = grad_momentum * grad_prev + (1.0 - grad_momentum) * raw_grad
        grad_prev = raw_grad

        # ---- 8.5) Adam step -------------------------------------------------
        t += 1
        m = b1 * m + (1.0 - b1) * raw_grad
        v = b2 * v + (1.0 - b2) * (raw_grad ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # ---- 8.6) Stochastic local moves (SA acceptance) -------------------
        # Block perturbation
        if rng.random() < block_prob:
            a_cand = _random_block_perturb(a, rng)
            conv_cand = _conv_self(a_cand, fft_len_conv)
            cand_score = 2.0 * n * conv_cand.max()
            delta = cand_score - cur_score
            accept = (delta < 0.0) or (rng.random() < np.exp(-delta / max(temp, 1e-12)))
            if accept:
                a = a_cand
                conv = conv_cand
                cur_max = conv_cand.max()
                cur_score = cand_score

        # Pairwise transfer
        if rng.random() < pair_prob:
            a_cand = _pairwise_transfer(a, rng)
            conv_cand = _conv_self(a_cand, fft_len_conv)
            cand_score = 2.0 * n * conv_cand.max()
            delta = cand_score - cur_score
            accept = (delta < 0.0) or (rng.random() < np.exp(-delta / max(temp, 1e-12)))
            if accept:
                a = a_cand
                conv = conv_cand
                cur_max = conv_cand.max()
                cur_score = cand_score

        # ---- 8.7) Restart when progress stalls -----------------------------
        if now - last_improve > restart_gap:
            # Jitter around the best solution seen so far.
            jitter = np.exp(rng.normal(0.0, 0.13, size=n))
            a = best * jitter
            a = _project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            grad_prev.fill(0.0)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------------------
    # 9) Return the best solution (apply the same clipping that evaluate_sequence does)
    # ------------------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    total_elapsed = time.monotonic() - start_time
    print(f"[{total_elapsed:6.2f}s] final best score = {best_score:.6f}")
    return [float(x) for x in best]
