        # gradient for weighted sum (size n)
        a_rev = a[::-1]
        L = (2 * n - 1) + n - 1
        fft_len_grad = _fft_len_for_len(L)
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        grad_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)
        grad = 2.0 * grad_full[n-1:n-1+n] - a * w[2*np.arange(n)]

        # gradient of objective F = 2*n * softmaxBeta(conv)
        grad_f = 2.0 * n * grad

        # gradient wrt logits u
        dot = np.dot(a, grad_f)
        grad_u = a * (grad_f - dot)

        # Adam update
        t_iter += 1
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        u -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random block perturbation
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            a_pert = a.copy()
            a_pert[start:start+width] *= factor
            a_pert = np.clip(a_pert, 0, None)
            a_pert /= a_pert.sum()
            u = np.log(a_pert + 1e-30)

        # occasional pair move
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta_max = min(a[i], 0.2 * a[i])
                if delta_max > 0:
                    delta = rng.random() * delta_max
                    a_pair = a.copy()
                    a_pair[i] -= delta
                    a_pair[j] += delta
                    a_pair = np.clip(a_pair, 0, None)
                    a_pair /= a_pair.sum()
                    # recompute conv only for acceptance test
                    conv_pair = _conv_self(a_pair, fft_len_conv)
                    score_pair = 2.0 * n * conv_pair.max()
                    if score_pair < cur_score:
                        # accept by updating u
                        u = np.log(a_pair + 1e-30)

        # restart if stalled
        if now - last_improve > restart_gap:
            # perturb best and restart
            a_restart = best.copy()
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            a_restart = a_restart * noise
            a_restart = np.clip(a_restart, 0, None)
            a_restart /= a_restart.sum()
            u = np.log(a_restart + 1e-30)
            m[:] = 0.0
            v[:] = 0.0
            t_iter = 0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # final best clip
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

propose_candidate
-----------------

Search for a non‑negative step function that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

by optimisation on the probability simplex.  The algorithm works in the
log‑softmax (logit) space and uses an Adam optimiser on a smooth
log‑sum‑exp surrogate of the max‑norm.  The surrogate weights are obtained
from a softmax over the convolution entries, whose gradient is computed by a
single extra FFT‑based convolution.  A simple linear schedule for the
learning‑rate and the soft‑max temperature is applied, together with
occasional random block perturbations, pairwise mass transfers and a
restart mechanism when progress stalls.  The best vector seen during the

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to a unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out

    default_n = 4000  # a comfortable size for the time budget
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a0 = np.clip(cand, 0.0, None)
                s = a0.sum()
                if s > 0.0:
                    a0 = a0 / s
                else:
                    a0 = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # arcsine‑shaped start (often better than uniform)
    idx = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
    arcsine = np.clip(arcsine, 0.0, None)
    arcsine /= arcsine.sum()

    # pick whichever start is currently better (cheap evaluation)
    fft_len_n = _fft_len_for_n(n)
    conv_uniform = _conv_self(a0, fft_len_n)
    conv_arc = _conv_self(arcsine, fft_len_n)
    score_uniform = 2.0 * n * conv_uniform.max()
    score_arc = 2.0 * n * conv_arc.max()
    if score_arc < score_uniform:
        a0 = arcsine.copy()
        best_score = score_arc
    else:
        best_score = score_uniform

    # log‑softmax representation (logits)
    u = np.log(a0 + 1e-30)               # small epsilon avoids log(0)

    # ------------------------------------------------------------------
    # 2) Optimisation bookkeeping
    # ------------------------------------------------------------------
    best_a = np.exp(u - np.max(u))
    best_a /= best_a.sum()

    m = np.zeros_like(u)  # first moment (Adam)
    v = np.zeros_like(u)  # second moment (Adam)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_iter = 0

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters / schedules
    # ------------------------------------------------------------------
    eta_start, eta_end = 0.3, 0.001      # learning‑rate schedule
    beta_start, beta_end = 0.5, 2000.0   # soft‑max temperature schedule
    restart_gap = 30.0                   # seconds without improvement → restart
    block_perturb_prob = 0.02
    pair_move_prob = 0.05

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ------- current probabilities (soft‑max) -----------------------
        u_shift = u - np.max(u)                     # prevent overflow
        a = np.exp(u_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ------- convolution and current score -------------------------
        conv = _conv_self(a, fft_len_n)            # length 2n‑1
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ------- keep the best solution --------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------- schedules --------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ------- soft‑max weights (log‑sum‑exp surrogate) -------------
        max_conv = cur_max
        shifted = conv - max_conv                    # shift for stability
        w_unnorm = np.exp(beta * shifted)
        Z = w_unnorm.sum()
        w = w_unnorm / Z                             # normalized soft‑max

        # ------- gradient of the weighted sum -------------------------
        a_rev = a[::-1]
        L = (2 * n - 1) + n - 1                      # length needed for convolution
        fft_len_grad = _fft_len_for_len(L)

        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        grad_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)

        # slice the n components that correspond to the gradient entries
        grad = 2.0 * grad_full[n - 1 : n - 1 + n] - a * w[2 * np.arange(n)]

        # gradient of the surrogate objective  f = 2*n*log‑sum‑exp(conv)
        grad_f = 2.0 * n * grad

        # ------- gradient w.r.t. logits (soft‑max Jacobian) -----------
        dot = np.dot(a, grad_f)
        grad_u = a * (grad_f - dot)

        # ------- Adam update -----------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        u -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # ------- occasional random block perturbation -----------------
        if rng.random() < block_perturb_prob:
            a_pert = _random_block_perturb(a, rng)
            u = np.log(a_pert + 1e-30)

        # ------- occasional pair‑wise mass transfer --------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_delta = min(a[i], 0.2 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_pair = a.copy()
                    a_pair[i] -= delta
                    a_pair[j] += delta
                    a_pair = np.clip(a_pair, 0.0, None)
                    a_pair /= a_pair.sum()
                    conv_pair = _conv_self(a_pair, fft_len_n)
                    score_pair = 2.0 * n * conv_pair.max()
                    if score_pair < cur_score:
                        u = np.log(a_pair + 1e-30)

        # ------- soft restart if progress stalls -----------------------
        if now - last_improve > restart_gap:
            # start from the current best with a mild jitter
            a_restart = best_a.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a_restart *= jitter
            a_restart = np.clip(a_restart, 0.0, None)
            a_restart /= a_restart.sum()
            u = np.log(a_restart + 1e-30)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
