"""
Simulated‑annealing optimizer for the step‑function evaluation.
----------------------------------------------------------------
We work on the probability simplex (∑a_i = 1, a_i ≥ 0) because the
evaluation function

    score = 2 * n * max(conv(a,a)) / (∑a)^2

is invariant under a common scaling of the coefficients.
The algorithm repeatedly proposes cheap moves (pairwise mass transfer,
block scaling or a global jitter), evaluates the (fast FFT‑based)
self‑convolution, and accepts the move either if it improves the
score or with a Metropolis probability that gradually cools down.
Occasional stagnation triggers a restart from the best solution seen
so far with a fresh random perturbation.  The best vector found before
the time budget expires is returned (clipped to the limits enforced by
``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors whose total length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Returns a new array.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose an initial vector – use the supplied one if it looks OK
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                a = a / s if s > 0 else np.full(n, 1.0 / n)
            else:
                raise ValueError
        except Exception:
            n = 3000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 3000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute the FFT length for the self‑convolution
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_len(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Helper to evaluate the objective (the factor 2·n is kept)
    # ------------------------------------------------------------------
    def _score(vec):
        conv = _conv_self(vec, fft_len)
        return 2.0 * n * conv.max()

    cur_score = _score(a)
    best = a.copy()
    best_score = cur_score

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # a tiny safety margin
    last_improve = start

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing schedule parameters
    # ------------------------------------------------------------------
    temp_start = 0.06                 # initial temperature
    temp_end = 0.0004                 # final temperature
    restart_gap = 30.0                # seconds without improvement → restart
    prob_transfer = 0.55              # how often we do a pairwise move
    prob_block = 0.30                 # how often we scale a random block
    # remaining probability → global jitter

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        # linear temperature decay
        temp = temp_start * (1.0 - elapsed / total) + temp_end * (elapsed / total)

        # ----- propose a candidate move ---------------------------------
        r = rng.random()
        if r < prob_transfer:                     # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-12:
                delta = rng.random() * a[i] * 0.5
                a_cand = a.copy()
                a_cand[i] -= delta
                a_cand[j] += delta
            else:
                a_cand = a
        elif r < prob_transfer + prob_block:       # random block perturbation
            width = rng.integers(1, max(2, n // 12))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a_cand = a.copy()
            a_cand[start_idx:start_idx + width] *= factor
            a_cand = np.clip(a_cand, 0.0, None)
            a_cand = _project_onto_simplex(a_cand)
        else:                                      # global jitter
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            a_cand = a * jitter
            a_cand = _project_onto_simplex(a_cand)

        # ----- evaluate and possibly accept ----------------------------
        new_score = _score(a_cand)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / (temp + 1e-12)):
            a = a_cand
            cur_score = new_score

        # ----- keep the best seen -------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now - start:6.2f}s] new best score = {best_score:.6f}")

        # ----- restart if we are stuck --------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            cur_score = _score(a)
            last_improve = now
            print(f"[{now - start:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best vector (final clipping respects evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
