"""
Hybrid simulated‑annealing + mirror‑descent optimizer.

The search works on the probability simplex (∑a = 1, a_i ≥ 0).  The
objective that evaluate_sequence finally returns (with a normalised
vector) is

    F(a) = 2·n·max(conv(a,a))

where n = len(a).  The algorithm therefore tries to minimise the
maximum entry of the self‑convolution.

Strategy
--------
* Initialise from the best known construction (`height_sequence_1`);
  otherwise start from a uniform distribution.
* Perform a classic Metropolis simulated‑annealing loop.  The proposal
  operators are cheap, element‑wise multiplicative jitter,
  block‑wise jitter, a transfer of a random amount of mass between two
  coordinates and a simple swap.  The temperature is linearly cooled
  from `T0` to a tiny `T_end`.
* Whenever a new best vector is found it is printed.
* Every few thousand iterations a short mirror‑descent (exponentiated‑
  gradient on the simplex) refinement is applied to the current best
  vector – this is the same “soft‑max mirror descent” used in the
  reference implementation but with a more aggressive schedule.
* After a long period without improvement the algorithm restarts from
  the current best with a small multiplicative log‑normal jitter.

The whole routine runs for the allotted budget (default 1000 s, a few
seconds left as safety margin) and finally returns the best vector
found, clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (pure functions, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT."""
    n = a.size
    # length needed for the full convolution
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Compute the evaluation function for a normalised vector (∑a = 1).

    Returns 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = _conv_self(a)
    return 2.0 * n * float(conv.max())


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator,
                         sigma: float = 0.15) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _mirror_descent_refine(a: np.ndarray,
                           steps: int = 200,
                           eta_start: float = 0.5,
                           eta_end: float = 0.01,
                           beta_start: float = 4.0,
                           beta_end: float = 200.0) -> np.ndarray:
    """
    A few iterations of the soft‑max mirror‑descent from the reference
    implementation, used as a local optimiser.
    """
    n = a.size
    a = a.copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s

    for step in range(steps):
        # temperature for the soft‑max (smooth approximation of max)
        beta = beta_start + (beta_end - beta_start) * (step / steps)

        # self‑convolution
        conv = _conv_self(a)

        # soft‑max weights
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(w * a_rev) on the appropriate slice
        a_rev = a[::-1]
        L = len(w) + len(a_rev) - 1
        N = _next_pow2(L)
        fw = np.fft.rfft(w, n=N)
        fa_rev = np.fft.rfft(a_rev, n=N)
        conv_rev = np.fft.irfft(fw * fa_rev, n=N)[:L]
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # stepsize schedule (linear)
        eta = eta_start * (1.0 - step / steps) + eta_end * (step / steps)

        # exponentiated‑gradient (mirror descent) update on the simplex
        log_factor = -eta * grad
        # keep exponent numerically stable
        log_factor = np.clip(log_factor, -50.0, 50.0)
        factor = np.exp(log_factor)
        a *= factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a

    The algorithm combines simulated annealing with occasional mirror‑descent
    refinements and a restart mechanism.

    # -----------------------------------------------------------------

    block_sigma = 0.12       # block perturb jitter
    transfer_fraction = 0.25  # maximal fraction of mass moved (scaled by T)
    block_prob = 0.30        # probability of block perturb
    jitter_prob = 0.30       # probability of global jitter
    transfer_prob = 0.30     # probability of a mass‑transfer move
    swap_prob = 0.10         # probability of a trivial swap

    # -----------------------------------------------------------------

        r = rng.random()

                # amount scales with the current temperature
                max_move = cand[i] * (T / T0) * transfer_fraction
                move = rng.random() * max_move
                j = rng.integers(0, n)
                cand[i] -= move
                cand[j] += move

        else:
            # simple swap of two coordinates
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # Renormalise onto the simplex
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand[:] = 1.0 / n
        else:
            cand /= s

        # ---------- evaluate the proposal ---------------------------
        cand_score = _objective(cand)

        # ---------- Metropolis acceptance ----------------------------
        if cand_score < cur_score:
            accept = True
        else:
            # probability = exp(-(cand_score - cur_score) / T)
            delta = cur_score - cand_score
            prob = np.exp(delta / T) if T > 0 else 0.0
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score

        # ---------- keep global best ---------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---------- occasional gradient refinement --------------------
        max_iter += 1
        if max_iter % refine_every == 0:
            # use the current best as a starting point for a short mirror‑descent
            best = _mirror_descent_refine(best,
                                          steps=200,
                                          eta_start=0.8,
                                          eta_end=0.005,
                                          beta_start=2.0,
                                          beta_end=500.0)
            best_score = _objective(best)
            cur = best.copy()
            cur_score = best_score
            if time.monotonic() < deadline:
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] after refinement best = {best_score:.6f}")

        # ---------- restart after long stagnation --------------------
        if time.monotonic() - last_improve > 45.0:
            # restart from the best seen so far, add a tiny jitter
            jitter = np.exp(rng.normal(0.0, 0.05, size=n))
            cur = best * jitter
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()
            cur_score = _objective(cur)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5) Return the best solution found (clipped to evaluate_sequence limits)
    # -----------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
