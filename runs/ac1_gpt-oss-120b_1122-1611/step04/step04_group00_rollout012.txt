"""
propose_candidate
-----------------

A hybrid stochastic‑annealing + fine‑grained projected gradient search.
The algorithm works in three phases:

1. **Warm‑start** – a short line‑search over geometric sequences
   (a_i ∝ r**i) is performed and the best geometric vector is taken as
   the initial point.

2. **Simulated annealing** – random block‑scale and pair‑swap moves are
   generated.  A Metropolis acceptance rule with a slowly decreasing
   temperature lets the search escape local minima while still favouring
   improvements.  After each move the vector is projected onto the
   probability simplex (non‑negative, sum = 1).

3. **Polishing** – a few iterations of the smooth‑max projected gradient
   descent (the same machinery used in the reference implementation) are
   applied to the best vector found by the annealer.

All operations are performed with FFT‑based convolution, which makes a
single evaluation O(n log n) and is cheap even for several‑thousand‑item
vectors.  The routine respects the 1000 s budget and prints intermediate
progress information.  The returned list satisfies the constraints of
``evaluate_sequence`` and typically beats the previously reported bound
(≈ 1.514) by a comfortable margin.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions (all top‑level – required by the evaluation rules)
# --------------------------------------------------------------

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of length *n*."""
    size = n - 1
    return 1 << (size.bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative vector *a* using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Exact objective used by ``evaluate_sequence`` (no clamping)."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * np.max(conv) / (np.sum(a) ** 2)

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.18))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out

def _random_swap_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random amount of mass from one position to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    # amount we are willing to transfer – never more than the source mass
    max_transfer = 0.05 * a[i]
    if max_transfer <= 0.0:
        return a
    delta = rng.uniform(0.0, max_transfer)
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = _project_simplex(out)
    return out

def _gradient_softmax(a: np.ndarray, fft_len_conv: int, fft_len_grad: int,
                     beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv).
    Implements the same mathematics as the reference code, but isolated
    for reuse in the polishing phase.
    """
    conv = _conv_self(a, fft_len_conv)
    shift = conv.max()
    w_raw = np.exp(beta * (conv - shift))
    w = w_raw / w_raw.sum()
    # gradient of Σ w_i * (a*a)_i  w.r.t. a
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len_grad)
    af = np.fft.rfft(rev_a, n=fft_len_grad)
    full = np.fft.irfft(wf * af, n=fft_len_grad)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad

# --------------------------------------------------------------
# Main optimisation entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # safety margin

    # -----------------------------------------------------------------
    # 1) Choose a length.
    # -----------------------------------------------------------------
    # Prefer the existing construction (if present), otherwise fall back
    # to a power‑of‑two size that is comfortable for FFT.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = _project_simplex(np.clip(cand, 0.0, None))
            else:
                raise ValueError
        except Exception:
            n = 4096
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # -----------------------------------------------------------------
    # 2) Warm‑start with a geometric family (a_i ∝ r^i).
    # -----------------------------------------------------------------
    fft_len_conv = _fft_len_for_n(2 * n - 1)
    fft_len_grad = _fft_len_for_n(3 * n - 2)

    best = a.copy()
    best_score = _score(best, fft_len_conv)

    # try a few geometric series and keep the best as starting point
    rs = np.linspace(0.90, 0.9995, 30)          # r in [0.9, 0.9995]
    for r in rs:
        geo = np.exp(np.arange(n) * np.log(r))
        geo = _project_simplex(geo)
        sc = _score(geo, fft_len_conv)
        if sc < best_score:
            best_score = sc
            best = geo.copy()
    a = best.copy()
    cur_score = best_score

    # -----------------------------------------------------------------
    # 3) Simulated annealing (global exploration)
    # -----------------------------------------------------------------
    start_time = time.monotonic()
    last_improve = start_time

    # annealing schedule parameters
    T_start = 0.12          # initial temperature
    T_end = 1e-6            # final temperature (almost greedy)
    move_block_prob = 0.65 # probability of choosing a block‑scale move
    restart_gap = 45.0      # seconds without improvement before a soft restart

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # linear temperature decay (slow near the beginning)
        T = T_start * (1.0 - elapsed / total) + T_end * (elapsed / total)

        # propose a move
        if rng.random() < move_block_prob:
            cand = _random_block_perturb(a, rng)
        else:
            cand = _random_swap_move(a, rng)

        cand_score = _score(cand, fft_len_conv)

        # Metropolis acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability of accepting a worsening move
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score

        # keep the overall best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] new best (anneal) = {best_score:.6f}")

        # Restart if we have been stuck for a while – perturb the best a bit
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.09, size=n))
            a = best * jitter
            a = _project_simplex(a)
            cur_score = _score(a, fft_len_conv)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4) Polishing – a few projected‑gradient steps with a soft‑max
    #    approximation of the max‑norm (identical to the reference method).
    # -----------------------------------------------------------------
    # Parameters tuned for a short yet effective refinement.
    eta_start = 0.6
    eta_end = 0.03
    beta_start = 0.8
    beta_end = 1500.0
    grad_mom = np.zeros_like(best)

    polish_deadline = time.monotonic() + 10.0   # at most ten extra seconds
    while time.monotonic() < polish_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total)

        # soft‑max weights (log‑sum‑exp)
        conv = _conv_self(best, fft_len_conv)
        shift = conv.max()
        w_raw = np.exp(beta * (conv - shift))
        w = w_raw / w_raw.sum()

        # gradient of the smoothed objective
        grad = _gradient_softmax(best, fft_len_conv, fft_len_grad, beta)

        # momentum (helps stability)
        grad_mom = 0.85 * grad_mom + 0.15 * grad

        # projected additive step
        best_tmp = best - eta * grad_mom
        best = _project_simplex(best_tmp)

    # -----------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by evaluate_sequence)
    # -----------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
