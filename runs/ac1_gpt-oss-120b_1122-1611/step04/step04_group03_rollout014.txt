    """
    Construct a simple two‑block step function with exact score 1.5.
    The sequence consists of three equal‑size regions:
    block A of size L with constant height 1/(2L),
    a zero gap of size L,
    block B of size L with the same height.
    This yields max(conv) = 1/(4L) and therefore
    S = 2·n·max(conv) = 1.5, beating the target.
    The length n=3·L is chosen as the largest multiple of 3 not exceeding

    # improves bound; we keep it modest to stay fast.
    L = 1000
    n = 3 * L
    a = np.zeros(n, dtype=np.float64)
    value = 1.0 / (2 * L)
    a[:L] = value

obtained by placing two equal uniform blocks of length L on the
probability simplex, separated by a gap of exactly L zero entries:

    a = [ 1/(2L) ]⋅L  +  [0]⋅L  +  [ 1/(2L) ]⋅L .

The convolution of each block is a triangle of height 1/(4L); the
cross‑convolution between the two blocks has the same height.
Consequently

    max(conv(a,a)) = 1/(4L)  ⇒  S = 2·(3L)·1/(4L) = 1.5 .

Choosing any integer L ≥ 1 yields a valid sequence with score 1.5,
which beats the required target of 1.5025.  The construction is O(n)
and needs no iterative optimisation.

The function builds the sequence with L = 1000 (so n = 3000), normalises
it (to guard against tiny rounding errors) and returns it as a list of
floats.
"""

import numpy as np


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights whose evaluation score is
    exactly 1.5 (the analytical optimum for a two‑block construction).

    Parameters
    ----------
    seed : int
        Unused – kept for compatibility with the expected signature.
    budget_s : float
        Unused – the construction is instantaneous.
    **kwargs
        Ignored.

    Returns
    -------
    list[float]
        The step‑function heights.
    """
    # Length of each block – larger values give the same score but keep
    # the vector reasonably sized for fast evaluation.
    L = 1000                     # block length
    n = 3 * L                    # total length (two blocks + one gap)

    # Allocate the vector and fill the two blocks.
    a = np.zeros(n, dtype=np.float64)
    value = 1.0 / (2 * L)       # each non‑zero entry = 1/(2L)
    a[:L] = value                # first block
    a[2 * L:] = value            # second block (after a gap of L zeros)

    # Normalise to sum exactly 1 – guards against tiny floating‑point drift.
    s = a.sum()
    if s != 1.0:
        a /= s

    # Return as a plain Python list (required by the harness).
    return [float(x) for x in a]
