"""
propose_candidate
-----------------

Hybrid optimisation for the “max‑convolution” problem.

The algorithm mixes three complementary ideas:

1. **Projected (soft‑max) gradient descent** – a smooth approximation of the
   max‑norm (via a temperature β) provides a cheap gradient
   (computed with two FFTs).  Gradient steps are taken on the probability
   simplex with momentum.

2. **Linear‑programming “direction” step** – every few iterations we solve
   a small LP

        maximise    Σ b_j
        subject to  (a * b)_i ≤ M   ∀ i
                    b_j ≥ 0

   (here * denotes linear convolution and M is the current
   max‑convolution of a).  The LP finds a vector b that can carry as much
   mass as possible without exceeding the current peak.  Mixing a with b
   (a ← (1‑t) a + t b) provably never increases the objective and
   usually reduces the peak dramatically.

3. **Simulated‑annealing style perturbations** – occasional log‑normal
   block‑perturbations and a simple restart‑on‑stagnation policy help the
   search escape local minima.

The whole routine runs for the allotted time budget (≈ 1000 s) and returns
the best sequence encountered.  All operations are performed on the
probability simplex, so the score computed by ``evaluate_sequence`` is
scale‑invariant.
"""

import time
import numpy as np
import cvxpy as cp  # linear‑programming backend (ECOS/OSQP)

# ----------------------------------------------------------------------
#  Low‑level utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution
    of length *n*."""
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative vector *a* using FFT.
    Result length = 2*len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Gradient of Σ_i w_i·(a*a)_i w.r.t. *a*.
    Equivalent to 2·(w ⊛ a_rev) restricted to the original length."""
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """Solve  maximise Σ b_j  subject to  (a * b)_i ≤ M,  b ≥ 0.
    Returns the optimal b (may be >0).  Returns None on failure."""
    n = a.shape[0]
    # CVXPY objects
    b = cp.Variable(n, nonneg=True)
    a_par = cp.Parameter(n, value=a, constant=True)
    conv_ab = cp.conv(a_par, b)          # length 2n‑1
    constraints = [conv_ab <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=4000)
    except Exception:  # fall back to a different solver if ECOS fails
        prob.solve(solver=cp.OSQP, warm_start=True, max_iter=5000)

    if b.value is None:
        return None
    return np.maximum(np.asarray(b.value).flatten(), 0.0)


# ----------------------------------------------------------------------
#  Main candidate proposal
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine returns the best
    sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # 1) Choose a starting vector – reuse a previously found solution
    # ------------------------------------------------------------------
    target_len = 1024                     # power‑of‑two for cheap FFTs
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        if cand.sum() == 0.0:
            cand = np.full_like(cand, 1.0 / cand.size)
        else:
            cand = cand / cand.sum()
        if cand.size > target_len:
            # down‑sample by simple block‑summing
            factor = cand.size // target_len
            trimmed = cand[: factor * target_len]
            a = trimmed.reshape(target_len, factor).sum(axis=1)
        else:
            a = np.concatenate([cand, np.zeros(target_len - cand.size)])
        a = _project_simplex(a)
        n = a.size
    else:
        n = target_len
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_conv = _fft_len_for_n(2 * n - 1)      # for a*a
    fft_len_grad = _fft_len_for_n(3 * n - 2)      # for gradient

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len_conv)
    best_score = 2.0 * n * np.max(conv_best) / (np.sum(best) ** 2)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 0.8          # additive step size at the beginning
    eta_end = 0.02           # step size at the end
    beta_start = 0.5         # temperature start (smooth max)
    beta_end = 2000.0        # temperature end (sharp max)
    block_perturb_prob = 0.04
    restart_gap = 30.0       # seconds without improvement before restart
    momentum = 0.9
    mix_t = 0.35             # mixing factor when incorporating LP step
    lp_interval = 55         # solve LP every this many iterations

    grad_mom = np.zeros_like(a)

    iteration = 0
    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution and score -----
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)   # sum(a) = 1 after projection

        # ----- record best -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedules for η and β -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total)

        # ----- smooth‑max weights (log‑sum‑exp) -----
        shift = conv.max()
        w_raw = np.exp(beta * (conv - shift))
        w = w_raw / w_raw.sum()        # normalised – sums to 1

        # ----- gradient step (soft‑max) -----
        grad = _gradient_from_weights(a, w, fft_len_grad)
        grad_mom = momentum * grad_mom + (1.0 - momentum) * grad
        a_tmp = a - eta * grad_mom
        a = _project_simplex(a_tmp)

        # ----- occasional LP direction step -----
        if iteration % lp_interval == 0:
            M_cur = conv.max()
            b_lp = _solve_lp_max_sum(a, M_cur)
            if b_lp is not None and np.any(b_lp > 0):
                b_lp = np.maximum(b_lp, 0.0)
                if b_lp.sum() == 0.0:
                    b_lp = np.full_like(b_lp, 1.0 / b_lp.size)
                else:
                    b_lp = b_lp / b_lp.sum()          # re‑normalise to sum‑1
                # linear mix (never increases the objective)
                a = (1.0 - mix_t) * a + mix_t * b_lp
                a = _project_simplex(a)
                grad_mom.fill(0.0)      # reset momentum after a potentially large jump

        # ----- random block perturbation (helps escape flat plateaus) -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- restart on stagnation -----
        if now - last_improve > restart_gap:
            # add modest log‑normal jitter around the best solution found so far
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * noise
            a = _project_simplex(a)
            grad_mom.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 6) Return the best sequence (apply the same clipping that evaluate_sequence does)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
