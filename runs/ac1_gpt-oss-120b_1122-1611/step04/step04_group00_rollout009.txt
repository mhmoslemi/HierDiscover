"""
propose_candidate
-----------------

Optimises a non‑negative step function under the evaluation
``score = 2·n·max(conv(a,a)) / sum(a)^2``.

The algorithm works on the probability simplex (∑a=1) and employs
a smooth (soft‑max) surrogate for the max‑norm together with an
Adam‑style projected optimiser.  Temperature β for the soft‑max
is increased exponentially so that the objective gradually turns
into the true max.  After each gradient step we project back onto
the simplex.  Random block perturbations and a simple
restart‑on‑stagnation policy (jittering the best solution) help
escape local minima.  The best vector seen before the time budget
expires is returned.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold a linear convolution
    of length *n*.
    """
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative vector *a* using FFT.
    Result length is 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a*a)_i with respect to *a*.
    Implements 2·(w ⊛ reverse(a)) restricted to the original length.
    """
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``
    before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    # Keep a small safety margin for the outer driver.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector.
    # ------------------------------------------------------------------
    # Prefer a previously found solution if it exists.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a0 = np.clip(cand, 0.0, None)
                if a0.sum() == 0.0:
                    a0 = np.full_like(a0, 1.0 / a0.size)
                else:
                    a0 = a0 / a0.sum()
            else:
                a0 = None
        except Exception:
            a0 = None
    else:
        a0 = None

    # Target length – a power of two not smaller than 8192.
    target_len = 8192
    if a0 is not None:
        n0 = a0.size
        if n0 < target_len:
            a = np.concatenate([a0, np.zeros(target_len - n0)])
            a = _project_simplex(a)
        elif n0 > target_len:
            a = _project_simplex(a0[:target_len])
        else:
            a = a0.copy()
    else:
        a = np.full(target_len, 1.0 / target_len)

    n = a.size

    # FFT lengths – reused throughout the optimisation.
    fft_len_conv = _fft_len_for_n(2 * n - 1)          # for self‑convolution
    fft_len_grad = _fft_len_for_n(3 * n - 2)          # for gradient convolution

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len_conv)
    best_score = 2.0 * n * np.max(conv_best) / (np.sum(best) ** 2)

    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)   # first‑moment estimate
    v = np.zeros_like(a)   # second‑moment estimate
    t_iter = 0

    # Step‑size schedule (additive after Adam scaling)
    alpha_start, alpha_end = 0.6, 0.02

    # Soft‑max temperature schedule
    beta_start, beta_end = 0.5, 1e5

    block_perturb_prob = 0.07               # more frequent than before
    restart_gap = 20.0                      # seconds without improvement
    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution and objective value -----
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        s = a.sum()                     # should stay =1, kept for safety
        cur_score = 2.0 * n * cur_max / (s * s)

        # ----- keep track of the best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedules for step‑size α and temperature β -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear decay for α
        alpha = alpha_start * (1.0 - elapsed / total) + alpha_end * (elapsed / total)
        # exponential increase for β
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total)

        # ----- soft‑max weights (log‑sum‑exp) -----
        shift = conv.max()
        w_raw = np.exp(beta * (conv - shift))
        w = w_raw / w_raw.sum()          # normalized, sum=1

        # ----- gradient of the smoothed objective -----
        grad = _gradient_from_weights(a, w, fft_len_grad)

        # ----- Adam update -----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        a_tmp = a - alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a_tmp)

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- restart if stagnation detected -----
        if now - last_improve > restart_gap:
            # Combine the current best with fresh randomness.
            noise = np.exp(rng.normal(0.0, 0.18, size=n))
            a = best * 0.6 + (rng.random(n) * 0.4)
            a *= noise
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (final safety clipping)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
