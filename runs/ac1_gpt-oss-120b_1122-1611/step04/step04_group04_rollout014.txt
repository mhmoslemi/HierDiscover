def _project_simplex(v):
    """Projects onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    """
    Solve linear program: max sum(g) s.t. a*g <= M (componentwise), g >= 0.
    Returns vector g (numpy array) of length n.

"""
Hybrid optimisation for the L∞ self‑convolution problem.

The routine works in three stages inside a fixed time budget (default 1000 s):
1. Exponentiated‑gradient descent with a soft‑max approximation of the
   ``max`` and occasional random block jitter.  Momentum and a
   temperature schedule accelerate convergence.
2. One (or a few) linear‑programming refinements.  For the current
   sequence *a* we solve
       max   Σ g_j                 (i.e. maximise the L1‑norm of g)
       s.t.  (a * g)_k ≤ M  for all k,
             g_j ≥ 0,
   where M = max(a*a).  The optimal g is mixed with a (convex combination)
   to obtain a strictly better point.
3. Fine‑grained local search: pairwise mass transfers, random block swaps
   and additional jitter are tried and accepted only if they lower the
   objective.  The best candidate seen throughout the run is returned.

All operations keep the vector non‑negative and normalised to unit
L1‑norm (the evaluation function is homogeneous, so this does not affect
the score).  The final sequence is clipped to the limits imposed by
``evaluate_sequence`` before being returned.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (returns length 2*len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)          # clip tiny negatives
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Formula: ∂/∂a_j Σ_k w_k·(a*a)_k = 2· Σ_i a_i·w_{i+j}
    computed via a single FFT‑based correlation.
    """
    n = a.size
    Lw = w.size                 # = 2*n-1
    wr = w[::-1]                # reverse for correlation
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)          # positions of the gradient entries
    grad = 2.0 * conv_full[idx]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)

    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_transfer(a: np.ndarray, conv: np.ndarray, rng: np.random.Generator):
    """
    Move a small amount of mass from one coordinate to another.
    Returns (new_a, new_conv, new_score) if the move improves the
    objective, otherwise the original tuple (a, conv, None).
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    new_conv = conv.copy()
    lo_i = i
    hi_i = i + n
    lo_j = j
    hi_j = j + n

    new_conv[lo_i:hi_i] -= delta * a
    new_conv[lo_j:hi_j] += delta * a

    new_score = 2.0 * n * np.max(new_conv)
    cur_score = 2.0 * n * np.max(conv)

    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.clip(a2, 0.0, None)
        a2 /= a2.sum()
        return a2, new_conv, new_score
    else:
        return a, conv, None


def _block_swap(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two non‑overlapping blocks of equal length."""
    n = a.size
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start1 = rng.integers(0, n - width + 1)

    # find a second block that does not overlap the first one
    for _ in range(5):
        start2 = rng.integers(0, n - width + 1)
        if start2 + width <= start1 or start2 >= start1 + width:
            break
    else:
        return a                                    # give up – nothing to swap

    out = a.copy()
    tmp = out[start1:start1 + width].copy()
    out[start1:start1 + width] = out[start2:start2 + width]
    out[start2:start2 + width] = tmp
    return out


def _solve_lp_for_g(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        max   Σ g_j
        s.t.  (a * g)_k ≤ M   for all k = 0 … 2n‑2
              g_j ≥ 0
    Returns the optimal vector g (may be dense).
    """
    n = a.size
    rows = 2 * n - 1

    # Build the Toeplitz‐type constraint matrix A where
    #   (A g)_k = Σ_i a_i * g_{k-i}
    # Each diagonal offset i contains the constant a_i.
    diags_data = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diags_data, offsets, shape=(rows, n), format='csc')

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximise Σ g_j  <=> minimise -Σ g_j
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise problem size and starting point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full(n, 1.0 / n)
        else:
            n = 4096
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # FFT length for self‑convolution (power of two for speed)
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _self_conv_fft(best, fft_len)
    best_score = 2.0 * n * conv_best.max()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # tiny safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters for the three phases
    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent
    momentum = 0.85
    block_perturb_prob = 0.025
    restart_gap = 30.0

    # Phase 2 – LP refinement
    lp_mix_steps = (0.3, 0.5, 0.7)   # convex‑mixing coefficients

    # Phase 3 – local search
    prob_pairwise = 0.07
    prob_swap = 0.33
    prob_jitter = 0.20

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop (split into three timed phases)
    # ------------------------------------------------------------------
    total_budget = deadline - start_time
    phase1_end = start_time + 0.45 * total_budget
    phase2_end = start_time + 0.80 * total_budget

    # -------------------------- Phase 1 -------------------------------
    while time.monotonic() < phase1_end:
        conv = _self_conv_fft(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep track of the best solution
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best {best_score:.6f}")

        # time‑dependent schedules
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_budget

        eta = 0.9 * (1.0 - frac) + 0.02 * frac           # learning‑rate decay
        beta = 0.5 + (3000.0 - 0.5) * frac                # temperature (soft‑max sharpness)

        # soft‑max weight vector approximating the max
        w = np.exp(beta * (conv - cur_max))               # max exponent = 0

        # gradient of the smooth max
        grad = _gradient_from_weights(a, w)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient (mirror descent) step
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        a = a / a_sum if a_sum > 0.0 else np.full_like(a, 1.0 / n)

        # occasional random block jitter
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # cheap pairwise transfer move
        if rng.random() < prob_pairwise:
            a, _, _ = _pairwise_transfer(a, conv, rng)

        # restart if we have been stuck for a while
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.14, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # -------------------------- Phase 2 -------------------------------
    while time.monotonic() < phase2_end:
        conv = _self_conv_fft(a, fft_len)
        M = conv.max()

            g = _solve_lp_for_g(a, M)

            print(f"[{time.monotonic() - start_time:6.2f}s] LP failed: {e}")
            break

        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        g_norm = g / sum_g                      # bring back to unit L1‑norm

        improved = False
        for t in lp_mix_steps:
            a_candidate = (1.0 - t) * a + t * g_norm
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate /= a_candidate.sum()
            conv_c = _self_conv_fft(a_candidate, fft_len)
            score_c = 2.0 * n * conv_c.max()

            if score_c < best_score:
                best_score = score_c
                best = a_candidate.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start_time:6.2f}s] LP‑mix t={t:.2f} new best {best_score:.6f}")

            if score_c < cur_score:
                a = a_candidate
                improved = True
                break

            # No improvement from mixing – give up further LP attempts
            break

    # -------------------------- Phase 3 -------------------------------
    while time.monotonic() < deadline:
        conv = _self_conv_fft(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best (fine) {best_score:.6f}")

        # decide which local move to try
        r = rng.random()
        if r < prob_jitter:
            a = _random_block_perturb(a, rng)
        elif r < prob_pairwise + prob_jitter:
            a, _, _ = _pairwise_transfer(a, conv, rng)
        elif r < prob_pairwise + prob_jitter + prob_swap:
            a2 = _block_swap(a, rng)
            conv2 = _self_conv_fft(a2, fft_len)
            if 2.0 * n * conv2.max() < cur_score:
                a = a2
                conv = conv2
                cur_score = 2.0 * n * conv.max()

        # ensure feasibility (non‑negative + unit sum)
        a = np.clip(a, 0.0, None)
        a /= a.sum()

        # tiny gradient “polish” step every now and then
        if rng.random() < 0.05:
            beta_small = 15.0
            w = np.exp(beta_small * (conv - cur_max))
            grad = _gradient_from_weights(a, w)
            lr = 0.01
            factor = np.exp(-lr * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # early exit if target already beaten
        if best_score <= 1.5025:
            break

    # ------------------------------------------------------------------
    # 5) Return best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
