Propose a step‑function (non‑negative heights) that minimizes
evaluate_sequence().  The algorithm combines a provably good analytic
initialisation (the “√‑difference” shape) with a fast projected
mirror‑descent that works on the true L∞‑objective via an active‑set
sub‑gradient.  Occasional random perturbations and soft restarts keep the
search from stagnating.  The whole optimisation runs for the allotted
budget (≈1000 s) and returns the best vector found.

Key ideas
---------
*   The objective
        score = 2·n·max_k (a∗a)_k / (∑a_i)^2
    is homogeneous, so we keep the vector on the probability simplex
    (∑a_i = 1).  Then  score = 2·n·max_k (a∗a)_k .
*   The convolution (a∗a) is a linear combination of the outer‑product
    matrices M_k with entries (M_k)_ij = 1_{i+j=k}.  At an index k
    the gradient of (a∗a)_k is 2·a shifted by k, i.e.
        ∂/∂a_i (a∗a)_k = 2·a_{k‑i}   (zero if the index is out of range).
*   We minimise the L∞‑norm by a mirror‑descent (exponentiated‑gradient)

    i.e. those whose convolution value is close to the current maximum.
*   The starting point a_i = √(i+1)‑√i (normalised) gives a very flat

    the previously obtained 1.514.

    restarts inject diversity and help escape shallow local minima.

# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT.
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a, returned length = 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Compute the gradient of the soft‑max surrogate (used only as a fallback).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Grad of  Σ_k w_k (a∗a)_k  = 2·a ⋆ w  (correlation)."""
    n = a.size
    L = w.size                # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

    beta_end = 2500.0        # very sharp soft‑max (almost true max)
    block_perturb_prob = 0.035
    pair_move_prob = 0.08
    restart_gap = 120.0      # seconds without improvement before a restart
    momentum = 0.85

    grad_prev = np.zeros_like(a)

    # --------------------------------------------------------------
    # 5) Main optimisation loop ---------------------------------------
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- self‑convolution & current score -----------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----- keep the best solution ---------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- scheduling ---------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ----- active‑set (indices close to the maximum) -------------
        # Use a relative tolerance; keep at most 12 indices.
        tol = cur_max * 0.995   # within 0.5 % of the max
        active = np.where(conv >= tol)[0]
        if active.size > 12:
            # keep the 12 largest values (they are already close to max)
            active = np.argsort(conv)[-12:]

        # ----- gradient of the (soft) max -----------------------------
        if active.size > 0:
            # Soft weights over the active set (more weight for larger values)
            w_raw = np.exp(beta * (conv[active] - cur_max))
            w_sum = w_raw.sum()
            if w_sum > 0.0:
                w = w_raw / w_sum
            else:
                w = np.full_like(w_raw, 1.0 / w_raw.size)

            grad = np.zeros_like(a)
            for idx, weight in zip(active, w):
                # The gradient of (a∗a)_idx is 2·a shifted by idx.
                lo = max(0, idx - (n - 1))
                hi = min(n - 1, idx)
                i_range = np.arange(lo, hi + 1, dtype=np.int64)
                grad[i_range] += weight * 2.0 * a[idx - i_range]
        else:
            # Very unlikely, but fall back to the full‑soft‑max surrogate.
            w = np.exp(beta * (conv - cur_max))
            w = w / w.sum()
            grad = _gradient_correlation(a, w)

        # ----- momentum -----------------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- mirror‑descent (exponentiated‑gradient) update ----------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- occasional random block perturbation --------------------
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            factor_blk = np.exp(rng.normal(0.0, 0.2))
            a[start:start + width] *= factor_blk
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # ----- occasional pairwise mass transfer -----------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-14:
                # move at most half of the mass at i to j
                delta = rng.random() * a[i] * 0.5
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

        # ----- soft restart if stuck ----------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # jitter every component multiplicatively (log‑normal noise)
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (respect the clipping performed by
    #    evaluate_sequence).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
