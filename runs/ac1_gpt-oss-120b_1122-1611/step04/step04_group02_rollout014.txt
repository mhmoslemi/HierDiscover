    # maybe subtract max for numerical stability

def solve_lp_step(a, M):
    n = a.size
    # Build A_ub matrix
    rows = 2*n - 1
    A_ub = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):

    # objective
    c = -np.ones(n, dtype=np.float64)  # minimize -sum b

    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.full(rows, M), bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    return b

"""
Propose a step‑function (non‑negative vector on the simplex) that beats the
current best bound (≈1.514).  The algorithm consists of two stages.

Stage 1 – full‑softmax mirror descent.
    * The objective 2·n·max(convolution) is smoothed by a soft‑max
      (temperature β).  The gradient of the smooth objective is obtained
      via a single FFT‑based convolution, i.e. it uses *all* convolution
      entries (no hand‑picked active set).
    * An exponentiated‑gradient (mirror‑descent) update keeps the iterate
      on the probability simplex.  Step‑size η and temperature β follow
      a linear schedule, but they are *reset* after every restart so that
      the search can re‑heat.

Stage 2 – stochastic annealing refinement.
    * Pairwise mass transfers, block‑wise log‑normal jitter and a short
      Gaussian smoothing move are tried.  Acceptance follows a simple
      Metropolis rule with a linearly decreasing temperature.
    * If no improvement has been seen for a while a soft restart is
      performed: the current best solution is jittered (multiplicative
      log‑normal noise) and the schedules are re‑initialised.

The whole routine respects the 1000 s time budget, prints occasional
progress reports and finally returns the best vector found, clipped to
the limits enforced by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two length‑n vectors."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector ``a`` using FFT.
    Returned array has length ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of   G(a)=∑_k exp(β·(conv_k−max(conv)))·conv_k   w.r.t. a.
    Uses a single FFT convolution and therefore incorporates *all*
    convolution entries (no active‑set truncation).
    """
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))            # weight vector, length 2n‑1
    a_rev = a[::-1]                              # reversed copy of a
    size = a_rev.size + w.size - 1                # length of the linear convolution a_rev * w
    fft_len = 1 << ((size - 1).bit_length())
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, n=fft_len)[:size]
    start = a_rev.size - 1                        # = n‑1
    grad = 2.0 * conv_rev[start : start + a.size]  # 2·∑_j a_j w_{i+j}
    return grad


def _smooth_move(a: np.ndarray) -> np.ndarray:
    """Small Gaussian smoothing move (3‑point kernel)."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    sm = np.convolve(a, kernel, mode='same')
    sm = np.clip(sm, 0.0, None)
    s = sm.sum()
    if s == 0.0:
        sm[:] = 1.0 / a.size
    else:
        sm /= s
    return sm


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Global time‑budget handling
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # safety margin

    # --------------------------------------------------------------
    # 1) Choose a starting vector (use previous construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            a = rng.dirichlet(np.ones(256)).astype(np.float64)
    else:
        a = rng.dirichlet(np.ones(256)).astype(np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 2) Book‑keeping
    # --------------------------------------------------------------
    conv = _conv_self(a, fft_len)
    cur_score = 2.0 * n * conv.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time
    # 70 % of the budget for the soft‑max mirror‑descent phase
    phase1_end = start_time + 0.70 * total_time

    # --------------------------------------------------------------
    # 3) Hyper‑parameters for phase 1 (soft‑max mirror descent)
    # --------------------------------------------------------------
    eta_start, eta_end = 1.2, 0.01        # step‑size schedule
    beta_start, beta_end = 0.5, 800.0    # temperature schedule
    momentum = 0.90
    grad_prev = np.zeros_like(a)
    eps = 1e-12                         # tiny floor to keep positivity

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – soft‑max mirror descent (dominant part of the budget)
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedules (reset after every restart)
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # gradient of the smooth max‑approximation (full active set)
            grad = _softmax_gradient(a, conv, beta)

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient (mirror descent) update
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, eps, None)
            s = a.sum()
            a /= s

            # occasional jitter to preserve diversity
            if rng.random() < 0.015:
                a = _random_block_perturb(a, rng)

            # re‑evaluate convolution and score
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Phase 2 – stochastic local refinement (final 30 %)
        # --------------------------------------------------------------
        else:
            # linear temperature schedule for simulated annealing
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = max(1e-8, 0.06 * (1.0 - t_frac))

            move_type = rng.random()

            if move_type < 0.70:  # pairwise mass transfer
                i = rng.integers(0, n)
                if a[i] < 1e-14:
                    continue
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_delta = 0.35 * a[i]
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
            elif move_type < 0.85:  # block‑perturbation (multiplicative jitter)
                a_new = _random_block_perturb(a, rng)
            else:                  # smoothing move
                a_new = _smooth_move(a)

            conv_new = _conv_self(a_new, fft_len)
            new_score = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temperature):
                a = a_new
                conv = conv_new
                cur_score = new_score

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart when stuck for a while
        # --------------------------------------------------------------
        restart_gap = 30.0
        if now - last_improve > restart_gap:
            # jittered restart from the current best
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, eps, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            # reset the schedules so that we can “re‑heat”
            start_time = now
            phase1_end = now + (phase1_end - start_time)  # keep same remaining fraction
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (safety‑clipped)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
