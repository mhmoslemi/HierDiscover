The algorithm consists of three stages

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] best improved = {best_score:.6f}")

        # ----- schedule -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # ----- active set -----
        active = np.where(conv >= max_conv - eps_active)[0]   # indices achieving the max
        # gradient of max‖a∗a‖ (sub‑gradient):
        #   grad_j = 2 * sum_{k in active} a_{k-j} - a_j * I[2*j in active]
        # (out‑of‑range accesses are ignored)
        grad = np.zeros_like(a)

        # contribution from the first term (2 * sum a_{k-j})
        for k in active:
            # shift = k - np.arange(n)
            shift = k - np.arange(n)
            pos = (shift >= 0) & (shift < n)
            grad[pos] += 2.0 * a[shift[pos]]

        # second term: - a_j if 2*j ∈ active
        idx2 = 2 * np.arange(n)
        mask2 = np.isin(idx2, active, assume_unique=True)
        grad[mask2] -= a[mask2]

        # ----- exponentiated‑gradient step -----
        # safe clipping of the exponent to avoid overflow/underflow
        exponent = -eta * grad
        exponent = np.clip(exponent, -700, 700)   # np.exp range safeguard
        factor = np.exp(exponent)
        a = a * factor
        a = _project_simplex(a)   # keep a on the probability simplex

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- occasional pair‑wise mass transfer -----
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                max_delta = min(a[i], 0.5 * a[i])
                delta = rng.random() * max_delta
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                a_new = _project_simplex(a_new)
                # cheap evaluation
                conv_new = _conv_self(a_new, fft_len)
                new_score = 2.0 * n * conv_new.max()
                if new_score < cur_score:
                    a = a_new
                    cur_score = new_score

        # ----- soft restart when stuck -----
        if now - last_improve > restart_gap:
            a = best_a.copy()
            # add modest log‑normal jitter before restarting
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = _project_simplex(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
