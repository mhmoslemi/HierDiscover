"""
Propose candidate – ADAM‐based soft‑max gradient descent with
restart and final hill‑climbing.

The sequence is parametrised by log‑probabilities x (a_i = softmax(x)).
A smooth approximation of the maximum self‑convolution
    f(x) = (1/β)·log∑_k exp(β·(a*a)_k)
is minimised by ADAM updates on x.
β is increased linearly from a small to a large value, so the
approximation tightens to the true L∞‑norm.
If the algorithm stalls for a while it restarts from the current best
solution with a small random perturbation.
When the time budget is almost exhausted a cheap pairwise‑mass‑transfer
local search (the same move used in the reference code) is performed
to squeeze out the last improvements.

All operations are FFT‑based (O(n log n)) and respect the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Clip tiny negatives caused by round‑off.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements 2·corr(a,w) via one FFT‑based convolution.
    """
    n = a.size
    Lw = w.size                     # = 2*n-1
    wr = w[::-1]                    # reversed for correlation

    conv_len = _next_pow2(n + Lw - 1)   # length for a + wr - 1
    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]

    idx = Lw - 1 - np.arange(n)          # indices for gradient formula
    grad = 2.0 * conv_full[idx]
    return grad


def _pairwise_transfer(a: np.ndarray, conv: np.ndarray, rng: np.random.Generator):
    """
    Cheap hill‑climbing move: move a small fraction of mass from one index
    to another.  The convolution is updated incrementally.
    Returns (new_a, new_conv, new_score) if the move improves the score,
    otherwise the original triple (a, conv, None).
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    # Move at most 30 % of the donor's current mass.
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # Incremental update of the full convolution.
    new_conv = conv.copy()
    lo_i = i
    hi_i = i + n
    lo_j = j
    hi_j = j + n

    new_conv[lo_i:hi_i] -= delta * a
    new_conv[lo_j:hi_j] += delta * a

    new_score = 2.0 * n * np.max(new_conv)
    cur_score = 2.0 * n * np.max(conv)

    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.clip(a2, 0.0, None)
        s = a2.sum()
        if s == 0.0:
            a2[:] = 1.0 / n
        else:
            a2 /= s
        return a2, new_conv, new_score
    else:
        return a, conv, None


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` (i.e. 2·n·max(a*a) / (sum a)^2).

    The method:
        * initialise from the globally stored best sequence if present,
          otherwise a uniform distribution;
        * optimise the smooth log‑sum‑exp of the self‑convolution using
          ADAM on the log‑softmax parametrisation;
        * periodically restart from the current best when progress stalls;
        * finish with a cheap pairwise‑mass‑transfer hill‑climber.
    Returns the best coefficient list (plain Python floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector a (sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a0 = np.clip(cand0, 0.0, None)
            s = a0.sum()
            if s > 0.0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = 4096
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # make sure we have no exact zeros for the log‑parametrisation
    eps = 1e-12
    a0 = np.clip(a0, eps, None)
    a0 /= a0.sum()

    # log‑probability vector
    x = np.log(a0)

    # FFT lengths (powers of two)
    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)      # for correlation

    # ------------------------------------------------------------------
    # 2) Book‑keeping and ADAM hyper‑parameters
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_conv = _self_conv_fft(best_a, fft_len_self)
    best_score = 2.0 * n * best_conv.max()

    m = np.zeros_like(x)          # ADAM first moment
    v = np.zeros_like(x)          # ADAM second moment
    t = 0                         # step counter

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    last_improve = start_time

    # ADAM defaults
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # Schedule parameters
    beta_start = 0.5
    beta_end = 2000.0
    lr_start = 0.15
    lr_end = 0.001
    restart_gap = 30.0           # seconds without improvement → restart
    noise_prob = 0.02            # probability of a random log‑noise perturbation
    noise_sigma = 0.2            # std‑dev of the log‑noise

    # ------------------------------------------------------------------
    # 3) Main optimisation loop (soft‑max + ADAM)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- obtain a from the current logits -----
        x_shift = x - np.max(x)          # numerical stability
        a = np.exp(x_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a_sum

        # ----- current self‑convolution and score -----
        conv = _self_conv_fft(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        # ----- keep best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_conv = conv.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- time‑aware schedules -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start * (1.0 - frac) + lr_end * frac
        t += 1

        # ----- smooth max weights (log‑sum‑exp) -----
        max_conv = conv.max()
        # stable exponentials: max conv subtracted already
        w_raw = np.exp(beta * (conv - max_conv))
        w = w_raw / w_raw.sum()          # normalized weight vector

        # ----- gradient of the smoothed objective w.r.t. a -----
        grad_a = _gradient_from_weights(a, w)   # shape (n,)

        # ----- convert to gradient w.r.t. logits (softmax‑reparam.) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)                # dL/dx

        # ----- ADAM update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ----- occasional random perturbation to escape shallow plateaus -----
        if rng.random() < noise_prob:
            x += rng.normal(0.0, noise_sigma, size=n)

        # ----- restart if stuck long enough -----
        if now - last_improve > restart_gap:
            # re‑initialise logits around the current best
            x = np.log(best_a + eps) + rng.normal(0.0, noise_sigma, size=n)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

        # ------------------------------------------------------------------
        # optional cheap pairwise transfer (occasionally) – helps local tweaks
        # ------------------------------------------------------------------
        if rng.random() < 0.07:
            # use the already‑computed convolution for the current a
            a, conv, _ = _pairwise_transfer(a, conv, rng)
            # re‑normalise a (the transfer routine already does it)
            # we keep the updated a for the next iteration; logits are
            # re‑derived from a at the top of the loop, so we simply
            # replace x with the new logits.
            x = np.log(np.clip(a, eps, None))

    # ------------------------------------------------------------------
    # 4) Final local hill‑climbing using cheap pairwise transfers
    # ------------------------------------------------------------------
    # Use the best solution found so far as the starting point.
    a = best_a.copy()
    conv = best_conv.copy()
    cur_score = best_score
    end_time = start_time + float(budget_s) - 0.5   # leave a tiny safety margin

    while time.monotonic() < end_time:
        a, conv, new_score = _pairwise_transfer(a, conv, rng)
        if new_score is not None and new_score < cur_score:
            cur_score = new_score
            best_a = a.copy()
            best_score = cur_score
            best_conv = conv.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] hill‑climb improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped for safety)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
