def shape_power(n, p):
    i = np.arange(n)
    val = ((i + 0.5) * (n - i - 0.5))**(-p)
    val = np.clip(val,0,None)
    return val / val.sum()

def shape_geo(n, r):
    i = np.arange(n)
    val = np.power(r, i) + np.power(r, n-1-i)
    val = np.clip(val,0,None)
    return val / val.sum()

    i = np.arange(n)

"""
propose_candidate
-----------------

Search for a non‑negative step function that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The method combines a cheap parametric “shape” search on a small problem,
followed by a staged mirror‑descent optimisation (exponentiated‑gradient
with a soft‑max approximation of the max‑norm).  The optimisation first
runs on a modest size (≈ 512) to quickly discover a good shape, then the
solution is interpolated to the target length (≈ 5000) and refined for
the remaining time.  Random block perturbations, pairwise mass transfers
and occasional soft‑restarts help to escape plateaus.

The algorithm runs entirely in‑memory, uses only NumPy and the provided
`evaluate_sequence` (via a lightweight replica for progress reporting) and
keeps the best sequence found before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a real non‑negative vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    then re‑normalise to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.18))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _interpolate(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    # map both old and new indices to [0,1)
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.full(new_n, 1.0 / new_n)
    return a_new


def _shape_power(n: int, p: float) -> np.ndarray:
    """Arcsine‑type shape with arbitrary exponent p (p=0.5 gives the classic
    1/√(i·(n‑1‑i)) curve)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


def _shape_geo(n: int, r: float) -> np.ndarray:
    """Symmetric geometric shape a_i ∝ r^i + r^{n‑1‑i}."""
    i = np.arange(n, dtype=np.float64)
    vals = np.power(r, i) + np.power(r, n - 1 - i)
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


def _shape_double_tail(n: int, p: float) -> np.ndarray:
    """Two power‑law tails a_i ∝ (i+½)^{-p} + (n‑i‑½)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-p) + (n - i - 0.5) ** (-p)
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


def _compute_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence
    up to clipping – our vectors stay safely within the limits)."""
    n = a.size
    conv = _conv_self(a, fft_len)
    max_conv = conv.max()
    return 2.0 * n * max_conv / (a.sum() ** 2)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global‑time bookkeeping
    # ------------------------------------------------------------------
    safety = 5.0  # keep a few seconds margin
    total_budget = max(0.0, float(budget_s) - safety)
    start_time = time.monotonic()
    deadline = start_time + total_budget

    # ------------------------------------------------------------------
    # 1) Small‑scale parametric search for a good seed shape
    # ------------------------------------------------------------------
    n_small = 512                     # cheap to evaluate
    fft_len_small = _fft_len_for_n(n_small)

    best_a = np.full(n_small, 1.0 / n_small, dtype=np.float64)
    best_score = _compute_score(best_a, fft_len_small)

    # 1a) Shapes based on power‑law / arcsine family
    for p in np.linspace(0.30, 0.80, 6):
        cand = _shape_power(n_small, p)
        s = _compute_score(cand, fft_len_small)
        if s < best_score:
            best_score, best_a = s, cand

    # 1b) Symmetric geometric tails
    for r in np.linspace(0.88, 0.99, 6):
        cand = _shape_geo(n_small, r)
        s = _compute_score(cand, fft_len_small)
        if s < best_score:
            best_score, best_a = s, cand

    # 1c) Double‑tail power‑law
    for p in np.linspace(0.30, 0.80, 6):
        cand = _shape_double_tail(n_small, p)
        s = _compute_score(cand, fft_len_small)
        if s < best_score:
            best_score, best_a = s, cand

    # 1d) If a previous solution is available, try to assimilate it
    if "height_sequence_1" in globals():
        try:
            previous = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if previous.ndim == 1 and previous.size > 0 and np.isfinite(previous).all():
                # down‑sample / up‑sample to the small size
                cand = _interpolate(previous, n_small)
                cand = np.clip(cand, 0.0, None)
                cand_sum = cand.sum()
                if cand_sum > 0.0:
                    cand /= cand_sum
                else:
                    cand = np.full(n_small, 1.0 / n_small)
                s = _compute_score(cand, fft_len_small)
                if s < best_score:
                    best_score, best_a = s, cand
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) Stage‑1 optimisation on the small problem (quick exploration)
    # ------------------------------------------------------------------
    def _optimize_stage(a: np.ndarray,
                        n: int,
                        time_limit: float,
                        eta_start: float,
                        eta_end: float,
                        beta_start: float,
                        beta_end: float,
                        pair_move_prob: float,
                        block_perturb_prob: float,
                        restart_gap: float,
                        momentum: float):
        """Mirror‑descent optimiser on a fixed length `n` for at most
        `time_limit` seconds.  Returns the best vector seen."""
        ff_len = _fft_len_for_n(n)
        a = a.copy()
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        best_local = a.copy()
        best_local_score = _compute_score(a, ff_len)

        grad_prev = np.zeros_like(a)
        start = time.monotonic()
        last_improve = start

        while True:
            now = time.monotonic()
            if now - start >= time_limit:
                break

            # convolution and current score
            conv = _conv_self(a, ff_len)
            cur_max = float(conv.max())
            cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)   # sum(a) == 1 → 2·n·max

            # record improvement
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best (stage {n}) = {best_local_score:.6f}")

            # schedule hyper‑parameters linearly in the stage’s time
            elapsed_stage = now - start
            frac = elapsed_stage / time_limit
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # soft‑max weights for the max‑norm (smooth approximation)
            max_conv = conv.max()
            w = np.exp(beta * (conv - max_conv))   # w_max == 1

            # gradient of  Σ w_k·conv_k  w.r.t. a (see analysis notes)
            a_rev = a[::-1]
            L = (2 * n - 1) + n - 1
            fft_len_grad = _fft_len_for_len(L)
            fw = np.fft.rfft(w, n=fft_len_grad)
            fa = np.fft.rfft(a_rev, n=fft_len_grad)
            grad_full = np.fft.irfft(fw * fa, n=fft_len_grad)
            # slice the central n components
            grad = 2.0 * grad_full[n - 1 : n - 1 + n] - a * w[2 * np.arange(n)]

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient update (mirror descent on simplex)
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # occasional random block perturbation
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # occasional pairwise mass transfer
            if rng.random() < pair_move_prob:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    # Move a fraction of mass from i to j
                    delta = rng.random() * min(a[i], 0.3 * a[i])
                    if delta > 0.0:
                        a_new = a.copy()
                        a_new[i] -= delta
                        a_new[j] += delta
                        a_new = np.clip(a_new, 0.0, None)
                        a_new /= a_new.sum()
                        conv_new = _conv_self(a_new, ff_len)
                        new_score = 2.0 * n * np.max(conv_new) / (np.sum(a_new) ** 2)
                        if new_score < cur_score:
                            a = a_new
                            cur_score = new_score
                            conv = conv_new
                            cur_max = conv_new.max()

            # soft restart if stuck for a while
            if now - last_improve > restart_gap:
                a = best_local.copy()
                noise = np.exp(rng.normal(0.0, 0.12, size=n))
                a = a * noise
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                grad_prev = np.zeros_like(a)
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] restart (stage {n}) – best {best_local_score:.6f}")

        return best_local, best_local_score

    # Stage‑1: work on the small problem for about 15 % of the total budget
    t_stage1 = total_budget * 0.15
    best_small, best_small_score = _optimize_stage(
        a=best_a,
        n=n_small,
        time_limit=t_stage1,
        eta_start=1.0,
        eta_end=0.02,
        beta_start=0.5,
        beta_end=800.0,
        pair_move_prob=0.10,
        block_perturb_prob=0.06,
        restart_gap=25.0,
        momentum=0.85,
    )

    # ------------------------------------------------------------------
    # 2) Upscale to the final problem size and continue optimisation
    # ------------------------------------------------------------------
    n_final = 5000                     # final length – a good compromise
    a_up = _interpolate(best_small, n_final)
    a_up = np.clip(a_up, 0.0, None)
    a_up /= a_up.sum()

    # adapt hyper‑parameters for the long final stage
    t_remaining = max(0.0, deadline - time.monotonic())
    # allocate the rest of the budget to the final stage
    best_final, best_final_score = _optimize_stage(
        a=a_up,
        n=n_final,
        time_limit=t_remaining,
        eta_start=0.9,
        eta_end=0.015,
        beta_start=0.8,
        beta_end=1500.0,
        pair_move_prob=0.12,
        block_perturb_prob=0.07,
        restart_gap=30.0,
        momentum=0.9,
    )

    # ------------------------------------------------------------------
    # 3) Very short local refinement on the best vector (if time permits)
    # ------------------------------------------------------------------
    # The refinement consists of many cheap random pairwise moves.
    end_time = time.monotonic()
    time_left = deadline - end_time
    if time_left > 2.0:
        a_refine = best_final.copy()
        n = a_refine.size
        ff_len = _fft_len_for_n(n)
        cur_score = _compute_score(a_refine, ff_len)

        # simple hill‑climbing with a fixed amount of trials
        trials = int(5e4 * time_left / total_budget) + 1000
        for _ in range(trials):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * min(a_refine[i], 0.2 * a_refine[i])
            if delta == 0.0:
                continue
            a_new = a_refine.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()
            conv_new = _conv_self(a_new, ff_len)
            new_score = 2.0 * n * np.max(conv_new) / (np.sum(a_new) ** 2)
            if new_score < cur_score:
                a_refine = a_new
                cur_score = new_score

        # keep the refined version if it improved
        if cur_score < best_final_score:
            best_final = a_refine
            best_final_score = cur_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] refined best score = {best_final_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to what evaluate_sequence accepts)
    # ------------------------------------------------------------------
    best_final = np.clip(best_final, 0.0, 1000.0)
    return [float(x) for x in best_final]
