Adam‑based mirror‑descent with soft‑max smoothing and stochastic
perturbations.

The objective that ``evaluate_sequence`` finally reports is
    S(a) = 2·n·max(conv(a,a))  (the denominator (∑a)² is fixed to 1
when we stay on the probability simplex).

The algorithm works entirely on the simplex:

* a is kept non‑negative and normalised to sum to one.
* At each iteration we compute the self‑convolution a∗a with a fast FFT.
* A temperature β(t) gradually grows, turning a soft‑max over the
  convolution entries into a sharper approximation of the true max.
* The resulting smooth gradient is fed to an Adam optimiser that updates
  the *log‑probabilities* (mirroring an exponentiated‑gradient step).
* To avoid getting stuck we repeatedly jitter a random block, swap two
  coordinates, inject fresh random Dirichlet candidates and, after a
  period without improvement, restart from the current best solution.
* The best sequence encountered is returned (clipped to the limits used
  inside ``evaluate_sequence``).

The routine respects the 1000 s time budget, prints progress, and
produces a list of non‑negative floats ready to be evaluated.
"""

import time
import numpy as np

# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, N: int = None) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    if N is None:
        N = _next_pow2(n)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:n]
    return conv


def _random_block_perturb(
    a: np.ndarray,
    rng: np.random.Generator,
    width_frac: float = 0.1,
    sigma: float = 0.3,
) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and renormalise.
    width_frac controls the maximal block size as a fraction of the full length.
    sigma is the standard deviation of the underlying normal distribution.
    """
    n = a.size
    max_width = max(2, int(width_frac * n))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- keep the best -------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule (linear interpolation) -------------------------------
        elapsed = time.monotonic() - start_time
        total = (budget_s - 5.0)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- soft‑max weights (smooth approximation of the max) ------------
        shift = conv.max()                        # avoid overflow
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical under‑flow – fall back to uniform weights
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective ----------------------------
        a_rev = a[::-1]                           # reversed copy
        grad_full = _conv_fft(w, a_rev, N=fft_len_grad)  # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n] # length n

        # ---- Adam update on log‑probabilities -------------------------------
        m = beta1_adam * m + (1.0 - beta1_adam) * grad
        v = beta2_adam * v + (1.0 - beta2_adam) * (grad ** 2)

        m_hat = m / (1.0 - beta1_adam ** t_iter)
        v_hat = v / (1.0 - beta2_adam ** t_iter)

        step = lr_base / (np.sqrt(v_hat) + eps_adam)
        log_a = np.log(a + eps_adam) - step * m_hat
        a = np.exp(log_a)

        # ---- projection onto the simplex (entropy mirror descent) ----------
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- occasional stochastic perturbations ---------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(
                a, rng, width_frac=0.15, sigma=jitter_sigma_block
            )

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]

        # ---- restart if we have been idle for a while ----------------------
        now = time.monotonic()
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma_restart, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

        # ---- inject a fresh random candidate every ``inject_gap`` seconds ---
        if (now - start_time) % inject_gap < 0.1:
            # a Dirichlet with smaller alpha creates a sparse candidate
            cand = rng.dirichlet(np.full(n, 0.5))
            conv_cand = _conv_self(cand)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                last_improve = now
                print(
                    f"[{now - start_time:6.2f}s] injected better candidate = {best_score:.6f}"
                )
            # possibly replace the working point if it is worse
            if cand_score < cur_score:
                a = cand.copy()
                m.fill(0.0)
                v.fill(0.0)

    # ----------------------------------------------------------------------
    # 5) Return the best solution (clipped to the limits used by evaluate_sequence)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
