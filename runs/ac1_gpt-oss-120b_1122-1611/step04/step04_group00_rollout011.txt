```python
def _gradient_from_weights(a, w, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad
```

    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _init_uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _init_exponential(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    total = a.sum()
    if total == 0.0:
        return _init_uniform(n)
    return a / total


def _init_linear_desc(n: int) -> np.ndarray:
    """Linearly decreasing sequence a_i ∝ (n‑i)."""
    a = np.arange(n, 0, -1, dtype=np.float64)  # n, n‑1, …, 1
    return a / a.sum()


def _init_quad_desc(n: int) -> np.ndarray:
    """Quadratically decreasing sequence a_i ∝ (n‑i)²."""
    a = np.square(np.arange(n, 0, -1, dtype=np.float64))
    return a / a.sum()


def _init_power_law_desc(n: int, p: float) -> np.ndarray:
    """Power‑law decreasing a_i ∝ (i+1)^{-p} (reversed to be descending)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -p)[::-1]
    return a / a.sum()


def _initial_shapes(n: int, rng: np.random.Generator):
    """Generate a modest pool of candidate initial vectors."""
    shapes = []

    # uniform
    shapes.append(_init_uniform(n))

    # linear and quadratic descending
    shapes.append(_init_linear_desc(n))
    shapes.append(_init_quad_desc(n))

    # exponential families
    for r in (0.90, 0.94, 0.96, 0.98, 0.99):
        shapes.append(_init_exponential(n, r))

    # power‑law families
    for p in (0.5, 1.0, 1.5, 2.0):
        shapes.append(_init_power_law_desc(n, p))

    # a few random Dirichlet draws
    for _ in range(5):
        shapes.append(rng.dirichlet(np.ones(n)))

    return shapes


def _score_vector(a: np.ndarray, n: int, fft_len_conv: int) -> float:
    """Exact objective value 2·n·max(conv(a,a)) / (sum a)²."""
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1000.0):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose problem size (power‑of‑two, not larger than 8192)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            base = np.clip(base, 0.0, None)
            if base.sum() == 0.0:
                base = np.full_like(base, 1.0 / base.size)
            else:
                base = base / base.sum()
            n0 = base.size
        except Exception:
            n0 = 1024
    else:
        n0 = 1024

    target_len = max(n0, 1024)
    n = 1 << ((target_len - 1).bit_length())          # next power of two
    n = min(n, 8192)                                   # keep FFT cheap

    # ------------------------------------------------------------------
    # 2) Initialise a feasible vector a on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            base = np.clip(base, 0.0, None)
            if base.sum() == 0.0:
                base = np.full_like(base, 1.0 / base.size)
            else:
                base = base / base.sum()
            if base.size >= n:
                a = base[:n].copy()
            else:
                a = np.concatenate([base, np.zeros(n - base.size, dtype=np.float64)])
            a = _project_simplex(a)
        except Exception:
            a = _init_uniform(n)
    else:
        a = _init_uniform(n)

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT lengths used repeatedly
    # ------------------------------------------------------------------
    fft_len_conv = _fft_len_for_n(2 * n - 1)          # for self‑convolution
    fft_len_grad = _fft_len_for_n(3 * n - 2)          # for gradient convolution

    # ------------------------------------------------------------------
    # 4) Quick search over a handful of simple shapes
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = _score_vector(best, n, fft_len_conv)

    candidates = _initial_shapes(n, rng)
    candidates.append(a)  # ensure the current vector is considered

    for cand in candidates:
        cand = _project_simplex(cand)
        sc = _score_vector(cand, n, fft_len_conv)
        if sc < best_score:
            best_score = sc
            best = cand.copy()

    # start optimisation from the best candidate found so far
    a = best.copy()

    # ------------------------------------------------------------------
    # 5) Adam optimiser with temperature schedule and restarts
    # ------------------------------------------------------------------
    m = np.zeros_like(a)           # first‑moment buffer
    v_buf = np.zeros_like(a)       # second‑moment buffer
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    start_time = time.monotonic()
    last_improve = start_time

    # hyper‑parameters (linear learning‑rate schedule, exponential β schedule)
    eta_start = 0.5
    eta_end = 0.02
    beta_temp_start = 0.5
    beta_temp_end = 5000.0
    block_perturb_prob = 0.03
    restart_gap = 30.0            # seconds without improvement before a restart
    iteration = 0

    while time.monotonic() < deadline:
        # ----- current objective -----
        conv = _conv_self(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a.sum() ** 2)   # a.sum() == 1 after projection

        # ----- keep best -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule updates -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # linear decay of the learning‑rate
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # exponential growth of the temperature β
        beta_temp = beta_temp_start * (beta_temp_end / beta_temp_start) ** (elapsed / total)

        # ----- compute weights for the smoothed objective -----
        if beta_temp < 50.0:
            # soft‑max (log‑sum‑exp) weights
            shift = conv.max()
            w_raw = np.exp(beta_temp * (conv - shift))
            w = w_raw / w_raw.sum()
        else:
            # sub‑gradient: equal weight on all entries within a tiny tolerance of the max
            tol = max(1e-12, cur_max * 1e-8)
            active = np.where(conv >= cur_max - tol)[0]
            w = np.zeros_like(conv)
            if active.size > 0:
                w[active] = 1.0 / float(active.size)
            else:
                w[conv.argmax()] = 1.0

        # ----- gradient (with respect to a) -----
        grad = _gradient_from_weights(a, w, fft_len_grad)

        # ----- Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_buf / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_tmp = a - step
        a = _project_simplex(a_tmp)

        # ----- occasional random block perturbation (helps escape plateaus) -----
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a_block = a.copy()
            a_block[start:start + width] *= factor
            a = _project_simplex(a_block)

        # ----- restart on stagnation -----
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            a = best * jitter
            a = _project_simplex(a)
            m.fill(0.0)
            v_buf.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence seen (clipped to the allowed range)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best.tolist()]
