"""
Algorithm: Exponentiated‑gradient descent with top‑k soft‑max smoothing.

We minimise
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

under the constraints a_i ≥ 0 and ∑a_i = 1 (the simplex).  After each
iteration we

1. compute the self‑convolution of a with an FFT,
2. identify the K largest convolution entries (K = 5 by default) and
   form a weighted gradient of those entries (weights are a soft‑max with a
   temperature β that is annealed from a low to a very high value),
3. perform a multiplicative (exponentiated) update
        a_i ← a_i · exp(−η·g_i)
   and renormalise to the simplex (the update automatically respects
   non‑negativity),
4. enforce mirror symmetry a = (a + a[::-1]) / 2 (the optimal solution
   is known to be symmetric) and occasionally apply a random block
   perturbation to escape shallow minima,
5. restart from the current best after a stagnation period.

All operations are vectorised and use only FFT‑based convolutions, so the
method easily fits the 1000 s time budget even for several‑thousand‑length
vectors.  The returned sequence is the best one seen during the run.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using a pre‑allocated FFT length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _topk_gradient(a: np.ndarray, conv: np.ndarray, k: int,
                   beta: float) -> np.ndarray:
    """
    Weighted gradient obtained from the K largest entries of the convolution.
    For each selected index k0 the exact gradient of (a∗a)[k0] is
        g_i = 2 * a_{k0-i}   (zero outside the support).
    The final gradient is the convex combination of those K gradients,
    where the combination weights are a soft‑max with temperature ``beta``.
    """
    n = a.size
    rev = a[::-1]                     # reversed a for fast slice handling
    # indices of the K largest convolution entries
    if k >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(-conv, k - 1)[:k]

    # soft‑max weights (shifted for numerical stability)
    c_max = conv.max()
    w_raw = np.exp(beta * (conv[idx] - c_max))
    w = w_raw / w_raw.sum()

    # accumulate weighted gradient
    g = np.zeros_like(a)
    for weight, kk in zip(w, idx):
        offset = n - 1 - kk
        if offset >= 0:
            length = n - offset
            g[:length] += weight * 2.0 * rev[offset:offset + length]
        else:
            off = -offset
            length = n - off
            g[off:] += weight * 2.0 * rev[:length]
    return g


def _symmetrify(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} and renormalise."""
    a = (a + a[::-1]) * 0.5
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    return a


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 15))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.22))
    b = a.copy()
    b[start:start + width] *= factor
    # renormalise to the simplex
    b = b / b.sum()
    return b

    start_time = time.monotonic()
    last_improve = start_time

    # current best
    conv_best = _self_conv_fft(a, fft_len)
    best_score = 2.0 * n * conv_best.max()
    best = a.copy()

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters.
    # ------------------------------------------------------------------
    eta_start = 5.0          # initial multiplicative step size
    eta_end = 0.15           # final step size
    beta_start = 0.5
    beta_end = 8000.0
    top_k = 5                # number of top convolution entries to use
    perturb_prob = 0.03
    restart_gap = 28.0       # seconds without improvement → restart
    eps = 1e-12              # for log‑stability

    # ------------------------------------------------------------------
    # 5) Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- compute convolution and current score -----
        conv = _self_conv_fft(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep the best solution seen so far -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- annealing schedules for η and β -----
        now = time.monotonic()
        t_elapsed = now - start_time
        t_total = deadline - start_time
        eta = eta_start * (1.0 - t_elapsed / t_total) + eta_end * (t_elapsed / t_total)
        beta = beta_start * (beta_end / beta_start) ** (t_elapsed / t_total)

        # ----- weighted gradient from the top‑k convolution entries -----
        g = _topk_gradient(a, conv, top_k, beta)

        # ----- exponentiated‑gradient (multiplicative) update -----
        # work in the log‑domain for stability
        log_a = np.log(a + eps)
        log_a_new = log_a - eta * g
        # shift to avoid overflow in exp
        log_a_new -= log_a_new.max()
        a = np.exp(log_a_new)
        a /= a.sum()                     # projection onto the simplex

        # ----- enforce symmetry (known to be advantageous) -----
        a = _symmetrify(a)

        # ----- occasional random block perturbation -----
        if rng.random() < perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- restart if stagnation lasts too long -----
        if now - last_improve > restart_gap:
            # jitter the current best and continue from there
            noise = np.exp(rng.normal(0.0, 0.15, size=n))
            a = best * noise
            a /= a.sum()
            a = _symmetrify(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best vector (clipped only to satisfy evaluate_sequence).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
