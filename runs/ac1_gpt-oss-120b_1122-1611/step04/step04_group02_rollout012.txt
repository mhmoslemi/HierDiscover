                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):

                    cur_score = new_score

        # update best

            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best: {best_score:.6f}")

            last_improve = now
            print(f"[{elapsed:6.2f}s] restart from best.")

        # optionally occasionally increase n by adding zeros?
        # Not needed for now.

    # end while

    # final best scaled and clipped
    best = np.clip(best, 0.0, 1000.0)
    # return as list of floats

"""
Propose a step function (non‑negative heights) with a low value of
``evaluate_sequence``.  The method combines

1. **Full‑softmax mirror descent** – a smooth approximation of the
   infinity norm of the self‑convolution (via the log‑sum‑exp) is
   minimised on the probability simplex.  The gradient is obtained
   exactly by a single FFT‑based convolution with the softmax weights.
   Step‑size η decays linearly, while the temperature β grows
   exponentially, so the update gradually focuses on the true peak.

2. **Stochastic annealing refinement** – after the first stage a
   simulated‑annealing style local search (pairwise mass transfers and
   block perturbations) tries to escape shallow minima and to polish
   the solution.

3. **Restart on stagnation** – if no improvement is seen for a while
   the current best solution is jittered and used as a new start.

The routine respects a 1000 s time budget (a 5 s safety margin is kept)
and returns the best sequence found.  All helper functions are top‑level
so that the harness can import them without closures.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x (x > 0)."""
    if x <= 0:
        return 1
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using a FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    return conv_full[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, beta: float, fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the soft‑max approximation of max(conv) w.r.t.
    the probability vector ``a``.
    The gradient is 2 * (a * w_rev) where w_i = exp(beta*(conv_i-max))/Z.
    """
    # soft‑max weights (stabilised by subtracting the max)
    w_raw = np.exp(beta * (conv - conv.max()))
    w = w_raw / w_raw.sum()

    # convolution a * w_rev  (full length = 3n-2)
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    grad_full = np.fft.irfft(fa * fw, n=fft_len)[: 3 * a.size - 2]

    n = a.size
    # slice n‑1 … 2n‑2 (inclusive) and reverse to obtain the correct order
    grad_slice = grad_full[n - 1 : 2 * n - 1][::-1]
    return 2.0 * grad_slice


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector (sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full_like(a, 1.0 / a.size)
        else:
            a = rng.dirichlet(np.ones(256)).astype(np.float64)
    else:
        # try a moderately large length; the exact value is not critical
        a = rng.dirichlet(np.ones(3809)).astype(np.float64)  # 3809 was the previous best length

    n = a.size
    fft_len = _next_pow2(3 * n - 2)          # big enough for both convolutions

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len)
    cur_score = 2.0 * n * conv.max()        # sum(a)=1 ⇒ evaluate_sequence == this
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    total_time = deadline - start_time
    phase1_end = start_time + 0.75 * total_time   # 75 % for mirror‑descent

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters for the mirror‑descent stage
    # ------------------------------------------------------------------
    eta_start = 1.0
    eta_end = 0.015
    beta_start = 0.5
    beta_end = 800.0
    momentum = 0.9
    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – full‑softmax mirror descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear decay for η, exponential growth for β
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start * (beta_end / beta_start) ** frac

            # compute self‑convolution and soft‑max gradient
            conv = _conv_self_fft(a, fft_len)
            grad = _grad_softmax(a, conv, beta, fft_len)

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient (mirror‑descent) update on the simplex
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # occasional random jitter to retain diversity
            if rng.random() < 0.015:
                a = _random_block_perturb(a, rng)

            # recompute score after the update
            conv = _conv_self_fft(a, fft_len)
            cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Phase 2 – stochastic local refinement (annealing)
        # --------------------------------------------------------------
        else:
            # temperature schedule (linear decay)
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = max(1e-8, 0.07 * (1.0 - t_frac))

            # ----- pairwise mass transfer --------------------------------
            if rng.random() < 0.85:
                i = rng.integers(0, n)
                if a[i] < 1e-12:
                    continue
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                max_delta = 0.30 * a[i]
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta

                # fast score evaluation via FFT
                conv_new = _conv_self_fft(a_new, fft_len)
                new_score = 2.0 * n * conv_new.max()

                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                    cur_score = new_score

            # ----- block‑perturbation (global jitter) -------------------
            else:
                a = _random_block_perturb(a, rng)
                conv = _conv_self_fft(a, fft_len)
                cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Restart if stuck for a while (30 s without improvement)
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
