def project_onto_simplex(v):
    # ensures sum(v) = 1 and v >=0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

A hybrid optimisation that starts from a family of analytically‑motivated
step‑functions (uniform, arcsine, mixtures and exponentials), picks the best
as a seed and then improves it with a mirror‑descent / exponentiated‑gradient
scheme.  The gradient is a blend of a *soft‑max* of all convolution entries
(and therefore a smooth approximation of the sub‑gradient of the ∞‑norm) and
the exact sub‑gradient at the current maximising lag (hard‑max).  The blend
shifts gradually from soft to hard as the temperature β grows, allowing a
broad exploration early on and an aggressive focus on the true max later.

Additional ingredients:
* linear schedules for the step‑size η and the temperature β,
* momentum on the gradient,
* occasional log‑normal block perturbations,
* random pair‑wise mass transfers,
* a soft restart when no improvement is seen for a while,
* a quick parametric sweep (arcsine‑mix, power‑law, exponential decay) before
  the main loop to avoid poor local minima.

The routine respects the 1000 s wall‑time limit, prints progress and returns
the best non‑negative sequence found (clipped to the limits enforced by
``evaluate_sequence``).  It works entirely with NumPy – no external solvers
are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _score_of(a: np.ndarray, fft_len: int) -> float:
    """Return the evaluation score 2·n·max(conv(a,a)) (sum(a) is assumed 1)."""
    n = a.size
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the allotted wall‑time expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine length and an initial distribution
    # ------------------------------------------------------------------
    default_n = 3800                               # a sensible default
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a0 = np.clip(cand, 0.0, None)
                s = a0.sum()
                a0 = a0 / s if s > 0.0 else np.full(n, 1.0 / n)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Quick parametric sweep to obtain a good seed
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # basic shapes
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = np.clip(arcsine, 0.0, None)
    arcsine /= arcsine.sum()
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # helper for scoring (self‑convolution length for this n)
    fft_len_n = _fft_len_for_n(n)

    best_a = a0.copy()
    best_score = _score_of(best_a, fft_len_n)

    # 2a) convex combination of arcsine & uniform
    for lam in np.linspace(0.0, 1.0, 21):
        cand = lam * arcsine + (1.0 - lam) * uniform   # already normalised
        sc = _score_of(cand, fft_len_n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()

    # 2b) power‑law family  a_i ∝ ((i+½)(n‑i‑½))^{-p}
    for p in np.linspace(0.30, 0.80, 26):
        shape = ((i + 0.5) * (n - i - 0.5)) ** (-p)
        shape = shape / shape.sum()
        sc = _score_of(shape, fft_len_n)
        if sc < best_score:
            best_score, best_a = sc, shape.copy()

    # 2c) symmetric exponential decay from the centre
    centre = (n - 1) / 2.0
    for decay in np.linspace(0.001, 0.010, 10):
        shape = np.exp(-decay * np.abs(i - centre))
        shape = shape / shape.sum()
        sc = _score_of(shape, fft_len_n)
        if sc < best_score:
            best_score, best_a = sc, shape.copy()

    a = best_a.copy()                     # current working vector

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters for the mirror‑descent phase
    # ------------------------------------------------------------------
    eta_start = 1.5          # aggressive start
    eta_end = 0.02           # fine finish
    beta_start = 0.1
    beta_end = 2000.0        # very sharp soft‑max → hard max
    momentum = 0.9
    block_perturb_prob = 0.05
    pair_move_prob = 0.07
    restart_gap = 30.0       # seconds without improvement → restart

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --- evaluate current candidate ---------------------------------
        conv = _conv_self(a, fft_len_n)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max          # sum(a) == 1 by construction

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --- schedule hyper‑parameters ---------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # --- soft‑max weights (smooth approximation of max) ----------
        max_conv = conv.max()
        w = np.exp(beta * (conv - max_conv))           # max weight = 1

        # gradient of the soft‑max weighted sum (as in the reference code)
        a_rev = a[::-1]
        L = (2 * n - 1) + n - 1
        fft_len_gr = _fft_len_for_len(L)
        fw = np.fft.rfft(w, n=fft_len_gr)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_gr)
        grad_full = np.fft.irfft(fw * fa_rev, n=fft_len_gr)
        grad_soft = 2.0 * grad_full[n - 1 : n - 1 + n] - a * w[2 * np.arange(n)]

        # --- hard‑max gradient (exact sub‑gradient of the max) ---------
        k_max = int(np.argmax(conv))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_hard = np.zeros_like(a)
        grad_hard[mask] = 2.0 * a[idx[mask]]

        # --- blend soft and hard gradients ----------------------------
        alpha = (beta - beta_start) / (beta_end - beta_start)   # 0 → 1
        grad = (1.0 - alpha) * grad_soft + alpha * grad_hard

        # --- momentum smoothing ---------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad

        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -50.0, 50.0)   # avoid overflow
        factor = np.exp(exp_arg)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:

        else:
            a /= s

        # --- occasional random block perturbation ----------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # --- occasional pair‑wise mass transfer ------------------------
        if rng.random() < pair_move_prob:
            i1, i2 = rng.integers(0, n, size=2)
            if i1 != i2:
                max_delta = min(a[i1], 0.3 * a[i1])
                if max_delta > 0.0:

                    a_new[i1] -= delta

    # ------------------------------------------------------------------
