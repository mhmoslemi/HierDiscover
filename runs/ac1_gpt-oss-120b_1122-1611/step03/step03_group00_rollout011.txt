def propose_candidate(...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5
    start = time.monotonic()
    best_seq = None
    best_score = np.inf

    # baseline from height_sequence_1 if exists
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            cand = np.clip(cand, 0., None)
            if cand.sum() > 0.:
                cand = cand / cand.sum()
                score = compute_score(cand)
                best_seq = cand.copy()
                best_score = score
                print(f"Baseline score: {best_score:.6f}")
        except Exception:
            pass

    # Stage 1: grid search over shapes
    ns = [1024, 2048, 4096, 8192]   # choose perhaps 2048, 4096 for speed
    gammas = np.linspace(0, 2.0, 41)  # inclusive
    for n in ns:
        fft_len = _fft_len_for_n(n)
        for gamma in gammas:
            idx = np.arange(n)
            weight = ((idx+1)*(n - idx)) ** (-gamma)
            # avoid infinities for gamma large: compute with float64
            # ensure nonzero: weight may become zero? Not; product >0.
            seq = weight / weight.sum()
            score = 2.0 * n * np.max(_conv_self(seq, fft_len))
            if score < best_score:
                best_score=score
                best_seq=seq.copy()
                print(f"[{time.monotonic()-start:.2f}s] shape n={n} gamma={gamma:.3f} score={score:.6f}")
    
    # Stage 2: refinement using mirror descent on best_seq length n_best
    if best_seq is not None:
        a = best_seq.copy()
        n = a.size
        fft_len = _fft_len_for_n(n)
        # parameters
        eta_start = 0.5
        eta_end = 0.02
        restart_gap = 20.0
        last_improve = time.monotonic()
        grad_prev = np.zeros_like(a)
        while time.monotonic() < deadline:
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max   # sum=1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve-start:.2f}s] improved to {best_score:.6f}")
            # schedule
            elapsed = time.monotonic() - start
            total = deadline - start
            eta = eta_start * (1 - elapsed / total) + eta_end * (elapsed / total)
            # compute subgradient at argmax
            max_idx = np.where(conv == cur_max)[0]   # might be multiple
            grad = np.zeros_like(a)
            rev_a = a[::-1]
            for k in max_idx:
                if k < n:
                    grad[: k+1] += 2 * a[k::-1]
                else:
                    start_shift = k - (n-1)
                    length = 2*n-1 - k
                    grad[start_shift:] += 2 * rev_a[:length]
            # momentum
            momentum = 0.9
            grad = momentum * grad_prev + (1 - momentum) * grad
            grad_prev = grad
            # mirror descent update
            a = a * np.exp(-eta * grad)
            a = np.clip(a, 0, None)
            s = a.sum()
            if s == 0:
                a = np.full_like(a, 1.0/n)
            else:
                a /= s
            # occasional random jitter
            if rng.random() < 0.05:
                noise = np.exp(rng.normal(0., 0.1, size=n))
                a *= noise
                a = np.clip(a, 0, None)
                a /= a.sum()
            # restart if stuck
            if time.monotonic() - last_improve > restart_gap:
                a = best_seq.copy()
                # jitter
                noise = np.exp(rng.normal(0., 0.2, size=n))

                a = np.clip(a, 0, None)
                a /= a.sum()
                grad_prev = np.zeros_like(a)
                last_improve = time.monotonic()
                print(f"[{last_improve-start:.2f}s] restart")
    else:
        # fallback: random vector
        pass

    # final returns
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

def propose_candidate(seed=42, budget_s=1000.0):

    deadline = time.monotonic() + budget_s - 5

    # baseline
    ...

    best_seq = baseline_seq or None

    shape_params = []

            # regularize by adding small epsilon to avoid zero
            seq = w / w.sum()
            score = compute_score(seq)  # 2*n*max(conv)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_n = n
                best_gamma = gamma
                print(f"New best shape n={n} gamma={gamma:.3f} score={score:.6f}")

    # Stage 1b: refine gamma in local region
    if best_seq is not None:
        n = best_seq.size
        fft_len = _fft_len_for_n(n)

        gamma = best_gamma
        for _ in range(200):
            if time.monotonic() >= deadline:
                break
            new_gamma = gamma + rng.normal(scale=0.05)
            if new_gamma < 0:
                new_gamma = 0.0
            w = ((np.arange(n)+1)*(n-np.arange(n))) ** (-new_gamma)
            seq = w / w.sum()
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                gamma = new_gamma
                print(f"Improved gamma {new_gamma:.3f} -> score {score:.6f}")
            else:
                # occasional acceptance for diversification?
                if rng.random() < 0.1:
                    gamma = new_gamma

    # Stage 2: mirror descent refinement starting from best_seq
    if best_seq is not None:
        a = best_seq.copy()
        n = a.size
        fft_len = _fft_len_for_n(n)
        # hyperparameters
        eta_start = 0.3
        eta_end = 0.001
        beta_start = 0.5
        beta_end = 400.
        K_active = max(100, n // 4)   # choose
        restart_gap = 15.0
        last_improve = time.monotonic()
        grad_prev = np.zeros_like(a)
        momentum = 0.9
        while time.monotonic() < deadline:
            # compute conv and score
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start:.2f}s] refined best {best_score:.6f}")
            # schedule
            now = time.monotonic()
            elapsed = now - start
            total = deadline - start
            eta = eta_start * (1 - elapsed/total) + eta_end * (elapsed/total)

            if K_active >= conv.size:
                active_idx = np.arange(conv.size)
            else:
                kth = conv.size - K_active
                active_idx = np.argpartition(conv, kth)[kth:]
            # compute gradient
            grad = np.zeros_like(a)
            rev_a = a[::-1]
            max_conv = conv.max()

                if w == 0.0:

            a = np.clip(a, 0.0, None)

                            a = np.clip(a, 0.0, None)
                            a /= a.sum()
                            conv = new_conv
                            cur_score = new_score

            # restart if stuck
            if now - last_improve > restart_gap:
                a = best_seq.copy()

                a = a * noise

                grad_prev = np.zeros_like(a)
                last_improve = now

"""
propose_candidate
-----------------
A two‑stage heuristic for minimising

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm first scans a low‑dimensional family of “edge‑heavy”
shapes a_i ∝ ((i+1)*(n-i))^{-γ} for several lengths `n` and several
exponents `γ`.  Those shapes already give a substantial reduction of the
max‑convolution compared with the uniform vector.  The best shape found
is then refined with a mirror‑descent (exponentiated‑gradient) optimiser
that works directly on the simplex and uses a soft‑max weighting of the
largest convolution entries.  Random block perturbations, pair‑wise
mass transfers and timed restarts help escape local minima.

The whole search runs until the time budget (default 1000 s) expires and
returns the best sequence seen.  All entries are non‑negative, finite and
clipped to the limits enforced by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two that can hold the full linear convolution
    of two length‑`n` vectors."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using FFT.  Returns a vector of length
    2*len(a)-1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    then renormalise to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """Contribution of a single entry a[idx] to the full linear convolution.
    Length is 2*n-1."""
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx : idx + n] = a
    return v


def _score_from_seq(a: np.ndarray) -> float:
    """Fast evaluation of the objective when the sum of ``a`` is 1.
    Returns 2*n*max(conv(a,a))."""
    n = a.size
    fft_len = _fft_len_for_n(n)
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # 1) baseline (if a previous construction is available)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            cand = np.clip(cand, 0.0, None)
            if cand.sum() > 0.0:
                cand = cand / cand.sum()
                s = _score_from_seq(cand)
                best_seq = cand.copy()
                best_score = s
                print(f"[{time.monotonic() - start_time:.2f}s] baseline score = {s:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2) Stage‑1 : scan a simple one‑parameter family of edge‑heavy shapes
    # --------------------------------------------------------------
    # lengths we will try (they are powers of two for fast FFT)
    lengths = [1024, 2048, 4096, 8192]
    gammas = np.linspace(0.0, 3.0, 61)          # 0 … 3 in steps of 0.05
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        idx = np.arange(n, dtype=np.float64) + 1.0          # 1 … n
        rev_idx = n - idx + 1.0
        for gamma in gammas:
            # weight = ((i+1)*(n-i))^{-gamma}
            # to avoid overflow / underflow add a tiny epsilon
            w = (idx * rev_idx) ** (-gamma) + 1e-30
            seq = w / w.sum()
            score = _score_from_seq(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_len = n
                best_gamma = gamma
                print(
                    f"[{time.monotonic() - start_time:.2f}s] "
                    f"shape n={n} gamma={gamma:.3f} → score={score:.6f}"
                )
        # ------------------------------------------------------------------
        # quick local refinement of gamma around the best found for this n
        # ------------------------------------------------------------------
        if best_seq is not None and best_len == n:
            gamma = best_gamma
            for _ in range(120):
                if time.monotonic() >= deadline:
                    break
                new_gamma = gamma + rng.normal(scale=0.03)
                if new_gamma < 0.0:
                    new_gamma = 0.0
                w = (idx * rev_idx) ** (-new_gamma) + 1e-30
                seq = w / w.sum()
                score = _score_from_seq(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    gamma = new_gamma
                    best_gamma = new_gamma
                    print(
                        f"[{time.monotonic() - start_time:.2f}s] "
                        f"refined gamma≈{new_gamma:.3f} → score={score:.6f}"
                    )
                else:
                    # occasional exploration
                    if rng.random() < 0.1:
                        gamma = new_gamma

    # --------------------------------------------------------------
    # 3) Stage‑2 : mirror‑descent refinement starting from the best shape
    # --------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        n = a.size
        fft_len = _fft_len_for_n(n)

        # hyper‑parameters for the descent
        eta_start = 0.6          # initial learning‑rate
        eta_end = 0.001          # final learning‑rate
        beta_start = 0.8         # temperature at the beginning
        beta_end = 300.0         # temperature at the end
        K_active = max(200, n // 5)   # how many of the largest conv entries we use
        restart_gap = 20.0               # seconds without improvement → restart
        pair_move_prob = 0.12
        block_perturb_prob = 0.04
        momentum = 0.9

        grad_prev = np.zeros_like(a)
        last_improve = time.monotonic()

        while time.monotonic() < deadline:
            # current convolution and score (sum(a) == 1 by construction)
            conv = _conv_self(a, fft_len)
            cur_max = float(conv.max())
            cur_score = 2.0 * n * cur_max

            # keep the best solution seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.monotonic()
                print(
                    f"[{last_improve - start_time:.2f}s] new best (refined) = {best_score:.6f}"
                )

            # schedule for η and β (linear interpolation)
            now = time.monotonic()
            elapsed = now - start_time
            total = deadline - start_time
            eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
            beta = beta_start + (beta_end - beta_start) * (elapsed / total)

            # ---- soft‑max weighting of the K largest convolution entries ----------
            if K_active >= conv.size:
                active_idx = np.arange(conv.size)
            else:
                kth = conv.size - K_active
                active_idx = np.argpartition(conv, kth)[kth:]

            # compute a weighted sub‑gradient of the (smooth) max‑norm
            rev_a = a[::-1]
            max_conv = conv.max()
            grad = np.zeros_like(a)
            for k in active_idx:
                w = np.exp(beta * (conv[int(k)] - max_conv))
                if w == 0.0:
                    continue
                k = int(k)
                if k < n:
                    # contribution from the left side
                    grad[: k + 1] += w * a[k :: -1]
                else:
                    start = k - (n - 1)
                    length = (2 * n - 1) - k
                    grad[start:] += w * rev_a[:length]

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # ---- mirror‑descent (exponentiated‑gradient) update -------------------
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # ---- occasional random block perturbation -----------------------------
            if rng.random() < block_perturb_prob:
                a = _random_block_perturb(a, rng)

            # ---- cheap pair‑wise mass transfer -------------------------------------------------
            if rng.random() < pair_move_prob:
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    max_delta = min(a[i], 0.3 * a[i])
                    if max_delta > 0.0:
                        delta = rng.random() * max_delta
                        # incremental update of the convolution
                        contrib_i = _contribution_vector(a, i)
                        contrib_j = _contribution_vector(a, j)
                        new_conv = conv + delta * (contrib_j - contrib_i)
                        new_score = 2.0 * n * new_conv.max()
                        if new_score < cur_score:
                            a[i] -= delta
                            a[j] += delta
                            a = np.clip(a, 0.0, None)
                            a /= a.sum()
                            conv = new_conv
                            cur_score = new_score
                            cur_max = conv.max()

            # ---- soft restart when progress stalls ----------------------------------
            if now - last_improve > restart_gap:
                a = best_seq.copy()
                noise = np.exp(rng.normal(0.0, 0.12, size=n))
                a = a * noise
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                grad_prev = np.zeros_like(a)
                last_improve = now
                print(
                    f"[{now - start_time:.2f}s] restart – current best {best_score:.6f}"
                )

    # ----------------------------------------------------------------------
    # 4) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ----------------------------------------------------------------------
    final = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in final]
