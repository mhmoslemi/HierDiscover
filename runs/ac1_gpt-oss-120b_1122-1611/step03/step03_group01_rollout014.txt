"""
propose_candidate
-----------------

A two‑phase optimizer for the step‑function evaluation problem.

Phase 1 – cheap parametric sweep:
    * Generate candidate sequences of the form
          w[i] ∝ ((i+½)(n‑i‑½)+c)^{-p}
      (a symmetric power‑law family that includes the Catalan / arcsine shapes).
    * Scan a coarse grid over the exponent p, mixture weight α and a tiny
      regularising constant c.  The best candidate from this cheap sweep is
      taken as a warm‑start.

Phase 2 – smoothed‑max Adam descent:
    * Minimise the smooth approximation
          F(w)= (1/β)·log Σ_i exp(β·(w★w)[i])
      of the L∞‑norm of the (full) autocorrelation.
    * The gradient is obtained analytically via a second convolution.
    * Adam with a decaying stepsize projects each iterate back onto the
      probability simplex (Σw=1, w≥0).  Symmetry is enforced periodically.
    * Occasionally a random block perturbation is applied to escape local minima.

The final vector is scaled to Σw=√(2·n) (the scale used by the official
evaluation) and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.  Returns the full (len(a)+len(b)-1) result.
    Small negative rounding errors are clipped to zero.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # protect against -0 due to rounding
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Duchi et al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Large cheap random jump: multiply a random contiguous block by a
    log‑normally distributed factor; optionally enforce symmetry.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetric averaging (helps for the convolution objective)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= cand.sum()
    return cand


def _score_of(w: np.ndarray) -> float:
    """Convenient wrapper returning the evaluation score for a simplex‑normalised w."""
    n = w.size
    conv = convolve_fft(w, w)               # full convolution, length 2n-1
    return 2.0 * n * np.max(conv)

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) with a low
    `evaluate_sequence` score.  The routine respects the ~1000 s
    time budget, combines a parametric sweep with a smoothed‑max
    Adam optimiser and finishes by scaling the result to √(2·n).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length (reuse previous construction if possible)
    # --------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(seq0.size)
        if n < 2:
            raise ValueError
    except Exception:
        n = 6000                               # comfortable default size

    # --------------------------------------------------------------
    # Pre‑computed index vector for the parametric family
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64) + 0.5

    # --------------------------------------------------------------
    # Phase 1 – coarse grid search over a simple symmetric power‑law family
    # --------------------------------------------------------------
    best_w = None
    best_score = np.inf

    # coarse ranges
    p_vals = np.linspace(0.35, 0.80, 10)          # exponent
    a_vals = np.linspace(0.60, 1.00, 9)           # mixture weight of the power‑law part
    c_factors = [0.0, 0.001, 0.005, 0.01]         # regularising constant as fraction of (n/2)²

    print("[grid] starting coarse sweep")
    for p in p_vals:
        for a in a_vals:
            for cf in c_factors:
                # power‑law core (symmetric)
                c = cf * (n / 2.0) ** 2
                core = ((idx) * (n - idx) + c) ** (-p)
                w = a * core + (1.0 - a) / n        # optional uniform mix
                w /= w.sum()
                score = _score_of(w)
                if score < best_score:
                    best_score = score
                    best_w = w.copy()
                    print(f"[grid] p={p:.3f}, a={a:.3f}, cf={cf:.5f} → score={score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – finer sweep centred around the current best
    # --------------------------------------------------------------
    if best_w is not None:
        # locate the parameters that generated best_w (fallback to defaults)
        # we re‑derive them by a tiny optimisation, but for speed we just
        # explore a narrow neighbourhood.
        p_center = np.clip(p_vals[np.argmin(np.abs(p_vals - 0.5))], 0.1, 2.0)
        a_center = np.clip(a_vals[np.argmin(np.abs(a_vals - 0.8))], 0.2, 1.0)
    else:
        p_center = 0.5
        a_center = 0.9

    p_fine = np.linspace(max(p_center - 0.04, 0.1), p_center + 0.04, 21)
    a_fine = np.linspace(max(a_center - 0.06, 0.2), min(a_center + 0.06, 1.0), 21)
    c_fine = [0.0, 0.001, 0.003]

    print("[grid] starting fine sweep")
    for p in p_fine:
        for a in a_fine:
            for cf in c_fine:
                c = cf * (n / 2.0) ** 2
                core = ((idx) * (n - idx) + c) ** (-p)
                w = a * core + (1.0 - a) / n
                w /= w.sum()
                score = _score_of(w)
                if score < best_score:
                    best_score = score
                    best_w = w.copy()
                    print(f"[fine] p={p:.4f}, a={a:.4f}, cf={cf:.4f} → score={score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – smoothed‑max Adam descent starting from the best sweep point
    # --------------------------------------------------------------
    if best_w is None:                         # fallback – uniform distribution
        best_w = np.full(n, 1.0 / n)
        best_score = _score_of(best_w)

    w = best_w.copy()
    m = np.zeros_like(w)                      # Adam first moment
    v = np.zeros_like(w)                      # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05                                 # base learning rate
    it = 0
    no_improve = 0

    start_time = time.monotonic()
    max_time = max(0.0, float(budget_s) - 5.0)  # keep a few seconds for final scaling

    print("[opt] entering Adam descent")
    while time.monotonic() - start_time < max_time:
        # ----- full autocorrelation and soft‑max weights (smooth max) -----
        conv = convolve_fft(w, w)                     # length 2n‑1
        max_val = np.max(conv)

        # smoothing parameter grows slowly, capping at ~35
        beta_soft = min(35.0, 5.0 + it * 0.01)

        # numerically stable soft‑max around the active entries
        exp_vals = np.exp(beta_soft * (conv - max_val))
        probs = exp_vals / exp_vals.sum()             # shape (2n‑1,)

        # ----- gradient of the smoothed max (via a second convolution) -----
        # grad_i = 2 * Σ_j w_j * probs_{i+j}  (indices gestion performed by reversal)
        probs_rev = probs[::-1]
        grad_full = convolve_fft(w, probs_rev)        # length 3n‑2
        grad = 2.0 * grad_full[2 * n - 2 - np.arange(n)]

        # ----- Adam update with projection onto the simplex -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        w_temp = w - step
        w = simplex_projection(w_temp)                # enforce Σw = 1, w≥0

        # ---- occasional symmetry enforcement (helps the convolution) ----
        if it % 2000 == 0 and it > 0:
            w = 0.5 * (w + w[::-1])
            w /= w.sum()

        # ---- evaluate current solution ----
        cur_score = _score_of(w)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={it:6d} score={best_score:.6f}")
        else:
            no_improve += 1

        # ---- random block jump every few thousand iterations ----
        if it % 5000 == 0 and it > 0:
            cand = block_perturbation(best_w, rng)
            cand_score = _score_of(cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[block] iter={it:6d} new best={best_score:.6f}")

        # ---- stagnation reset (helps escape flat regions) ----
        if no_improve > 200_000:
            w = block_perturbation(best_w, rng)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={it:6d} resetting to perturbed best")

        it += 1

    # --------------------------------------------------------------
    # Final scaling – bring Σw to √(2·n) as required by `evaluate_sequence`
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()

    # The harness will also call `evaluate_sequence` itself; we report the value for logging.
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score_before_scaling={best_score:.6f}, "
          f"final_score={final_score:.6f}")

    # Return a plain Python list.
    return [float(x) for x in final_seq]
