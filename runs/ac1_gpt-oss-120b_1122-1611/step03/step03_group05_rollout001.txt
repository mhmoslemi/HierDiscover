def lp_optimize(w, M, active_idx, t_mix=0.2):
    n = w.size
    num_cons = len(active_idx)
    # Build sparse matrix
    rows = []
    cols = []
    data = []
    for r, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i_end inclusive, but we can vectorize
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        coeffs = w[k - i_vals]
        rows.extend([r] * len(i_vals))
        cols.extend(i_vals.tolist())
        data.extend(coeffs.tolist())
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(num_cons, n))
    b_ub = np.full(num_cons, M, dtype=np.float64)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    # Scale g to same sum as w (which = 1)
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    g_scaled = g / sum_g  # sum = 1
    # Combine
    new_w = (1 - t_mix) * w + t_mix * g_scaled
    # Ensure nonneg and sum=1 (should already hold)
    new_w = np.maximum(new_w, 0.0)
    new_w = new_w / np.sum(new_w)
    return new_w

"""
propose_candidate
-----------------

Hybrid optimizer for the self‑convolution peak problem.

* Phase 1 – stochastic Adam on a smooth log‑sum‑exp surrogate of the
  convolution peak (softmax weighting).  The temperature β is increased
  homotopically from a small value (≈2) to a large one (≈200) so that the
  optimisation first explores a smooth landscape and later concentrates
  on the true maximum.

* Phase 2 – active‑set linear programming.  Every few thousand Adam
  iterations we collect the indices where the current self‑convolution
  is close to its maximum, build a tiny LP
      maximise   ∑ g_i
      subject to (w * g)_k ≤ M  for the active k,
                 g_i ≥ 0,
  solve it with ``scipy.optimize.linprog`` (high‑s performance simplex),
  rescale the solution to keep the ℓ¹‑norm unchanged and mix it into the
  current iterate (convex combination with a small mixing factor).

* Random jumps – cheap block perturbations are tried periodically and
  accepted if they improve the best peak.

The algorithm works entirely with non‑negative vectors, projects onto
the probability simplex after each update and optionally enforces
symmetry (a ↔ a[::-1]).  After the time budget expires the best vector
found is scaled to the conventional √(2 n) ℓ¹‑norm; because the objective
is scale‑invariant the returned list can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sparse

# ----------------------------------------------------------------------
# Helper utilities (must stay at top‑level)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers divided by ``4**i``.
    Recurrence a_{i+1}=a_i·(2i+1)/(2i+2) (stable)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):  # numerical safety
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random multiplicative perturbation of a contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # ≈0.5 % of length
    hi = max(2, n // 12)           # up to ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ~±12 % multiplicative noise
    cand[start:stop] *= factor
    if rng.random() < 0.35:        # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution w*w via FFT of length N (full result)."""
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # clip tiny negatives
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate L_β(w) = (1/β)·log∑exp(β·(w*w)[k]).
    ``probs`` are the soft‑max weights (sum 1, length 2·len(w)−1).
    """
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = probs.size
    offset = m - 1
    grad = 2.0 * corr_full[offset : offset + w.size]
    return grad


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry v[i] = v[-i-1] by averaging."""
    return 0.5 * (v + v[::-1])

    best_max = np.max(self_convolve_fft(w, N_self))

    lr = 0.07
    beta1, beta2 = 0.9, 0.999

    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta_start = 2.0
    beta_end = 200.0

    max_active = 150            # number of top convolution indices used as constraints

    iteration = 0
    no_improve = 0
    start_time = time.monotonic()

        b = self_convolve_fft(w, N_self)
        cur_max = np.max(b)

                best_max = cand_max
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best max‑conv={best_max:.6e}")

        if iteration % lp_interval == 0:

            data = []
            for r, k in enumerate(idx_top):
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                # indices of g participating in (w * g)_k
                i_vals = np.arange(i_start, i_end + 1, dtype=int)
                coeffs = w[k - i_vals]               # w_{k-i}
                rows.extend([r] * len(i_vals))
                cols.extend(i_vals.tolist())
                data.extend(coeffs.tolist())

            A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
            b_ub = np.full(K, M, dtype=np.float64)

            c = -np.ones(n)            # maximise ∑g_i  ⇔ minimise -∑g_i
            bounds = [(0, None)] * n

            try:
                res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                  bounds=bounds, method='highs',
                                  options={'presolve': True, 'time_limit': 5.0})
                if res.success:
                    g = res.x
                    sum_g = np.sum(g)
                    if sum_g > 1e-12:          # avoid degenerate solution
                        g = g / sum_g          # normalise to ℓ¹‑norm 1
                        # convex combination with current w
                        w = (1.0 - mix_t) * w + mix_t * g
                        w = simplex_projection(w)
                        w = symmetrize(w)
                        w = simplex_projection(w)
                        print(f"[lp] iter={iteration:6d}, LP improved, max‑conv still {cur_max:.6e}")
                else:
                    # LP failed – simply ignore this round
                    pass
            except Exception:
                # In case linprog is unavailable or crashes
                pass

        # ----- stagnation handling -----
        if no_improve > 50000:
            # perturb current best to escape a flat region
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

    # --------------------------------------------------------------
    # Final scaling – the evaluation function is scale‑invariant, so we
    # return the best normalised vector (ℓ¹‑norm = 1) scaled to
    # √(2 n) as required by the convention used in the reference code.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * (target_sum / np.sum(best_w))

    # Optional sanity check using the provided evaluator
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best max‑conv (norm‑1)={best_max:.6e}, "
              f"after scaling evaluate={final_score:.6f}")
    except Exception:
        pass

    # Return a plain Python list
    return [float(x) for x in final_seq]
