Key ideas

def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Tiny negative values can appear due to rounding – clip them.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_swap_perturbation(base: np.ndarray,
                             rng: np.random.Generator,
                             max_frac: float = 0.2) -> np.ndarray:
    """
    Move a random amount of probability mass from one index to another.
    The transferred amount is at most `max_frac` of the donor weight.
    """
    cand = base.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    # Amount we are allowed to move from i (cannot become negative)
    allowable = min(cand[i], max_frac * cand[i])
    if allowable <= 0.0:
        return cand
    delta = float(rng.uniform(0.0, allowable))
    cand[i] -= delta
    cand[j] += delta
    return cand


def block_perturbation(base: np.ndarray,
                       rng: np.random.Generator) -> np.ndarray:
    """A cheap O(n) block‑scale perturbation (used for restarts)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

        mu = maxb * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        shifted = (b - maxb) / mu               # max(b) → 0, others ≤ 0
        exp_term = np.exp(shifted)              # safe, lies in [0,1]
        p = exp_term / np.sum(exp_term)         # soft‑max over convolution entries

        # ---------- gradient of the smooth‑max ----------
        # 2·conv(p, w_rev) and take the centre n entries
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # ---------- Adam update ----------
        iteration_f = iteration + 1.0
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration_f)
        v_hat = v / (1.0 - beta2 ** iteration_f)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        w_candidate = w - step
        w = simplex_projection(w_candidate)

        # decay the smoothness parameter
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        # ---------- occasional swap‑perturbation ----------
        if iteration % swap_interval == 0 and iteration > 0:
            cand = random_swap_perturbation(best_w, rng, max_frac=0.2)
            cand = cand / np.sum(cand)
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[swap] iter={iteration:6d}, new best={best_score:.6f}")

        # ---------- restart if stuck ----------
        if no_improve > reset_interval:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final sub‑gradient polishing (focus on the active peak).
    # --------------------------------------------------------------
    sub_iter = 0
    lr_sub = 0.5
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 0.5):
        b = convolve_fft(w, w)
        maxb = np.max(b)
        i_max = int(np.argmax(b))

        # sub‑gradient for the active index i_max
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        grad[j] = 2.0 * w[i_max - j]   # 2·w shifted

        step = lr_sub / np.sqrt(sub_iter + 1.0)
        w = simplex_projection(w - step * grad)

        # check improvement
        score = 2.0 * n * np.max(convolve_fft(w, w))
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[sub‑improve] sub_iter={sub_iter:6d}, score={best_score:.6f}")

        sub_iter += 1
        if sub_iter > 30000:
            break

    # --------------------------------------------------------------
    # Rescale to the conventional √(2 n) ℓ₁‑norm and return.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_iter={best_iter}, best_score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
