"""
propose_candidate
-----------------

A hybrid search that first sweeps a few low‑dimensional families of
step‑functions (power‑law, exponentially tilted Catalan and
shifted‑power‑law) to locate a good starting point, then refines the
candidate with a projected sub‑gradient (Adam) method that directly
optimises the true max‑convolution objective.  The algorithm works on
the probability simplex (∑a_i = 1) – the final sequence is rescaled to
the conventional √(2 n) ℓ¹‑norm expected by the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x≥0,∑x=1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                # numerical safety
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers divided by 4**i:
    a_i = C(2i,i)/4**i.  Recurrence a_{i+1}=a_i*(2i+1)/(2i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def convolution_max(a: np.ndarray, N_fft: int) -> float:
    """Return max of the linear self‑convolution a*a (full length)."""
    A = np.fft.rfft(a, N_fft)
    c = np.fft.irfft(A * A, N_fft)[: 2 * a.size - 1]
    # tiny negative values appear due to round‑off – clamp them
    np.maximum(c, 0.0, out=c)
    return float(c.max())


def objective_from_vector(a: np.ndarray, N_fft: int) -> float:
    """Scale‑invariant objective: 2·n·max(a*a)/(∑a)²."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    max_c = convolution_max(a, N_fft)
    n = a.shape[0]
    return 2.0 * n * max_c / (s * s)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block multiplication (used for occasional jumps)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)               # ≈0.5 %
    hi = max(2, n // 12)                # ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ≈±12 % multiplicative noise
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])           # occasional symmetrisation
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

                if obj < best_obj:
                    best_obj, best_vec = obj, a
                    print(f"[grid] δ={delta:.1f}, p={p:.3f}, λ={lam:.3f} → obj={obj:.6f}")
                if time.monotonic() - start_time > budget_s * 0.55:
                    break
            if time.monotonic() - start_time > budget_s * 0.55:
                break
        if time.monotonic() - start_time > budget_s * 0.55:
            break

    # ------------------------------------------------------------------
    # Sub‑gradient (Adam) refinement on the true max‑convolution objective
    # ------------------------------------------------------------------
    w = best_vec.copy()          # current iterate (already normalized)
    m = np.zeros_like(w)         # Adam first moment
    v = np.zeros_like(w)         # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_initial = 0.07
    iteration = 0
    no_improve = 0
    best_obj_val = best_obj
    best_seq = best_vec.copy()

    # Adaptive learning‑rate schedule (homotopy on β not needed here)
    while time.monotonic() - start_time < max(0.0, budget_s - 5.0):
        # ---- convolution and max index ----
        A = np.fft.rfft(w, N_fft)
        conv_full = np.fft.irfft(A * A, N_fft)[: 2 * n - 1]
        np.maximum(conv_full, 0.0, out=conv_full)
        max_c = float(conv_full.max())
        cur_obj = 2.0 * n * max_c / (w.sum() ** 2)   # w is already normalised → 2·n·max_c

        # ---- keep global best ----
        if cur_obj < best_obj_val - 1e-12:
            best_obj_val = cur_obj
            best_seq = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, obj={cur_obj:.6f}")
        else:
            no_improve += 1

        # ---- sub‑gradient of the active max ----
        k = int(np.argmax(conv_full))
        # indices that actually contribute to this convolution entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad = np.zeros_like(w)
        # gradient of c_k w.r.t w_i is 2·w_{k‑i}
        grad[idx] = 2.0 * w[k - idx]

        # ---- Adam update with projection onto simplex ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr_initial * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step
        w_new = np.maximum(w_new, 0.0)          # keep non‑negative
        w_new = simplex_projection(w_new)       # enforce ∑w=1

        w = w_new

        # ---- occasional random jump to escape stagnation ----
        if iteration % 4000 == 0:
            cand = block_perturbation(best_seq, rng)
            cand = cand / cand.sum()
            obj_cand = objective_from_vector(cand, N_fft)
            if obj_cand < best_obj_val:
                best_seq = cand.copy()
                best_obj_val = obj_cand
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[jump] iter={iteration:6d}, obj={obj_cand:.6f}")

        # ---- reset if no improvement for a long while ----
        if no_improve > 25000:
            w = block_perturbation(best_seq, rng)
            w = w / w.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

    # ------------------------------------------------------------------
    # Final scaling to √(2 n) ℓ¹‑norm (evaluation is scale‑invariant)
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_seq * target_sum / best_seq.sum()

    # (optional sanity check – cheap)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best obj≈{best_obj_val:.6f}, evaluate={final_score:.6f}")
    except Exception:
        pass

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
