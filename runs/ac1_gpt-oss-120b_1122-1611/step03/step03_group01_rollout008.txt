def compute_grad(w, alpha):

"""
Propose a candidate step‑function by minimizing the
∞‑norm of its self‑convolution.

The algorithm:
  * Start from several diverse initial distributions (Catalan,
    arcsine‑law, Dirichlet, previous best) and keep the best.
  * Optimise the smooth approximation of max_i (w∗w)[i] using a
    projected Adam update on the probability simplex.
    The smooth max is a soft‑max (log‑sum‑exp) with a temperature
    α that is slowly increased during the run.
  * Gradient of the smooth objective is obtained analytically via
    FFT‑based convolutions, i.e.  ∇ = 2·(p∗w_rev) where
    p_i = exp(α·conv_i)/∑exp(α·conv) and w_rev is w reversed.
  * Occasionally enforce symmetry, apply a large random block
    perturbation and restart when no improvement is seen.
  * In the last few seconds a simple simulated‑annealing polish
    is performed.
  * Finally the best distribution (sum = 1) is scaled to √(2·n) as
    required by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First `length` Catalan numbers a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def arcsine_sequence(length: int) -> np.ndarray:
    """Discrete approximation of the arcsine law 1/√(x·(1‑x))."""
    x = np.arange(length) + 0.5
    seq = 1.0 / np.sqrt(x * (length + 0.5 - x))
    return seq


def convolve_linear(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()               # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    c = np.fft.irfft(Fa * Fb, N)[:n]
    np.maximum(c, 0.0, out=c)                  # eliminate tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random block scaling used as a large jump."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                      # optional symmetry averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def compute_conv_and_gradient(w: np.ndarray, alpha: float):
    """
    Returns (conv, grad) where
        conv = w ∗ w   (linear convolution, length 2n‑1)
        grad = ∇  ( 1/α·log∑exp(α·conv) )
    The gradient is 2·(p ∗ w_rev) with p_i = softmax(α·conv_i).
    """
    n = w.size
    # 1) convolution w∗w
    conv = convolve_linear(w, w)               # length 2n-1
    # 2) soft‑max weights
    max_val = conv.max()
    exp_term = np.exp(alpha * (conv - max_val))
    p = exp_term / exp_term.sum()              # length 2n-1, non‑negative, sum=1
    # 3) gradient via correlation (implemented as convolution with the reversed w)
    w_rev = w[::-1]
    corr = convolve_linear(p, w_rev)           # length 3n-2
    grad = 2.0 * corr[n - 1:n - 1 + n]         # slice of length n
    return conv, grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is as small as possible.
    The routine respects the 1000 s time budget, uses a smoothed
    gradient‑descent optimiser with Adam, occasional symmetry
    enforcement, large‑block perturbations, and a final simulated‑annealing
    polish.  The returned list is scaled to √(2·n) as expected by the
    evaluator.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a length (use previous construction if available).
    # ------------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(seq0.size)
        if n < 2:
            raise ValueError
    except Exception:
        n = 6000                     # comfortable default

    # ------------------------------------------------------------------
    # Build a few diverse starting points and keep the best one.
    # ------------------------------------------------------------------
    candidates = []

    # 1. Catalan distribution (classical baseline)
    cat = catalan_sequence(n)
    candidates.append(cat / np.sum(cat))

    # 2. Approximate arcsine law
    arcs = arcsine_sequence(n)
    candidates.append(arcs / np.sum(arcs))

    # 3. Random Dirichlet (α = 0.5 gives a spiky start)
    rand = rng.gamma(0.5, 1.0, size=n)
    candidates.append(rand / np.sum(rand))

    # 4. Re‑use a previous construction if it matches the length
    prev = globals().get('height_sequence_1')
    if isinstance(prev, (list, np.ndarray)):
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.size == n:
                candidates.append(prev_arr / np.sum(prev_arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Pick the best among the initial candidates (using the exact score).
    # ------------------------------------------------------------------
    best_w = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        conv = convolve_linear(cand, cand)
        score = 2.0 * n * np.max(conv)          # identical to evaluate_sequence after scaling
        print(f"[init cand {idx}] score={score:.6f}")
        if score < best_score:
            best_score = score
            best_w = cand.copy()
    print(f"[init] chosen length={n}, initial best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    w = best_w.copy()               # current iterate (sum = 1)
    m = np.zeros_like(w)            # first moment (Adam)
    v = np.zeros_like(w)            # second moment (Adam)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                       # base learning rate
    iteration = 0
    no_improve = 0
    start_time = time.monotonic()
    max_time = max(0.0, float(budget_s) - 8.0)   # keep a safety margin

    # smoothing parameter α (soft‑max temperature)
    alpha = 2.0
    alpha_max = 150.0
    # increase α linearly; chosen so that after ~150 000 iterations we hit α_max
    alpha_inc = (alpha_max - alpha) / 150_000.0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < max_time:
        # compute convolution and gradient of the smooth max
        conv, grad = compute_conv_and_gradient(w, alpha)

        # Adam update (bias‑corrected)
        iteration += 1
        lr = lr0 / np.sqrt(iteration)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_temp = w - step
        w_new = simplex_projection(w_temp)

        # Occasionally try the mirror‑descent (KL) update and keep the better
        if rng.random() < 0.5:
            factor = np.exp(-lr * grad)
            w_mirror = w * factor
            w_mirror /= w_mirror.sum()
            conv_m = convolve_linear(w_mirror, w_mirror)
            score_m = 2.0 * n * np.max(conv_m)
            conv_e = convolve_linear(w_new, w_new)
            score_e = 2.0 * n * np.max(conv_e)
            if score_m < score_e:
                w_new = w_mirror

        # Periodic symmetry enforcement (helps the search)
        if iteration % 500 == 0:
            sym = 0.5 * (w_new + w_new[::-1])
            w_new = simplex_projection(sym)

        w = w_new

        # Exact score for the current iterate
        conv_now = convolve_linear(w, w)
        score_now = 2.0 * n * np.max(conv_now)

        if score_now < best_score - 1e-9:
            best_score = score_now
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # Large random block perturbation every few thousand iterations
        if iteration % 2000 == 0:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()
            cand_score = 2.0 * n * np.max(convolve_linear(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # Increase the smoothing parameter α gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha + alpha_inc)

        # Reset if stuck for a very long time
        if no_improve > 120_000:
            w = block_perturbation(best_w, rng)
            w /= w.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, resetting to perturbed best.")

    # ------------------------------------------------------------------
    # Simulated‑annealing polish (uses the remaining time)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.monotonic() - start_time)
    if remaining > 6.0:
        temperature = 0.02
        cur_w = best_w.copy()
        cur_score = best_score
        anneal_start = time.monotonic()
        while time.monotonic() - anneal_start < remaining - 2.0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_transfer = min(cur_w[i], 0.02)
            delta = rng.uniform(0.0, max_transfer)
            if delta <= 0.0:
                continue
            cand = cur_w.copy()
            cand[i] -= delta
            cand[j] += delta
            conv_c = convolve_linear(cand, cand)
            cand_score = 2.0 * n * np.max(conv_c)
            if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / temperature):
                cur_w = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_w = cur_w.copy()
                    best_score = cur_score
            temperature *= 0.9999
            if temperature < 1e-6:
                temperature = 0.001
        print(f"[anneal] polishing finished, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final scaling – the evaluation is invariant to scaling.
    # Return the conventional √(2·n) ℓ¹‑norm.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()

    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best score (smooth)={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
