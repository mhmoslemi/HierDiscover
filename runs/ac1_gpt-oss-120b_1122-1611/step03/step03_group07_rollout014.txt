"""
Propose a candidate step‑function for the “max‑convolution” evaluation.
The algorithm is a two‑phase mirror‑descent on the simplex:

* Phase 1 – a soft‑max smoothed gradient is used (large temperature) to
  move quickly while keeping the update stable.
* Phase 2 – the smoothing temperature is annealed to zero and the
  gradient becomes the exact sub‑gradient of the max‑convolution.
  This drives the current maximum (usually at k=0) down further.

All updates are performed by an exponentiated‑gradient step (mirror
descent w.r.t. the KL‑divergence) followed by renormalisation onto the
probability simplex.  Periodic symmetry enforcement and cheap random
block jumps help to escape tiny plateaux.  The whole search runs for the
available time budget (≈1000 s) and finally rescales the best probability
vector to the conventional √(2 n)‑ℓ¹‑norm required by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – required by the harness)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan‑type numbers a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        # recurrence a_{i} = a_{i-1} * (2i-1)/(2i)
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Fast linear convolution using real‑valued FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off
    np.maximum(c, 0.0, out=c)
    return c


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max probabilities p_k ∝ exp(conv_k / γ).  Returns a vector that sums to 1."""
    shift = conv - np.max(conv)           # avoid overflow
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:                      # fallback (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float):
    """
    Gradient of the soft‑max smoothed objective f_γ(w) = γ·log Σ_k exp((w⋆w)_k/γ).
    Returns (grad, max_b) where grad has shape w.size.
    """
    b = convolve_fft(w, w)                     # linear self‑convolution
    max_b = np.max(b)
    p = _softmax_weights(b, gamma)             # soft‑max over convolution entries
    p_rev = p[::-1]
    grad = 2.0 * convolve_fft(w, p_rev)        # length = 3n‑2
    # align and truncate to length n
    grad = grad[::-1][: w.size]
    return grad, max_b


def hard_gradient_max(w: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max_i (w⋆w)_i at the currently maximal index k_max.
    For any j with 0 ≤ k_max‑j < n the derivative is 2·w_{k_max‑j}, else 0.
    """
    n = w.size
    grad = np.zeros_like(w)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * w[k_max - idx]
    return grad


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation – used as a jump."""
    cand = base.copy()
    n = cand.size
    # block width is a modest fraction of n
    lo = max(1, n // 250)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps for the symmetric nature of the problem)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights whose `evaluate_sequence` score
    is as low as possible.  The routine runs a time‑bounded mirror‑descent
    optimisation on the simplex, starting from the known Catalan construction
    (or from a previously saved best sequence).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    time_limit = max(0.0, float(budget_s) - 4.0)   # keep a small safety margin

    # --------------------------------------------------------------
    # Choose a length.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
            w = prev / np.sum(prev)                     # normalise to a probability vector
        except Exception:
            n = 6000
            w = catalan_sequence(n)
            w = w / np.sum(w)
    else:
        n = 6000
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # keep the best probability vector seen so far
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)       # this equals evaluate_sequence after scaling
    best_w = w.copy()
    print(f"[init] n={n}, init_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    eta0 = 0.6               # initial stepsize for exponentiated‑gradient
    gamma_start = 0.25       # start temperature factor (γ = factor * max_b)
    gamma_end = 0.01         # final temperature factor
    alpha_start = 1.0        # start weight for soft‑gradient
    alpha_end = 0.0          # final weight (hard gradient only)
    transition_iters = 250000   # after this many iters the temperature is essentially zero
    sym_interval = 1500
    block_interval = 4000
    no_improve_limit = 30000
    max_exponent = 50.0
    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() - start_time < time_limit:
        # a) current convolution and its max
        b = convolve_fft(w, w)
        max_b = np.max(b)
        k_max = int(np.argmax(b))
        score = 2.0 * n * max_b

        # b) record improvement
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # c) schedule of temperature and soft/hard blend
        #    temperature γ = max(gamma_end * max_b, gamma_start * max_b * decay)
        decay = max(0.0, 1.0 - iteration / transition_iters)
        gamma_factor = gamma_end + decay * (gamma_start - gamma_end)
        gamma = max(gamma_factor * max_b, 1e-14)

        #    blend factor α: start with soft gradient, anneal to hard
        alpha = alpha_end + decay * (alpha_start - alpha_end)

        # d) compute gradients
        grad_soft, _ = _smoothed_gradient(w, gamma)
        grad_hard = hard_gradient_max(w, k_max)

        grad = alpha * grad_soft + (1.0 - alpha) * grad_hard

        # e) exponentiated‑gradient update
        eta = eta0 / np.sqrt(iteration + 1.0)
        exponent = -eta * grad
        np.clip(exponent, -max_exponent, max_exponent, out=exponent)
        w = w * np.exp(exponent)

        # f) renormalise onto the simplex
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # g) occasional symmetrisation (helps because the target problem is symmetric)
        if iteration % sym_interval == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])

        # h) occasional random block jump – keep it only if it improves the best known value
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new_best={best_score:.6f}")

        # i) stagnation handling – reset from a perturbed version of the best vector
        if no_improve > no_improve_limit:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluator is scale‑invariant, but we output the
    # conventional √(2 n) ℓ¹‑norm for readability.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum   # best_w already sums to 1

    # sanity check (silently ignored if evaluate_sequence is not available)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best_score={best_score:.6f}, after_scaling={final_score:.6f}")
    except Exception:
        print(f"[final] n={n}, best_score={best_score:.6f}")

    # Return a plain Python list
    return [float(x) for x in final_seq]
