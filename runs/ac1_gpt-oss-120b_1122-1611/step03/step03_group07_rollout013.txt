def propose_candidate(seed=0, budget_s=1100):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    best_score = np.inf
    best_seq = None
    # provide list of lengths based on previous best we have
    lengths = [4000, 6000, 8000, 10000] # maybe chosen adaptively
    for n in lengths:
        # run family search
        # 1. Catalan (maybe truncated) as baseline
        cat = catalan_sequence(n)
        cat = cat / cat.sum()
        score = eval_seq(cat, n) # compute 2n*max(conv)
        # update best
        if score < best_score:
            best_score = score
            best_seq = cat.copy()
        # 2. Power law families: a_i = (i+1)^(-alpha)
        for alpha in np.linspace(0.1, 2.0, 19):
            w = np.arange(1, n+1, dtype=np.float64) ** (-alpha)
            w /= w.sum()
            score = eval_seq(w, n)
            if score < best_score:
                best_score = score
                best_seq = w.copy()
        # 3. Dirichlet draws: maybe 200 draws per n
        for _ in range(200):
            # Sample Dirichlet with concentration 0.5 maybe
            conc = rng.uniform(0.1, 5.0)
            g = rng.gamma(conc, 1.0, size=n)
            w = g / g.sum()
            score = eval_seq(w, n)
            if score < best_score:
                best_score = score
                best_seq = w.copy()
        # Early exit if time low
        if time.monotonic() - start_time > budget_s - 5:
            break
    # Further local refinement: apply exponentiated gradient starting from best_seq for some iterations
    w = best_seq.copy()
    n = w.size
    iteration = 0
    while time.monotonic() - start_time < budget_s - 1:
        # compute gradient using smoothed approach with decreasing gamma
        b = convolve_fft(w, w)
        max_b = np.max(b)
        gamma = max(0.01 * max_b, 1e-12) # smaller gamma to focus
        grad, _ = _smoothed_gradient(w, gamma)
        eta = 0.1 / np.sqrt(iteration + 1)
        exponent = -eta * grad
        # clip exponent
        np.clip(exponent, -20, 20, out=exponent)
        w = w * np.exp(exponent)
        w /= w.sum()
        # evaluate
        score = 2 * n * np.max(convolve_fft(w, w))
        if score < best_score:
            best_score = score
            best_seq = w.copy()
            print(f"[local improvement] iter={iteration} score={best_score:.6f}")
        iteration += 1
        if iteration % 200 == 0:
            # occasional random perturbation
            if rng.random() < 0.2:
                w = block_perturbation(w, rng)
                w /= w.sum()
    # final scaling
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_seq * target_sum
    return final_seq.tolist()

        w = w * np.exp(exponent)
        w /= w.sum()

            if cand_score < best_score:

                w = cand.copy()

            else:

            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            no_improve = 0
            print(f"[reset] iter={iteration}")

        iteration += 1
    # final scaling
    final_seq = best_w * np.sqrt(2.0 * n)
    return final_seq.tolist()

"""
Propose a candidate non‑negative step function that minimises
`evaluate_sequence`.  

The algorithm proceeds in three stages:

1. **Coarse global search** –  Several parametric families
   (Catalan, power‑law, shifted power‑law, Beta‑pdf, exponential)
   are sampled for a collection of lengths (500 … 8000).  Each candidate
   is normalised to a probability vector (∑p = 1) and its score
   `2·n·max(p∗p)` is evaluated by an FFT‑based convolution.  The best
   candidate so far is stored.

2. **Local refinement** – Starting from the best probability vector,
   a smoothed‑max gradient (soft‑max with temperature γ) is computed and
   an exponentiated‑gradient update is performed.  The step‑size η decays
   like 1/√t and the temperature γ adapts to the current maximal
   convolution entry.  Every few thousand iterations a random block
   perturbation is tried and kept if it improves the objective.
   Prolonged stagnation triggers a restart from a perturbed copy of the
   current best.

3. **Final scaling** – The probability vector is scaled to the
   conventional √(2 n) L¹‑norm required by `evaluate_sequence` and
   returned as a plain Python list.

The whole routine respects the 1000 s time budget, prints occasional
progress messages and never returns a vector containing NaN,
inf, negative or >1000 entries, thus satisfying the security checks
in `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level as required by the harness.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` numbers a_i = C(2i,i)/4**i (Catalan‑like)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for n up to a few ×10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # eliminate tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative block perturbation (cheap O(n))."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 10 % of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetric averaging (helps the geometry)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max weights p ∝ exp(conv/γ).  Returns a distribution (sums to 1)."""
    shift = conv - np.max(conv)               # avoid overflow
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float):
    """
    Gradient of the smoothed max‑convolution:
        ∇f_γ(w) = 2 · w * (softmax‑weights of w∗w)  (reversed)
    Returns (gradient, max(b)) where b = w∗w.
    """
    n = w.size
    b = convolve_fft(w, w)                     # length 2n‑1
    p = _softmax_weights(b, gamma)             # same length
    p_rev = p[::-1]
    conv = convolve_fft(w, p_rev)              # length 3n‑2
    grad = 2.0 * conv[::-1][:n]                # first n entries, correctly aligned
    return grad, np.max(b)


def _score_of_prob(p: np.ndarray) -> float:
    """Score 2·n·max(p∗p) for a probability vector p (∑p = 1)."""
    n = p.size
    b = convolve_fft(p, p)
    return 2.0 * n * np.max(b)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights whose `evaluate_sequence`
    score is as low as possible.  The routine respects the 1000 s
    wall‑clock budget and uses a mixture of global parametric search
    and a smoothed‑gradient local optimiser.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    # --------------------------------------------------------------
    # Global coarse search over several lengths and families.
    # --------------------------------------------------------------
    best_score = np.inf
    best_w = None
    best_n = None

    # lengths to try – tiny ones are cheap and sometimes give a lower
    # finite‑size constant.
    lengths = [500, 1000, 2000, 4000, 6000, 8000]

    for n in lengths:
        # 1) Catalan (baseline)
        w0 = catalan_sequence(n)
        w0 = w0 / w0.sum()
        s = _score_of_prob(w0)
        if s < best_score:
            best_score, best_w, best_n = s, w0.copy(), n
            print(f"[init] n={n} Catalan  score={s:.6f}")

        # 2) Random Dirichlet draws (different concentration parameters)
        for _ in range(80):
            conc = rng.uniform(0.2, 5.0)
            g = rng.gamma(conc, 1.0, size=n)
            w = g / g.sum()
            s = _score_of_prob(w)
            if s < best_score:
                best_score, best_w, best_n = s, w.copy(), n
                print(f"[dir]   n={n} conc={conc:.2f}  score={s:.6f}")

        # 3) Power‑law families   w_i ∝ (i+δ)^(-α)
        for alpha in np.linspace(0.40, 0.60, 7):          # α around 0.5
            for delta_frac in (0.0, 0.001, 0.005, 0.01):
                delta = int(delta_frac * n)
                idx = np.arange(n, dtype=np.float64) + delta + 1.0
                w = idx ** (-alpha)
                w = w / w.sum()
                s = _score_of_prob(w)
                if s < best_score:
                    best_score, best_w, best_n = s, w.copy(), n
                    print(f"[pow]   n={n} α={alpha:.3f} δ={delta}  score={s:.6f}")

        # 4) Beta‑pdf shape  (symmetric a=b)  w_i ∝ x^{a-1}(1-x)^{b-1}
        for _ in range(70):
            a = rng.uniform(0.2, 2.0)
            b = rng.uniform(0.2, 2.0)
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
            # numerical safety
            w[np.isnan(w)] = 0.0
            w = w / w.sum()
            s = _score_of_prob(w)
            if s < best_score:
                best_score, best_w, best_n = s, w.copy(), n
                print(f"[beta]  n={n} a={a:.2f} b={b:.2f}  score={s:.6f}")

        # 5) Simple exponential decay  w_i ∝ exp(-λ·i)
        for lam in np.linspace(0.0005, 0.0200, 7):
            w = np.exp(-lam * np.arange(n, dtype=np.float64))
            w = w / w.sum()
            s = _score_of_prob(w)
            if s < best_score:
                best_score, best_w, best_n = s, w.copy(), n
                print(f"[exp]   n={n} λ={lam:.4f}  score={s:.6f}")

        # Early exit if we are close to the wall clock limit
        if time.monotonic() - start > budget_s - 12.0:
            break

    # --------------------------------------------------------------
    # If a previous solution is present, use it as a starting point.
    # --------------------------------------------------------------
    if best_w is None and "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            best_n = prev.size
            best_w = prev / prev.sum()
            best_score = _score_of_prob(best_w)
            print(f"[reuse] using global height_sequence_1, n={best_n}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Local smoothed‑gradient optimisation (exponentiated‑gradient).
    # --------------------------------------------------------------
    if best_w is None:
        # fallback – a modest Catalan sequence
        best_n = 3000
        best_w = catalan_sequence(best_n)
        best_w = best_w / best_w.sum()
        best_score = _score_of_prob(best_w)

    w = best_w.copy()
    n = best_n

    iteration = 0
    no_improve = 0

    # hyper‑parameters – chosen to be a bit more aggressive than the
    # previous baseline.
    eta0 = 0.7                # initial learning rate
    gamma_factor = 0.06       # γ = gamma_factor * max(b)
    max_exponent = 30.0       # clipping for the exponent in EG updates
    block_interval = 2500     # how often to try a random block jump
    stagnation_limit = 25000  # reset after this many non‑improving steps

    while time.monotonic() - start < budget_s - 4.0:
        # current objective
        max_b = np.max(convolve_fft(w, w))
        score = 2.0 * n * max_b

        # record improvement
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] it={iteration:6d}  n={n}  score={best_score:.6f}")
        else:
            no_improve += 1

        # ---- smoothed‑max gradient step ------------------------------------
        gamma = max(gamma_factor * max_b, 1e-12)
        grad, _ = _smoothed_gradient(w, gamma)

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(iteration + 1.0)
        exponent = -eta * grad
        np.clip(exponent, -max_exponent, max_exponent, out=exponent)
        w = w * np.exp(exponent)

        # renormalise onto the simplex
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # occasional symmetry enforcement (helps with the convolution geometry)
        if iteration % 1200 == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])

        # random block jump – keep it only if it leads to an improvement
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block]  it={iteration:6d}  new best {best_score:.6f}")

        # stagnation handling – restart from a perturbed best vector
        if no_improve > stagnation_limit:
            w = block_perturbation(best_w, rng)
            w /= w.sum()
            no_improve = 0
            print(f"[reset]  it={iteration:6d}  restart from perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the required √(2 n) L¹‑norm.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum   # best_w already sums to 1

    # sanity check (silent if evaluate_sequence is not available)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n} best score (raw)={best_score:.6f}  "
              f"evaluate={final_score:.6f}")
    except Exception:
        print(f"[final] n={n} best score (raw)={best_score:.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in final_seq]
