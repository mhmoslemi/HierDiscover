"""
propose_candidate
-----------------
Hybrid optimizer for the L∞‑self‑convolution problem.

* Phase 1 – LP refinement  
  Given a probability vector w (∑w=1) we compute M = max(w∗w).  
  We solve the linear programme  

      maximise   Σ g_i
      subject to  (w∗g)_k ≤ M    for k=0,…,2n‑2
                  g_i ≥ 0

  (the convolution is linear in g).  The solution g is rescaled to a
  probability vector and mixed with the current w:  

      w ← (1‑t)·w + t·g ,   t starts at 0.3 and is slowly decreased.

  The LP is built with a Toeplitz matrix; SciPy’s HiGHS solver handles
  the dense (but structured) system efficiently.

* Phase 2 – smooth‑max Adam (as in the previous baseline)  
  We fine‑tune the resulting distribution using a soft‑max approximation
  of the max‑operator and Adam with a simplex projection.  Periodic
  random block perturbations and an occasional symmetrisation escape
  local minima.

* The best distribution seen during the run is returned after scaling
  to the conventional √(2 n) L¹‑norm, i.e. the value fed to
  ``evaluate_sequence`` is exactly the maximal entry of the
  self‑convolution, which is the quantity we minimise.

The whole routine respects the 1000 s budget, works with any previous
construction stored in the global ``height_sequence_1`` variable and
requires only NumPy, SciPy and the provided ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (pure functions, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan numbers scaled by 4⁻ⁱ:
    a_i = C(2i,i) / 4**i  (recurrence a_{i+1}=a_i*(2i+1)/(2i+2))."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT.  Small negative artefacts are
    clipped to 0 to keep the result numerically valid."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()  # next power of 2
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1} (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump – multiply a random contiguous block by a
    log‑normal factor; optionally enforce a symmetric averaging."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Core optimisation primitives
# ----------------------------------------------------------------------


def _build_toeplitz_matrix(w: np.ndarray) -> tuple:
    """
    Build the data, row, column arrays defining the Toeplitz matrix
    A such that (w * g)[k] = Σ_j A[k,j] * g_j.

    The matrix is returned as three 1‑D arrays (data, rows, cols) that
    can be fed directly to scipy.sparse.csr_matrix.
    """
    n = w.size
    # rows = concatenation of ranges [j, j+n) for j=0..n-1
    rows = np.concatenate([np.arange(j, j + n, dtype=np.int64) for j in range(n)])
    cols = np.repeat(np.arange(n, dtype=np.int64), n)
    data = np.tile(w, n)  # each w_j repeated n times
    return data, rows, cols


def _solve_lp(w: np.ndarray, M: float, rows: np.ndarray, cols: np.ndarray) -> np.ndarray | None:
    """
    Solve the LP

        maximise   Σ g_i
        subject to  (w * g)_k ≤ M  (k = 0 … 2n‑2)
                    g_i ≥ 0

    The matrix (w * g) is represented by the pre‑computed rows/cols
    of the Toeplitz matrix; ``data`` is recomputed from the current w.
    Returns the optimal g as a NumPy array, or ``None`` if the solver
    fails.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    n = w.size
    # repeat the current w to obtain the actual data vector
    data = np.tile(w, n)
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)  # maximise Σg → minimise −Σg
    bounds = [(0.0, None) for _ in range(n)]

    # HiGHS (default) is very fast for dense problems.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30.0})

    if not res.success:
        return None
    g = res.x
    return g


def _score_w(w: np.ndarray, n: int) -> float:
    """Return the objective value 2 n·max(w∗w) for a probability vector w."""
    conv = convolve_fft(w, w)
    return float(2.0 * n * np.max(conv))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts
    to minimise ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    budget = float(budget_s)

    # --------------------------------------------------------------
    # Length selection – reuse a known construction if possible.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init_arr = np.asarray(height_sequence_1, dtype=np.float64)
            n = init_arr.size
            if n < 2:
                raise ValueError
        except Exception:
            n = 5000
    else:
        n = 5000

    print(f"[setup] chosen length n={n}")

    # --------------------------------------------------------------
    # Initialise a probability vector w (Σw = 1).
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            w = np.asarray(height_sequence_1, dtype=np.float64)
            w = np.maximum(w, 0.0)
            w = w / np.sum(w)
        except Exception:
            w = catalan_sequence(n)
    else:
        w = catalan_sequence(n)

    w = w / np.sum(w)   # safety normalisation

    # --------------------------------------------------------------
    # Pre‑compute the index layout of the convolution matrix – it only
    # depends on n, not on the actual values of w.
    # --------------------------------------------------------------
    data_dummy, rows, cols = _build_toeplitz_matrix(np.ones(n, dtype=np.float64))

    # --------------------------------------------------------------
    # Book‑keeping of the best solution seen so far.
    # --------------------------------------------------------------
    best_w = w.copy()
    best_score = _score_w(best_w, n)
    print(f"[init] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – LP driven refinement.
    # --------------------------------------------------------------
    lp_iters = 8                # a modest number – each LP is costly
    t_mix = 0.30                # initial mixing factor
    for it in range(lp_iters):
        # time‑guard
        if time.monotonic() - start_time > budget - 120.0:
            print("[lp] time budget exhausted, leaving LP phase")
            break

        conv = convolve_fft(w, w)
        M = np.max(conv)

        # Solve LP
        g0 = _solve_lp(w, M, rows, cols)
        if g0 is None:
            print("[lp] LP failed – aborting LP phase")
            break

        # Normalise the LP solution to a probability vector
        sum_g0 = np.sum(g0)
        if sum_g0 <= 1e-12:
            print("[lp] LP returned near‑zero mass – aborting")
            break
        g = g0 / sum_g0

        # Mix the new direction into w
        w = (1.0 - t_mix) * w + t_mix * g
        # Keep on simplex (mixing two simplex‑vectors already stays on it)
        w = np.maximum(w, 0.0)
        w = w / np.sum(w)

        # Optional symmetrisation every few iterations
        if (it + 1) % 2 == 0:
            w = 0.5 * (w + w[::-1])
            w = w / np.sum(w)

        cur_score = _score_w(w, n)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_w = w.copy()
            print(f"[lp‑improve] iter={it+1:02d}, score={best_score:.6f}")

        # Reduce mixing magnitude gradually
        t_mix = max(0.05, t_mix * 0.85)

    # --------------------------------------------------------------
    # Phase 2 – smooth‑max Adam optimiser (the successful baseline).
    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for this hybrid approach)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    alpha = 12.0          # soft‑max temperature; will be increased slowly
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    iteration = 0
    no_improve = 0

    while time.monotonic() - start_time < budget - 5.0:
        # ---- 1. current self‑convolution ---------------------------------
        conv = convolve_fft(w, w)

        # ---- 2. soft‑max over the convolution ----------------------------
        maxc = np.max(conv)
        exp_vals = np.exp(alpha * (conv - maxc))
        p = exp_vals / np.sum(exp_vals)           # soft‑max probabilities

        # ---- 3. gradient of the smoothed max‑objective -------------------
        # ∂/∂w Σ_i p_i (w∗w)_i = 2 * corr(p, w)
        w_rev = w[::-1]
        grad_full = convolve_fft(p, w_rev)        # length 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # ---- 4. Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / math.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w = simplex_projection(w_new)

        # ---- 5. occasional symmetry enforcement -------------------------
        if iteration % 700 == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

        # ---- 6. evaluation of the true objective every few steps -------
        if iteration % 350 == 0:
            cur_score = _score_w(w, n)
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_w = w.copy()
                no_improve = 0
                print(f"[adam‑improve] iter={iteration:06d}, score={best_score:.6f}")
            else:
                no_improve += 1

        # ---- 7. random block perturbation -------------------------------
        if iteration % 1100 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = _score_w(cand, n)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:06d}, new best={best_score:.6f}")

        # ---- 8. temperature schedule ------------------------------------
        if iteration % 2400 == 0 and iteration > 0:
            alpha = min(alpha * 1.08, 2500.0)

        # ---- 9. restart if stuck ----------------------------------------
        if no_improve > 25000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:06d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling to the conventional √(2 n) L¹‑norm.
    # --------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # Compute the official score (for logging only – the harness will do it as well)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")
    except Exception as exc:
        # In the unlikely event that evaluate_sequence is unavailable
        print(f"[final] n={n}, best_score={best_score:.6f}, evaluate failed: {exc}")

    # Return a plain Python list.
    return [float(x) for x in final_seq]
