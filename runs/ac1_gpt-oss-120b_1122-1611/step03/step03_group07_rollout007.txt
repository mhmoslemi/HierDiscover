"""
Propose a candidate step function for the “max‑convolution” problem.

Algorithm
---------
1. Build a small library of symmetric, non‑negative “shape” vectors
   (Catalan, arcsine, geometric, random Dirichlet).  All are normalised
   to sum = 1 (i.e. a probability vector).  The best of these seeds is
   used as a start point.

2. Perform a projected *exponentiated* sub‑gradient descent on the
   objective

        F(w) = max_k (w ∗ w)[k]               (∗ = linear convolution)

   The sub‑gradient for the active (maximal) indices `K` is

        g = 2·Σ_{k∈K} shift_k(w) ,

   which can be evaluated efficiently by a second FFT‑based convolution.
   The update `w ← w·exp(−η g)` keeps the iterates non‑negative; a
   Euclidean projection onto the simplex restores Σw = 1.

3. Periodically enforce symmetry (`w ← (w+w[::-1])/2`), apply a cheap
   random block perturbation, and reset to a perturbed best solution if
   no improvement is seen for many iterations.  This gives the method
   enough exploration to escape shallow local minima.

4. The loop runs until the allotted time budget (~1000 s) is exhausted.
   The best vector `best_w` (still a probability vector) is finally
   scaled to the conventional √(2 n) L¹‑norm required by
   `evaluate_sequence` and returned as a plain Python list.

All heavy work (convolutions and gradients) is performed with
`numpy.fft.rfft`, i.e. O(n log n) per iteration, so tens of thousands
of iterations comfortably fit into the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of size power‑2."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off artefacts
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def catalan_sequence(n: int) -> np.ndarray:
    """Catalan numbers normalised by 4**i: a_i = C(2i,i)/4**i  (i=0,…,n‑1)."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return seq


def arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine law: w_i ∝ 1/√[(i+½)(n‑½‑i)].  Symmetric, non‑negative."""
    i = np.arange(n, dtype=np.float64)
    # avoid exact zero in the denominator – the endpoints are already >0
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return w


def geometric_sequence(n: int, r: float) -> np.ndarray:
    """Symmetric geometric shape: w_i ∝ r**i + r**(n‑1‑i).  r∈(0,1)."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(r, i) + np.power(r, n - 1 - i)
    w = (w + w[::-1]) * 0.5   # enforce symmetry (numerical safety)
    return w


def block_perturbation(base: np.ndarray,
                       rng: np.random.Generator,
                       min_width_frac: float = 1 / 200,
                       max_width_frac: float = 1 / 12,
                       sigma: float = 0.12) -> np.ndarray:
    """
    Cheap O(n) random block perturbation.
    A random contiguous block is multiplied by exp(N(0,σ²)).
    """
    n = base.size
    lo = max(1, int(n * min_width_frac))
    hi = max(2, int(n * max_width_frac))
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand = base.copy()
    cand[start:stop] *= factor
    # optional symmetrisation (helps a lot for even‑length vectors)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """Return a non‑negative step function whose `evaluate_sequence` score
    is as low as possible.  The routine respects the 1000 s wall‑time
    limit and always returns a list of floats (default length ≈7 k)."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # Choose a problem size.
    # --------------------------------------------------------------
    # Re‑use the previous best construction if it exists; otherwise start
    # with a default length of 7200 (large enough for a low ratio, small
    # enough to keep each FFT cheap).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            w0 = prev / np.sum(prev)   # already a probability vector
            if n < 2:
                raise ValueError
        except Exception:
            n = 7200
            w0 = catalan_sequence(n)
            w0 = w0 / np.sum(w0)
    else:
        n = 7200
        w0 = catalan_sequence(n)
        w0 = w0 / np.sum(w0)

    # --------------------------------------------------------------
    # Assemble a small library of diverse initial shapes.
    # --------------------------------------------------------------
    candidate_pool = []

    # 1. the (possibly) previously discovered vector
    candidate_pool.append(w0)

    # 2. pure Catalan (always a solid baseline)
    cat = catalan_sequence(n)
    candidate_pool.append(cat / cat.sum())

    # 3. arcsine law (edge‑heavy)
    arcs = arcsine_sequence(n)
    candidate_pool.append(arcs / arcs.sum())

    # 4. geometric families with several decay rates
    for r in (0.99, 0.995, 0.999, 0.9995):
        geo = geometric_sequence(n, r)
        candidate_pool.append(geo / geo.sum())

    # 5. a few Dirichlet draws (low‑α gives spikier candidates,
    #    high‑α approaches uniform)
    for alpha in (0.3, 0.5, 0.8):
        for _ in range(2):
            cand = rng.dirichlet(np.full(n, alpha))
            candidate_pool.append(cand)

    # --------------------------------------------------------------
    # Pick the best among the initial pool (fast cheap evaluation).
    # --------------------------------------------------------------
    best_w = None
    best_score = np.inf
    for w in candidate_pool:
        conv = convolve_fft(w, w)
        sc = 2.0 * n * np.max(conv)          # w sums to 1 → score = 2·n·max(conv)
        if sc < best_score:
            best_score = sc
            best_w = w.copy()
    print(f"[init] length={n}, best initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop – exponentiated sub‑gradient descent.
    # --------------------------------------------------------------

    w = best_w.copy()           # current iterand (also a probability vector)
    iteration = 0
    no_improve = 0

    # Hyper‑parameters (empirically reasonable for the chosen size)
    eta0 = 0.8                 # initial step‑size for the EG update
    max_exponent = 45.0        # clip the exponent to avoid overflow
    sym_interval = 2000        # enforce symmetry every this many iters
    block_interval = 5000      # try a random block jump every this many iters
    reset_limit = 20000        # reset to a perturbed best after this many no‑improve steps

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # (a) evaluate current objective
        conv = convolve_fft(w, w)
        max_val = np.max(conv)
        score = 2.0 * n * max_val

        # (b) record any improvement
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:5d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # (c) sub‑gradient of the max‑convolution (hard max, not smoothed)
        # indices attaining the maximum (within machine precision)
        idx = np.where(conv >= max_val - 1e-15)[0]
        # weight vector for the gradient: equal weight on each active index
        p = np.zeros_like(conv)
        p[idx] = 1.0 / max(1, idx.size)

        # gradient = 2 * Σ_{k∈K} shift_k(w)  (computed via a second convolution)
        grad = 2.0 * convolve_fft(w, p[::-1])[::-1][:n]

        # (d) exponentiated‑gradient update (multiplicative, stays non‑negative)
        eta = eta0 / np.sqrt(iteration + 1.0)
        exponent = -eta * grad
        np.clip(exponent, -max_exponent, max_exponent, out=exponent)
        w = w * np.exp(exponent)

        # (e) renormalise onto the simplex Σw=1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # (f) occasional symmetry enforcement (helps the convolution structure)
        if iteration % sym_interval == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])

        # (g) occasional random block perturbation – accept if it improves
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:5d}, new best score={best_score:.6f}")

        # (h) long‑run stagnation handling – reset to a perturbed best solution
        if no_improve > reset_limit:
            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            no_improve = 0
            print(f"[reset] iter={iteration:5d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but the
    # convention is to output a sequence whose L¹‑norm equals √(2 n).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum   # best_w already sums to 1

    # ------------------------------------------------------------------
    # Optional sanity check (silent in the real test harness)
    # ------------------------------------------------------------------
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best score={best_score:.6f}, "
              f"after scaling evaluate={final_score:.6f}")
    except Exception:   # evaluate_sequence may be unavailable in a dry run
        print(f"[final] length={n}, best score={best_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(v) for v in final_seq]
