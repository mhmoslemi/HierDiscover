def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Gradient‑based search on the probability simplex.

We minimise the evaluation score

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

by fixing sum(a)=1 (the denominator becomes 1) and applying a
smooth approximation of the max‑norm:
    fβ(a) = (1/β)·log Σ_i exp(β·conv_i(a))

∂fβ/∂a_j = a_j + ( Σ_i w_i·a_{i‑j} ) / Σ_i w_i ,
where w_i = exp(β·conv_i) (soft‑max weights).

The algorithm repeatedly
* evaluates conv(a,a) with an FFT,
* builds the soft‑max weights,
* obtains the gradient by another FFT,
* performs an Adam‑style step on the simplex,
* occasionally applies random block perturbations,
* performs cheap pairwise mass‑transfer moves,
* restarts with jitter when no improvement is seen for a while.

All operations are O(n log n) and run comfortably within the 1000 s budget.
The best sequence seen is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors whose total length is ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT.
    The result length is ``len(x)+len(y)-1`` (truncated to that)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Returns a new array."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u > cssv / ind
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _project_onto_simplex(out)

# Main optimisation routine
# ----------------------------------------------------------------------

    while time.monotonic() < deadline:
        # ----- 6.1) Convolution, max and current score -----------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max          # sum(a) == 1
        # ----- 6.2) Record best ---------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- 6.3) Schedules -----------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear interpolation (clamped)
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ----- 6.4) Soft‑max weighting of the whole convolution ----------
        # shift to avoid overflow
        w = np.exp(beta * (conv - cur_max))
        sum_w = w.sum()
        # ----- 6.5) Gradient of the smooth max -------------------------
        a_rev = a[::-1]                     # reversed copy
        conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
        # extract the segment that corresponds to Σ_i w_i a_{i‑j}
        grad_part = conv_w_rev[n - 1 : n - 1 + n] / sum_w
        raw_grad = grad_part + a                       # ∂fβ/∂a

        # momentum smoothing of the raw gradient (helps stability)
        raw_grad = momentum * grad_prev + (1.0 - momentum) * raw_grad
        grad_prev = raw_grad

        # ----- 6.6) Adam update (additive) ----------------------------
        t += 1
        m = b1 * m + (1.0 - b1) * raw_grad
        v = b2 * v + (1.0 - b2) * (raw_grad ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                     # descent step
        a = _project_onto_simplex(a)     # stay on simplex

        # ----- 6.7) Occasional random block perturbation -------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- 6.8) Pair‑wise mass transfer (cheap local search) -------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 0.0:
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    a_candidate = _project_onto_simplex(a_candidate)
                    conv_candidate = _conv_self(a_candidate, fft_len_conv)
                    score_candidate = 2.0 * n * conv_candidate.max()
                    if score_candidate < cur_score:
                        a = a_candidate
                        # keep the cheap variables up‑to‑date
                        conv = conv_candidate
                        cur_max = conv_candidate.max()
                        cur_score = score_candidate

        # ----- 6.9) Soft restart when stalled -------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # small log‑normal jitter on every coordinate
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            # reset Adam statistics
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            grad_prev[:] = 0.0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best vector (clipping per evaluate_sequence's limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
