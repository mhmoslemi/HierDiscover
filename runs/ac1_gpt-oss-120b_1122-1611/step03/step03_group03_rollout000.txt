    for i in range(pop_size):
        idxs = rng.choice([j for j in range(pop_size) if j != i], size=3, replace=False)
        a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
        mutant = a + F * (b - c)

        # Crossover
        trial = np.empty_like(mutant)

            if rng.random() < CR or j == jrand:

                trial[j] = pop[i][j]

"""
Propose a candidate step‑function using a two‑phase meta‑heuristic:

1. **Global exploration** – a small population differential‑evolution
   (DE) optimiser on the probability simplex.  Each individual is a
   non‑negative vector of length *n* that sums to 1.  DE provides large
   jumps in the search space and is good at escaping the local minima
   that Adam often settles in.

2. **Local refinement** – an Adam optimiser with a soft‑max (smooth‑max)
   surrogate for the ∞‑norm, exactly as in the reference code but run
   only on the best DE individual.  This sharpens the solution
   obtained by DE.

The algorithm respects the security checks of ``evaluate_sequence``,
projects every candidate onto the simplex, and finally rescales the
best vector to the conventional √(2 n) ℓ¹‑norm before returning it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns a non‑negative array of length len(a)+len(b)-1.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)          # clip tiny negatives from rounding
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Simple O(n) random jump – multiply a random contiguous block by a
    log‑normal factor and optionally enforce symmetry.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that seeks to
    minimise ``evaluate_sequence``.  The search consists of a global
    differential‑evolution phase followed by a local Adam refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose the length of the sequence.
    # --------------------------------------------------------------
    # Use the length of a previously discovered construction if it
    # looks reasonable; otherwise fall back to a medium‑size vector.
    if "height_sequence_1" in globals():
        try:
            n0 = int(np.asarray(height_sequence_1).size)
            n = max(3000, min(8000, n0))
        except Exception:
            n = 6000
    else:
        n = 6000

    # --------------------------------------------------------------
    # Initialise a population on the simplex.
    # --------------------------------------------------------------
    pop_size = 18
    population = []
    scores = []

    # Seed one individual with the previous best if dimensions match.
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size == n:
                w0 = np.maximum(prev, 0.0)
                w0 = w0 / np.sum(w0)
                population.append(w0)
                conv0 = convolve_fft(w0, w0)
                scores.append(2.0 * n * np.max(conv0))
            else:
                raise ValueError
        except Exception:
            pass

    # Fill the remaining slots with random Dirichlet draws.
    while len(population) < pop_size:
        v = rng.random(n)                     # uniform Dirichlet(1,…,1)
        v = simplex_projection(v)
        conv = convolve_fft(v, v)
        score = 2.0 * n * np.max(conv)
        population.append(v)
        scores.append(score)

    # Identify the current best.
    best_idx = int(np.argmin(scores))
    best_w = population[best_idx].copy()
    best_score = scores[best_idx]
    print(f"[init] n={n}, pop_size={pop_size}, best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Differential‑Evolution loop – consumes roughly 70 % of the budget.
    # --------------------------------------------------------------
    F = 0.78                         # differential weight
    CR = 0.92                        # crossover probability
    start_time = time.monotonic()
    max_time = float(budget_s) - 5.0          # safety margin
    de_time_limit = max_time * 0.70

    iteration = 0
    while time.monotonic() - start_time < de_time_limit:
        new_population = []
        new_scores = []
        for i in range(pop_size):
            # --- mutation -------------------------------------------------
            idxs = rng.choice([j for j in range(pop_size) if j != i], size=3,
                              replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + F * (b - c)

            # --- crossover ------------------------------------------------
            trial = np.empty_like(mutant)
            jrand = rng.integers(n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial[mask] = mutant[mask]
            trial[~mask] = population[i][~mask]

            # --- projection onto simplex ----------------------------------
            trial = np.clip(trial, 0.0, 1000.0)
            trial = simplex_projection(trial)

            # occasional symmetry enforcement (helps the landscape)
            if rng.random() < 0.30:
                trial = 0.5 * (trial + trial[::-1])
                trial = simplex_projection(trial)

            # --- evaluate -------------------------------------------------
            conv = convolve_fft(trial, trial)
            trial_score = 2.0 * n * np.max(conv)

            # --- selection ------------------------------------------------
            if trial_score < scores[i]:
                new_population.append(trial)
                new_scores.append(trial_score)
                if trial_score < best_score - 1e-12:
                    best_score = trial_score
                    best_w = trial.copy()
                    print(f"[de‑improve] iter={iteration:5d}, i={i}, score={best_score:.6f}")
            else:
                new_population.append(population[i])
                new_scores.append(scores[i])

        population = new_population
        scores = new_scores
        iteration += 1

        # Periodic random perturbation of the global best (helps escape plateaus)
        if iteration % 1100 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            conv_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(conv_cand)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                print(f"[de‑block] iter={iteration:5d}, new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Local Adam refinement on the best DE individual.
    # --------------------------------------------------------------
    remaining_time = max_time - (time.monotonic() - start_time)
    # safety clamp
    remaining_time = max(0.0, remaining_time)

    # Adam parameters (mirroring the reference implementation)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    alpha = 12.0          # start temperature for the smooth‑max
    adam_start = time.monotonic()
    adam_iter = 0

    while time.monotonic() - adam_start < remaining_time:
        conv = convolve_fft(best_w, best_w)          # length 2n‑1
        maxc = np.max(conv)

        # soft‑max over convolution values
        exp_vals = np.exp(alpha * (conv - maxc))
        p = exp_vals / np.sum(exp_vals)              # probability distribution (sum=1)

        # gradient of the smooth objective: 2 * corr(p, w)
        grad_full = convolve_fft(p, best_w[::-1])    # length 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = adam_iter + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_new = best_w - step
        best_w = simplex_projection(w_new)

        # optional symmetry enforcement every few iterations
        if adam_iter % 600 == 0 and adam_iter > 0:
            best_w = 0.5 * (best_w + best_w[::-1])
            best_w = simplex_projection(best_w)

        # gradual increase of the temperature (sharpen the max)
        if adam_iter % 2000 == 0 and adam_iter > 0:
            alpha = min(alpha * 1.09, 2500.0)

        # true objective check (cheap, done sparsely)
        if adam_iter % 350 == 0:
            cur_score = 2.0 * n * np.max(convolve_fft(best_w, best_w))
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                print(f"[adam‑improve] iter={adam_iter:5d}, score={best_score:.6f}")

        adam_iter += 1

    # --------------------------------------------------------------
    # Rescale to the conventional √(2 n) ℓ¹‑norm and return.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # Debug print of the official score (the harness will recompute it as well)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score≈{best_score:.6f}, evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
