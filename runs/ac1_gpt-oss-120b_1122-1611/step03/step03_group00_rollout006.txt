"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a soft‑max approximation of the max‑norm and a mirror‑descent
(update on the probability simplex via exponentiated gradients).
The algorithm works on any initial vector – we start from the provided
``height_sequence_1`` if it exists, otherwise from a random Dirichlet
draw – and runs until the time budget expires, returning the best
vector seen.

Key ideas
~~~~~~~~~
* The objective is approximated by  fβ(a)= (1/β)·log∑ exp(β·conv_k(a)),
  whose gradient w.r.t. a is 2·corr(softmax(conv), a).  The full
  soft‑max (instead of a truncated top‑K) gives a more accurate
  descent direction.
* Exponentiated‑gradient (mirror‑descent on the simplex) automatically
  enforces non‑negativity and the Σa_i = 1 constraint.
* β (temperature) is ramped up linearly so the optimisation first
  smooths the landscape and then focuses sharply on the peak.
* η (stepsize) is slowly decreased to allow fine‑grained tuning.
* Occasional log‑normal block perturbations and soft restarts prevent
  premature stagnation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two length that can hold the full
    linear convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that
    minimises ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # leave a short safety margin (a few seconds) for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise the vector a
    # ------------------------------------------------------------------
    default_n = 4096  # a convenient power‑of‑two size
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full_like(cand, 1.0 / cand.size, dtype=np.float64)
                n = a.size
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    # compute initial score (fast estimate, not the full evaluate_sequence)
    conv = _conv_self(a, fft_len)
    cur_max = float(conv.max())
    cur_score_est = 2.0 * n * cur_max / (np.sum(a) ** 2)  # sum(a) == 1 → 2*n*max
    # exact score for the initial vector (used for the best record)
    best_score = float(cur_score_est)
    best = a.copy()
    last_improve = time.monotonic()
    start_time = last_improve

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 0.8   # initial mirror‑descent stepsize
    eta_end = 0.02    # final stepsize
    beta_start = 0.5  # start temperature (smooth max)
    beta_end = 500.0  # final temperature (sharp max)

    momentum = 0.9          # exponential smoothing of the gradient
    grad_prev = np.zeros_like(a)

    block_perturb_prob = 0.03
    restart_gap = 30.0      # seconds without improvement → restart
    print_interval = 30.0   # seconds between status prints

    next_print = start_time + print_interval

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ---- linear schedules for η and β ---------------------------------
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- compute convolution and fast score estimate ------------------
        conv = _conv_self(a, fft_len)
        max_conv = float(conv.max())
        cur_score_est = 2.0 * n * max_conv / (np.sum(a) ** 2)  # sum(a) == 1 → 2*n*max

        # ---- update best record using the *exact* evaluation ----------------
        if cur_score_est < best_score:
            # evaluate with the official function (takes a little longer)
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best (exact) score = {best_score:.6f}")

        # ---- periodic status print -----------------------------------------
        if now >= next_print:
            print(f"[{elapsed:6.2f}s] current est. score = {cur_score_est:.6f}  "
                  f"best = {best_score:.6f}")
            next_print = now + print_interval

        # ---- soft‑max weighting of *all* convolution entries ----------------
        # stability: shift by the maximal value before exponentiation
        shift = max_conv
        w_raw = np.exp(beta * (conv - shift))          # max weight = 1
        w_sum = w_raw.sum()
        if w_sum == 0.0:  # numerically unlikely, but guard
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ---- gradient of the soft‑max objective (efficient via correlation) --
        # g = 2 * corr(w, a) where corr takes the valid part (length n)
        grad = 2.0 * np.correlate(w, a, mode='valid')
        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- mirror‑descent update (exponentiated gradient) -----------------
        factor = np.exp(-eta * grad)
        a = a * factor
        # Numerical safety – clip tiny negatives that could appear due to
        # floating‑point round‑off, then renormalise.
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---- occasional random block perturbation --------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ---- soft restart when stuck ---------------------------------------
        if now - last_improve > restart_gap:
            # jitter the current best and continue from there
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
