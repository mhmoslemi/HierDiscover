def refine_max(w, n, max_iter=2000):
    best_w = w.copy()
    best_score = 2*n*np.max(convolve_fft(w,w))
    for it in range(max_iter):
        b = convolve_fft(w,w)
        max_b = np.max(b)
        score = 2*n*max_b
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
        # identify indices achieving max (within tol)
        idx = np.where(np.abs(b - max_b) <= 1e-12)[0]
        # create p vector: weight 1/len(idx) for each idx
        p = np.zeros_like(b)
        p[idx] = 1.0 / len(idx)
        # compute gradient = 2 * convolution(w, p_rev)[:n] reversed
        # we can compute as in smoothed but p is not softmax.
        p_rev = p[::-1]
        grad = 2 * convolve_fft(w, p_rev)[::-1][:n]
        # EG update
        eta = 0.5 / np.sqrt(it+1.0)
        exponent = -eta * grad
        np.clip(exponent, -30, 30, out=exponent)
        w = w * np.exp(exponent)
        w = w / w.sum()
        # symmetrize each step perhaps
        w = 0.5 * (w + w[::-1])
        w = w / w.sum()
    return best_w, best_score

"""
Propose a low‑score step function for the `evaluate_sequence` objective.

The algorithm mixes several ideas:

1. **Multiple deterministic starts** – Catalan, symmetric arcsine,
   sine‑square and the previous best sequence (if available).

2. **Soft‑max‐smoothed exponentiated gradient** – the same smooth
   max used in the reference code, but with a *temperature schedule*
   that slowly cools (γ factor multiplied by 0.9999 each iteration).
   After each update the vector is projected onto the probability
   simplex by simple renormalisation; a symmetry enforcement is applied
   on every iteration.

3. **Stochastic jumps** – occasional block‑perturbations of the best
   solution, and a hard reset after long periods without improvement.

4. **Final refinement** – after the main EG phase a pure‑max sub‑gradient
   (the exact gradient of the max‑convolution entry) is applied for a
   few thousand extra steps, together with a tiny smoothing kernel,
   to push the value a little further down.

The whole routine is time‑bounded (≈1000 s).  It tries a few different
lengths (the length of the previous solution, +2000 and +5000) and
keeps the best result over all runs.  The returned list is scaled to
the conventional √(2 n) L¹‑norm.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # clip tiny negatives caused by round‑off
    np.maximum(c, 0.0, out=c)
    return c


# ----------------------------------------------------------------------
# Helper sequences (deterministic seeds)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan‑like sequence a_i = Π_{j< i} (2j+1)/(2j+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def arcsine_sequence(length: int) -> np.ndarray:
    """Symmetric arcsine law: a_i ∝ 1/√((i+½)*(n‑½‑i))."""
    idx = np.arange(length, dtype=np.float64)
    a = 1.0 / np.sqrt((idx + 0.5) * (length - 0.5 - idx))
    return a


def sine_square_sequence(length: int) -> np.ndarray:
    """Sin² shape: a_i = sin²(π·(i+½)/n)."""
    idx = np.arange(length, dtype=np.float64)
    a = np.sin(np.pi * (idx + 0.5) / length) ** 2
    return a


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex Σx=1, x≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max utilities (soft‑max gradient)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max probabilities p_k ∝ exp(conv_k / γ)."""
    shift = conv - np.max(conv)                     # avoid overflow
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float):
    """
    ∇f_γ(w) where f_γ = γ·log Σ_k exp((w*w)_k / γ).
    Returns (gradient, max(b)).
    """
    n = w.size
    # self‑convolution
    b = convolve_fft(w, w)                     # length 2n‑1
    # soft‑max weights
    p = _softmax_weights(b, gamma)             # same length
    # convolution with reversed weight vector
    grad = 2.0 * convolve_fft(w, p[::-1])[::-1][:n]
    return grad, np.max(b)


# ----------------------------------------------------------------------
# Random block perturbation (cheap jump)
# ----------------------------------------------------------------------


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetry blend (probability 0.4)
    if rng.random() < 0.4:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Pure‑max sub‑gradient (used for final refinement)
# ----------------------------------------------------------------------


def max_subgradient(w: np.ndarray, n: int):
    """Exact sub‑gradient of max_k (w*w)_k."""
    b = convolve_fft(w, w)
    max_b = np.max(b)
    # indices where the max is attained (within a tiny tolerance)
    tol = 1e-12
    max_idx = np.where(np.abs(b - max_b) <= tol)[0]
    # form a weight vector p that puts uniform mass on the active indices
    p = np.zeros_like(b)
    if max_idx.size:
        p[max_idx] = 1.0 / max_idx.size
    # gradient = 2 * conv(w, rev(p)) limited to first n entries
    grad = 2.0 * convolve_fft(w, p[::-1])[::-1][:n]
    return grad, max_b


# ----------------------------------------------------------------------
# Scoring helper (vector already normalised to Σ=1)
# ----------------------------------------------------------------------


def score_of(w: np.ndarray) -> float:
    """Return the objective 2·n·max(b) for a probability vector w."""
    n = w.size
    b = convolve_fft(w, w)
    return 2.0 * n * np.max(b)

        seeds.append(uni)

        # If a previous solution of the same length exists, use it.
        if "height_sequence_1" in globals():
            try:
                prev = np.asarray(height_sequence_1, dtype=np.float64)
                if prev.size == n:
                    tmp = prev / prev.sum()
                    seeds.append(tmp)
            except Exception:
                pass

        # Pick the seed with the lowest initial score.
        best_local_w = None
        best_local_score = float("inf")
        for s in seeds:
            sc = score_of(s)
            if sc < best_local_score:
                best_local_score = sc
                best_local_w = s.copy()
        w = best_local_w
        best_w = w.copy()
        best_score = best_local_score

        print(f"[init] best seed score = {best_score:.6f}")

        # ------- Optimisation hyper‑parameters -------
        eta0 = 0.55                 # initial EG step‑size
        gamma_factor = 0.5         # temperature multiplier (γ = factor·max_b)
        gamma_decay = 0.9999       # cools the temperature slowly
        sym_interval = 1           # enforce symmetry every iteration
        block_interval = 1500      # try a random jump every block_interval steps
        max_exponent = 45.0        # clipping for exponent in EG
        no_improve_limit = 25000   # reset after this many stagnant steps
        iter_limit = 200000        # safety cap (won’t be reached under time limit)

        no_improve = 0
        iteration = 0

        while (time.monotonic() - length_start) < (length_budget - 0.5):
            # ----- evaluate current objective -----
            cur_score = score_of(w)

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_w = w.copy()
                no_improve = 0
                print(f"[improve] it={iteration:6d}, score={best_score:.6f}")
            else:
                no_improve += 1

            # ----- smoothed gradient (soft‑max) -----
            # temperature proportional to current max entry
            _, max_b = _smoothed_gradient(w, gamma_factor * max(np.max(convolve_fft(w, w)), 1e-12))
            gamma = max(gamma_factor * max_b, 1e-12)
            grad, _ = _smoothed_gradient(w, gamma)

            # ----- exponentiated‑gradient update -----
            eta = eta0 / np.sqrt(iteration + 1.0)
            exponent = -eta * grad
            np.clip(exponent, -max_exponent, max_exponent, out=exponent)
            w = w * np.exp(exponent)

            # ---- projection onto the simplex (renormalise) ----
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = np.full_like(w, 1.0 / n)
            else:
                w /= w_sum

            # ---- enforce symmetry every iteration (cheap) ----
            if iteration % sym_interval == 0:
                w = 0.5 * (w + w[::-1])
                w /= w.sum()

            # ---- occasional block perturbation (jump) ----
            if iteration % block_interval == 0 and iteration > 0:
                cand = block_perturbation(best_w, rng)
                cand = cand / cand.sum()
                cand_score = score_of(cand)
                if cand_score < best_score:
                    best_score = cand_score
                    best_w = cand.copy()
                    w = cand.copy()
                    print(f"[block] it={iteration:6d}, new best={best_score:.6f}")

            # ---- stagnation handling ----
            if no_improve > no_improve_limit:
                w = block_perturbation(best_w, rng)
                w = w / w.sum()
                no_improve = 0
                print(f"[reset] it={iteration:6d}, reset to perturbed best")

            # ---- cool the temperature a little ----
            gamma_factor *= gamma_decay

            iteration += 1
            if iteration >= iter_limit:
                break

        # --------------------------------------------------------------
        # Final pure‑max sub‑gradient refinement (few thousand steps)
        # --------------------------------------------------------------
        refine_iters = min(8000, max(2000, iteration // 5))
        w = best_w.copy()
        for it in range(refine_iters):
            # compute exact max‑gradient
            grad, max_b = max_subgradient(w, n)
            # EG update with a smaller step‑size
            eta = 0.35 / np.sqrt(it + 1.0)
            exponent = -eta * grad
            np.clip(exponent, -30.0, 30.0, out=exponent)
            w = w * np.exp(exponent)
            w = w / w.sum()
            # enforce symmetry each 10 steps
            if it % 10 == 0:
                w = 0.5 * (w + w[::-1])
                w = w / w.sum()
            # optional tiny smoothing kernel every 500 steps
            if it % 500 == 0 and it > 0:
                kern = np.array([0.25, 0.5, 0.25], dtype=np.float64)
                w = np.convolve(w, kern, mode="same")
                w = np.maximum(w, 0.0)
                w = w / w.sum()
        # evaluate final refinement score
        refined_score = score_of(w)
        if refined_score < best_score:
            best_score = refined_score
            best_w = w.copy()
            print(f"[refine] length={n}, refined score={best_score:.6f}")

        # --------------------------------------------------------------
        # Keep the globally best solution across all lengths.
        # --------------------------------------------------------------
        if best_score < best_score_global:
            best_score_global = best_score
            best_w_global = best_w.copy()
            best_n_global = n
            print(f"[global] new best overall length={n}, score={best_score_global:.6f}")

        # Safety check – stop early if time is exhausted.
        if (time.monotonic() - start_time) > (total_budget - 5.0):
            break

    # --------------------------------------------------------------
    # Scale to the conventional L¹‑norm √(2 n) and return.
    # --------------------------------------------------------------
    if best_w_global is None:
        # fallback – should never happen
        best_n_global = base_len
        best_w_global = np.full(best_n_global, 1.0 / best_n_global)

    target_sum = np.sqrt(2.0 * best_n_global)
    final_seq = best_w_global * target_sum

    # Sanity check (silent unless an exception occurs)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={best_n_global}, best score={best_score_global:.6f}, "
              f"evaluate={final_score:.6f}")
    except Exception:
        print(f"[final] length={best_n_global}, best score={best_score_global:.6f}")

    return [float(x) for x in final_seq]
