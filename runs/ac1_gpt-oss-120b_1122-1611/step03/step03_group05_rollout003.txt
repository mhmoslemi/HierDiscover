"""
Hybrid Adam + Sub‑gradient optimizer for the self‑convolution peak problem.

* Phase 1 – Adam on a soft‑max smooth approximation of max (a∗a).
  β (softness) is increased from 2 to 200, the learning rate is
  gradually decayed, and the iterate is kept symmetric (w←½(w+wᵀ))
  after each projection.  Random block jumps are injected every few
  thousand steps.

* Phase 2 – Pure sub‑gradient descent (exact max) with a diminishing
  step size  η_t = η₀/√t.  The direction is the gradient of the
  active convolution entry; after each step we project onto the
  probability simplex and enforce symmetry again.

* Local perturbations – a cheap “block perturbation’’ (multiplying a
  contiguous segment by a random factor) is tried periodically and
  kept if it improves the current best peak.

The routine starts from the previously discovered best sequence
(`height_sequence_1` if present) or from the Catalan distribution.
It returns a list of floats whose ℓ¹‑norm is √(2 n), which is exactly
the value that `evaluate_sequence` reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level as required)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers divided by 4**i:
    a_i = C(2i,i)/4**i.  Recurrence a_{i+1}=a_i*(2i+1)/(2i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random perturbation – used as a jump to escape local minima."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 % of length
    hi = max(2, n // 12)           # up to ~8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # about ±12 %
    cand[start:stop] *= factor
    if rng.random() < 0.35:        # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Return the next power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution w*w using an FFT of length N."""
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)  # discard numerical negatives
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth max‑approximation L_β at point w.
    ``probs`` are the soft‑max weights (sum 1, length 2·len(w)‑1).
    Returns a vector of length len(w).
    """
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = probs.size
    offset = m - 1
    grad = 2.0 * corr_full[offset : offset + w.size]
    return grad


# ----------------------------------------------------------------------
# Main function required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function with a small self‑convolution peak.
    Returns a list of floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size and an initial vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=np.float64)
            if init_seq.ndim != 1 or init_seq.size < 2:
                raise ValueError
            n = init_seq.size
            w0 = np.maximum(init_seq, 0.0)
            if np.sum(w0) < 1e-12:
                raise ValueError
            w0 = w0 / np.sum(w0)          # probability simplex
        except Exception:
            # fallback to Catalan
            n = 8000
            w0 = catalan_sequence(n)
            w0 = w0 / np.sum(w0)
    else:
        n = 8000
        w0 = catalan_sequence(n)
        w0 = w0 / np.sum(w0)

    # occasionally try a random Dirichlet start and keep the better one
    rand_dir = rng.dirichlet(np.ones(n))
    if np.max(self_convolve_fft(rand_dir, _next_pow2(2 * n - 1))) < \
       np.max(self_convolve_fft(w0, _next_pow2(2 * n - 1))):
        w0 = rand_dir

    # ensure symmetry from the very beginning – many optimal solutions are symmetric
    w0 = 0.5 * (w0 + w0[::-1])

    # --------------------------------------------------------------
    # Pre‑computed FFT lengths
    # --------------------------------------------------------------
    N_self = _next_pow2(2 * n - 1)          # for self‑convolution
    N_corr = _next_pow2(3 * n - 2)          # for smooth‑gradient correlation

    # --------------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------------
    w = w0.copy()
    best_w = w.copy()
    best_score = np.max(self_convolve_fft(w, N_self))

    print(f"[init] n={n}, initial max‑conv={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on smooth surrogate (β ramps from 2 → 200)
    # --------------------------------------------------------------
    lr = 0.07
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta_start = 2.0
    beta_end = 200.0
    beta_ramp_iters = 200_000          # after this β stays at beta_end

    start_time = time.monotonic()
    time_limit = max(0.0, float(budget_s) - 5.0)   # keep a 5 s safety margin
    iteration = 0
    no_improve = 0
    last_best_iter = 0

    # --------------------------------------------------------------
    # Main Adam loop (runs until about 60 % of the allocated time)
    # --------------------------------------------------------------
    while time.monotonic() - start_time < time_limit * 0.6:
        # 1. self‑convolution and current max
        b = self_convolve_fft(w, N_self)
        cur_max = np.max(b)

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_w = w.copy()
            no_improve = 0
            last_best_iter = iteration
            print(f"[improve] iter={iteration:7d}, max‑conv={best_score:.9f}")
        else:
            no_improve += 1

        # 2. current β (smoothness)
        if iteration < beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * (iteration / beta_ramp_iters)
        else:
            beta = beta_end

        # 3. soft‑max weights over convolution values
        shift = beta * (b - cur_max)          # ≤ 0
        exp_shift = np.exp(shift)            # the max entry gets weight 1
        probs = exp_shift / np.sum(exp_shift)

        # 4. gradient of the smooth surrogate
        grad = gradient_smooth(w, probs, N_corr)

        # 5. Adam update + projection onto the simplex
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step
        w = simplex_projection(w_new)

        # 6. enforce symmetry (helps to keep the optimum in the symmetric subspace)
        w = 0.5 * (w + w[::-1])

        # 7. occasional random block jump
        if iteration % 5_000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_max = np.max(self_convolve_fft(cand, N_self))
            if cand_max < best_score:
                best_w = cand.copy()
                best_score = cand_max
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best max‑conv={best_score:.9f}")

        # 8. reset if stagnated for a long time
        if no_improve > 30_000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

    # --------------------------------------------------------------
    # Phase 2 – Pure sub‑gradient polishing (exact max)
    # --------------------------------------------------------------
    sub_iter = 0
    eta0 = 0.12

    while time.monotonic() - start_time < time_limit:
        b = self_convolve_fft(best_w, N_self)
        cur_max = np.max(b)
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_w = best_w.copy()
            print(f"[sub‑improve] sub_iter={sub_iter:8d}, max‑conv={best_score:.9f}")

        # index (or one of the indices) attaining the maximum
        k_star = int(np.argmax(b))

        # sub‑gradient of the active quadratic term
        grad = np.zeros_like(best_w)
        if k_star < n:
            # the active shift lies completely inside the vector
            grad[: k_star + 1] = 2.0 * best_w[k_star :: -1]
        else:
            start = k_star - n + 1
            grad[start:] = 2.0 * best_w[start:][::-1]

        # diminishing step size
        eta = eta0 / np.sqrt(sub_iter + 1.0)

        w_tmp = best_w - eta * grad
        w_proj = simplex_projection(w_tmp)
        w_proj = 0.5 * (w_proj + w_proj[::-1])          # keep symmetry

        # acceptance test (only keep if we improve the max‑conv)
        new_max = np.max(self_convolve_fft(w_proj, N_self))
        if new_max < cur_max - 1e-13:
            best_w = w_proj
            best_score = new_max
        else:
            # occasional stochastic acceptance to escape flat plateaus
            if rng.random() < np.exp(-(new_max - cur_max) / 0.001):
                best_w = w_proj

        sub_iter += 1

        # cheap random block perturbation every few thousand sub‑steps
        if sub_iter % 4_000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_max = np.max(self_convolve_fft(cand, N_self))
            if cand_max < best_score:
                best_w = cand
                best_score = cand_max
                print(f"[sub‑block] sub_iter={sub_iter:8d}, new best max‑conv={best_score:.9f}")

    # --------------------------------------------------------------
    # Final scaling – we rescale the best probability vector to the
    # conventional √(2 n) ℓ¹‑norm.  The evaluation value then equals the
    # stored max‑convolution peak.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # optional sanity check (cheap)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best max‑conv={best_score:.9f}, evaluate={final_score:.9f}")
    except Exception:
        pass

    # Return a plain Python list as required.
    return [float(x) for x in final_seq]
