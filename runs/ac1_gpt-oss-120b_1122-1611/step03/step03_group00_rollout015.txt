"""
Propose a candidate step‑function sequence.

This implementation improves on the previous mirror‑descent code by
using the full soft‑max gradient (all convolution entries) instead of a
fixed‑size active set, and by computing the gradient via FFT‑based
convolution.  A simple exponential‑gradient (mirror‑descent) update on
the probability simplex drives the self‑convolution towards a flat
profile, which provably lowers the score
    2·n·max(conv(a,a)) / (sum(a)**2)
towards its theoretical limit of 1.
The routine respects the 1000 s time budget, periodically injects tiny
random perturbations and performs soft restarts when progress stalls.
The best sequence seen is returned as a plain list of floats.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper functions – top level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(size: int) -> int:
    """
    Smallest power‑of‑two length that can hold a linear convolution
    of total size ``size``.
    """
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``.
    """
    fa = rfft(a, n=fft_len)
    fc = fa * fa
    conv = irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_linear(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two vectors using FFT.
    Returned length is ``len(x)+len(y)-1``.
    """
    fx = rfft(x, n=fft_len)
    fy = rfft(y, n=fft_len)
    fc = fx * fy
    conv = irfft(fc, n=fft_len)[: x.size + y.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise

        score = 2 * n * max(conv(a, a)) / (sum(a)**2)

    The optimisation runs until the time budget expires and returns the
    best sequence found.  The algorithm uses a soft‑max approximation of
    the ∞‑norm together with exponentiated‑gradient (mirror descent) updates
    computed via FFT for speed.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose the problem size.
    # ------------------------------------------------------------------
    default_n = 1024                     # a size that is fast enough for many FFTs
    if "height_sequence_1" in globals():
        try:
            hs = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if hs.ndim == 1 and hs.size > 0 and np.isfinite(hs).all():
                n = hs.size
            else:
                n = default_n
        except Exception:
            n = default_n
    else:
        n = default_n

    # ------------------------------------------------------------------
    # 2) Initialise a random Dirichlet vector (sum = 1)
    # ------------------------------------------------------------------
    a = rng.dirichlet(alpha=np.full(n, 0.5))          # slightly sparse start
    a = np.clip(a, 0.0, None)
    a /= a.sum()                                      # normalise to 1

    # Pre‑compute FFT lengths to avoid repeated work
    fft_len_self = _fft_len_for_n(2 * n - 1)          # for a*a
    fft_len_grad = _fft_len_for_n(3 * n - 2)          # for w * rev(a)

    # Book‑keeping for the best solution seen so far
    best_a = a.copy()
    best_score = float("inf")                         # will be updated first iteration

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (linear schedules)
    # ------------------------------------------------------------------
    eta_start = 1.0
    eta_end = 0.001
    beta_start = 0.2
    beta_end = 800.0
    block_perturb_prob = 0.02
    restart_gap = 30.0          # seconds without improvement → soft restart
    momentum = 0.85
    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution & score (internal cheap estimate) -----
        conv = _conv_self(a, fft_len_self)                     # length 2n‑1
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)       # sum(a) is 1 → 2·n·max

        # ----- keep the best solution ---------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule of learning‑rate η and temperature β ------------
        now = time.monotonic()
        elapsed = now - start_time
        progress = min(1.0, elapsed / (deadline - start_time))
        eta = eta_start * (1.0 - progress) + eta_end * progress
        beta = beta_start + (beta_end - beta_start) * progress

        # ----- soft‑max weights for *all* convolution entries ------------
        #   w_k ∝ exp(β·(c_k – max_c))   (non‑negative, sum to 1)
        max_conv = conv.max()
        # numerical stability: shift by max_conv (negative or zero)
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:                     # extremely high β may underflow
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth max (via correlation) ------------
        rev_a = a[::-1]
        # convolution of w with reversed a gives the needed correlation
        corr = _conv_linear(w, rev_a, fft_len_grad)          # length 3n‑2
        grad = 2.0 * corr[n - 1 : n - 1 + n]                # slice length n

        # momentum smoothing (helps stability)
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror‑descent) update ----------
        # Clip the exponent to avoid overflow/underflow
        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -700, 700)                # np.exp limits
        factor = np.exp(exp_arg)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation --------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- soft restart if stuck ----------------------------------
        if now - last_improve > restart_gap:
            a = best_a.copy()
            # Small log‑normal jitter on every coordinate
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
