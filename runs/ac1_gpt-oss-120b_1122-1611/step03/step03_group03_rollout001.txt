"""
Propose a candidate step‑function sequence that (approximately) minimizes
``evaluate_sequence``.

Algorithm Overview
------------------
1. **Probability representation** – The objective is scale‑invariant, so we
   work with ``w`` such that ``w_i ≥ 0`` and ``sum(w)=1``.  The true score
   for a given ``w`` (and length ``n``) is

        score(w) = 2 * n * max_k (w * w)[k]               (1)

   where ``*`` denotes linear convolution.  This can be computed quickly
   via an FFT‐based convolution.

2. **Initialisation** – If a previous construction ``height_sequence_1``
   is present we reuse it; otherwise we start from the Catalan sequence,
   which already gives a good shape.

3. **Parametric sweep** – We generate several families of symmetric
   distributions whose shape is controlled by a small number of parameters
   and keep the best one:
   * Beta(α,α) density (U‑shaped for α<1, hill‑shaped for α>1).
   * A simple power‑law “U” shape: ``w_i ∝ (i+1)^{-p} + (n-i)^{-p}``.

   For each parameter set we normalise to a probability vector and evaluate
   (1).  This sweep is cheap (a few hundred FFTs) and provides a strong
   starting point.

4. **Local stochastic refinement** – Starting from the best vector found
   above we run a simulated‑annealing style loop:
   * Randomly pick a contiguous block and multiply it by a log‑normal factor.
   * Optionally enforce a tiny amount of symmetry.
   * Re‑normalise onto the simplex.
   * Accept the new vector if it improves the score, otherwise accept
     with probability ``exp(-(Δscore)/T)`` where the temperature ``T`` is
     slowly decreased.

   The loop runs until the time budget is exhausted (≈ 1000 s).  All
   operations are O(n log n) and the inner loop evaluates at most a few
   thousand candidates per second.

5. **Final scaling** – The probability vector is scaled to the conventional
   L¹‑norm ``√(2 n)`` (the objective is invariant to this scaling) and
   returned as a plain ``list[float]``.  A final call to ``evaluate_sequence``
   is printed for verification.

The whole procedure respects the time budget, uses only NumPy, and keeps
the search space low‑dimensional enough to explore many candidates while
still being able to escape local minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan numbers scaled by 4⁻ⁱ:
    a_i = C(2i,i) / 4ⁱ ."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (non‑circular) via real FFT.  Small negative
    rounding errors are clipped to zero."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project ``v`` onto the probability simplex {x ≥ 0, Σx = 1}
    using the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_symmetric(alpha: float, n: int) -> np.ndarray:
    """Return a discretised symmetric Beta(α,α) density on ``n`` points,
    normalised to sum 1."""
    # mid‑point rule for integration
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # compute log‑pdf to avoid under/overflow
    logpdf = (alpha - 1.0) * (np.log(x) + np.log1p(-x))
    pdf = np.exp(logpdf)
    pdf_sum = np.sum(pdf)
    if pdf_sum == 0.0:
        pdf = np.full(n, 1.0 / n)
    else:
        pdf = pdf / pdf_sum
    return pdf


def powerlaw_u_shape(p: float, n: int) -> np.ndarray:
    """U‑shaped power‑law:  w_i ∝ (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    left = (i) ** (-p)
    right = (n + 1 - i) ** (-p)
    w = left + right
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w_sum
    return w


def score_of(w: np.ndarray, n: int) -> float:
    """Return the objective value (1) for a probability vector ``w``."""
    conv = convolve_fft(w, w)               # length 2n‑1
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The method consists of a parametric sweep followed
    by a simple simulated‑annealing style refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a reasonable length.  6000 works well and matches the
    # previous best construction; we keep it fixed for simplicity.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Initialise the probability vector ``w``.
    # ------------------------------------------------------------------
    best_w = None
    best_score = np.inf

    # Try to reuse a previously discovered construction.
    if 'height_sequence_1' in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.size == n:
                w0 = np.maximum(arr, 0.0)
                w0 = w0 / np.sum(w0)
                sc = score_of(w0, n)
                best_w, best_score = w0.copy(), sc
                print(f"[init] reused previous construction, score={sc:.6f}")
        except Exception:
            pass

    # Fall back to the Catalan sequence.
    if best_w is None:
        w0 = catalan_sequence(n)
        w0 = np.maximum(w0, 0.0)
        w0 = w0 / np.sum(w0)
        best_w = w0.copy()
        best_score = score_of(best_w, n)
        print(f"[init] Catalan baseline, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Parametric sweep over a few low‑dimensional families
    # ------------------------------------------------------------------
    start = time.monotonic()
    # Alpha sweep for the symmetric Beta distribution.
    alphas = np.concatenate((
        np.linspace(0.15, 0.45, 7),
        np.linspace(0.5, 1.5, 11),
        np.linspace(1.6, 3.0, 8),
    ))
    gammas = np.linspace(0.0, 1.0, 6)               # mixture weight with uniform
    uniform = np.full(n, 1.0 / n)

    for alpha in alphas:
        if time.monotonic() - start > budget_s - 20:
            break
        beta_w = beta_symmetric(alpha, n)
        for gamma in gammas:
            # Mixture with the uniform distribution (helps with smoothing)
            w = (1.0 - gamma) * beta_w + gamma * uniform
            w = np.clip(w, 0.0, None)
            w = w / np.sum(w)
            sc = score_of(w, n)
            if sc < best_score:
                best_score, best_w = sc, w.copy()
                print(f"[sweep] Beta α={alpha:.3f}, γ={gamma:.2f} → score={sc:.6f}")

    # Power‑law U‑shapes.

    for p in exponents:
        if time.monotonic() - start > budget_s - 20:
            break
        w = powerlaw_u_shape(p, n)
        sc = score_of(w, n)
        if sc < best_score:
            best_score, best_w = sc, w.copy()
            print(f"[sweep] Power‑law p={p:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Local stochastic refinement (simulated annealing)
    # ------------------------------------------------------------------
    it = 0

    T0 = 0.02

        cand[start_idx:start_idx + width] *= factor

        if rng.random() < 0.10:
            cand = 0.5 * (cand + cand[::-1])

        # projection back onto the simplex
        cand = np.clip(cand, 0.0, None)
        cand_sum = np.sum(cand)
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        # evaluate
        sc = score_of(cand, n)

        # acceptance test
        if sc < best_score:
            best_score, best_w = sc, cand.copy()
            print(f"[accept] iter={it:6d} improved score={sc:.6f}")
        else:

            elapsed = time.monotonic() - start
            T = T0 * (1.0 - elapsed / budget_s)
            if T > 0 and rng.random() < np.exp(-(sc - best_score) / T):

                best_score = sc
                print(f"[accept] iter={it:6d} accepted worsening score={sc:.6f} (T={T:.5f})")
        it += 1

    # 3. Final scaling to the conventional L¹‑norm √(2 n)
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # Verify the official score (the harness will also compute it)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score≈{best_score:.6f}, evaluate={final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
