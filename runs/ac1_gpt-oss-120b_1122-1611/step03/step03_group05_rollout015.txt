def max_convolution_and_subgradient(w, N, n):
    # Compute self-convolution via FFT
    wf = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(wf * wf, N)[:2*n-1]
    cur_max = np.max(conv_full)
    # Determine active indices - those within tolerance of max
    # Use absolute tolerance based on max magnitude and machine epsilon
    tol = 1e-12
    active = conv_full >= cur_max - tol
    # If no active (should not happen), set one index
    if not np.any(active):

        active[np.argmax(conv_full)] = True
    # Build weight vector v, uniform weight over active set
    v = np.zeros_like(conv_full, dtype=np.float64)
    cnt = np.count_nonzero(active)

propose_candidate
-----------------

A two‑stage optimizer for the self‑convolution peak problem.

Stage 1 – cheap exploration of simple parametric families
    * power‑law a_i ∝ (i+1)^{-p}
    * exponential a_i ∝ r^{i}
    * linear decreasing a_i = n‑i
    * the classic Catalan sequence
    * a few random Dirichlet draws
  The best member (lowest max‑convolution) is kept as a warm‑start.

Stage 2 – projected sub‑gradient descent on the exact L∞ objective.
    * For a current weight vector w (∑w=1, w≥0) we compute the full
      self‑convolution via a single FFT.
    * The active set of indices attaining the maximal convolution value
      yields a sub‑gradient:   g = 2·(v * rev(w))[n‑1:n‑1+n],
      where v is the uniform weight on the active positions.
    * An Adam‑style update is applied and the iterate is projected back
      onto the probability simplex.
    * Every few thousand iterations we try a cheap random block
      perturbation of the best solution seen so far; long stagnation
      triggers a reset to a perturbed copy of the best solution.

The routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence (scaled to the customary
‖·‖₁ = √(2 n) norm) as a plain Python list.

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, ∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:  # all entries equal (or negative) → uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a random factor (≈±12 %)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 %
    hi = max(2, n // 12)           # ~8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈±12 %
    cand[start:stop] *= factor
    if rng.random() < 0.35:        # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan numbers divided by 4**i
    (a_i = C(2i,i)/4**i).  Recurrence a_{i+1}=a_i·(2i+1)/(2i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = np.power(i, -p)
    seq /= np.sum(seq)
    return seq


def generate_exponential(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.power(r, i)
    seq /= np.sum(seq)
    return seq


def generate_linear_decreasing(n: int) -> np.ndarray:
    seq = np.arange(n, 0, -1, dtype=np.float64)
    seq = seq.astype(np.float64)
    seq /= np.sum(seq)

def max_conv_and_subgrad(w: np.ndarray, N: int, n: int):
    """Compute max self‑convolution value and a sub‑gradient w.r.t. w.
    Returns (max_value, subgradient, full_convolution)."""
    # self‑convolution via FFT (full length 2n‑1)
    wf = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(wf * wf, N)[: 2 * n - 1]
    cur_max = np.max(conv_full)

    # active set (indices attaining the maximum up to tiny tolerance)
    eps = 1e-12
    active = conv_full >= cur_max - eps
    if not np.any(active):
        active = np.zeros_like(conv_full, dtype=bool)
        active[np.argmax(conv_full)] = True
    cnt = np.count_nonzero(active)
    v = np.zeros_like(conv_full, dtype=np.float64)

    w_rev = w[::-1]

    corr_full = np.fft.irfft(vf * wrf, N)
    subgrad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return cur_max, subgrad, conv_full

            best_score = score
            best_w = cand.copy()
            print(f"[sweep] new best (idx={idx}) score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient (Adam) optimisation
    # --------------------------------------------------------------
    w = best_w.copy()                     # start from the best sweep result
    m = np.zeros_like(w)                  # Adam first moment
    v = np.zeros_like(w)                  # Adam second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07                             # base learning rate

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0
    last_best_iter = 0

    # main optimisation loop – stops a few seconds before the hard deadline
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        cur_max, grad, _ = max_conv_and_subgrad(w, N, n)
        cur_score = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            last_best_iter = iteration
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # Adam update (projected onto simplex)
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step
        w = simplex_projection(w_new)

        # occasional random jump (cheap block perturbation)
        if iteration % 5000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_max, _, _ = max_conv_and_subgrad(cand, N, n)
            cand_score = 2.0 * n * cand_max
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best score={best_score:.6f}")

        # reset if stagnating for a long time
        if no_improve > 25000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

    # --------------------------------------------------------------
    # Final scaling – convert the best probability vector into the
    # conventional √(2 n) ℓ¹‑norm.  The evaluate_sequence value then equals
    # the stored max‑convolution value (i.e. the score).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum(best_w) is 1, guard against rounding

    # Optional sanity check with the evaluator (cheap)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best score={best_score:.6f}, after scaling evaluate={final_score:.6f}")
    except Exception:
        pass

    # Return a plain Python list as required.
    return [float(x) for x in final_seq]
