"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. Initialise a non‑negative probability vector `w` on the simplex.
   * If a previous candidate (`height_sequence_1`) exists, reuse it.
   * Otherwise start from the classic decreasing central‑binomial
     distribution (a good known baseline).

2. Perform Adam‑optimised projected gradient descent on a smooth
   approximation of the ∞‑norm of the self‑convolution.
   * The smooth max is realised via a soft‑max with a temperature
     `μ = μ_factor * max(b)`.
   * Gradients are obtained analytically: `grad = 2·conv(p, rev(w))`.
   * After each Adam step the iterate is projected onto the simplex
     and forced to be symmetric (`w = (w + w[::-1]) / 2`).

3. Every `block_interval` iterations a cheap random block perturbation
   of the best solution so far is tried; if it improves the score the
   current iterate is reset to this candidate.

4. If no improvement is seen for `reset_interval` iterations the
   current iterate is replaced by a perturbed copy of the best
   solution (escape from stagnation).

5. With the remaining time run a multiplicative hill‑climbing phase
   (simulated‑annealing style) that perturbs the current best with
   log‑normal factors, keeps symmetry, and accepts occasional uphill
   moves.

6. Finally a short sub‑gradient refinement focuses on the active peak
   of the convolution.

7. The best vector is scaled to the conventional ℓ₁‑norm
   `sqrt(2·n)` and returned as a plain Python list.

All steps respect the constraints enforced by `evaluate_sequence`
(non‑negative entries, sum > 0).  The code is self‑contained and uses
only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation, used as a jump operator."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def multiplicative_perturb(w: np.ndarray, sigma: float,
                          rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation followed by simplex projection."""
    factor = np.exp(rng.normal(0.0, sigma, size=w.shape))
    cand = w * factor
    cand = simplex_projection(cand)
    # enforce mirror symmetry (the optimum is believed to be symmetric)
    cand = (cand + cand[::-1]) * 0.5
    return cand

# Main candidate generator required by the harness.

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Return a non‑negative step‑function (list of floats) with a low
    `evaluate_sequence` score.  The routine runs for roughly `budget_s`
    seconds, using Adam‑based smooth‑max gradient descent followed by
    random hill‑climbing and a short sub‑gradient clean‑up.
    """
    rng = np.random.default_rng(seed)

    # -------------------------
    # Choose a length (reuse previous if possible)
    # -------------------------
    if "height_sequence_1" in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=np.float64)
            n = init_seq.size
            if n < 2:
                raise ValueError
        except Exception:
            init_seq = None
            n = 6000
    else:
        init_seq = None
        n = 6000

    # -------------------------
    # Initialise w on the simplex
    # -------------------------
    if init_seq is not None:
        w = np.maximum(init_seq, 0.0)
        s = np.sum(w)
        if s < 1e-12:
            w = np.full(n, 1.0 / n)
        else:
            w = w / s
    else:
        # classic decreasing central‑binomial distribution
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2 * i)
        w = a / np.sum(a)

    # enforce symmetry and simplex feasibility
    w = (w + w[::-1]) * 0.5
    w = simplex_projection(w)

    # ------------------------------------------------------------------
    # Adam‑optimised smooth‑max gradient descent
    # ------------------------------------------------------------------
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.05                     # base learning rate
    mu_factor = 0.6
    mu_factor_min = 0.02
    mu_decay = 0.9998
    block_interval = 2000
    reset_interval = 25000

    iteration = 0
    no_improve = 0
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # main optimisation (≈70 % of the time budget)
    while time.monotonic() - start_time < budget_s * 0.70:
        b = convolve_fft(w, w)
        maxb = np.max(b)
        score = 2.0 * n * maxb

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # smooth‑max (soft‑max) weights
        mu = maxb * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        shifted = (b - maxb) / mu
        exp_term = np.exp(shifted)
        p = exp_term / np.sum(exp_term)

        # gradient of the smooth‑max objective
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # Adam step with projection
        w_candidate = w - alpha * m_hat / (np.sqrt(v_hat) + eps)
        w = simplex_projection(w_candidate)
        w = (w + w[::-1]) * 0.5                     # enforce symmetry

        # schedule for the temperature of the smooth‑max
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        # occasional cheap random jump
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand = (cand + cand[::-1]) * 0.5
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # restart from a perturbed best if stagnating
        if no_improve > reset_interval:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            w = (w + w[::-1]) * 0.5
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Hill‑climbing with multiplicative perturbations (simulated annealing)
    # ------------------------------------------------------------------
    sigma = 0.5
    sigma_min = 0.001
    sigma_decay = 0.9995
    temperature = 1.0
    temp_min = 0.001
    temp_decay = 0.9997
    w_hill = best_w.copy()
    hill_score = best_score
    hill_iter = 0

    while time.monotonic() - start_time < budget_s * 0.95:
        cand = multiplicative_perturb(w_hill, sigma, rng)
        cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))

        if cand_score < hill_score:
            w_hill = cand
            hill_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_w = cand.copy()
                print(f"[hill‑improve] iter={hill_iter:6d}, score={best_score:.6f}")
        else:
            # accept uphill moves with a temperature‑dependent probability
            prob = np.exp(-(cand_score - hill_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                w_hill = cand
                hill_score = cand_score

        sigma = max(sigma * sigma_decay, sigma_min)
        temperature = max(temperature * temp_decay, temp_min)
        hill_iter += 1
        if hill_iter > 200000:
            break

    # ------------------------------------------------------------------
    # Short sub‑gradient clean‑up phase (focus on the active peak)
    # ------------------------------------------------------------------
    sub_iter = 0
    eta0_sub = 0.3
    while time.monotonic() - start_time < max(0.0, budget_s - 0.5):
        b = convolve_fft(best_w, best_w)
        maxb = np.max(b)
        max_idxs = np.flatnonzero(b >= maxb - 1e-15)
        i_max = int(rng.choice(max_idxs))

        grad = np.zeros_like(best_w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        grad[j] = 2.0 * best_w[i_max - j]

        eta = eta0_sub / np.sqrt(sub_iter + 1.0)
        best_w = simplex_projection(best_w - eta * grad)
        best_w = (best_w + best_w[::-1]) * 0.5

        score = 2.0 * n * np.max(convolve_fft(best_w, best_w))
        if score < best_score - 1e-12:
            best_score = score
            print(f"[sub‑improve] sub_iter={sub_iter:6d}, score={best_score:.6f}")

        sub_iter += 1
        if sub_iter > 50000:
            break

    # ------------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but we output the
    # conventional √(2 n) ℓ₁‑norm for readability.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    print(f"[final] length={n}, best_score={best_score:.6f}, "
          f"after evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
