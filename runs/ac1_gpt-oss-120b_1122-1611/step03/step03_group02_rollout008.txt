"""
Propose a step‑function sequence that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm:

1. Work on the probability simplex (weights sum to 1).  The objective
   2·n·max(conv(w,w)) is invariant to scaling, therefore we keep
   `w` normalised to 1 and only optimise the shape.

2. Initialise three different candidates:
   * a discrete arcsine law (w_i ∝ 1/√((i+½)(N‑1‑i+½)))
   * the classic central‑binomial law (w_i ∝ C(2i,i)/4**i)
   * a random Dirichlet distribution.
   If a previous solution ``height_sequence_1`` is present we also
   start from it.

3. Optimise a smooth approximation of the ℓ∞‑norm using
   the log‑sum‑exp (smooth‑max) of the convolution:
       f(w) = μ·log Σ_k exp((w∗w)[k]/μ)
   Its gradient is 2·conv(p, w_rev) where p is the soft‑max over
   the convolution values.  We perform Adam updates on the logits
   `z` (w = softmax(z)), which automatically respects positivity
   and the simplex constraint.

4. The smoothing parameter μ = μ_factor·max_b is slowly reduced.
   An occasional “reset” (perturb the current best with log‑normal
   noise) helps to escape plateaus.

5. After the smooth optimisation a short sub‑gradient phase focuses
   directly on the active peak(s) of the (unsmoothed) convolution.
   This phase often squeezes a few more digits out of the optimal
   value.

6. The best sequence found for any tried length is scaled to the
   conventional √(2·n) ℓ₁‑norm and returned.  The final evaluation is
   printed for verification.

The whole search is bounded by the supplied ``budget_s`` (≈1100 s)
and is deliberately lightweight on the CPU – FFT‑based convolutions
are O(N log N) and the inner loop runs for tens of thousands of
iterations within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for N≈10⁴–10⁵)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Tiny negative values can appear because of rounding; clip them.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # This can only happen when the input is all zeros – return uniform.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_logits(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random block perturbation.  Used as a jump operator
    when the optimiser stalls.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetric blending (helps to keep the distribution balanced)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Core optimisation routine for a single length
# ----------------------------------------------------------------------


def _optimise_length(
    n: int,
    rng: np.random.Generator,
    global_start: float,
    budget_s: float,
    previous_seq: np.ndarray | None,
) -> tuple[np.ndarray, float]:
    """
    Returns the best weight vector (sum = 1) found for length ``n`` and its
    un‑scaled score  ``2·n·max(conv(w,w))``.
    """
    # ------------------------------------------------------------------
    # Preparation of initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # 0 – discrete arcsine law (β(½,½) density)
    i = np.arange(n, dtype=np.float64)
    eps = 0.5               # shift to avoid division by zero at the borders
    arcsine = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    arcsine = arcsine / arcsine.sum()
    init_candidates.append(arcsine)

    # 1 – central‑binomial (the classic start from the previous work)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    binom = a / a.sum()
    init_candidates.append(binom)

    # 2 – random Dirichlet
    dirichlet = rng.dirichlet(np.ones(n))
    init_candidates.append(dirichlet)

    # 3 – If a previous solution of the same length is available, reuse it.
    if previous_seq is not None and previous_seq.size == n:
        start = np.maximum(previous_seq, 0.0)
        s = start.sum()
        if s > 1e-12:
            init_candidates.append(start / s)

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters (chosen after a few empirical tests)
    # ------------------------------------------------------------------
    alpha = 0.48               # base learning‑rate for Adam
    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-8

    mu_factor = 0.70           # start smoothing factor (relative to max_b)
    mu_min = 0.02
    mu_decay = 0.9995

    max_iters = 250_000        # hard cap – we break earlier if time expires
    reset_after = 30_000       # number of iterations without improvement that
                               # trigger a random restart
    block_every = 5_000        # attempt a cheap random jump every this many steps

    best_w = None
    best_score = np.inf
    best_iter = -1

    # ------------------------------------------------------------------
    # Loop over all initial guesses
    # ------------------------------------------------------------------
    for init_idx, w0 in enumerate(init_candidates):
        # Convert the probability vector to logits (so that we can use Adam).
        z = np.log(np.maximum(w0, 1e-12))

        m = np.zeros_like(z)
        v = np.zeros_like(z)

        no_improve = 0
        iteration = 0

        while iteration < max_iters:
            # Time guard – leave a small safety margin.
            if time.monotonic() - global_start > budget_s - 2.0:
                break

            # --------- 1) current weights and convolution -----------------
            w = softmax_logits(z)                # w lives on the simplex, ∑w = 1
            b = convolve_fft(w, w)               # linear self‑convolution
            maxb = np.max(b)
            score = 2.0 * n * maxb                # the value that evaluate_sequence would return

            # --------- 2) keep the best seen so far ----------------------
            if score < best_score - 1e-12:
                best_score = float(score)
                best_w = w.copy()
                best_iter = iteration
                no_improve = 0
                print(f"[len={n}] init={init_idx} iter={iteration:6d} new best {best_score:.6f}")
            else:
                no_improve += 1

            # --------- 3) smooth‑max weights (log‑sum‑exp) ---------------
            mu = maxb * mu_factor
            if mu <= 0.0:
                mu = 1e-12
            shifted = (b - maxb) / mu          # max → 0, others ≤ 0
            # Clamp to avoid overflow in the exponential.
            exp_shifted = np.exp(np.clip(shifted, -50.0, 0.0))
            p = exp_shifted / exp_shifted.sum()  # soft‑max over convolution entries

            # --------- 4) gradient of the smooth objective ---------------
            # grad_w = 2·conv(p, w_rev)  → slice to length n
            grad_w = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

            # --------- 5) convert to gradient on logits ------------------
            # For softmax, ∇_z L = w ⊙ (∇_w L - (w·∇_w L)·1)
            dot = np.dot(w, grad_w)
            grad_z = w * (grad_w - dot)

            # --------- 6) Adam update ------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            t = iteration + 1
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            z -= alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

            # --------- 7) schedule smoothing factor ----------------------
            mu_factor = max(mu_factor * mu_decay, mu_min)

            # --------- 8) occasional random block jump --------------------
            if iteration % block_every == 0 and iteration > 0:
                cand = block_perturbation(best_w, rng)
                cand = cand / cand.sum()
                z_cand = np.log(np.maximum(cand, 1e-12))
                # One quick evaluation – no Adam state is kept.
                w_cand = cand
                b_cand = convolve_fft(w_cand, w_cand)
                score_cand = 2.0 * n * np.max(b_cand)
                if score_cand < best_score:
                    best_score = float(score_cand)
                    best_w = w_cand.copy()
                    best_iter = iteration
                    z = z_cand
                    m.fill(0.0)
                    v.fill(0.0)
                    no_improve = 0
                    print(f"[len={n}] block‑jump iter={iteration:6d} new best {best_score:.6f}")

            # --------- 9) restart if stuck --------------------------------
            if no_improve > reset_after:
                # Perturb the current best with log‑normal noise.
                noise = rng.normal(0.0, 0.12, size=n)
                w_pert = best_w * np.exp(noise)
                w_pert = w_pert / w_pert.sum()
                z = np.log(np.maximum(w_pert, 1e-12))
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[len={n}] restart iter={iteration:6d}")

            iteration += 1

        # ------------------------------------------------------------------
        # Short sub‑gradient polishing (focuses on the true ℓ∞‑norm)
        # ------------------------------------------------------------------
        if best_w is None:
            continue
        w = best_w.copy()
        eta_sub = 0.70
        eta_decay = 0.9996
        sub_iters = 0
        while sub_iters < 20_000:
            if time.monotonic() - global_start > budget_s - 1.0:
                break
            b = convolve_fft(w, w)
            maxb = np.max(b)
            # Gather all indices whose value is within a tiny tolerance of the maximum.
            active = np.flatnonzero(b >= maxb - 1e-15)
            i_max = int(rng.choice(active))

            # Sub‑gradient of the unsmoothed ℓ∞‑norm for the active index.
            grad = np.zeros_like(w)
            j_start = max(0, i_max - (n - 1))
            j_end = min(i_max, n - 1)
            j = np.arange(j_start, j_end + 1, dtype=int)
            grad[j] = 2.0 * w[i_max - j]          # contribution of each pair

            # Projected sub‑gradient step.
            w = simplex_projection(w - eta_sub * grad)

            # Check for improvement.
            new_score = 2.0 * n * np.max(convolve_fft(w, w))
            if new_score < best_score - 1e-12:
                best_score = float(new_score)
                best_w = w.copy()
                print(f"[len={n}] sub‑improve iter={sub_iters:5d} score={best_score:.6f}")

            eta_sub *= eta_decay
            sub_iters += 1

        # If budget is exhausted we can stop early.
        if time.monotonic() - global_start > budget_s - 0.5:
            break

    # ------------------------------------------------------------------
    # Return the best weight vector (still normalized to sum=1) and its
    # un‑scaled score 2·n·max_b.
    # ------------------------------------------------------------------
    return best_w, best_score


# ----------------------------------------------------------------------
# Public entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine tries a few sequence lengths
    (starting with the length of a previously found solution if present)
    and applies Adam optimisation on a smooth approximation of the ℓ∞‑norm.
    A short sub‑gradient polishing phase finishes the search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # Determine which lengths to explore.  We start with the length of a
    # previous solution (if any) and then try a couple of larger sizes.
    # ------------------------------------------------------------------
    lengths = []
    prev_seq = None
    if "height_sequence_1" in globals():
        try:
            prev_seq = np.asarray(height_sequence_1, dtype=np.float64)
            prev_len = int(prev_seq.size)
            if prev_len >= 4:
                lengths.append(prev_len)
        except Exception:
            prev_seq = None

    # Additional lengths – chosen to keep the FFT cheap while still
    # allowing a finer discretisation.
    extra_lengths = [6000, 8000, 12000, 16000]
    for L in extra_lengths:
        if L not in lengths:
            lengths.append(L)

    overall_best_seq = None
    overall_best_score = np.inf
    overall_best_len = None

    # ------------------------------------------------------------------
    # Run optimisation for each selected length.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() - start_time > budget_s - 5.0:
            break
        print(f"[main] Starting optimisation for length n={n}")
        w_opt, score = _optimise_length(
            n=n,
            rng=rng,
            global_start=start_time,
            budget_s=budget_s,
            previous_seq=prev_seq,
        )
        # Scale to the conventional √(2·n) ℓ₁‑norm (does not change the score).
        seq = w_opt * np.sqrt(2.0 * n) / w_opt.sum()
        final_score = float(evaluate_sequence(seq.tolist()))
        print(f"[main] length={n} -> evaluate={final_score:.6f}")

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq
            overall_best_len = n

    # ------------------------------------------------------------------
    # Fallback – should never happen, but we protect against an empty result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 8000
        uniform = np.full(n, 1.0 / n)
        overall_best_seq = uniform * np.sqrt(2.0 * n)

    # Final sanity check (the evaluation may raise if something went wrong).
    final_score = float(evaluate_sequence(overall_best_seq.tolist()))
    print(
        f"[final] best length={overall_best_len}, final evaluate={final_score:.6f}"
    )

    # Convert to a plain Python list as required by the harness.
    return [float(x) for x in overall_best_seq]
