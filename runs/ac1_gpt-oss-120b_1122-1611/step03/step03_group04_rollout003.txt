"""
propose_candidate
-----------------

Momentum‑enhanced exponentiated‑gradient descent for the
scale‑invariant problem

    minimise   max_k (w * w)[k]      s.t.  w_i ≥ 0 , Σ w_i = 1 .

The algorithm works on the probability vector w (ℓ¹‑normalised),
updates the log‑weights u = log w by Adam‑style momentum on the
gradient of a smoothed max (soft‑max) of the self‑convolution.
A temperature β that grows like √iteration turns the non‑smooth
objective into a smooth one.  After each multiplicative update the
vector is re‑normalised and symmetrised (the optimal solution is
symmetric), which halves the effective search space.  Periodic
block‑perturbations and a reset mechanism help escape plateaus.
The final vector is scaled to the conventional √(2 n) ℓ¹‑norm before
returning.

The implementation respects the 1000 s runtime limit, prints a few
progress messages, and works with the global variable
`height_sequence_1` (when available) as a warm start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negative values that appear because of floating noise.
    np.maximum(c, 0.0, out=c)
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max:   g = 2·corr(w, α)

    α – probability vector of length 2·n‑1, w – probability vector of
    length n.  The correlation is evaluated via a second FFT
    convolution.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)            # length = 3·n‑2
    start = w.shape[0] - 1                       # index that aligns with j=0
    return 2.0 * conv[start:start + w.shape[0]]


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump: multiply a random block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)        # 0.5 % of length, at least 1
    hi = max(2, n // 12)         # up to ~8 % of length
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal multiplier
    cand[start:stop] *= factor
    # occasional symmetric smoothing
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def sin_initial(length: int) -> np.ndarray:
    """Half‑wave sine shape – a smooth, centrally‑peaked start."""
    i = np.arange(length, dtype=np.float64)
    seq = np.sin(np.pi * (i + 0.5) / length)
    return seq

def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low
    `evaluate_sequence` score.  The routine respects the time budget
    (≈ 1000 s) and prints intermediate progress.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a length and an initial probability vector w (ℓ¹‑normalised).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
            w = prev.copy()
        except Exception:
            n = 8000
            w = sin_initial(n)
    else:
        n = 8000
        w = sin_initial(n)

    # Guard against negatives, normalise to sum‑one and enforce symmetry.
    w = np.maximum(w, 0.0)
    w = w / np.sum(w)
    w = 0.5 * (w + w[::-1])
    w = w / np.sum(w)

    # Momentum term (same shape as w).  We use a simple exponential moving
    # average of the raw gradient.
    m = np.zeros_like(w)

    # Keep the best solution seen so far.
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = float(2.0 * n * np.max(b))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    iteration = 0
    last_improve = 0

    # Hyper‑parameters – values were tuned empirically.
    eta0 = 3.0                # base learning‑rate for the multiplicative update
    beta_mom = 0.9            # momentum decay
    beta0 = 0.10              # temperature growth factor (β ≈ β0·√t)
    beta_max = 800.0          # safety cap for temperature
    block_every = 2500        # block perturbation interval
    reset_patience = 80000    # iterations without improvement before a reset

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ---- self‑convolution and raw score --------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            last_improve = iteration
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")

        # ---- smoothed max (soft‑max) weights α -----------------------------
        beta = beta0 * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b                      # ≤ 0, max element becomes 0
        exp_term = np.exp(beta * diff)        # stable: max term = 1
        alpha = exp_term / np.sum(exp_term)   # probability vector, length 2·n‑1

        # ---- gradient of the smoothed objective ---------------------------
        grad = correlation_grad(w, alpha)     # shape (n,)

        # ---- momentum (exponential moving average) -----------------------
        m = beta_mom * m + (1.0 - beta_mom) * grad

        # ---- multiplicative (exponentiated) update -------------------------
        eta = eta0 / np.sqrt(iteration + 1)
        w = w * np.exp(-eta * m)

        # ---- re‑normalise, enforce symmetry --------------------------------
        total = w.sum()
        if total <= 0.0 or not np.isfinite(total):
            w = np.full_like(w, 1.0 / n)
            total = 1.0
        w = w / total
        w = 0.5 * (w + w[::-1])   # symmetry
        w = w / np.sum(w)

        # ---- occasional cheap random jump ---------------------------------
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand = 0.5 * (cand + cand[::-1])
            cand = cand / np.sum(cand)
            cand_score = float(2.0 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                w = cand.copy()
                m = np.zeros_like(w)   # wipe momentum after a jump
                print(f"[block] iter={iteration:7d}, new best={best_score:.6f}")

        # ---- reset if stuck for a long time -------------------------------
        if iteration - last_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            w = 0.5 * (w + w[::-1])
            w = w / np.sum(w)
            m = np.zeros_like(w)
            last_improve = iteration
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final fine‑grained local search (time‑limited)
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 2.0):
        cand = block_perturbation(best_w, rng)
        cand = cand / np.sum(cand)
        cand = 0.5 * (cand + cand[::-1])
        cand = cand / np.sum(cand)
        cand_score = float(2.0 * n * np.max(convolve_fft(cand, cand)))
        if cand_score < best_score:
            best_score = cand_score
            best_w = cand.copy()
            print(f"[local] iter={iteration:7d}, new best={best_score:.6f}")
        iteration += 1
        if iteration % 10000 == 0:
            # occasional status line
            elapsed = time.monotonic() - start_time
            print(f"[status] it={iteration}, best={best_score:.6f}, elapsed={elapsed:.1f}s")
            # avoid overrunning the budget
            if elapsed > budget_s - 2.0:
                break

    # ------------------------------------------------------------------
    # Scale to the conventional √(2 n) ℓ¹‑norm (the evaluator is
    # scale‑invariant, but this gives a nice looking output).
    # ------------------------------------------------------------------
    scale = np.sqrt(2.0 * n) / np.sum(best_w)   # sum(best_w)=1 → scale=√(2 n)
    final_seq = best_w * scale

    # --------------------------------------------------------------
    # Optional sanity check with the provided evaluator (for logging)
    # --------------------------------------------------------------
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:
        final_score = float('nan')
    print(f"[final] length={n}, raw best score={best_score:.6f}, "
          f"evaluate={final_score:.6f}")

    # Return a plain Python list of floats.
    return [float(x) for x in final_seq]
