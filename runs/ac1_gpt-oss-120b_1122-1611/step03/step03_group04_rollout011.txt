"""
propose_candidate
-----------------

Hybrid search for a step‑function that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)²)

(The evaluation function is scale‑invariant, so we keep the vector
normalised to sum = 1 during optimisation and rescale at the end.)

The method consists of three stages:

1. **Smart initialisation** – scan a few one‑parameter families
   (uniform, power‑law, exponential, Catalan) and keep the best
   candidate.

2. **Entropic mirror descent** – the same smooth‑max gradient descent
   used in the reference solution, but with more aggressive step‑size,
   always‐on symmetry, and several independent restarts (different
   random seeds and initial shapes).  Convolutions are performed with a
   real FFT for speed.

3. **Late‑stage random‑block perturbations** – cheap O(n) jumps that
   occasionally replace the current best by a nearby candidate.

All operations stay within the allowed budget (≈1000 s).  The final
sequence is scaled to the conventional ℓ¹‑norm √(2 n) before being
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First `length` Catalan numbers C_i / 4**i (ratio form)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump: multiply a random block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 % of length
    hi = max(2, n // 12)           # up to ~8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    #‑‑ optional symmetric smoothing
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max:
        g = 2·corr(w, α) .
    ``α`` is a probability vector of length ``2·n‑1``.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = 3·n‑2
    start = w.shape[0] - 1                     # centre index aligns with lag 0
    return 2.0 * conv[start:start + w.shape[0]]


def compute_score(w: np.ndarray) -> float:
    """Raw, scale‑invariant score: 2·n·max(conv(w,w))  (w is already normalised)."""
    n = w.size
    b = convolve_fft(w, w)
    max_b = np.max(b)
    return float(2 * n * max_b)

# Main candidate proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # ------------------------------------------------------------------

    # 10000 is a sweet spot for the 1000 s budget.
    n = 10000

    # ------------------------------------------------------------------
    candidates = []

    uniform = np.full(n, 1.0 / n)
    candidates.append(uniform)

    # Catalan (classic start)
    cat = catalan_sequence(n)
    cat = cat / np.sum(cat)
    candidates.append(cat)

    # power‑law a_i ∝ (i+1)^(-p), symmetrised
    idx = np.arange(n)
    for p in np.linspace(0.0, 2.0, 17):          # 0 .. 2 in 0.125 steps
        w = (idx + 1) ** (-p)
        w = 0.5 * (w + w[::-1])                 # enforce symmetry
        w = w / np.sum(w)
        candidates.append(w)

    # exponential decay a_i ∝ exp(-λ·i), symmetrised
    for lam in np.linspace(1e-4, 5e-3, 15):

        w = w / np.sum(w)

        iteration = 0
        no_improve = 0
        local_best_w = w.copy()
        local_best_score = compute_score(w)

        while time.monotonic() - start_time < budget_s - 10.0:
            # ---- convolution & current (hard) score ----
            b = convolve_fft(w, w)
            max_b = np.max(b)
            score = 2.0 * n * max_b

            # ---- keep global best ----
            if score < best_score - 1e-12:

                best_w = w.copy()
                print(f"[improve] restart={restart_idx}, iter={iteration}, score={best_score:.6f}")
