"""
propose_candidate
-----------------

A multi‑start, symmetry‑aware Adam optimiser that first minimises a
log‑sum‑exp smooth surrogate of the peak self‑convolution and then
switches to the exact sub‑gradient (max‑convolution) to polish the
solution.  The optimiser works on a probability simplex (∑a=1,
a_i≥0) and periodically performs cheap symmetric block perturbations
to escape local minima.  Several analytically motivated initial
distributions (Catalan, power‑law, exponential) are tried and the
best run is returned.  The final sequence is scaled to the conventional
ℓ¹‑norm √(2 n) so that the value returned by ``evaluate_sequence`` equals
the achieved maximum of the self‑convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    Return the first ``length`` Catalan numbers divided by 4**i,
    i.e. a_i = C(2i,i)/4**i.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, ∑x=1}.
    Duchi et al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation: multiply a contiguous block by a random
    factor and (with some probability) symmetrise the result.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # ≈0.5 %
    hi = max(2, n // 12)           # ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ≈±12 % multiplicative noise
    cand[start:stop] *= factor
    # optional symmetric averaging – useful because the optimal shape is
    # essentially symmetric w.r.t. reversal.
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """
    Linear self‑convolution (full length 2·len(w)‑1) using an FFT of length N.
    Tiny negative rounding errors are clipped to zero.
    """
    F = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(F * F, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth max‑approximation L_β at point w.
    ``probs`` are the soft‑max weights (sum 1, length 2·len(w)‑1).
    """
    # correlation w ⋆ probs  (note the conjugate because we use real FFT)
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = probs.size
    offset = m - 1
    # the linear correlation has length len(w)+len(probs)-1 = 3·n‑2;
    # we need the slice that aligns with w.
    grad = 2.0 * corr_full[offset : offset + w.size]
    return grad


def grad_subconv(w: np.ndarray, lag: int) -> np.ndarray:
    """
    Exact sub‑gradient of the peak convolution at a given lag.
    ``lag`` is the index of the convolution (0 … 2·n‑2).  Because the
    convolution is symmetric we mirror lags larger than n‑1.
    """
    n = w.size
    # map to the range 0 … n‑1
    if lag > n - 1:
        lag = 2 * n - 2 - lag
    # now lag ∈ [0, n‑1]
    grad = np.zeros_like(w)
    if lag == 0:
        # derivative of w[0] * w[0] + … = 2·w   (but we only have w_i*w_i terms)
        grad = 2.0 * w
    else:
        # contributions from w_i * w_{i+lag} and w_{i+lag} * w_i
        grad[: n - lag] += w[lag:]
        grad[lag:] += w[: n - lag]
        grad *= 2.0
    return grad


def max_conv(w: np.ndarray, N_self: int) -> float:
    """Convenient wrapper returning max self‑convolution of w."""
    return float(np.max(self_convolve_fft(w, N_self)))

        if iteration % block_interval == 0:
            cand = block_perturbation(best_w, rng)
            cand = 0.5 * (cand + cand[::-1])
            cand = cand / np.sum(cand)
            cand_score = evaluate_candidate(cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[block] iter={iteration:6d}, new best max‑conv={best_score:.6f}")

        # 6) reset on prolonged stagnation
        if no_improve > stagnation_limit:
            w = block_perturbation(best_w, rng)
            w = 0.5 * (w + w[::-1])
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        # safety stop if we already beat the target
        if best_score * 2.0 * n < 1.5025:
            print("[target] reached desired bound – early exit")
            break

        # safeguard against exceeding a hard iteration ceiling
        if iteration >= max_iters:
            break

    # --------------------------------------------------------------
    # Final scaling to the conventional ℓ¹‑norm √(2 n)
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum should already be 1

    # Optional sanity check – cheap and useful while debugging
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best max‑conv={best_score:.6f}, "
              f"evaluate={final_score:.6f}")
    except Exception:
        pass

    # Return as a plain Python list (required by the harness)
    return [float(x) for x in final_seq]
