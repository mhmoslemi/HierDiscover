"""
Search for a low‑peak step function.

The evaluation metric is
    score = 2·n·max(a*a) / (∑a)² .
Because the score is invariant under scaling we work on the probability simplex
(∑w = 1, w ≥ 0) and finally scale the best w to the conventional ℓ₁‑norm
√(2 n).

The algorithm proceeds in three stages:

1. **Exploratory families** – for several lengths we generate candidates
   from a few parametric families (uniform, sinusoidal, symmetric Beta,
   discrete Gaussian).  Each candidate is evaluated using a fast FFT‑based
   linear convolution and the best one is kept.

2. **Local refinement** – the best candidate of a given length is refined
   by projected (multiplicative) Adam on a smooth approximation of the
   L∞‑norm:
       f(w) = μ·log Σ exp((w*w)[k]/μ) .
   The gradient of f is
       ∇f = 2·conv(p, rev(w)),
   where p is the soft‑max over the convolution entries.
   The update is performed in the probability simplex by the
   exponentiated‑gradient trick (w ← w·exp(‑η·∇f) and renormalisation).

3. **Random perturbations** – occasional block‑perturbations (the same
   cheap O(n) jump used in the reference code) are applied to escape
   local minima and to explore the neighbourhood of the current best.

All work is bounded by the supplied time budget (≈ 1000 s).  The function
returns the best sequence found (as a plain Python list) and prints a
few progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear convolution via a real FFT (fast for n up to a few 10⁴).
# ----------------------------------------------------------------------
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using an FFT of size a power of two."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # smallest power‑of‑two ≥ n
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # tiny negative values can appear because of round‑off – clip them.
    np.maximum(c, 0.0, out=c)
    return c


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, ∑x = 1}.
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Cheap random block perturbation – a cheap O(n) “jump” operator.
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Modify a random contiguous block of `base` by a multiplicative factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal factor
    cand[start:stop] *= factor
    # with a small probability also enforce (approximate) symmetry
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Helper: compute the maximum of the (linear) self‑convolution.
# The result equals the score when the vector is scaled to √(2 n).
# ----------------------------------------------------------------------
def max_conv(seq: np.ndarray) -> float:
    """Return max_i (seq * seq)[i] for a non‑negative vector."""
    return float(convolve_fft(seq, seq).max())


# ----------------------------------------------------------------------
# Helper: smooth‑max gradient for a single step of Adam.
# ----------------------------------------------------------------------
def smooth_max_grad(w: np.ndarray, mu_factor: float):
    """
    Compute a smooth approximation of the L∞ norm of w*w and its gradient.

    Returns (max_b, grad, p, mu) where
        max_b = max_i (w*w)[i],
        grad   = 2·conv(p, rev(w))  (vector of length n),
        p      = soft‑max(b/μ) (probability vector of length 2n‑1),
        mu     = smoothing parameter (≈ mu_factor·max_b).
    """
    n = w.size
    b = convolve_fft(w, w)
    maxb = b.max()
    mu = maxb * mu_factor
    if mu <= 0.0:
        mu = 1e-12
    # centre the exponent to avoid overflow
    shifted = (b - maxb) / mu          # maxb → 0, others ≤ 0
    max_shift = shifted.max()
    exp_shift = np.exp(shifted - max_shift)
    p = exp_shift / exp_shift.sum()
    # gradient = 2·conv(p, rev(w)) and we need the central n entries
    rev = w[::-1]
    full = convolve_fft(p, rev)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return maxb, grad, p, mu

        eps = 1e-8
        eta = 0.12          # base learning‑rate for the multiplicative step
        mu_factor = 0.6     # initial smoothing factor (as a fraction of max_b)

        # Perform a few thousand Adam steps (still cheap thanks to FFT).
        max_iter = 6000
        for it in range(1, max_iter + 1):
            if time_left() < 5.0:
                break

            maxb, grad, _, mu = smooth_max_grad(w, mu_factor)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            # bias‑corrected estimates
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # multiplicative Adam step
            step = eta * m_hat / (np.sqrt(v_hat) + eps)
            w = w * np.exp(-step)      # keep positivity
            w = w / w.sum()            # simplex projection (just normalise)

            # slowly tighten the smoothing
            mu_factor = max(0.04, mu_factor * 0.9999)

            # Periodically evaluate the true (unsmoothed) objective.
            if it % 200 == 0:
                seq = w * target_sum
                sc = max_conv(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[adam] iter={it:4d}  score={best_score:.6f}")

        # ----------------------------------------------------------------
        # 5.  A short sub‑gradient “cleanup” phase on the exact L∞ norm.
        # ----------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            w = best_seq / target_sum
            eta_sub = 0.6
            for sub_it in range(5000):
                if time_left() < 2.0:
                    break
                # exact convolution and active index
                b = convolve_fft(w, w)
                maxb = b.max()
                # pick one of the (maybe many) maximising indices uniformly
                idxs = np.flatnonzero(b >= maxb - 1e-12)
                i_max = int(rng.choice(idxs))
                # sub‑gradient of max entry:
                grad = np.zeros_like(w)
                j0 = max(0, i_max - (n - 1))
                j1 = min(n - 1, i_max)
                j = np.arange(j0, j1 + 1, dtype=int)
                grad[j] = 2.0 * w[i_max - j]
                # step size diminishing like 1/√t
                step = eta_sub / np.sqrt(sub_it + 1.0)
                w = simplex_projection(w - step * grad)
                # occasional evaluation
                if sub_it % 250 == 0:
                    seq = w * target_sum
                    sc = max_conv(seq)
                    if sc < best_score:
                        best_score, best_seq = sc, seq.copy()
                        print(f"[sub‑grad] it={sub_it:4d}  score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – if everything failed fall back to a simple uniform vector.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = max_conv(best_seq)
        print("[fallback] using uniform sequence")

    # ----------------------------------------------------------------------
    # Final sanity check: run the provided evaluator (it also applies the
    # clipping and scaling safeguards) and print the resulting score.
    # ----------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"\n[final] length={best_seq.size}, best_score (max_conv)={best_score:.6f}, "
          f"evaluate_sequence={final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq.tolist()]
