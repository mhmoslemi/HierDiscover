def propose_candidate(...):
    rng = np.random.default_rng(seed)
    # Determine length n (maybe 8000)
    n = 8000
    # Initial vector candidates: Catalan, random uniform, maybe Beta distribution.
    candidates = []
    cand_catalan = catalan_sequence(n)
    cand_catalan /= cand_catalan.sum()
    candidates.append(cand_catalan)
    # Add some random ones:
    for _ in range(2):
        v = rng.random(n)
        v = np.maximum(v, 0)
        v /= v.sum()
        candidates.append(v)

    # For each candidate run mirror descent for some iterations, keep best.
    best_w = None; best_max = np.inf
    start_time = time.monotonic()
    # allocate time share:
    total_budget = max(0., float(budget_s) - 4.0)  # keep margin
    # We'll loop using while time remains:
    while time.monotonic() - start_time < total_budget:
        # pick a candidate base w, maybe random from candidates or best_w
        if best_w is not None and rng.random() < 0.5:
            w = best_w.copy()
        else:
            w = candidates[rng.integers(len(candidates))].copy()
        # inner optimization loop
        iteration = 0
        no_improve_iter = 0
        max_iters = 20000  # maybe limit inner loop.
        while iteration < max_iters and time.monotonic() - start_time < total_budget:
            b = convolve_fft(w, w)
            max_b = np.max(b)
            if max_b < best_max:
                best_max = max_b
                best_w = w.copy()
                no_improve_iter = 0
                # print improvement
                print(f"[improve] iter_total={iteration_total}, max_b={best_max:.6f}")
            else:
                no_improve_iter += 1

            # Choose active shifts: indices where b within tolerance of max_b
            tol = 1e-12
            active = np.where(b >= max_b - tol)[0]
            # pick a subset: up to 5 random indices
            if active.size > 5:
                idxs = rng.choice(active, size=5, replace=False)
            else:
                idxs = active

            # compute gradient as average of shifts
            grad = np.zeros_like(w)
            for k in idxs:
                grad += shift_gradient(w, k)
            grad /= len(idxs)

            # exponentiated gradient step
            eta0 = 0.6
            eta = eta0 / np.sqrt(iteration + 1.0)
            # apply clipping
            exp_factor = -eta * grad
            max_exp = 30.0
            np.clip(exp_factor, -max_exp, max_exp, out=exp_factor)
            w *= np.exp(exp_factor)
            # renormalize
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / n)
            else:
                w /= w_sum

def generate_shape(n, alpha, a):
    idx = np.arange(n)
    d = np.minimum(idx, n-1-idx)
    w = (d + a) ** (-alpha)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full(n, 1.0/n)
    else:
        w /= w_sum
    return w
```

"""
propose_candidate
~~~~~~~~~~~~~~~~~

Search for a non‑negative step function that minimizes

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation is invariant under a global scaling, therefore we work with a
probability vector `w` (Σ w = 1) and finally rescale it to the conventional
√(2 n) L¹‑norm required by ``evaluate_sequence``.

The search consists of two stages:

1. **Parametric grid + local random refinement**  
   We explore a three‑parameter symmetric family

       w_i(α, a, c) ∝ ( min(i, n-1-i) + a )^{-α} + c ,

   where `α` controls the power‑law decay, `a` is an offset preventing a
   singularity at the boundary and `c ≥ 0` adds a tiny uniform component.
   A coarse grid (α, a, c) is scanned, the best candidate is kept and then
   refined by a short random walk in the (α, a, c) space.

2. **Hard‑max mirror descent**  
   Starting from the best parametric vector we run a mirror‑descent (exponentiated‑gradient)
   iteration that uses the true sub‑gradient of the max‑convolution
   (i.e. only the shift(s) attaining the maximum).  The update respects the
   simplex geometry and a periodic symmetrisation keeps the vector nearly
   symmetric.  The loop runs for the remaining time budget.

All operations are O(n log n) thanks to FFT‑based convolution; the
implementation stays well within the 1000 s limit.

The function returns a plain Python list of floats – the evaluator will
clip values to [0, 1000] and compute the score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # numerical safety – negative round‑off noise is clipped to zero
    np.maximum(c, 0.0, out=c)
    return c


def shape_vector(n: int, alpha: float, offset: float, uniform: float) -> np.ndarray:
    """
    Symmetric power‑law shape:
        w_i ∝ (min(i, n-1-i) + offset)^{-alpha} + uniform .
    Returns a probability vector (sum = 1).
    """
    idx = np.arange(n, dtype=np.float64)
    d = np.minimum(idx, n - 1 - idx)           # distance to the nearest edge
    w = (d + offset) ** (-alpha) + uniform
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / n)
    else:
        w /= s
    return w


def hard_max_gradient(w: np.ndarray, k: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(w * w) with respect to w,
    for a single active shift index ``k``.
    """
    n = w.size
    grad = np.zeros_like(w)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    grad[i] = w[k - i]          # shift_k(w)
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights whose `evaluate_sequence`
    score is as low as possible (the lower the better).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # Choose the length of the step function.
    # Prefer the previously discovered construction if it exists,
    # otherwise fall back to a modest size that still allows a fine
    # discretisation.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 10:
                raise ValueError
            # start from the previous shape, normalised to a probability vector
            w_start = prev / np.sum(prev)
        except Exception:
            n = 6000
            w_start = shape_vector(n, alpha=0.5, offset=1.0, uniform=0.0)
    else:
        n = 6000
        w_start = shape_vector(n, alpha=0.5, offset=1.0, uniform=0.0)

    # ------------------------------------------------------------------
    # Stage 1 – parametric exploration (grid + random refinement)
    # ------------------------------------------------------------------
    best_w = w_start.copy()
    best_val = np.max(convolve_fft(best_w, best_w))

    # coarse grid
    alphas = np.linspace(0.30, 0.80, 26)          # 0.30 … 0.80
    offsets = np.linspace(0.5, 4.0, 24)          # 0.5 … 4.0
    uniforms = np.linspace(0.0, 0.05, 6)         # 0 … 0.05

    # we stop the grid early if the overall time budget is almost exhausted
    max_grid_time = budget_s * 0.25
    for alpha in alphas:
        for off in offsets:
            for uni in uniforms:
                if time.monotonic() - start_time > max_grid_time:
                    break
                w = shape_vector(n, alpha, off, uni)
                cur = np.max(convolve_fft(w, w))
                if cur < best_val:
                    best_val = cur
                    best_w = w.copy()
                    best_alpha, best_off, best_uni = alpha, off, uni
                    print(f"[grid] new best {best_val:.6f} (α={alpha:.3f}, a={off:.3f}, c={uni:.3f})")
    else:
        # make sure the variables exist even if the loop never updated them
        best_alpha = best_alpha if 'best_alpha' in locals() else 0.5
        best_off   = best_off   if 'best_off'   in locals() else 1.0
        best_uni   = best_uni   if 'best_uni'   in locals() else 0.0

    # local random walk in the three‑parameter space
    refine_time = budget_s * 0.20
    while time.monotonic() - start_time < refine_time:
        cand_alpha = np.clip(best_alpha + rng.normal(scale=0.015), 0.10, 1.0)
        cand_off   = np.clip(best_off   + rng.normal(scale=0.20 ), 0.01, 10.0)
        cand_uni   = np.clip(best_uni   + rng.normal(scale=0.008), 0.0 , 0.10)
        w = shape_vector(n, cand_alpha, cand_off, cand_uni)
        cur = np.max(convolve_fft(w, w))
        if cur < best_val:
            best_val = cur
            best_w   = w.copy()
            best_alpha, best_off, best_uni = cand_alpha, cand_off, cand_uni
            print(f"[refine] new best {best_val:.6f} (α={cand_alpha:.3f}, a={cand_off:.3f}, c={cand_uni:.3f})")

    # ------------------------------------------------------------------
    # Stage 2 – hard‑max mirror‑descent starting from the best shape.
    # ------------------------------------------------------------------
    remaining = max(0.0, budget_s - (time.monotonic() - start_time) - 0.5)
    w = best_w.copy()
    best_local_w = w.copy()
    best_local_val = best_val
    iteration = 0
    no_improve = 0

    # hyper‑parameters
    eta0 = 0.6
    max_exp = 30.0                # safety clipping for the exponent
    sym_interval = 800
    stagnation_limit = 6000

    while time.monotonic() - start_time < remaining:
        # evaluate
        conv = convolve_fft(w, w)
        cur_max = np.max(conv)

        # record improvement
        if cur_max < best_local_val - 1e-12:
            best_local_val = cur_max
            best_local_w = w.copy()
            no_improve = 0
            print(f"[improve] it={iteration:06d}, score={best_local_val:.6f}")
        else:
            no_improve += 1

        # compute sub‑gradient from a random active index
        active = np.where(conv >= cur_max - 1e-12)[0]
        k = int(rng.choice(active))
        grad = hard_max_gradient(w, k)

        # exponentiated‑gradient step with diminishing step size
        eta = eta0 / np.sqrt(iteration + 1.0)
        exponent = -eta * grad
        np.clip(exponent, -max_exp, max_exp, out=exponent)
        w = w * np.exp(exponent)

        # renormalise onto the simplex
        s = w.sum()
        if s == 0.0:
            w.fill(1.0 / n)
        else:
            w /= s

        # periodic symmetrisation
        if iteration % sym_interval == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])

        # handle long stagnation – jump to a perturbed best
        if no_improve > stagnation_limit:
            w = best_local_w.copy()
            # a cheap random block is enough to escape a flat region
            block_width = int(rng.integers(max(1, n // 250), max(2, n // 30)))
            start = int(rng.integers(0, n - block_width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            w[start:start + block_width] *= factor
            w = np.clip(w, 0.0, 1000.0)
            w /= w.sum()
            no_improve = 0
            print(f"[reset] it={iteration:06d}, random block jump")

        iteration += 1
        # safety break if we run out of time
        if time.monotonic() - start_time >= budget_s:
            break

    # --------------------------------------------------------------
    # Assemble final result – scale to the conventional √(2 n) norm.
    # --------------------------------------------------------------
    final_w = best_local_w      # best vector found (probability simplex)
    target_sum = np.sqrt(2.0 * n)
    final_seq = final_w * target_sum
    final_seq = np.clip(final_seq, 0.0, 1000.0)   # obey the evaluator's caps

    # sanity check (silently ignored if the function is not imported)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best max={best_local_val:.6f}, evaluate={final_score:.6f}")
    except Exception:
        print(f"[final] length={n}, best max={best_local_val:.6f}")

    # Return plain Python list as required by the harness
    return [float(x) for x in final_seq]
