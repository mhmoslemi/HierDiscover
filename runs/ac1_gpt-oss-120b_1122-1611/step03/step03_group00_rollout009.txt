"""
Propose a candidate step‑function via full‑gradient mirror descent.
The algorithm works directly on the probability simplex (a_i ≥ 0, Σa_i = 1)
and minimises the smooth surrogate
    F(a) = log‑sum‑exp( β·(a∗a – max(a∗a)) )
using an exponentiated‑gradient (mirror descent) update.
A full gradient is obtained by a single FFT‑based correlation, i.e.

    ∇F(a)_j = 2·∑_k w_k·a_{k−j}   with   w_k ∝ exp(β·(conv_k−max_conv))

β is increased linearly (soft‑max becomes sharper) while the stepsize η
is decreased linearly.  Occasional random block perturbations,
pairwise mass transfers and soft restarts prevent stagnation.
The best sequence seen during the 1000 s budget is returned.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n`` (i.e. length 2·n‑1).
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative real vector using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the soft‑max surrogate:
        grad_j = 2·∑_k w_k·a_{k−j}
    where w has length 2·n‑1 and a has length n.
    The operation is a linear correlation, performed via FFT.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out
