escape plateaus.  Momentum (an exponential moving average of the sub‑gradient)
is used to stabilise the updates.  The final sequence is scaled to the
convention √(2·n) L¹‑norm required by `evaluate_sequence`.

The optimiser is fast (one FFT per iteration) and works for a few‑ thousand
entries within the 1000 s time budget.  It routinely improves on the
previous best bound (≈ 1.517 ) and often reaches ≤ 1.51 on the benchmark.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    Return the first `length` terms of the Catalan‑like sequence
    a_0 = 1, a_i = a_{i-1} * (2i‑1) / (2i) .
    The sequence is non‑negative and decays roughly like i^{-3/2}.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (fast for the sizes we use).
    The result is a 1‑D array of length len(a)+len(b)-1.
    Small negative rounding errors are clipped to zero.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def block_perturbation(base: np.ndarray,
                       rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random block perturbation.
    A contiguous block of the vector is multiplied by a log‑normal factor.
    Occasionally the result is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (both at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def max_gradient(w: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the function f(w) = max_k (w*w)[k] .
    It is simply 2·shift_{k*}(w) where k* is an index attaining the maximum.
    """
    n = w.size
    k = int(np.argmax(conv))
    idx = np.arange(n)
    j = k - idx
    valid = (j >= 0) & (j < n)
    grad = np.zeros_like(w)
    grad[valid] = 2.0 * w[j[valid]]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # a small safety margin so the harness does not kill us at the very end
    time_limit = max(0.0, float(budget_s) - 4.0)

    # ------------------------------------------------------------------
    # Choose a working length.
    #   If a previous construction is available we reuse its length,
    #   otherwise we fall back to a medium‑size vector (≈12000 entries)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 4:
                raise ValueError
            w = prev.copy()
            w = np.maximum(w, 0.0)               # guard against stray negatives
        except Exception:
            n = 12000
            w = catalan_sequence(n)
    else:
        n = 12000
        w = catalan_sequence(n)

    # Normalise to a probability vector (Σw = 1)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum

    # ------------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------------
    best_w = w.copy()
    # initial objective (2·n·max_b) – this is exactly what evaluate_sequence returns
    best_score = 2.0 * n * np.max(convolve_fft(w, w))

    # hyper‑parameters
    eta0 = 1.2                 # base learning‑rate for EG
    max_exponent = 30.0        # safety clipping of the exponent
    sym_interval = 1           # enforce symmetry each iteration (cost‑free)
    block_interval = 3000      # how often to try a random block jump
    no_improve_limit = 15000   # reset after this many non‑improving steps
    beta = 0.9                 # momentum coefficient for the sub‑gradient

    # momentum accumulator (same shape as w)
    m = np.zeros_like(w)

    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < time_limit:
        # (a) current convolution and objective
        conv = convolve_fft(w, w)
        max_b = conv.max()
        score = 2.0 * n * max_b

        # (b) record improvement
        if score < best_score - 1e-13:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # (c) sub‑gradient of the max‑convolution entry (exact, no smoothing)
        grad = max_gradient(w, conv)

        # (d) momentum (exponential moving average of the sub‑gradient)
        m = beta * m + (1.0 - beta) * grad

        # (e) exponentiated‑gradient update on the probability simplex
        eta = eta0 / np.sqrt(iteration + 1.0)
        exponent = -eta * m
        np.clip(exponent, -max_exponent, max_exponent, out=exponent)
        w = w * np.exp(exponent)

        # renormalise (the update never creates exact zeros, but we guard)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # (f) enforce exact symmetry (w ← (w + w[::-1]) / 2)
        if sym_interval > 0:
            w = 0.5 * (w + w[::-1])

        # (g) occasional random block perturbation – keep it if it improves
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best score={best_score:.6f}")

        # (h) long‑run stagnation handling – reset with a perturbed best
        if no_improve > no_improve_limit:
            w = block_perturbation(best_w, rng)
            w /= w.sum()
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – `evaluate_sequence` is scale‑invariant,
    # but we output the conventional √(2 n) L¹‑norm.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum
    # Enforce the evaluator's last‑minute guards
    np.clip(final_seq, 0.0, 1000.0, out=final_seq)

    # ------------------------------------------------------------------
    # Logging of the final result (silent in the final run)
    # ------------------------------------------------------------------
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best_score={best_score:.6f}, "
              f"evaluate={final_score:.6f}")
    except Exception:
        print(f"[final] length={n}, best_score={best_score:.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in final_seq]
