"""
LP‑based minimisation of the peak of self‑convolution.

The objective to be minimised is
    score(w) = 2·n·max(conv(w,w)),
with w ≥ 0 and Σ w = 1 (the final scaling to √(2 n) is done at the end).
For a fixed w we compute M = max(conv(w,w)).  The convex programme

    maximise    Σ g_i
    subject to  conv(w,g) ≤ M   (2 n‑1 linear inequalities)
                g_i ≥ 0

finds a vector g that spreads the mass as much as possible while keeping the
convolution with w bounded by the current peak M.  A small convex combination

    w_new = (1‑t)·w + t·g_norm,   g_norm = g / Σg

strictly reduces (or at least does not increase) the peak.  Repeating the
LP‑step yields a monotone descent.  The implementation uses SciPy’s HiGHS
linear‑program solver on a sparse constraint matrix (the pattern of the
convolution matrix does not change, only the non‑zero values do).  A final
projection onto the probability simplex guarantees feasibility after each
update.

The routine runs for at most ``budget_s`` seconds, retains the best vector
found, rescales it to the required ℓ₁‑norm √(2 n) and returns it as a plain
Python list.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities – all defined at module level.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan‑type coefficients a_i = C(2i,i)/4**i (all positive)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (full) convolution via real FFT (fast for large vectors)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Numerical noise may create tiny negatives; clip to zero.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(w: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(w,w)) for a vector w with Σw = 1."""
    b = convolve_fft(w, w)
    return 2.0 * n * np.max(b)

            w = prev[:n].copy()
            w = np.maximum(w, 0.0)
            if w.sum() < 1e-12:
                raise ValueError
            w = w / w.sum()
        except Exception:
            n = DEFAULT_N
            w = catalan_sequence(n)
            w = w / w.sum()
    else:
        n = DEFAULT_N
        w = catalan_sequence(n)
        w = w / w.sum()

    # ------------------------------------------------------------------
    # Pre‑compute the sparsity pattern of the convolution matrix:
    #   (conv(w,g))[k] = Σ_i w_i·g_{k-i}
    #   => row k has entries w_i at column j = k-i.
    # The pattern (row, col) does not depend on w, only the values do.
    # ------------------------------------------------------------------
    m = 2 * n - 1                  # length of the linear convolution
    row_idx = []
    col_idx = []
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i runs over valid indices of w; column = j = k-i.
        for i in range(i_min, i_max + 1):
            row_idx.append(k)
            col_idx.append(k - i)   # j = k-i
    row_idx = np.array(row_idx, dtype=np.int32)
    col_idx = np.array(col_idx, dtype=np.int32)
    # the corresponding w‑index for each non‑zero entry
    idx_i = row_idx - col_idx       # always in 0 … n‑1

    # ------------------------------------------------------------------
    # State for the optimiser.
    # ------------------------------------------------------------------
    best_w = w.copy()
    best_score = compute_score(best_w, n)
    print(f"[init] n={n}, score={best_score:.6f}")

    start_time = time.monotonic()
    max_time = max(0.0, budget_s - 5.0)   # keep a few seconds for finalising
    iteration = 0
    t = 0.3               # interpolation weight (will be shrunk gradually)
    no_improve = 0

    while time.monotonic() - start_time < max_time and iteration < 30:
        # ------------------- current peak ---------------------------
        b = convolve_fft(w, w)
        M = np.max(b)

        # ------------------- build LP matrix ------------------------
        # each non‑zero gets the current coefficient w_i where i = row‑col.
        A_data = w[idx_i]                     # shape (nnz,)
        A_ub = sp.csr_matrix((A_data, (row_idx, col_idx)), shape=(m, n))

        # ------------------- linear programme -----------------------
        # maximise Σ g_i  <=> minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        rhs = np.full(m, M, dtype=np.float64)

        # Optional time limit for the solver (remaining budget)
        remaining = max_time - (time.monotonic() - start_time)
        lp_opts = {"presolve": True}
        if remaining > 0:
            lp_opts["time_limit"] = remaining

        res = opt.linprog(
            c,
            A_ub=A_ub,
            b_ub=rhs,
            bounds=bounds,
            method="highs",
            options=lp_opts,
        )

        if not res.success:
            # If the LP fails we fall back to a simple gradient step.
            print(f"[lp] iteration {iteration}: failed ({res.message}), "
                  "switching to gradient fallback.")
            # ----- simple sub‑gradient step on the current max ------------
            max_k = int(np.argmax(b))
            # sub‑gradient of max_k entry: 2 * (shifted w)[:n]
            grad = 2.0 * np.zeros_like(w)
            # contribution for indices where both i and max_k‑i are in range
            i_start = max(0, max_k - (n - 1))
            i_end = min(n - 1, max_k)
            # grad[j] += w[max_k - j] for each feasible j
            for j in range(i_start, i_end + 1):
                grad[j] = w[max_k - j] * 2.0
            # step size heuristics
            eta = 0.5 / (iteration + 1.0)
            w_candidate = w - eta * grad
            w = simplex_projection(w_candidate)
            iteration += 1
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("[lp] sum(g) non‑positive, aborting.")
            break
        g_norm = g / sum_g

        # ------- convex combination (small step) -------------------
        w_new = (1.0 - t) * w + t * g_norm
        w_new = simplex_projection(w_new)

        new_score = compute_score(w_new, n)

        if new_score < best_score - 1e-9:
            best_score = new_score
            best_w = w_new.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:3d}, score={best_score:.6f}")
            # slowly reduce the step size as we approach a plateau
            t = max(0.05, t * 0.9)
        else:
            no_improve += 1
            # shrink t more aggressively if we are stuck
            t = max(0.05, t * 0.85)

        # occasional random jump to avoid stagnation
        if no_improve >= 5:
            perturb = best_w * np.exp(rng.normal(0.0, 0.15, size=n))
            perturb = np.clip(perturb, 0.0, 1000.0)
            perturb = simplex_projection(perturb)
            w = perturb
            no_improve = 0
            print(f"[reset] iter={iteration:3d}")
        else:
            w = w_new

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling to the conventional ℓ₁‑norm √(2 n) required by the
    # evaluation harness.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score={best_score:.6f}, "
          f"evaluate={final_score:.6f}")

    # Return a plain Python list.
    return [float(x) for x in final_seq]
