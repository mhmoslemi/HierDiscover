"""
propose_candidate
-----------------
Implements a fast gradient–based search for a probability vector `a`
(minimum L∞ self‑convolution) using a smooth max approximation.
The algorithm works on the simplex by exponentiated‑gradient updates
and a temperature schedule that sharpens from a soft‑max to the hard
maximum.  A full‑gradient is obtained efficiently via FFT (no O(n²)
loops).  Random block jitter, occasional hard “pair‑wise mass transfer”
moves and a simple restart strategy prevent stagnation.
The routine respects the time‑budget (default 1000 s) and returns the
best coefficient list found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may introduce tiny negative values – clip.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of  Σ_k w_k·(a*a)_k  w.r.t. a.
    The formula is   ∂/∂a_j  Σ_k w_k·conv_k = 2· Σ_i a_i·w_{i+j}
    which is a cross‑correlation between a and w.
    This implementation uses a single FFT‑based convolution.
    """
    n = a.size
    Lw = w.size                         # = 2*n-1
    # reverse w for correlation via convolution
    wr = w[::-1]

    # length required for a (len=n) + wr (len=Lw) - 1 = n+Lw-1 = 3n-2
    conv_len = _next_pow2(n + Lw - 1)

    Fa = np.fft.rfft(a, n=conv_len)
    Fwr = np.fft.rfft(wr, n=conv_len)

    # full linear convolution (real part)
    conv_full = np.fft.irfft(Fa * Fwr, n=conv_len)
    conv_full = conv_full[: n + Lw - 1]   # keep only needed part

    # gradient_j = 2 * conv_full[Lw-1 - j]   (see derivation in analysis)
    idx = Lw - 1 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a multiplicative log‑normal perturbation to a random contiguous
    block of the vector, then renormalise onto the simplex.
    """
    n = a.size
    # block width at most 20 % of the vector, at least 1 element
    max_width = max(1, n // 5)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)

    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_transfer(a: np.ndarray, conv: np.ndarray, rng: np.random.Generator):
    """
    Attempt a cheap pairwise mass transfer.
    Returns a new (a, conv, score) tuple if the move improves the
    current objective, otherwise the original tuple.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, conv, None

    # move at most 30 % of donor's mass
    max_delta = min(a[i], 0.3 * a[i])
    if max_delta <= 0.0:
        return a, conv, None
    delta = rng.random() * max_delta

    # contribution vectors for indices i and j (length 2n‑1)
    # Instead of constructing them fully we update conv using the
    # linear relationship: new_conv = conv + delta * (c_j - c_i)
    n2 = 2 * n - 1
    # c_i[k] = a_{k-i}   for k in [i, i+n-1] otherwise 0
    # c_j similarly.  We'll add/subtract delta efficiently.
    new_conv = conv.copy()
    # range where c_i contributes
    lo_i = i
    hi_i = i + n
    # range where c_j contributes
    lo_j = j
    hi_j = j + n

    # The overlapping indices are handled automatically by slicing.
    new_conv[lo_i:hi_i] -= delta * a         # subtract donor's contribution
    new_conv[lo_j:hi_j] += delta * a         # add to receiver

    # compute new score from conv (sum(a) stays 1)
    new_score = 2.0 * n * np.max(new_conv)
    cur_score = 2.0 * n * np.max(conv)

    if new_score < cur_score:
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.clip(a2, 0.0, None)
        a2 /= a2.sum()
        return a2, new_conv, new_score
    else:
        return a, conv, None

def propose_candidate(seed: int = 1234, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function minimizing
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine problem size and initial vector
    # ------------------------------------------------------------------
    # If a previous construction exists we reuse its length; otherwise
    # we start from an ample size (≈4000) because larger n gives more
    # room for minimising the max‑convolution.
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = cand0.size
            a = np.clip(cand0, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            # fallback
            n = 4096
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # FFT lengths (powers of two) for the two convolutions we need
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for correlation a ⋆ w

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _self_conv_fft(best, fft_len_self)
    best_score = 2.0 * n * conv_best.max()      # sum(best) == 1

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # leave small margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (tuned for this budget)
    # ------------------------------------------------------------------
    eta_start = 0.9          # initial learning rate (exponentiated‑gradient)
    eta_end = 0.02           # final learning rate
    beta_start = 0.5         # start temperature (soft‑max softness)
    beta_end = 2000.0        # final temperature – almost hard max
    momentum = 0.85          # exponential smoothing of the gradient
    block_perturb_prob = 0.025
    restart_gap = 30.0       # seconds without improvement → restart
    hard_max_phase = 0.85    # fraction of time after which we switch to hard max

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current self‑convolution (FFT) -----
        conv = _self_conv_fft(a, fft_len_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ----- keep best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule for learning rate & temperature -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- construct weight vector w (soft‑max or hard max) -----
        if frac < hard_max_phase:
            # soft‑max on the full convolution vector
            w = np.exp(beta * (conv - cur_max))          # stable: max exponent = 0
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0                     # exact sub‑gradient

        # ----- gradient of the weighted max (full‑gradient) -----
        grad = _gradient_from_weights(a, w)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror descent) update -----
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- cheap pairwise mass transfer (optional) -----
        if rng.random() < 0.07:
            a, conv, _ = _pairwise_transfer(a, conv, rng)

        # ----- restart when stuck -----
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.14, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
