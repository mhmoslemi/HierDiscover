def catalan_seq(n):
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1,n):
        seq[i] = seq[i-1] * (2*(i-1)+1) / (2*i)
    return seq
```

```
def power_law_seq(n,p):
    i = np.arange(n)
    seq = 1.0 / (i+1)**p
    return seq
```

```
def beta_seq(n,alpha):
    x = (np.arange(n) + 0.5) / n
    # add epsilon for boundaries
    eps = 1e-12
    x = np.clip(x, eps, 1-eps)
    seq = (x**(alpha-1)) * ((1-x)**(alpha-1))
    return seq
```

```
def gaussian_seq(n,sigma):
    i = np.arange(n)
    center = (n-1)/2
    seq = np.exp(-0.5 *((i - center)/sigma)**2)
    return seq
```

```
def exponential_seq(n,rate):
    i = np.arange(n)
    seq = np.exp(-rate*i)
    return seq
```

```
def normalize(seq):
    s = np.sum(seq)
    if s <= 0:
        return seq
    return seq / s

def approx_score(seq):
    n = seq.shape[0]
    conv = convolve_fft(seq, seq)
    maxb = np.max(conv)
    sum_a = np.sum(seq)
    return float(2 * n * maxb / (sum_a**2))

def refine(seq, steps=2000):
    n = len(seq)
    w = seq.copy()
    w_prev = w.copy()
    eta0 = 0.5
    momentum=0.8
    for it in range(steps):
        conv = convolve_fft(w,w)
        max_val = conv.max()
        idx_candidates = np.flatnonzero(conv >= max_val - 1e-12)
        i_max = int(rng.choice(idx_candidates))
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n-1))
        j_end = min(n-1, i_max)
        j = np.arange(j_start, j_end+1)
        idx = i_max - j
        grad[j] = 2.0*w[idx]
        eta = eta0 / np.sqrt(it+1)
        w_temp = w - eta*grad + momentum*(w-w_prev)
        w_new = simplex_projection(w_temp)
        # enforce nonneg and small bound

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm works in three stages:

1. **Coarse grid search** –  a family of analytically‑motivated
   shapes (Catalan, power‑law, Beta, Gaussian, exponential and random
   Dirichlet) is generated for several sequence lengths (≈200‑8000).
   Each candidate is normalised to unit ℓ¹‑norm and scored with a fast
   FFT‑based convolution (the score is invariant under scaling).

2. **Local refinement** – the best candidate from the grid is polished
   by a few iterations of projected sub‑gradient descent on the
   non‑smooth objective 2·n·max((w∗w))/‖w‖₁².  The gradient is taken at
   an active index of the convolution, a Nesterov‑style momentum step
   is applied and the vector is projected onto the probability simplex.

3. **Final scaling & verification** – the refined vector is handed to the
   provided `evaluate_sequence` (which performs the required clipping)
   and returned as a plain Python list.

All helper functions are defined at module level; the routine respects
the 1000 s time budget and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_seq(length: int) -> np.ndarray:
    """Catalan numbers a_i = C(2i,i)/4**i for i=0,…,length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def power_law_seq(length: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{-p}."""
    idx = np.arange(length, dtype=np.float64) + 1.0
    return 1.0 / np.power(idx, p)


def beta_seq(length: int, alpha: float) -> np.ndarray:
    """Discrete Beta(α,α) density:  x^{α‑1}(1‑x)^{α‑1},  x∈(0,1)."""
    x = (np.arange(length, dtype=np.float64) + 0.5) / length
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    return np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)


def gaussian_seq(length: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian shape centred in the middle of the interval."""
    i = np.arange(length, dtype=np.float64)
    centre = (length - 1) / 2.0
    return np.exp(-0.5 * ((i - centre) / sigma) ** 2)


def exponential_seq(length: int, rate: float) -> np.ndarray:
    """Monotonically decreasing exponential."""
    i = np.arange(length, dtype=np.float64)
    return np.exp(-rate * i)


def normalize(v: np.ndarray) -> np.ndarray:
    """Scale to ℓ¹‑norm = 1 (leaves the zero vector untouched)."""
    s = np.sum(v)
    if s <= 0.0:
        return v
    return v / s


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n up to a few 10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.ifft(Fc, N).real[:n]
    np.maximum(c, 0.0, out=c)               # eliminate tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Large random O(n) perturbation – useful for escaping plateaus."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:      # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative floats whose `evaluate_sequence` score
    is (hopefully) below the target 1.5025.  The routine respects a
    ~1000 s time budget, searches several parametric families, performs a
    short projected sub‑gradient polish and finally returns the best
    construction found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    best_seq = None
    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # 1.  Choice of lengths to probe
    # ------------------------------------------------------------------
    lengths = [200, 400, 600, 800, 1000, 1500, 2000,
               3000, 4000, 6000, 8000]

    # If a previous construction is supplied, also try its length.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size not in lengths:
            lengths.append(prev.size)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2.  Grid search over shape families and parameters
    # ------------------------------------------------------------------
    for n in lengths:
        # keep an eye on the remaining time – leave ~30 s for polishing
        if time.monotonic() - start_time > budget_s - 30.0:
            break

        candidates = []

        # Catalan
        candidates.append(normalize(catalan_seq(n)))

        # Discrete arcsine (Beta with α=0.5)
        candidates.append(normalize(beta_seq(n, 0.5)))

        # Power‑law family
        for p in (0.4, 0.5, 0.6, 0.8, 1.0):
            candidates.append(normalize(power_law_seq(n, p)))

        # Beta family (α∈[0.3,1.0])
        for a in (0.3, 0.4, 0.5, 0.6, 0.8, 1.0):
            candidates.append(normalize(beta_seq(n, a)))

        # Gaussian bells
        for sigma in (n / 8.0, n / 6.0, n / 4.0):
            candidates.append(normalize(gaussian_seq(n, sigma)))

        # Simple exponentials
        for rate in (0.001, 0.002, 0.004, 0.008):
            candidates.append(normalize(exponential_seq(n, rate)))

        # Random Dirichlet (α=0.5) – two independent draws
        for _ in range(2):
            cand = rng.gamma(0.5, 1.0, size=n)
            candidates.append(normalize(cand))

        # Symmetrised copies (helpful for convolution)
        for seq in list(candidates):
            sym = 0.5 * (seq + seq[::-1])
            candidates.append(normalize(sym))

        # ------------------------------------------------------------------
        #   fast evaluation (uses FFT, no clipping – fine for our scales)
        # ------------------------------------------------------------------
        for seq in candidates:
            conv = convolve_fft(seq, seq)
            max_b = conv.max()
            s = np.sum(seq)
            score = float(2.0 * n * max_b / (s * s))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_n = n
                print(f"[grid] n={n} → new best approx score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2b.  If nothing was found (should never happen) fall back to Catalan
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = normalize(catalan_seq(6000))
        best_n = 6000
        best_score = float('inf')   # will be overwritten later

    # ------------------------------------------------------------------
    # 3.  Local refinement with projected sub‑gradient descent
    # ------------------------------------------------------------------
    # Estimate the remaining time and allocate a modest number of steps.
    remaining = budget_s - (time.monotonic() - start_time)
    # about 150–200 iterations per second on a typical CPU
    max_iterations = max(0, int(remaining * 180.0))

    if max_iterations > 0:
        w = best_seq.copy()
        w_prev = w.copy()
        eta0 = 0.6
        momentum = 0.8
        n = w.size

        for it in range(max_iterations):
            # convolution and active index
            conv = convolve_fft(w, w)
            max_val = conv.max()
            idx_candidates = np.flatnonzero(conv >= max_val - 1e-15)
            i_max = int(rng.choice(idx_candidates))

            # sub‑gradient of the active quadratic form
            grad = np.zeros_like(w)
            j_start = max(0, i_max - (n - 1))
            j_end = min(n - 1, i_max)
            j = np.arange(j_start, j_end + 1, dtype=int)
            idx = i_max - j
            grad[j] = 2.0 * w[idx]

            # step size with a basic schedule
            eta = eta0 / np.sqrt(it + 1)

            # Nesterov‑style momentum + projection onto the simplex
            w_temp = w - eta * grad + momentum * (w - w_prev)
            w_new = simplex_projection(w_temp)

            w_prev = w
            w = w_new

            # occasional aggressive perturbation to avoid stagnation
            if it % 1000 == 0 and it > 0:
                cand = block_perturbation(w, rng)
                cand = cand / cand.sum()
                # cheap score check
                cand_score = float(2.0 * n *
                                   convolve_fft(cand, cand).max() /
                                   (cand.sum() ** 2))
                if cand_score < best_score:
                    best_seq = cand.copy()
                    best_score = cand_score
                    w = cand.copy()
                    w_prev = w.copy()
                    print(f"[perturb] iter={it}, improved approx score={best_score:.6f}")

        # store refined candidate
        best_seq = w.copy()
        # re‑evaluate with the fast metric (for printing only)
        best_score = float(2.0 * best_n *
                           convolve_fft(best_seq, best_seq).max() /
                           (best_seq.sum() ** 2))

    # 4.  Final evaluation with the official routine and return result

    # The score is scale‑invariant, we keep the ℓ¹‑norm = 1.
    final_seq = best_seq.copy()
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={best_n}, final approx score={best_score:.6f}, "
          f"official evaluate={final_score:.6f}")
