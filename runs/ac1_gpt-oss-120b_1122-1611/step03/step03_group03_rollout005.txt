def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    # Setup RNG
    rng = np.random.default_rng(seed)

    # Determine n, initial w
    if 'height_sequence_1' in globals():
        try:
            init_arr = np.asarray(height_sequence_1, dtype=np.float64)
            n = init_arr.size
            if n < 2:
                raise ValueError
        except Exception:
            n = 8000
            init_arr = None
    else:
        n = 8000
        init_arr = None

    if init_arr is not None:
        # ensure non-negative
        w = np.maximum(init_arr, 0.0)
        # symmetrize? optional
    else:
        # use Catalan sequence
        w = catalan_sequence(n)

    # normalize to sum=1
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full(n, 1.0/n)
    else:
        w = w / w_sum

    # ensure symmetry
    w = 0.5*(w + w[::-1])

    # track best
    conv = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(conv)
    best_w = w.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Adam parameters
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12   # maybe smaller.

    # other hyperparameters
    eta0 = 1.0   # for step size schedule (maybe combine with Adam)
    iter = 0
    no_improve = 0
    start_time = time.monotonic()
    max_iters = 10**9

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0) and iter < max_iters:
        # compute conv
        conv = convolve_fft(w, w)
        max_val = np.max(conv)
        score = 2.0 * n * max_val

        # update best
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iter:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # gradient of max (subgradient)
        tol = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - tol).astype(np.float64)
        # correlation
        corr = convolve_fft(mask, w[::-1])  # length 3n-2
        grad = 2.0 * corr[n-1:n-1+n]  # length n

        # Adam update (descent)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad*grad)
        t = iter + 1
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        lr = lr0 / np.sqrt(t)  # decays
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step

        # projection onto simplex (ensuring nonneg)
        w = simplex_projection(w_new)

        # enforce symmetry
        w = 0.5 * (w + w[::-1])
        # optionally renorm after symmetrization (but already sum 1 due to linear)
        # Since symmetrizing may alter sum: compute sum and renorm.
        w_sum = np.sum(w)
        if w_sum != 0:
            w = w / w_sum
        else:
            # fallback uniform
            w = np.full(n, 1.0/n)

        # occasional block perturbation
        if iter % 1500 == 0 and iter > 0:
            cand = block_perturbation(best_w, rng)
            # re-symmetrize
            cand = 0.5 * (cand + cand[::-1])
            cand = cand / np.sum(cand)
            cand_conv = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(cand_conv)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[block] iter={iter:6d}, new best={best_score:.6f}")

        # restart if no improvement long
        if no_improve > 50000:
            w = block_perturbation(best_w, rng)
            w = 0.5 * (w + w[::-1])
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iter:6d}, reset to perturbed best")

        iter += 1

    # final sequence scaling
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]

"""
propose_candidate
-----------------
Hybrid projected‑Adam optimizer that works on the exact sub‑gradient of the
L∞‑norm of the self‑convolution.  The objective is

    score(w) = 2·n·max(conv(w,w)),   w ≥ 0, Σw = 1.

The gradient of the (nonsmooth) max is obtained analytically:
if M = max(conv) and 𝕀 = {k : conv_k = M},
    ∇_w score = 2·∑_{k∈𝕀} (mask_k ⋆ w),

where “⋆” denotes discrete correlation.  Correlation is evaluated with a
real‑FFT based convolution (O(n log n)).  Adam’s first‑ and second‑moment
estimates are updated with this sub‑gradient and the iterate is projected
onto the probability simplex.  Every few thousand iterations a random
block perturbation (multiplying a contiguous block by a log‑normal factor)
is tried; if it improves the true objective the current state is reset to
that perturbed best point.  The iterate is kept symmetric (w = w[::-1])
because the optimal step function is known to be even, which reduces the
search space and stabilises the convolution.

The best vector found is finally rescaled to the conventional √(2 n) L¹‑norm
required by the evaluation harness and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic signal utilities
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT, with a tiny negative‑value guard."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # clip rounding artefacts
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are the same (or negative) → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly select a contiguous block, multiply it by a log‑normal factor,
    optionally enforce symmetry, and clip to the allowed range.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps to stay in the even subspace)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def catalan_sequence(length: int) -> np.ndarray:
    """
    Catalan numbers divided by 4ⁱ, i.e. a_i = C(2i,i)/4ⁱ.
    They form a smooth decreasing shape that is a good generic start.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def half_sine_sequence(length: int) -> np.ndarray:
    """
    a_i = sin(π·(i+1)/(length+1)), a simple even‑ish shape.
    """
    i = np.arange(1, length + 1, dtype=np.float64)
    return np.sin(np.pi * i / (length + 1))
