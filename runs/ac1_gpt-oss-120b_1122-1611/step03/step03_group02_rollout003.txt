def propose_candidate(seed=0,budget_s=1100, **kwargs):
    import time, numpy as np
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    best_score = float('inf')
    best_seq = None
    restarts = 0
    while time.monotonic() - start < budget_s - 10 and restarts < 10:
        # Choose n randomly between 5000 and 12000
        n = int(rng.integers(5000,12001))
        # initialize v: zeros or random
        # Use either central binomial shape:
        if restarts == 0 and 'height_sequence_1' in globals():
            # Use previous best if available
            base_seq = np.asarray(height_sequence_1, dtype=np.float64)
            if base_seq.size > 0:
                n = base_seq.size
                w0 = np.maximum(base_seq, 0)
                w0 = w0 / np.sum(w0)
                v = np.log(w0+1e-12)  # initial logits
        else:
            # random Dirichlet
            w0 = rng.dirichlet(np.ones(n))
            v = np.log(w0+1e-12)
        # Initialize Adam parameters m,vh = zeros
        m = np.zeros(n)
        vh = np.zeros(n)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr0 = 0.5
        mu_factor = 0.8
        mu_factor_min = 0.02
        mu_decay = 0.9997
        max_iter = 200000  # maybe limited by time
        iter = 0
        while True:
            # check time
            if time.monotonic() - start > budget_s - 5:
                break
            # compute w
            # Using softmax:
            maxv = np.max(v)
            expv = np.exp(v - maxv)
            w = expv / np.sum(expv)
            # compute convolution
            b = convolve_fft(w, w)
            maxb = np.max(b)
            score = 2.0 * n * maxb
            # record best
            if score < best_score:
                best_score = score
                best_seq = w.copy()
                best_n = n
                # optionally record iteration and time
                print(f"[restart {restarts}] new global best: n={n}, iter={iter}, score={score:.6f}")
            # compute p (smooth softmax of b)
            mu = max(maxb * mu_factor, 1e-12)
            shifted = (b - maxb) / mu
            # To avoid overflow, maybe clip shifted to -30 to 0
            shifted = np.clip(shifted, -30, 0)
            exp_shift = np.exp(shifted)
            p = exp_shift / np.sum(exp_shift)
            # gradient w.r.t. w
            grad_w = 2.0 * convolve_fft(p, w[::-1])[n-1:n-1+n]
            # gradient w.r.t. v (chain rule)
            dot = np.dot(w, grad_w)
            grad_v = w * (grad_w - dot)
            # Adam update
            iter += 1
            beta1_t = beta1 ** iter
            beta2_t = beta2 ** iter
            m = beta1 * m + (1 - beta1) * grad_v
            vh = beta2 * vh + (1 - beta2) * (grad_v * grad_v)
            m_hat = m / (1 - beta1_t)
            vh_hat = vh / (1 - beta2_t)
            lr = lr0 / np.sqrt(iter + 1)
            v -= lr * m_hat / (np.sqrt(vh_hat) + eps)
            # mu schedule
            mu_factor = max(mu_factor * mu_decay, mu_factor_min)
            # occasional random perturbation
            if iter % 3000 == 0:
                # produce a perturbed version of best_seq (global best) possibly
                cand = best_seq.copy()
                # random block scaling
                width = int(rng.integers(max(1, n // 200), max(2, n // 12) + 1))
                start_idx = int(rng.integers(0, max(1, n - width + 1)))
                factor = np.exp(rng.normal(0.0, 0.15))
                cand[start_idx:start_idx+width] *= factor
                # keep positivity
                cand = np.clip(cand, 0.0, 1000.0)
                # renormalize
                cand = cand / np.sum(cand)
                # set v accordingly
                v = np.log(cand + 1e-12)
                m.fill(0.0)
                vh.fill(0.0)
            # maybe break if iter large
            if iter >= max_iter:
                break
        restarts += 1
    # final scaling of best_seq to sqrt(2n)
    if best_seq is None:
        # fallback to simple uniform?
        n = 5000
        best_seq = np.full(n, 1.0 / n)
        best_n = n
    target_sum = np.sqrt(2.0 * best_n)

    n = a.size + b.size - 1

        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v * grad_v)
        # bias correction
        m_hat = m / (1 - beta1**iteration)
        v_hat_corr = v_hat / (1 - beta2**iteration)
        # step size
        lr = lr0 / np.sqrt(iteration)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)
        # decay mu factor
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)
        # occasional random block perturbation
        if iteration % 2500 == 0:
            # apply perturbation to best_w as a jump
            cand = best_w.copy()
            lo = max(1, n // 200)
            hi = max(2, n // 12)
            width = int(rng.integers(lo, hi + 1))
            start_idx = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.18))
            cand[start_idx:start_idx + width] *= factor
            # optional symmetrization
            if rng.random() < 0.4:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand / np.sum(cand)
            # reset logits from candidate
            v = np.log(cand + 1e-12)
            m.fill(0.0)
            v_hat.fill(0.0)
            # evaluate its score
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                print(f"[block] iter={iteration}, new best={best_score:.6f}")
    # subgradient refinement
    w = best_w.copy()
    sub_iter = 0
    lr0_sub = 0.5
    while time.monotonic() - start < total_budget - 2.0 and sub_iter < 20000:
        b = convolve_fft(w, w)
        maxb = np.max(b)
        # get indices near max
        idxs = np.flatnonzero(b >= maxb - 1e-14)
        i_max = int(rng.choice(idxs))
        # compute subgradient for active index
        grad_sub = np.zeros_like(w)
        # indices j where 0 <= j <= n-1 and 0 <= i_max - j <= n-1
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        # vectorized fill
        j_range = np.arange(j_start, j_end + 1, dtype=int)
        grad_sub[j_range] = 2.0 * w[i_max - j_range]
        # step size
        eta = lr0_sub / np.sqrt(sub_iter + 1.0)
        # update and project onto simplex
        w = w - eta * grad_sub
        # simplex projection
        # using simple Euclidean projection
        w = simplex_projection(w)
        # evaluate score
        score = 2.0 * n * np.max(convolve_fft(w, w))
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[sub‑improve] sub_iter={sub_iter}, score={best_score:.6f}")
        sub_iter += 1
    # final scaling
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score={best_score:.6f}, final evaluate={final_score:.6f}")
    # store global for possible reuse
    globals()['height_sequence_1'] = final_seq.tolist()
    return [float(x) for x in final_seq]

"""
Propose a low‑score step‑function.

The algorithm works in two phases.

1.  Smooth‑max Adam on the probability simplex.
    The vector `v` holds logits; `w = softmax(v)` is a non‑negative
    sequence with ∑w = 1.  At each iteration we compute the self‑convolution
    `b = w * w`, form a smooth approximation of max(b) using a
    temperature μ = max(b)·μ_factor and a softmax weight `p`.  The gradient
    of the smooth objective is 2·conv(p, rev(w)).  Adam updates are applied
    to the logits `v`.  μ_factor is slowly decayed, the step‑size follows
    η ∝ 1/√t and occasional random block perturbations are used to escape
    plateaus.

2.  Sub‑gradient cleanup.
    After the smooth phase a pure sub‑gradient method for the true
    L∞‑norm (max of the convolution) refines the best candidate found.
    For the active index of the peak we compute the exact sub‑gradient,
    take a diminishing step and project back onto the simplex.

The best sequence is finally rescaled to the conventional ℓ₁‑norm
√(2 n) (the evaluation function is scale‑invariant) and returned.
The resulting sequence is stored in the global variable
`height_sequence_1` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – defined at top level.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    maxv = np.max(v)
    e = np.exp(v - maxv)
    return e / np.sum(e)


def block_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block scaling, used as an escape jump."""
    n = seq.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.18)))
    cand = seq.copy()
    cand[start:start + width] *= factor
    if rng.random() < 0.4:                     # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= np.sum(cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is (hopefully) better than the previous best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = float(budget_s)

    # --------------------------------------------------------------
    # Choose length and initial sequence.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = base.size
            if n < 2:
                raise ValueError
            init_w = np.maximum(base, 0.0)
            init_w = init_w / np.sum(init_w)
        except Exception:                          # fallback
            base = None
    if "base" not in locals():
        # moderate size, power‑of‑two for fast FFT
        n = 8192
        # central‑binomial (Catalan) distribution – a good starting shape
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
        init_w = a / np.sum(a)

    # --------------------------------------------------------------
    # Logits for softmax (guarantees positivity and simplex constraint).
    # --------------------------------------------------------------
    v = np.log(init_w + 1e-12)

    # --------------------------------------------------------------
    # Adam state.
    # --------------------------------------------------------------
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)            # second‑moment accumulator
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.8                           # base learning rate

    # --------------------------------------------------------------
    # Smoothing parameters for the log‑sum‑exp approximation.
    # --------------------------------------------------------------
    mu_factor = 0.8
    mu_factor_min = 0.01
    mu_decay = 0.9997

    best_w = init_w.copy()
    best_score = 2.0 * n * np.max(convolve_fft(best_w, best_w))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    iteration = 0
    # --------------------------------------------------------------
    # Main smooth‑max Adam optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, total_budget - 30.0):
        # ---------- current w ----------
        w = softmax(v)

        # optional symmetrisation every 1000 steps (helps stability)
        if iteration % 1000 == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])
            v = np.log(w + 1e-12)

        # ---------- convolution ----------
        b = convolve_fft(w, w)
        maxb = np.max(b)
        score = 2.0 * n * maxb

        # ---------- best‑track ----------
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")

        # ---------- smooth‑max weighting ----------
        mu = max(maxb * mu_factor, 1e-12)
        shifted = (b - maxb) / mu               # max(b) → 0, others ≤ 0
        shifted = np.clip(shifted, -40.0, 0.0)   # avoid under/overflow
        exp_shift = np.exp(shifted)
        p = exp_shift / np.sum(exp_shift)       # softmax over 2n‑1 entries

        # ---------- gradient wrt w ----------
        grad_w = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # ---------- chain rule to logits ----------
        dot = np.dot(w, grad_w)
        grad_v = w * (grad_w - dot)

        # ---------- Adam update ----------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v * grad_v)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat_corr = v_hat / (1.0 - beta2 ** iteration)

        lr = lr0 / np.sqrt(iteration)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # ---------- temperature schedule ----------
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        # ---------- occasional random jump ----------
        if iteration % 2500 == 0:
            cand = block_perturbation(best_w, rng)
            # re‑initialise logits from the perturbed candidate
            v = np.log(cand + 1e-12)
            m.fill(0.0)
            v_hat.fill(0.0)

            # evaluate jump immediately
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient refinement (optimises the true L∞‑norm).
    # --------------------------------------------------------------
    w = best_w.copy()
    sub_iter = 0
    lr0_sub = 0.5
    while (time.monotonic() - start_time < max(0.0, total_budget - 2.0)
           and sub_iter < 20000):
        b = convolve_fft(w, w)
        maxb = np.max(b)
        # pick an index attaining (or almost attaining) the maximum
        idxs = np.flatnonzero(b >= maxb - 1e-15)
        i_max = int(rng.choice(idxs))

        # sub‑gradient for the active peak (only the contributing pairs)
        grad_sub = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j_range = np.arange(j_start, j_end + 1, dtype=int)
        grad_sub[j_range] = 2.0 * w[i_max - j_range]

        # step size and projection
        eta = lr0_sub / np.sqrt(sub_iter + 1.0)
        w = w - eta * grad_sub
        w = simplex_projection(w)

        # check improvement
        new_score = 2.0 * n * np.max(convolve_fft(w, w))
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_w = w.copy()
            print(f"[sub‑improve] sub_iter={sub_iter:6d}, score={best_score:.6f}")

        sub_iter += 1

    # --------------------------------------------------------------
    # Final scaling to the conventional √(2 n) ℓ₁‑norm.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # (optional) final verification – the harness will also call evaluate_sequence
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # expose the result for possible reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq.tolist()

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
