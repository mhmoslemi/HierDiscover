"""
propose_candidate
-----------------
A self‑convolution minimiser based on projected Adam‑style gradients
with a soft‑max surrogate for the max‑operator and an auxiliary
flatness term (∑b_k²).  The iterate lives on the probability simplex
(∑w_i = 1, w_i ≥ 0) and is forced to be symmetric after each
projection – symmetry is known to be beneficial for the aperiodic
auto‑correlation problem.  Occasionally a random “block” perturbation
re‑injects diversity and a simple restart mechanism avoids long
stagnation periods.  After the search the best probability vector is
scaled to the required ℓ¹‑norm √(2 n) and returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – defined at top level.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.  Returns a non‑negative
    array of length len(a)+len(b)-1.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # remove tiny negatives caused by round‑off
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation used as a “jump’’.
    A random contiguous block is multiplied by a log‑normal factor and,
    with some probability, the whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def init_triangular(n: int) -> np.ndarray:
    """
    Symmetric triangular shape: ramps up then down.
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    tri = np.minimum(i + 1, n - i)
    return tri / np.sum(tri)


def init_arcsine(n: int) -> np.ndarray:
    """
    Discrete arcsine‑type density ∝ 1/√(x·(1‑x)).
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    density = 1.0 / np.sqrt(x * (1.0 - x))
    density = np.nan_to_num(density, nan=0.0, posinf=0.0, neginf=0.0)
    return density / np.sum(density)


def init_random_dirichlet(rng: np.random.Generator, n: int, alpha: float = 1.0) -> np.ndarray:
    """
    Random Dirichlet distribution with concentration ``alpha``.
    """
    vec = rng.gamma(alpha, 1.0, size=n)
    if np.sum(vec) == 0.0:
        vec = np.ones_like(vec)
    return vec / np.sum(vec)

# ----------------------------------------------------------------------

    Return a non‑negative step function (list of floats) whose

    The routine combines a smooth soft‑max surrogate, a flatness

        # 1) compute self‑convolution and current score
        # ----------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # 2) soft‑max weights for the smoothed max
        # ----------------------------------------------------------
        exp_arg = beta * (b - max_b)          # ≤ 0
        p = np.exp(exp_arg)
        sum_p = np.sum(p)
        if sum_p == 0.0:                      # underflow guard
            p = np.full_like(p, 1.0 / p.size)
            sum_p = 1.0
        p /= sum_p

        # ----------------------------------------------------------
        # 3) gradient (soft‑max + flattening)
        # ----------------------------------------------------------
        grad_soft = 2.0 * convolve_fft(p, w)[:n]
        grad_flat = 4.0 * convolve_fft(b, w)[:n] if lambda_flat > 0 else 0.0
        grad = grad_soft + lambda_flat * grad_flat

        # ----------------------------------------------------------
        # 4) heavy‑ball / Adam‑style projected step
        # ----------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1.0)
        velocity = momentum * velocity + eta * grad
        w_candidate = w - velocity
        w = simplex_projection(w_candidate)
        w = 0.5 * (w + w[::-1])   # enforce symmetry

        # ----------------------------------------------------------
        # 5) occasional random block perturbation
        # ----------------------------------------------------------
        if iteration % 4000 == 0 and iteration > 0:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            velocity.fill(0.0)
            print(f"[perturb] iter={iteration:6d}")

        # ----------------------------------------------------------
        # 6) increase smoothing parameter β slowly
        # ----------------------------------------------------------
        if iteration % beta_inc_interval == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_growth)

        # ----------------------------------------------------------
        # 7) reset if stagnation persists
        # ----------------------------------------------------------
        if no_improve > 30000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            velocity.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – the evaluation harness expects a sequence whose
    # ℓ¹‑norm equals √(2 n).  The value returned by ``evaluate_sequence``
    # equals the best_score we tracked, but we call it once for safety.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum          # Σ final_seq = √(2 n)
    final_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"[final] length={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Return plain Python list for the harness.
    return [float(x) for x in final_seq]
```
