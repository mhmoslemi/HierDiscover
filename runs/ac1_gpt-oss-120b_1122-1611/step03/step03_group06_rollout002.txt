"""
Propose a step‑function sequence using an alternating
convolution‑LP + simplex mixing scheme.

The core idea is the classic “linear‑programming improvement”
step: given a probability vector w (∑w=1) we compute
   M = max_k (w*w)[k].
We then solve the linear program

    maximise   Σ g_j
    subject to   (w * g)_k ≤ M   for all k
                g_j ≥ 0

which finds a vector g that places as much mass as possible in
positions where the current self‑convolution is small.
A convex combination w←(1−τ)w+τ·ĝ (with ĝ normalized to sum 1)
strictly reduces the maximal self‑convolution.
Repeating this step (with occasional random block perturbations
to escape plateaus) drives the objective

    score = 2·n·max_k (w*w)[k]

downwards.  The final sequence is obtained by scaling w so that
∑a_i = √(2 n), exactly the format required by
``evaluate_sequence``.

The implementation works with moderate lengths (≈2000‑3000) to keep
the LP tractable, but it automatically falls back to the length of
the previously discovered best construction if that is available.
"""

import time
import numpy as np
import cvxpy as cp
from scipy import sparse


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan‑type coefficients a_i = C(2i,i)/4**i, normalised later."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear full convolution via real FFT (fast for large n)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    conv = np.fft.irfft(Fa * Fb, N)[:n]
    # Numerical noise may produce tiny negatives; clip.
    np.maximum(conv, 0.0, out=conv)
    return conv


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Random contiguous block scaling (log‑normal factor) and optional symmetrisation.
    Used as a cheap “jump’’ when the algorithm stalls.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:        # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _build_convolution_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1)×n convolution matrix L such that L·g = f*g.
    The i‑th diagonal (offset i) contains the constant f[i].
    """
    n = f.size
    # For offset i there are (n‑i) non‑zero entries equal to f[i].
    diags_data = [np.full(n - i, f[i]) for i in range(n)]
    offsets = list(range(n))
    L = sparse.diags(diags_data, offsets, shape=(2 * n - 1, n), format='csr')
    return L

            # Gradually reduce the mixing factor to fine‑tune later.
            tau = max(0.05, tau * 0.92)
            print(f"[improve] iter={iter_cnt:6d}, score={best_sc:.6f}, tau={tau:.3f}")
        else:
            no_improve += 1
            # If we have been stuck for a while, inject a random jump.
            if no_improve >= 30:
                w = block_perturbation(best_w, rng)
                w = w / np.sum(w)
                tau = 0.45
                no_improve = 0
                print(f"[reset]   iter={iter_cnt:6d}")

        iter_cnt += 1

        # Simple early‑stop if we already beat the target bound.
        if best_sc < 1.5025:
            print("[target] reached target bound (or better).")
            break

    # ------------------------------------------------------------------
    # Final scaling: Σ a_i must equal sqrt(2 n).  After scaling,
    # evaluate_sequence returns exactly best_sc (up to tiny rounding).
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum
    final_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"[final] length={n}, best_score={best_sc:.6f}, evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
