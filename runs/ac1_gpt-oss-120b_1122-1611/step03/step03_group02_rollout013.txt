    if window_type == 'sin':
        x = np.arange(n) / (n-1)
        base = np.sin(np.pi * x)

        x = np.arange(n) / (n-1)

        x = np.arange(n) / (n-1)

"""
Propose a step‑function sequence with a low self‑convolution peak.

Algorithm
---------
1. **Random shape exploration** – generate many candidate sequences using
   simple parametric families (uniform, exponential, power‑law,
   symmetric power‑law, sine, Hann, Blackman, triangular).  Each candidate
   is scaled to the conventional ℓ₁‑norm √(2 n) and its peak
   `max(a*a)` is evaluated via an FFT based convolution.
   The best candidate found so far is kept.

2. **Sub‑gradient refinement** – starting from the best candidate,
   minimise the exact objective `max(a*a)` using projected sub‑gradient
   descent.  At each iteration the active constraints (indices where
   the convolution attains its maximum) are identified; the sub‑gradient
   is the average of the gradients of these active constraints and is
   obtained cheaply by a second convolution.  A momentum term and occasional
   random block perturbations are used to escape plateaus.  After each
   update the vector is clipped to non‑negative values and re‑scaled to
   the target ℓ₁‑norm.

3. **Output** – the final sequence (list of Python floats) is returned.
   The sequence is also stored in the global name `height_sequence_1`
   so that a subsequent call can reuse it as a warm start.

The whole search respects the time budget supplied by `budget_s`
(approximately 1000 s).  Intermediate progress is printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level as required by the harness.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Simple O(n) random block perturbation – used as a jump operator.
    Multiplies a random contiguous block by a random factor and optionally
    averages with its mirror image.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def generate_shape(n: int, kind: str, p: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative raw shape (before scaling) of length `n`.
    `kind` selects a base window; `p` is an exponent that sharpens
    (`p>1`) or flattens (`p<1`) the window.
    """
    x = np.arange(n) / max(1, n - 1)

    if kind == "uniform":
        base = np.ones(n)
    elif kind == "exponential":
        # rate close to 1, sampled uniformly in [0.95, 0.9999]
        r = rng.uniform(0.95, 0.9999)
        base = r ** np.arange(n)
    elif kind == "power":
        base = 1.0 / (np.arange(n) + 1) ** p
    elif kind == "sym_power":
        half = (n + 1) // 2
        left = 1.0 / (np.arange(half) + 1) ** p
        if n % 2 == 0:
            right = left[::-1]
        else:
            right = left[-2::-1]
        base = np.concatenate([left, right])
    elif kind == "sin":
        base = np.sin(np.pi * x)          # zero at the ends, peak in the middle
    elif kind == "hann":
        base = 0.5 * (1 - np.cos(2 * np.pi * x))
    elif kind == "blackman":
        base = 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
    elif kind == "triangular":
        base = 1.0 - 2.0 * np.abs(x - 0.5)    # linear ramp up then down
    else:
        base = np.ones(n)

    # Avoid exact zeros (they would stay zero after exponentiation)
    base = np.maximum(base, 1e-12)

    # Apply a shaping exponent (for kinds where it makes sense)
    if kind not in {"power", "sym_power"}:
        base = base ** p

    # Small random jitter (up to a few percent) – helps escape symmetry traps
    jitter = rng.uniform(0.0, 0.05)
    if jitter > 0.0:
        base *= (1.0 + rng.normal(0.0, jitter, size=n))
        base = np.maximum(base, 0.0)

    return base


def evaluate_peak(seq: np.ndarray) -> float:
    """
    Compute the objective value `max(a*a)` after scaling the sequence to
    ℓ₁‑norm √(2 n).  This is exactly the score returned by `evaluate_sequence`
    for a correctly scaled input.
    """
    n = seq.size
    target_sum = np.sqrt(2.0 * n)
    s = target_sum / np.sum(seq)
    # Convolution of the scaled sequence; we only need the max value.
    b = convolve_fft(seq * s, seq * s)
    return float(b.max())

# Main function required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is as low as possible.

    The routine first performs a fast random search over several
    parametric families of shapes, then refines the best candidate
    with a projected sub‑gradient method that directly minimizes the
    peak of the self‑convolution.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Total time we will actually use; leave a small margin for finalisation.
    total_budget = float(budget_s) - 0.5

    # --------------------------------------------------------------
    # Grab a previously found construction if it exists.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_n = None

    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= 2:
                sc = evaluate_peak(prev)
                best_seq = prev.copy()
                best_score = sc
                best_n = prev.size
                print(f"[init] using previous sequence, n={best_n}, score={best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Random shape exploration – occupies roughly 30 % of the budget.
    # --------------------------------------------------------------
    shape_kinds = [
        "uniform", "exponential", "power", "sym_power",
        "sin", "hann", "blackman", "triangular"
    ]
    # Desired length range – larger lengths give finer granularity but cost
    # more per iteration.  We stay between 2000 and 10000.
    min_n, max_n = 2000, 10000

    # Heuristic split of the budget
    explore_deadline = start_time + 0.3 * total_budget

    while time.monotonic() < explore_deadline:
        # Random length
        n = int(rng.integers(min_n, max_n + 1))
        # Random window type and exponent
        kind = rng.choice(shape_kinds)
        # exponent roughly in [0.2, 4]; use log‑uniform sampling
        p = float(10 ** rng.uniform(np.log10(0.2), np.log10(4.0)))
        # Build raw shape and evaluate
        raw = generate_shape(n, kind, p, rng)

        if raw.sum() == 0.0:
            continue

        # Scale to the canonical ℓ₁ norm and evaluate the peak.
        score = evaluate_peak(raw)

        if score < best_score:
            best_score = score
            best_seq = raw.copy()
            best_n = n
            print(f"[search] new best: n={n}, kind={kind}, p={p:.3f}, score={score:.6f}")

    # If random search failed to produce any candidate (unlikely), fall back
    # to the classic central‑binomial distribution.
    if best_seq is None:
        n = 8000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2 * i)
        best_seq = a
        best_n = n
        best_score = evaluate_peak(best_seq)
        print(f"[fallback] using central‑binomial, n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Sub‑gradient refinement – the main optimisation phase.
    # --------------------------------------------------------------
    n = best_n
    target_sum = np.sqrt(2.0 * n)

    # Normalise the current best to the target ℓ₁‑norm.
    w = np.maximum(best_seq, 0.0)
    if np.sum(w) == 0.0:
        w = np.full(n, target_sum / n)
    else:
        w = w * (target_sum / np.sum(w))

    best_w = w.copy()
    best_score = evaluate_peak(best_w)

    # Momentum / Adam‑like accumulators
    momentum = np.zeros_like(w)

    # Hyper‑parameters for the sub‑gradient loop
    eta0 = 0.5          # initial step‑size (will be divided by √t)
    beta = 0.9          # momentum coefficient
    block_interval = 1500   # how often we try a random jump
    reset_interval = 20000  # restart if no improvement for a long time
    no_improve = 0
    iteration = 0

    refinement_deadline = start_time + total_budget

    while time.monotonic() < refinement_deadline:
        # 1) current convolution and peak
        b = convolve_fft(w, w)
        maxb = np.max(b)

        # 2) record improvement
        if maxb < best_score - 1e-12:
            best_score = float(maxb)
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        tol = maxb * 1e-12
        active = np.flatnonzero(b >= maxb - tol)
        # probability vector over the 2n‑1 convolution entries
        p = np.zeros_like(b)

        # 4) gradient = 2·conv(p, w_rev)  (take the central n entries)
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        eta = eta0 / np.sqrt(iteration + 1.0)

        w_candidate = w + momentum

        else:

            cand_score = evaluate_peak(cand)

                best_score = cand_score
                w = cand.copy()
                momentum = np.zeros_like(w)
                no_improve = 0
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        if no_improve > reset_interval:
            w = block_perturbation(best_w, rng)
            w = w * (target_sum / np.sum(w))
            momentum = np.zeros_like(w)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but we output

    final_seq = best_w.copy()

    print(f"[final] length={n}, best iter={iteration}, best score (peak)={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Store the result for possible reuse in a later run.
    globals()["height_sequence_1"] = final_seq.tolist()

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
