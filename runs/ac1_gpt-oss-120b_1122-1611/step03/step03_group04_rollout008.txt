"""
propose_candidate
-----------------
A fast convex‑optimization based search for a non‑negative step function
that minimises  2 n·max(conv(w,w))  (equivalently the ℓ∞‑norm of the
self‑convolution).  The algorithm works on the probability vector
w (∑w=1) and consists of three stages:

1. **Hard‑max sub‑gradient descent**  
   The exact sub‑gradient of f(w)=max(conv(w,w)) is obtained by a single
   FFT‑based correlation with the indicator of the maximal entries.
   A multiplicative (entropic) update with momentum is applied and the
   vector is kept symmetric after every step.

2. **Occasional block perturbations**  
   To escape plateaus we repeatedly multiply a random contiguous block
   by a log‑normal factor and re‑symmetrise.  If the perturbed candidate
   improves the score it is kept as the new current iterate.

3. **Final local random swaps**  
   In the last few seconds a cheap random‑exchange hill‑climber performs
   tiny mass transfers between two positions; every improvement is kept.

The whole procedure respects the time budget (≈ 1000 s), uses only NumPy
(and the already imported ``evaluate_sequence``) and finally rescales
the best vector to the conventional √(2 n) ℓ¹‑norm before returning it
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (FFT‑based linear convolutions)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution using a real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negatives caused by numerical noise.
    np.maximum(c, 0.0, out=c)
    return c


def max_subgradient(w: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  f(w)=max(conv(w,w)) .
    Returns  g_i = 2·∑_{k∈K} w_{k-i},   K = argmax conv.
    Implemented via a single FFT‑based correlation.
    """
    n = w.size
    max_val = conv.max()
    # mask of the active (maximal) convolution entries
    mask = (conv >= max_val - 1e-12).astype(np.float64)      # length 2n‑1
    # correlation: g = 2 * corr(w, mask)
    # corr(w,mask)[i] = ∑_j w[j]·mask[i+j]
    # we can obtain it as a convolution with the reversed mask.
    conv_full = convolve_fft(w, mask[::-1])                # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)                      # map to i‑range
    g = 2.0 * conv_full[idx]
    return g


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and
    (optionally) re‑symmetrise.  The operation keeps the vector non‑negative
    and respects the [0,1000] clipping required by the evaluator."""
    cand = base.copy()
    n = cand.size
    # block width between 0.3 % and ~10 % of the length
    lo = max(1, n // 300)
    hi = max(2, n // 10)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.15)))  # log‑normal multiplier
    cand[start:stop] *= factor

    # Symmetrise with probability 0.5 – the optimal solution is symmetric.
    if rng.random() < 0.5:
        cand = 0.5 * (cand + cand[::-1])

    np.clip(cand, 0.0, 1000.0, out=cand)
    # Re‑normalise to stay on the probability simplex.
    cand_sum = cand.sum()
    if cand_sum == 0.0 or not np.isfinite(cand_sum):
        cand = np.full_like(cand, 1.0 / n)
    else:
        cand /= cand_sum
    return cand

    """
    Returns a list of non‑negative floats (the step heights) whose
    ``evaluate_sequence`` score is as low as possible.  The implementation
    follows the three‑stage strategy described in the module docstring.
    """

    best_score = float(2 * n * np.max(convolve_fft(w, w)))

    iteration = 0

    while True:

        conv = convolve_fft(w, w)

            best_w = w.copy()
            no_improve = 0

            no_improve += 1

        else:

                best_score = cand_score
