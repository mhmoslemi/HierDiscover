        x = (idx + 0.5) / n
        shape = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))

        shape[np.isinf(shape)] = 0.0
        return shape

"""
Hybrid simulated‑annealing / projected‑gradient search for a
non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works on the probability simplex (∑a = 1, a ≥ 0) –
scaling is irrelevant for the final score.  It proceeds in three
phases:

1. **Initialisation** – choose a length `n` (re‑use the existing
   global `height_sequence_1` if present) and start from a
   Beta‑distribution on the simplex (U‑shaped for α<1).  The best
   among a few α‑values is kept.

2. **Simulated‑annealing exploration** – repeatedly apply cheap block
   multiplicative perturbations and global log‑normal noise,
   accepting moves with a Metropolis rule.  The temperature decays
   exponentially and the current best solution is tracked.

3. **Smooth‑max projected gradient descent** – after the stochastic
   phase we refine the current best point using the smooth‑max
   approximation of the `ℓ∞`‑norm (the same objective used in the
   reference implementation).  Momentum and a diminishing step‑size
   are employed, together with Euclidean projection onto the simplex.

The final sequence is scaled to the conventional √(2 n) ℓ₁‑norm
(only for readability – the score is scale‑invariant).  The function
`propose_candidate` returns the list of floats required by the
evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top‑level as required by the harness.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # very small negatives can appear because of round‑off
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(vec: np.ndarray) -> float:
    """Raw objective 2·n·max(conv)/sum² (scale‑invariant)."""
    n = vec.shape[0]
    conv = convolve_fft(vec, vec)
    maxb = np.max(conv)
    s = np.sum(vec)
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is (hopefully) better than the current
    best (~1.51786).  The routine runs a stochastic search followed
    by a deterministic gradient refinement within the given time
    budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose a length and an initial point on the simplex.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` is supplied by the harness if a previous
        # construction exists.
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = prev_seq.size
        if n < 2:
            raise ValueError
        # Normalise just in case the previous run used a different scale.
        prev_seq = np.maximum(prev_seq, 0.0)
        prev_seq = prev_seq / np.sum(prev_seq)
    except Exception:
        # No previous sequence – pick a moderate size.
        n = 8000
        prev_seq = None

    # ------------------------------------------------------------------
    # 2) Build a few Beta‑shaped candidates on the simplex and pick the
    #    one with the smallest raw score.
    # ------------------------------------------------------------------
    def beta_shape(alpha):
        """U‑shaped Beta(α,α) density on {0,…,n‑1} (discrete)."""
        i = np.arange(n, dtype=np.float64)
        # map to (0,1) without hitting the exact boundaries
        x = (i + 0.5) / n
        # density ∝ x^{α-1} (1‑x)^{α-1}
        shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        shape[np.isinf(shape)] = 0.0
        return shape

    candidates = []
    alphas = [0.2, 0.3, 0.4, 0.6, 1.0]       # 1.0 → uniform
    for a in alphas:
        w = beta_shape(a)
        if np.all(w == 0):
            continue
        w /= np.sum(w)
        candidates.append(w)

    # add the previous construction (if any) as a candidate
    if prev_seq is not None:
        candidates.append(prev_seq)

    # pick the best among the few cheap candidates
    best_init = None
    best_val = np.inf
    for w in candidates:
        val = compute_score(w)
        if val < best_val:
            best_val = val
            best_init = w.copy()

    if best_init is None:                # fallback – uniform
        best_init = np.full(n, 1.0 / n)
        best_val = compute_score(best_init)

    w = best_init.copy()                 # current point (on simplex)
    cur_score = best_val
    best_w = w.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial raw score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 3) Phase‑1: Simulated‑annealing exploration.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)   # a few seconds spare
    phase1_budget = total_budget * 0.60
    phase2_budget = total_budget * 0.35
    # temperature schedule (exponential decay)
    T_start = best_score * 0.25          # start fairly warm
    T_end = best_score * 1e-5

    iteration = 0
    while time.monotonic() - start_time < phase1_budget:
        # fraction of phase 1 elapsed → temperature
        elapsed = time.monotonic() - start_time
        frac = elapsed / phase1_budget
        T = T_start * (T_end / T_start) ** frac

        # ----- propose a neighbour ------------------------------------
        cand = w.copy()
        if rng.random() < 0.65:
            # block multiplicative perturbation (local move)
            lo = max(1, n // 500)
            hi = max(2, n // 20)
            width = int(rng.integers(lo, hi + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.13)))   # log‑normal factor
            cand[start:start + width] *= factor
        else:
            # global log‑normal noise (small‑scale exploration)
            noise = rng.normal(0.0, 0.02, size=n)
            cand = cand * np.exp(noise)

        # ensure non‑negativity and re‑normalise to the simplex
        np.maximum(cand, 0.0, out=cand)
        s = np.sum(cand)
        if s == 0.0:
            continue
        cand /= s

        cand_score = compute_score(cand)

        # ----- Metropolis acceptance ---------------------------------
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            w = cand
            cur_score = cand_score

        # ----- update global best ------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_w = w.copy()
            print(f"[SA improve] iter={iteration:7d}, raw score={best_score:.8f}")

        # occasional symmetric averaging – cheap way to enforce symmetry
        if iteration % 5000 == 0 and iteration > 0:
            sym = 0.5 * (w + w[::-1])
            sym /= np.sum(sym)
            sym_score = compute_score(sym)
            if sym_score < best_score:
                best_score = sym_score
                best_w = sym.copy()
                w = sym.copy()
                cur_score = sym_score
                print(f"[sym improve] iter={iteration:7d}, raw score={best_score:.8f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4) Phase‑2: Smooth‑max projected gradient descent (deterministic fine‑tune).
    # ------------------------------------------------------------------
    w = best_w.copy()
    momentum = np.zeros_like(w)
    eta0 = 0.55
    beta = 0.88
    mu_factor = 0.45
    mu_factor_min = 0.02
    mu_decay = 0.9996

    cur_score = compute_score(w)
    gd_iter = 0
    while time.monotonic() - start_time < phase1_budget + phase2_budget:
        # convolution and smooth‑max weights
        conv = convolve_fft(w, w)
        maxb = np.max(conv)
        mu = maxb * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        shifted = (conv - maxb) / mu              # max → 0, others ≤0
        exp_term = np.exp(shifted)
        p = exp_term / np.sum(exp_term)           # softmax (length 2n‑1)

        # gradient of the smooth‑max objective
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # step‑size & momentum update
        eta = eta0 / np.sqrt(gd_iter + 1.0)
        momentum = beta * momentum - eta * grad
        w_candidate = w + momentum

        # project back onto the simplex
        w = simplex_projection(w_candidate)

        # evaluate
        cur_score = compute_score(w)
        if cur_score < best_score:
            best_score = cur_score
            best_w = w.copy()
            print(f"[GD improve] iter={gd_iter:7d}, raw score={best_score:.8f}")

        # cool the smoothing factor
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        gd_iter += 1
        if gd_iter > 250000:          # safety guard
            break

    # ------------------------------------------------------------------
    # 5) Final scaling (the evaluation is scale‑invariant, but we output
    #    the conventional √(2 n) ℓ₁‑norm for readability).
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # Compute the official score (the harness will also call evaluate_sequence,
    # but printing it here is useful for debugging).
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best raw score={best_score:.8f}, evaluate={final_score:.8f}")

    # Convert to a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
