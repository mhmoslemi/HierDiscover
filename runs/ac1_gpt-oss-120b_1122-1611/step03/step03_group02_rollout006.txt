"""
Propose a candidate sequence for the self‑convolution L∞ minimisation problem.

The algorithm consists of three stages:

1. **Initialisation** – reuse a previously found sequence if available,
   otherwise start from a uniform distribution on a moderate length (≈8000).

2. **Randomised shape search** – generate many symmetric “edge‑heavy’’
   distributions from a few low‑dimensional families (inverse‑power,
   exponential decay, distance‑power).  Each shape is optionally mixed with
   the uniform distribution.  The best candidate (lowest 2·n·max(conv))
   seen so far is kept.

3. **Fine‑tuning** – run a short Adam‑style projected gradient descent
   on the smooth‑max approximation of the L∞ norm, starting from the best
   random candidate.  The iterate is projected onto the probability simplex
   after each step.

The final vector is scaled to the conventional ℓ₁‑norm √(2 n) before being
returned – this does not change the objective value returned by
`evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top‑level as required by the harness.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Numerical noise can create tiny negative values – clip them.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, ∑x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main function required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is (hopefully) better than the previous best.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a length.  Use the previously discovered sequence if present,
    # otherwise pick a moderate size (≈8000) that balances quality and speed.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(init_seq.size)
            if n < 2:
                raise ValueError
            # Normalise to a probability vector (∑ = 1)
            sum_init = np.sum(init_seq)
            if sum_init < 1e-12:
                raise ValueError
            w0 = np.maximum(init_seq, 0.0) / sum_init
            print(f"[init] re‑using previous sequence, length={n}")
        except Exception:
            init_seq = None
            n = 8000
            w0 = None
    else:
        init_seq = None
        n = 8000
        w0 = None

    # ------------------------------------------------------------------
    # Uniform baseline – used for mixture and as a safe fallback.
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initialise the best candidate.
    # ------------------------------------------------------------------
    best_w = uniform.copy()
    best_score = np.inf

    if w0 is not None:
        # evaluate the imported candidate
        b0 = convolve_fft(w0, w0)
        best_score = 2.0 * n * np.max(b0)
        best_w = w0.copy()
        print(f"[init] score of imported sequence = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Randomised shape search.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # reserve a few seconds for the final scaling / return
    alloc_time = max(0.0, float(budget_s) - 20.0)

    # Pre‑compute the index array – used by the three families.
    idx = np.arange(n, dtype=np.float64)

    while time.monotonic() - start_time < alloc_time:
        # Randomly pick one of three simple families.
        family = rng.integers(0, 3)  # 0 = inverse‑power, 1 = exponential, 2 = distance‑power

        if family == 0:
            # Inverse‑power of the product (edge‑heavy “arcsine‑like’’ shape)
            gamma = rng.uniform(0.30, 0.80)          # shape exponent
            offset = rng.uniform(0.0, 2.0)           # tiny shift to avoid exact zeros
            base = ((idx + offset) * (n - 1 - idx + offset)) ** (-gamma)

        elif family == 1:
            # Symmetric exponential decay from the edges toward the centre
            gamma = rng.uniform(0.001, 0.05)         # decay rate
            d = np.minimum(idx, n - 1 - idx)
            base = np.exp(-gamma * d)

        else:
            # Power of the distance to the centre (also edge‑heavy)
            gamma = rng.uniform(0.5, 3.0)
            d = np.minimum(idx, n - 1 - idx) + 1.0
            base = d ** (-gamma)

        # Normalise to a probability vector.
        base_sum = np.sum(base)
        if base_sum <= 0.0:
            continue
        base /= base_sum

        # Mix with the uniform distribution (mixing weight λ ∈ [0,1]).
        lam = rng.random()
        w = lam * base + (1.0 - lam) * uniform  # already sums to 1

        # Evaluate the score (2·n·max(conv)).
        b = convolve_fft(w, w)
        maxb = np.max(b)
        score = 2.0 * n * maxb

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[improve] time={time.monotonic() - start_time:6.1f}s, "
                  f"family={family}, gamma={gamma:.4f}, λ={lam:.3f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fine‑tuning with projected Adam on the smooth‑max surrogate.
    # ------------------------------------------------------------------
    remaining = float(budget_s) - (time.monotonic() - start_time) - 5.0
    if remaining > 15.0 and best_w is not None:
        w = best_w.copy()
        m = np.zeros_like(w)
        v = np.zeros_like(w)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-9
        lr = 0.02                     # base learning rate
        mu_factor = 0.5               # start smoothing as a fraction of max(conv)
        mu_decay = 0.9995
        it = 0
        while time.monotonic() - start_time < float(budget_s) - 5.0:
            b = convolve_fft(w, w)
            maxb = np.max(b)
            score = 2.0 * n * maxb
            if score < best_score - 1e-12:
                best_score = score
                best_w = w.copy()
                print(f"[adam] iter={it:6d}, score={best_score:.6f}")

            # Smooth‑max weights (log‑sum‑exp approximation of the max)
            mu = maxb * mu_factor
            if mu <= 0.0:
                mu = 1e-12
            shifted = (b - maxb) / mu           # max → 0, others ≤ 0
            exp_shift = np.exp(shifted)         # in [0,1]
            p = exp_shift / np.sum(exp_shift)   # soft‑max over 2n‑1 entries

            # Gradient of the smooth‑max objective: 2·conv(p, w_rev)
            grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

            # Adam update (minimisation → subtract the step)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            w_candidate = w - step

            # Project back onto the simplex {x ≥ 0, ∑x = 1}
            w = simplex_projection(w_candidate)

            # Decay smoothing factor slowly.
            mu_factor = max(mu_factor * mu_decay, 0.04)

            it += 1
            if it >= 200000:          # safety cap
                break

    # ------------------------------------------------------------------
    # Final scaling – we output the conventional √(2 n) ℓ₁‑norm.
    # ------------------------------------------------------------------
    if best_w is None:
        best_w = uniform.copy()

    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # Evaluate with the provided (secure) function for an honest log.
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score≈{best_score:.6f}, "
          f"evaluate_sequence={final_score:.6f}")

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
