    start = time.monotonic()
    best_w = w.copy()
    b = convolve_fft(w,w)
    best_score = 2*n*np.max(b)
    iteration = 0
    no_improve=0
    while time.monotonic() - start < time_limit:
        b = convolve_fft(w,w)
        max_b = np.max(b)
        score = 2*n*max_b
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve=0
        else:
            no_improve+=1

        # decide gradient method
        if iteration % 5 == 0:
            # compute subgradient at max positions
            max_idxs = np.where(b >= max_b - 1e-12)[0]
            grad = np.zeros_like(w)
            for m in max_idxs:
                lo = max(0, m-(n-1))
                hi = min(m, n-1)
                idx_range = np.arange(lo, hi+1)
                grad[idx_range] += w[m - idx_range]
            grad *= 2.0 / len(max_idxs)
        else:

"""
Propose a step‑function (non‑negative sequence) that minimises
    evaluate_sequence(seq) = max_k (seq * seq)[k] .
The algorithm consists of two phases:

1. **Coarse shape search** – try a handful of simple parametric families
   (Catalan, power‑law, exponential, uniform) on a fixed length `n`.
   Each candidate is symmetrised, normalised to a probability vector and
   scored by `2·n·max(conv(w,w))`.  The best shape is kept.

2. **Refined EG optimisation** – start from the best shape and perform an
   exponentiated‑gradient descent on a smooth approximation of the max‑convolution.
   The gradient is either a soft‑max weighted combination of all shifts
   (temperature γ proportional to the current maximum) or, every few
   iterations, the exact sub‑gradient of the true max (averaged over all
   maximising indices).  After each update the vector is projected back onto
   the probability simplex and forced to be symmetric.  Occasional random
   block perturbations and a stagnation‑reset keep the search from getting
   stuck.

The whole routine respects the 1000 s budget, prints status messages and
returns the final sequence scaled to the conventional √(2 n) L¹‑norm.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for lengths up to ~10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)        # clip tiny negative round‑off errors
    return c


# ----------------------------------------------------------------------
# Simple families of trial sequences (always returned normalised)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Scaled Catalan numbers a_i = C(2i,i)/4**i (decreasing ~ i⁻¹ᐟ²)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def power_law_sequence(length: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    return idx ** (-p)


def exponential_sequence(length: int, alpha: float) -> np.ndarray:
    """a_i = exp(-alpha * i)."""
    idx = np.arange(length, dtype=np.float64)
    return np.exp(-alpha * idx)


def uniform_sequence(length: int) -> np.ndarray:
    """Uniform vector."""
    return np.ones(length, dtype=np.float64)


def symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force exact symmetry (w_i = w_{n-1-i})."""
    return 0.5 * (seq + seq[::-1])


def normalise_to_one(seq: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (L1‑norm = 1)."""
    total = np.sum(seq)
    if total == 0.0:
        return np.full_like(seq, 1.0 / seq.size)
    return seq / total


def evaluate_w(w: np.ndarray) -> float:
    """Return the objective value for a probability vector `w` (sum=1)."""
    n = w.size
    b = convolve_fft(w, w)
    return 2.0 * n * np.max(b)


# ----------------------------------------------------------------------
# Gradient of the soft‑max smoothed max‑convolution
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max probabilities p_k ∝ exp(conv_k / γ)."""
    shift = conv - np.max(conv)               # for numerical stability
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float):
    """
    Compute ∇f_γ(w) where f_γ(w)=γ·log Σ_k exp((w*w)_k/γ).
    Returns (gradient, max_convolution_entry).
    """
    n = w.size
    # self‑convolution
    b = convolve_fft(w, w)                     # length 2n‑1
    # soft‑max over convolution entries
    p = _softmax_weights(b, gamma)             # length 2n‑1
    # gradient = 2 * (w * p_rev)  (first n entries, after reversing)
    p_rev = p[::-1]
    conv = convolve_fft(w, p_rev)              # length 3n‑2
    grad = 2.0 * conv[::-1][:n]                # align and keep first n entries
    return grad, np.max(b)


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (averaged over all maximisers)
# ----------------------------------------------------------------------


def _max_subgradient(w: np.ndarray, b: np.ndarray):
    """
    Return the sub‑gradient of max_k (w*w)[k].
    If several indices attain the maximum, the gradient is the average
    of the corresponding shift vectors.
    """
    n = w.size
    max_val = np.max(b)
    # treat entries within 1e-12 of the max as active (covers ties)
    active = np.where(b >= max_val - 1e-12)[0]
    grad = np.zeros_like(w)
    for m in active:
        lo = max(0, m - (n - 1))
        hi = min(m, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[m - idx]               # w_{m-i}
    grad *= 2.0 / max(1, len(active))         # 2·average shift
    return grad


# ----------------------------------------------------------------------
# Cheap random block jump (used as perturbation)
# ----------------------------------------------------------------------


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:            # occasional forced symmetry
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function with a low `evaluate_sequence`
    score. The routine spends ~35 % of the time exploring simple parametric
    families, then refines the best candidate by an exponentiated‑gradient
    descent on a smoothed max‑convolution objective.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = float(budget_s)

    # ------------------------------------------------------------------
    # Choose a length.  Re‑use a previously found construction if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 5:
                raise ValueError
        except Exception:
            n = 8000
    else:
        n = 8000

    # ------------------------------------------------------------------
    # 1) Coarse parametric search
    # ------------------------------------------------------------------
    shape_candidates = [
        ("catalan", None),
        ("uniform", None),
    ]
    # power‑law exponents
    for p in np.linspace(0.10, 1.50, 15):
        shape_candidates.append(("power", float(p)))
    # exponential decay rates
    for alpha in [1e-4, 5e-4, 1e-3, 2e-3, 5e-3, 1e-2, 2e-2, 5e-2]:
        shape_candidates.append(("exp", float(alpha)))

    time_for_shape = total_budget * 0.34
    best_w = None
    best_score = float("inf")

    for typ, param in shape_candidates:
        if time.monotonic() - start_time > time_for_shape:
            break

        # generate raw shape
        if typ == "catalan":
            raw = catalan_sequence(n)
        elif typ == "uniform":
            raw = uniform_sequence(n)
        elif typ == "power":
            raw = power_law_sequence(n, param)
        elif typ == "exp":
            raw = exponential_sequence(n, param)
        else:
            continue

        # enforce symmetry and normalise to a probability vector
        w = normalise_to_one(symmetrize(raw))

        score = evaluate_w(w)
        if score < best_score:
            best_score = score
            best_w = w.copy()
            print(f"[shape] type={typ:<8} param={param!s:<8} score={score:.6f}")

    # If no candidate survived (unlikely) fall back to uniform
    if best_w is None:
        best_w = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_w(best_w)

    # ------------------------------------------------------------------
    # 2) Refined exponentiated‑gradient optimisation
    # ------------------------------------------------------------------
    opt_time = total_budget - (time.monotonic() - start_time) - 5.0
    if opt_time > 10.0:
        w = best_w.copy()
        iteration = 0
        no_improve = 0
        eta0 = 0.65                # base learning rate for EG
        gamma_factor = 0.03        # temperature as a fraction of current max
        max_exponent = 50.0        # clipping for the exponent
        sym_interval = 1           # enforce symmetry every iteration (fast)
        block_interval = 2500      # occasional random jump
        reset_after = 25000        # stagnation limit

        while time.monotonic() - start_time < total_budget - 5.0:
            # current convolution and score
            b = convolve_fft(w, w)
            max_b = np.max(b)
            score = 2.0 * n * max_b

            # record improvement
            if score < best_score - 1e-12:
                best_score = score
                best_w = w.copy()
                no_improve = 0
                print(f"[improve] iter={iteration:6d} score={best_score:.6f}")
            else:
                no_improve += 1

            # choose gradient source
            if iteration % 5 == 0:
                # exact sub‑gradient of the true max (more aggressive)
                grad = _max_subgradient(w, b)
            else:
                # smooth gradient via soft‑max (helps when many entries are close)
                gamma = max(gamma_factor * max_b / np.sqrt(iteration + 1.0), 1e-12)
                grad, _ = _smoothed_gradient(w, gamma)

            # EG update (multiplicative) with decaying step size
            eta = eta0 / np.sqrt(iteration + 1.0)
            exponent = -eta * grad
            np.clip(exponent, -max_exponent, max_exponent, out=exponent)
            w = w * np.exp(exponent)

            # projection onto the simplex
            sum_w = np.sum(w)
            if sum_w == 0.0:
                w = np.full_like(w, 1.0 / n)
            else:
                w /= sum_w

            # enforce symmetry (cheap – O(n))
            if sym_interval and (iteration % sym_interval == 0):
                w = 0.5 * (w + w[::-1])

            # occasional random block perturbation – keep it only if it improves
            if block_interval and iteration % block_interval == 0 and iteration > 0:
                cand = block_perturbation(best_w, rng)
                cand = normalise_to_one(cand)
                cand_score = evaluate_w(cand)
                if cand_score < best_score:
                    best_w = cand.copy()
                    best_score = cand_score
                    w = cand.copy()
                    print(f"[block] iter={iteration:6d} new best={best_score:.6f}")

            # stagnation handling: reset to a perturbed version of the best seen
            if no_improve > reset_after:
                w = block_perturbation(best_w, rng)
                w = normalise_to_one(w)
                no_improve = 0
                print(f"[reset] iter={iteration:6d} reset to perturbed best")

            iteration += 1

    # ------------------------------------------------------------------
    # 3) Final scaling to the conventional √(2 n) L¹‑norm
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum

    # sanity check (silent when the harness does not provide evaluate_sequence)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n} best_score={best_score:.6f} after scaling={final_score:.6f}")
    except Exception:
        print(f"[final] n={n} best_score={best_score:.6f}")

    # Return a plain Python list as required by the harness
    return [float(x) for x in final_seq]
