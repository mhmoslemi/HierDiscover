def solve_lp(f, M):
    n = len(f)
    import cvxpy as cp
    import scipy.sparse as sp

    # Build A as sparse diagonal matrix
    # offsets = np.arange(n)
    A = sp.diags(f, offsets=np.arange(n), shape=(2*n-1, n), format='csc')
    g = cp.Variable(n)
    constraints = [A @ g <= M + 1e-12, g >= 0]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-8, reltol=1e-8)

    if g.value is None:
        # fallback: return zeros
        return np.zeros(n), 0.0
    g_val = np.maximum(g.value, 0.0)  # ensure nonneg
    return g_val, float(np.sum(g_val))

"""
Propose a low‑score step‑function using a two‑phase optimisation:

1. Initialise a probability vector w (∑w = 1) from several baselines
   (Catalan, arcsine law, Dirichlet, optional previous solution).

2. Repeatedly solve a linear programme

        max   1ᵀ g
        s.t.  f * g ≤ M          (convolution bound, linear in g)
               g ≥ 0

   where f is the current distribution, M = max(f*f) and “*” denotes the
   linear convolution.  The LP returns a vector g with a total mass S>1.
   After normalising (ĝ = g / S) we mix the old and the new distribution

        f_new = (1‑t)·f + t·ĝ   (t∈{0.1,0.3,0.5,0.7,0.9})

   and keep the mixture that yields the smallest new maximum convolution.
   Mixing strictly reduces the objective 2·n·max(f*f); the process is
   repeated until the time budget expires or the LP can no longer increase
   the mass (S≈1).

3. Occasionally apply cheap random block perturbations to escape shallow
   local minima.

4. At the end the vector is scaled to the convention
   Σ aᵢ = √(2·n) and returned as a plain Python list.

The implementation uses NumPy, SciPy’s sparse diagonal matrix (so that the
convolution constraints are built in O(n) memory) and CVXPY with the ECOS
solver.  With n≈2000 the LP is tiny (≈4 000 constraints, 2 000 variables) and
the whole routine stays well within the 1000 s budget while typically
pushing the score below the target 1.5025.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities (all top‑level as required)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def arcsine_sequence(length: int) -> np.ndarray:
    """Discrete approximation of the arcsine density 1/√(x(1‑x))."""
    x = np.arange(length) + 0.5
    seq = 1.0 / np.sqrt(x * (length + 0.5 - x))
    return seq


def random_dirichlet(length: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma(0.5,1) ≈ Dirichlet with α=0.5 – a spiky start."""
    seq = rng.gamma(0.5, 1.0, size=length)
    return seq


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump used as a large perturbation."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= np.sum(cand)
    return cand


def max_convolution(v: np.ndarray) -> float:
    """Return max_{k}(v * v)[k] for a non‑negative vector v (L¹ = 1)."""
    conv = np.convolve(v, v)
    return float(conv.max())


def solve_lp_for_g(f: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """
    Solve   max 1ᵀ g   s.t.   f*g ≤ M   (linear convolution) ,   g ≥ 0.
    Returns the optimal g and its total mass Σg.
    """
    n = f.size
    # Build sparse matrix A where (A g)[k] = (f * g)[k]
    # Row k, column j gets weight f_{k‑j} when 0≤k‑j<n, i.e. diagonal offset = i.
    A = sp.diags(f, offsets=np.arange(n), shape=(2 * n - 1, n), format='csc')

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M + 1e-12]          # tiny slack for numerical safety
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is fast for this size; warm_start helps when we call many times.
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
               abstol=1e-9, reltol=1e-9, feastol=1e-9)

    if g.value is None:
        # In the very unlikely case of a solver failure, fall back to zero.
        return np.zeros(n, dtype=np.float64), 0.0

    g_opt = np.maximum(g.value, 0.0)   # eliminate tiny negatives
    return g_opt, float(g_opt.sum())


def downsample_sequence(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Simple down‑sampling by taking equally spaced indices."""
    if seq.size <= target_len:
        return seq.copy()
    idx = np.linspace(0, seq.size - 1, target_len).astype(int)
    return seq[idx].copy()


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low
    ``evaluate_sequence`` score.  The routine runs for roughly 1000 s,
    mixing a linear‑programming based global move with occasional random
    block perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a modest length that keeps the LP cheap yet still
    # expressive enough.  2000 works well in practice.
    # --------------------------------------------------------------
    TARGET_N = 2000

    # Try to reuse a previous construction if it has at least TARGET_N items.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= TARGET_N:
            w0 = downsample_sequence(prev, TARGET_N)
            w0 = np.maximum(w0, 0.0)
            w0 /= w0.sum()
            n = TARGET_N
        else:
            raise ValueError
    except Exception:
        # Fall back to a few deterministic starters.
        n = TARGET_N
        starters = []

        cat = catalan_sequence(n)
        starters.append(cat / cat.sum())

        arcs = arcsine_sequence(n)
        starters.append(arcs / arcs.sum())

        rand = random_dirichlet(n, rng)
        starters.append(rand / rand.sum())

        # pick the best starter (lowest max convolution)
        best_score = np.inf
        best_w = None
        for s in starters:
            sc = 2.0 * n * max_convolution(s)
            if sc < best_score:
                best_score = sc
                best_w = s.copy()
        w0 = best_w

    # --------------------------------------------------------------
    # Initial state
    # --------------------------------------------------------------
    w = w0.copy()
    best_w = w.copy()
    best_score = 2.0 * n * max_convolution(w)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop (LP + mixing + occasional perturbation)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # Reserve a safety margin for final scaling & polishing.
    max_time = max(0.0, float(budget_s) - 12.0)

    iteration = 0
    while time.monotonic() - start_time < max_time:
        # current bound
        M = max_convolution(w)

        # -----------------------------------------------------------------
        # LP step: maximise total mass of a feasible g (f*g ≤ M)
        # -----------------------------------------------------------------
        g_opt, mass = solve_lp_for_g(w, M)

        # If the LP cannot increase the total mass we are (almost) optimal.
        if mass <= 1.000_001:
            # No further improvement possible from LP – fall back to a
            # modest sub‑gradient move (tiny random transfer) to shake out of
            # possible plateaus.
            if iteration % 7 == 0:
                w = block_perturbation(w, rng)
                w /= np.sum(w)
            iteration += 1
            continue

        # Normalise the candidate (so it stays a probability vector)
        g_norm = g_opt / mass

        # -----------------------------------------------------------------
        # Try several mixing factors and keep the best mixture.
        # -----------------------------------------------------------------
        mix_factors = (0.1, 0.3, 0.5, 0.7, 0.9)
        local_best_w = None
        local_best_sc = np.inf
        for t in mix_factors:
            cand = (1.0 - t) * w + t * g_norm
            sc = 2.0 * n * max_convolution(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_w = cand

        # Accept the mixture if it improves the global best.
        if local_best_sc < best_score - 1e-9:
            best_score = local_best_sc
            best_w = local_best_w.copy()
            w = local_best_w.copy()
            print(f"[lp improve] iter={iteration:5d}, score={best_score:.6f}, mass={mass:.4f}")
            # occasional symmetry enforcement (helps the objective)
            if iteration % 13 == 0:
                sym = 0.5 * (w + w[::-1])
                sym /= sym.sum()
                sc_sym = 2.0 * n * max_convolution(sym)
                if sc_sym < best_score:
                    best_score = sc_sym
                    best_w = sym.copy()
                    w = sym.copy()
                    print(f"[sym] iter={iteration:5d}, improved to {best_score:.6f}")
        else:
            # No improvement – apply a small random block perturbation every few steps.
            if iteration % 11 == 0:
                w = block_perturbation(w, rng)
                w /= np.sum(w)

        iteration += 1

    # --------------------------------------------------------------
    # Final polishing: a short simulated‑annealing style walk.
    # --------------------------------------------------------------
    remaining = budget_s - (time.monotonic() - start_time)
    if remaining > 8.0:
        temperature = 0.02
        cur_w = best_w.copy()
        cur_score = best_score
        anneal_end = time.monotonic() + remaining - 3.0
        while time.monotonic() < anneal_end:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_transfer = min(cur_w[i], 0.02)
            delta = rng.uniform(0.0, max_transfer)
            if delta == 0.0:
                continue
            cand = cur_w.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = 2.0 * n * max_convolution(cand)
            if sc < cur_score or rng.random() < np.exp((cur_score - sc) / temperature):
                cur_w = cand
                cur_score = sc
                if cur_score < best_score:
                    best_score = cur_score
                    best_w = cur_w.copy()
            temperature *= 0.9999
            if temperature < 1e-5:
                temperature = 0.001
        print(f"[anneal] polishing finished, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Scale to the convention Σ a_i = √(2·n) and return as a list.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    # The external harness will call evaluate_sequence; we print the final score.
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best score={best_score:.6f}, after scaling evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
