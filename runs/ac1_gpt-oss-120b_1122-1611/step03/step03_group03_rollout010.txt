def solve_lp_for_g(f, M):
    n = len(f)
    conv_len = 2*n - 1
    # Build Toeplitz matrix C shape (conv_len, n) where C @ g = f * g.
    # Use scipy.sparse.diags to construct efficiently.
    # Approach: For each i in range(n):
    #   rows i..i+n-1 have f[i].
    # Represent using spdia with constant diagonal across columns.
    # Actually we can use circulant? but simpler: build dense matrix using np.zeros and fill columns.
    # For moderate n (<=2500) dense matrix is okay.

    # Let's create dense matrix A (conv_len, n) zero.
    A = np.zeros((conv_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]   # slice assignment
    # Now constraints: A @ g <= M
    # c = -np.ones(n) to maximize sum(g). But linprog by default minimizes c^T x, so use c = -np.ones(n).
    bounds = [(0, None)] * n
    res = linprog(c=np.full(n, -1.0), A_ub=A, b_ub=np.full(conv_len, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a low‑max‑convolution step‑function.

Algorithm
---------
* Choose a length *n* (use the length of the previously found
  `height_sequence_1` if available, otherwise 8000).
* Initialise a probability vector `w` (non‑negative, Σw = 1) either from the
  known best construction or from a Dirichlet (exponential) draw.
* Optimise *w* by a projected Adam scheme that minimises a smooth approximation
  of the infinity‑norm of the self‑convolution `w*w`.  The smooth maximum is
  obtained with a soft‑max (temperature *α*) that is slowly increased.
* Every few hundred iterations an exact sub‑gradient of the true max is
  mixed in to help push down the peak.
* Periodic random block perturbations and occasional symmetry enforcement
  increase diversity and avoid stagnation.
* The best vector seen so far is stored; if no improvement occurs for a long
  time a restart based on a perturbed version of the current best is performed.
* After the time budget the best vector is rescaled to the conventional
  √(2 n) L¹‑norm and returned as a plain Python list.

The evaluation metric
    2·n·max_k (w*w)[k]  (with Σw = 1)
is scale‑invariant, so we work with a probability vector and only
rescale at the end for readability.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan numbers scaled by 4⁻ⁱ: a_i = C(2i,i)/4ⁱ."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT; negative round‑off is clipped."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor.
    With probability 0.35 the result is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 300)
    hi = max(2, n // 10)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.15)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _grad_from_weights(p: np.ndarray, w_rev: np.ndarray, n: int) -> np.ndarray:
    """
    Gradient of Σ p_k (w*w)[k] w.r.t. w:
        2 * corr(p, w)   (size‑n slice of the full correlation)
    """
    full = convolve_fft(p, w_rev)                # length 3n‑2
    start = n - 1
    return 2.0 * full[start:start + n]


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that aims
    to minimise ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a length.  Prefer the length of a previously found
    #    construction if it is available.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 8000
    else:
        n = 8000

    # ------------------------------------------------------------------
    # 2. Initial probability vector w (Σw = 1, w ≥ 0)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init_arr = np.asarray(height_sequence_1, dtype=np.float64)
            if init_arr.size == n:
                w = np.maximum(init_arr, 0.0)
            else:
                raise ValueError
        except Exception:
            w = catalan_sequence(n)
    else:
        w = catalan_sequence(n)

    # Dirichlet perturbation – a small random nudge that destroys exact
    # integer structure while keeping the vector in the simplex.
    w = w + rng.exponential(scale=0.01, size=n)
    w = simplex_projection(w)

    # ------------------------------------------------------------------
    # Book‑keeping of the best vector seen so far.
    # ------------------------------------------------------------------
    conv = convolve_fft(w, w)
    best_score = float(2 * n * np.max(conv))
    best_w = w.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.25                     # base learning rate
    alpha = 30.0                   # soft‑max temperature
    alpha_inc = 1.03               # multiplicative increase
    subgrad_lambda = 0.5           # weight of exact max sub‑gradient
    subgrad_period = 200           # how often to add the exact sub‑gradient
    eval_interval = 250            # evaluate true objective this often
    block_interval = 1200          # random block jump frequency
    symmetry_interval = 1000       # enforce w ≈ w[::-1] occasionally
    restart_patience = 25000       # iterations without improvement → restart
    max_alpha = 5000.0

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ----- current convolution (needed for gradient & metrics) -----
        conv = convolve_fft(w, w)                     # length 2n‑1

        # ----- soft‑max weights over the convolution ------------------
        maxc = np.max(conv)
        exp_vals = np.exp(alpha * (conv - maxc))      # stabilised exp
        p_soft = exp_vals / np.sum(exp_vals)

        # ----- gradient from soft‑max --------------------------------
        w_rev = w[::-1]
        grad = _grad_from_weights(p_soft, w_rev, n)

        # ----- exact max sub‑gradient (occasional) -------------------
        if iteration % subgrad_period == 0:
            max_idx = np.flatnonzero(conv == maxc)
            p_max = np.zeros_like(conv)
            p_max[max_idx] = 1.0 / max_idx.size
            grad_max = _grad_from_weights(p_max, w_rev, n)
            grad = grad + subgrad_lambda * grad_max

        # ----- Adam update -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w = simplex_projection(w_new)

        # ----- symmetry enforcement (helps against edge effects) -----
        if iteration % symmetry_interval == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

        # ----- evaluate the true (non‑smooth) objective ---------------
        if iteration % eval_interval == 0:
            conv_now = convolve_fft(w, w)
            score = float(2 * n * np.max(conv_now))
            if score < best_score - 1e-12:
                best_score = score
                best_w = w.copy()
                no_improve = 0
                print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
            else:
                no_improve += 1

        # ----- random block perturbation -------------------------------
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            conv_cand = convolve_fft(cand, cand)
            cand_score = float(2 * n * np.max(conv_cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # ----- temperature schedule ------------------------------------
        if iteration % 500 == 0 and iteration > 0:
            alpha = min(alpha * alpha_inc, max_alpha)

        # ----- restart if we have been stuck for a while -------------
        if no_improve > restart_patience:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[restart] iter={iteration:6d}, refreshed from best")

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Final scaling – the evaluation is scale‑invariant; we adopt the
    #    conventional √(2 n) L¹‑norm for readability.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)
    np.clip(final_seq, 0.0, 1000.0, out=final_seq)

    # Optional debug print of the official score
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")
    except Exception:
        # In the harness evaluate_sequence is guaranteed to be present;
        # this block is defensive.
        pass

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
