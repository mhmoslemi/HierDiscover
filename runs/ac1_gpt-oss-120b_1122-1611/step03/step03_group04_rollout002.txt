"""
propose_candidate
-----------------

Hybrid optimisation for the “peak of self‑convolution” problem.

* Stage 1 – cheap parametric sweep:
  Search over a few simple symmetric families (U‑shaped power‑law and
  inverse‑product) and keep the shape giving the smallest raw score
  (2·n·max(w*w)).

* Stage 2 – entropic mirror descent (multiplicative updates) with a
  smoothed max‑objective (soft‑max).  The temperature (β) grows
  √iteration, the step‑size (η) decays as 1/√iteration.  The iterate
  is always projected onto the probability simplex, forced symmetric,
  and periodically perturbed by a random block‑scale jump to escape
  plateaus.

* Small Gaussian smoothing and occasional full‑reset are added for
  robustness.

The algorithm finishes within the allotted time and returns the best
vector found (scaled to the conventional √(2 n) ℓ¹‑norm).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution / correlation helpers (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = _next_pow2(n)
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negatives caused by numerical noise.
    np.maximum(c, 0.0, out=c)
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        g = 2 * corr(w, alpha)

    alpha is a probability vector of length 2·n−1.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = 3·n−2
    start = w.shape[0] - 1                     # aligns j=0
    return 2.0 * conv[start:start + w.shape[0]]


# ----------------------------------------------------------------------
# Simple parametric families (U‑shaped) for the sweep
# ----------------------------------------------------------------------


def ushape_product(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    Symmetric U‑shape w_i ∝ ((i+0.5)*(n‑i‑0.5) + eps)⁻ᵖ .
    p = 0 gives uniform, larger p pushes mass toward the ends.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5) + eps
    w = prod ** (-p)
    return w


def ushape_sum(n: int, p: float, eps: float = 1e-12) -> np.ndarray:
    """
    Symmetric U‑shape w_i ∝ (i+0.5)⁻ᵖ + (n‑i‑0.5)⁻ᵖ .
    """
    i = np.arange(n, dtype=np.float64)
    left = (i + 0.5) ** (-p) + eps
    right = (n - i - 0.5) ** (-p) + eps
    w = left + right
    return w


# ----------------------------------------------------------------------
# Random block perturbation (cheap O(n) jump)
# ----------------------------------------------------------------------


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 % of length
    hi = max(2, n // 12)           # up to ~8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation of the perturbed block
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):
    """
    Return a non‑negative step function (list of floats) with a low
    evaluate_sequence() score.  The routine combines a fast parametric
    sweep with a long entropic mirror‑descent phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose length n and initial vector w (probability simplex).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.ndim != 1 or prev.size < 2:
                raise ValueError
            n = int(prev.size)
            w = prev / np.sum(prev)                      # normalise to Σ=1
            print(f"[init] Using previous sequence, n={n}")
        except Exception:
            n = 8000
            w = None
            print("[init] Failed to use previous sequence – falling back to sweep")
    else:
        n = 8000
        w = None
        print(f"[init] No previous sequence – using n={n}")

    # --------------------------------------------------------------
    # Stage 1 – fast parametric sweep (U‑shapes)
    # --------------------------------------------------------------
    # We only sweep if we do not already have a good candidate.
    # The sweep is cheap (≈80 FFTs) and often yields a better start.
    best_score = np.inf
    best_w = None

    # Evaluate any supplied w from previous run.
    if w is not None:
        conv = convolve_fft(w, w)
        best_score = 2.0 * n * np.max(conv)
        best_w = w.copy()
        print(f"[sweep] Score of previous vector = {best_score:.6f}")

    # Parameter grid
    p_vals = np.linspace(0.0, 2.0, 41)  # 0 … 2 step 0.05

    for shape_func, name in [(ushape_product, "product"), (ushape_sum, "sum")]:
        for p in p_vals:
            cand = shape_func(n, p)
            # Normalise to sum=1 (the objective is scale‑invariant)
            cand_sum = cand.sum()
            if cand_sum == 0.0 or not np.isfinite(cand_sum):
                continue
            cand /= cand_sum
            conv = convolve_fft(cand, cand)
            score = 2.0 * n * np.max(conv)
            if score < best_score - 1e-15:
                best_score = score
                best_w = cand.copy()
                print(f"[sweep] New best ({name}, p={p:.3f}) → score={score:.6f}")

    # If the sweep did not improve the previous vector (or there was no
    # previous vector), initialise w with the sweep result.
    if best_w is None:
        # safety fallback – uniform distribution
        best_w = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * np.max(convolve_fft(best_w, best_w))
        print(f"[fallback] Uniform start, score={best_score:.6f}")
    w = best_w.copy()
    best_w = w.copy()
    print(f"[init] Starting mirror‑descent, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – entropic mirror descent with soft‑max smoothing
    # --------------------------------------------------------------
    # Hyper‑parameters (tuned empirically)
    eta0 = 1.0                # base step size for multiplicative updates
    beta0 = 0.3               # temperature prefactor (β = beta0·√iter)
    beta_max = 500.0
    reset_patience = 30000   # iterations without improvement before a reset
    block_every = 5000       # cheap random jump interval
    smooth_every = 2000      # Gaussian smoothing interval
    symmetrize_every = 1     # we keep the vector symmetric each iteration
    tolerance = 1e-12

    # Simple 3‑point Gaussian smoother (kernel sums to 1)
    smooth_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # Pre‑allocate a few arrays that will be reused to minimise allocation.
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ----------------------------------------------------------
        # 1. Self‑convolution, current (hard) score
        # ----------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - tolerance:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # 2. Soft‑max weights α over the 2·n−1 convolution entries
        # ----------------------------------------------------------
        beta = beta0 * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b                      # max element becomes 0
        # stable soft‑max (shift already performed)
        exp_term = np.exp(beta * diff)
        alpha = exp_term / exp_term.sum()     # probability vector, length 2·n−1

        # ----------------------------------------------------------
        # 3. Gradient of the smooth objective
        # ----------------------------------------------------------
        grad = correlation_grad(w, alpha)     # shape (n,)

        # ----------------------------------------------------------
        # 4. Multiplicative (entropic) mirror‑descent update
        # ----------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1)
        # Clip gradient magnitude to avoid overflow in the exponent.
        np.clip(grad, -50.0, 50.0, out=grad)
        w *= np.exp(-eta * grad)

        # Re‑project onto the probability simplex (non‑negative, sum=1)
        s = w.sum()
        if s <= 0.0 or not np.isfinite(s):
            w.fill(1.0 / n)
        else:
            w /= s

        # ----------------------------------------------------------
        # 5. Enforce symmetry (optimal solution is symmetric)
        # ----------------------------------------------------------
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= w.sum()

        # ----------------------------------------------------------
        # 6. Occasional Gaussian smoothing (helps keep the shape smooth)
        # ----------------------------------------------------------
        if iteration % smooth_every == 0 and iteration > 0:
            w = np.convolve(w, smooth_kernel, mode="same")
            w = np.maximum(w, 0.0)
            w /= w.sum()

        # ----------------------------------------------------------
        # 7. Random block perturbation (escape plateaus)
        # ----------------------------------------------------------
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()
            cand_score = float(2.0 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best score={best_score:.6f}")

        # ----------------------------------------------------------
        # 8. Reset if stuck for too long
        # ----------------------------------------------------------
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w /= w.sum()
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – evaluation is scale‑invariant, but we keep the
    # conventional √(2 n) ℓ¹‑norm for nicer numbers.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()

    # Optional sanity check (does not affect the returned value)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:
        final_score = float("nan")
    print(f"[final] n={n}, best raw score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
