L∞‑self‑convolution problem.

The algorithm starts from an analytically motivated “arcsine”

def arcsine_sequence(length: int) -> np.ndarray:
    """
    Discrete arcsine law:
        a_i ∝ 1 / sqrt( (i+0.5) * (length-0.5-i) )
    The endpoints are finite because of the +0.5 shift.
    """
    i = np.arange(length, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (length - 0.5 - i))
    return vals


def catalan_sequence(length: int) -> np.ndarray:
    """Fallback: first ``length`` Catalan numbers divided by 4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (fast for n up to a few 10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off artifacts
    np.maximum(c, 0.0, out=c)
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation (log‑normal scaling)."""
    cand = base.copy()
    n = cand.size
    # block width is a modest fraction of the length
    lo = max(1, n // 300)
    hi = max(2, n // 30)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.13)))  # log‑normal multiplier
    cand[start:stop] *= factor
    # optional symmetric averaging
    if rng.random() < 0.33:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max probabilities p_k ∝ exp(conv_k / γ).  Returns a vector that sums to 1."""
    shift = conv - np.max(conv)           # for numerical stability
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float):
    """
    Gradient of the soft‑max‑smoothed max‑convolution.
    Returns (grad, max_convolution_entry).
    """
    n = w.size
    # 1. self‑convolution
    b = convolve_fft(w, w)                   # length 2n‑1
    # 2. soft‑max weights
    p = _softmax_weights(b, gamma)           # same length
    # 3. reverse the weight vector
    p_rev = p[::-1]
    # 4. convolution w * p_rev
    conv = convolve_fft(w, p_rev)            # length 3n‑2
    # 5. extract the first n entries (re‑aligned) and scale
    grad = 2.0 * conv[::-1][:n]
    return grad, np.max(b)


def _hard_max_subgradient(w: np.ndarray, conv: np.ndarray):
    """
    Exact sub‑gradient of   f(w)=max_k (w*w)_k .
    ``conv`` must be the linear convolution w*w (length 2n‑1).
    The gradient is 2 * Σ_{k∈K} shift_k(w) where K = {k : conv_k = max}.
    """
    n = w.size
    max_val = np.max(conv)
    # select all indices attaining the maximum (within a tiny tolerance)
    idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]
    grad = np.zeros_like(w)
    for k in idxs:
        if k < n:
            # overlap i = 0 … k
            grad[:k + 1] += 2.0 * w[k::-1]
        else:
            # overlap i = k-(n-1) … n-1
            start = k - (n - 1)
            length = n - start
            grad[start:] += 2.0 * w[:length][::-1]
    return grad

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step function (list of floats) whose
    ``evaluate_sequence`` score is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # Choose a problem size.
    # Larger n gives a smaller theoretical lower bound, but also a
    # larger per‑iteration cost.  10 000 is a good compromise for the
    # 1000 s budget on a typical 2‑core machine.
    # --------------------------------------------------------------
    n = 10000

    # --------------------------------------------------------------
    # Build the initial probability vector.
    # Use the analytical arcsine law if possible, otherwise fall back
    # to Catalan numbers.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= 2:
                w = prev.copy()
                w = np.maximum(w, 0.0)
                w_sum = np.sum(w)
                if w_sum == 0.0:
                    raise ValueError
                w = w / w_sum
            else:
                raise ValueError
        except Exception:
            w = arcsine_sequence(n)
            w = w / np.sum(w)
    else:
        w = arcsine_sequence(n)
        w = w / np.sum(w)

    # ensure symmetry – the problem is symmetric under reversal
    w = 0.5 * (w + w[::-1])

    # --------------------------------------------------------------
    # State variables for the best solution seen so far.
    # The score we minimise is 2·n·max(b) (because w already sums to 1).
    # --------------------------------------------------------------
    conv = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(conv)
    best_w = w.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameter schedules (linear annealing over the allotted time)
    # --------------------------------------------------------------
    total_budget = float(budget_s) - 2.0                # keep a tiny safety margin
    max_exponent = 50.0                                 # clamp exponent in EG updates
    sym_interval = 200                                  # enforce symmetry every … iterations
    block_interval = 500                                # try a random block jump every …
    stagnation_limit = 40000                           # reset after many non‑improving steps

    # schedule endpoints
    gamma_start = 0.25          # initial γ = gamma_start * max_b
    gamma_end = 0.02            # final γ = gamma_end * max_b
    eta_start = 1.0             # initial step size for EG
    eta_end = 0.08              # final step size

    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop – runs until the time budget is exhausted.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < total_budget:
        # fraction of elapsed budget (0 → 1)
        elapsed = time.monotonic() - start_time
        t_frac = min(elapsed / total_budget, 1.0)

        # linear interpolation of hyper‑parameters
        gamma_factor = gamma_start * (1.0 - t_frac) + gamma_end * t_frac
        eta = eta_start * (1.0 - t_frac) + eta_end * t_frac

        # ------------------------------------------------------------------
        # (a) evaluate the current vector (needed for the score and for γ)
        # ------------------------------------------------------------------
        conv = convolve_fft(w, w)
        max_b = np.max(conv)
        cur_score = 2.0 * n * max_b

        # ------------------------------------------------------------------
        # (b) record improvement
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # (c) choose a gradient direction
        # ------------------------------------------------------------------
        if iteration % 7 == 0:
            # soft‑max‑smoothed gradient (most of the time)
            gamma = max(gamma_factor * max_b, 1e-12)
            grad, _ = _smoothed_gradient(w, gamma)
        else:
            # hard max‑sub‑gradient – focuses exactly on the peaks
            grad = _hard_max_subgradient(w, conv)

        # ------------------------------------------------------------------
        # (d) exponentiated‑gradient update (mirror descent on the simplex)
        # ------------------------------------------------------------------
        exponent = -eta * grad
        np.clip(exponent, -max_exponent, max_exponent, out=exponent)
        w = w * np.exp(exponent)
        w_sum = np.sum(w)
        if w_sum == 0.0:                     # safety fallback
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # ------------------------------------------------------------------
        # (e) periodic symmetry enforcement (helps to keep the shape balanced)
        # ------------------------------------------------------------------
        if iteration % sym_interval == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])

        # ------------------------------------------------------------------
        # (f) occasional random block perturbation – keep it if it improves
        # ------------------------------------------------------------------
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # ------------------------------------------------------------------
        # (g) long‑run stagnation handling – reset from a perturbed best
        # ------------------------------------------------------------------
        if no_improve > stagnation_limit:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluator expects the conventional √(2 n) L¹‑norm.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum

    # ------------------------------------------------------------------
    # A sanity check (silent unless an exception occurs)
    # ------------------------------------------------------------------
    try:
        final_val = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best score={best_score:.6f}, evaluate={final_val:.6f}")
    except Exception:
        # In dry‑run environments evaluate_sequence may not be present.
        print(f"[final] n={n}, best score={best_score:.6f}")

    # Store the best sequence globally for possible reuse by a next run.
    globals()["height_sequence_1"] = final_seq.tolist()

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
