"""
candidate_generator
-------------------
A fast search for a low‑value step function based on the
family

    a_i(p) ∝ 1 / [(i+1)^p (n-i)^p] , 0 < p < ½ .

For p = ¼ the L¹‑norm scales like √n and the self‑convolution
is essentially constant (no logarithmic blow‑up), giving a
much smaller max‑convolution value than the Catalan construction.
The routine scans a few lengths n and a dense set of p‑values,
keeps the best sequence and finally returns it (already
normalised to Σ a_i = √(2 n) as required by evaluate_sequence).

The whole search stays well below the 1000 s budget
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT.
    Equivalent to np.convolve(a, b) but O(n log n).
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off errors
    np.maximum(c, 0.0, out=c)
    return c


def beta_shape(n: int, p: float) -> np.ndarray:
    """
    Return an un‑normalised vector of length n with entries
        a_i = 1 / ((i+1)^p * (n-i)^p) .
    The vector is symmetric and vanishes slowly at the ends.
    """
    i = np.arange(n, dtype=np.float64)
    # (i+1) and (n-i) are never zero for i = 0 … n‑1
    a = 1.0 / ((i + 1) ** p * (n - i) ** p)
    return a


def normalise_to_target(a: np.ndarray) -> np.ndarray:
    """
    Scale a vector so that its L¹ norm becomes √(2 n).
    """
    n = a.size
    target = np.sqrt(2.0 * n)
    scale = target / np.sum(a)
    a_scaled = a * scale
    # the evaluation harness clips values >1000; keep us safe
    np.clip(a_scaled, 0.0, 1000.0, out=a_scaled)
    return a_scaled


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search over the beta‑shape family a_i(p) to find a sequence
    with a low evaluate_sequence() value.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Reserve a few seconds for the final call to evaluate_sequence()
    max_search_time = max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Parameter grid.
    # ------------------------------------------------------------------
    # Lengths to try – a few powers of two give a good trade‑off.
    n_candidates = [2000, 4000, 6000, 8000, 10000, 12000, 15000]
    # p values – denser near the theoretically promising 0.25.
    p_grid = np.concatenate(
        [np.linspace(0.15, 0.30, 32, endpoint=False),
         np.linspace(0.30, 0.35, 16)]
    )

    best_score = np.inf
    best_seq = None
    best_n = None
    best_p = None

    print("[search] start – budget {:.1f}s".format(max_search_time))

    for n in n_candidates:
        # stop early if we are out of time
        if time.monotonic() - start_time > max_search_time:
            break

        # Pre‑compute the denominator that does not depend on p
        i = np.arange(n, dtype=np.float64)
        base_left = i + 1                # (i+1)
        base_right = n - i               # (n-i)

        for p in p_grid:
            if time.monotonic() - start_time > max_search_time:
                break

            # Construct the candidate shape
            a = 1.0 / (base_left ** p * base_right ** p)

            # Normalise so that Σ a_i = √(2 n)
            a = normalise_to_target(a)

            # Compute the maximum self‑convolution value.
            # Since the vector is already L¹‑normalised, the evaluation
            # result equals max(b) (the scaling factor in evaluate_sequence
            # cancels out).
            b = convolve_fft(a, a)
            max_b = np.max(b)

            # Direct objective (identical to evaluate_sequence output)
            score = max_b

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                best_n = n
                best_p = p
                print("[improve] n={:5d}  p={:.4f}  score={:.6f}".format(n, p, score))

        # ------------------------------------------------------------------
        # Small random refinement around the current best for this n.
        # ------------------------------------------------------------------
        if best_seq is not None and best_n == n:
            # try a few random perturbations of p in a narrow window
            for _ in range(30):
                if time.monotonic() - start_time > max_search_time:
                    break
                p_new = best_p + rng.normal(0.0, 0.015)  # narrow Gaussian jitter
                if not (0.10 < p_new < 0.45):
                    continue
                a = 1.0 / (base_left ** p_new * base_right ** p_new)
                a = normalise_to_target(a)
                b = convolve_fft(a, a)
                score = np.max(b)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    best_p = p_new
                    print("[jitter] n={:5d}  p={:.4f}  score={:.6f}".format(n, p_new, score))

    # ----------------------------------------------------------------------
    # If the search failed for any reason, fall back to the previous best
    # (or a simple uniform distribution) – this guarantees a valid return.
    # ----------------------------------------------------------------------
    if best_seq is None:
        if 'height_sequence_1' in globals():
            print("[fallback] using provided height_sequence_1")
            return height_sequence_1
        else:
            n = 6000
            a = np.full(n, 1.0)
            a = normalise_to_target(a)
            return a.tolist()

    # ----------------------------------------------------------------------
    # Final sanity check with the official evaluator (once only).
    # ----------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print("[final] length={:5d}  p={:.4f}  best_score={:.6f}  evaluate={:.6f}"
          .format(best_n, best_p, best_score, final_score))

    # Convert to plain Python list for the harness.
    return [float(x) for x in best_seq]
