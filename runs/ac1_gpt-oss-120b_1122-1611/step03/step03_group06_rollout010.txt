import cvxpy as cp
def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # indices of g: j = i, f_coef = f[k-i]
        # Build linear expression sum_{i=i_start}^{i_end} f[k-i] * g[i] <= M
        expr = cp.sum(cp.multiply(f[k - np.arange(i_start, i_end+1)], g[i_start:i_end+1]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return g.value, prob.value
```

"""
propose_candidate
-----------------
Hybrid optimizer for the non‑negative step‑function problem.

* Stage 1 – Adam‑gradient descent on a smooth log‑sum‑exp surrogate of the
  self‑convolution maximum (the same idea as the reference code, but with
  Adam, a more aggressive β‑schedule and multiple random restarts).

* Stage 2 – After a few thousand Adam steps we solve a small linear program:
  for the current vector f we maximise Σ g_j under the linear constraints
  (f∗g)_k ≤ M (M = max(f∗f)).  The solution g tends to concentrate mass in
  “cheap’’ positions, allowing a convex combination f←(1‑t)f+t g to lower the
  actual max‑convolution value.  The LP is solved with SciPy’s HiGHS solver.

The whole procedure is time‑boxed (default 1000 s).  Several independent
restarts are performed; the best vector (according to the exact
`evaluate_sequence` value) is finally rescaled to the required L¹‑norm
√(2 n) and returned.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First `length` Catalan‑type coefficients a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)      # eliminate tiny negative round‑off errors
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation used as a “jump’’.
    A random contiguous block is multiplied by a log‑normal factor.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetrisation – useful for escaping plateaus
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def softmax_gradient(w: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate Sβ(w) = (1/β)·log Σ exp(β·(w∗w)[k]).
    Returns the gradient ∇Sβ (size = len(w)).
    """
    b = convolve_fft(w, w)
    max_b = np.max(b)
    exp_arg = beta * (b - max_b)               # ≤ 0  (numerically stable)
    p = np.exp(exp_arg)
    p_sum = np.sum(p)
    if p_sum == 0.0:                           # extreme β → underflow
        p = np.full_like(b, 1.0 / b.size)
        p_sum = 1.0
    p /= p_sum                                 # Σ p = 1
    grad = 2.0 * convolve_fft(p, w)[:w.size]
    return grad


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the linear operator A with rows = (f∗g)_k  (k=0..2n‑2) such that
    A @ g = f ∗ g  (full convolution).  A is stored as a CSR sparse matrix.
    """
    n = f.size
    rows = 2 * n - 1
    # There are n² non‑zero entries – build them in three flat arrays.
    data = np.empty(n * n, dtype=np.float64)
    row_ind = np.empty(n * n, dtype=np.int32)
    col_ind = np.empty(n * n, dtype=np.int32)

    ptr = 0
    for i in range(n):
        # column indices 0 … n‑1
        cols = np.arange(n, dtype=np.int32)
        rows_i = i + cols                     # row = i + column
        data[ptr:ptr + n] = f[i]
        row_ind[ptr:ptr + n] = rows_i
        col_ind[ptr:ptr + n] = cols
        ptr += n
    A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    return A


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve max Σ g_j  subject to g ≥ 0 and (f∗g)_k ≤ M  (for all k).
    Returns the optimal g (not normalised) or None if the solver fails.
    """
    n = f.size
    A = build_conv_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -Σ g ⇒ maximise Σ g
    # HiGHS via scipy.optimize.linprog is very fast on these modest sizes.
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if res.success:
        return res.x
    else:
        print("[lp] solver failed:", res.message)
        return None

        w0 = w0 / np.sum(w0)

    # ------------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------------
    best_w = w0.copy()
    best_score = 2.0 * n * np.max(convolve_fft(best_w, best_w))

    print(f"[init] length={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    # keep a few seconds free for final scaling / printing
    max_time = max(0.0, float(budget_s) - 5.0)

    # -----------------------------
    # Adam hyper‑parameters
    # -----------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.42           # moderate base learning rate
    t_mix = 0.25             # mixing factor for LP update
    beta = 1.0               # start of the soft‑max smoothing
    beta_max = 800.0
    beta_inc = 1.04
    beta_inc_interval = 2500

    # --------------------------------------------------------------
    # Multiple independent restarts (time‑sliced)
    # --------------------------------------------------------------
    restart_budget = max_time / 3.0   # three major phases
    while time.monotonic() - start_time < max_time:
        # ----------------------------------------------------------
        # 1) initialise a fresh start (random Dirichlet or Catalan)
        # ----------------------------------------------------------
        if rng.random() < 0.5:
            # Dirichlet with parameter 0.7 → slightly spiky but still dense
            alpha = 0.7
            w = rng.gamma(alpha, 1.0, size=n)
        else:
            w = catalan_sequence(n).astype(np.float64)
        w = w / np.sum(w)

        m = np.zeros_like(w)        # Adam first moment
        v = np.zeros_like(w)        # Adam second moment
        iteration = 0
        local_best_w = w.copy()
        local_best_score = 2.0 * n * np.max(convolve_fft(w, w))
        phase_start = time.monotonic()

        while time.monotonic() - phase_start < restart_budget:
            # ------------------------------------------------------
            # Gradient descent step (Adam + simplex projection)
            # ------------------------------------------------------
            grad = softmax_gradient(w, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (iteration + 1))
            v_hat = v / (1.0 - beta2 ** (iteration + 1))
            lr = base_lr / np.sqrt(iteration + 1.0)   # diminishing step size
            w_candidate = w - lr * m_hat / (np.sqrt(v_hat) + eps)
            w = simplex_projection(w_candidate)

            # ------------------------------------------------------
            # Record progress
            # ------------------------------------------------------
            b = convolve_fft(w, w)
            cur_score = 2.0 * n * np.max(b)
            if cur_score < local_best_score - 1e-9:
                local_best_score = cur_score
                local_best_w = w.copy()
                if cur_score < best_score - 1e-9:
                    best_score = cur_score
                    best_w = w.copy()
                print(f"[improve] iter={iteration:7d}, score={cur_score:.6f}")

            # ------------------------------------------------------
            # β‑schedule (smooth max → true max)
            # ------------------------------------------------------
            if iteration % beta_inc_interval == 0 and beta < beta_max:
                beta = min(beta_max, beta * beta_inc)

            # ------------------------------------------------------
            # Random block perturbation to avoid stagnation
            # ------------------------------------------------------
            if iteration % 5000 == 0 and iteration > 0:
                w = block_perturbation(local_best_w, rng)
                w = w / np.sum(w)
                m.fill(0.0)
                v.fill(0.0)
                print(f"[perturb] iter={iteration:7d}")

            # ------------------------------------------------------
            # Linear‑programming refinement (occasionally)
            # ------------------------------------------------------
            if iteration % 12000 == 0 and iteration > 0:
                M = np.max(b)
                g = solve_lp_max_sum(w, M)
                if g is not None:
                    sum_g = np.sum(g)
                    if sum_g > 0.0:
                        g = g / sum_g                     # renormalise to sum = 1
                        # convex combination (mixing)
                        w = (1.0 - t_mix) * w + t_mix * g
                        w = simplex_projection(w)
                        # recompute score after mixing
                        b = convolve_fft(w, w)
                        mix_score = 2.0 * n * np.max(b)
                        if mix_score < local_best_score - 1e-9:
                            local_best_score = mix_score
                            local_best_w = w.copy()
                            if mix_score < best_score - 1e-9:
                                best_score = mix_score
                                best_w = w.copy()
                            print(f"[lp‑mix] iter={iteration:7d}, score={mix_score:.6f}")

            # ------------------------------------------------------
            # Termination check
            # ------------------------------------------------------
            iteration += 1
            if time.monotonic() - start_time >= max_time:
                break

        # end of a restart – keep the best of this run
        if local_best_score < best_score - 1e-9:
            best_score = local_best_score
            best_w = local_best_w.copy()
        # reduce remaining budget for the next restart
        elapsed = time.monotonic() - start_time
        if elapsed >= max_time:
            break
        # shrink the per‑restart budget a little (to guarantee we finish)
        restart_budget = max(0.0, (max_time - elapsed) / max(1, (3 - 0)))  # simple heuristic

    # ------------------------------------------------------------------
    # Final scaling – the evaluation harness expects Σ a = √(2 n).
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum
    final_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"[final] length={n}, best_score≈{best_score:.6f}, "
          f"evaluate={final_score:.6f}")

    # Convert to plain Python list for the harness.
    return [float(x) for x in final_seq]
