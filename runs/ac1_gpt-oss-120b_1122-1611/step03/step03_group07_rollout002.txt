"""
Hybrid multi‑start search for a low‑autocorrelation step function.

The algorithm proceeds in three stages:

1. **Initialisation** – generate a portfolio of candidate probability
   vectors (Catalan, power‑law, exponential, Gaussian, Dirichlet draws,
   plus the previous best if available).  The one with the smallest
   ‑max‑convolution score is kept.

2. **Adam optimisation on the log‑probabilities** – we minimise the
   smooth approximation 
       f_γ(w)=γ·log Σ_k exp((w*w)[k]/γ)
   using Adam on the unconstrained variables 
       y_i = log w_i .
   The gradient ∇_w f_γ is obtained analytically via a soft‑max‑weighted
   convolution, and the chain rule yields ∇_y f_γ = w·(∇_w f_γ –
   (w·∇_w f_γ)).  After each update we recover w by a numerically
   stable soft‑max and optionally enforce symmetry w←(w+w[::-1])/2.

3. **Exploration jumps** – every few thousand iterations a cheap random
   block perturbation is applied to the current best vector; if it
   improves the score the state is reset to that perturbation.  A
   stagnation counter triggers a reset to a noisy copy of the best
   solution.

All convolutions are performed with an FFT‑based routine
`convolve_fft` (O(N log N)).  The search runs until the allocated time
budget (≈1000 s) is exhausted and finally returns the best vector
scaled to the conventional √(2 n) L¹‑norm required by
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – pure top‑level definitions.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for up to ~10⁴ points)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # clip minuscule negatives caused by round‑off
    np.maximum(c, 0.0, out=c)
    return c


def softmax(y: np.ndarray) -> np.ndarray:
    """Numeric‑stable soft‑max; returns a probability vector that sums to 1."""
    shift = y - np.max(y)
    exp = np.exp(shift)
    s = np.sum(exp)
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(y, 1.0 / y.size)
    return exp / s


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max weights p ∝ exp(conv/γ) that sum to 1."""
    shift = conv - np.max(conv)           # improve stability
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float, N_fft: int):
    """
    Return (∇_w f_γ(w), max(b)), where f_γ is the soft‑max‑smoothed max‑convolution.
    The gradient is 2·Σ_k p_k·shift_k(w) with p_k the soft‑max weights.
    """
    # 1. self‑convolution w*w
    w_fft = np.fft.rfft(w, N_fft)
    b = np.fft.irfft(w_fft * w_fft, N_fft)[:2 * w.size - 1]

    # 2. soft‑max probabilities over the convolution entries
    p = _softmax_weights(b, gamma)           # length 2n‑1, sums to 1

    # 3. reverse the probability vector and convolve with w
    p_rev = p[::-1]
    p_rev_fft = np.fft.rfft(p_rev, N_fft)
    grad_fft = w_fft * p_rev_fft
    conv = np.fft.irfft(grad_fft, N_fft)[:2 * w.size - 1]

    # 4. gradient (first n entries, reversed to align indices)
    grad = 2.0 * conv[::-1][: w.size]
    max_b = np.max(b)

    return grad, max_b


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Low‑cost multiplicative perturbation of a random contiguous block.
    The factor is drawn from a log‑normal distribution.
    """
    cand = base.copy()
    n = cand.size
    # block length between 0.5 % and 8 % of the vector
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1.0, out=cand)      # stay on simplex before renorm
    return cand

# Families for the multi‑start initialisation.
# ----------------------------------------------------------------------


def catalan_sequence(n: int) -> np.ndarray:
    """Catalan numbers divided by 4**i, normalised to sum‑1."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    seq /= seq.sum()
    return seq


def power_law_sequence(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ (i+1)^(-α), normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = 1.0 / np.power(i, alpha)
    seq /= seq.sum()
    return seq


def exponential_sequence(n: int, lam: float) -> np.ndarray:
    """w_i ∝ exp(-λ·i), normalised."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    seq /= seq.sum()
    return seq


def gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, normalised."""
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    seq /= seq.sum()
    return seq


def dirichlet_sequence(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a Dirichlet(α) distribution."""
    seq = rng.gamma(alpha, 1.0, size=n)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the evaluation score.  Returns the best sequence found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 8000
    else:
        n = 8000                     # a comfortable default (> previous 6733)

    # FFT length (next power of two ≥ 2n‑1)
    N_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Generate a pool of initial candidates and pick the best.
    # --------------------------------------------------------------
    def score_of(w):
        """Return 2·n·max(w*w) – the value that `evaluate_sequence` will finally output."""
        b = convolve_fft(w, w)
        return 2.0 * n * np.max(b)

    candidates = []

    # 1. prior best (if present)
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            if cand.size == n:
                cand = cand / cand.sum()
                candidates.append(cand)
        except Exception:
            pass

    # 2. Catalan
    candidates.append(catalan_sequence(n))

    # 3. Power‑law family
    for alpha in (1.0, 1.3, 1.5, 1.7, 2.0):
        candidates.append(power_law_sequence(n, alpha))

    # 4. Exponential family
    for lam in (0.0005, 0.001, 0.002, 0.003):
        candidates.append(exponential_sequence(n, lam))

    # 5. Gaussian family
    for sigma in (n / 12.0, n / 8.0, n / 5.0):
        candidates.append(gaussian_sequence(n, sigma))

    # 6. Random Dirichlet draws
    for a in (0.3, 0.8, 1.5, 3.0):
        for _ in range(2):
            candidates.append(dirichlet_sequence(n, a, rng))

    # Evaluate all candidates (cheap, only a convolution each)
    best_w = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        sc = score_of(cand)
        if sc < best_score:
            best_score = sc
            best_w = cand.copy()
        # tiny progress report
        if idx % 5 == 0:
            print(f"[init‑eval] candidate {idx:2d}, score={sc:.6f}")

    # --------------------------------------------------------------
    # Optimisation parameters (Adam on log‑probabilities).
    # --------------------------------------------------------------
    y = np.log(best_w + 1e-30)                 # log‑probability vector
    m = np.zeros_like(y)                      # first moment
    v = np.zeros_like(y)                      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.12                                # base learning rate
    gamma_factor = 0.07                        # temperature multiplier
    sym_interval = 400                         # enforce symmetry every …
    pert_interval = 2500                       # block‑jump every …
    no_improve_limit = 25000
    no_improve = 0
    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- recover probability vector from y -----
        w = softmax(y)

        # ----- evaluate current objective -----
        b = convolve_fft(w, w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b

        # ----- keep global best -----
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ----- compute smoothed gradient -----
        gamma = max(gamma_factor * max_b, 1e-12)
        grad_w, _ = _smoothed_gradient(w, gamma, N_fft)

        # ----- transform gradient to log‑prob space (chain rule) -----
        dot = np.dot(w, grad_w)
        grad_y = w * (grad_w - dot)          # ∂f/∂y

        # ----- Adam update on y -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        t = iteration + 1.0
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = eta0 / np.sqrt(t)               # simple decay
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- optional symmetrisation -----
        if iteration % sym_interval == 0 and iteration > 0:
            y = 0.5 * (y + y[::-1])

        # ----- occasional random block perturbation (jump) -----
        if iteration % pert_interval == 0 and iteration > 0:
            cand_w = block_perturbation(best_w, rng)
            cand_w /= cand_w.sum()
            cand_score = score_of(cand_w)
            if cand_score < best_score:
                best_w = cand_w.copy()
                best_score = cand_score
                y = np.log(cand_w + 1e-30)
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[jump]   iter={iteration:6d}, new best={best_score:.6f}")

        # ----- stagnation handling -----
        if no_improve > no_improve_limit:
            # restart from a noisy copy of the best solution
            noise = rng.normal(0.0, 0.02, size=n)
            y = np.log(best_w + 1e-30) + noise
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset]  iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling to the conventional √(2 n) L¹‑norm.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum

    # Basic sanity check (will be silent in the final evaluation environment)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best score={best_score:.6f}, evaluate={final_score:.6f}")
    except Exception:
        print(f"[final] length={n}, best score={best_score:.6f}")

    # Return plain Python list as required by the harness
    return [float(v) for v in final_seq]
