def optimize(w, time_left):
    start = time.monotonic()
    iteration = 0
    best_w = w.copy()
    best_score = compute_score(w)
    # hyperparameters
    eta0 = 0.8
    beta0 = 0.5
    beta_max = 100.0
    beta1,beta2 = 0.9,0.999
    eps = 1e-8
    block_every = 3000
    symmetrize_every = 500
    while time.monotonic() - start < time_left:
        # compute conv and score
        conv = convolve_fft(w,w)
        max_b = np.max(conv)
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_w = w.copy()
            # maybe print improvement
            print(f"[improve] iter={iteration} score={best_score}")
        # compute gradient
        beta = beta0 * np.sqrt(iteration+1)
        beta = min(beta, beta_max)
        diff = conv - max_b
        exp_term = np.exp(beta * diff)
        alpha = exp_term / exp_term.sum()
        grad = correlation_grad(w, alpha)
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        eta = eta0 / np.sqrt(iteration+1)
        w = w * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        # renorm
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/n)
        else:
            w /= w_sum
        # symmetrize
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= w.sum()
        # block perturb
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
        iteration += 1
        # maybe early break if time about up
        if time.monotonic() - start > time_left:
            break
    return best_w, best_score

        except Exception:
            n = 8000
    else:
        n = 8000

    # generate initial candidates
    init_candidates = []
    # catalan
    init_candidates.append(catalan_sequence(n))
    # powerlaw exponents
    for alpha in [0.55, 0.6, 0.65, 0.7]:
        seq = np.arange(1, n+1, dtype=np.float64) ** (-alpha)
        init_candidates.append(seq)
    # gaussian
    x = np.arange(n, dtype=np.float64)
    sigma = n / 5.0
    seq = np.exp(-((x - (n-1)/2)**2) / (2 * sigma**2))
    init_candidates.append(seq)

    # uniform
    init_candidates.append(np.ones(n, dtype=np.float64))

    best_w_global = None
    best_score_global = float('inf')
    start_time = time.monotonic()
    # allocate time proportionally: each candidate gets at most remaining_time/len(candidates)
    for idx, seq in enumerate(init_candidates):
        # normalize to sum 1
        w = seq.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0: continue
        w = w / w_sum
        # compute remaining time
        remaining = max(0.0, budget_s - (time.monotonic() - start_time) - 10)
        if remaining <= 2:
            break
        # allocate time for this run, maybe 0.2*remaining or fixed 120 sec
        alloc_time = min(remaining, max(5.0, remaining/len(init_candidates)))
        try:
            best_w, best_score = optimize(w, alloc_time)
        except Exception as e:
            print("Error during optimize:", e)
            continue
        if best_score < best_score_global:
            best_score_global = best_score
            best_w_global = best_w.copy()
            print(f"[global improve] candidate {idx} score={best_score_global:.6f}")

    if best_w_global is None:
        # fallback: uniform
        best_w_global = np.ones(n) / n
        best_score_global = float('inf')

    # final scaling
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w_global * target_sum / best_w_global.sum()
    # validate score
    try:
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"[final] length={n}, score={final_score:.6f}")
    except Exception:
        pass
    return [float(x) for x in final_seq]

"""
Propose a low‐score step‑function sequence.

The algorithm is a multi‑start, mirror‑descent optimisation on the
soft‑max approximation of the self‑convolution supremum.
It uses several well‑chosen initialisations (Catalan, power‑law,
Gaussian and uniform), runs an Adam‑style multiplicative update,
periodically symmetrises, random‑shifts/blocks and a cheap mixing with
the current best candidate.  The whole search is bounded by the
``budget_s`` argument (default 960 s).  The final sequence is scaled to
the conventional ℓ¹‑norm √(2 n) so that ``evaluate_sequence`` returns the
max‑convolution value directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers C_i / 4**i (monotone decreasing)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution via a real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 % of length
    hi = max(2, n // 12)           # ~8 % of length
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetric smoothing
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """Gradient of smooth max:  g = 2·corr(w, α)."""
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = 3·n‑2
    start = w.shape[0] - 1                     # centre index for j=0
    return 2.0 * conv[start:start + w.shape[0]]


def random_shift_mix(w: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Mix the vector with a randomly shifted copy (linear blend)."""
    n = w.size
    shift = int(rng.integers(1, n))            # non‑zero shift
    shifted = np.concatenate([np.zeros(shift), w[: n - shift]])
    mixed = (w + shifted) * 0.5
    mixed_sum = mixed.sum()
    if mixed_sum == 0.0:
        mixed = np.full_like(w, 1.0 / n)
    else:
        mixed /= mixed_sum
    return mixed


# ----------------------------------------------------------------------
# Core optimisation routine (single start)
# ----------------------------------------------------------------------


def optimise_one_start(
    w0: np.ndarray,
    n: int,
    rng: np.random.Generator,
    time_budget: float,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the smooth max of w*w.
    Returns the best vector (sum‑normalised to 1) and its raw score
    ``2·n·max(conv)``.
    """
    w = w0.copy()
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = float(2 * n * np.max(b))

    # Adam hyper‑parameters
    eta0 = 0.9            # base learning‑rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    beta0 = 0.4
    beta_max = 150.0

    # periodic actions
    symmetrize_every = 400
    block_every = 2500
    shift_mix_every = 1800
    reset_patience = 35000

    m = np.zeros_like(w)       # first moment
    v = np.zeros_like(w)       # second moment
    it = 0
    no_improve = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        # ---- 1. self‑convolution and current score -------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] it={it:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ---- 2. soft‑max weights α ------------------------------------
        beta = beta0 * np.sqrt(it + 1)
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b
        exp_term = np.exp(beta * diff)          # max entry -> 1
        alpha = exp_term / np.sum(exp_term)     # probability vector

        # ---- 3. gradient ---------------------------------------------
        grad = correlation_grad(w, alpha)

        # ---- 4. Adam multiplicative update ---------------------------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        eta = eta0 / np.sqrt(it1)

        update_factor = np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        w *= update_factor

        # re‑normalise to a probability vector
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w.fill(1.0 / n)
        else:
            w /= w_sum

        # ---- 5. occasional symmetrisation -----------------------------
        if it % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= w.sum()

        # ---- 6. cheap random block perturbation -----------------------
        if it % block_every == 0 and it > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = float(2 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] it={it:7d}, new best={best_score:.6f}")

        # ---- 7. random shift mixing -----------------------------------
        if it % shift_mix_every == 0 and it > 0:
            cand = random_shift_mix(best_w, rng)
            cand = cand / cand.sum()
            cand_score = float(2 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[shift] it={it:7d}, new best={best_score:.6f}")

        # ---- 8. reset if stuck too long -------------------------------
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] it={it:7d}, reset to perturbed best")

        it += 1

    return best_w, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):
    """
    Search for a step‑function with minimal ``evaluate_sequence`` value.
    Returns a list of floats (length = n) whose sum is √(2 n).
    """
    rng = np.random.default_rng(seed)

    # ---- length choice -------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 8000
    else:
        n = 8000

    print(f"[setup] target length n={n}")

    # ---- build a collection of start vectors ---------------------------
    start_vectors = []

    # Catalan (default)
    start_vectors.append(catalan_sequence(n))

    # Power‑law families (α in (0.5, 0.8))
    for alpha in (0.55, 0.60, 0.65, 0.70, 0.75):
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
        start_vectors.append(seq)

    # Discrete Gaussian centred in the middle
    x = np.arange(n, dtype=np.float64)
    sigma = n / 5.0
    gauss = np.exp(-((x - (n - 1) / 2) ** 2) / (2 * sigma ** 2))
    start_vectors.append(gauss)

    # Uniform
    start_vectors.append(np.ones(n, dtype=np.float64))

    # ---- optimisation across the different starts ----------------------
    best_overall_w = None
    best_overall_score = float('inf')
    start_time = time.monotonic()
    remaining_budget = lambda: max(0.0, budget_s - (time.monotonic() - start_time) - 5.0)

    for idx, vec in enumerate(start_vectors):
        # normalise to a probability vector (∑=1)
        w = vec.astype(np.float64)
        s = w.sum()
        if s == 0.0 or not np.isfinite(s):
            continue
        w /= s

        # allocate a slice of the remaining time (fair share)
        time_left = remaining_budget()
        if time_left < 5.0:
            break
        # give each restart at most 1/4 of the remaining budget, but not less than 5 s
        alloc = min(time_left / 4.0, max(5.0, time_left * 0.25))

        print(f"[run] start #{idx} – allocated {alloc:.1f}s")
        try:
            cand_w, cand_score = optimise_one_start(w, n, rng, alloc)
        except Exception as exc:
            print(f"[error] start #{idx} raised {exc}")
            continue

        if cand_score < best_overall_score:
            best_overall_score = cand_score
            best_overall_w = cand_w.copy()
            print(f"[global] new best after start #{idx}: score={best_overall_score:.6f}")

        # If the global best is already very low we can stop early
        if best_overall_score < 1.45:   # generous early‑exit threshold
            break

    # ------------------------------------------------------------------
    # If for some reason nothing succeeded fall back to a simple uniform
    # distribution (the evaluation function will still be finite).
    # ------------------------------------------------------------------
    if best_overall_w is None:
        best_overall_w = np.ones(n, dtype=np.float64) / n
        best_overall_score = float('inf')
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant; we adopt the
    # conventional √(2 n) ℓ¹‑norm.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_overall_w * target_sum / best_overall_w.sum()

    # sanity check (optional)
    try:
        final_eval = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, final evaluate={final_eval:.6f}")
    except Exception:
        pass

    return [float(x) for x in final_seq]
