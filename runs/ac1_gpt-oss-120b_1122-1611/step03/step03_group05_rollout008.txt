"""
propose_candidate
-----------------

A fast Adam‑based optimizer for the self‑convolution peak problem.
The algorithm improves on the previous version by

* using a larger problem size (n≈15000) – this gives a smaller
  asymptotic constant in 2·n·max(conv),
* sharpening the smooth‐max surrogate (β up to 300) and lengthening the
  homotopy,
* decaying the learning rate in three stages,
* switching to an exact sub‑gradient once β has saturated,
* occasional symmetrisation and cheap block‑perturb jumps,
* a short “polishing’’ phase with a pure sub‑gradient and a
  √t‑decay step size.

All updates keep the vector on the probability simplex
{w ≥ 0, ∑w = 1}.  The best vector found is finally rescaled to the
convention √(2 n) ℓ¹‑norm, which makes the value returned by
``evaluate_sequence`` exactly ``2·n·max(conv)``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers divided by 4**i:
    a_i = C(2i,i)/4**i.  Recurrence a_{i+1}=a_i*(2i+1)/(2i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    # Duchi et al. (2008) algorithm.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random perturbation – used as a jump to escape local minima."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)      # 0.5 % of the length
    hi = max(2, n // 12)       # up to ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈ ±12 % multiplicative noise
    cand[start:stop] *= factor
    if rng.random() < 0.35:    # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution w*w using an FFT of length N.
    Returns the full (2·len(w)‑1) result."""
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # eliminate tiny negatives
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth max‑approximation L_β at point w.
    ``probs`` are the soft‑max weights (sum = 1, length = 2·len(w)‑1).
    Returns a vector of length len(w).
    """
    # correlation w ⋆ probs (shifted) → 2·∑_i probs[i]·w[i‑j]
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = probs.size
    offset = m - 1
    grad = 2.0 * corr_full[offset: offset + w.size]
    return grad


def shift_gradient(w: np.ndarray, k: int, out: np.ndarray) -> None:
    """
    Add the contribution 2·w_{k‑i} to ``out[i]`` for all i where the
    indices are valid.  Used to build the exact sub‑gradient for a
    given convolution index k.
    """
    n = w.size
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1)
    j = k - i
    out[i] += 2.0 * w[j]
