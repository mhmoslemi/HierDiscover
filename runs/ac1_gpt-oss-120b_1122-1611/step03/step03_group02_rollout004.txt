"""
Smooth‑max Adam optimizer for the ℓ∞‑self‑convolution problem.

The algorithm works on the probability simplex (∑w = 1, w ≥ 0) and
minimises the peak of the self‑convolution w∗w.
It uses a smooth approximation of the max operator
    f(w)=μ·log∑_k exp((w∗w)[k]/μ)
with an Adam optimiser, occasional random block jumps and
a simple sub‑gradient clean‑up at the end.  The final sequence is
scaled to the conventional √(2 n) ℓ₁‑norm, which does not change the
score returned by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required by the harness)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for n up to a few ×10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()     # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Small negative values arise from numerical noise – clip them.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should rarely happen)
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random block perturbation used as a jump operator.
    A contiguous block is multiplied by a log‑normal factor;
    optionally the sequence is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) whose score,
    as measured by `evaluate_sequence`, is (hopefully) better than the
    previous best (~1.51786).  The routine runs for at most `budget_s`
    seconds, uses an Adam optimiser on a smooth‑max surrogate of the
    L∞ convolution peak, and finishes with a short sub‑gradient cleanup.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose problem size.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=np.float64)
            n = init_seq.size
            if n < 2:
                raise ValueError
        except Exception:
            init_seq = None
            n = 10000
    else:
        init_seq = None
        n = 10000                     # a moderately large size for a good asymptote

    # --------------------------------------------------------------
    # Initialise w on the simplex (∑w = 1, w ≥ 0).
    # --------------------------------------------------------------
    if init_seq is not None:
        w = np.maximum(init_seq, 0.0)
        s = np.sum(w)
        if s < 1e-12:
            w = np.full(n, 1.0 / n)
        else:
            w = w / s
    else:
        # classic central‑binomial (a_i = C(2i,i)/4**i) – known to be a good start
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2 * i)
        w = a / np.sum(a)

    # --------------------------------------------------------------
    # State variables.
    # --------------------------------------------------------------
    best_w = w.copy()
    b = convolve_fft(w, w)
    max_b = np.max(b)
    best_score = float(2.0 * n * max_b)      # equivalent to evaluate_sequence after scaling
    best_iter = 0
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam hyper‑parameters.
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05                # base learning rate
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # smooth‑max parameters
    mu_factor = 0.5
    mu_factor_min = 0.04
    mu_decay = 0.9997

    # control parameters
    start_time = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 30000        # after this many stagnant steps we reset
    block_interval = 3000         # try a random jump every few thousand steps
    reset_interval = 20000        # fall back to a perturbed best if stuck

    # --------------------------------------------------------------
    # Main optimisation loop (smooth‑max Adam).
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # 1) self‑convolution and current score
        b = convolve_fft(w, w)
        maxb = np.max(b)
        score = 2.0 * n * maxb

        # 2) record improvement
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            best_iter = iteration
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # 3) smooth‑max weights (soft‑max over convolution values)
        mu = maxb * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        shifted = (b - maxb) / mu
        exp_term = np.exp(shifted)               # max(b) maps to 1, others ≤ 1
        p = exp_term / np.sum(exp_term)          # softmax vector (length 2n‑1)

        # 4) gradient of the smooth‑max surrogate:
        #    ∇ = 2·conv(p, w_rev)  (take the central n entries)
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # 5) Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        t = iteration + 1.0
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)                     # simple decay on top of Adam’s adaptivity
        w_candidate = w - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 6) project back onto the simplex
        w = simplex_projection(w_candidate)

        # 7) slowly reduce the smoothing factor
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        # 8) occasional cheap random jump (helps escape shallow local minima)
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)                 # re‑normalise to the simplex
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m = np.zeros_like(w)
                v = np.zeros_like(w)
                no_improve = 0
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # 9) reset from a perturbed best if we seem stuck
        if no_improve > reset_interval:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m = np.zeros_like(w)
            v = np.zeros_like(w)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Sub‑gradient cleanup phase (focus on the active peak(s)).
    # --------------------------------------------------------------
    sub_iter = 0
    eta0_sub = 0.6
    tolerance = 1e-12
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 0.5):
        b = convolve_fft(w, w)
        maxb = np.max(b)
        active = np.flatnonzero(b >= maxb - tolerance)
        # sub‑gradient: sum of 2·a_{k−i} over all active k
        grad = np.zeros_like(w)
        for k in active:
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            # indices i such that 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
            i = np.arange(j_start, j_end + 1, dtype=int)
            grad[i] += 2.0 * w[k - i]        # 2·w shifted
        eta = eta0_sub / np.sqrt(sub_iter + 1.0)
        w = simplex_projection(w - eta * grad)

        # assess improvement
        score = 2.0 * n * np.max(convolve_fft(w, w))
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[sub‑improve] sub_iter={sub_iter:6d}, score={best_score:.6f}")

        sub_iter += 1
        if sub_iter > 30000:
            break

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but we output the
    # conventional √(2 n) ℓ₁‑norm for readability.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum          # now ∑final_seq = √(2 n)

    # sanity‑check with the provided scorer (optional, but useful for logging)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best iter={best_iter}, best score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
