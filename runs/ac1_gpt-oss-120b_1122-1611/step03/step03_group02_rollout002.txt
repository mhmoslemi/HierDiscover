"""
Arc‑sine‑initialized projected Adam for minimizing the peak of a self‑convolution.

The algorithm works on the probability simplex (∑w = 1, w ≥ 0).  
* Initialise a symmetric “arcsine” shape   w_i ∝ 1/√((i+½)(n‑i‑½)).
* Optimise the smooth approximation   f_μ(w)=μ·log∑_k exp((w∗w)[k]/μ)  
  using an Adam‑style projected gradient descent (projection onto the simplex).
* Every few thousand iterations perform a cheap random block perturbation
  (escape heuristic) and keep the best sequence seen so far.
* A short sub‑gradient cleanup phase focuses on the currently active peak.
* Finally the best vector is rescaled to the conventional ℓ₁‑norm √(2 n)
  (the scaling does not affect the score returned by `evaluate_sequence`).

The whole search runs for at most ``budget_s`` seconds and prints progress.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required by the harness)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # tiny negative round‑off can appear – clip to zero
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of  v  onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation, used as a jump operator."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetric averaging – keeps the sequence symmetric
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact left‑right symmetry (preserves the ℓ₁‑norm)."""
    return 0.5 * (v + v[::-1])

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is hopefully better than the previous best.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose the length of the sequence.
    # Use a moderately large size (≈12 000).  If a previous candidate
    # `height_sequence_1` is available we reuse its length, otherwise we
    # keep the new length.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            n = init.size
            if n < 2:
                raise ValueError
        except Exception:
            n = 12000
            init = None
    else:
        n = 12000
        init = None

    # --------------------------------------------------------------
    # Initialise w on the simplex.
    #   * If a previous sequence exists – take it, clip negatives,
    #     renormalise and symmetrise.
    #   * Otherwise start from a symmetric arcsine‑like distribution:
    #         w_i ∝ 1 / sqrt((i+½)(n‑i‑½)).
    # --------------------------------------------------------------
    if init is not None:
        w = np.maximum(init, 0.0)
        if w.sum() < 1e-12:
            w = np.full(n, 1.0 / n)
        else:
            w = w / w.sum()
        w = symmetrize(w)
        w = simplex_projection(w)          # guarantee exact simplex condition
    else:
        i = np.arange(n, dtype=np.float64)
        # arcsine‑type shape, symmetric by construction
        w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        w = w / w.sum()
        w = symmetrize(w)

    # --------------------------------------------------------------
    # State variables for the optimisation loop.
    # --------------------------------------------------------------
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = float(2.0 * n * b.max())   # objective value (same as evaluate_sequence)
    best_iter = 0
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam hyper‑parameters (projected onto simplex after each step).
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5               # learning‑rate before Adam correction
    m = np.zeros_like(w)        # first moment
    v = np.zeros_like(w)        # second moment

    # smoothing for the smooth‑max (μ = mu_factor * maxb)
    mu_factor = 0.7
    mu_factor_min = 0.04
    mu_decay = 0.9995

    # Misc. counters
    start_time = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 25000
    block_interval = 2500
    reset_interval = 20000

    # --------------------------------------------------------------
    # Main optimisation loop (smooth‑max Adam + occasional jumps).
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # 1) current convolution and score
        b = convolve_fft(w, w)
        maxb = b.max()
        score = 2.0 * n * maxb

        # 2) keep the best vector ever seen
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            best_iter = iteration
            no_improve = 0
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # 3) smooth‑max weights (softmax over convolution values)
        mu = maxb * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        shifted = (b - maxb) / mu          # shift so that max → 0, others ≤ 0
        exp_term = np.exp(shifted)        # safe: values in [0,1]
        p = exp_term / exp_term.sum()     # softmax – vector of length 2n‑1

        # 4) gradient of the smooth‑max (2·conv(p, w_rev) sliced to length n)
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # 5) Adam update (projected onto the simplex)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        w_candidate = w - base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # 6) projection onto the simplex and enforce symmetry
        w = simplex_projection(w_candidate)
        w = symmetrize(w)

        # 7) slowly reduce the smoothing factor μ
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        # 8) random block jump every few thousand steps
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_b = convolve_fft(cand, cand)
            cand_score = 2.0 * n * cand_b.max()
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # 9) restart from a perturbed best if we have been stagnant for a while
        if no_improve > reset_interval:
            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Sub‑gradient refinement – concentrate on the currently active peak.
    # --------------------------------------------------------------
    sub_iter = 0
    eta0_sub = 0.6
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 0.5):
        b = convolve_fft(w, w)
        maxb = b.max()
        # indices where convolution is essentially maximal (tolerance 1e-15)
        max_idxs = np.flatnonzero(b >= maxb - 1e-15)
        i_max = int(rng.choice(max_idxs))

        # sub‑gradient for the active lag i_max
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        grad[j] = 2.0 * w[i_max - j]      # 2·w shifted

        eta = eta0_sub / np.sqrt(sub_iter + 1.0)
        w = simplex_projection(w - eta * grad)
        w = symmetrize(w)

        # record possible improvement
        score = 2.0 * n * convolve_fft(w, w).max()
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[sub‑improve] sub_iter={sub_iter:6d}, score={best_score:.6f}")

        sub_iter += 1
        if sub_iter >= 20000:
            break

    # --------------------------------------------------------------
    # Final scaling – evaluation is scale‑invariant, but we output the
    # conventional √(2 n) ℓ₁‑norm for readability.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best iter={best_iter}, best score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
