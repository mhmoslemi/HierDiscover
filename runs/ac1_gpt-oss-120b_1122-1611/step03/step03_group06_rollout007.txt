"""
A fast multi‑restart search for a non‑negative step function that minimizes

    evaluate_sequence(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm works entirely on the probability simplex (∑w = 1, w ≥ 0) and
optimises the scale‑invariant objective 2·n·max(conv(w,w)).  It combines
three ideas:

1.  **Adam‑based soft‑max gradient descent** – a smooth surrogate of the
    pointwise maximum (log‑sum‑exp) gives a well‑behaved gradient.
    The temperature (β) is increased gradually, and after a threshold the
    gradient is taken w.r.t. the true maximum (hard‐max sub‑gradient).

2.  **Symmetrisation** – after each projection the vector is mirrored and
    re‑projected; the optimum is known to be symmetric, this reduces the
    search space without harming the optimum.

3.  **Local simulated‑annealing refinement** – starting from the best
    vector found, random block perturbations are accepted with a
    temperature schedule, allowing the search to escape shallow minima.

Different lengths (≈ 2 k–10 k) are tried in parallel, each with a cheap random
initialisation (mix of Catalan, Dirichlet and sparse support).  The best
sequence seen before the time budget expires is finally scaled to
∑a = √(2 n) and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan‑type coefficients a_i = C(2i,i)/4**i for i = 0…length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT (fast for large inputs).
    Returns a * b of length len(a)+len(b)-1.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negatives caused by round‑off
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation:
    a contiguous block is multiplied by a log‑normal factor,
    optionally symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def random_sparse_w(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a simplex‑vector with a random sparse support.
    The support size is a random fraction of n (between 2 % and 40 %).
    """
    frac = rng.uniform(0.02, 0.4)
    s = max(2, int(frac * n))
    idx = rng.choice(n, size=s, replace=False)
    w = np.zeros(n, dtype=np.float64)
    w[idx] = 1.0 / s
    return w


def init_random_w(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random initial vector on the simplex.
    Mixture of Catalan, a Dirichlet draw and optionally a sparse component.
    """
    cat = catalan_sequence(n)
    cat = cat / np.sum(cat)                     # sum = 1
    dirichlet = rng.random(n)
    dirichlet = dirichlet / np.sum(dirichlet)   # sum = 1
    alpha = rng.random()
    w = alpha * cat + (1.0 - alpha) * dirichlet
    # occasional sparse mixture
    if rng.random() < 0.25:
        w_sparse = random_sparse_w(n, rng)
        w = 0.5 * w + 0.5 * w_sparse
    return simplex_projection(w)


def softmax_adam_descent(w0: np.ndarray,
                         n: int,
                         max_iters: int,
                         rng: np.random.Generator,
                         time_limit: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of max(conv(w,w)).
    Returns the best vector seen (still on the simplex) and its
    scale‑invariant score 2·n·max(conv(w,w)).
    The run stops early if time_limit is exceeded.
    """
    w = w0.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.15
    beta = 0.5            # start of smoothing
    beta_max = 300.0
    beta_factor = 1.015
    hard_beta = 250.0     # switch to exact max‑gradient beyond this
    best_w = w.copy()
    # initial score
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)

    start = time.monotonic()
    for it in range(1, max_iters + 1):
        # time guard
        if time.monotonic() - start > time_limit:
            break

        # 1) convolution and max
        b = convolve_fft(w, w)
        max_b = np.max(b)

        # 2) soft‑max or hard‑max weights
        if beta < hard_beta:
            exp_arg = beta * (b - max_b)            # ≤ 0
            p = np.exp(exp_arg)
            s = np.sum(p)
            if s == 0.0:                           # under‑flow, fallback
                p = np.full_like(b, 1.0 / b.size)
                s = 1.0
            p /= s
        else:
            # exact sub‑gradient: uniform weight over all arg‑max positions
            idx = np.flatnonzero(b == max_b)
            p = np.zeros_like(b)
            p[idx] = 1.0 / idx.size

        # 3) gradient of the surrogate
        grad = 2.0 * convolve_fft(p, w)[:n]

        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_candidate = w - step
        # 5) projection onto simplex
        w = simplex_projection(w_candidate)

        # optional symmetrisation (helps for the known symmetric optimum)
        if rng.random() < 0.35:
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

        # 6) record improvement
        cur_score = 2.0 * n * np.max(convolve_fft(w, w))
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()

        # 7) increase smoothing temperature
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

    return best_w, best_score


def simulated_annealing(w_start: np.ndarray,
                        n: int,
                        rng: np.random.Generator,
                        time_limit: float,
                        init_temp: float = 0.02,
                        final_temp: float = 1e-5,
                        steps: int = 5000) -> tuple[np.ndarray, float]:
    """
    Very cheap SA that repeatedly applies ``block_perturbation``.
    Returns the best vector found (still on the simplex) and its score.
    """
    w = w_start.copy()
    cur_score = 2.0 * n * np.max(convolve_fft(w, w))
    best_w, best_score = w.copy(), cur_score
    start = time.monotonic()

    # linear temperature schedule
    for i in range(steps):
        if time.monotonic() - start > time_limit:
            break
        T = init_temp + (final_temp - init_temp) * (i / steps)
        cand = block_perturbation(w, rng)
        cand = cand / np.sum(cand)
        cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
        delta = cand_score - cur_score
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            w, cur_score = cand, cand_score
            if cand_score < best_score:
                best_w, best_score = cand.copy(), cand_score
    return best_w, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights whose ``evaluate_sequence``
    value is as small as possible.  Uses several random restarts,
    Adam‑based gradient descent on a soft‑max surrogate, optional
    hard‑max sub‑gradient, symmetrisation, and a final simulated‑annealing
    polishing phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Keep a global best (across all lengths) for the whole run.
    # ------------------------------------------------------------------
    best_sequence = None
    best_score = float('inf')
    best_n = None

    # possibly reuse an already known construction
    if 'height_sequence_1' in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            n0 = seq0.size
            w0 = seq0 / np.sum(seq0)               # normalise to sum = 1
            score0 = 2.0 * n0 * np.max(convolve_fft(w0, w0))
            best_sequence = seq0.copy()
            best_score = score0
            best_n = n0
            print(f"[init] previous best length={n0}, score={best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameters governing the search.
    # ------------------------------------------------------------------
    # a few useful lengths – we let the algorithm wander a bit
    length_pool = [4000, 5600, 6733, 8000, 10000]
    # per‑run budgets (seconds)
    now = time.monotonic()
    deadline = now + float(budget_s) - 0.5          # tiny safety margin
    # allocate ~70 % of the time to gradient runs, rest for SA polishing
    gradient_time_ratio = 0.70

    # ------------------------------------------------------------------
    # Main random‑restart loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # decide remaining time and split for this iteration
        remaining = deadline - time.monotonic()
        # Reserve a bit for a final polishing phase
        if remaining < 2.0:
            break
        # Choose a length (biased towards the current best length if any)
        if best_n is not None and rng.random() < 0.5:
            n = best_n
        else:
            n = int(rng.choice(length_pool))

        # ----------------------------------------------------------------
        # Random initial vector on the simplex.
        # ----------------------------------------------------------------
        w0 = init_random_w(n, rng)

        # ----------------------------------------------------------------
        # Gradient descent phase (soft‑max + Adam).
        # ----------------------------------------------------------------
        grad_time = remaining * gradient_time_ratio * 0.5   # a fraction for this run
        max_iters = 8000                                     # generous upper bound
        w_opt, score_opt = softmax_adam_descent(
            w0, n, max_iters, rng, time_limit=grad_time
        )

        if score_opt < best_score:
            best_score = score_opt
            best_n = n
            best_sequence = w_opt * np.sqrt(2.0 * n)
            print(f"[gradient] new best length={n}, score={best_score:.6f}")

        # ----------------------------------------------------------------
        # Local simulated‑annealing refinement around the current best.
        # ----------------------------------------------------------------
        # Use only a few seconds for the SA; adapt to the leftover budget.
        leftover = deadline - time.monotonic()
        if leftover <= 0.0:
            break
        sa_time = min(leftover * (1.0 - gradient_time_ratio), 1.5)
        # number of SA steps is chosen so that we stay within sa_time
        # (empirically a step takes ~0.3‑0.5 ms for n≈6000)
        approx_step_sec = 0.0004
        steps = max(1, int(sa_time / approx_step_sec))
        w_sa, score_sa = simulated_annealing(
            w_opt, n, rng, time_limit=sa_time, steps=steps
        )
        if score_sa < best_score:
            best_score = score_sa
            best_n = n
            best_sequence = w_sa * np.sqrt(2.0 * n)
            print(f"[SA]       improved length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety check – evaluate through the provided harness.
    # ------------------------------------------------------------------
    if best_sequence is None:
        # fallback: a very simple uniform vector
        n_fallback = 2000
        w_fallback = np.full(n_fallback, 1.0 / n_fallback)
        best_sequence = w_fallback * np.sqrt(2.0 * n_fallback)
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    else:
        # sanity evaluation (the harness clamps values anyway)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"[final] best length={best_sequence.size}, evaluate={best_score:.6f}")
    # Convert to plain Python list as required by the harness.
    return [float(x) for x in best_sequence]
