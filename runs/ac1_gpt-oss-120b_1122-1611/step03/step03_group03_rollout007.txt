"""
propose_candidate
-----------------
A two‑phase optimizer for the convolution‑max problem.

Phase 1 – a smooth‑max / Adam run (global exploration).  
Phase 2 – a sub‑gradient descent on the true max‐convolution
           (sharp local refinement).  The iterate is always kept on
           the probability simplex and symmetrised (w = w[::-1]) to
           exploit the inherent symmetry of the objective.
Random Dirichlet restarts and block perturbations are used when the
search stagnates.  The best sequence seen is finally scaled to the
conventional √(2 n) L¹‑norm before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities – all defined at module level (no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan numbers scaled by 4⁻ⁱ: a₀=1, a_{i+1}=a_i*(2i+1)/(2i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}. (Duchi et al. 2008)"""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and ~8 % of n
    lo = max(2, n // 200)
    hi = max(3, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.10)))   # mild factor
    cand[start:stop] *= factor
    # occasional symmetrisation
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def compute_score(w: np.ndarray) -> float:
    """Return 2·n·max(conv(w,w)).  This equals the official evaluation
    for any normalisation because the objective is scaling‑invariant."""
    n = w.size
    conv = convolve_fft(w, w)
    return float(2.0 * n * np.max(conv))

    Returns a non‑negative step‑function (list of floats) that aims
    to minimise ``evaluate_sequence``.  The algorithm combines a
    smooth‑max Adam phase with a sharp sub‑gradient phase and several
    random restarts.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose the vector length.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():

    else:

            if init_arr.size == n:
                w = np.maximum(init_arr, 0.0)
            else:
                raise ValueError
        except Exception:
            w = catalan_sequence(n)
    else:
        w = catalan_sequence(n)

    # Normalise to the probability simplex and enforce symmetry.
    w = w / np.sum(w)
    w = 0.5 * (w + w[::-1])
    w = simplex_projection(w)

    # --------------------------------------------------------------
    # Book‑keeping of the best sequence seen so far.
    # --------------------------------------------------------------
    best_w = w.copy()
    best_score = compute_score(best_w)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    time_budget = float(budget_s)

    # --------------------------------------------------------------
    # Phase 1 – Adam with soft‑max approximation of the max.
    # --------------------------------------------------------------
    m = np.zeros_like(w)          # first moment
    v = np.zeros_like(w)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12                    # modest learning rate
    temperature = 5.0             # start temperature (soft max)
