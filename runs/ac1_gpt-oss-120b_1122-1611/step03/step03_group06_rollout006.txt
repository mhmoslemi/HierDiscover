"""
propose_candidate
-----------------
A multi‑restart projected‑gradient search on a smooth “soft‑max”
approximation of the max self‑convolution, followed by a cheap
hill‑climbing refinement.

The objective (scale‑invariant) is

    minimise   max_k (w·w)[k]   subject to   w ≥ 0 ,  Σ w = 1 .

The hard non‑smooth max is replaced by a log‑sum‑exp surrogate

    S_β(w) = (1/β)·log∑_k exp(β·(w·w)[k]),   β≫0 ,

whose gradient is   ∇S_β = 2·(p * w)[:n]   with
    p_k = exp(β·((w·w)[k]‑max))/∑_ℓ exp(β·((w·w)[ℓ]‑max)) .

A simple heavy‑ball momentum step is taken and the candidate is
projected back onto the probability simplex.  Several very different
initialisations (uniform, Catalan, sine‑square, semi‑circle,
arcsine‑like, random Dirichlet) are tried in turn; the best vector is
kept.  The smoothing parameter β is increased far beyond the previous
limit (up to 5000) so that the surrogate approaches the true max.
Occasional “block” perturbations help the optimiser escape shallow
local minima.

When the time budget is almost exhausted a cheap pairwise
hill‑climbing phase moves a small fraction of mass between two random
indices and keeps the move only if it reduces the peak of the
self‑convolution.  The final vector is scaled to the required
ℓ¹‑norm √(2 n) and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – defined at top level.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan‑type coefficients a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using real FFT.
    Equivalent to np.convolve(a, b) but O(n log n) for large n.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off errors
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries equal – return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation used as a “jump’’.
    A random contiguous block is multiplied by a log‑normal factor and the
    whole vector is symmetrised (averaging with its reverse) to keep the
    distribution roughly symmetric.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # minimal block width
    hi = max(2, n // 12)           # maximal block width
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # enforce symmetry (helps the self‑convolution be more uniform)
    cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
