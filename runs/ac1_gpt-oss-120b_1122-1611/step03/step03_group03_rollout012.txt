    # -----------------------

    m = np.zeros_like(w); v = np.zeros_like(w)
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    lr0 = 0.08  # smaller learning

    iter = 0

    best_w = w.copy()

        maxc = np.max(conv)

The algorithm consists of three stages:

   (geometric, power‑law, Beta‑shaped, sine, cosine).  The best of these
   initialisations is kept as a starting point.

   The temperature of the soft‑max is increased slowly so that the iterates
   approach the true non‑smooth objective.

3. A short “LP‑mixing’’ phase motivated by the state‑of‑the‑art approach:
   for the current iterates we solve a small linear programme that
   produces a direction g whose convolution with the current f does not
   increase the present peak value M.  A convex combination of f and g
   is accepted if it reduces the real objective.  A few random block
   perturbations are interleaved to avoid stagnation.

All computation respects the 1000 s time budget; the final sequence is

def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers scaled by 4⁻ⁱ:
    a_i = C(2i,i) / 4ⁱ   (recurrence a_{i+1}=a_i*(2i+1)/(2i+2))."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (non‑circular) using real FFT.
    Small negative round‑off errors are clipped to 0."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # guard against -0 caused by rounding
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump – multiply a random contiguous block by a
    log‑normal factor, optionally symmetrise."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def geometric_sequence(n: int, r: float) -> np.ndarray:
    """Finite geometric series normalised to unit L¹."""
    if r <= 0.0:
        return np.full(n, 1.0 / n)
    # Use analytical sum of finite geometric series
    idx = np.arange(n, dtype=np.float64)
    if r == 1.0:
        w = np.full(n, 1.0 / n)
    else:
        # avoid catastrophic cancellation for r≈1 using series expansion if needed
        # but numpy handles it well for double precision.
        geom = r ** idx
        # Normalise to sum 1
        total = (1.0 - r ** n) / (1.0 - r)
        w = geom / total
    # Guard against tiny negative components due to rounding
    w = np.maximum(w, 0.0)
    return w


def power_sequence(n: int, p: float) -> np.ndarray:
    """Sequence proportional to (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n)
    w = w / w_sum
    return w


def beta_sequence(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta‑like shape: (i+1)^{a} * (n-i)^{b}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = (i) ** a
    right = (n - i + 1) ** b
    w = left * right
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n)
    w = w / w_sum
    return w


def sine_sequence(n: int) -> np.ndarray:
    """Sine‑shaped distribution on [0,n)."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n)
    w_sum = w.sum()
    w = w / w_sum
    return w


def cosine_sequence(n: int) -> np.ndarray:
    """Half‑cosine shape, zero at the ends, max at centre."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(np.pi * i / (n - 1))
    w_sum = w.sum()
    w = w / w_sum
    return w

        # r in (0.9, 0.9999)
        r = rng.uniform(0.9, 0.9999)
        families.append(geometric_sequence(n, r))

    for _ in range(30):
        p = rng.uniform(0.1, 3.0)
        families.append(power_sequence(n, p))

    # random beta‑like trials
    for _ in range(30):

        b = rng.uniform(-2.0, 2.0)
        families.append(beta_sequence(n, a, b))

    if 'height_sequence_1' in globals():
        try:

            if external.shape[0] == n:
                families.append(np.maximum(external, 0.0))
        except Exception:
            pass

    # Evaluate all candidates, but stop early if we run out of time
    for cand in families:
        if time.monotonic() - start_time > 0.1 * budget:  # spend at most 10% on scan
            break
        maybe_update(cand)

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimisation on a smooth max surrogate
    # ------------------------------------------------------------------
    w = best_w.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.08                # smaller learning‑rate than the reference
    alpha = 15.0              # initial soft‑max temperature
    iter_cnt = 0

    phase1_deadline = start_time + 0.55 * budget

    while time.monotonic() < phase1_deadline:
        conv = convolve_fft(w, w)
        maxc = np.max(conv)
        # soft‑max over the convolution
        exp_vals = np.exp(alpha * (conv - maxc))
        p = exp_vals / np.sum(exp_vals)

        # gradient of the smooth surrogate (2 * correlation)
        w_rev = w[::-1]
        grad_full = convolve_fft(p, w_rev)          # length 3n‑2
        grad = 2.0 * grad_full[len(w) - 1:len(w) - 1 + n]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iter_cnt + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w = simplex_projection(w_new)

        # periodic evaluation of the true objective
        if iter_cnt % 500 == 0:
            cur_score = 2.0 * n * np.max(convolve_fft(w, w))
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_w = w.copy()
                print(f"[adam] iter={iter_cnt:5d} score={best_score:.6f}")

        # schedule temperature increase
        if iter_cnt % 2500 == 0 and iter_cnt > 0:
            alpha = min(alpha * 1.07, 2000.0)

        # occasional symmetrisation (helps when the optimum is asymmetric)
        if iter_cnt % 3000 == 0 and iter_cnt > 0:
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

        iter_cnt += 1

    # Stage 2 – LP based mixing (only if scipy is available)
    # ------------------------------------------------------------------
    if _has_scipy:
        # mix for the rest of the budget; each LP is tiny (≤300 constraints)
        lp_deadline = start_time + budget
        while time.monotonic() < lp_deadline:
            # compute current convolution and the peak value M
            conv = convolve_fft(w, w)
            M = np.max(conv)

            # pick a modest set of “active’’ indices – top values of conv
            rel_tol = 0.03                         # keep constraints within 3 % of M
            active = np.where(conv >= M * (1.0 - rel_tol))[0]
            # enforce a hard cap on the number of constraints for speed
            max_constraints = 200
            if active.size > max_constraints:
                # keep the largest values only
                order = np.argsort(-conv[active])
                active = active[order[:max_constraints]]

            rows = active.size
            if rows == 0:
                break

            # build the (rows × n) constraint matrix A_ub
            A = np.zeros((rows, n), dtype=np.float64)
            for r, k in enumerate(active):
                lo = max(0, k - (n - 1))
                hi = min(k, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.intp)
                A[r, idx] = w[k - idx]          # coefficient w_{k‑j}

            # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0.0, None) for _ in range(n)]
            lp_res = linprog(c, A_ub=A, b_ub=np.full(rows, M),
                             bounds=bounds, method='highs',
                             options={'presolve': True, 'disp': False})

            if not lp_res.success:
                # fall back to a cheap perturbation and continue
                cand = block_perturbation(w, rng)
                w = simplex_projection(cand)
                continue

            g = lp_res.x
            sum_g = np.sum(g)
            if sum_g <= 0.0:
                # nothing useful – perform a random jump
                w = block_perturbation(w, rng)
                w = simplex_projection(w)
                continue

            # normalise the direction to stay on the simplex
            g_norm = g / sum_g

            # convex combination – mixing weight slowly decays
            t_mix = 0.25
            w_candidate = (1.0 - t_mix) * w + t_mix * g_norm
            w_candidate = simplex_projection(w_candidate)

            cand_score = 2.0 * n * np.max(convolve_fft(w_candidate, w_candidate))
            if cand_score < best_score - 1e-9:
                best_score = cand_score
                best_w = w_candidate.copy()
                w = w_candidate
                print(f"[lp] improved score={best_score:.6f} rows={rows}")
            else:
                # occasional random shake‑up to escape shallow plateaus
                if rng.random() < 0.15:
                    w = block_perturbation(w, rng)
                    w = simplex_projection(w)

    # Final scaling to √(2 n) L¹‑norm and return
    # ------------------------------------------------------------------
    target_sum = sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum(best_w)=1, but keep for safety
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n} best_score={best_score:.6f} evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]

"""
Propose candidate for minimizing ``evaluate_sequence``.

The optimisation proceeds in three stages:

1. **Parametric scan** – a cheap search over a handful of analytic families
   (geometric, power‑law, Beta‑like, sine, cosine).  The best shape from this
   scan is taken as the start point.

2. **Smooth‑max Adam descent** – we minimise a soft‑max approximation of the
   true objective  max_k (w∗w)[k]  while staying on the probability simplex.
   The temperature of the soft‑max is increased slowly so that the iterates
   converge to the non‑smooth problem.

3. **LP‑mixing phase** – inspired by the published state‑of‑the‑art
   iteration.  For the current iterate `w` we solve a tiny linear programme
   that yields a direction `g` whose convolution with `w` does not increase the
   present peak value `M`.  A convex combination of `w` and a normalised `g`
   is accepted if it reduces the true objective.  Random block perturbations
   are interleaved to avoid stagnation.

If ``scipy`` is unavailable the LP phase is skipped and the algorithm falls
back to the Adam stage only.  The final sequence is scaled to the conventional
√(2 n) L¹‑norm before being returned.

All helper functions are defined at top level, there are no closures and the
implementation respects the 1000 s time budget.
"""

def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers scaled by 4⁻ⁱ:
    a_i = C(2i,i) / 4ⁱ   (recurrence a_{i+1}=a_i*(2i+1)/(2i+2))."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (non‑circular) using real FFT.
    Small negative round‑off errors are clipped to 0."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # guard against -0 caused by rounding
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump – multiply a random contiguous block by a
    log‑normal factor, optionally symmetrise."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def geometric_sequence(n: int, r: float) -> np.ndarray:
    """Finite geometric series normalised to unit L¹."""
    if r <= 0.0:
        return np.full(n, 1.0 / n)
    idx = np.arange(n, dtype=np.float64)
    if r == 1.0:
        w = np.full(n, 1.0 / n)
    else:
        geom = r ** idx
        total = (1.0 - r ** n) / (1.0 - r)    # sum_{i=0}^{n-1} r^i
        w = geom / total
    w = np.maximum(w, 0.0)
    return w


def power_sequence(n: int, p: float) -> np.ndarray:
    """Sequence proportional to (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return w / s


def beta_sequence(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta‑like shape: (i+1)^{a} * (n-i)^{b}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** a
    right = (n - i + 1) ** b
    w = left * right
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return w / s


def sine_sequence(n: int) -> np.ndarray:
    """Sine‑shaped distribution on [0,n)."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n)
    return w / w.sum()


def cosine_sequence(n: int) -> np.ndarray:
    """Half‑cosine shape, zero at the ends, maximal in the centre."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(np.pi * i / (n - 1))
    return w / w.sum()

    # --------------------------------------------------------------
    target_sum = sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum(best_w)=1, but keep for safety
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n} best_score={best_score:.6f} evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
