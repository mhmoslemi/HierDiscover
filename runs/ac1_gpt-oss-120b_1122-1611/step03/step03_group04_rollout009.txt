"""
Improved search for a low‑value step‑function.

The algorithm follows the same “entropic gradient descent on a smooth max”
as the reference code, but uses a few refinements that usually push the
objective a little further:

*   a larger temperature cap (β_max = 4000) – the soft‑max becomes much
    closer to the true max, giving a more accurate gradient;
*   a smaller base step‑size (η₀ = 0.8) together with the classic 1/√t decay;
*   Adam‑style moment estimates for the gradient, which stabilises the
    multiplicative updates;
*   always keep the candidate symmetric (optimal solutions are symmetric);
*   more aggressive random “block‑perturbations” and an occasional
    “swap‑perturbation” to escape plateaux;
*   the whole run is split into two phases – a long warm‑up with the
    smooth‑max, then a short “hard‑max” phase where the probability weight
    α is concentrated uniformly on the (few) indices that attain the
    current maximal convolution value.

The routine works for any existing `height_sequence_1` (its length is kept
and used as a warm start) or, if none is present, builds a default
Catalan‑based sequence of size 10000.  The returned list is scaled to the
conventional ℓ¹‑norm √(2 n) – this does not change the score but makes the
output look familiar.

The code is completely self‑contained, uses only NumPy and the provided
`evaluate_sequence`, and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helper functions
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan numbers divided by 4**i.
    Recurrence: a[i+1] = a[i] * (2*i+1)/(2*i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        g = 2 * corr(w, α)
    α has length 2·n‑1.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = 3·n‑2
    start = w.shape[0] - 1                     # centre index aligning with w[0]
    return 2.0 * conv[start:start + w.shape[0]]


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 400)          # 0.25 % of length
    hi = max(2, n // 25)           # up to ~4 % of length
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.15)))   # moderate jitter
    cand[start:start + width] *= factor
    # optional symmetric smoothing
    if rng.random() < 0.4:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def swap_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exchange a tiny amount of mass between two random positions."""
    cand = base.copy()
    n = cand.size
    i, j = rng.choice(n, size=2, replace=False)
    eps = cand[i] * rng.uniform(0.001, 0.01)
    cand[i] -= eps
    cand[j] += eps
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main candidate proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: float = 960.0):
    """
    Return a non‑negative step‑function (list of floats) with a low
    ``evaluate_sequence`` score.  The routine runs an Adam‑accelerated
    entropic mirror‑descent on a temperature‑smoothed max of the
    self‑convolution, with occasional random jumps.
    """
    rng = np.random.default_rng(seed)

    # ---- choose length and initialise -------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
            w = prev.astype(np.float64)
            w = np.maximum(w, 0.0)
        except Exception:
            n = 10000
            w = catalan_sequence(n)
    else:
        n = 10000
        w = catalan_sequence(n)

    # ensure symmetry right from the start
    w = 0.5 * (w + w[::-1])

    # normalise to a probability vector (∑w = 1)
    w = w / np.sum(w)

    # ---- optimisation bookkeeping ------------------------------------------------------
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = float(np.max(b))          # after final scaling score = max_b
    print(f"[init] n={n}, initial max_b={best_score:.6f}")

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # Adam style accumulators
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # hyper‑parameters
    eta0 = 0.8               # base multiplicative step size
    beta0 = 0.4              # temperature growth factor
    beta_max = 4000.0        # larger than the reference code
    log_interval = 2000      # printing frequency
    block_every = 2500       # how often to apply a cheap random jump
    symmetrize_every = 300   # enforce symmetry regularly
    reset_patience = 35000   # resets after many non‑improving iters
    warmup_iters = 50000     # iterations before switching to hard‑max

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ----------------------------------------------------------------------
        # 1. self‑convolution and current (hard) score
        # ----------------------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = max_b                     # after sum‑normalisation this is the true score

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, max_b={best_score:.6f}")
        else:
            no_improve += 1

        # ----------------------------------------------------------------------
        # 2. Build probability vector α (smooth‑max or hard‑max)
        # ----------------------------------------------------------------------
        if iteration < warmup_iters:
            beta = beta0 * np.sqrt(iteration + 1)
            if beta > beta_max:
                beta = beta_max
            diff = b - max_b
            # exponential weights; shift by max for stability
            exp_term = np.exp(beta * diff)     # max element → 1
            alpha = exp_term / np.sum(exp_term)
        else:
            # hard‑max: uniform weight on all indices attaining the maximum
            idxs = np.where(np.abs(b - max_b) < 1e-12)[0]
            alpha = np.zeros_like(b)
            alpha[idxs] = 1.0 / max(1, idxs.size)

        # ----------------------------------------------------------------------
        # 3. Gradient of the (smooth) objective
        # ----------------------------------------------------------------------
        grad = correlation_grad(w, alpha)

        # ----------------------------------------------------------------------
        # 4. Adam‑style multiplicative update (entropic mirror descent)
        # ----------------------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))

        # multiplicative factor in log‑domain
        log_factor = -eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        # avoid overflow/underflow in the exponent
        log_factor = np.clip(log_factor, -50.0, 50.0)
        w *= np.exp(log_factor)

        # re‑normalise to a probability vector
        w_sum = w.sum()
        if w_sum <= 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # ----------------------------------------------------------------------
        # 5. Periodic symmetrisation (the optimum is symmetric)
        # ----------------------------------------------------------------------
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= np.sum(w)

        # ----------------------------------------------------------------------
        # 6. Occasional cheap random jump (helps escape plateaux)
        # ----------------------------------------------------------------------
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = 0.5 * (cand + cand[::-1])
            cand /= np.sum(cand)
            cand_score = float(np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best max_b={best_score:.6f}")

            # occasionally also try a tiny swap‑perturbation
            if rng.random() < 0.3:
                cand2 = swap_perturbation(best_w, rng)
                cand2 = 0.5 * (cand2 + cand2[::-1])
                cand2 /= np.sum(cand2)
                cand2_score = float(np.max(convolve_fft(cand2, cand2)))
                if cand2_score < best_score:
                    best_w = cand2.copy()
                    best_score = cand2_score
                    w = cand2.copy()
                    print(f"[swap]  iter={iteration:7d}, new best max_b={best_score:.6f}")

        # ----------------------------------------------------------------------
        # 7. Reset if stuck for a long time
        # ----------------------------------------------------------------------
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w = 0.5 * (w + w[::-1])
            w /= np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        # ----------------------------------------------------------------------
        # 8. Logging
        # ----------------------------------------------------------------------
        if iteration % log_interval == 0:
            elapsed = time.monotonic() - start_time
            print(f"[log] iter={iteration:7d}, max_b={score:.6f}, elapsed={elapsed:.1f}s")

        iteration += 1

    # ----------------------------------------------------------------------
    # Final scaling – the evaluation function is scale‑invariant, but we use
    # the conventional √(2 n) ℓ¹‑norm for tidy output.
    # ----------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # sanity check with the provided evaluator (optional, for logging)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:          # the harness will call evaluate_sequence itself
        final_score = float('nan')
    print(f"[final] length={n}, best max_b (raw)={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
