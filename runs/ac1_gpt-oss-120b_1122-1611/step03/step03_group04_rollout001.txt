        max_b = np.max(b)
        score = 2.0*n*max_b

        if score < best_score - 1e-12:

        # Compute gradient
        if iteration < int(0.5*reset_patience) or iteration < 200000:
            # softmax
            beta = min(beta_max, beta0*np.sqrt(iteration+1))
            diff = b - max_b

            alpha = exp_term / np.sum(exp_term)
        else:

            tol = 1e-12
            mask = b >= max_b - tol

            cnt = np.count_nonzero(mask)

        # Adam update for log w
        iteration += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # update log w
        log_w = np.log(w+eps) - step
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum <= 0 or not np.isfinite(w_sum):

            w /= w_sum

                best_score = cand_score

                w = cand.copy()

"""
propose_candidate
-----------------

Optimises the step‑function shape by (i) mirror‑descent on the
soft‑max approximation of the convolution peak, (ii) an Adam‑style
adaptive step‑size for the log‑weights (which keeps the vector on the
probability simplex), (iii) periodic symmetrisation, (iv) occasional
random block perturbations and a final stochastic hill‑climb.  The
algorithm works on the normalised vector (∑a_i = 1); the final
sequence is rescaled to the conventional √(2 n) ℓ¹‑norm before being
returned.

The method improves on the previous solution by using a more robust
adaptive step‑size (Adam) and by switching from a smooth soft‑max
gradient to the true sub‑gradient of the max once the temperature has
grown large enough.  A cheap local search at the end helps to escape
tiny plateaus, aiming to push the evaluation score below the target
1.5025.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Low‑level helpers – all defined at module level (no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    First ``length`` Catalan numbers C_i / 4**i.
    Recurrence a_{i+1}=a_i·(2 i+1)/(2 i+2).
    The sequence is positive, monotone decreasing and often a good
    starting point for the optimisation.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (zero‑padding to a power of two).
    The result is clipped at 0 to avoid tiny negative artefacts.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip numerical noise
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max:   g = 2·corr(w, α)
    where α is a probability vector of length 2·n−1.
    Implemented via a second FFT convolution:
        corr(w,α)[j] = Σ_i w[i]·α[i+j]  (j = 0…n−1)
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = 3·n−2
    start = w.shape[0] - 1                     # aligns with j=0
    return 2.0 * conv[start:start + w.shape[0]]


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random jump: multiply a random block by a log‑normal factor.
    Occasionally mirrors the block to keep the vector roughly symmetric.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 % of length
    hi = max(2, n // 12)           # up to ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal multiplier
    cand[start:stop] *= factor
    if rng.random() < 0.35:        # optional symmetric smoothing
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) whose
    ``evaluate_sequence`` score is as low as possible.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.
    # --------------------------------------------------------------
    # Longer vectors give a smaller asymptotic constant, but increase
    # per‑iteration cost.  10000 works well in the time budget.
    n = 10000

    # --------------------------------------------------------------
    # Initialise the probability vector w (∑w = 1).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= 2:
                w = prev / np.sum(prev)
                if w.shape[0] != n:
                    # if the previous length differs, interpolate linearly
                    x_old = np.linspace(0, 1, prev.size)
                    x_new = np.linspace(0, 1, n)
                    w = np.interp(x_new, x_old, prev)
                    w = np.maximum(w, 0.0)
                w /= np.sum(w)
            else:
                raise ValueError
        except Exception:
            w = catalan_sequence(n)
            w = w / np.sum(w)
    else:
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # Enforce exact symmetry from the start – the optimal solution is symmetric.
    w = 0.5 * (w + w[::-1])
    w /= np.sum(w)

    # --------------------------------------------------------------
    # Adam state for log‑weights.
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    beta1_pow = beta1
    beta2_pow = beta2
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    lr = 0.15                     # base learning‑rate for the log‑update
    eps = 1e-12

    # --------------------------------------------------------------
    # Parameters for the smooth‑max (soft‑max) approximation.
    # --------------------------------------------------------------
    beta_start = 0.5
    beta_max = 200.0

    # --------------------------------------------------------------
    # Book‑keeping.
    # --------------------------------------------------------------
    conv = convolve_fft(w, w)
    best_w = w.copy()
    best_score = float(2 * n * np.max(conv))
    iteration = 0
    no_improve = 0
    reset_patience = 60000          # iterations without improvement before a reset
    symmetrize_every = 500
    block_every = 3000

    start_time = time.monotonic()
    # ensure a small safety buffer for final scaling/checks
    time_budget = max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < time_budget:
        # ---- current convolution and score ----
        conv = convolve_fft(w, w)
        max_b = np.max(conv)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d} score={best_score:.6f}")
        else:
            no_improve += 1

        # ---- compute the gradient (soft‑max or hard max) ----
        # Use a smooth soft‑max while beta is moderate; switch to the true
        # sub‑gradient of the max once beta becomes large enough.
        beta = beta_start * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max

        if beta < 30.0:                     # soft‑max regime
            diff = conv - max_b
            # For numerical stability shift by max (already done)
            exp_term = np.exp(beta * diff)  # max element becomes 1
            alpha = exp_term / np.sum(exp_term)   # probability vector, length 2·n‑1
        else:                               # hard‑max sub‑gradient
            # tolerance to capture ties caused by floating‑point noise
            tol = 1e-15
            mask = conv >= max_b - tol
            cnt = np.count_nonzero(mask)
            if cnt == 0:                    # safety fallback
                alpha = np.zeros_like(conv)
                alpha[np.argmax(conv)] = 1.0
            else:
                alpha = np.where(mask, 1.0 / cnt, 0.0)

        # ---- gradient of the (smoothed) max ----
        grad = correlation_grad(w, alpha)   # shape (n,)

        # ---- Adam update on log‑weights (mirror descent on the simplex) ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # update bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # multiplicative (mirror) update: log‑w ← log‑w – step
        log_w = np.log(w + eps) - step
        w = np.exp(log_w)

        # renormalise to stay on the probability simplex
        w_sum = w.sum()
        if w_sum <= 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # ---- periodic symmetrisation (optimal solution is symmetric) ----
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= np.sum(w)

        # ---- occasional cheap random block jumps (helps escape plateaus) ----
        if iteration % block_every == 0:
            cand = block_perturbation(best_w, rng)
            cand /= np.sum(cand)
            cand_score = float(2 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d} new best score={best_score:.6f}")

        # ---- reset if stuck for a long time ----
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w /= np.sum(w)
            # reset Adam accumulators – otherwise they keep stale momentum
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta1
            beta2_pow = beta2
            no_improve = 0
            print(f"[reset] iter={iteration:7d} reset to perturbed best")

    # --------------------------------------------------------------
    # Final stochastic hill‑climb (tiny local moves) – runs only while time permits.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < float(budget_s) - 1.0:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # propose moving a tiny amount from i to j
        delta = 0.0005 * best_w[i]          # 0.05 % of the source mass
        if delta <= 0.0:
            continue
        new_w = best_w.copy()
        new_w[i] -= delta
        new_w[j] += delta
        # keep the vector normalised (sum unchanged) and non‑negative
        if new_w[i] < 0.0:                 # reject if we would go negative
            continue
        cand_score = float(2 * n * np.max(convolve_fft(new_w, new_w)))
        if cand_score < best_score:
            best_w = new_w
            best_score = cand_score
            print(f"[hill] iter={iteration:7d} improved to {best_score:.6f}")
        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but we output with
    # the conventional √(2 n) ℓ¹‑norm for aesthetics.
    # --------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = best_w * (target_sum / best_w.sum())
    final_seq = np.clip(final_seq, 0.0, 1000.0)   # obey the evaluator's clipping

    # sanity check (optional, just for logging)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:
        final_score = float('nan')
    print(f"[final] length={n} best raw score={best_score:.6f} "
          f"evaluate_sequence={final_score:.6f}")

    return [float(x) for x in final_seq]
