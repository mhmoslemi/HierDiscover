"""
Propose a step‑function using a symmetric mirror‑gradient descent.

The objective after normalising the vector to sum 1 is
    J(w) = max_k (w*w)[k]               (w*w = linear self‑convolution)
and the score returned by ``evaluate_sequence`` is 2·n·J(w).
The global optimum is attained when the convolution is perfectly flat,
i.e. J(w) = 1/(2n) and the score equals 1.  The algorithm below tries
to flatten the convolution by minimising a smooth log‑sum‑exp
approximation of the maximum, followed by occasional exact sub‑gradient
updates and symmetric block perturbations.  By enforcing mirror symmetry
the search space is halved and the resulting sequences tend to be much
flatter.

Key ideas
---------
* Work on a probability vector (sum = 1) and keep it on the simplex by
  Euclidean projection.
* After each optimisation step enforce w = (w + w[::-1]) / 2 so the
  sequence is symmetric.
* Use a β‑homotopy: start with a small β (smooth loss) and increase it
  gradually up to a large value (≈200) to approach the true max.
* Gradient of the smooth surrogate is obtained by a correlation
  (FFT‑based, O(n log n)).
* Every few thousand iterations a cheap exact sub‑gradient of the
  current maximum convolution entry is applied; this sharpens the
  result.
* Random contiguous‑block perturbations (mirrored) are tried periodically
  and kept when they improve the best‑ever peak.

The routine respects the 1000 s budget, prints progress, and finally
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers divided by 4**i.
    Recurrence a_{i+1}=a_i·(2i+1)/(2i+2) – stable and fast."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex
    {x≥0,∑x=1}.  Implements the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Fallback – should rarely happen
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry: v = (v + v[::-1]) / 2."""
    return 0.5 * (v + v[::-1])


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random multiplicative perturbation of a contiguous block,
    with an optional symmetric copy."""
    cand = base.copy()
    n = cand.size
    # Width between 0.5 % and ~8 % of n
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈±12 % multiplicative noise
    cand[start:stop] *= factor

    # keep symmetry (mirrored block)
    mir_start = n - stop
    mir_stop = n - start
    cand[mir_start:mir_stop] = cand[start:stop][::-1]

    # occasional extra averaging with the reversed vector
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])

    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution w*w using an FFT of length N.
    Returns the full (2·len(w)‑1) result."""
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    # tiny negative rounding artefacts are killed
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """Gradient of the smooth max‑approximation
    L_β(w)= (1/β)·log ∑_k exp(β·(w*w)[k]).
    ``probs`` must be a probability vector of length 2·len(w)‑1.
    """
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    # correlation of w with probs (shifted)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    offset = probs.size - 1
    grad = 2.0 * corr_full[offset : offset + w.size]
    return grad


def gradient_max_sub(w: np.ndarray, k_max: int) -> np.ndarray:
    """Exact sub‑gradient of max_k (w*w)[k] at the current arg‑max."""
    n = w.size
    grad = np.zeros_like(w)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    grad[idx] = 2.0 * w[k_max - idx]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the competition harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function with a small
    self‑convolution peak.  Returns a list of floats ready for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size (power of two is convenient for FFT).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            if seq0.ndim != 1 or seq0.size < 2:
                raise ValueError
            n = int(seq0.size)
            w = np.maximum(seq0, 0.0)
        except Exception:
            n = 4096
            w = catalan_sequence(n)
    else:
        n = 4096
        w = catalan_sequence(n)

    # Normalise to a probability vector and enforce symmetry.
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w_sum
    w = symmetrize(w)

    # Pre‑computed FFT lengths (powers of two for speed)
    N_self = _next_pow2(2 * n - 1)          # for self‑convolution
    N_corr = _next_pow2(3 * n - 2)          # for gradient correlation

    # --------------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------------
    best_w = w.copy()
    best_score = np.max(self_convolve_fft(w, N_self))
    print(f"[init] n={n}, initial max‑conv={best_score:.9f}")

    # Adam optimiser hyper‑parameters
    lr = 0.15
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # β‑homotopy (smooth → sharp)
    beta_start = 2.0
    beta_end = 200.0
    beta_ramp_iters = 200_000

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ---- 1. compute current convolution and its peak ----
        b = self_convolve_fft(w, N_self)
        cur_max = np.max(b)

        # keep best ever seen
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, max‑conv={best_score:.9f}")
        else:
            no_improve += 1

        # ---- 2. choose current β (smoothness) ----
        if iteration < beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * (iteration / beta_ramp_iters)
        else:
            beta = beta_end

        # ---- 3. soft‑max weighting of convolution entries ----
        # shift ≤ 0, numerically safe
        shift = beta * (b - cur_max)
        exp_shift = np.exp(shift)            # max weight == 1
        probs = exp_shift / np.sum(exp_shift)

        # ---- 4. gradient of the smooth surrogate ----
        grad = gradient_smooth(w, probs, N_corr)

        # ---- 5. Adam update with projection onto the simplex ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step
        w = simplex_projection(w_new)

        # enforce symmetry after every update
        w = symmetrize(w)

        # ---- 6. occasional exact sub‑gradient refinement ----
        if iteration % 3000 == 0:
            k_max = int(np.argmax(b))
            sub_grad = gradient_max_sub(w, k_max)
            sub_eta = 0.05                     # modest step size
            w2 = w - sub_eta * sub_grad
            w = simplex_projection(w2)
            w = symmetrize(w)

        # ---- 7. random symmetric block perturbation ----
        if iteration % 5000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = symmetrize(cand)
            cand = cand / np.sum(cand)          # back to probability simplex
            cand_max = np.max(self_convolve_fft(cand, N_self))
            if cand_max < best_score:
                best_w = cand.copy()
                best_score = cand_max
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best max‑conv={best_score:.9f}")

        # ---- 8. reset if stagnation persists ----
        if no_improve > 30_000:
            w = block_perturbation(best_w, rng)
            w = symmetrize(w)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, resetting to perturbed best")

    # --------------------------------------------------------------
    # Final scaling – the evaluator is scale‑invariant.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # should already be 1, but guard

    # Optional sanity check (cheap)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best max‑conv={best_score:.9f}, evaluate={final_score:.9f}")
    except Exception:
        pass

    # Return plain Python list as required.
    return [float(x) for x in final_seq]
