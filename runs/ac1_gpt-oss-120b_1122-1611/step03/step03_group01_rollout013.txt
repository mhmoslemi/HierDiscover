"""
Propose a step‑function that minimizes the evaluation metric.

The algorithm works on the probability simplex (∑w = 1, w≥0) and
optimises the smooth surrogate

    Lβ(w) = (1/β)·log(∑ₖ e^{β·(w∗w)[k]})

which converges to the original objective maxₖ (w∗w)[k] as β→∞.
A projected Adam optimiser is used, and the surrogate parameter β is
gradually increased (continuation).  The method is supplemented by
occasional random block perturbations, monotone‑sorting and a short
simulated‑annealing polishing phase.  All operations are performed by
FFT‑based convolutions, so the routine scales to sequences of several
10 000 entries within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan numbers a_i = C(2i,i)/4**i, recurrence is stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution via real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # tiny negative values caused by rounding are killed
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) large‑scale move; used for escape/re‑start."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def correlation_valid(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Valid (length‑n) cross‑correlation of p (length 2n‑1) with w (length n):
       c[j] = Σ_i p[i]·w[i‑j]   for 0≤j<n  (indices out of range are ignored)
    Implemented with FFT for O(n log n) speed.
    """
    n = w.size
    m = p.size                     # 2·n‑1
    N = 1 << ((n + m - 1).bit_length())
    Fp = np.fft.rfft(p, N)
    Fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(Fp * np.conj(Fw), N)
    start = n - 1
    end = m
    return corr[start:end]

    """Convenient wrapper: 2·n·max(conv) equals the raw evaluate_sequence() score."""
    return float(2.0 * n * np.max(conv))

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = eta0 / np.sqrt(iteration)          # modest global decay
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        w_temp = w - update
        w = simplex_projection(w_temp)            # stay on the simplex

        # ---- continuation on β ------------------------------------
        if iteration % 2000 == 0:
            beta = min(beta_max, beta * beta_growth)

        # ---- occasional large random jump -------------------------
        if iteration % 1500 == 0:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()
            cand_score = score_from_conv(convolve_fft(cand, cand), n)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[block] iter={iteration:06d}, new best={best_score:.6f}")

        # ---- enforce monotone decreasing order every few iters -----
        if iteration % 3000 == 0:
            sorted_w = np.sort(w)[::-1]           # decreasing
            sorted_w /= sorted_w.sum()
            sorted_score = score_from_conv(convolve_fft(sorted_w, sorted_w), n)
            if sorted_score < best_score:
                best_w = sorted_w.copy()
                best_score = sorted_score
                w = sorted_w.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[sort] iter={iteration:06d}, improved to {best_score:.6f}")

        # ---- tiny random transfer of mass (0‑2 % of a coordinate) ---
        if rng.random() < 0.02:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = rng.uniform(0.0, min(w[i], 0.01))
            if delta > 0.0:
                cand = w.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_score = score_from_conv(convolve_fft(cand, cand), n)
                if cand_score < best_score:
                    best_w = cand.copy()
                    best_score = cand_score
                    w = cand.copy()
                    m.fill(0.0)
                    v.fill(0.0)
                    print(f"[move] iter={iteration:06d}, improved to {best_score:.6f}")

        # ---- stagnation reset --------------------------------------
        if no_improve > 150_000:
            w = block_perturbation(best_w, rng)
            w /= w.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:06d}, resetting to perturbed best.")

    # --------------------------------------------------------------
    # Final simulated‑annealing polishing (uses the remaining budget)
    # --------------------------------------------------------------
    remaining = budget_s - (time.monotonic() - start_time)
    if remaining > 6.0:
        temperature = 0.02
        cur_w = best_w.copy()
        cur_score = best_score
        anneal_start = time.monotonic()
        while time.monotonic() - anneal_start < remaining - 3.0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_transfer = min(cur_w[i], 0.02)
            delta = rng.uniform(0.0, max_transfer)
            if delta == 0.0:
                continue
            cand = cur_w.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = score_from_conv(convolve_fft(cand, cand), n)
            # Metropolis acceptance
            if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / temperature):
                cur_w = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_w = cur_w.copy()
                    best_score = cur_score
            temperature *= 0.9999
            if temperature < 1e-5:
                temperature = 0.001
        print(f"[anneal] polishing finished, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final scaling – the evaluation is invariant to a uniform scaling.
    # We output the conventional √(2·n) ℓ¹‑norm.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = (best_w * target_sum).tolist()

    # sanity check with the provided evaluator (mostly for logging)
    final_score = float(evaluate_sequence(final_seq))
    print(f"[final] length={n}, best raw score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Return a plain Python list.
    return [float(x) for x in final_seq]
