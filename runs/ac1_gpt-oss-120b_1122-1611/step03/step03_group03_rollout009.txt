def propose_candidate(seed=...):
    rng = np.random.default_rng(seed)
    n = choose length; maybe 8000 if time permits else 6000.
    # init w:
    w = rng.random(n)
    w = w / w.sum()
    # Adam state:
    m = np.zeros(n)
    v = np.zeros(n)
    beta1=0.9; beta2=0.999; eps=1e-8; lr0=0.2
    # parameters for smooth phase:
    alpha = 5.0; alpha_max=5000.
    # time markers:
    total_budget = budget_s - 5.0
    phase1_end = total_budget * 0.7
    iteration = 0
    start = time.monotonic()
    best_w = w.copy()
    best_score = float('inf')
    no_improve=0

    while time.monotonic() - start < total_budget:
        elapsed = time.monotonic() - start
        phase1 = elapsed < phase1_end

        if phase1:
            # compute softmax gradient
            conv = convolve_fft(w,w)
            maxc = conv.max()
            exp_vals = np.exp(alpha * (conv - maxc))
            p = exp_vals / exp_vals.sum()
            grad = 2.0 * convolve_fft(p, w[::-1])
            # Adam update
            m = beta1 * m + (1-beta1) * grad
            v = beta2 * v + (1-beta2) * (grad*grad)
            t = iteration+1
            m_hat = m/(1 - beta1**t)
            v_hat = v/(1 - beta2**t)
            lr = lr0/np.sqrt(t)
            w_new = w - lr * m_hat / (np.sqrt(v_hat)+eps)
            w = simplex_projection(w_new)
            # after fixed intervals, eval objective update best
            if iteration % 300 == 0:
                conv_now = convolve_fft(w,w)
                score = float(2*n*conv_now.max())
                if score < best_score:
                    best_score = score
                    best_w = w.copy()
                    no_improve = 0
                    print(f"[improve] iter={iteration} score={best_score}")
                else:
                    no_improve+=1
            # adjust alpha gradually
            if iteration % 1000 == 0 and alpha < alpha_max:
                alpha = min(alpha * 1.1, alpha_max)
        else:
            # hard max gradient
            conv = convolve_fft(w,w)
            k_max = int(conv.argmax())
            # compute gradient for max shift only
            # use e_k vector for conv gradient
            e = np.zeros_like(conv)
            e[k_max]=1.0
            grad = 2.0 * convolve_fft(e, w[::-1])
            # simple gradient step with diminishing lr
            lr = 0.2 / np.sqrt(iteration+1)
            w_new = w - lr * grad
            w = simplex_projection(w_new)
            if iteration % 200 == 0:
                conv_now = convolve_fft(w,w)
                score = float(2*n*conv_now.max())
                if score < best_score:
                    best_score = score
                    best_w = w.copy()
                    no_improve = 0
                    print(f"[hard improve] iter={iteration} score={best_score}")
                else:
                    no_improve+=1

        # occasional random perturbation:
        if iteration % 1500 == 0 and iteration>0:
            cand = block_perturbation(best_w, rng)
            cand = cand/ cand.sum()
            conv_cand = convolve_fft(cand,cand)
            cand_score = float(2*n*conv_cand.max())
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration} score={best_score}")

        # resets if stuck
        if no_improve > 50000:
            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            m.fill(0.0); v.fill(0.0); no_improve = 0
            print(f"[reset] iter={iteration}")

        iteration += 1

    # after loop finalize
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"[final] n={n} best_score={best_score} eval={final_score}")
    return [float(x) for x in final_seq]

"""
propose_candidate
-----------------
Hybrid optimiser for the step‑function problem.

* Phase 1 – Adam on a soft‑max (log‑sum‑exp) surrogate of the
  L∞‑norm of the self‑convolution.  The temperature α is raised
  gradually so the optimisation first smooths the landscape and then
  focuses on the current worst shift.

* Phase 2 – a hard‑max sub‑gradient step (gradient of the single
  maximal convolution entry).  This sharpens the solution once the
  temperature is already large.

* Phase 3 – aggressive random block perturbations for local escape.

The iterate is always projected onto the probability simplex and
symmetrised (w = (w+w[::-1])/2) because the objective is symmetric.
The best candidate seen so far is kept and finally rescaled to the
conventional √(2 n) L¹‑norm before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (non‑circular) via real FFT.
    The result has length len(a)+len(b)-1.  Small negative values caused by
    rounding are clipped to 0."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Randomly multiply a contiguous block by a log‑normal factor.
    Occasionally enforces symmetry."""
    cand = base.copy()
    n = cand.size
    # block width proportional to the size (between 0.5 % and 8 % of n)
    lo = max(2, n // 200)
    hi = max(4, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # sometimes symmetrise the whole vector
    if rng.random() < 0.4:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) that attempts
    to minimise ``evaluate_sequence``.  The optimiser works in three
    phases as described in the module docstring.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.  Prefer the length of a previously found
    # construction if it is available, otherwise fall back to 6000.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 6000
    else:
        n = 6000

    # --------------------------------------------------------------
    # Initialise w.  Use the previously found sequence if it exists,
    # otherwise start from a Dirichlet draw (more diverse than the
    # Catalan start).
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            w = np.asarray(height_sequence_1, dtype=np.float64)
            if w.size != n:
                raise ValueError
        except Exception:
            w = rng.random(n)
    else:
        w = rng.random(n)
    w = np.maximum(w, 0.0)
    w = w / np.sum(w)                     # enforce Σw = 1
    # enforce symmetry from the start
    w = 0.5 * (w + w[::-1])
    w = simplex_projection(w)

    # --------------------------------------------------------------
    # Book‑keeping of the best candidate seen so far.
    # --------------------------------------------------------------
    conv = convolve_fft(w, w)
    best_score = float(2 * n * np.max(conv))
    best_w = w.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser state for the smooth‑max phase.
    # --------------------------------------------------------------
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.20                # slightly larger base learning rate
    alpha = 5.0               # start temperature
    alpha_max = 5000.0        # final temperature

    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)   # safety margin
    phase1_end = total_budget * 0.65                # 65 % of the time
    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < total_budget:
        elapsed = time.monotonic() - start_time
        in_phase1 = elapsed < phase1_end

        # ----- common data -------------------------------------------------
        w_rev = w[::-1]

        if in_phase1:
            # --------- Phase 1 : smooth‑max Adam -------------------------
            conv = convolve_fft(w, w)
            maxc = np.max(conv)
            # stabilise exponentials
            exp_vals = np.exp(alpha * (conv - maxc))
            p = exp_vals / np.sum(exp_vals)          # soft‑max weights, sum=1

            # gradient of the soft‑max surrogate (see the docstring)
            grad_full = convolve_fft(p, w_rev)       # length 3n‑2
            start = n - 1
            grad = 2.0 * grad_full[start:start + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            t = iteration + 1
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / np.sqrt(t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            w_new = w - step
            w = simplex_projection(w_new)
            # keep exact symmetry (helps the objective)
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

            # slowly raise the temperature
            if iteration % 800 == 0 and alpha < alpha_max:
                alpha = min(alpha * 1.12, alpha_max)

        else:
            # --------- Phase 2 : hard‑max sub‑gradient -------------------
            conv = convolve_fft(w, w)
            k_max = int(np.argmax(conv))

            # unit vector e_k_max (length 2n‑1)
            e = np.zeros(len(conv), dtype=np.float64)
            e[k_max] = 1.0

            # gradient of the true max (sub‑gradient)
            grad_full = convolve_fft(e, w_rev)
            start = n - 1
            grad = 2.0 * grad_full[start:start + n]

            # simple decaying step size
            lr = 0.12 / np.sqrt(iteration + 1)
            w_new = w - lr * grad
            w = simplex_projection(w_new)
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

        # ----- occasional exact evaluation ---------------------------------
        if iteration % 300 == 0:
            conv_now = convolve_fft(w, w)
            score = float(2 * n * np.max(conv_now))
            if score < best_score - 1e-12:
                best_score = score
                best_w = w.copy()
                no_improve = 0
                print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
            else:
                no_improve += 1

        # ----- random block perturbations (Phase 3 style) -----------------
        if iteration % 900 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            conv_cand = convolve_fft(cand, cand)
            cand_score = float(2 * n * np.max(conv_cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # ----- reset if we seem stuck ------------------------------------
        if no_improve > 40000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant; we adopt the
    # conventional √(2 n) L¹‑norm for readability.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # Compute the official score (the harness will also do this, but we
    # print it for debugging).
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Return plain Python list as required.
    return [float(x) for x in final_seq]
