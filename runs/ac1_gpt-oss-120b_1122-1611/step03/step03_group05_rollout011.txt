"""
propose_candidate
-----------------

A hybrid optimiser for the self‑convolution peak problem.

* **Phase 1 – smooth Adam descent**  
  Minimise a soft‑max approximation of maxₖ (w*w)[k] on the probability
  simplex (w ≥0, ∑w=1).  The surrogate loss
      L_β(w)= (1/β)·log ∑ₖ exp(β·Cₖ(w)),   Cₖ(w)=(w*w)[k],
  is differentiable; its gradient is computed by an FFT‑based
  correlation 2·w⋆p where p are the soft‑max weights.
  β is raised linearly from 2 to a large value (≈ 200) – a homotopy
  that gradually focuses on the true peak.

* **Phase 2 – occasional linear‑programming polishing**  
  Every few thousand iterations we collect the “active’’ convolution
  indices (those attaining the current maximum) and build a tiny
  linear program

        maximise   ∑ g_i
        subject to (w ∗ g)[k] ≤ M   for every active k,
                   g_i ≥ 0,

  where M = maxₖ (w*w)[k].  The LP yields a vector g that can carry
  more ℓ₁‑mass than w while respecting the same peak bound.  After
  rescaling g to unit ℓ₁‑norm we mix it into w (α≈0.2) – this provably
  reduces the peak when ∑g>1.

* **Phase 3 – stochastic block jumps & resets**  
  To escape plateaux we repeatedly perturb the best‑ever solution by
  scaling a random contiguous block (≈ 1‑8 % of the length) and,
  if no improvement is seen for a long while, we reset w to a
  perturbed copy of the current best.

The whole routine respects the 1000 s budget, projects onto the
simplex after each update, and finally rescales the best probability
vector to the conventional √(2 n) ℓ¹‑norm required by
``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

def catalan_sequence(length: int) -> np.ndarray:
    """Return a₀,…,a_{length‑1} with a_i = C(2i,i)/4ⁱ.
    Recurrence a_{i+1}=a_i·(2i+1)/(2i+2) is numerically stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # numerical fallback
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # ≈ 0.5 % of length
    hi = max(2, n // 12)           # ≈ 8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ≈ ±12 %
    cand[start:stop] *= factor
    if rng.random() < 0.35:                       # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution w∗w via FFT of length N (full length 2|w|‑1)."""
    F = rfft(w, N)
    conv_full = irfft(F * F, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # eliminate tiny negatives
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth max‑approximation L_β at w.
    'probs' are the soft‑max weights (∑ probs = 1, length 2|w|-1).
    """
    Fw = rfft(w, N)
    Fs = rfft(probs, N)
    corr = irfft(Fw * np.conj(Fs), N)          # correlation w⋆probs
    m = probs.size
    offset = m - 1
    grad = 2.0 * corr[offset: offset + w.size]
    return grad

                print(f"[block] iter={iteration:6d}, new best max‑conv={best_max:.6f}")

        # ----- 7. Linear‑programming polishing (active‑set) -----
        if iteration % 6_000 == 0:
            # gather active indices (within a tiny absolute tolerance)
            tol = 1e-12
            active_idx = np.where(conv_full >= cur_max - tol)[0]
            # ensure a few edge constraints so the LP is bounded
            candidates = list(active_idx) + [0, 2 * n - 2]
            # limit the number of rows for speed
            max_rows = 30
            if len(candidates) > max_rows:
                candidates = rng.choice(candidates, size=max_rows, replace=False).tolist()
            rows = []
            for k in candidates:
                row = np.zeros(n, dtype=np.float64)
                j_start = max(0, k - (n - 1))
                j_end = min(n - 1, k)
                # vectorised assignment
                j = np.arange(j_start, j_end + 1, dtype=int)
                row[j] = w[k - j]
                rows.append(row)
            A_ub = np.vstack(rows)                      # shape (K, n)
            b_ub = np.full(A_ub.shape[0], cur_max)

            # solve LP: maximise 1ᵀg  ⇔ minimise -1ᵀg
            from scipy.optimize import linprog
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.success:
                sum_g = np.sum(res.x)
                if sum_g > 1.0 + 1e-4:               # genuine improvement possible
                    g_norm = res.x / sum_g            # unit ℓ¹‑norm
                    mix = 0.2
                    w = (1.0 - mix) * w + mix * g_norm
                    w = simplex_projection(w)        # re‑project (numerical drift)
                    print(f"[LP] iter={iteration:6d}, sum_g={sum_g:.6f}")

        # ----- 8. stagnation reset -----
        if no_improve > 25_000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

    # --------------------------------------------------------------
    # Final rescaling – transform the best probability vector to the
    # conventional √(2 n) ℓ¹‑norm (the evaluation is scale‑invariant).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum should already be 1

    # Optional sanity check (cheap):
    try:
        score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best max‑conv={best_max:.6f}, after scaling evaluate={score:.6f}")
    except Exception:
        pass

    return [float(x) for x in final_seq]
