def proj_monotone(v):
    # enforce decreasing: w[0] >= w[1] >= ... >= w[n-1]
    # Use PAVA: start with each element as a block.
    n = len(v)
    level = v.astype(float).copy()
    weight = np.ones(n)  # each weight 1
    i = 0
    while i < n-1:
        if level[i] < level[i+1] - 1e-12:  # violation (i.e., should be >=)
            # merge blocks i and i+1
            total_weight = weight[i] + weight[i+1]
            avg = (level[i]*weight[i] + level[i+1]*weight[i+1]) / total_weight
            level[i] = avg
            level[i+1] = avg
            weight[i] = total_weight
            weight[i+1] = total_weight
            # go back if needed
            j = i - 1
            while j >= 0 and level[j] < level[j+1] - 1e-12:
                total_weight = weight[j] + weight[j+1]
                avg = (level[j]*weight[j] + level[j+1]*weight[j+1]) / total_weight
                level[j] = avg
                level[j+1] = avg
                weight[j] = total_weight
                weight[j+1] = total_weight
                j -= 1
            i = max(j+1, 0)
        else:
            i += 1
    return level
```

"""
Propose a new step‑function using an Adam‑style projected gradient descent.
The optimizer works on the probability simplex (∑w = 1, w ≥ 0) and repeatedly
projects onto the simplex after an Adam update.  To keep the search space
well‑behaved we add cheap heuristics:
* periodic symmetry averaging (w ← (w+w[::-1])/2)
* occasional sorting to enforce a decreasing shape (isotonic projection)
* random block perturbations to escape plateaus
* a tiny multiplicative‐mirror update (exponential weights) used at random
The routine starts from several diverse seeds (Catalan, arcsine‑law,
random Dirichlet, previous best) and keeps the best candidate seen.
All computation is O(n log n) per iteration thanks to FFT‑based convolution.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities – top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers a_i = C(2i,i)/4**i.
    Recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def arcsine_sequence(length: int) -> np.ndarray:
    """Discrete version of the arcsine law: 1/√(x·(1‑x))."""
    x = np.arange(length) + 0.5
    seq = 1.0 / np.sqrt(x * (length + 0.5 - x))
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, ∑x=1}.
    Implementation of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = rfft(a, N)
    Fb = rfft(b, N)
    Fc = Fa * Fb
    c = irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)  # eliminate tiny negatives from round‑off
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Large cheap random jump – multiply a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetric smoothing
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Core optimisation routine.
# ----------------------------------------------------------------------


def _adam_optimize(
    w0: np.ndarray,
    n: int,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Run an Adam style projected gradient descent on the simplex.
    Returns the best vector (sum=1) and its raw score 2·n·max(conv).
    """
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5

    w = w0.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    best_w = w.copy()
    conv = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(conv)

    start = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_limit:
        # ---- current convolution and active index -----------------------
        conv = convolve_fft(w, w)
        max_val = np.max(conv)
        # pick a random index among the ties (helps avoid bias)
        idxs = np.flatnonzero(conv >= max_val - 1e-14)
        i_max = int(rng.choice(idxs))

        # ---- sub‑gradient of the active quadratic form -----------------
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        idx = i_max - j
        grad[j] = 2.0 * w[idx]

        # ---- Adam moment updates ---------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias correction
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- gradient step + simplex projection -------------------------
        w_temp = w - step
        w_temp = np.maximum(w_temp, 0.0)      # ensure non‑negative before projection
        w_new = simplex_projection(w_temp)

        # ---- occasional multiplicative (mirror) update ------------------
        if rng.random() < 0.5:
            factor = np.exp(-lr * grad)      # KL‑mirror step
            w_mirror = w * factor
            w_mirror /= np.sum(w_mirror)
            # keep the better of the two proposals
            conv_m = convolve_fft(w_mirror, w_mirror)
            score_m = 2.0 * n * np.max(conv_m)
            if score_m < 2.0 * n * np.max(conv):
                w_new = w_mirror

        w = w_new

        # ---- periodic symmetry / monotonicity enforcement ---------------
        if it % 1200 == 0 and it > 0:
            # symmetry averaging (always applied)
            w = 0.5 * (w + w[::-1])
            w /= np.sum(w)

        if it % 2500 == 0 and it > 0:
            # enforce decreasing shape by sorting (cheap projection)
            w = np.sort(w)[::-1]
            w /= np.sum(w)

        # ---- cheap random block perturbation ---------------------------
        if it % 3000 == 0 and it > 0:
            cand = block_perturbation(w, rng)
            cand /= np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                m = np.zeros_like(w)
                v = np.zeros_like(w)
                print(f"[block] iter={it:6d} new best={best_score:.6f}")

        # ---- evaluate current solution ----------------------------------
        conv = convolve_fft(w, w)
        cur_score = 2.0 * n * np.max(conv)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={it:6d} score={best_score:.6f}")
        else:
            no_improve += 1

        # ---- stagnation reset -----------------------------------------
        if no_improve > 120_000:
            w = block_perturbation(best_w, rng)
            w /= np.sum(w)
            m = np.zeros_like(w)
            v = np.zeros_like(w)
            no_improve = 0
            print(f"[reset] iter={it:6d} resetting to perturbed best.")

        it += 1

    return best_w, best_score
