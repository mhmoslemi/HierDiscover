def lp_refine(w, niter=8, t=0.5):
    n = w.size
    best_w = w.copy()
    best_M = np.max(convolve_fft(w,w))
    for it in range(niter):
        # build Toeplitz matrix
        col = np.concatenate([best_w, np.zeros(n-1)])
        row = np.concatenate([best_w[:1], np.zeros(n-1)])
        A = toeplitz(col, row)  # shape (2n-1, n)
        # Set up LP
        c = -np.ones(n)
        b_ub = np.full(2*n-1, best_M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = np.sum(g)
        if sum_g <= 1 + 1e-12:
            # no gain possible
            break
        # Create new w
        w_new = (1 - t) * best_w + t * (g / sum_g)
        M_new = np.max(convolve_fft(w_new, w_new))
        if M_new < best_M:
            best_M = M_new
            best_w = w_new
        else:
            # maybe reduce t or break
            break
    return best_w, best_M

"""Algorithm: start from simple parametric distributions, improve by
     linear‑programming majorisation (max‑convolution bound) and a short
     random polishing phase.  The LP step follows the scheme described
     in the statement: given the current distribution f, solve
         max   1ᵀg
         s.t.  f * g ≤ M,   g ≥ 0                (M = max(f*f))
     Scale the solution back to unit ℓ¹‑norm and mix with the current
     vector.  The monotone decrease of M is guaranteed.  Several
     starting points are tried, the best final vector is returned.
"""
import time
import math
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan numbers a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def arcsine_like(length: int) -> np.ndarray:
    """1/√(x·(1‑x)) distribution on a discrete grid, normalised."""
    x = np.arange(length) + 0.5
    seq = 1.0 / np.sqrt(x * (length + 0.5 - x))
    return seq / np.sum(seq)


def random_dirichlet(length: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma(0.5) Dirichlet – produces a spiky but non‑negative vector."""
    seq = rng.gamma(0.5, 1.0, size=length)
    seq /= np.sum(seq)
    return seq


def linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Fast linear convolution using real FFT (output length len(a)+len(b)-1)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Numerical noise may produce tiny negatives – clamp to zero.
    np.maximum(c, 0.0, out=c)
    return c


def max_convolution(w: np.ndarray) -> float:
    """Return max value of the linear self‑convolution of ``w``."""
    conv = linear_convolution(w, w)
    return float(np.max(conv))


def build_toeplitz_matrix(w: np.ndarray) -> np.ndarray:
    """Build the (2·n‑1)×n matrix A such that A @ g = conv(w,g)."""
    n = w.size
    col = np.concatenate([w, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([w[:1], np.zeros(n - 1, dtype=np.float64)])
    return toeplitz(col, row)


def lp_expand(w: np.ndarray, M: float, t: float = 0.5):
    """
    One LP‑based refinement step.
    Returns (w_new, M_new) or (w, M) if the LP did not improve the bound.
    """
    n = w.size
    # Linear constraints A @ g ≤ M
    A = build_toeplitz_matrix(w)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise 1ᵀg  → minimise -1ᵀg
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Solve the LP
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # LP failed – fall back to the unmodified vector.
        return w, M

    g = res.x
    sum_g = np.sum(g)
    if sum_g <= 1.0 + 1e-12:
        # No room for improvement.
        return w, M

    # Convex combination of the old and the normalised LP solution.
    w_new = (1.0 - t) * w + t * (g / sum_g)
    M_new = max_convolution(w_new)
    if M_new < M - 1e-12:
        return w_new, M_new
    return w, M


def downsample(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Simple block‑averaging down‑sampling to ``target_len`` entries."""
    n = seq.size
    if n <= target_len:
        return seq.copy()
    # ensure an integer factor for simplicity
    factor = n // target_len
    trimmed = seq[: factor * target_len]
    reshaped = trimmed.reshape(target_len, factor)
    avg = reshaped.mean(axis=1)
    avg /= np.sum(avg)
    return avg

linear‑programming majorisation step (the method described in the
statement) and finish with a cheap random‑move polishing phase.
The LP step solves

    max   1ᵀg
    s.t.  f * g ≤ M ,  g ≥ 0

where M = max(f*f).  The solution g has a larger ℓ¹‑norm, therefore
after normalisation the new vector yields a strictly smaller maximum

result is returned (scaled to √(2·n) as required by `evaluate_sequence`)."""

def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq

    """Discretised 1/√(x·(1‑x)) distribution, normalised to sum 1."""
    x = np.arange(length) + 0.5
    seq = 1.0 / np.sqrt(x * (length + 0.5 - x))
    return seq / np.sum(seq)


def random_dirichlet(length: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet with concentration 0.5 – produces a spiky start."""

    c = np.fft.irfft(Fc, N)[:n]

    return c
