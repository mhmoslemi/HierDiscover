def propose_candidate(seed=1234, budget_s=960.0):
    rng = np.random.default_rng(seed)
    # Determine length n
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        n = prev.size
        w0 = prev.astype(np.float64)
        # Convert to probability by dividing by sum
        w0 = np.maximum(w0, 0.0)
        S0 = w0.sum()
        if S0 <= 0:
            n = 8000
            w0 = catalan_sequence(n)
        else:
            w0 = w0 / S0
    else:
        n = 8000
        w0 = catalan_sequence(n)
        w0 = w0 / np.sum(w0)
    # Ensure symmetry
    w0 = (w0 + w0[::-1]) / 2
    w0 = w0 / np.sum(w0)
    # initialize logits z = log(w0 + epsilon)
    eps = 1e-12
    z = np.log(w0 + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Hyperparameters
    lr = 0.1  # base learning rate
    beta_start = 0.2
    beta_max = 300.0
    sym_every = 200  # symmetrize every X iters
    block_every = 1500
    reset_patience = 20000
    no_improve = 0
    best_w = np.exp(z)
    best_w = best_w / best_w.sum()
    best_score = float('inf')
    # start time
    start = time.monotonic()
    iteration = 0
    while time.monotonic() - start < max(0.0, budget_s - 5.0):
        w = np.exp(z)
        w = w / w.sum()
        # compute conv and score
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b
        # update best
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration}, score={best_score:.6f}")
        else:
            no_improve += 1

        # compute softmax weights α
        beta = beta_start * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b
        exp_term = np.exp(beta * diff)
        alpha = exp_term / np.sum(exp_term)

        # gradient wrt w
        grad_w = correlation_grad(w, alpha) * 2.0

        # gradient wrt logits
        dot = np.dot(w, grad_w)
        grad_z = w * (grad_w - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        # bias corrections
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step

        # symmetrize
        if iteration % sym_every == 0:
            w = np.exp(z)
            w = w / w.sum()
            w = 0.5 * (w + w[::-1])
            w = w / w.sum()
            # reset logits from w
            z = np.log(w + eps)

        # block perturbation occasional
        if iteration % block_every == 0 and iteration > 0:
            # apply perturbation to best_w
            cand = block_perturbation(best_w, rng)
            # renorm to sum=1
            cand = cand / cand.sum()
            cand_b = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(cand_b)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                # set z from cand
                z = np.log(cand + eps)
                print(f"[block] iter={iteration}, new best score={best_score:.6f}")

        # reset if stuck
        if no_improve > reset_patience:
            # random perturb best
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            z = np.log(cand + eps)
            w = cand
            no_improve = 0
            print(f"[reset] iter={iteration}, reset to perturbed best")

        iteration += 1

    # final scaling to sqrt(2n) L1 norm
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    # optional evaluate
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best score raw={best_score:.6f}, evaluate={final_score:.6f}")
    except Exception:
        print("[final] evaluation error")
    return [float(x) for x in final_seq]

def block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:start+width] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

"""
Propose a candidate step‐function using Adam‐style mirror descent.

The objective to minimise is
    F(w) = 2·n·max_k (w*w)[k] / (∑w)^2 .
Because the ratio is scale‑invariant we work with a probability vector
   p_i ≥ 0 , Σ p_i = 1.
For a fixed β the smoothed objective is
   f_β(p) = log‑sum‑exp( β·(p*p) )
and its gradient is
   ∇f_β(p) = 2·corr(p,α)  with   α_k ∝ exp(β·((p*p)_k−max(p*p))).

We optimise the (log‑softmax) parameters  z_i (so p_i = softmax(z)_i)
with Adam.  Periodic symmetrisation, cheap block perturbations and a
restart policy are used to escape plateaus.  The best probability vector
found is finally rescaled to the conventional ℓ¹‑norm √(2 n) before
returning it as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT helpers
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negatives caused by round‑off.
    np.maximum(c, 0.0, out=c)
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max:   g = 2·corr(w, α)
    w – probability vector (size n)
    α – probability vector of length 2·n‑1
    Returned array has length n.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = 3·n‑2
    start = w.shape[0] - 1                     # aligns with output index 0
    return 2.0 * conv[start:start + w.shape[0]]


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers C_i / 4**i (a convenient smooth start)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)      # ≈0.5 % of n
    hi = max(2, n // 12)       # up to ≈8 % of n
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetric smoothing
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main optimiser
# ----------------------------------------------------------------------

    """
    Return a non‑negative step function (list of floats) with a low
    `evaluate_sequence` score.  The optimiser runs for up to ``budget_s``
    seconds and prints progress.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length and an initial probability vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:

            if n < 2:

            w0 = np.maximum(prev, 0.0)
            if w0.sum() == 0.0:
                raise ValueError
            w0 = w0 / w0.sum()
        except Exception:
            n = 8000
            w0 = catalan_sequence(n)
            w0 = w0 / w0.sum()
    else:
        n = 8000
        w0 = catalan_sequence(n)
        w0 = w0 / w0.sum()

    # Enforce symmetry – the optimal solution is believed to be symmetric.
    w0 = 0.5 * (w0 + w0[::-1])
    w0 = w0 / w0.sum()

    # Work in the log‑softmax (mirror‑descent) space.
    eps = 1e-12
    z = np.log(w0 + eps)

    # Adam bookkeeping
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.12                     # base learning rate for Adam

    # Hyper‑parameters controlling the soft‑max temperature and misc. actions
    beta_start = 0.20
    beta_max = 250.0
    sym_every = 250               # enforce symmetry every this many iters
    block_every = 2500            # cheap random block jump
    reset_patience = 30000        # iterations without improvement → reset

    # Book‑keeping of the best solution seen so far
    best_w = w0.copy()
    b = convolve_fft(best_w, best_w)
    best_score = float(2.0 * n * np.max(b))      # raw (2·n·max) – the scale‑invariant value
    print(f"[init] n={n}, raw score={best_score:.6f}")

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ----------------------------------------------------------
        # Current probability vector and its (hard) score
        # ----------------------------------------------------------
        w = np.exp(z)
        w /= w.sum()
        b = convolve_fft(w, w)
        max_b = np.max(b)
        raw_score = 2.0 * n * max_b

        if raw_score < best_score - 1e-12:
            best_score = raw_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, raw score={best_score:.6f}")
        else:
            no_improve += 1

        # ----------------------------------------------------------
        # Smoothed max (soft‑max) weights α
        # ----------------------------------------------------------
        beta = beta_start * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b                     # shift for numerical stability
        exp_term = np.exp(beta * diff)       # the max element becomes 1.0
        alpha = exp_term / np.sum(exp_term)  # length 2·n‑1, sums to 1

        # ----------------------------------------------------------
        # Gradient of the smooth objective (w.r.t. w)
        # ----------------------------------------------------------
        grad_w = correlation_grad(w, alpha)   # already includes the factor 2

        # ----------------------------------------------------------
        # Transform to gradient w.r.t. logits (softmax mirror map)
        # ----------------------------------------------------------
        dot = np.dot(w, grad_w)               # scalar ⟨w,∇f⟩
        grad_z = w * (grad_w - dot)           # ∂f/∂z_i = w_i (∂f/∂w_i – ⟨w,∇f⟩)

        # ----------------------------------------------------------
        # Adam update in log‑space
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z -= step

        # ----------------------------------------------------------
        # Periodic symmetrisation (helps enforce the known symmetry)
        # ----------------------------------------------------------
        if iteration % sym_every == 0:
            w = np.exp(z)
            w /= w.sum()
            w = 0.5 * (w + w[::-1])
            w /= w.sum()
            z = np.log(w + eps)

        # ----------------------------------------------------------
        # Occasional cheap random block perturbation (exploration)
        # ----------------------------------------------------------
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()

            cand_raw = 2.0 * n * np.max(cand_b)
            if cand_raw < best_score:
                best_score = cand_raw
                best_w = cand.copy()
                z = np.log(best_w + eps)
                print(f"[block] iter={iteration:7d}, new raw best={best_score:.6f}")

        # ----------------------------------------------------------
        # Reset if stuck for a long time (jump from the best so far)
        # ----------------------------------------------------------
        if no_improve > reset_patience:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()
            z = np.log(cand + eps)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – the evaluator is scale‑invariant, but we output the
    # conventional √(2 n) ℓ¹‑norm for readability.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()

              f"evaluate={final_score:.6f}")
    except Exception:
        print("[final] evaluation error (not critical)")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
