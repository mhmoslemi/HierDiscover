"""
Propose a candidate step function for the self‑convolution minimisation task.

Algorithm
---------
1. Initialise a probability vector `w` (sum‑to‑one) from the previously
   discovered sequence `height_sequence_1` if it exists, otherwise use a
   uniform distribution.

2. **Random parametric search** (≈30 % of the time budget)  
   Generate a large number of candidate shapes defined by a handful of
   simple parametric families (power‑law, exponential, beta‑type, etc.),
   normalise each candidate to a probability vector and evaluate its
   objective `score = 2·n·max(conv(w,w))`.  Keep the best candidate.

3. **Local hill‑climbing** (remaining time)  
   Starting from the best candidate, repeatedly perform cheap stochastic
   moves:
   * move a small amount of mass from one index to another,
   * or take a sub‑gradient step at the currently maximal convolution
     index (the sub‑gradient of `max(conv(w,w))` is `2·w` shifted by the
     maximising index),
   * occasionally apply the cheap block‑perturbation used in the
     reference code.
   After each move keep the candidate if the objective improves.
   All moves are projected back onto the simplex.

4. **Final scaling** – the evaluator is scale‑invariant; we scale the best
   probability vector to `ℓ¹`‑norm `sqrt(2·n)` (the usual normalisation)
   and return it as a plain Python list.

The whole routine respects the 1000 s budget, uses only NumPy and SciPy,
and prints occasional progress messages.
"""

import time
import numpy as np
from scipy.optimize import linprog  # only imported for type checking; not used directly


# -------------------------------------------------------------
# Helper utilities – all top‑level as required by the harness
# -------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def max_convolution(seq: np.ndarray, N_fft: int) -> float:
    """
    Return the maximum value of the linear self‑convolution of `seq`.
    ``seq`` is assumed to sum to one (probability vector).  The
    convolution is performed by FFT for speed.
    """
    f = np.fft.rfft(seq, N_fft)
    conv = np.fft.irfft(f * f, N_fft)[: 2 * seq.size - 1]
    # tiny negative values can appear due to round‑off
    np.maximum(conv, 0.0, out=conv)
    return float(conv.max())


def compute_score(seq: np.ndarray, N_fft: int) -> float:
    """Objective used throughout the search: 2·n·max(conv(seq,seq))."""
    n = seq.size
    return 2.0 * n * max_convolution(seq, N_fft)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation – used as a jump."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 % of length
    hi = max(2, n // 12)           # ≈ 8 % of length
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ≈±12 % multiplicative noise
    cand[start:stop] *= factor
    if rng.random() < 0.35:                       # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

        elif shape_type == 4:
            p = rng.uniform(0.0, 2.0)
            q = rng.uniform(0.0, 4.0)
            raw = (i_arr ** (-p)) * np.exp(-q * np.arange(n) / n)

        # ----- mirrored mixed shape (decreasing on the right side) -----------------
        else:
            p = rng.uniform(0.0, 2.0)
            q = rng.uniform(0.0, 4.0)
            raw = (rev_i ** (-p)) * np.exp(-q * np.arange(n) / n)

        # optional multiplicative noise (20 % of candidates)
        if rng.random() < 0.2:
            raw *= np.exp(rng.normal(0.0, 0.02, size=n))

        # clip and normalise
        raw = np.maximum(raw, 0.0)
        s = raw.sum()
        if s <= 0.0:
            continue
        cand = raw / s                                   # probability vector

        # evaluate
        score = compute_score(cand, N_fft)
        if score < best_score:
            best_score = score
            best_w = cand
            print(f"[rand] time={int(time.monotonic() - start_time)}s "
                  f"new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local hill‑climbing / sub‑gradient polishing (≈70 % of budget)
    # ------------------------------------------------------------------
    move_end = start_time + budget - 5.0   # keep a few seconds for final scaling
    step_factor = 0.15                    # controls the size of a random transfer
    min_step = 0.001
    no_improve = 0
    while time.monotonic() < move_end:
        # --------------------------------------------------------------
        # 3a – decide the type of local move
        # --------------------------------------------------------------
        if rng.random() < 0.7:
            # ---- random mass transfer between two indices ----------------
            i = rng.integers(0, n)
            if best_w[i] < 1e-14:
                continue
            delta = best_w[i] * rng.uniform(0.0, step_factor)
            if delta <= 0.0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            w_new = best_w.copy()
            w_new[i] -= delta
            w_new[j] += delta
        else:
            # ---- sub‑gradient step at the current maximal convolution -----
            conv_full = np.convolve(best_w, best_w)
            cur_max = conv_full.max()
            max_idx = int(conv_full.argmax())
            # sub‑gradient of max‑conv at `max_idx` is 2 * best_w shifted
            grad = np.zeros_like(best_w)
            lo = max(0, max_idx - (n - 1))
            hi = min(n - 1, max_idx)
            idxs = np.arange(lo, hi + 1, dtype=int)
            grad[idxs] = 2.0 * best_w[max_idx - idxs]
            # gradient descent step with an adaptive stepsize
            eta = step_factor * (cur_max / (np.mean(best_w) + 1e-12))
            w_new = best_w - eta * grad
            w_new = np.clip(w_new, 0.0, None)
            total = w_new.sum()
            if total <= 0.0:
                continue
            w_new /= total

        # --------------------------------------------------------------
        # 3b – evaluate the candidate
        # --------------------------------------------------------------
        score = compute_score(w_new, N_fft)

        # --------------------------------------------------------------
        # 3c – accept if improvement
        # --------------------------------------------------------------
        if score < best_score - 1e-12:
            best_w = w_new
            best_score = score
            no_improve = 0
            # slowly shrink the transfer size when we are improving
            step_factor = max(min_step, step_factor * 0.995)
            print(f"[improve] time={int(time.monotonic() - start_time)}s "
                  f"score={best_score:.6f}")
        else:
            no_improve += 1
            # slowly increase the step size if we have been stuck
            if no_improve % 5000 == 0:
                step_factor = min(0.5, step_factor * 1.02)

        # occasional cheap block perturbation – helps to escape flat plateaus
        if rng.random() < 0.001:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            score = compute_score(cand, N_fft)
            if score < best_score:
                best_w = cand
                best_score = score
                print(f"[block] time={int(time.monotonic() - start_time)}s "
                      f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final scaling to the conventional √(2 n) ℓ¹‑norm
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()    # still = target_sum
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best_score={best_score:.6f}, "
              f"evaluate={final_score:.6f}")
    except Exception:
        # In case evaluate_sequence is absent in the sandbox
        pass

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
