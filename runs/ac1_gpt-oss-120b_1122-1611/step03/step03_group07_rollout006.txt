def solve_lp_for_g(f, M):
    n = len(f)
    # Build sparse matrix
    data = [np.full(n, f[k]) for k in range(n)]
    offsets = np.arange(n)
    A = diags(data, offsets, shape=(2*n-1, n), format='csc')
    b = np.full(2*n-1, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence() score.

The algorithm mixes two complementary ideas:

1. **Smooth exponential‑gradient descent** – we minimise a
   soft‑max approximation of   max_k (p * p)[k]   (p ≥ 0,
   Σp = 1).  The gradient is obtained by a soft‑max weighting of
   the self‑convolution entries and the update is performed with an
   exponentiated‑gradient step followed by a renormalisation onto
   the probability simplex.

2. **Linear‑programming refinement** – when the gradient method
   stalls we solve the LP

        maximise   Σ g_i
        subject to (p * g)_k ≤ M   for all k,
                   g_i ≥ 0 ,

   where M = max(p * p).  The LP pushes mass to positions where the
   current distribution p is small (the constraints are weakest)
   and therefore flattens the auto‑correlation.  The resulting g is
   mixed with p using a small step‑size; the mixture is guaranteed
   to keep Σ = 1 and usually lowers the objective.  The LP is
   solved with SciPy’s high‑performance “highs” backend.

   The two phases are interleaved, together with occasional
   symmetry enforcement, random block perturbations and a simple
   stagnation‑reset.

The final vector is scaled to the conventional √(2 n) L¹‑norm so
that evaluate_sequence() returns the true max‑convolution value.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers C_i = (2i choose i) / 4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (real) using FFT – fast for a few thousand points."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = np.fft.irfft(A * B, N)[:n]
    # Clip tiny negative artefacts (they would be rejected by evaluate_sequence)
    np.maximum(C, 0.0, out=C)
    return C


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max over the convolution entries, returns a probability vector."""
    shift = conv - np.max(conv)            # improve numerical stability
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(p: np.ndarray, gamma: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max‑smoothed max‑convolution.
    Returns (grad, max_convolution) where ``grad`` has the same shape as ``p``.
    """
    # self‑convolution
    conv_pp = convolve_fft(p, p)                     # length 2n‑1
    # soft‑max weights
    w = _softmax_weights(conv_pp, gamma)             # length 2n‑1
    # reverse for the correct shift pattern
    w_rev = w[::-1]
    # convolve p with w_rev and adjust the scaling factor 2
    tmp = convolve_fft(p, w_rev)                     # length 3n‑2
    grad = 2.0 * tmp[::-1][:p.size]                  # first n entries, reversed back
    return grad, np.max(conv_pp)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random block scaling – used as a jump when the search stagnates.
    """
    cand = base.copy()
    n = cand.size
    # pick a block size proportional to n (but at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal perturbation
    cand[start:stop] *= factor
    if rng.random() < 0.35:                         # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def solve_lp_for_g(p: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to (p * g)_k ≤ M   and g ≥ 0.
    Returns the optimal g (not normalised) or None on failure.
    """
    n = p.size
    # Build the (2n‑1) × n convolution matrix with p on its diagonals.
    # Each diagonal offset k (0 … n‑1) contains the constant value p[k].
    data = [np.full(n, p[k]) for k in range(n)]
    offsets = np.arange(n)                         # 0,1,…,n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M)

    # maximise Σ g_i  ⇔ minimise -Σ g_i
    c = -np.ones(n)

    # SciPy’s “highs” solver handles large sparse LPs efficiently.
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative floats – the step function that
    minimises evaluate_sequence().  The routine runs for up to
    ``budget_s`` seconds and reports intermediate bounds.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    end_time = start_time + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    max_n = 2500                     # keeps the LP comfortably sized
    if "height_sequence_1" in globals():
        try:

            if arr.ndim != 1 or arr.size == 0:
                raise ValueError
            # Down‑sample if the existing construction is too long

                step = int(np.ceil(arr.size / max_n))
                arr = arr[::step]
                if arr.size > max_n:
                    arr = arr[:max_n]
            n = int(arr.size)
            p = np.maximum(arr, 0.0)
            s = np.sum(p)
            if s == 0.0:
                p = np.full(n, 1.0 / n)
            else:
                p = p / s
        except Exception:
            n = 2000
            p = catalan_sequence(n)
            p = p / np.sum(p)
    else:
        n = 2000
        p = catalan_sequence(n)
        p = p / np.sum(p)

    # Ensure strict positivity – zero entries would make the LP unbounded.
    eps = 1e-12
    p = np.maximum(p, eps)
    p = p / np.sum(p)

    best_p = p.copy()
    conv = convolve_fft(p, p)

    # ------------------------------------------------------------------

    no_improve_limit = 25000       # stagnation threshold for a reset
