    lengths = [2000, 4000, 6000, 8000]  # but ensure not exceed memory/time
    best_score = np.inf

    start_time = time.monotonic()
    total_budget = budget_s

    for n in lengths:
        # allocate time budget for this n
        remaining = total_budget - (time.monotonic() - start_time)

        if score < best_score:
            best_score, best_seq = score, w

        # if time very tight break
        if time.monotonic() - start_time > total_budget - 5:

    # scale final best_seq to sqrt(2*n) L1 as done before
    n = best_seq.size
    target_sum = np.sqrt(2.0*n)
    final_seq = best_seq * target_sum / best_seq.sum()

        # restart if stagnated
        if iter_cnt % 20000 == 0 and iter_cnt > 0 and best_score > 1.51: # maybe seldom

            w = w / w.sum()
            m.fill(0.0); v.fill(0.0)
            print(f"[reset] n={n} iter={iter_cnt}")
        iter_cnt += 1
        # check overall time
        if time.monotonic() - start_time > budget_s - 1.0:
            break
    # final best sequence also optionally local search
    # Perform a hill climbing local search for short time
    local_start = time.monotonic()
    while time.monotonic() - local_start < min(5.0, budget_s*0.1):

        if cand_score < best_score:
            best_score = cand_score
            best_w = cand.copy()
            print(f"[local] n={n} local improve score={best_score:.6f}")
    return best_w, best_score

def refine_sequence(w, max_iters, time_budget):
    n = w.shape[0]
    best_w = w.copy()
    conv = convolve_fft(w,w)
    best_score = 2.0 * n * np.max(conv)
    m = np.zeros(n); v=np.zeros(n)
    beta1 = 0.9; beta2 = 0.999; eps=1e-8
    lr0 = 0.2
    alpha = 5.0
    iter =0
    start = time.monotonic()
    while time.monotonic() - start < time_budget:
        conv = convolve_fft(w,w)
        maxc = np.max(conv)
        k = np.argmax(conv)
        # compute gradient
        grad = np.zeros(n)
        start_i = max(0, k-(n-1))
        stop_i = min(n-1, k)
        if start_i <= stop_i:
            i = np.arange(start_i, stop_i+1)
            grad[i] = 2 * w[k - i]
        # Adam
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        t = iter+1
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        step = (lr0/np.sqrt(t)) * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step
        w = simplex_projection(w_new)
        # occasional symmetrization
        if iter % 500 == 0 and iter>0:
            w = (w + w[::-1]) * 0.5
            w = simplex_projection(w)
        # occasional evaluation
        if iter % 200 == 0:
            conv_cur = convolve_fft(w,w)
            cur_score = 2.0 * n * np.max(conv_cur)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_w = w.copy()
                # maybe print
        # occasional random perturbation
        if iter % 1000 == 0 and iter>0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            conv_cand = convolve_fft(cand,cand)
            cand_score = 2.0 * n * np.max(conv_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
        iter += 1
        if iter >= max_iters:
            break
    return best_w, best_score

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    best_score = np.inf
    best_w = None
    best_n = None
    # choose lengths
    lengths = [4000, 6000, 8000]
    for n in lengths:
        # if global seq with same n exists, include it
        candidates = []
        if 'height_sequence_1' in globals():
            try:
                hs = np.asarray(height_sequence_1, dtype=np.float64)
                if hs.size == n:
                    w = np.maximum(hs, 0.0)
                    w = w / np.sum(w)
                    candidates.append(w)
            except Exception:
                pass
        # Catalan
        w_cat = catalan_sequence(n)
        w_cat = w_cat / np.sum(w_cat)
        candidates.append(w_cat)
        # Exponential rates
        for lam in np.linspace(0.0005, 0.005, 5):
            w_exp = np.exp(-lam * np.arange(n))
            w_exp = w_exp / w_exp.sum()
            candidates.append(w_exp)
        # Power law exponents
        for p in np.linspace(0.2, 1.5, 5):
            w_pow = 1.0 / (np.arange(1, n+1) ** p)
            w_pow = w_pow / w_pow.sum()
            candidates.append(w_pow)
        # Random Dirichlet few
        for _ in range(3):
            w_rand = rng.dirichlet(np.ones(n))
            candidates.append(w_rand)
        # Evaluate each quickly and keep top few
        evaluated = [(compute_score(w, n), w) for w in candidates]
        evaluated.sort(key=lambda x: x[0])
        top_candidates = [w for _, w in evaluated[:3]]
        # For each, run refinement limited to remaining time per length
        remaining = budget_s - (time.monotonic() - start_time) - 5.0
        if remaining <= 0:
            break
        # allocate per candidate: equal share
        per_candidate = remaining / len(top_candidates)
        for w0 in top_candidates:
            # refine
            w_opt, score_opt = refine_sequence(w0, n, rng, max_iters=200000,
                                                time_budget=per_candidate)
            if score_opt < best_score:
                best_score = score_opt
                best_w = w_opt.copy()
                best_n = n
                print(f"[global] new best length {n} score {best_score:.6f}")
            # check time
            if time.monotonic() - start_time > budget_s - 5.0:
                break
        # break if out of time
        if time.monotonic() - start_time > budget_s - 5.0:
            break

    # Final scaling
    if best_w is None:
        # fallback uniform
        n = 2000
        best_w = np.full(n, 1.0/n)
        best_n = n

    target_sum = np.sqrt(2.0 * best_n)

    return [float(x) for x in final_seq]

    # Solve min ||u - v||^2 s.t. u_0 >= u_1 >= ... >= u_{n-1}
    # Equivalent to isotonic regression on -v for non-decreasing.
    n=len(v)
    u = -v.copy()
    # implement PAVA for non-decreasing
    avg = []
    weight = []
    for x in u:
        avg.append(x)
        weight.append(1)
        # check for violation
        while len(avg) >= 2 and avg[-2] > avg[-1]:
            # merge last two
            w_sum = weight[-2] + weight[-1]
            a_new = (weight[-2]*avg[-2] + weight[-1]*avg[-1]) / w_sum
            avg[-2] = a_new
            weight[-2] = w_sum
            avg.pop()
            weight.pop()
    # Now expand merged averages to output
    u_iso = np.empty(n, dtype=np.float64)
    idx = 0
    for a, w in zip(avg, weight):
        u_iso[idx:idx+w] = a
        idx += w
    # revert sign
    return -u_iso

"""
Propose a low‑score step‑function using a multi‑stage search.

Stage‑1 : generate a diverse set of simple analytic shapes
          (Catalan, exponential, power‑law, random Dirichlet) for a few
          sequence lengths and keep the best few candidates.

Stage‑2 : refine each candidate with an Adam optimiser that minimises a
          smooth‑max surrogate of max_k (w*w)[k].  The temperature of the
          soft‑max is increased gradually; we periodically project the
          vector onto the probability simplex and (optionally) enforce a
          monotone decreasing shape.

Stage‑3 : a short stochastic local‑search (greedy hill‑climbing) is
          applied to the refined candidate to escape remaining shallow
          minima.

The final vector is scaled to the conventional √(2 n) L¹‑norm before
returning, because the evaluation function is homogeneous with respect to
scaling.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # numeric safety
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def monotone_decreasing(v: np.ndarray) -> np.ndarray:
    """
    Projection onto the cone {x₀ ≥ x₁ ≥ … ≥ x_{n‑1}} (PAVA).
    The result is not normalised; caller should renormalise afterward.
    """
    # PAVA on -v (enforces non‑decreasing → original non‑increasing)
    u = -v.copy()
    avg = []      # block averages
    weight = []   # block sizes
    for x in u:
        avg.append(x)
        weight.append(1)
        while len(avg) >= 2 and avg[-2] > avg[-1] + 1e-15:
            wsum = weight[-2] + weight[-1]
            a = (weight[-2] * avg[-2] + weight[-1] * avg[-1]) / wsum
            avg[-2] = a
            weight[-2] = wsum
            avg.pop()
            weight.pop()
    # expand
    out = np.empty_like(v)
    pos = 0
    for a, wgt in zip(avg, weight):
        out[pos:pos + wgt] = a
        pos += wgt
    return -out


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:          # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Analytic families of initial vectors (sum = 1)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan–type sequence a_{i+1}=a_i·(2i+1)/(2i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq / np.sum(seq)


def exponential_sequence(length: int, lam: float) -> np.ndarray:
    """exp(‑lam·i) normalised."""
    w = np.exp(-lam * np.arange(length, dtype=np.float64))
    return w / w.sum()


def power_law_sequence(length: int, p: float) -> np.ndarray:
    """i⁻ᵖ (i starts at 1) normalised."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    w = 1.0 / (idx ** p)
    return w / w.sum()


def random_dirichlet_sequence(length: int, rng: np.random.Generator) -> np.ndarray:
    """Single Dirichlet draw (α_i = 1)."""
    w = rng.dirichlet(np.ones(length))
    return w


# ----------------------------------------------------------------------
# Scoring (sum of entries is assumed to be 1)
# ----------------------------------------------------------------------


def compute_score(w: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(w,w)).  Works for normalized w (Σw = 1)."""
    conv = convolve_fft(w, w)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Refinement with Adam + optional monotone projection
# ----------------------------------------------------------------------


def refine_with_adam(init_w: np.ndarray,
                     n: int,
                     rng: np.random.Generator,
                     max_iters: int,
                     time_budget: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a soft‑max surrogate of max(conv).
    Returns the best vector seen and its score.
    """
    w = init_w.copy()
    best_w = w.copy()
    best_score = compute_score(w, n)

    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.20               # a bit larger than the previous script
    temperature = 5.0        # initial soft‑max temperature

    start = time.monotonic()
    it = 0
    while it < max_iters and (time.monotonic() - start) < time_budget:
        # convolution of current iterate
        conv = convolve_fft(w, w)
        maxc = np.max(conv)

        # soft‑max probabilities over all shifts
        exp_vals = np.exp(temperature * (conv - maxc))
        p = exp_vals / np.sum(exp_vals)                      # length 2n‑1

        # gradient of the smoothed objective:
        # g = 2 * corr(p, w)  (correlation = conv(p, w_rev) restricted)
        w_rev = w[::-1]
        grad_full = convolve_fft(p, w_rev)                   # length 3n‑2
        start_idx = n - 1
        grad = 2.0 * grad_full[start_idx:start_idx + n]

        # Adam update on the probability simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = it + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w = simplex_projection(w_new)

        # occasional monotone decreasing projection (helps to keep the
        # “Catalan‑like” shape)
        if it % 1200 == 0 and it > 0:
            w = monotone_decreasing(w)
            w = simplex_projection(w)

        # occasional symmetrisation (helps to avoid asymmetric traps)
        if it % 2500 == 0 and it > 0:
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

        # improve record every few hundred steps
        if it % 200 == 0:
            cur_score = compute_score(w, n)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_w = w.copy()
                # print progress (non‑essential for final result)
                print(f"[adam] n={n} it={it:6d} score={best_score:.6f}")

        # block perturbation as a “restart” move
        if it % 1500 == 0 and it > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = compute_score(cand, n)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[adam‑block] n={n} it={it:6d} new best={best_score:.6f}")

        # inflate temperature slowly (makes the surrogate sharper)
        if it % 500 == 0 and it > 0:
            temperature = min(temperature * 1.08, 3000.0)

        it += 1

    return best_w, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (greedy)
# ----------------------------------------------------------------------


def hill_climb(w: np.ndarray,
               n: int,
               rng: np.random.Generator,
               time_budget: float) -> tuple[np.ndarray, float]:
    """
    Very cheap local search that transfers a random mass chunk between two
    coordinates and keeps the move if it reduces the true objective.
    """
    cur_w = w.copy()
    cur_score = compute_score(cur_w, n)
    best_w = cur_w.copy()
    best_score = cur_score

    start = time.monotonic()
    it = 0
    while (time.monotonic() - start) < time_budget:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer up to 30 % of the donor mass
        eps = rng.random() * cur_w[i] * 0.3
        if eps <= 0.0:
            continue
        cand = cur_w.copy()
        cand[i] -= eps
        cand[j] += eps
        # remain on the simplex automatically (sum unchanged)
        cand_score = compute_score(cand, n)
        if cand_score < cur_score - 1e-12:
            cur_w = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_w = cur_w.copy()
                print(f"[hill] n={n} it={it:6d} score={best_score:.6f}")
        it += 1
    return best_w, best_score

def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,

    """

    score ≈ 1.5025.  The function works in three stages:
    * generate a pool of analytic initial guesses,
    * refine a few promising candidates with Adam,
    * finish with a short greedy hill‑climb.
    The best vector (scaled to √(2 n) L¹‑norm) is returned.
    """
    rng = np.random.default_rng(seed)
    start_overall = time.monotonic()

    # ------------------------------------------------------------------
    # Choose several lengths to try – the number of entries influences the
    # scaling factor 2 n in the objective.
    # ------------------------------------------------------------------
    length_candidates = [4000, 6000, 8000]

    global_best_w = None
    global_best_score = np.inf
    global_best_n = None

    for n in length_candidates:
        # ------------------------------------------------------------------
        # Build a short list of diverse initial vectors for this length.
        # ------------------------------------------------------------------
        init_list = []

        # (a) Catalan‑type (already a strong baseline)
        init_list.append(catalan_sequence(n))

        # (b) Exponential families
        for lam in np.linspace(0.0005, 0.0045, 5):
            init_list.append(exponential_sequence(n, lam))

        for p in np.linspace(0.25, 0.75, 5):
            init_list.append(power_law_sequence(n, p))

        # (d) Random Dirichlet draws
        for _ in range(4):
            init_list.append(random_dirichlet_sequence(n, rng))

        # (e) Re‑use the previously found construction if it has matching size
        if 'height_sequence_1' in globals():
            try:
                hs = np.asarray(height_sequence_1, dtype=np.float64)
                if hs.size == n:
                    w = np.maximum(hs, 0.0)
                    w = w / np.sum(w)
                    init_list.append(w)
            except Exception:
                pass

        # ------------------------------------------------------------------
        # Evaluate all initial candidates (very cheap) and keep the top few.
        # ------------------------------------------------------------------
        scored = [(compute_score(w, n), w) for w in init_list]
