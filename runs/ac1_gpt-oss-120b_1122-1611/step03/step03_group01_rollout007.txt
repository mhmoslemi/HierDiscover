"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimizes
the evaluation metric

    score = 2 * n * max( w * w )     (w is the vector, * denotes linear convolution)

where the vector is finally scaled so that its ℓ¹‑norm equals √(2n) –
the scaling does not affect the objective.  The algorithm works in three
phases:

1. **Fast parametric sweep** – a handful of one‑parameter families
   (inverse‑power, sin‑inverse, power‑law, Catalan, uniform…) are
   generated, normalised to the simplex and the best shape is kept.
2. **Refinement with adaptive projected Adam** – starting from the best
   candidate a projected Adam optimiser (sub‑gradient of the active
   convolution entry) is run on the probability simplex.  After every
   update the vector is symmetrised (w←½(w+w[::-1])) and projected back
   onto the simplex.  Occasional block perturbations and tiny mass
   transfers are tried to escape plateaus.
3. **Final simulated‑annealing polish** – with the remaining time a
   simple annealing that moves tiny mass chunks is performed.

All operations use FFT‑based convolution (O(n log n)) and the routine
automatically respects the 1000 s time budget supplied by the harness.

The returned list is a plain Python list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def autocorr_fft(w: np.ndarray, N_fft: int) -> np.ndarray:
    """Linear autocorrelation w ★ w using a single FFT pair (real‑valued)."""
    F = np.fft.rfft(w, N_fft)
    c = np.fft.irfft(F * F, N_fft)[: 2 * w.size - 1]
    # eliminate tiny negatives caused by round‑off
    np.maximum(c, 0.0, out=c)
    return c


def score_of(w: np.ndarray, N_fft: int) -> float:
    """Scaled score = 2·n·max(w★w) (identical to evaluate_sequence after scaling)."""
    n = w.size
    conv = autocorr_fft(w, N_fft)
    return 2.0 * n * np.max(conv)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Large random jump: multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetrisation (often beneficial)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= np.sum(cand)                     # re‑normalise to the simplex
    return cand


# ----------------------------------------------------------------------
# Parametric families – cheap to evaluate
# ----------------------------------------------------------------------


def family_inverse_power(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ 1 / ((i+0.5)*(n-i+0.5))**alpha   (symmetric, heavy tails)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    vals = 1.0 / ((idx * (n + 0.5 - idx)) ** alpha)
    vals = (vals + vals[::-1]) * 0.5  # enforce exact symmetry
    vals /= np.sum(vals)
    return vals


def family_sin_inverse(n: int, beta: float) -> np.ndarray:
    """w_i ∝ sin(π·(i+0.5)/n)^{-beta}  (blows up at the ends, symmetric)."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    sinv = np.sin(theta)
    # avoid division by zero at the edges – add a tiny epsilon
    sinv = np.maximum(sinv, 1e-12)
    vals = sinv ** (-beta)
    vals = (vals + vals[::-1]) * 0.5
    vals /= np.sum(vals)
    return vals


def family_power_law(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ (i+0.5)^{-alpha} + (n-i+0.5)^{-alpha} (symmetric power‑law)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    left = idx ** (-alpha)
    right = (n + 0.5 - idx) ** (-alpha)
    vals = left + right
    vals = (vals + vals[::-1]) * 0.5
    vals /= np.sum(vals)
    return vals


def catalan_distribution(n: int) -> np.ndarray:
    """First n Catalan numbers a_i = C(2i,i)/4**i, normalised to sum 1."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    seq /= np.sum(seq)
    return seq


def uniform_distribution(n: int) -> np.ndarray:
    w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a list of non‑negative heights whose evaluation score is as low
    as possible.  The routine respects the 1000 s budget (a safety margin
    of 5 s is kept), uses a short parametric sweep for a good start and
    then refines the solution with a projected Adam optimiser and a final
    simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length of the vector.
    # --------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(seq0.size)
        if n < 2:
            raise ValueError
    except Exception:
        n = 8000                     # comfortable default size

    # Pre‑compute FFT length (next power of two ≥ 2n‑1)
    N_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep – build a small diverse portfolio.
    # --------------------------------------------------------------
    candidates = []

    # uniform baseline
    candidates.append(uniform_distribution(n))

    # Catalan sequence – classic strong baseline
    candidates.append(catalan_distribution(n))

    # arcsine‑like (inverse‑power) family
    for alpha in np.linspace(0.05, 0.55, 11):
        candidates.append(family_inverse_power(n, alpha))

    # sin‑inverse family
    for beta in np.linspace(0.10, 2.00, 20):
        candidates.append(family_sin_inverse(n, beta))

    # power‑law family
    for alpha in np.linspace(0.2, 2.0, 10):
        candidates.append(family_power_law(n, alpha))

    # random Dirichlet (α=0.5 gives spiky starts)
    for _ in range(3):
        rand = rng.gamma(0.5, 1.0, size=n)
        candidates.append(rand / np.sum(rand))

    # keep the best coarse candidate
    best_w = None
    best_sc = np.inf
    for idx, cand in enumerate(candidates):
        sc = score_of(cand, N_fft)
        print(f"[sweep] cand {idx:02d} → score {sc:.6f}")
        if sc < best_sc:
            best_sc = sc
            best_w = cand.copy()
    print(f"[sweep] chosen initial best score = {best_sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Refinement with projected Adam on the simplex.
    # --------------------------------------------------------------
    w = best_w.copy()
    # enforce exact symmetry – the optimum is always symmetric
    w = 0.5 * (w + w[::-1])
    w /= np.sum(w)

    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                     # initial learning rate (before decay)

    start_time = time.monotonic()
    max_time = max(0.0, float(budget_s) - 5.0)   # keep a safety margin
    iteration = 0
    no_improve = 0

    while time.monotonic() - start_time < max_time - 30.0:  # leave ~30 s for SA
        # ---- current convolution & active index ---------------------------------
        conv = autocorr_fft(w, N_fft)
        max_val = np.max(conv)
        # indices whose value is within a tiny tolerance of the maximum
        active = np.flatnonzero(conv >= max_val - 1e-15)
        i_max = int(rng.choice(active))

        # ---- sub‑gradient of the active quadratic term -------------------------
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        idx_pair = i_max - j
        grad[j] = 2.0 * w[idx_pair]          # d (w★w)[i_max] / d w_j

        # ---- Adam update (bias‑corrected) ---------------------------------------
        t = iteration + 1
        lr = lr0 / np.sqrt(t)                # simple decay on top of Adam's adaptivity
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- project back onto the simplex ---------------------------------------
        w_temp = w - step
        w_new = simplex_projection(w_temp)

        # ---- enforce symmetry every iteration (very cheap) -----------------------
        w_new = 0.5 * (w_new + w_new[::-1])

        w_prev = w
        w = w_new

        # ---- evaluation -----------------------------------------------------------
        sc = score_of(w, N_fft)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:07d}  score={best_sc:.6f}")
        else:
            no_improve += 1

        # ---- occasional large random jump (helps escape plateaus) ---------------
        if iteration % 2000 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            sc_cand = score_of(cand, N_fft)
            if sc_cand < best_sc:
                best_w = cand.copy()
                best_sc = sc_cand
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[block] iter={iteration:07d}  new best={best_sc:.6f}")

        # ---- tiny random mass transfer (probabilistic local move) ---------------
        if rng.random() < 0.02:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = rng.uniform(0.0, min(w[i], 0.01))
            if delta > 0.0:
                cand = w.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = simplex_projection(cand)
                cand = 0.5 * (cand + cand[::-1])
                sc_cand = score_of(cand, N_fft)
                if sc_cand < best_sc:
                    best_w = cand.copy()
                    best_sc = sc_cand
                    w = cand.copy()
                    m.fill(0.0)
                    v.fill(0.0)
                    print(f"[move] iter={iteration:07d}  improved to {best_sc:.6f}")

        # ---- reset if we have been stagnant far too long ------------------------
        if no_improve > 150_000:
            w = block_perturbation(best_w, rng)
            w = simplex_projection(w)
            w = 0.5 * (w + w[::-1])
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:07d}  resetting to perturbed best.")

        iteration += 1

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polishing (uses the remaining time)
    # --------------------------------------------------------------
    remaining = budget_s - (time.monotonic() - start_time)
    if remaining > 10.0:
        temperature = 0.02
        cur_w = best_w.copy()
        cur_sc = best_sc
        anneal_start = time.monotonic()
        t_iter = 0
        while time.monotonic() - anneal_start < remaining - 3.0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_transfer = min(cur_w[i], 0.02)
            delta = rng.uniform(0.0, max_transfer)
            if delta == 0.0:
                continue
            cand = cur_w.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = simplex_projection(cand)
            cand = 0.5 * (cand + cand[::-1])
            sc_cand = score_of(cand, N_fft)
            # Metropolis acceptance
            if sc_cand < cur_sc or rng.random() < np.exp((cur_sc - sc_cand) / temperature):
                cur_w = cand
                cur_sc = sc_cand
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_w = cur_w.copy()
            # exponential cooling
            temperature *= 0.9999
            temperature = max(temperature, 1e-6)
            t_iter += 1
        print(f"[anneal] finished after {t_iter} steps, best score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Final scaling to √(2·n) (the evaluation function is invariant to scaling)
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # final sanity check using the already‑imported evaluate_sequence
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best score (pre‑scale)={best_sc:.6f}, "
          f"after full evaluation={final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
