```
def evaluate_sequence(sequence):
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]  # enforce non-negative
    sequence = [min(1000.0, x) for x in sequence] # max bound
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

Thus we want to minimize: `2 * n * max(conv(seq, seq)) / (sum(seq)**2)` under non-negative and bounded constraints.

We can scale the sequence arbitrarily because the objective is homogeneous with respect to scaling: Suppose we multiply all elements by factor λ > 0, then sum scales as λ, max_conv scales as λ^2 (since convolution is bilinear). So objective becomes `2 * n * λ^2 * max_c / (λ^2 * sum^2) = 2 * n * max_c / sum^2`. Actually sum scales as λ sum_orig, max_c scales as λ^2 max_conv_orig. """
Propose a low‑peak step‑function using a simple simulated‑annealing
search on a symmetric probability vector.

The algorithm works as follows:

1.  Initialise a symmetric non‑negative vector `w` with Σw = 1.
    If a previously found construction (`height_sequence_1`) exists it is
    used, otherwise a uniform distribution is taken.

2.  The quality of a vector is measured by
        score = 2·n·max(conv(w,w))
    (this is exactly the value returned by ``evaluate_sequence`` after the
    required scaling).

3.  Repeatedly propose a *symmetric* mass transfer:
        – choose two positions i and j,
        – move a random amount δ from the (i, n‑1‑i) pair to the
          (j, n‑1‑j) pair (the centre element if n is odd belongs to a
          single pair),
        – keep the vector non‑negative and renormalise onto the simplex.

    The move is accepted if it improves the score or with a Metropolis
    probability that depends on a temperature `T`.  The temperature is
    slowly decreased (simulated annealing).

4.  Every few thousand iterations a larger “block‑perturbation”
    (multiplying a random contiguous block by a log‑normal factor) is
    tried – this helps escaping shallow local minima.

5.  The best vector seen is finally scaled to the conventional
    √(2 n) L¹‑norm and returned as a plain Python list.

The whole search fits comfortably into the 1000 s budget; with a
length of a few thousand the FFT‑based convolutions are cheap and the
method typically improves the previous best bound (≈1.517) toward the
target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution using a real FFT.
    Input arrays are one‑dimensional; result length = len(a)+len(b)-1.
    Small negative round‑off errors are clipped to zero.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x≥0, Σx = 1}.  Implements the O(n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly select a contiguous block and multiply it by a log‑normal factor.
    With probability 0.35 the result is symmetrised by averaging with its
    mirror image.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def evaluate_score(w: np.ndarray) -> float:
    """
    Return the optimisation score for a probability vector `w` (Σw = 1):
        score = 2 * n * max(conv(w,w))
    The convolution is performed with the FFT routine.
    """
    n = w.size
    conv = convolve_fft(w, w)
    return 2.0 * n * np.max(conv)

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    the supplied `evaluate_sequence`.  A simple simulated‑annealing
    optimiser on a symmetric probability vector is used.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose length – reuse a known construction if available,
    # otherwise fall back to a few thousand entries.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 6000
    else:
        n = 6000

    # ------------------------------------------------------------------
    # Initialise a symmetric probability vector w (Σw = 1).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.shape[0] == n:
                w = np.maximum(init, 0.0)
            else:
                raise ValueError
        except Exception:
            w = np.ones(n, dtype=np.float64) / n
    else:
        w = np.ones(n, dtype=np.float64) / n

    # enforce symmetry (mirror around the centre)
    w = 0.5 * (w + w[::-1])
    w = simplex_projection(w)          # guard against tiny numerical drift

    # ------------------------------------------------------------------
    # Book‑keeping of the best candidate.
    # ------------------------------------------------------------------
    best_w = w.copy()
    best_score = evaluate_score(w)
    print(f"[init] n={n}, start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters.
    # ------------------------------------------------------------------
    T = 1.0               # initial temperature
    T_min = 1e-4
    decay = 0.9999        # exponential cooling per iteration
    it = 0

    start_time = time.monotonic()
    time_limit = max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() - start_time < time_limit:
        # ---- 1) propose a symmetric mass transfer --------------------
        i = rng.integers(0, n)                     # source index
        j = rng.integers(0, n)                     # destination index
        if i == j:
            continue

        i_sym = n - 1 - i
        j_sym = n - 1 - j

        # maximum transferable mass from the source pair
        if i_sym == i:
            max_delta = w[i]
        else:
            max_delta = min(w[i], w[i_sym])

        if max_delta <= 0.0:
            continue

        # step size – scaled with the current temperature
        delta = rng.random() * max_delta * min(0.5, T)
        if delta <= 0.0:
            continue

        # build the candidate vector (respecting symmetry)
        cand = w.copy()
        # subtract from source pair
        cand[i] -= delta
        if i_sym != i:
            cand[i_sym] -= delta
        # add to destination pair
        cand[j] += delta
        if j_sym != j:
            cand[j_sym] += delta

        # numerical safety
        np.maximum(cand, 0.0, out=cand)

        # Minor drift away from the simplex may appear (especially for the
        # centre element when n is odd); re‑project.
        cand = simplex_projection(cand)

        # ---- 2) evaluate candidate ----------------------------------
        cand_score = evaluate_score(cand)

        # ---- 3) acceptance test ------------------------------------
        if cand_score < best_score or rng.random() < np.exp((best_score - cand_score) / T):
            w = cand
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                print(f"[improve] iter={it:6d}, score={best_score:.6f}")

        # ---- 4) temperature decay ----------------------------------
        T = max(T_min, T * decay)

        # ---- 5) occasional large block perturbation -----------------
        if it % 2000 == 0 and it > 0:
            blk = block_perturbation(best_w, rng)
            blk = blk / np.sum(blk)               # renormalise
            blk_score = evaluate_score(blk)
            if blk_score < best_score:
                best_w = blk.copy()
                best_score = blk_score
                w = blk.copy()
                print(f"[block] iter={it:6d}, score={best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant.  The benchmark
    # convention uses the √(2 n) L¹‑norm.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()

    # Compute the official score (the harness will also do this,
    # but printing is useful for debugging).
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Return as a plain Python list.
    return [float(x) for x in final_seq]
