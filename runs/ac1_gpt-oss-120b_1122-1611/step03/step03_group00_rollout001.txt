def _gradient_fft(a, conv, beta, fft_len_self):
    n = a.size
    w = np.exp(beta * (conv - conv.max()))
    # Compute convolution of w and reversed a.
    a_rev = a[::-1]
    # FFT size for gradient conv
    grad_fft_len = 1 << ((len(w) + len(a_rev) - 1 - 1).bit_length())
    fw = np.fft.rfft(w, n=grad_fft_len)
    fa_rev = np.fft.rfft(a_rev, n=grad_fft_len)
    conv2 = np.fft.irfft(fw * fa_rev, n=grad_fft_len)  # length = grad_fft_len
    # now slice out indices n-1 to n-1 + n
    start = n - 1
    grad = 2.0 * conv2[start:start + n]
    return grad

    minima.

The schedule for the learning‑rate (η) and the temperature (β) is linear
in time, starting with a large step‑size and a very smooth surrogate,
and ending with a small step‑size and a sharp focus on the maximal
convolution entry.  The best sequence seen during the allotted budget
is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n(n: int) -> int:
    """Length of the FFT that can hold the linear convolution of two vectors of size n."""
    size = 2 * n - 1
    return _next_pow2(size)


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution ``a * a`` using an FFT of length ``fft_len``.
    Returns a vector of length ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_w(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given the weight vector ``w`` (same size as ``conv(a,a)``) compute

        g_j = 2 * Σ_k w_k * a_{k-j}

    which is the gradient of the soft‑max surrogate of the max‑norm.
    The convolution is performed with an FFT for speed.
    """
    n = a.size
    a_rev = a[::-1]

    total_len = len(w) + len(a_rev) - 1
    fft_len = _next_pow2(total_len)

    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(fw * fa, n=fft_len)

    start = n - 1
    grad = 2.0 * conv[start : start + n]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that
    minimizes ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initialise the candidate vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full(cand.size, 1.0 / cand.size, dtype=np.float64)
            else:
                raise ValueError
        except Exception:
            a = np.full(256, 1.0 / 256, dtype=np.float64)
    else:
        a = np.full(256, 1.0 / 256, dtype=np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Upsample to a comfortable power‑of‑two length (≥ 8192)
    # ------------------------------------------------------------------
    target_n = max(8192, _next_pow2(n))
    if n < target_n:
        old_idx = np.linspace(0.0, 1.0, n)
        new_idx = np.linspace(0.0, 1.0, target_n)
        a = np.interp(new_idx, old_idx, a)
        a = a / a.sum()
        n = target_n

    fft_len_self = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 3) Book‑keeping for the best solution
    # ------------------------------------------------------------------
    conv = _conv_self_fft(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * np.max(conv)
    last_improve = time.monotonic()

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters and schedules
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    eta_start = 5.0          # large step‑size initially
    eta_end = 0.02           # tiny step‑size at the end
    beta_start = 0.1         # very smooth soft‑max at the beginning
    beta_end = 250.0         # focus sharply on the current peak finally
    momentum = 0.9
    restart_gap = 30.0       # seconds without improvement → restart
    block_perturb_prob = 0.04
    pair_move_prob = 0.07
    max_grad_clip = 30.0     # clip gradient before exponentiation

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0

        # linear schedules
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- self‑convolution and current score -----
        conv = _conv_self_fft(a, fft_len_self)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        # ----- record the best solution so far -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] new best score = {best_score:.6f}")

        # ----- gradient of the soft‑max surrogate -----
        # w_k = exp(beta * (conv_k - max_conv))  (all entries are ≤ 1)
        w = np.exp(beta * (conv - max_conv))
        grad = _gradient_from_w(a, w)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient (mirror descent) update -----
        factor = np.exp(-eta * np.clip(grad, -max_grad_clip, max_grad_clip))
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- optional pair‑wise mass transfer with acceptance test -----
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta_max = min(a[i], 0.3 * a[i])
                if delta_max > 0.0:
                    delta = rng.random() * delta_max
                    a_candidate = a.copy()
                    a_candidate[i] -= delta
                    a_candidate[j] += delta
                    # sum unchanged → no re‑normalisation needed
                    conv_cand = _conv_self_fft(a_candidate, fft_len_self)
                    cand_score = 2.0 * n * np.max(conv_cand)
                    if cand_score < cur_score:
                        a = a_candidate
                        cur_score = cand_score
                        max_conv = np.max(conv_cand)
                        # possibly update the global best
                        if cand_score < best_score:
                            best_score = cand_score
                            best = a.copy()
                            last_improve = time.monotonic()
                            print(f"[{last_improve - start_time:6.2f}s] new best (pair move) = {best_score:.6f}")

        # ----- soft restart when progress stalls -----
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.4, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
