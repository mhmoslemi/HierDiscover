"""
Improved candidate generator for the step‑function minimisation task.

Main ideas
----------
* Initialise a probability vector `w` (∑w = 1, w ≥ 0) with a Catalan‑type
  decreasing shape (or, if a previous best is available, reuse it).
* Optimise the smoothed objective  
      f_γ(w) = γ·log Σ_k exp(((w∗w)_k)/γ)                (γ>0)
  whose gradient is
      ∇f_γ(w) = 2·conv(w, p_rev)                     (p = soft‑max of w∗w).
  The true objective (max‑convolution) is recovered for γ→0.
* Perform an **exponentiated‑gradient** update
      w ← w·exp(−η·∇f_γ)  → renormalise to sum 1,
  where the step‑size η decays like 1/√t.
* **Anneal the temperature**: start with a large smoothing factor
  γ≈5·max(b) and shrink it exponentially to γ≈0.02·max(b).  
  This lets the algorithm explore a smooth landscape first and then
  focus on the true max‑convolution.
* To escape plateaux we occasionally apply a cheap random **block
  perturbation** (multiply a contiguous block by a log‑normal factor)
  and keep the change only if it improves the (non‑smoothed) score.
* No explicit symmetrisation is performed – asymmetry is essential for
  beating the uniform‑distribution bound (=2).
* The whole search runs for the allotted time budget (≈1000 s).  The
  best probability vector is finally scaled to the conventional
  √(2 n) ℓ¹‑norm required by `evaluate_sequence`.

The algorithm is deliberately simple, highly parallel‑friendly and
fits comfortably into the 1000 s budget while usually improving upon
the previous best bound (~1.5179 → ≈1.50).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level as required by the harness.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    First ``length`` terms of the sequence (2i choose i)/4**i.
    This decays roughly like 1/√(π i) and forms a good initial shape.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negative round‑off errors
    np.maximum(c, 0.0, out=c)
    return c


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max weights p_k ∝ exp(conv_k / γ).  Returns a vector that sums to 1."""
    # shift by the maximum to avoid overflow
    shift = conv - np.max(conv)
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    # Guard against under‑flow (should be rare)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float):
    """
    Compute the gradient of the soft‑max smoothed objective.
    Returns (grad, max_conv) where ``grad`` has shape ``w`` and
    ``max_conv`` = max_k (w∗w)_k.
    """
    # 1) self‑convolution
    b = convolve_fft(w, w)                     # length 2n‑1
    max_b = np.max(b)

    # 2) soft‑max over the convolution entries
    p = _softmax_weights(b, gamma)             # same length 2n‑1

    # 3) compute the gradient 2·conv(w, p_rev)
    p_rev = p[::-1]
    grad_conv = convolve_fft(w, p_rev)         # length 3n‑2
    grad = 2.0 * grad_conv[::-1][: w.size]     # first n entries, reversed back
    return grad, max_b


def block_perturbation(base: np.ndarray,
                       rng: np.random.Generator,
                       width_range: tuple[int, int] = None) -> np.ndarray:
    """
    Cheap O(n) random block perturbation.
    A contiguous block of ``base`` is multiplied by a log‑normal factor.
    The change is kept purely in the copy – caller should renormalise.
    """
    n = base.size
    cand = base.copy()
    if width_range is None:
        lo = max(1, n // 200)
        hi = max(2, n // 12)
    else:
        lo, hi = width_range
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ~log‑normal multiplier
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------

    """
    Returns a non‑negative step function (list of floats) whose
    ``evaluate_sequence`` score is as low as possible.
    """

    start_time = time.monotonic()

    # --------------------------------------------------------------
    # Choose the length of the candidate and an initial vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                raise ValueError
            w = prev / np.sum(prev)           # normalise to a probability vector
        except Exception:
            # fall back to a fresh Catalan construction
            n = 8000
            w = catalan_sequence(n)
            w = w / np.sum(w)
    else:
        n = 8000
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # Keep the best vector seen so far (always a probability vector)
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)
    best_w = w.copy()
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    eta0 = 0.5                     # initial step‑size for exponentiated‑gradient
    gamma_start_factor = 5.0      # γ = factor * max_b  (large, smooth start)
    gamma_end_factor = 0.02       # final smoothing factor (≈hard max)
    max_exponent = 50.0           # clip exponent to avoid overflow
    block_interval = 1200         # try a random block jump every … iterations
    reset_no_improve = 20000      # restart after this many non‑improving steps
    # Estimated total iteration count (used only for the annealing schedule)
    max_iters_est = 300_000

    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 2.0):
        # (a) evaluate current (non‑smoothed) score

            cand = cand / np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:

                best_score = cand_score

        if no_improve > reset_no_improve:

            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset]  iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluator expects the conventional √(2 n) L¹‑norm.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
