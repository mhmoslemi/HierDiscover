```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def convolve_fft(a,b):
    # same as previous, maybe using FFT
    n = a.size + b.size - 1
    N = 1 << (n-1).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    fc = fa * fb
    c = np.fft.irfft(fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c

def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # b_i contributes to rows i..i+n-1 with weights f[0..n-1]
        start_row = i
        end_row = i + n    # exclusive
        # for j in 0..n-1, row = i + j, col = i, data = f[j]
        # we can add all at once via repeated pattern
        rows.extend(range(start_row, end_row))
        cols.extend([i] * n)
        data.extend(f.tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A
```

def solve_lp_max_sum(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # minimization of -sum => maximize sum
    # call linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    b_opt = np.maximum(res.x, 0)
    return b_opt
```

"""
propose_candidate
-----------------

A hybrid stochastic search for the step‑function that minimises the
objective of ``evaluate_sequence``.  The method combines three complementary
ideas:

1.  **Projected stochastic local moves** – transfer a small amount of mass
    between two coordinates while staying on the probability simplex.
2.  **Targeted reduction of the peak convolution entry** – the current
    maximum of the autocorrelation is identified and a move reduces the
    product that creates it.
3.  **Periodic block perturbations and symmetry enforcement** – occasional
    larger jumps (log‑normal scaling of a random block) and averaging with
    the mirror image help escape shallow local minima.

All operations are performed on a normalised vector (L¹‑norm = 1); the
final sequence is rescaled to the conventional √(2 n) norm required by the
harness.  The algorithm runs until the allotted time budget is exhausted
(≈ 1000 s) and keeps the best solution seen.

The search is deliberately simple, fast (FFT‑based convolution) and
does not depend on external optimisation libraries, making it robust
under the competition constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers scaled by 4⁻ⁱ (a convenient non‑negative start)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using an FFT (real, non‑circular).  Small negatives are clipped."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()           # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # protect against -0 caused by rounding
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) jump – pick a random contiguous block and multiply it by a
    log‑normal factor.  With probability 0.35 the result is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def target_score(conv_max: float, n: int) -> float:
    """
    Given the maximum entry of the (unit‑sum) self‑convolution,
    return the score that ``evaluate_sequence`` would produce after the
    final √(2 n) scaling.
    """
    return 2.0 * n * conv_max                     # equivalent to the official score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The algorithm is a fast stochastic
    optimiser that works directly on the probability simplex.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose the length of the candidate.  Prefer a previously discovered
    # construction (height_sequence_1); otherwise fall back to a few thousand.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init_arr = np.asarray(height_sequence_1, dtype=np.float64)
            n = init_arr.size
            if n < 3:
                raise ValueError
            w = np.maximum(init_arr, 0.0)
        except Exception:
            n = 6000
            w = catalan_sequence(n)
    else:
        n = 6000
        w = catalan_sequence(n)

    # --------------------------------------------------------------
    # Normalise to a probability vector (L¹ = 1).  The evaluation is
    # scale‑invariant, so we keep the sum = 1 throughout the search.
    # --------------------------------------------------------------
    w_sum = np.sum(w)
    if w_sum <= 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / w_sum

    # --------------------------------------------------------------
    # Book‑keeping of the best candidate seen so far.
    # --------------------------------------------------------------
    conv = convolve_fft(w, w)
    best_conv_max = np.max(conv)
    best_w = w.copy()
    best_score = target_score(best_conv_max, n)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for the stochastic search
    # --------------------------------------------------------------
    start_time = time.monotonic()
    it = 0
    no_improve = 0

    # Temperature schedule for Metropolis acceptance (starts relatively warm)
    T0 = 0.015                                 # initial temperature (in score units)
    T_decay = 0.9995                           # exponential decay per iteration
    eps_frac = 0.025                           # maximal fraction of a coordinate that can be moved
    sym_interval = 500                         # how often to enforce symmetry
    block_interval = 2000                      # how often to try a block jump
    restart_patience = 25000                   # iterations without improvement before a restart

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ----------------------------------------------------------
        # Decide which type of move to perform
        # ----------------------------------------------------------
        move_type = rng.integers(0, 100)

        # ----------------------------------------------------------
        # 1) Simple local transfer between two random indices
        # ----------------------------------------------------------
        if move_type < 55:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # transfer a small amount from i to j
            max_transfer = min(w[i], eps_frac * w[i])
            delta = rng.random() * max_transfer
            if delta <= 0.0:
                continue
            cand = w.copy()
            cand[i] -= delta
            cand[j] += delta

        # ----------------------------------------------------------
        # 2) Targeted reduction of the current peak of the convolution
        # ----------------------------------------------------------
        elif move_type < 80:
            conv = convolve_fft(w, w)
            k_max = int(np.argmax(conv))
            # valid i range for the peak k_max
            i_lo = max(0, k_max - (n - 1))
            i_hi = min(n - 1, k_max)
            idx = np.arange(i_lo, i_hi + 1, dtype=np.int64)
            opp = k_max - idx
            # contribution of each pair (i, k_max-i)
            contrib = w[idx] * w[opp]
            i_star = idx[np.argmax(contrib)]              # index giving the largest product
            # pick a destination far from the interval so it does not feed k_max
            while True:
                j = rng.integers(0, n)
                if j < i_lo or j > i_hi:
                    break
            max_transfer = eps_frac * w[i_star]
            delta = rng.random() * max_transfer
            if delta <= 0.0:
                continue
            cand = w.copy()
            cand[i_star] -= delta
            cand[j] += delta

        # ----------------------------------------------------------
        # 3) Block perturbation (larger but still cheap jump)
        # ----------------------------------------------------------
        else:
            cand = block_perturbation(w, rng)

        # ----------------------------------------------------------
        # 4) Project back onto the probability simplex (simple scaling)
        # ----------------------------------------------------------
        cand = np.maximum(cand, 0.0)
        s = np.sum(cand)
        if s <= 0.0:
            continue
        cand /= s

        # ----------------------------------------------------------
        # 5) Evaluate the new candidate (fast FFT based)
        # ----------------------------------------------------------
        conv_cand = convolve_fft(cand, cand)
        conv_max_cand = np.max(conv_cand)
        score_cand = target_score(conv_max_cand, n)

        # ----------------------------------------------------------
        # 6) Accept / reject based on Metropolis criterion
        # ----------------------------------------------------------
        # Temperature decays gradually
        T = T0 * (T_decay ** it)

        improved = score_cand < best_score - 1e-12
        if improved:
            best_w = cand.copy()
            best_score = score_cand
            best_conv_max = conv_max_cand
            no_improve = 0
            print(f"[improve] it={it:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # Accept if better, otherwise with probability exp(-(Δ)/T)
        if improved or (T > 1e-12 and rng.random() < np.exp(-(score_cand - best_score) / T)):
            w = cand

        # ----------------------------------------------------------
        # 7) Occasionally enforce symmetry (helps the search landscape)
        # ----------------------------------------------------------
        if it % sym_interval == 0 and it > 0:
            w = 0.5 * (w + w[::-1])
            w = np.maximum(w, 0.0)
            w /= np.sum(w)

        # ----------------------------------------------------------
        # 8) Occasionally try a global block jump from the current best
        # ----------------------------------------------------------
        if it % block_interval == 0 and it > 0:
            cand = block_perturbation(best_w, rng)
            cand = np.maximum(cand, 0.0)
            cand /= np.sum(cand)
            conv_cand = convolve_fft(cand, cand)
            conv_max_cand = np.max(conv_cand)
            score_cand = target_score(conv_max_cand, n)
            if score_cand < best_score:
                best_w = cand.copy()
                best_score = score_cand
                best_conv_max = conv_max_cand
                w = cand.copy()
                no_improve = 0
                print(f"[block]  it={it:7d}, new best={best_score:.6f}")

        # ----------------------------------------------------------
        # 9) Restart if we have not improved for a long time
        # ----------------------------------------------------------
        if no_improve > restart_patience:
            w = block_perturbation(best_w, rng)
            w = np.maximum(w, 0.0)
            w /= np.sum(w)
            no_improve = 0
            print(f"[restart] it={it:7d} – jumping out of a plateau")

        it += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant; we adopt the
    # conventional √(2 n) L¹‑norm for readability.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Return a plain Python list, as required by the harness.
    return [float(x) for x in final_seq]
