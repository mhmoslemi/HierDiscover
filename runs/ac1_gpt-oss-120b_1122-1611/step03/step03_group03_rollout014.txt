"""
Improved geometric‑seed Adam optimizer for minimising max autocorrelation.
...
"""

import time
import numpy as np

def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c

def simplex_projection(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 300)
    hi = max(2, n // 20)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.20)))   # larger jumps than before
    cand[start:stop] *= factor
    # random small Gaussian noise everywhere
    cand += rng.normal(0.0, 0.005, size=n)
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= np.sum(cand)
    return cand

def geometric_sequence(n: int, epsilon: float) -> np.ndarray:
    # w_i ∝ r^i with r = 1 - epsilon, normalized to sum 1

def sine_sequence(n: int, power: float = 1.0) -> np.ndarray:

def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (non‑circular).  Small rounding errors
    are clipped to zero to stay within the feasible region."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # guard against -0 caused by rounding
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implements the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
