def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    best_seq = None
    best_eval = float('inf')
    # set candidate n values to try
    n_values = [8000, 12000, 16000]  # could be modified
    for n in n_values:
        if time.time() - start > budget_s - 5: break
        # search over alpha
        def shape_w(alpha):
            x = (np.arange(n) + 0.5) / n
            w = (x * (1 - x)) ** alpha
            total = np.sum(w)
            if total == 0:
                w = np.full(n, 1.0 / n)
            else:
                w = w / total
            return w
        # coarse
        alphas = np.linspace(0.0, 3.0, 31)
        best_alpha = None
        best_raw = float('inf')
        for a in alphas:
            if time.time() - start > budget_s - 5: break
            w = shape_w(a)
            b = convolve_fft(w,w)
            maxb = np.max(b)
            if maxb < best_raw:
                best_raw = maxb
                best_alpha = a
        if best_alpha is None: continue
        # refine using golden-section
        left = max(0.0, best_alpha - 0.2)
        right = best_alpha + 0.2
        phi = (1 + 5 ** 0.5) / 2
        for _ in range(20):
            if time.time() - start > budget_s - 5: break
            m1 = right - (right - left) / phi
            m2 = left + (right - left) / phi
            w1 = shape_w(m1)
            b1 = convolve_fft(w1,w1)
            f1 = np.max(b1)
            w2 = shape_w(m2)
            b2 = convolve_fft(w2,w2)
            f2 = np.max(b2)
            if f1 < f2:
                right = m2
                if f1 < best_raw:
                    best_raw = f1
                    best_alpha = m1
                # keep left
            else:
                left = m1
                if f2 < best_raw:
                    best_raw = f2
                    best_alpha = m2
        # Now have best_alpha for this n
        w = shape_w(best_alpha)
        # optional gradient descent fine-tune
        momentum = np.zeros_like(w)
        beta = 0.95
        eta0 = 0.5
        # we will use subgradient descent for max directly (mask)
        max_iter = 2000
        for it in range(max_iter):
            if time.time() - start > budget_s - 5: break
            b = convolve_fft(w,w)
            maxb = np.max(b)
            # compute mask of near-max indices
            eps = 1e-12
            # include all indices equal to max, but rarely multiple.
            mask = (b >= maxb - 1e-15).astype(float)  # binary mask
            # probability distribution of active constraints:
            if np.sum(mask) == 0:
                break
            p = mask / np.sum(mask)
            # gradient (subgradient) of max is 2 * conv(p, w_rev)
            grad = 2.0 * convolve_fft(p, w[::-1])[n-1:n-1+n]
            # momentum step
            eta = eta0 / np.sqrt(it + 1.0)
            momentum = beta * momentum - eta * grad
            w = w + momentum
            # project onto simplex
            w = simplex_projection(w)
            # enforce symmetry
            w = (w + w[::-1]) / 2.0
        # after fine-tune, compute final score
        seq = w * np.sqrt(2.0 * n)  # sum = sqrt(2n)
        eval_score = evaluate_sequence(seq.tolist())
        print(f"[n={n}] best_alpha={best_alpha:.6f}, raw_max={np.max(convolve_fft(w,w)):.6f}, eval={eval_score:.6f}")

"""
Propose a step‑function sequence that minimises the evaluate_sequence score.

The algorithm explores a one‑parameter family of symmetric “beta‑shaped”
distributions  a_i ∝ (x_i·(1−x_i))^α  (x_i = (i+0.5)/n).  For each candidate
length n a coarse grid on α is scanned, then a golden‑section refinement
searches the neighbourhood of the best α.  The best shape is further
polished by a projected sub‑gradient descent on the true L∞‑norm of the
self‑convolution (the active‑constraint mask provides a sub‑gradient).
Symmetry and the simplex constraint (∑a_i = 1) are enforced after every
update; finally the sequence is scaled to the conventional ℓ₁‑norm
√(2 n), which does not change the evaluation score.

Multiple lengths are tried (subject to the remaining time budget) and the
best candidate encountered is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # numerical noise may produce tiny negatives
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_shape(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric beta‑shape on the simplex:
        w_i ∝ (x_i·(1−x_i))^α,   x_i = (i+0.5)/n .
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x * (1.0 - x)) ** alpha
    s = np.sum(w)
    if s == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= s
    return w


def subgradient_step(w: np.ndarray, beta: float, eta0: float,
                     it: int) -> np.ndarray:
    """
    One projected sub‑gradient step for the L∞‑norm of self‑convolution.
    Uses the active‑set mask as a sub‑gradient.
    """
    n = w.size
    b = convolve_fft(w, w)
    maxb = np.max(b)

    # active set (all indices attaining the maximum, up to numerical tolerance)
    mask = (b >= maxb - 1e-15).astype(np.float64)
    if mask.sum() == 0:
        return w
    p = mask / mask.sum()

    # sub‑gradient of the max‑convolution (2·conv(p, w_rev))
    grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

    # simple diminishing step size + momentum
    eta = eta0 / np.sqrt(it + 1.0)
    w_new = w - eta * grad
    w_proj = simplex_projection(w_new)

    # enforce symmetry (the optimum is symmetric)
    w_sym = (w_proj + w_proj[::-1]) * 0.5
    return simplex_projection(w_sym)

# Main entry point required by the harness
# ----------------------------------------------------------------------
