def compute_score(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n - 2).bit_length())
    # Actually we want smallest power of two >= 2*n-1
    L = 1 << ((2*n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]
    max_conv = np.max(conv)
    return 2.0 * n * max_conv / (np.sum(a)**2)
```

"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in two stages:

1️⃣ Global parametric scan  
   – Try a handful of classic windows (Hann, Hamming, Blackman,
     triangular, Kaiser, Gaussian, exponential, power‑law, cosine‑power)
     for several lengths (128 … 4096).  
   – Random Dirichlet draws are also sampled.  
   – The best candidate from this cheap exhaustive search is kept.

2️⃣ Mirror‑descent refinement  
   – Starting from the best vector, a temperature‑controlled
     soft‑max approximation of the max‑norm is optimised with an
     exponentiated‑gradient (mirror‑descent) update on the probability
     simplex.  
   – The schedule of learning‑rate (η) and temperature (β) is linear,
     while the number of active convolution entries K grows from a
     small to a larger fraction of the support.  
   – Small random block perturbations and occasional pair‑wise mass
     transfers improve exploration.  
   – If the objective does not improve for a while, a soft restart is
     performed (jittered copy of the current best).

The whole procedure respects the 1000 s budget (a 5 s safety margin is
kept).  The returned list contains only non‑negative floats, is
normalised to have unit sum (the scaling does not affect the score) and
is clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for convolution and score computation
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of two length‑n vectors."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using FFT (real input). Returns an array of length 2*n‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int, sum_a: float = 1.0) -> float:
    """Exact objective value given a convolution array."""
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def compute_score(a: np.ndarray) -> float:
    """Convenient wrapper – normalises a to unit sum and returns the objective."""
    s = a.sum()
    if s <= 0.0:
        return float('inf')
    a_norm = a / s
    n = a_norm.size
    fft_len = _fft_len_for_n(n)
    conv = _conv_self_fft(a_norm, fft_len)
    return _score_from_conv(conv, n, sum_a=1.0)


# ----------------------------------------------------------------------
# Window families (all return non‑negative vectors)
# ----------------------------------------------------------------------
def shape_hann(n):
    return np.hanning(n)


def shape_hamming(n):
    return np.hamming(n)


def shape_blackman(n):
    return np.blackman(n)


def shape_triangular(n):
    i = np.arange(n, dtype=float)
    w = 1.0 - np.abs((i - (n - 1) / 2) / ((n - 1) / 2))
    return np.clip(w, 0.0, None)


def shape_kaiser(n, beta):
    return np.kaiser(n, beta)


def shape_gaussian(n, sigma):
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma_abs = sigma * (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma_abs) ** 2)
    return w


def shape_exponential(n, lam):
    i = np.arange(n, dtype=float)
    w = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return w


def shape_power(n, p):
    i = np.arange(n, dtype=float) + 1.0
    w = (i ** p) * ((n - i + 1) ** p)
    return w


def shape_cosine_power(n, p):
    i = np.arange(n, dtype=float)
    w = np.sin(np.pi * i / (n - 1)) ** p
    return w


# ----------------------------------------------------------------------
# Random perturbations used during refinement
# ----------------------------------------------------------------------
def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Mirror‑descent refinement (core of the optimiser)
# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Mirror‑descent on a soft‑max approximation of the max‑norm.
    Starts from ``a0`` (already normalised to sum = 1) and runs until ``deadline``.
    Returns the best vector found.
    """
    a = a0.copy()
    n = a.size
    fft_len = _fft_len_for_n(n)

    best = a.copy()
    best_score = compute_score(best)
    last_improve = time.monotonic()
    start = last_improve

    # schedules – values tuned for a wide range of n
    eta_start, eta_end = 0.9, 0.02
    beta_start, beta_end = 1.0, 250.0
    K_start, K_end = max(5, int(0.04 * n)), max(10, int(0.20 * n))

    block_perturb_prob = 0.04
    pair_move_prob = 0.07
    restart_gap = 25.0  # seconds without improvement → soft restart
    momentum = 0.85
    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:
        # ----- convolution and current score -----
        conv = _conv_self_fft(a, fft_len)
        cur_score = _score_from_conv(conv, n, sum_a=1.0)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] new best (refine) score = {best_score:.6f}")

        # ----- linear schedules -----
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        K_active = int(K_start * (1.0 - elapsed / total) + K_end * (elapsed / total))
        K_active = max(1, min(K_active, conv.size))

        # ----- soft‑max weighting of the K largest convolution entries -----
        if K_active == conv.size:
            active_idx = np.arange(conv.size)
        else:
            kth = conv.size - K_active
            active_idx = np.argpartition(conv, kth)[kth:]

        max_c = conv.max()
        grad = np.zeros_like(a)
        rev_a = a[::-1]                     # reversed view – used for the right part
        for k in active_idx:
            w = np.exp(beta * (conv[k] - max_c))  # 0 ≤ w ≤ 1
            if w == 0.0:
                continue
            if k < n:
                # left side: indices 0 … k
                grad[: k + 1] += w * a[k :: -1]   # a[k], a[k-1], … , a[0]
            else:
                # right side: indices k‑(n‑1) … 2n‑2‑k
                start = k - (n - 1)
                length = (2 * n - 1) - k
                grad[start:] += w * rev_a[:length]

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- mirror‑descent update on the simplex (exponentiated gradient) -----
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- optional pair‑wise mass transfer (cheap, no recompute of conv) -----
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # move a fraction of the donor's mass
                max_delta = min(a[i], 0.3 * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    a[i] -= delta
                    a[j] += delta
                    a = np.clip(a, 0.0, None)
                    a /= a.sum()

        # ----- soft restart when stuck -----
        if now - last_improve > restart_gap:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    return best


# ----------------------------------------------------------------------
# Global parametric scan (cheap exploration of many shapes)
# ----------------------------------------------------------------------
def _global_scan(deadline: float, rng: np.random.Generator):
    """Return a (vector, score) pair that is the best among a collection of parametric windows."""
    best_vec = None
    best_score = float('inf')

    # include the supplied sequence if it looks reasonable
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    cand /= cand.sum()
                    sc = compute_score(cand)
                    best_vec, best_score = cand, sc
                    print(f"Starting from height_sequence_1 – score {sc:.6f}")
        except Exception:
            pass

    # --- lengths to try -------------------------------------------------
    lengths = [128, 256, 512, 1024, 2048, 4096]

    # --- deterministic windows -----------------------------------------
    det_funcs = [shape_hann, shape_hamming, shape_blackman, shape_triangular]

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for f in det_funcs:
            w = f(n).astype(np.float64)
            if w.sum() == 0.0:
                continue
            a = w / w.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f"[scan] n={n} {f.__name__:12s} → score {sc:.6f}")

    # --- parametrised windows -------------------------------------------
    # Parameters are sampled uniformly in a sensible range.
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # Kaiser (beta ∈ [0, 12])
        for beta in rng.uniform(0.0, 12.0, size=5):
            w = shape_kaiser(n, beta).astype(np.float64)
            if w.sum() == 0.0:
                continue
            a = w / w.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f"[scan] n={n} kaiser(beta={beta:.2f}) → {sc:.6f}")

        # Gaussian (sigma ∈ [0.1, 0.5] relative to half‑width)
        for sigma in rng.uniform(0.1, 0.5, size=5):
            w = shape_gaussian(n, sigma).astype(np.float64)
            if w.sum() == 0.0:
                continue
            a = w / w.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f"[scan] n={n} gaussian(σ={sigma:.2f}) → {sc:.6f}")

        # Exponential decay (λ ∈ [0.5, 4])
        for lam in rng.uniform(0.5, 4.0, size=5):
            w = shape_exponential(n, lam).astype(np.float64)
            if w.sum() == 0.0:
                continue
            a = w / w.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f"[scan] n={n} exponential(λ={lam:.2f}) → {sc:.6f}")

        # Power law (p ∈ [0.5, 4])
        for p in rng.uniform(0.5, 4.0, size=5):
            w = shape_power(n, p).astype(np.float64)
            if w.sum() == 0.0:
                continue
            a = w / w.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f"[scan] n={n} power(p={p:.2f}) → {sc:.6f}")

        # Cosine‑power (p ∈ [0.5, 4])
        for p in rng.uniform(0.5, 4.0, size=5):
            w = shape_cosine_power(n, p).astype(np.float64)
            if w.sum() == 0.0:
                continue
            a = w / w.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f"[scan] n={n} cos_pow(p={p:.2f}) → {sc:.6f}")

    # --- random Dirichlet draws -----------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # concentration parameters 0.3, 1.0, 2.0 (sparser to smoother)
        for alpha in (0.3, 1.0, 2.0):
            for _ in range(8):
                a = rng.dirichlet(np.full(n, alpha))
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_vec = sc, a.copy()
                    print(f"[scan] n={n} Dirichlet(alpha={alpha}) → {sc:.6f}")

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # reserve a few seconds for the harness itself
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Global parametric scan – cheap exploration
    # ------------------------------------------------------------------
    best_vec, best_score = _global_scan(deadline, rng)
    if best_vec is None:
        # fallback: uniform distribution
        n = 256
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_score(best_vec)

    # ------------------------------------------------------------------
    # 2) Mirror‑descent refinement (uses remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        refined = _refine_sequence(best_vec, deadline, rng)
        refined_score = compute_score(refined)
        if refined_score < best_score:
            best_vec, best_score = refined, refined_score

    # ------------------------------------------------------------------
    # 3) Final clean‑up: trim near‑zero entries to keep the length minimal
    # ------------------------------------------------------------------
    eps = 1e-9
    mask = best_vec > eps
    if mask.sum() == 0:
        # extremely unlikely – fall back to uniform
        n = 256
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec[mask]
        best_vec = best_vec / best_vec.sum()  # re‑normalise to unit sum

    # ------------------------------------------------------------------
    # 4) safety clipping (evaluate_sequence caps values itself, we do it as well)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 5) Return plain Python list of floats
    # ------------------------------------------------------------------
    return [float(x) for x in best_vec]
