"""
Propose a step‑function sequence that minimizes the evaluation metric.

Algorithm summary
-----------------
* Work on the probability simplex (weights sum to 1, w ≥ 0).  Scaling to
  √(2 n) is done only at the very end because the score is homogeneous.
* Use a relatively large length (default 12 000) – longer sequences
  give the optimizer more freedom to flatten the self‑convolution.
* Initialise with a symmetric “edge‑heavy’’ exponential shape
  w[i] ≈ exp(−c·i)+exp(−c·(n‑1‑i)).
* Main optimisation loop:
    – Compute the self‑convolution b = w∗w.
    – Approximate the L∞‑norm by the smooth‑max μ·log∑exp(b/μ)
       (μ = μ_factor·max(b) and μ_factor is slowly decayed).
    – Gradient of the smooth‑max is 2·conv(softmax(b), w_rev).
    – Update w with the Adam optimiser, then project back onto the simplex.
    – Enforce symmetry (w ← (w+w[::-1])/2) after each projection.
    – Keep the best‑ever vector (lowest score) and occasionally try a
      random block perturbation of the best vector (global jump).
    – If no improvement for a long time, restart from a perturbed best.
* After the smooth‑max phase, perform a short sub‑gradient refinement
  that directly works on the active peak of the true L∞ norm.
* Finally, rescale the best vector to have ℓ₁‑norm √(2 n) and return it
  as a plain Python list.

The combination of a better initial shape, Adam optimisation, enforced
symmetry and a dedicated sub‑gradient cleanup yields a lower peak of the
self‑convolution than the previous baseline.
"""

import time
import math

# Helper functions – must be defined at top level for the harness.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # tiny negative rounding errors are removed
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    O(n) random block perturbation used as a jump operator.
    A contiguous block is multiplied by a log‑normal factor.
    The result is optionally symmetrised and clipped to the allowed range.
    """
    cand = base.copy()
    n = cand.size
    # block length – between ~0.5% and ~8% of the vector length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # log‑normal multiplicative factor
    cand[start:stop] *= factor
    # optional symmetric averaging (helps enforce the final symmetry)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
