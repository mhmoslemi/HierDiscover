"""
Propose a step‑function (non‑negative sequence) that minimizes the
`evaluate_sequence` objective.

The strategy is a multi‑stage heuristic:

1. Start from the Catalan (central binomial) distribution, which is a
   strong baseline.  Additional initial points are generated from power‑law
   families (i⁻ᶿ) with random exponents to diversify the search.

2. For a fixed length *n* we run a mirror‑descent optimisation on the
   simplex using an Adam update of the log‑weights.  The objective is
   smoothed by a temperature‑scaled log‑sum‑exp (soft‑max) of the
   self‑convolution entries; the gradient is computed by a cheap FFT‑based
   convolution.

3. Periodically we enforce symmetry, apply a cheap random block jump,
   perform a targeted “mass‑transfer” between two positions and restart
   from the current best when progress stalls.

4. When the allotted time permits we increase the length (interpolating
   the best distribution from the previous stage) and repeat the
   optimisation.

5. After the iterative phase a short grid search over power‑law
   exponents and a final mixing with the uniform distribution are used
   to squeeze out any remaining improvement.

The final sequence is scaled to the conventional √(2 n) L¹‑norm required
by `evaluate_sequence` before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` central binomial coefficients divided by 4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)  # clip tiny negatives
    return c


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max probabilities p_k ∝ exp(conv_k/γ).  Returns a vector that sums to 1."""
    shift = conv - np.max(conv)
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float) -> (np.ndarray, float):
    """
    Gradient of the soft‑max‑smoothed maximum of the self‑convolution.
    Returns (∇fγ(w),  max_b) where max_b = max_k (w*w)[k].
    """
    # self‑convolution
    b = convolve_fft(w, w)                     # length 2n‑1
    # soft‑max weights over the convolution entries
    p = _softmax_weights(b, gamma)             # length 2n‑1
    # convolution with the reversed weight vector gives Σ_k p_k·shift_k(w)
    grad = 2.0 * convolve_fft(w, p[::-1])[::-1][: w.size]
    return grad, np.max(b)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random block perturbation – used as a jump.
    A contiguous block of the vector is multiplied by a log‑normal factor.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _powerlaw_sequence(length: int, theta: float, beta: float) -> np.ndarray:
    """Sequence proportional to (i+β)^{-θ} (β>0)."""
    idx = np.arange(length, dtype=np.float64)
    seq = (idx + beta + 1e-12) ** (-theta)
    return seq

def _optimize_w(w0: np.ndarray,
                n: int,
                rng: np.random.Generator,
                time_budget: float) -> (np.ndarray, float):
    """
    Mirror‑descent optimisation (Adam on log‑weights) for a fixed length.
    Returns the best vector found and its 2·n·max(b) score.
    """
    start = time.monotonic()
    # initialise log‑probabilities
    v = np.log(w0 + 1e-12)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # hyper‑parameters (tuned experimentally)
    alpha = 0.12            # base learning rate for Adam
    gamma_factor = 0.04     # temperature = gamma_factor * max_b
    sym_interval = max(500, n // 10)
    block_interval = max(1500, n // 4)
    transfer_interval = max(1200, n // 6)
    no_improve_limit = max(8000, n * 2)

    best_w = w0.copy()
    best_score = 2.0 * n * np.max(convolve_fft(best_w, best_w))
    iteration = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # ---- convert to simplex (softmax) ----
        vmax = np.max(v)
        expv = np.exp(v - vmax)
        w = expv / np.sum(expv)

        # ---- evaluate ----
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b

        if score < best_score - 1e-13:
            best_score = score
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- smoothed gradient ----
        gamma = max(gamma_factor * max_b, 1e-12)
        grad_w, _ = _smoothed_gradient(w, gamma)

        # ---- gradient w.r.t. log‑weights (mirror‑descent) ----
        avg = np.dot(w, grad_w)
        grad_v = w * (grad_w - avg)

        # ---- Adam update on v ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)
        m_hat = m / (1.0 - beta1 ** iteration)
        v2_hat = v2 / (1.0 - beta2 ** iteration)
        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v -= step

        # ---- symmetry enforcement ----
        if iteration % sym_interval == 0:
            w_sym = 0.5 * (w + w[::-1])
            v = np.log(w_sym + 1e-12)
            m.fill(0.0)
            v2.fill(0.0)

        # ---- random block jump ----
        if iteration % block_interval == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                v = np.log(cand + 1e-12)
                m.fill(0.0)
                v2.fill(0.0)

        # ---- targeted mass transfer (move a small chunk from a left‑hand
        #      heavy index to a right‑hand lighter index) ----
        if iteration % transfer_interval == 0:
            # pick a source index among the first 15 % of the vector
            src = rng.integers(0, max(1, n // 7))
            # pick a destination anywhere else
            dst = rng.integers(0, n)
            if src != dst and w[src] > 1e-12:
                delta = w[src] * rng.uniform(0.01, 0.15)   # move 1‑15 % of src
                w_new = w.copy()
                w_new[src] -= delta
                w_new[dst] += delta
                score_new = 2.0 * n * np.max(convolve_fft(w_new, w_new))
                if score_new < best_score:
                    best_w = w_new.copy()
                    best_score = score_new
                    v = np.log(w_new + 1e-12)
                    m.fill(0.0)
                    v2.fill(0.0)
                    no_improve = 0

        # ---- stagnation handling ----
        if no_improve > no_improve_limit:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            v = np.log(w + 1e-12)
            m.fill(0.0)
            v2.fill(0.0)
            no_improve = 0

    return best_w, best_score

    # ------------------------------------------------------------------
    if overall_best_w is None:

        overall_best_n = 2000

    return [float(x) for x in final_seq]

`evaluate_sequence` objective.

Strategy
--------
1.  Start from the Catalan (central‑binomial) distribution, a strong baseline,

    execute a targeted *mass‑transfer* between two positions, and restart
    from the current best when progress stalls.

4.  When time permits increase the length (interpolating the best distribution
    from the previous stage) and repeat the optimisation.

5.  After the iterative phase run a short grid search over power‑law exponents

            if rng.random() < 0.07:
                w_opt, sc_opt = _optimize_w(w0, n, rng, sub_budget)
                if sc_opt < stage_best_score:
                    stage_best_score = sc_opt
                    stage_best_w = w_opt.copy()
                    print(f"[stage‑cat] n={n} new best={stage_best_score:.6f}")

        # ----- post‑processing: uniform mixing and a fine power‑law grid -----
        if stage_best_w is not None:
            uniform = np.full(n, 1.0 / n)

            # mix with uniform (small amounts)
            for alpha_mix in np.linspace(0.0, 0.36, 13):
                mixed = (1.0 - alpha_mix) * stage_best_w + alpha_mix * uniform
                mixed = mixed / np.sum(mixed)
                sc = 2.0 * n * np.max(convolve_fft(mixed, mixed))
                if sc < stage_best_score:
                    stage_best_score = sc
                    stage_best_w = mixed.copy()

            # fine grid over power‑law exponent close to 0.5
            for theta in np.linspace(0.45, 0.55, 11):
                w_cand = _powerlaw_sequence(n, theta, 0.0)
                w_cand = w_cand / np.sum(w_cand)
                sc = 2.0 * n * np.max(convolve_fft(w_cand, w_cand))
                if sc < stage_best_score:
                    stage_best_score = sc
                    stage_best_w = w_cand.copy()

        # ----- record the global best across stages -----
        if stage_best_score < overall_best_score:
            overall_best_score = stage_best_score
            overall_best_w = stage_best_w.copy()
            overall_best_n = n
            print(f"[global] new overall best at n={n}: {overall_best_score:.6f}")

        prev_w = stage_best_w   # seed the next resolution stage

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    # ------------------------------------------------------------------
    if overall_best_w is None:
        overall_best_n = 2000
        overall_best_w = catalan_sequence(overall_best_n)
        overall_best_w = overall_best_w / np.sum(overall_best_w)

    # ------------------------------------------------------------------
    # Final scaling to √(2 n) L¹‑norm required by `evaluate_sequence`
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * overall_best_n)
    final_seq = overall_best_w * target_sum

    # sanity check (silently ignored if the evaluator is unavailable)
    try:
        final_eval = evaluate_sequence(final_seq.tolist())
        print(f"[final] n={overall_best_n}, evaluate={final_eval:.6f}")
    except Exception:
        pass

    # the harness expects a plain Python list
    return [float(x) for x in final_seq]
