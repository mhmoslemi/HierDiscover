"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1️⃣  **Diverse seeds** – several analytically‑motivated shapes are built:
    * Catalan numbers (classic baseline)
    * Arcsine law (Beta(0.5,0.5))
    * Symmetric Beta(a,a) for a grid of a‑values
    * A few random Dirichlet draws
    * The previous answer (`height_sequence_1`) if present.

   All seeds are normalised to a probability vector (sum = 1) and scored
   with `2·n·max(conv)` (identical to `evaluate_sequence` after the final
   √(2·n) scaling).

2️⃣  **Local refinement** – for the best few seeds a projected sub‑gradient
   descent with momentum (Nesterov‑style) is executed.  The sub‑gradient of
   the active convolution entry `max(conv)` is the simple vector
   `grad[k] = 2·w[i_max‑k]` (zero outside the support).  After each step the
   iterate is projected onto the probability simplex by the Duchi et al.
   (2008) algorithm.  Occasionally a “mirror‑descent” KL‑step,
   a block‑perturbation jump and a forced symmetry operation are applied
   to avoid stagnation.

3️⃣  **Polishing** – a short simulated‑annealing phase moves tiny mass
   packets between random positions while accepting uphill moves with a
   slowly decreasing temperature.

The whole routine respects the 1000 s budget (a 5‑s safety margin) and
returns the best sequence found, scaled to the required ℓ¹‑norm
`√(2·n)`.  The quality of the result is printed and also checked with the
provided `evaluate_sequence` function.
"""

import time
import numpy as np

def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def beta_shape(length: int, a: float) -> np.ndarray:
    """
    Symmetric Beta(a,a) density on a discrete grid.
    Returns a non‑negative vector of length ``length`` whose sum is 1.
    """
    # centre of each bin (avoid zero)
    i = np.arange(length, dtype=np.float64) + 0.5
    # scale to [0,1]
    x = i / length
    # density ∝ x^{a‑1} (1‑x)^{a‑1}
    w = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    # enforce symmetry explicitly (tiny numerical asymmetries)
    w = 0.5 * (w + w[::-1])
    w_sum = w.sum()
    if w_sum == 0:
        w[:] = 1.0 / length
    else:
        w /= w_sum
    return w


def arcsine_shape(length: int) -> np.ndarray:
    """Arcsine law – the Beta(0.5,0.5) case."""
    return beta_shape(length, 0.5)


def random_dirichlet(length: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a Dirichlet‑distributed probability vector."""
    vec = rng.gamma(alpha, 1.0, size=length)
    total = vec.sum()
    if total == 0:
        vec[:] = 1.0 / length
    else:
        vec /= total
    return vec


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n≈10 k)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Numerical noise may produce tiny negatives – clamp to zero.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """A cheap O(n) random jump used as a large perturbation."""
    cand = base.copy()
    n = cand.size
    # pick a block whose width scales with n
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation to keep the distribution roughly even
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def subgradient_descent(
    w0: np.ndarray,
    n: int,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum and occasional KL‑step.
    Returns the best vector found (sum = 1) and its score 2·n·max(conv).
    """
    w = w0.copy()
    w_prev = w.copy()
    best_w = w.copy()
    # initial score
    conv = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(conv)

    eta0 = 0.8
    momentum = 0.9
    iteration = 0
    no_improve = 0
    start = time.monotonic()

    while time.monotonic() - start < time_limit:
        # --- current convolution & active index ---------------------------------
        conv = convolve_fft(w, w)
        max_val = np.max(conv)
        # pick a random index among the (almost) active set
        idxs = np.flatnonzero(conv >= max_val - 1e-15)
        i_max = int(rng.choice(idxs))

        # --- sub‑gradient of the active quadratic form -------------------------
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        # grad[j] = 2 * w[i_max - j]
        grad[j] = 2.0 * w[i_max - j]

        # --- step size ---------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1)

        # --- momentum + Euclidean projection -----------------------------------
        w_temp = w - eta * grad + momentum * (w - w_prev)
        w_new = simplex_projection(w_temp)

        # --- occasional KL‑mirror step (chosen with prob 0.5) -------------------
        if rng.random() < 0.5:
            factor = np.exp(-eta * grad)
            w_mirror = w * factor
            w_mirror_sum = w_mirror.sum()
            if w_mirror_sum > 0:
                w_mirror /= w_mirror_sum
                # evaluate which variant is better
                max_new = np.max(convolve_fft(w_new, w_new))
                max_mirror = np.max(convolve_fft(w_mirror, w_mirror))
                if max_mirror < max_new:
                    w_new = w_mirror

        w_prev = w
        w = w_new

        # --- evaluate the new point --------------------------------------------
        conv = convolve_fft(w, w)
        cur_score = 2.0 * n * np.max(conv)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1

        # --- deterministic symmetry enforcement every 2000 iter -----------------
        if iteration % 2000 == 0 and iteration > 0:
            sym = 0.5 * (w + w[::-1])
            sym /= sym.sum()
            # keep symmetry only if it improves the score
            sym_score = 2.0 * n * np.max(convolve_fft(sym, sym))
            if sym_score < best_score:
                w = sym
                w_prev = w.copy()
                best_w = sym.copy()
                best_score = sym_score
                no_improve = 0

        # --- large random block‑perturbation every 4000 iter --------------------
        if iteration % 4000 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                w = cand
                w_prev = cand.copy()
                best_w = cand.copy()
                best_score = cand_score
                no_improve = 0

        # --- reset if stuck for a very long time --------------------------------
        if no_improve > 150_000:
            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            w_prev = w.copy()
            no_improve = 0

        iteration += 1

    return best_w, best_score


def annealing_polish(
    w0: np.ndarray,
    n: int,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing: move tiny mass packets,
    accept uphill moves with a slowly cooling temperature.
    """
    cur_w = w0.copy()
    cur_score = 2.0 * n * np.max(convolve_fft(cur_w, cur_w))
    best_w = cur_w.copy()
    best_score = cur_score
    temperature = 0.02
    start = time.monotonic()

    while time.monotonic() - start < time_limit:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        max_transfer = min(cur_w[i], 0.02)
        delta = rng.uniform(0.0, max_transfer)
        if delta == 0.0:
            continue
        cand = cur_w.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / temperature):
            cur_w = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_w = cur_w.copy()
                best_score = cur_score
        temperature *= 0.9999
        if temperature < 1e-5:
            temperature = 0.001
    return best_w, best_score

    # Evaluate all seeds, keep the best few.
    # ------------------------------------------------------------------
    scored = []
    for idx, seed_vec in enumerate(seeds):
        conv = convolve_fft(seed_vec, seed_vec)
        score = 2.0 * n * np.max(conv)
        scored.append((score, seed_vec))
        print(f"[seed {idx:02d}] score = {score:.6f}")

    scored.sort(key=lambda x: x[0])
    # keep up to 5 best seeds for local optimisation
    top_k = min(5, len(scored))
    best_overall_w = scored[0][1].copy()
    best_overall_score = scored[0][0]
    print(f"[init] best seed score = {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    safety_margin = 5.0                     # seconds reserved for final scaling
    max_time = max(0.0, float(budget_s) - safety_margin)
    time_per_seed = max_time / (top_k + 1)   # rough split

    # ------------------------------------------------------------------
    # Local refinement for each top seed.
    # ------------------------------------------------------------------
    for rank in range(top_k):
        seed_score, seed_vec = scored[rank]
        remaining = max_time - (time.monotonic() - start_time)
        if remaining < safety_margin:
            break
        run_time = min(time_per_seed, remaining - safety_margin)
        print(f"[refine] start rank {rank} (seed score={seed_score:.6f})")
        refined_w, refined_score = subgradient_descent(
            seed_vec, n, rng, run_time
        )
        print(f"[refine] rank {rank} finished – refined score = {refined_score:.6f}")

        if refined_score < best_overall_score:
            best_overall_score = refined_score
            best_overall_w = refined_w.copy()
            print(f"[update] new global best = {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final annealing polish with whatever time is left.
    # ------------------------------------------------------------------
    remaining = max_time - (time.monotonic() - start_time)
    if remaining > 8.0:
        print("[anneal] starting polishing phase")
        polished_w, polished_score = annealing_polish(
            best_overall_w, n, rng, remaining - 3.0
        )
        if polished_score < best_overall_score:
            best_overall_score = polished_score
            best_overall_w = polished_w.copy()
        print(f"[anneal] polishing done – best score = {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Scale to the required ℓ¹‑norm √(2·n) and return.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_overall_w * target_sum / best_overall_w.sum()

    # sanity check using the provided evaluate_sequence (already imported)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best internal score={best_overall_score:.6f}, "
          f"evaluate_sequence score={final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
