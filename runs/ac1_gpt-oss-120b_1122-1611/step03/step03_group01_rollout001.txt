"""
Propose a near‑optimal step‑function for the self‑convolution bound.

Algorithm
---------
1. Choose a (fairly large) length `n`.  If a previous construction
   (`height_sequence_1`) exists we use the larger of its length and a
   safe default (≈40 000).  Larger `n` moves the discrete optimum closer
   to the continuous limit (≈1.5) and gives us room to beat the target.

2. Build several *beta‑shaped* initial candidates:
      p_i(γ) ∝ (x_i·(1−x_i))^{γ−1} ,  x_i = (i+½)/n ,   γ∈(0,1).
   The classic arcsine law corresponds to γ=½.  We sweep a modest grid
   of γ values, evaluate the bound `score = 2·n·max(p★p)`, and keep the
   best.

3. Refine the best candidate with a projected (Nesterov) sub‑gradient
   descent on the probability simplex.  At each iteration we:
      * evaluate the convolution,
      * pick a random index where the maximum is attained,
      * compute the sub‑gradient of that active quadratic form,
      * take a momentum‑augmented step,
      * project back onto the simplex,
      * enforce symmetry (w ← (w+wᵀ)/2) – the optimum is known to be
        symmetric,
      * occasionally apply a large random block perturbation to escape
        plateaus.

4. When the time budget is almost exhausted we run a short simulated‑
   annealing polish that moves tiny mass chunks and accepts uphill
   moves with a slowly cooling temperature.

5. Finally we rescale the probability vector to the required
   ℓ¹‑norm `√(2·n)` and return it as a plain Python list.  The
   `evaluate_sequence` function will then return the (scale‑invariant)
   bound, which we try to push below the target 1.5025.

The whole procedure runs completely inside the given time budget,
uses only NumPy (and the provided `evaluate_sequence`) and avoids any
file or network I/O.

import time

def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.  Returns a real array of length
    len(a)+len(b)-1.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative values caused by round‑off
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σx = 1}.  Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_distribution(n: int, gamma: float) -> np.ndarray:
    """
    Return a discretised symmetric Beta(γ,γ) density on n points,
    normalised to sum to 1.  For γ=1 we obtain the uniform distribution,
    for γ=½ the classic arcsine law.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    # power may be negative → large values at the ends; clip to avoid overflow
    shape = np.power(x * (1.0 - x), gamma - 1.0)
    shape = np.clip(shape, 0.0, 1e6)
    total = shape.sum()
    if total == 0:
        # fallback to uniform
        shape = np.full(n, 1.0 / n)
    else:
        shape /= total
    return shape

        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= cand.sum()
    return cand

    """
    n = w.size

        cur_score = compute_score(w)

        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:6d} → score = {best_score:.6f}")
        else:
            no_improve += 1

        # --- large random block jump every few thousand iterations ------
        if iteration % 3000 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                w_prev = w.copy()
                print(f"[block] iter={iteration:6d} → new best = {best_score:.6f}")

        # --- reset if stagnated for a long time -----------------------
        if no_improve > 100_000:
            w = block_perturbation(best_w, rng)
            w_prev = w.copy()
            no_improve = 0
            print(f"[reset] iter={iteration:6d} – resetting from best.")

        iteration += 1

    # --------------------------------------------------------------
    # Simulated‑annealing polishing phase (uses remaining time).
    # --------------------------------------------------------------
    time_left = budget_s - (time.monotonic() - start_time)
    if time_left > 8.0:
        temperature = 0.02
        cur_w = best_w.copy()
        cur_score = best_score
        anneal_start = time.monotonic()
        while time.monotonic() - anneal_start < time_left - 3.0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = rng.uniform(0.0, min(cur_w[i], 0.02))
            if delta == 0.0:
                continue
            cand = cur_w.copy()
            cand[i] -= delta
            cand[j] += delta
            cand_score = compute_score(cand)
            if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / temperature):
                cur_w = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_w = cur_w.copy()
                    best_score = cur_score
            temperature *= 0.9999
            if temperature < 1e-6:
                temperature = 0.001
        print(f"[anneal] polishing finished – best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final rescaling – the evaluation is invariant to ℓ¹‑scaling.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum(best_w) should be 1
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length = {n}, pre‑scale best = {best_score:.6f}, "
          f"after scaling evaluate = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
