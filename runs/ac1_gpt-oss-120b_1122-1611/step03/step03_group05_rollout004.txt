"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises the
self‑convolution peak (hence the evaluation score).  

The algorithm consists of three stages:

1️⃣ **Parameterised seeding** – generate many candidate vectors from a
   simple two‑parameter family  
   `w_i ∝ λ·1 + (1-λ)·((i+1)*(n-i))**α`  (α∈[-2,0], λ∈[0,1]).
   The family includes the uniform distribution (α=0, λ=1) and a
   U‑shaped “arcsine” distribution (α<0).  All candidates are normalised
   to a probability vector (∑w=1) and their peak convolution is measured
   with an FFT.  The best seed is kept.

2️⃣ **Projected sub‑gradient descent** – starting from the best seed we
   minimise the convex function  
   `f(w)=max_k (w*w)[k]`  with the constraint `w_i≥0, Σw_i=1`.  
   If `k*` is a maximising index, the sub‑gradient is
   `∂f/∂w_j = 2·w_{k*‑j}` (zero when the index is out of range).  When
   several indices are tied we average their gradients.  The update is
   performed with a decreasing step‑size `η_t = η₀/√t` and a Euclidean
   projection onto the probability simplex (Duchi et al., 2008).

3️⃣ **Escapes & polishing** – every few thousand iterations a cheap
   block‑perturbation (multiply a random contiguous block by a factor
   ≈ 1±12 %) is tried and kept if it improves the best value.
   The algorithm stops a few seconds before the overall budget so that
   the final sequence can be rescaled to the conventional ℓ¹‑norm
   `√(2n)` required by the evaluator.

The whole search runs under the time budget (`budget_s` seconds) and
returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx = 1}.
    Duchi et al. (2008) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical fallback – uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """
    Linear self‑convolution of a real vector w using an FFT of length N.
    Returns the full (2*len(w)-1) result.
    """
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    # Numerical noise may produce tiny negative values – clamp to zero.
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def correlation_gradient(w: np.ndarray, s: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the linear functional   g(w) = Σ_k s_k (w*w)[k] .
    Formula:  ∂g/∂w = 2 * (w ⋆ s)   where ⋆ denotes (linear) correlation.
    The implementation uses FFTs for O(N log N) speed.
    """
    # Correlation can be obtained by multiplying the FFT of w with the
    # conjugate of the FFT of s.
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(s, N)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = s.size
    offset = m - 1
    grad = 2.0 * corr_full[offset : offset + w.size]
    return grad


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation used as an escape move.
    A random contiguous block is multiplied by a log‑normally distributed
    factor (≈ ±12 %).  With probability 0.35 the result is symmetrised
    (averaged with its reversal).  The result is clipped to the allowed
    value range [0, 1000].
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈ ±12 %
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def generate_u_shape(alpha: float, mix: float, n: int, ext: np.ndarray) -> np.ndarray:
    """
    Return a normalised non‑negative vector of length n belonging to the
    two‑parameter family described in the doc‑string.
    * ext[i] = (i+1)*(n-i)  (i = 0 … n‑1) is pre‑computed once.
    * alpha  – exponent controlling the U‑shape (α∈[-2,0]).
    * mix    – weight of the uniform component (λ∈[0,1]).
    The result is already scaled to Σw = 1.
    """
    # Base shape (U‑shaped when alpha < 0, flat when alpha = 0)
    base = ext.astype(np.float64) ** alpha
    # Mix with a uniform component
    w = mix * np.ones(n, dtype=np.float64) + (1.0 - mix) * base
    # Enforce non‑negativity (tiny negative values may arise from rounding)
    np.maximum(w, 0.0, out=w)
    total = w.sum()
    if total <= 0.0:
        # fall back to uniform if something went wrong
        w.fill(1.0 / n)
    else:
        w /= total
    return w


# ----------------------------------------------------------------------
# Main search function required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) with a small
    self‑convolution peak.  The function respects the given time budget
    and returns a list of floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose the problem size.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=np.float64)
            if init_seq.ndim != 1 or init_seq.size < 2:
                raise ValueError
            n = int(init_seq.size)
            w0 = np.maximum(init_seq, 0.0)
            s = w0.sum()
            if s < 1e-12:
                raise ValueError
            w0 = w0 / s
        except Exception:
            # fallback to a moderate default size
            n = 5000
            w0 = None
    else:
        n = 5000
        w0 = None

    # Pre‑computed FFT lengths
    N_self = _next_pow2(2 * n - 1)      # for self‑convolution
    N_corr = _next_pow2(3 * n - 2)      # for correlation (gradient)

    # --------------------------------------------------------------
    # Pre‑compute the “extent” factor used in the two‑parameter family.
    # --------------------------------------------------------------
    i = np.arange(1, n + 1, dtype=np.float64)            # 1 … n
    ext = i * (n + 1 - i)                               # (i)*(n+1-i)

    # --------------------------------------------------------------
    # Helper to evaluate a probability vector (Σw = 1).
    # Returns the raw peak of the linear self‑convolution.
    # --------------------------------------------------------------
    def peak_of_self_conv(w: np.ndarray) -> float:
        b = self_convolve_fft(w, N_self)
        return float(b.max())

    # --------------------------------------------------------------
    # 1️⃣  Parameterised seeding – random search over (α, λ).
    # --------------------------------------------------------------
    start_time = time.monotonic()
    best_w = None
    best_score = np.inf
    best_peak = np.inf

    # If we have a valid initial vector, use it as a seed.
    if w0 is not None:
        best_w = w0.copy()
        best_peak = peak_of_self_conv(best_w)
        best_score = 2.0 * n * best_peak
        print(f"[seed] using provided sequence, n={n}, initial score={best_score:.6f}")

    # Number of random seeds – we cap it so that we keep enough time for
    # the sub‑gradient phase (roughly 1/3 of the total budget).
    max_seed_iters = 4000
    seed_iters = 0
    while (time.monotonic() - start_time < budget_s * 0.35) and (seed_iters < max_seed_iters):
        # Randomly draw parameters.
        alpha = rng.uniform(-2.0, 0.0)          # U‑shape exponent
        mix = rng.uniform(0.0, 1.0)            # uniform mix weight
        w = generate_u_shape(alpha, mix, n, ext)
        cur_peak = peak_of_self_conv(w)
        cur_score = 2.0 * n * cur_peak
        if cur_score < best_score:
            best_score = cur_score
            best_w = w.copy()
            best_peak = cur_peak
            print(f"[seed] iter={seed_iters:5d}, α={alpha:.3f}, λ={mix:.3f}, "
                  f"score={best_score:.6f}")
        seed_iters += 1

    # --------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient descent starting from the best seed.
    # --------------------------------------------------------------
    if best_w is None:
        # Should never happen, but guard against it.
        best_w = np.full(n, 1.0 / n, dtype=np.float64)

    w = best_w.copy()
    # Sub‑gradient step‑size schedule.
    eta0 = 0.5
    iteration = 0
    no_improve = 0
    last_best_iter = 0

    # Main optimisation loop (the remaining budget, leaving a small margin).
    while time.monotonic() - start_time < max(0.0, budget_s - 5.0):
        # 2.1 Evaluate current vector.
        b = self_convolve_fft(w, N_self)
        cur_peak = float(b.max())
        cur_score = 2.0 * n * cur_peak

        # Update best solution.
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            best_peak = cur_peak
            no_improve = 0
            last_best_iter = iteration
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # 2.2 Identify all indices attaining the maximal convoluted value
        # (within a tiny tolerance to avoid missing active constraints).
        tol = max(1e-12, cur_peak * 1e-12)
        active = np.where(np.abs(b - cur_peak) <= tol)[0]
        # Build the sub‑gradient weight vector (average over all active indices).
        s = np.zeros_like(b, dtype=np.float64)
        if active.size:
            s[active] = 1.0 / float(active.size)

        # 2.3 Gradient of the (nonsmooth) max – use correlation with s.
        grad = correlation_gradient(w, s, N_corr)

        # 2.4 Sub‑gradient step with diminishing step size.
        iteration += 1
        eta = eta0 / np.sqrt(iteration)
        w_new = w - eta * grad
        w = simplex_projection(w_new)

        # 2.5 Occasional random escape move.
        if iteration % 5000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
                cand_peak = peak_of_self_conv(cand)
                cand_score = 2.0 * n * cand_peak
                if cand_score < best_score:
                    best_score = cand_score
                    best_w = cand.copy()
                    best_peak = cand_peak
                    w = cand.copy()
                    print(f"[block] iter={iteration:6d}, improved score={best_score:.6f}")

        # 2.6 Reset if we have been stagnant for a long time.
        if no_improve > 25000:
            w = block_perturbation(best_w, rng)
            w = np.maximum(w, 0.0)
            ssum = w.sum()
            if ssum > 0:
                w /= ssum
            # Reset the adaptive moments – not needed for sub‑gradient, but we
            # clear the momentum of the Adam scheme used in the previous code.
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

    # --------------------------------------------------------------
    # 3️⃣  Final scaling to the conventional ℓ¹‑norm √(2 n).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()   # best_w already sums to 1.

    # Optional sanity check (does not affect the result if evaluate_sequence
    # is unavailable in the execution environment).
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best score={best_score:.6f}, "
              f"after scaling evaluate={final_score:.6f}")
    except Exception:
        pass

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
