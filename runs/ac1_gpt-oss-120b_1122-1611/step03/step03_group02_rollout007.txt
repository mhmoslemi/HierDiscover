def subgradient_optimize(w0, n, time_limit, rng, max_iter=100000, step0=0.6, beta=0.0):
    w = w0.copy()
    best_w = w.copy()
    best_score = compute_score(w, n)
    iter = 0
    no_improve = 0
    start = time.time()
    while (time.time() - start) < time_limit and iter < max_iter:
        # compute conv
        c = convolve_fft(w,w)
        maxc = c.max()
        score = 2.0 * n * maxc
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Identify index(s) with max
        i_max_candidates = np.flatnonzero(c >= maxc - 1e-14)  # tolerance
        i_max = int(rng.choice(i_max_candidates))

        # gradient
        g = np.zeros(n)
        start_idx = max(0, i_max - (n-1))
        end_idx = min(n-1, i_max)
        # Indices j: start_idx...end_idx
        idx = np.arange(start_idx, end_idx+1)
        g[idx] = 2.0 * w[i_max - idx]

        # step size
        step = step0 / np.sqrt(iter + 1.0)
        # update
        w_temp = w - step * g
        # project onto simplex
        w = simplex_projection(w_temp)

        # occasional random perturbation if stuck
        if no_improve > 5000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0

        iter += 1
        # optionally break early if time low
        if time.time() - start > time_limit - 0.1:
            break
    return best_w, best_score

"""
Multi‑start local search for the step‑function minimising
 2·n·max(w*w)   (with ∑w = 1, w ≥ 0).

The pipeline:

1. Choose a length n (reuse an existing construction if available,
   otherwise n≈8000).
2. Build a small library of parametric start‑up shapes
   (uniform, triangular, Hann, Blackman, Gaussian, exponential,
   central‑binomial, random Dirichlet).
3. For each start:
     – evaluate its score,
     – perform a few hundred iterations of sub‑gradient descent on the
       L∞‑objective (reduce the active peak),
     – retain the best view.
4. Take the overall best vector and run a longer smooth‑max + momentum
   projected gradient descent (the same idea used in the reference code,
   but with a more aggressive schedule and occasional random jumps).
5. Return the best vector, rescaled to the conventional √(2 n) ℓ₁‑norm.
   The returned list satisfies the security checks of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n up to ~2·10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # numerical noise can produce tiny negative values – clip them
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation – used as a jump operator."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetric averaging (helps regularity)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def compute_score(w: np.ndarray, n: int) -> float:
    """Score = 2·n·max(w*w) for a probability vector w (∑w = 1)."""
    c = convolve_fft(w, w)
    return float(2.0 * n * np.max(c))


# ----------------------------------------------------------------------
# Shape factories – all return a probability vector (sum = 1, non‑negative)
# ----------------------------------------------------------------------


def shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n)
    w = np.minimum(i + 1, n - i)
    return w / w.sum()


def shape_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def shape_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return w / w.sum()


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    ctr = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - ctr) / sigma) ** 2)
    return w / w.sum()


def shape_exponential(n: int, tau: float) -> np.ndarray:
    i = np.arange(n)
    ctr = (n - 1) / 2.0
    w = np.exp(-np.abs(i - ctr) / tau)
    return w / w.sum()


def shape_central_binomial(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return a / a.sum()


def shape_random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    sample = rng.gamma(alpha, 1.0, size=n)
    if sample.sum() == 0:
        sample = np.full(n, 1.0 / n)
    else:
        sample = sample / sample.sum()
    return sample


# ----------------------------------------------------------------------
# Local optimisation primitives
# ----------------------------------------------------------------------


def subgradient_optimize(
    w0: np.ndarray,
    n: int,
    rng: np.random.Generator,
    time_limit: float,
    step0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the L∞‑objective."""
    w = w0.copy()
    best_w = w.copy()
    best_score = compute_score(w, n)

    start = time.time()
    it = 0
    no_improve = 0

    while time.time() - start < time_limit:
        # current convolution and score
        c = convolve_fft(w, w)
        maxc = np.max(c)
        score = 2.0 * n * maxc

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1

        # pick a (random) active index
        active = np.flatnonzero(c >= maxc - 1e-14)
        i_max = int(rng.choice(active))

        # sub‑gradient of the active peak
        g = np.zeros_like(w)
        start_idx = max(0, i_max - (n - 1))
        end_idx = min(n - 1, i_max)
        idx = np.arange(start_idx, end_idx + 1, dtype=int)
        # for each j in idx we have contribution 2·w[i_max‑j]
        g[idx] = 2.0 * w[i_max - idx]

        # step‑size schedule
        step = step0 / np.sqrt(it + 1.0)
        w_candidate = w - step * g
        w = simplex_projection(w_candidate)

        # occasional random jump if we are stuck
        if no_improve > 5000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0

        it += 1
        # safety break if we are about to exceed the time budget
        if time.time() - start > time_limit - 0.05:
            break

    return best_w, best_score


def smooth_max_optimize(
    w0: np.ndarray,
    n: int,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """Smooth‑max projected gradient descent (momentum + occasional jumps)."""
    w = w0.copy()
    momentum = np.zeros_like(w)

    best_w = w.copy()
    best_score = compute_score(w, n)

    # hyper‑parameters (tuned for speed vs. quality)
    eta0 = 0.55          # base learning‑rate
    beta = 0.9           # momentum
    mu_factor = 0.7      # start with μ = mu_factor·max(c)
    mu_factor_min = 0.03
    mu_decay = 0.9998    # exponential decay each iteration
    block_interval = max(500, n // 30)
    reset_thresh = 15000
    no_improve = 0

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        c = convolve_fft(w, w)
        maxc = np.max(c)
        score = 2.0 * n * maxc

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1

        # smooth‑max weights (softmax over the convolution entries)
        mu = maxc * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        shifted = (c - maxc) / mu
        exp_term = np.exp(shifted)               # max(c) → 1, others ≤ 1
        p = exp_term / np.sum(exp_term)

        # gradient of the smooth‑max proxy
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1 : n - 1 + n]

        # momentum update + projection onto simplex
        eta = eta0 / np.sqrt(it + 1.0)
        momentum = beta * momentum - eta * grad
        w_candidate = w + momentum
        w = simplex_projection(w_candidate)

        # decay the smoothing factor
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        # cheap random “jump” every block_interval steps
        if it % block_interval == 0 and it > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = compute_score(cand, n)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                momentum = np.zeros_like(w)
                no_improve = 0

        # restart from a perturbed best if we have stagnated
        if no_improve > reset_thresh:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            momentum = np.zeros_like(w)
            no_improve = 0

        it += 1
        if time.time() - start > time_limit - 0.05:
            break

    return best_w, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score aims to be smaller than the previous best
    (~1.51786).  The procedure builds several parametric starts,
    refines them with cheap sub‑gradient steps, then applies a
    longer smooth‑max gradient descent.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Length selection – reuse a previous construction if available,
    # otherwise fall back to a moderate size.
    # ------------------------------------------------------------------
    global height_sequence_1
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, (list, tuple)):
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.ndim == 1 and init.size >= 2:
                n = init.size
                # normalise to a probability vector
                w_init = np.maximum(init, 0.0)
                s = w_init.sum()
                if s > 0:
                    w_init = w_init / s
                else:
                    w_init = None
            else:
                w_init = None
                n = 8000
        except Exception:
            w_init = None
            n = 8000
    else:
        w_init = None
        n = 8000

    # ------------------------------------------------------------------
    # Build a library of start‑up vectors (all are probability vectors)
    # ------------------------------------------------------------------
    starts = []

    if w_init is not None:
        starts.append(("prev", w_init))

    # deterministic families
    starts.append(("uniform", shape_uniform(n)))
    starts.append(("triangular", shape_triangular(n)))
    starts.append(("hann", shape_hann(n)))
    starts.append(("blackman", shape_blackman(n)))
    starts.append(("central_binomial", shape_central_binomial(n)))

    # Gaussian families (a few sigmas)
    for sigma_factor in (0.15, 0.20, 0.30):
        sigma = sigma_factor * n
        starts.append((f"gauss_{sigma_factor:.2f}", shape_gaussian(n, sigma)))

    # Exponential families
    for tau_factor in (0.12, 0.18):
        tau = tau_factor * n
        starts.append((f"exp_{tau_factor:.2f}", shape_exponential(n, tau)))

    # Random Dirichlet draws
    for alpha in (0.3, 0.7, 1.5):
        for _ in range(4):
            starts.append(
                (f"dirichlet_{alpha:.1f}", shape_random_dirichlet(n, alpha, rng))
            )

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_w = None
    best_score = np.inf
    start_time = time.time()
    total_budget = float(budget_s) - 5.0          # keep a few seconds for safety

    # ------------------------------------------------------------------
    # Phase 1 – quick sub‑gradient refinement of each start
    # ------------------------------------------------------------------
    per_start_budget = max(2.0, total_budget * 0.25 / max(1, len(starts)))
    for name, w0 in starts:
        elapsed = time.time() - start_time
        if elapsed > total_budget * 0.3:
            break

        # evaluate raw start
        score0 = compute_score(w0, n)
        if score0 < best_score:
            best_score = score0
            best_w = w0.copy()
            print(f"[init] {name:12s}  score={score0:.6f}")

        # short sub‑gradient sprint
        time_left = total_budget - (time.time() - start_time)
        limit = min(per_start_budget, time_left * 0.4)
        w_opt, sc_opt = subgradient_optimize(w0, n, rng, limit, step0=0.8)

        if sc_opt < best_score:
            best_score = sc_opt
            best_w = w_opt.copy()
            print(f"[sub‑refine] {name:12s}  improved to {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long smooth‑max optimisation from the current best
    # ------------------------------------------------------------------
    time_remaining = total_budget - (time.time() - start_time)
    if time_remaining > 5.0 and best_w is not None:
        w_opt, sc_opt = smooth_max_optimize(best_w, n, rng, time_remaining)
        if sc_opt < best_score:
            best_score = sc_opt
            best_w = w_opt.copy()
            print(f"[smooth‑max] final refinement score={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the traditional √(2 n) ℓ₁‑norm
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # --------------------------------------------------------------
    # Log the final evaluation (using the provided guard‑wrapped function)
    # --------------------------------------------------------------
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:  # safety – if evaluate_sequence is unavailable
        final_score = compute_score(best_w, n)
    print(f"[final] length={n}, best score={best_score:.6f}, after scaling evaluate={final_score:.6f}")

    # store for possible reuse by subsequent runs
    height_sequence_1 = final_seq.tolist()

    # --------------------------------------------------------------
    # Return as plain Python list
    # --------------------------------------------------------------
    return [float(x) for x in final_seq]
