def subgradient_descent(w, rng, time_limit, start_time):

    best_w = w.copy(); best_score = compute_score(w)

    eta0 = 0.8; beta = 0.9; momentum = np.zeros_like(w)

    iteration=0

    while time.monotonic() - start_time < time_limit:

        conv = convolve_fft(w,w)

        maxb = np.max(conv)

        score = 2 * len(w) * maxb

        if score < best_score - 1e-12:

            best_score = score; best_w = w.copy()

        # active set mask
        eps = maxb * 1e-8  # relative tolerance
        active_mask = (conv >= maxb - eps).astype(np.float64)  # shape 2n-1

        # gradient
        grad = 2.0 * convolve_fft(active_mask, w[::-1])[len(w)-1:len(w)-1+len(w)]

        # momentum update
        eta = eta0 / np.sqrt(iteration + 1.0)

        momentum = beta * momentum - eta * grad

        w_candidate = w + momentum

        # projection onto simplex
        w = simplex_projection(w_candidate)

        # enforce symmetry
        w = symmetrize(w)

        iteration+=1

        # occasionally random block perturbation
        if iteration % 5000 == 0:
            w = block_perturbation(w, rng)
            w = w / np.sum(w)
            momentum = np.zeros_like(w)

        # maybe break if too many iterations w/out improvement?

    return best_w, best_score

"""
Hybrid subgradient descent + local search for minimizing the peak of a self‑convolution
on the probability simplex.  The algorithm starts from several Beta‑shaped
initialisations, runs a projected sub‑gradient descent on the exact L∞‑norm
(using the active set as a tight approximation of the gradient), and finally
performs a cheap symmetric pair‑wise transfer local search.  All steps respect
the non‑negativity and simplex constraints, and the final sequence is scaled to
the conventional √(2 n) ℓ₁‑norm before returning.
"""

import time
import numpy as np
from typing import List

# Helper functions ---------------------------------------------------------

def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for moderate n)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # remove tiny negatives
    return c

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, ∑x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # v is already on the simplex (or all zeros)
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def symmetrize(v: np.ndarray) -> np.ndarray:
    """Make v symmetric (v[i] = v[-i-1]) while keeping it on the simplex."""
    v_sym = (v + v[::-1]) * 0.5
    s = np.sum(v_sym)
    if s > 0:
        v_sym /= s
    return v_sym

def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 250)
    hi = max(2, n // 15)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.09)))   # moderate multiplicative noise
    cand[start:stop] *= factor
    # optional mirroring – keeps symmetry if we want
    if rng.random() < 0.3:
        cand = (cand + cand[::-1]) * 0.5
    np.clip(cand, 0.0, 1000.0, out=cand)
    # re‑normalise to the simplex
    cand = cand / np.sum(cand)
    return cand

def compute_score(w: np.ndarray) -> float:
    """Return the objective 2·n·max(w*w) for a vector w that already satisfies ∑w=1."""
    n = w.size
    conv = convolve_fft(w, w)
    maxb = np.max(conv)
    return 2.0 * n * maxb

# -------------------------------------------------------------
#   Sub‑gradient descent on the exact L∞‑norm
# -------------------------------------------------------------

def subgradient_descent(w0: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        start_time: float) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent that uses the exact active set of the
    self‑convolution to build the gradient.  The step size is chosen by a
    simple line‑search‑like formula  t = η·max / ||g||²   and a slowly decaying
    factor η.  Symmetry is enforced after each projection.
    """
    w = w0.copy()
    n = w.size
    best_w = w.copy()
    best_score = compute_score(w)
    iteration = 0
    eta0 = 1.0                # initial multiplier for the line‑search step
    eta = eta0
    step_factor = 0.5         # multiplied with the line‑search step
    decay = 0.9999
    momentum = np.zeros_like(w)

    while time.monotonic() < deadline - 0.5:
        conv = convolve_fft(w, w)
        maxb = np.max(conv)
        score = 2.0 * n * maxb

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[sgd] iter={iteration:6d} score={best_score:.6f}")

        # ---- active set -------------------------------------------------
        eps = maxb * 1e-8
        active = (conv >= maxb - eps).astype(np.float64)

        # gradient = 2·conv(active, w_rev) restricted to the n‑dimensional slice
        grad = 2.0 * convolve_fft(active, w[::-1])[n - 1:n - 1 + n]

        # line‑search‑like step (t = step_factor * maxb / ||grad||²)
        grad_norm_sq = np.dot(grad, grad) + 1e-20
        t = step_factor * maxb / grad_norm_sq

        # momentum update (classical heavy‑ball)
        iteration += 1
        eta = eta0 / np.sqrt(iteration)
        momentum = 0.9 * momentum - eta * grad
        w_candidate = w - t * grad + momentum

        # project onto simplex and enforce symmetry
        w = simplex_projection(w_candidate)
        w = symmetrize(w)

        # cool‑down the step factor
        step_factor = max(step_factor * decay, 1e-5)

        # occasional random jumps to escape plateaus
        if iteration % 4000 == 0:
            w = block_perturbation(w, rng)
            momentum = np.zeros_like(w)
            step_factor = 0.5

    return best_w, best_score

# -------------------------------------------------------------
#   Simple symmetric local search (pairwise mass transfer)
# -------------------------------------------------------------

def local_search(w0: np.ndarray,
                 rng: np.random.Generator,
                 deadline: float,
                 start_time: float) -> (np.ndarray, float):
    """
    Randomly moves a small amount of probability mass from one index to another
    (together with their mirror positions) while keeping the vector on the
    simplex.  Accepts only improving moves.
    """
    w = w0.copy()
    n = w.size
    best_w = w.copy()
    best_score = compute_score(w)

    while time.monotonic() < deadline - 0.5:
        # pick two distinct positions in the left half (symmetry will handle the right)
        i = int(rng.integers(0, n // 2))
        j = int(rng.integers(0, n // 2))
        if i == j:
            continue

        # maximal admissible delta is limited by the smallest of the four entries
        delta_max = min(w[i], w[n - 1 - i])
        if delta_max <= 0:
            continue
        # sample a fraction of that amount
        delta = float(rng.random() * delta_max * 0.2 + 1e-12)

        # propose the move (keep symmetry)
        w_new = w.copy()
        w_new[i] -= delta
        w_new[j] += delta
        w_new[n - 1 - i] -= delta
        w_new[n - 1 - j] += delta

        # safety check – non‑negativity after the move
        if np.min(w_new) < -1e-12:
            continue

        score_new = compute_score(w_new)
        if score_new < best_score - 1e-12:
            w = w_new
            best_w = w_new.copy()
            best_score = score_new
            print(f"[local] improved score={best_score:.6f}")

    return best_w, best_score

# -------------------------------------------------------------
#   Initial shape generators
# -------------------------------------------------------------

def beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) density on {0,…,n‑1}.
    For α<1 the distribution is U‑shaped (large mass at the ends).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # the formula diverges for α=0.5 at the borders; add a tiny epsilon to avoid infinities
    eps = 1e-12
    vals = np.where(np.isfinite(vals), vals, eps)
    vals = vals / np.sum(vals)
    return vals

def central_binomial_shape(n: int) -> np.ndarray:
    """Classic central‑binomial (Catalan‑like) distribution."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    a = a / np.sum(a)
    return a

"""
Hybrid sub‑gradient descent + local search for minimizing the peak of a
self‑convolution on the probability simplex.

The algorithm proceeds as follows:

1.  Choose a length (re‑using a previously discovered construction if
    available, otherwise ≈ 8000) and generate several symmetric initial
    distributions – Beta‑shaped U‑distributions (α < 1) together with the
    classic central‑binomial shape.

2.  For each initialisation a projected sub‑gradient descent is run on the
    exact L∞‑norm of the convolution:
        •  the active set (indices where the convolution achieves its
           current maximum) defines the exact sub‑gradient;
        •  a line‑search‑type step size  t = η·max / ‖g‖²  is used;
        •  after each projection onto the simplex the vector is symmetrised.

    A cheap random block perturbation is performed every few thousand
    iterations to escape plateaus.

3.  A short symmetric pair‑wise local search refines the best vector:
    a small amount of probability mass is transferred between two
    positions (together with their mirror positions) and the move is kept
    only if it strictly reduces the objective.

4.  The best vector found over all initialisations is finally scaled to the
    conventional √(2 n) ℓ₁‑norm (the score is scale‑invariant) and returned.

All helper functions are defined at the top level; the harness will call
`propose_candidate` directly.
"""

import time

# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for moderate n)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, ∑x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Make v symmetric (v[i] = v[-i‑1]) while preserving the simplex."""
    v_sym = (v + v[::-1]) * 0.5
    s = np.sum(v_sym)
    if s > 0:
        v_sym /= s
    return v_sym


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block (keeps symmetry)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 250)
    hi = max(2, n // 15)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.09)))   # moderate noise
    cand[start:stop] *= factor
    if rng.random() < 0.3:                         # optional forced symmetry
        cand = (cand + cand[::-1]) * 0.5
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= np.sum(cand)                           # back to the simplex
    return cand

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale invariant, but we output
    # the conventional √(2 n) ℓ₁‑norm for readability.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_overall_w * target_sum / np.sum(best_overall_w)

    # safety clamp (required by `evaluate_sequence`)
    np.clip(final_seq, 0.0, 1000.0, out=final_seq)

    # one final call just to report the score
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n} internal best={best_overall_score:.6f} "
          f"evaluate_sequence={final_score:.6f}")

    # return a plain Python list
    return [float(x) for x in final_seq]
```
