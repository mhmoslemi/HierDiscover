"""
propose_candidate
-----------------
Search for a low‑max‑auto‑convolution step function.

The algorithm is an entropic (multiplicative) mirror‑descent on a
soft‑max approximation of   max_k (w * w)[k]  with the following
enhancements compared with the reference:

*   Use a larger support (default 20 000) – more degrees of freedom
    typically lower the achievable max‑convolution.
*   Initialise from the previously found best sequence (if available) by
    linear interpolation to the new length, otherwise fall back to the
    Catalan‑based shape.
*   Adam‑style moment adaptivity on the log‑domain update (still a
    multiplicative step, i.e. w ← w·exp(‑η·g) ).
*   Faster temperature growth for the soft‑max (β≈0.7·√iteration) and a
    higher cap – this focuses the gradient on the true max earlier.
*   More frequent symmetrisation (the optimum is symmetric) and cheap
    block‑perturbations to escape plateaus.
*   A simple reset mechanism when no improvement is seen for a while.
*   A short final random‑perturbation sweep for fine‑tuning.

The whole search runs until *budget_s* seconds have elapsed (default
~950 s, leaving a few seconds for final scaling).  The best sequence
found is returned, scaled to the conventional ℓ¹‑norm √(2 n) required
by `evaluate_sequence`.  The returned list consists of non‑negative
floats only.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers C_i / 4**i.
    Recurrence a_{i+1}=a_i·(2 i+1)/(2 i+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution using a real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negatives caused by round‑off.
    np.maximum(c, 0.0, out=c)
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump: multiply a random block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # ≈0.5 %
    hi = max(2, n // 12)           # ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal multiplier
    cand[start:stop] *= factor
    # Occasionally enforce a little symmetry – helps the optimum.
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        g = 2·corr(w, α)
    where α is a probability vector of length 2·n‑1.
    Implemented via a second FFT convolution.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = 3·n‑2
    start = w.shape[0] - 1                     # index that aligns with j=0
    return 2.0 * conv[start:start + w.shape[0]]


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is as low as possible.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a length – larger supports give a better asymptotic bound.
    # ------------------------------------------------------------------
    target_n = 20000                     # feel free to tune up/down
    if "height_sequence_1" in globals():
        try:
            prior = np.asarray(height_sequence_1, dtype=np.float64)
            if prior.ndim != 1 or prior.size < 2:
                raise ValueError
            # Interpolate the old shape onto the new grid.
            x_old = np.linspace(0.0, 1.0, prior.size)
            x_new = np.linspace(0.0, 1.0, target_n)
            w = np.interp(x_new, x_old, prior)
            w = np.maximum(w, 0.0)
        except Exception:
            # fallback to Catalan if anything goes wrong
            w = catalan_sequence(target_n)
    else:
        w = catalan_sequence(target_n)

    # Normalise to a probability vector (sum == 1) – the objective is
    # scale‑invariant, the ℓ¹‑norm will be fixed later.
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full(target_n, 1.0 / target_n)
    else:
        w = w / w_sum

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_w = w.copy()
    # Since w sums to 1, the raw objective is just max_b.
    b = convolve_fft(w, w)
    best_score = float(np.max(b))
    print(f"[init] n={target_n}, initial raw score={best_score:.6f}")

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # Hyper‑parameters (tuned empirically for the larger support)
    eta0 = 6.0                 # base step size for multiplicative update
    beta0 = 0.7                # temperature growth factor (≈β·√t)
    beta_max = 300.0
    reset_patience = 20000    # iterations without improvement before a reset
    block_every = 2500        # cheap random jumps
    symmetrize_every = 300    # enforce symmetry regularly

    # Adam‑style moments on the log‑domain update
    m = np.zeros(target_n, dtype=np.float64)
    v = np.zeros(target_n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # --------------------------------------------------------------
        # 1. self‑convolution and current (hard) score
        # --------------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = float(np.max(b))
        score = max_b                     # sum(w)=1 ⇒ objective = max_b

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, raw score={best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 2. Smooth‑max (soft‑max) weights α
        # --------------------------------------------------------------
        beta = beta0 * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b                     # ≤ 0, max entry becomes 0
        exp_term = np.exp(beta * diff)       # stability: max term = 1
        alpha = exp_term / np.sum(exp_term)  # length = 2·n‑1, sums to 1

        # --------------------------------------------------------------
        # 3. Gradient of the smoothed objective
        # --------------------------------------------------------------
        grad = correlation_grad(w, alpha)    # shape (n,)

        # --------------------------------------------------------------
        # 4. Adam‑style moment updates on the log‑domain (mirrored step)
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        # Multiplicative update with clipping to avoid overflow.
        eta = eta0 / np.sqrt(iteration + 1)
        delta = -eta * m_hat / (np.sqrt(v_hat) + eps)
        # Clip the exponent to a safe range.
        delta = np.clip(delta, -50.0, 50.0)
        w = w * np.exp(delta)

        # ------------------------------------------------------------------
        # 5. Projection back onto the simplex (non‑negative, sum = 1)
        # ------------------------------------------------------------------
        w = np.maximum(w, 0.0)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / target_n)
        else:
            w /= w_sum

        # ------------------------------------------------------------------
        # 6. Periodic symmetrisation – the optimal shape is symmetric.
        # ------------------------------------------------------------------
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= w.sum()

        # ------------------------------------------------------------------
        # 7. Occasional cheap random block jump (helps escape plateaus).
        # ------------------------------------------------------------------
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = float(np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best raw score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 8. Reset if stuck for a long time.
        # ------------------------------------------------------------------
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final fine‑tuning: a short burst of cheap random moves.
    # ------------------------------------------------------------------
    for _ in range(120):
        cand = block_perturbation(best_w, rng)
        cand = cand / np.sum(cand)
        cand_score = float(np.max(convolve_fft(cand, cand)))
        if cand_score < best_score:
            best_w = cand.copy()
            best_score = cand_score

    # ------------------------------------------------------------------
    # Scale to the conventional √(2 n) ℓ¹‑norm (the evaluator is
    # scale‑invariant, but this yields nicely‑shaped numbers).
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * target_n)
    final_seq = best_w * target_sum / best_w.sum()

    # Optional sanity check using the supplied evaluator.
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:   # the harness will call evaluate_sequence itself
        final_score = float('nan')
    print(f"[final] length={target_n}, best raw score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Return a plain Python list of floats.
    return [float(x) for x in final_seq]
