"""
Propose a step‑function with a low ∥f∗f∥∞ value.

The algorithm consists of three phases

1. **Linear‑programming refinement** –  for a fixed length `n` we solve
   ``max  ∑ g_i   s.t.  f∗g ≤ M`` where `M = max(f∗f)`.  The LP pushes
   mass to positions where `f` is small, thereby creating a new
   distribution with a strictly larger ℓ₁‑norm while respecting the same
   convolution bound.  Normalising the optimiser reduces the peak of
   the self‑convolution and gives a monotone improvement.

2. **Smooth‑max projected gradient descent** –  a fast log‑sum‑exp
   surrogate of the L∞ norm is minimised with Euclidean projection onto
   the probability simplex.  The parameters (`η₀`, momentum, smoothing
   factor) are tuned for a longer run than the reference implementation.

3. **Local stochastic hill‑climbing** –  the final distribution is
   refined by tiny mass transfers between two positions; moves that
   lower the exact objective are always accepted, occasional uphill
   moves are allowed according to a linear cooling schedule.

All phases respect the simplex constraints (`a_i ≥ 0`, `∑a_i = 1`);
the final sequence is scaled to the conventional √(2 n) ℓ₁‑norm,
which does not change the value returned by `evaluate_sequence`.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # tiny negative numbers are only numerical noise – clip them
    np.maximum(c, 0.0, out=c)
    return c


def build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Construct the sparse matrix A (size (2n‑1)×n) such that
    A @ g == convolve(f, g) for any vector g.
    Each diagonal of A is constant and equal to one entry of `f`.
    """
    n = f.shape[0]
    # data[d] is a length‑n vector filled with f[d] (constant diagonal)
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0 … n‑1
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def simplex_score(w: np.ndarray) -> float:
    """Exact 2 n·max((w∗w)) for a probability vector `w`."""
    n = w.size
    b = convolve_fft(w, w)
    return float(2.0 * n * np.max(b))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, ∑ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            w = np.maximum(init, 0.0)
    else:
        # fall back to the classic central‑binomial (≈1/√i) shape
        n = max_n
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2 * i)
        w = a

    # Normalise to the probability simplex (∑w = 1)
    s = np.sum(w)
    if s < 1e-12:
        w = np.full(n, 1.0 / n)
    else:
        w = w / s

    # --------------------------------------------------------------
    # Track the best solution seen so far.
    # --------------------------------------------------------------
    best_w = w.copy()
    best_score = simplex_score(best_w)
    print(f"[init] n={n}, score={best_score:.6f}")

    start_time = time.monotonic()
    elapsed = lambda: time.monotonic() - start_time

    # ------------------------------------------------------------------
    # 2️⃣ LP refinement phase (≈30 % of the time budget)
    # ------------------------------------------------------------------
    lp_time_budget = budget_s * 0.30
    iteration = 0
    while elapsed() < lp_time_budget:
        # current bound M = max(f∗f)
        M = np.max(convolve_fft(w, w))

        # Build the convolution matrix for the current w
        A = build_conv_matrix(w)

        # Solve: maximise ∑g_i  ⇔ minimise −∑g_i
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Use the efficient HiGHS interior‑point solver
        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(lp_time_budget - elapsed(), 1.0)},
        )

        if not res.success:
            print("[lp] linprog failed – exiting LP phase")
            break

        g = res.x
        sum_g = np.sum(g)

        # If the LP could not increase the ℓ₁‑norm, we are at a fixed point.
        if sum_g <= 1.0 + 1e-12:
            print("[lp] no ℓ₁‑norm improvement (sum_g≈1).")
            break

        # Normalise and enforce symmetry – symmetry is known to be beneficial.
        w_candidate = g / sum_g
        w_candidate = (w_candidate + w_candidate[::-1]) / 2.0
        w_candidate /= np.sum(w_candidate)

        cand_score = simplex_score(w_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_w = w_candidate.copy()
            print(f"[lp‑improve] iter={iteration:4d}, score={best_score:.6f}")

        # Accept the candidate (it never hurts the bound M)
        w = w_candidate
        iteration += 1

    # ------------------------------------------------------------------
    # 3️⃣ Smoothed‑max projected gradient descent (≈55 % of time budget)
    # ------------------------------------------------------------------
    gd_time_budget = budget_s * 0.55
    # hyper‑parameters
    eta0 = 0.8               # base step‑size
    beta = 0.95              # momentum
    mu_factor = 0.6          # start smoothing as μ = mu_factor·max(b)
    mu_factor_min = 0.001
    mu_decay = 0.9999
    momentum = np.zeros_like(w)

    while elapsed() < gd_time_budget:
        b = convolve_fft(w, w)
        maxb = np.max(b)
        score = 2.0 * n * maxb

        # Record best known
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[gd‑improve] iter={iteration:6d}, score={best_score:.6f}")

        # Smooth‑max (log‑sum‑exp) weights
        mu = maxb * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        shifted = (b - maxb) / mu          # max(b) → 0, others ≤ 0
        exp_term = np.exp(shifted)         # safely in (0,1]
        p = exp_term / np.sum(exp_term)    # soft‑max probabilities

        # Gradient of the smooth max: 2·conv(p, w_rev)
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1 : n - 1 + n]

        # Diminishing step‑size + momentum
        eta = eta0 / np.sqrt(iteration + 1.0)
        momentum = beta * momentum - eta * grad
        w_next = w + momentum

        # Projection onto the simplex
        w = project_simplex(w_next)

        # Optional symmetrisation (helps the structure)
        w = (w + w[::-1]) / 2.0

        # Reduce smoothing factor slowly
        mu_factor = max(mu_factor * mu_decay, mu_factor_min)

        iteration += 1

    # ------------------------------------------------------------------
    # 4️⃣ Local stochastic hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    hc_time_budget = budget_s - elapsed()
    if hc_time_budget > 5.0:
        # temperature schedule – starts relatively high and cools linearly
        T0 = 0.05 * best_score
        end_time = start_time + budget_s - 0.5
        current_w = best_w.copy()
        current_score = best_score
        hc_iter = 0

        while time.monotonic() < end_time:
            # choose two distinct indices
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            # amount to move – a fraction of the donor mass
            max_delta = min(current_w[i], 0.02)  # never shift more than 2 %
            delta = rng.random() * max_delta

            cand = current_w.copy()
            cand[i] -= delta
            cand[j] += delta

            # project back onto simplex (tiny violations may appear)
            cand = project_simplex(cand)

            # evaluate exact objective
            cand_score = simplex_score(cand)

            # acceptance rule (Metropolis)
            if cand_score < current_score:
                accept = True
            else:
                # linear cooling – temperature decreases to zero at the end
                remaining = max(end_time - time.monotonic(), 0.0)
                T = T0 * (remaining / hc_time_budget)
                accept = rng.random() < np.exp(-(cand_score - current_score) / max(T, 1e-12))

            if accept:
                current_w = cand
                current_score = cand_score
                if current_score < best_score - 1e-12:
                    best_score = current_score
                    best_w = current_w.copy()
                    print(f"[hc‑improve] iter={hc_iter:6d}, score={best_score:.6f}")

            hc_iter += 1
            # very occasional full reshuffle to avoid stagnation
            if hc_iter % 20000 == 0:
                current_w = best_w * (1.0 + 0.1 * rng.normal(size=n))
                current_w = project_simplex(np.clip(current_w, 0.0, None))

    # ------------------------------------------------------------------
    # 5️⃣ Finalisation – scale to √(2 n) for aesthetic compatibility.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    # sanity check (the harness will also check, but we like to see it)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] n={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Return plain Python list as required by the harness.
    return [float(x) for x in final_seq]
