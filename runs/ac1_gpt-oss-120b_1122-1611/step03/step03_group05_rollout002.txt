"""
propose_candidate
-----------------

A two‑stage heuristic for the min‑max self‑convolution problem.

*Stage 1 – shape exploration*  
  Sample a large number of simple parametric families (Beta,
  exponential decay, power‑law, Catalan, uniform).  Afterwards a
  “symmetrisation + simplex projection” step guarantees a feasible
  probability vector (non‑negative, ℓ¹‑norm = 1).  The best shape found
  is kept as a warm‑start for the next stage.

*Stage 2 – projected sub‑gradient descent*  
  Starting from the best shape we minimise the exact objective
  `max (a*a)` (the raw maximum of the self‑convolution).  For a
  current vector `w` we locate the index `k*` attaining the maximum,
  compute the sub‑gradient `g_i = 2·w_{k*‑i}` on its support and apply
  an Adam update.  After each Adam step the iterate is projected onto
  the probability simplex using Duchi’s algorithm and then
  symmetrised (`w ← (w+w[::-1])/2`).  Occasional block‑perturbations
  and a restart policy avoid stagnation.

The algorithm works directly with the invariant ℓ¹‑norm = 1; the final
sequence is scaled to the conventional `√(2 n)` norm required by
`evaluate_sequence`.  The whole search respects the supplied time
budget (default ≈ 1000 s) and prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def catalan_sequence(length: int) -> np.ndarray:
    """First `length` Catalan numbers divided by 4**i:
    a_i = C(2i,i)/4**i = product_{j=0}^{i-1} (2j+1)/(2j+2)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x ≥ 0, Σ x = 1} (Duchi et al., 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are (almost) equal; return uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise to exactly 1.
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation of `base`.  A random contiguous block
    is multiplied by a log‑normal factor (≈ ±12 %).  With probability 0.35
    the result is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and ~8 % of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈ ±12 %
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    # re‑normalise to a probability vector
    s = cand.sum()
    if s == 0:
        cand = np.full_like(cand, 1.0 / n)
    else:
        cand = cand / s
    return cand


def self_convolve_fft(w: np.ndarray, N_fft: int) -> np.ndarray:
    """
    Linear self‑convolution of `w` with itself using an FFT of length `N_fft`.
    Returns the full convolution (length 2*len(w)-1) with tiny negative
    values clamped to zero.
    """
    Fw = np.fft.rfft(w, N_fft)
    conv_full = np.fft.irfft(Fw * Fw, N_fft)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full

# Main entry point required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function with a small self‑convolution
    peak.  Returns a list of floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose problem size and an initial feasible vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.ndim != 1 or hs.size < 2:
                raise ValueError
            n = int(hs.size)
            w = np.maximum(hs, 0.0)
            s = w.sum()
            if s < 1e-12:
                raise ValueError
            w = w / s                         # normalise to ℓ¹‑norm = 1
        except Exception:
            n = 8000
            w = catalan_sequence(n).astype(np.float64)
            w = w / w.sum()
    else:
        n = 8000
        w = catalan_sequence(n).astype(np.float64)
        w = w / w.sum()

    # Pre‑compute FFT lengths
    N_self = _next_pow2(2 * n - 1)          # for self‑convolution
    # --------------------------------------------------------------
    # Helper for fast raw‑max evaluation
    # --------------------------------------------------------------
    def raw_max(seq: np.ndarray) -> float:
        """Maximum entry of the self‑convolution of `seq` (sum(seq)=1)."""
        return np.max(self_convolve_fft(seq, N_self))

    # --------------------------------------------------------------
    # Stage 1 – parametric shape exploration
    # --------------------------------------------------------------
    start_time = time.monotonic()
    best_w = w.copy()
    best_score = raw_max(best_w)

    # fixed grid of x values (mid‑point rule) for shape families
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                     # points in (0,1)

    families = ["beta", "exp", "power", "catalan", "uniform"]
    # maximum number of candidates; we stop earlier if time runs out
    max_candidates = 2500
    for cand_id in range(max_candidates):
        if time.monotonic() - start_time > budget_s * 0.25:
            break

        family = rng.choice(families)

        if family == "beta":
            # sample α, β log‑uniformly in [0.1, 5]
            log_a = rng.uniform(-1.0, 0.7)
            log_b = rng.uniform(-1.0, 0.7)
            a = 10 ** log_a
            b = 10 ** log_b
            seq = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
        elif family == "exp":
            lam = rng.uniform(0.0, 0.025)          # decay rate
            seq = np.exp(-lam * i)
        elif family == "power":
            p = rng.uniform(0.2, 2.0)
            seq = (i + 1.0) ** (-p)
        elif family == "catalan":
            seq = catalan_sequence(n).astype(np.float64)
        else:  # "uniform"
            seq = np.ones(n, dtype=np.float64)

        # Guard against pathological zero vectors
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0:
            continue
        seq = seq / s

        # Enforce symmetry – almost all good solutions are symmetric
        seq = 0.5 * (seq + seq[::-1])
        seq = seq / seq.sum()

        cur_score = raw_max(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_w = seq.copy()
            print(f"[shape] id={cand_id:04d}, fam={family}, max={best_score:.6e}")

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient (Adam) optimisation
    # --------------------------------------------------------------
    w = best_w.copy()

    # Adam hyper‑parameters (tuned for the sub‑gradient case)
    lr0 = 0.04
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    iteration = 0
    no_improve = 0
    start_opt = time.monotonic()
    while time.monotonic() - start_opt < budget_s * 0.70:
        # ----- 2.1 compute current convolution and max -----
        conv = self_convolve_fft(w, N_self)
        cur_max = np.max(conv)

        # keep global best
        if cur_max < best_score - 1e-15:
            best_score = cur_max
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:06d}, max={best_score:.6e}")
        else:
            no_improve += 1

        # ----- 2.2 sub‑gradient at the current maximiser -----
        k_max = int(np.argmax(conv))
        g = np.zeros_like(w)

        # Only indices that really contribute to (w*w)[k_max] have non‑zero gradient.
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g[idx] = 2.0 * w[k_max - idx]

        # ----- 2.3 Adam update + simplex projection + symmetrisation -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        w_tmp = w - step
        w_tmp = simplex_projection(w_tmp)

        # Enforce symmetry (helps flatten the autocorrelation)
        w_tmp = 0.5 * (w_tmp + w_tmp[::-1])
        w = w_tmp

        # ----- 2.4 occasional global perturbations -----
        if iteration % 4000 == 0:
            cand = block_perturbation(best_w, rng)
            cand_max = raw_max(cand)
            if cand_max < best_score:
                best_score = cand_max
                best_w = cand.copy()
                w = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[perturb] iter={iteration:06d}, new best={best_score:.6e}")

        # ----- 2.5 stagnation handling -----
        if no_improve > 60000:
            # Restart from a lightly perturbed version of the best known point.
            w = block_perturbation(best_w, rng)
            w = simplex_projection(w)
            w = 0.5 * (w + w[::-1])
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:06d}, reset to perturbed best")

    # --------------------------------------------------------------
    # Final scaling to the conventional ℓ¹‑norm √(2 n)
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()   # keeps ℓ¹‑norm = √(2 n)

    # Optional sanity check (does not affect the returned value)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, raw best={best_score:.6e}, evaluate={final_score:.6f}")
    except Exception:
        pass

    # Return a plain Python list as required by the harness.
    return [float(v) for v in final_seq]
