def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    # Setup RNG
    rng = np.random.default_rng(seed)

    # Determine n and initial a
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            # fallback
            n = 256
            a = np.full(n, 1.0/n, dtype=np.float64)
        else:
            a = np.clip(a, 0, None)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = np.full_like(a, 1.0/a.size)
    else:
        n = 256
        a = np.full(n, 1.0/n, dtype=np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)

    # compute initial conv and score
    conv = _conv_self(a, fft_len)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best = a.copy()
    best_score = cur_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # Phase schedule
    total_time = deadline - start_time
    early_phase = 0.70 * total_time

    # Pre-compute hyperparams
    # Mirror descent params:
    eta_start = 1.2
    eta_end = 0.01
    beta_start = 0.5
    beta_end = 500.0
    K_active = max(200, int(0.2 * conv.size))
    momentum = 0.95
    grad_prev = np.zeros_like(a)

    # Main loops
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        # Determine which phase
        if elapsed < early_phase:
            # Mirror descent
            eta = eta_start * (1.0 - elapsed/early_phase) + eta_end * (elapsed/early_phase)
            beta = beta_start + (beta_end - beta_start) * (elapsed/early_phase)
            # compute active indices
            if K_active >= conv.size:
                active_idx = np.arange(conv.size)
            else:
                kth = conv.size - K_active
                active_idx = np.argpartition(conv, kth)[kth:]
            # compute weighted gradient
            max_conv = conv.max()
            rev_a = a[::-1]
            grad = np.zeros_like(a)
            for k in active_idx:
                w = np.exp(beta * (conv[int(k)] - max_conv))
                if w == 0.0:
                    continue
                kk = int(k)
                if kk < n:
                    # indices 0..kk inclusive
                    # partner vector = a[kk::-1]
                    grad[:kk+1] += w * a[kk::-1]
                else:
                    start = kk - (n - 1)
                    length = (2*n - 1) - kk
                    grad[start:] += w * rev_a[:length]
            # momentum
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad
            # exponentiated gradient update
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s
            # occasional block perturb
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
            # recompute conv and score
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
        else:
            # Local random search
            # temperature schedule
            t_elapsed = elapsed - early_phase
            t_total = total_time - early_phase
            T = max(1e-6, 0.05 * (1.0 - t_elapsed / t_total))  # T decays to almost zero
            # propose move
            if rng.random() < 0.7:
                # pairwise transfer
                # donor must have > some mass
                i = rng.integers(0, n)
                if a[i] < 1e-12:
                    continue
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                # delta fraction
                max_frac = min(0.5, a[i] / a[i])  # just 0.5
                delta = a[i] * rng.random() * max_frac
                if delta <= 0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # ensure non-negative:
                if a_new[i] < 0:
                    continue
                conv_new = _conv_self(a_new, fft_len)
                new_score = 2.0 * n * conv_new.max()
                accept = False
                if new_score < cur_score:
                    accept = True
                else:
                    # probabilistic acceptance
                    if rng.random() < np.exp(-(new_score - cur_score) / T):
                        accept = True
                if accept:
                    a = a_new
                    conv = conv_new
                    cur_score = new_score
                    cur_max = conv.max()
            else:
                # block perturb (log-normal jitter) similar to earlier.
                a = _random_block_perturb(a, rng)
                conv = _conv_self(a, fft_len)
                cur_score = 2.0 * n * conv.max()
                cur_max = conv.max()
        # record best
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")
        # optional restart if no improvement for some time
        # we can track last_improve
        if elapsed - (last_improve if 'last_improve' in locals() else 0) > 30.0:
            # restart from best with jitter
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise

            a /= a.sum()
            grad_prev = np.zeros_like(a)
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            cur_max = conv.max()
            last_improve = time.monotonic()

    # return best clipped to limits
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""Propose a step function that beats the current best bound (≈1.514).

This implementation consists of two stages:

1. **Soft‑max mirror descent** – a smooth approximation of the
   max‑norm is minimised using an exponentiated‑gradient (mirror‑descent)
   update.  The gradient is formed from a weighted sum of the *K* largest
   convolution entries, with a temperature β that grows from 1 to a large
   value (≈500) so the algorithm increasingly concentrates on the true
   peak.  Momentum (0.95) stabilises the updates.

2. **Fine‑grained stochastic local search** – once the schedule reaches
   the end of the bulk stage, we switch to a simulated‑annealing style
   search that performs targeted pairwise mass transfers and occasional
   block‑perturbations.  The acceptance probability uses a linearly
   decreasing temperature, allowing the algorithm to escape shallow
   local minima and polish the solution.

Both stages keep the distribution normalised (∑a=1) and retain the best
vector seen so far.  The routine respects the 1000 s time budget and
returns a list of non‑negative heights ready for `evaluate_sequence`."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two length that can hold the full linear
    convolution of two length‑n vectors."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector `a` using FFT.
    The returned array has length 2*len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling
    # ------------------------------------------------------------------
    # a small safety margin (5 s) is left for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose an initial probability vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full_like(a, 1.0 / a.size)
        else:
            # fallback – a modest uniform start
            n = 256
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # no prior construction – start from a Dirichlet draw
        n = 256
        a = rng.dirichlet(np.ones(n)).astype(np.float64)

    n = a.size
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len)
    cur_score = 2.0 * n * conv.max()          # ∑a = 1 → score = 2 n max(conv)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time
    # we spend roughly 70 % of the budget on the mirror‑descent phase
    phase1_end = start_time + 0.70 * total_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters for phase‑1 (mirror descent)
    # ------------------------------------------------------------------
    eta_start = 1.2
    eta_end = 0.01
    beta_start = 0.5
    beta_end = 500.0
    # a fairly large active set – 20 % of the convolution entries
    K_active = max(200, int(0.20 * conv.size))
    momentum = 0.95
    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – smooth mirror descent (dominates the early budget)
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedules for stepsize η and temperature β
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # indices of the K largest convolution entries
            if K_active >= conv.size:
                active_idx = np.arange(conv.size)
            else:
                kth = conv.size - K_active
                active_idx = np.argpartition(conv, kth)[kth:]

            # weighted gradient from the active entries
            max_conv = conv.max()
            rev_a = a[::-1]                     # one reversed copy per iteration
            grad = np.zeros_like(a)

            for k in active_idx:
                w = np.exp(beta * (conv[int(k)] - max_conv))
                if w == 0.0:
                    continue
                kk = int(k)
                if kk < n:
                    # contribution from the left part of the convolution
                    grad[: kk + 1] += w * a[kk :: -1]
                else:
                    # contribution from the right part
                    start = kk - (n - 1)
                    length = (2 * n - 1) - kk
                    grad[start:] += w * rev_a[:length]

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated‑gradient (mirror‑descent) update on the simplex
            factor = np.exp(-eta * grad)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

            # occasional random block perturbation to keep diversity
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)

            # re‑evaluate convolution and score
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Phase 2 – stochastic local refinement (the final 30 %)
        # --------------------------------------------------------------
        else:
            # simple linear temperature schedule for simulated annealing
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = max(1e-7, 0.05 * (1.0 - t_frac))

            # ----- pairwise mass transfer --------------------------------
            if rng.random() < 0.85:          # most proposals are pair moves
                # choose a donor with non‑trivial mass
                i = rng.integers(0, n)
                if a[i] < 1e-12:
                    continue
                # choose a distinct recipient
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1

                # move at most 30 % of the donor's mass
                max_delta = 0.30 * a[i]
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta

                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta

                # compute the new score (full FFT convolution)
                conv_new = _conv_self(a_new, fft_len)
                new_score = 2.0 * n * conv_new.max()

                # Metropolis acceptance
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                    conv = conv_new
                    cur_score = new_score

            # ----- block‑perturbation (global jitter) -------------------
            else:
                a = _random_block_perturb(a, rng)
                conv = _conv_self(a, fft_len)
                cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart when stuck for a while
        # --------------------------------------------------------------
        restart_gap = 30.0
        if now - last_improve > restart_gap:
            # restart from the current best with a mild multiplicative jitter
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (clipped to the limits enforced by
    #    `evaluate_sequence` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
