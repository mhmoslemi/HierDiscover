def arcsine_sequence(n):
    i = np.arange(n, dtype=np.float64)
    raw = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    scale = np.sqrt(2.0 * n) / np.sum(raw)
    seq = raw * scale
    return seq

"""
propose_candidate
-----------------
A fast, low‑dimensional search for a near‑optimal step function.
The search explores a small family of simple parametric shapes
(power‑law, shifted power‑law and their mixtures with the uniform
distribution) and then refines the best candidate with a few steps
of projected sub‑gradient descent on the exact max‑convolution
objective.  All candidates are normalised to Σ a_i = √(2 n) as
required by ``evaluate_sequence``.  The routine works for any
reasonable length (a few thousand) and finishes well within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures.
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Equivalent to ``np.convolve(a, b)`` but O(n log n).
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Numerical noise may produce tiny negatives – clip safely.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def catalan_normalised(n: int) -> np.ndarray:
    """
    Catalan‑type decreasing coefficients a_i = C(2i,i)/4**i,
    normalised so that Σ a_i = 1 (i.e. a probability vector).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    a /= a.sum()
    return a


def power_law_normalised(n: int, p: float, beta: float) -> np.ndarray:
    """
    Power‑law shape  w_i ∝ (i + β)^{-p}.
    The vector is normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = 1.0 / np.power(idx + beta, p)
    raw[0] = 1.0 / np.power(beta, p)          # explicit for i=0
    raw /= raw.sum()
    return raw


def evaluate_score(prob_vec: np.ndarray) -> float:
    """
    Evaluate the score for a probability vector (i.e. Σ w = 1).
    The returned value equals the value that ``evaluate_sequence``
    would give after the final scaling Σ a = √(2 n).
    """
    n = prob_vec.size
    b = convolve_fft(prob_vec, prob_vec)
    max_b = np.max(b)
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Main routine required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a non‑negative step function (list of floats) with a small
    value of ``evaluate_sequence``.
    The method searches a tiny parametric family and then performs a
    few sub‑gradient steps starting from the best candidate found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Choose the length of the sequence.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(base.size)
            if n < 2:
                raise ValueError
        except Exception:
            n = 7000
    else:
        n = 7000                     # a comfortable size for the search

    print(f"[info] using length n = {n}")

    # --------------------------------------------------------------
    # 2) Build a few baseline candidates.
    # --------------------------------------------------------------
    candidates = []

    # a) the classic Catalan shape
    w_catalan = catalan_normalised(n)
    candidates.append(("catalan", w_catalan))

    # b) pure uniform (useful for mixing)
    w_uniform = np.full(n, 1.0 / n)
    candidates.append(("uniform", w_uniform))

    # c) small grid of power‑law shapes
    for p in np.linspace(0.45, 0.55, 11):            # 0.45 … 0.55
        for beta in (0.0, 0.5, 1.0, 2.0):
            w_pl = power_law_normalised(n, p, beta + 1.0)   # +1 to keep i=0 finite
            candidates.append((f"pl(p={p:.2f},β={beta})", w_pl))

    # --------------------------------------------------------------
    # 3) Evaluate all raw candidates and keep the best.
    # --------------------------------------------------------------
    best_w = None
    best_score = np.inf
    best_name = None
    for name, w in candidates:
        score = evaluate_score(w)
        if score < best_score:
            best_score = score
            best_w = w.copy()
            best_name = name
        print(f"[init] {name:20s} score = {score:.6f}")

    print(f"[best init] {best_name:20s} → {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Simple mixture optimisation.
    #    Mix the current best with the uniform vector and with the
    #    reversed version of the best shape; keep any improvement.
    # --------------------------------------------------------------
    w_rev = best_w[::-1]               # mirrored version
    for t in np.linspace(0.0, 0.25, 21):    # mixture weight
        # mix with uniform
        w_mix = (1.0 - t) * best_w + t * w_uniform
        w_mix = w_mix / w_mix.sum()       # re‑normalise (should stay 1)
        score = evaluate_score(w_mix)
        if score < best_score:
            best_score = score
            best_w = w_mix.copy()
            best_name = f"mix_uniform(t={t:.3f})"
            print(f"[mix] improved with uniform t={t:.3f} → {score:.6f}")

        # mix with reversed shape
        w_mix2 = (1.0 - t) * best_w + t * w_rev
        w_mix2 = w_mix2 / w_mix2.sum()
        score2 = evaluate_score(w_mix2)
        if score2 < best_score:
            best_score = score2
            best_w = w_mix2.copy()
            best_name = f"mix_rev(t={t:.3f})"
            print(f"[mix] improved with rev t={t:.3f} → {score2:.6f}")

    # --------------------------------------------------------------
    # 5) A few iterations of projected sub‑gradient descent.
    #    The objective is  max_k (w*w)[k]; its sub‑gradient is
    #    2·reverse(w) shifted to the active index.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    max_time = max(0.0, float(budget_s) - 5.0)   # keep a couple of seconds free

    eta0 = 0.5                                 # initial step size
    iteration = 0
    while time.monotonic() - start_time < max_time and iteration < 8000:
        # 1) full convolution and index of the maximum entry
        b = convolve_fft(best_w, best_w)
        k_max = int(np.argmax(b))

        # 2) sub‑gradient of the max‑convolution entry
        #    grad_i = 2 * w_{k_max-i} for indices that exist
        grad = np.zeros_like(best_w)
        # vectorised version: reverse the vector and pad
        rev = best_w[::-1]
        # the slice we need starts at (n-1 - k_max)
        start = max(0, n - 1 - k_max)
        stop = start + n
        padded = np.pad(rev, (n - 1, n - 1), mode='constant')
        grad = 2.0 * padded[start:start + n]

        # 3) projected step
        eta = eta0 / np.sqrt(iteration + 1.0)
        w_candidate = best_w - eta * grad
        best_w = simplex_projection(w_candidate)

        # 4) record improvement
        score = evaluate_score(best_w)
        if score < best_score - 1e-9:
            best_score = score
            best_name = f"sgd_iter{iteration}"
            print(f"[sgd] iter={iteration:4d} new best {score:.6f}")
        iteration += 1

    # --------------------------------------------------------------
    # 6) Final scaling to the required ℓ¹‑norm √(2 n) and safety check.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum                # Σ final_seq = √(2 n)
    # One final call to the provided evaluator (optional sanity‑check)
    final_val = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score={best_score:.6f}, "
          f"evaluate={final_val:.6f}, source={best_name}")

    # Return a plain Python list – the harness expects this.
    return [float(x) for x in final_seq]
