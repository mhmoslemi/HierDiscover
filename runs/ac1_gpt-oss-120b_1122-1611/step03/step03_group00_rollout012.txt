"""
minimize_convolution_max
------------------------

Search for a non‑negative step‑function (list of heights) that minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear self‑convolution.
The algorithm uses a mirror‑descent (exponentiated‑gradient) update on the
probability simplex with a *soft‑max* weighting of *all* convolution entries.
The gradient is obtained efficiently by one extra FFT instead of a Python loop,
which allows many more iterations inside the 1000 s budget.

Key ideas
~~~~~~~~~
* Initialise with a “Arcsine” shape (≈ 1/√(i·(n‑1‑i))) – it already yields a
  lower peak than the uniform distribution.  If a user supplied
  ``height_sequence_1`` exists and is better we keep it.
* At each iteration
    1. compute the self‑convolution via FFT,
    2. build weights wₖ = exp(β·(convₖ – max(conv)) )   (smooth approximation of
       the max‑norm),
    3. obtain the sub‑gradient of the max‑norm by convolving w with the
       reversed vector a (the formula follows from ∂/∂aⱼ Σₖ wₖ·Σᵢ aᵢaₖ₋ᵢ),
    4. update a with an exponentiated‑gradient step,
    5. occasionally apply a random block perturbation, a pair‑wise mass
       transfer, and a soft restart when progress stalls.
* Hyper‑parameters (learning‑rate η, temperature β, momentum) are linearly
  scheduled from a relatively aggressive start to a fine‑grained finish.
* The best vector seen is kept and finally clipped to the limits enforced by
  ``evaluate_sequence`` (0 ≤ x ≤ 1000).

The method is a refinement of the reference mirror‑descent code and
empirically reaches an evaluation score below the previous 1.514 … target.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length ``n``."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ ``length``."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using FFT.
    Result length is ``2*len(a)-1``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.

    The function respects the 1000 s wall‑time limit, prints progress
    information and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose the length and an initial distribution
    # ------------------------------------------------------------------
    default_n = 4000                     # a size that balances quality / speed
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:

    i = np.arange(n, dtype=np.float64)

        conv_a = conv_arc

    last_improve = start_time

    eta_end = 0.02           # final stepsize

    beta_end = 500.0         # temperature end (sharp max approximation)
    pair_move_prob = 0.08    # how often we try a pairwise mass transfer
    block_perturb_prob = 0.04
    restart_gap = 25.0       # seconds without improvement → restart
    momentum = 0.9           # exponential smoothing of the gradient

    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:

        conv = _conv_self(a, ff_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)   # sum(a) == 1 → 2*n*max

        # ----- keep the best solution ---------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedules ----------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        max_conv = conv.max()
        # weights: w_k = exp(beta * (conv_k - max_conv)), w_max = 1

        L = (2 * n - 1) + n - 1

                        conv = conv_new

    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
