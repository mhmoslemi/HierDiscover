evaluate_sequence() score.

---------
1. Choose a problem size n (the length of the sequence).  If a previous
   construction `height_sequence_1` is supplied we reuse its length,
   otherwise we use a comfortable default n≈8000.
2. Build a small diverse pool of starting points:
   * arcsine law a_i ∝ (i·(n‑1‑i))^{-p} with several exponents p,
   * Catalan numbers,
   * a Dirichlet sample,
   * the supplied previous construction (if any).
   All candidates are normalised to the probability simplex
   (∑w = 1).

   score(w)=2·n·max(conv(w,w))) as the starting point for a
   projected sub‑gradient descent.

     convolution matrix that correspond to the K largest convolution
     entries (K≈5).  This reduces the chance of getting stuck at a
     single active peak.
   – A Nesterov‐style momentum term is added.
   – After each update the vector is projected onto the simplex and
     symmetrised (w←½(w+w[::-1])).
4. Periodically (every few thousand iterations) apply a cheap random
   block perturbation and a tiny random mass‑transfer move to escape
   flat regions.
5. When progress stalls a “reset” perturbs the current best solution
   and continues.
6. With the remaining time a short simulated‑annealing polish moves
   small mass packets while accepting uphill moves with a slowly
   decreasing temperature.
7. Finally the probability vector is scaled to the conventional
   ℓ¹‑norm √(2·n) required by evaluate_sequence() and the list is
   returned.

The whole routine respects the 1000 s budget (a small safety margin
is kept) and prints progress information.
"""

import time
import numpy as np

#  Helper utilities (top‑level as required)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast for vectors of length up to ~10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # eliminate tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}
       (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Large cheap random jump – multiply a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 250)
    hi = max(2, n // 30)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.13)))
    cand[start:stop] *= factor
    if rng.random() < 0.4:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan numbers a_i = C(2i,i)/4**i (stable recurrence)."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def arcsine_sequence(n: int, p: float = 0.5, eps: float = 1e-8) -> np.ndarray:
    """Discrete approximation of the arcsine law:
       a_i ∝ ( (i+0.5)*(n+0.5-i) )^{-p} ."""
    x = np.arange(n, dtype=np.float64) + 0.5
    denom = (x * (n + 0.5 - x))
    seq = np.power(denom + eps, -p)
    return seq


def score_from_w(w: np.ndarray, n: int) -> float:
    """Objective 2·n·max(conv(w,w)).  w should already sum to 1."""
    conv = convolve_fft(w, w)
    return 2.0 * n * np.max(conv)
