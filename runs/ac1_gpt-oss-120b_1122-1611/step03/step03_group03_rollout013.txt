def mass_transfer_perturbation(base, rng, eps_factor=0.1):
    cand = base.copy()
    n = len(cand)
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i==j: return cand
    delta = min(cand[i], eps_factor * cand[i] * rng.random())
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

"""
propose_candidate
-----------------
Hybrid optimiser for ``evaluate_sequence``.

The optimiser works on a probability vector (∑w = 1) and minimises
    2·n·max_k (w*w)[k]           (the objective after the final √(2 n) scaling)

Two complementary phases are used:

1. **Soft‑max Adam phase** – a smooth approximation of the max is
   minimised with Adam on the probability simplex.  The temperature
   α is increased quickly so the proxy quickly becomes sharp.

2. **Hard‑max sub‑gradient phase** – after α is large we switch to the
   true max and use a sub‑gradient (the correlation of w with the
   indicator at the active index).  Adam’s moments are retained to
   give a stable descent.

Throughout the run random block perturbations are applied to the
current best candidate, occasional restarts are triggered when progress
stops and the best sequence found is finally rescaled to the
convention √(2 n) L¹‑norm.

The algorithm is deliberately lightweight (FFT‑based convolutions,
simplex projection) and runs for the full time budget (≈1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers scaled by 4⁻ⁱ:
    a_i = C(2i,i) / 4ⁱ   (recurrence a_{i+1}=a_i*(2i+1)/(2i+2))."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (non‑circular) using real FFT.
    Small negative round‑off errors are clipped to 0."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # guard against -0 caused by rounding
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump – multiply a random contiguous block by a
    log‑normal factor, optionally symmetrise."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def upsample_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a 1‑D sequence to a new length."""
    if seq.size == new_len:
        return seq.copy()
    old_x = np.linspace(0, 1, seq.size)
    new_x = np.linspace(0, 1, new_len)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) that aims
    to minimise ``evaluate_sequence``.  The optimiser merges a soft‑max
    Adam stage with a hard‑max sub‑gradient stage and random perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose the working length.
    # If a previous construction exists we up‑sample it to give the optimiser
    # a larger design space; otherwise we fall back to a few thousand entries.
    # --------------------------------------------------------------
    target_len = 8000                      # a safe size for the 1000 s budget
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            # upsample / downsample to the target length
            w0 = upsample_sequence(prev, target_len)
        except Exception:
            w0 = catalan_sequence(target_len)
    else:
        w0 = catalan_sequence(target_len)

    # Normalise to a probability simplex (∑w = 1)
    w0 = np.maximum(w0, 0.0)
    if np.sum(w0) == 0:
        w0 = np.full(target_len, 1.0 / target_len)
    w = w0 / np.sum(w0)

    # --------------------------------------------------------------
    # Book‑keeping of the best candidate seen so far.
    # --------------------------------------------------------------
    conv = convolve_fft(w, w)
    best_score = float(2.0 * target_len * np.max(conv))   # same as evaluate_sequence after scaling
    best_w = w.copy()
    print(f"[init] n={target_len}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser state (shared between the two phases)
    # --------------------------------------------------------------
    m = np.zeros(target_len, dtype=np.float64)   # first moment
    v = np.zeros(target_len, dtype=np.float64)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.25               # a slightly larger base learning rate
    alpha = 5.0              # initial soft‑max temperature
    alpha_max = 5000.0

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Helper: hard‑max gradient (sub‑gradient of the max‑conv)
    # --------------------------------------------------------------
    def hard_max_grad(vec: np.ndarray, k_star: int) -> np.ndarray:
        """Gradient of max_k (vec*vec)[k] w.r.t. vec at the active index k_star."""
        n = vec.size
        grad = np.zeros_like(vec)
        # grad[i] = 2 * vec[k_star - i]  when the index is inside the support
        idx = np.arange(n)
        j = k_star - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * vec[j[mask]]
        return grad

    # --------------------------------------------------------------
    # Main optimisation loop – terminates a few seconds before the
    # given budget to give a safety margin.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ---- 1. current self‑convolution ---------------------------------
        conv = convolve_fft(w, w)                 # length 2n‑1

        # ---- 2. Choose the update rule ----------------------------------
        if alpha < 2000.0:                         # soft‑max Adam phase
            maxc = np.max(conv)
            # stabilise exponentials
            exp_vals = np.exp(alpha * (conv - maxc))
            p = exp_vals / np.sum(exp_vals)      # soft‑max weights, sum = 1

            # gradient of the smoothed objective:
            grad_full = convolve_fft(p, w[::-1])   # length 3n‑2
            start = target_len - 1
            grad = 2.0 * grad_full[start:start + target_len]
        else:                                      # hard‑max sub‑gradient phase
            k_star = int(np.argmax(conv))
            grad = hard_max_grad(w, k_star)

        # ---- 3. Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)                     # 1/√t decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w = simplex_projection(w_new)            # keep on the simplex

        # ---- 4. occasional symmetry enforcement (optional) -------------
        if iteration % 1200 == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)

        # ---- 5. evaluate true objective every few hundred iterations ----
        if iteration % 200 == 0:
            conv_now = convolve_fft(w, w)
            score = float(2.0 * target_len * np.max(conv_now))
            if score < best_score - 1e-9:
                best_score = score
                best_w = w.copy()
                no_improve = 0
                print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
            else:
                no_improve += 1

        # ---- 6. random block perturbation (more frequent) ---------------
        if iteration % 800 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            conv_cand = convolve_fft(cand, cand)
            cand_score = float(2.0 * target_len * np.max(conv_cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # ---- 7. increase the smoothing temperature more aggressively ----
        if iteration % 500 == 0 and iteration > 0:
            alpha = min(alpha * 1.07, alpha_max)

        # ---- 8. restart if stuck -----------------------------------------
        if no_improve > 15000:
            # jump to a fresh random point near the current best
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant; we adopt the
    # conventional √(2 n) L¹‑norm for readability.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * target_len)
    final_seq = best_w * target_sum / np.sum(best_w)

    # Compute the official score (the harness will also do this, but we
    # print it for debugging).
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={target_len}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Return plain Python list as required.
    return [float(x) for x in final_seq]
