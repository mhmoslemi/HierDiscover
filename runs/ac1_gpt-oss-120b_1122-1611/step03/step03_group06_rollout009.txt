def run_descent(w, n, start_time, max_seconds, rng):
    # initialize optimizer variables
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 0
    best_w = w.copy()
    # compute initial best_val
    b = convolve_fft(w,w)
    best_val = np.max(b)
    # schedule beta (softmax smoothing)
    beta = 5.0
    beta_max = 5000.0
    beta_inc = 1.05  # multiply each iteration or every N
    beta_inc_interval = 1000

    while time.monotonic() - start_time < max_seconds:
        # compute convolution
        b = convolve_fft(w, w)
        max_b = np.max(b)
        if max_b < best_val - 1e-9:
            best_val = max_b
            best_w = w.copy()
        # softmax weights
        exp_arg = beta * (b - max_b)
        # The shape of exp_arg is 2n-1. For large negative, exp may underflow; safe.
        p = np.exp(exp_arg)
        sum_p = np.sum(p)
        if sum_p == 0.0:
            p = np.full_like(b, 1.0 / b.size)
            sum_p = 1.0
        p = p / sum_p
        # gradient
        grad = 2.0 * convolve_fft(p, w)[:n]
        # enforce symmetry on gradient (maybe)
        grad = symmetrize(grad)
        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        # step size schedule: base_alpha maybe 0.5
        alpha = 0.5 / (np.sqrt(iteration) + 1e-8)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step
        # ensure nonnegative
        w_new = np.maximum(w_new, 0.0)
        # project onto simplex
        w_new = simplex_projection(w_new)
        # symmetrize after projection
        w = symmetrize(w_new)

        # increase beta
        if iteration % beta_inc_interval == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

        # occasional perturbation to escape stagnation
        if iteration % 5000 == 0:
            # a random perturbation based on current best
            if rng.random() < 0.5:
                w = block_perturbation(best_w, rng)
                w = w / w.sum()
                m.fill(0.0); v.fill(0.0)
                # symmetrize
                w = symmetrize(w)
                print(f"[restart] iter={iteration}, time={time.monotonic()-start_time:.1f}s")
        # optional break if max_seconds reached (already loop condition)
    return best_w, best_val

"""
Propose a non‑negative step function with a low max self‑convolution value.

The algorithm consists of three stages:

1. **Initialisation** – draw several diverse starting vectors (Catalan,
   power‑law, exponential, random or the previously found best).  Each is
   normalised to lie on the probability simplex and symmetrised.

2. **Main optimisation** – projected Adam on a *soft‑max* surrogate of the
   max‑convolution.  The smoothing parameter β grows from 5 up to several
   thousand, turning the smooth objective into the true max.  After each
   update the vector is projected onto the simplex and forced to stay
   symmetric.  Random block perturbations occasionally jump out of plateaus.

3. **Focused refinement** – after the soft‑max phase a deterministic
   sub‑gradient descent on the exact maximum index (β → ∞) is performed.
   This further shrinks the peak of the self‑convolution.

All runs respect the overall time budget (`budget_s`).  The best sequence
found over all restarts is finally scaled to the required ℓ¹‑norm
√(2 n) and returned as a plain Python list.  The final sequence is also
stored in the global variable `height_sequence_1` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top level, no closures.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan‑type coefficients a_i = C(2i,i)/4**i, i=0,…,length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for large vectors)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off errors
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) perturbation: pick a random contiguous block and multiply it
    by a log‑normal factor; optionally symmetrise the whole vector.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact symmetry: v ← (v + v[::-1]) / 2 (preserves the ℓ¹‑sum)."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def run_main_optimisation(w0: np.ndarray,
                          budget_seconds: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max surrogate of max_k (w*w)[k].
    Returns the best vector seen and its associated max‑convolution value.
    """
    n = w0.size
    w = w0.copy()
    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    iteration = 0
    best_w = w.copy()
    best_val = np.max(convolve_fft(w, w))

    # soft‑max schedule
    beta = 5.0
    beta_max = 2000.0
    beta_inc_interval = 500
    beta_inc_factor = 1.05

    start = time.monotonic()
    while time.monotonic() - start < budget_seconds:
        # --- 1) self‑convolution and current max ---
        b = convolve_fft(w, w)
        max_b = np.max(b)
        if max_b < best_val - 1e-12:
            best_val = max_b
            best_w = w.copy()
            print(f"[improve] iter={iteration:6d}, max={best_val:.6f}")

        # --- 2) soft‑max weights p ---
        exp_arg = beta * (b - max_b)                 # ≤ 0
        p = np.exp(exp_arg)
        sum_p = np.sum(p)
        if sum_p == 0.0:                             # numerical safety
            p = np.full_like(b, 1.0 / b.size)
            sum_p = 1.0
        p /= sum_p

        # --- 3) gradient of the surrogate ---
        grad = 2.0 * convolve_fft(p, w)[:n]
        grad = symmetrize(grad)

        # --- 4) Adam step + projection ---
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        # step size slowly decreasing
        alpha = 0.4 / (np.sqrt(iteration) + 1.0)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w_new = np.maximum(w_new, 0.0)
        w = simplex_projection(w_new)
        w = symmetrize(w)

        # --- 5) schedule β ---
        if iteration % beta_inc_interval == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_inc_factor)

        # --- 6) occasional random jump ---
        if iteration % 4000 == 0 and rng.random() < 0.3:
            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            m.fill(0.0)
            v.fill(0.0)
            w = symmetrize(w)
            print(f"[jump] iter={iteration:6d}")

    return best_w, best_val


def run_focused_refinement(w0: np.ndarray,
                           budget_seconds: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max index (β → ∞).  A few thousand
    cheap iterations are enough to push the peak down a little further.
    """
    n = w0.size
    w = w0.copy()
    best_w = w.copy()
    best_val = np.max(convolve_fft(w, w))

    start = time.monotonic()
    iteration = 0
    while time.monotonic() - start < budget_seconds:
        b = convolve_fft(w, w)
        max_b = np.max(b)
        k_max = int(np.argmax(b))

        if max_b < best_val - 1e-12:
            best_val = max_b
            best_w = w.copy()
            print(f"[refine] iter={iteration:6d}, max={best_val:.6f}")

        # gradient for the exact max: 2 * shift of w by k_max
        # we implement the shift without an FFT for speed
        grad = np.zeros_like(w)
        if k_max < n:  # part of the shift fits without wrap‑around
            grad[k_max:] = w[: n - k_max]
        else:          # tail part of the shift (when k_max ≥ n)
            grad[: 2 * n - 1 - k_max] = w[k_max - n + 1 :]

        grad = symmetrize(grad)

        # simple diminishing step size
        step_sz = 0.15 / (np.sqrt(iteration + 1) + 1.0)
        w_new = w - step_sz * grad
        w_new = np.maximum(w_new, 0.0)
        w = simplex_projection(w_new)
        w = symmetrize(w)

        iteration += 1

    return best_w, best_val
