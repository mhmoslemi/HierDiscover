"""
Hybrid Adam / Sub‑gradient optimizer for the self‑convolution minimisation problem.

The algorithm works in two phases.
* Phase 1 (smooth) – minimise a soft‑max surrogate of the peak
  ‑‑L_β(w)= (1/β)·log Σ_k exp(β·(w*w)[k]) – using Adam on the probability
  simplex.  β is increased from a small value to a large one (≈200) so the
  landscape gradually concentrates on the true max.
* Phase 2 (exact) – once β has reached its final value the soft‑max is
  replaced by the exact sub‑gradient of the max‑convolution (the gradient of
  the peak).  Adam continues with a decreasing step‑size.
* Exploration – every few thousand iterations a cheap block perturbation of
  the best solution seen so far is tried and accepted if it improves the peak.
  Long periods without improvement trigger a reset to a perturbed copy of the
  best solution.

All vectors are kept on the probability simplex (non‑negative, Σ=1);
the final sequence is scaled to the conventional √(2 n) ℓ¹‑norm so that the
value returned by ``evaluate_sequence`` equals the achieved convolution peak.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random contiguous‑block multiplicative perturbation."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)            # ≈0.5 %
    hi = max(2, n // 12)             # ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ~±12 %
    cand[start:stop] *= factor
    if rng.random() < 0.35:          # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def arcsine_sequence(length: int) -> np.ndarray:
    """
    Discrete approximation of the arcsine density:
    a_i ∝ 1 / sqrt(x_i·(1−x_i)),  x_i = (i+0.5)/length.
    """
    idx = np.arange(length, dtype=np.float64)
    x = (idx + 0.5) / length
    # avoid division by zero at the ends
    eps = 1e-12
    a = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    return a


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution w*w using an FFT of length N."""
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # kill tiny negatives
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate L_β at point w.
    ``probs`` must be a non‑negative array of length 2·len(w)−1 that sums to 1.
    Returns a gradient vector of length len(w).
    """
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    # correlation (w ⋆ probs)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = probs.size
    offset = m - 1
    grad = 2.0 * corr_full[offset: offset + w.size]
    return grad


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function with a small self‑convolution peak.
    Returns a plain Python list of floats suitable for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size and an initial probability vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.ndim != 1 or init.size < 2:
                raise ValueError
            n = int(init.size)
            w = np.maximum(init, 0.0)
            if np.sum(w) < 1e-12:
                raise ValueError
            w = w / np.sum(w)
        except Exception:
            # fallback to a generic arcsine start
            n = 10000
            w = arcsine_sequence(n)
            w = w / np.sum(w)
    else:
        n = 10000
        w = arcsine_sequence(n)
        w = w / np.sum(w)

    # FFT lengths (next powers of two)
    N_self = _next_pow2(2 * n - 1)          # for self‑convolution
    N_corr = _next_pow2(3 * n - 2)          # for correlation (gradient)

    # --------------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------------
    best_w = w.copy()
    b = self_convolve_fft(w, N_self)
    best_score = float(np.max(b))          # this equals the final evaluation
    print(f"[init] n={n}, initial max‑conv={best_score:.8f}")

    # Adam hyper‑parameters
    lr0 = 0.07
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # β‑homotopy (soft‑max temperature)
    beta_start = 2.0
    beta_end = 200.0
    beta_ramp_iters = 300000               # linear increase up to this iter

    # bookkeeping
    start_time = time.monotonic()
    iteration = 0
    no_improve = 0
    last_best_iter = 0
    print_interval = 5000

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # 1️⃣  current self‑convolution and its peak
        b = self_convolve_fft(w, N_self)
        cur_max = float(np.max(b))

        # keep best ever seen
        if cur_max < best_score - 1e-14:
            best_score = cur_max
            best_w = w.copy()
            no_improve = 0
            last_best_iter = iteration
            print(f"[improve] iter={iteration:7d}, max‑conv={best_score:.8f}")
        else:
            no_improve += 1

        # 2️⃣  β schedule
        if iteration < beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * (iteration / beta_ramp_iters)
        else:
            beta = beta_end

        # 3️⃣  gradient (smooth → exact)
        if iteration < beta_ramp_iters:
            # smooth soft‑max weights
            shift = beta * (b - cur_max)               # ≤0, stable
            exp_shift = np.exp(shift)                 # max weight == 1
            probs = exp_shift / np.sum(exp_shift)     # length 2n‑1, sum 1
        else:
            # exact sub‑gradient: binary mask on the peak(s)
            tol = 1e-12
            mask = (b >= cur_max - tol).astype(np.float64)
            mask_sum = np.sum(mask)
            if mask_sum == 0:
                probs = np.full_like(b, 1.0 / b.size)
            else:
                probs = mask / mask_sum

        grad = gradient_smooth(w, probs, N_corr)

        # 4️⃣  Adam update with a simple annealing of the step size
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # decreasing effective lr (∝ 1/√iter)
        lr = lr0 / np.sqrt(iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w = simplex_projection(w_new)

        # 5️⃣  occasional random block jump
        if iteration % 5000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_b = self_convolve_fft(cand, N_self)
            cand_max = float(np.max(cand_b))
            if cand_max < best_score:
                best_w = cand.copy()
                best_score = cand_max
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best max‑conv={best_score:.8f}")

        # 6️⃣  reset if we have been stagnant for a while
        if no_improve > 25000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        # periodic logging
        if iteration % print_interval == 0:
            elapsed = time.monotonic() - start_time
            print(f"[status] iter={iteration:7d}, cur_max={cur_max:.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.1f}s")

    # --------------------------------------------------------------
    # Final scaling – transform the best probability vector to the
    # conventional √(2 n) ℓ¹‑norm.  The evaluation value then equals
    # 2 n·max_b, i.e. the stored ``best_score`` after scaling.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum already ≈1

    # Optional sanity check (cheap – does not affect the return value)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best max‑conv={best_score:.8f}, "
              f"evaluate={final_score:.8f}")
    except Exception:
        pass

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
