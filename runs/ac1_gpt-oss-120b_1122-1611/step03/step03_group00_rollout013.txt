def _conv_general(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    FA = np.fft.rfft(a, n=fft_len)
    FB = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(FA * FB, n=fft_len)
    return conv[: a.size + b.size - 1]

"""
propose_candidate
-----------------

A two‑phase optimiser for the “max‑self‑convolution” problem.

Phase 1 – smooth mirror‑descent:
   * Uses a full soft‑max weighting of *all* convolution entries
     (no arbitrary “top‑K” truncation).  The gradient is obtained
     efficiently as a correlation via FFT.
   * The temperature (β) and steps‑size (η) are linearly annealed.
   * Momentum and occasional random block jitter keep the dynamics
     from stalling.

Phase 2 – simple simulated‑annealing hill‑climber:
   * Random multiplicative perturbations of the whole vector are
     proposed (log‑normal noise) and accepted if they improve the
     score, otherwise with a temperature‑controlled Metropolis
     probability.
   * The second phase runs on the remaining time budget, starting
     from the best vector found by phase 1.

Both phases respect the non‑negativity, clipping and normalisation
rules of ``evaluate_sequence``.  The routine works for any starting
vector supplied through the global variable ``height_sequence_1``,
otherwise it creates a fresh uniform distribution (default length
≈ 4000).  The best sequence encountered is returned as a plain
Python list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------------
# Helper FFT utilities
# ------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold a linear convolution
    of a vector of length ``n`` with another vector of length ``n``.
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution ``a*a`` using a real FFT.
    Returns an array of length ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_general(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two arbitrary real vectors ``x`` and ``y``
    using an FFT of length ``fft_len``.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[: x.size + y.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ------------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise

        score = 2 * n * max(conv(a, a)) / (sum(a)**2)

    within the allotted wall‑clock budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose problem size and initial vector
    # ------------------------------------------------------------------
    default_n = 4000                     # a safe default that fits in the budget
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                n = cand.size
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    a = np.full(n, 1.0 / n)
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n)
    else:
        n = default_n
        a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _fft_len_for_n(n)                # for a*a
    fft_len_grad = 1 << ((3 * n - 2 - 1).bit_length())   # for soft‑max weight * rev(a)

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * np.max(_conv_self(best, fft_len_self)) / (np.sum(best) ** 2)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Phase‑1 – smooth mirror‑descent (full soft‑max)
    # ------------------------------------------------------------------
    eta_start, eta_end = 0.9, 0.02
    beta_start, beta_end = 0.5, 150.0
    momentum = 0.85
    grad_prev = np.zeros_like(a)
    block_perturb_prob = 0.04
    restart_gap = 30.0                # seconds without progress → soft restart

    while time.monotonic() < deadline:
        # ----- current convolution & score --------------------------------
        conv = _conv_self(a, fft_len_self)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)   # sum(a) ≈ 1 after normalisation

        # ----- keep global best -------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedule ----------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ----- soft‑max weights (full convolution) ------------------------
        max_c = cur_max
        # Subtract max for numerical stability
        w_raw = np.exp(beta * (conv - max_c))
        w_norm = w_raw / w_raw.sum()

        # ----- gradient = correlation (w * reverse(a)) --------------------
        rev_a = a[::-1]
        grad_full = _conv_general(w_norm, rev_a, fft_len_grad)
        grad = grad_full[n - 1 : n - 1 + n]               # slice of length n

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- mirror‑descent (exponentiated‑gradient) update -------------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block jitter ------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- soft restart when stuck ------------------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            # small multiplicative jitter to escape a shallow basin
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

        # -----------------------------------------------------------------
        # Exit to phase‑2 once we have spent about 75 % of the budget
        # -----------------------------------------------------------------
        if now - start_time > 0.75 * (deadline - start_time):
            break

    # ------------------------------------------------------------------
    # 5) Phase‑2 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    # Simple multiplicative noise proposal, Metropolis acceptance.
    # Temperature decays linearly to a small value.
    sa_start_temp = 0.3
    sa_end_temp = 0.001
    cur_a = a.copy()
    cur_score = 2.0 * n * np.max(_conv_self(cur_a, fft_len_self)) / (np.sum(cur_a) ** 2)

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear temperature schedule
        T = sa_start_temp * (1.0 - elapsed / total) + sa_end_temp * (elapsed / total)

        # propose: global log‑normal noise
        proposal = cur_a * np.exp(rng.normal(0.0, 0.12, size=n))
        proposal = np.clip(proposal, 0.0, None)
        proposal_sum = proposal.sum()
        if proposal_sum == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= proposal_sum

        # evaluate
        prop_conv = _conv_self(proposal, fft_len_self)
        prop_score = 2.0 * n * np.max(prop_conv) / (np.sum(proposal) ** 2)

        # Metropolis acceptance
        if prop_score < cur_score or rng.random() < np.exp(-(prop_score - cur_score) / max(T, 1e-12)):
            cur_a = proposal
            cur_score = prop_score

            if cur_score < best_score:
                best_score = cur_score
                best = cur_a.copy()
                last_improve = time.monotonic()
                elapsed_best = last_improve - start_time
                print(f"[{elapsed_best:6.2f}s] SA improvement → best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best vector (clip to limits enforced by ``evaluate_sequence``)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
