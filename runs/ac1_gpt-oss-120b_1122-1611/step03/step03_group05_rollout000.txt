"""
propose_candidate
-----------------

A two‑phase projected Adam optimiser for the max‑self‑convolution problem.
Phase 1 minimises a smooth soft‑max surrogate of the peak (β is increased
gradually up to a large value, ≈200).  
Phase 2 refines the solution using the exact sub‑gradient (the soft‑max
behaviour with a huge β already mimics it).  The iterates stay on the
probability simplex (∑w = 1, w ≥ 0) by Euclidean projection.
Occasional “block perturbations’’ (multiplying a random contiguous block
by a small random factor) provide stochastic jumps that help escape
local plateaus.  The algorithm runs until the time budget expires and
returns the best probability vector found, rescaled to the conventional
ℓ¹‑norm √(2 n) required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """
    Linear self‑convolution w*w using an FFT of length N.
    Returns the full (2*len(w)-1) result.
    """
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)  # eliminate tiny negatives
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth max‑approximation L_β at point w.
    ``probs`` are the soft‑max weights (sum 1, length 2*len(w)‑1).
    Returns a vector of length len(w).
    """
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = probs.size
    offset = m - 1
    grad = 2.0 * corr_full[offset : offset + w.size]
    return grad


def block_perturbation(base: np.ndarray,
                       rng: np.random.Generator,
                       sigma: float = 0.12) -> np.ndarray:
    """
    Cheap O(n) random perturbation used as a jump.
    A random contiguous block is multiplied by exp(N(0, sigma)).
    An optional symmetric averaging is applied with probability 0.35.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # ≈0.5 %
    hi = max(2, n // 12)           # up to ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point for the competition harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a small self‑convolution peak.
    The function is evaluated by ``evaluate_sequence`` which expects the
    ℓ¹‑norm to be √(2 n).  The optimiser works on the probability simplex
    (Σw = 1) and finally rescales the best vector found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose problem size and an initial probability vector.
    # --------------------------------------------------------------
    try:
        # If a previous solution is present, start from it.
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        if init_seq.ndim == 1 and init_seq.size >= 2 and np.sum(np.abs(init_seq)) > 0:
            n = int(init_seq.size)
            w = np.maximum(init_seq, 0.0)
            w = w / np.sum(w)
        else:
            raise ValueError
    except Exception:
        # No previous solution – fall back to a smooth sine‑squared shape.
        n = 8000
        i = np.arange(n, dtype=np.float64)
        w = np.sin(np.pi * (i + 0.5) / n)        # positive on (0, π)
        w = w ** 2                               # make it smoother and strictly positive
        w = np.maximum(w, 0.0)
        w = w / np.sum(w)                        # probability simplex

    # FFT lengths (powers of two for speed)
    N_self = next_pow2(2 * n - 1)          # for self‑convolution
    N_corr = next_pow2(3 * n - 2)          # for correlation (gradient)

    # --------------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------------
    best_w = w.copy()
    best_score = np.max(self_convolve_fft(w, N_self))   # this is the unscaled peak
    print(f"[init] n={n}, initial peak={best_score:.8f}")

    # Adam hyper‑parameters
    lr = 0.07
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # Soft‑max temperature schedule
    beta_start = 2.0
    beta_end = 200.0                 # large β ≈ exact max
    beta_ramp_iters = 150_000        # linear increase

    # Stagnation handling
    no_improve = 0
    last_best_iter = 0
    iter_cnt = 0

    start_time = time.monotonic()
    budget = float(budget_s)

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, budget - 5.0):
        # ----- current peak -----
        conv_full = self_convolve_fft(w, N_self)
        cur_max = conv_full.max()

        # keep the best ever seen
        if cur_max < best_score - 1e-14:
            best_score = cur_max
            best_w = w.copy()
            no_improve = 0
            last_best_iter = iter_cnt
            print(f"[improve] iter={iter_cnt:07d}, peak={best_score:.8f}")
        else:
            no_improve += 1

        # ----- β for smooth surrogate (homotopy) -----
        if iter_cnt < beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * (iter_cnt / beta_ramp_iters)
        else:
            beta = beta_end

        # ----- soft‑max weights over convolution entries -----
        shift = beta * (conv_full - cur_max)            # ≤ 0, max entry → 0
        # subtract the maximal shift (which is 0) for numerical safety
        exp_shift = np.exp(shift)                       # very small for non‑max entries
        probs = exp_shift / np.sum(exp_shift)           # length 2n‑1, sums to 1

        # ----- gradient of the smooth loss -----
        grad = gradient_smooth(w, probs, N_corr)

        # ----- Adam update with projection onto the simplex -----
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step
        w = simplex_projection(w_new)

        # ----- occasional stochastic jump -----
        if iter_cnt % 10_000 == 0:
            cand = block_perturbation(best_w, rng, sigma=0.09)
            cand = cand / np.sum(cand)
            cand_max = self_convolve_fft(cand, N_self).max()
            if cand_max < best_score:
                best_w = cand.copy()
                best_score = cand_max
                w = cand.copy()
                print(f"[block] iter={iter_cnt:07d}, new best peak={best_score:.8f}")

        # ----- very occasional reset when stuck for a long time -----
        # (kept deliberately infrequent to allow fine convergence)
        if no_improve > 200_000:
            w = block_perturbation(best_w, rng, sigma=0.12)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iter_cnt:07d}, reset to perturbed best")

    # --------------------------------------------------------------
    # Final scaling – convert the best probability vector to the required
    # ℓ¹‑norm √(2 n).  The evaluation value then equals the stored peak.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # guard against tiny drift

    # Optional sanity check (cheap)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))  # type: ignore
        print(f"[final] n={n}, best peak (prob‑simplex)={best_score:.8f}, "
              f"evaluate={final_score:.8f}")
    except Exception:
        pass

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
