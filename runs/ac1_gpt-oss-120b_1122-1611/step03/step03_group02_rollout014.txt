def adam_optimize(w, n_iters, eta=0.1, beta1=0.9, beta2=0.999, eps=1e-8, mu_factor=0.5, mu_decay=0.9999):
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    for t in range(1, n_iters+1):
        # compute conv
        b = convolve_fft(w,w)
        maxb = np.max(b)
        # compute smooth max softmax distribution p
        mu = maxb * mu_factor
        shifted = (b - maxb) / mu
        exp_term = np.exp(shifted - np.max(shifted)) # for numerical stability
        p = exp_term / np.sum(exp_term)
        grad = 2.0 * convolve_fft(p, w[::-1])[n-1:n-1+n]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        w = w - eta * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        w = simplex_projection(w)
        # decay mu factor
        mu_factor = max(mu_factor*mu_decay, 0.02)
        # optional early stop if time
    return w

"""
Optimization of self‑convolution peak via simple parametric families.
We generate a set of candidate step‑functions (Gaussian, Laplace,
Beta‑like) for several lengths, evaluate the L∞ convolution
peak, keep the best, then perform a short Adam‑style refinement.
The final sequence is rescaled to the conventional √(2 n) ℓ1‑norm.
"""

import time
import numpy as np

def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n up to a few 10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny numerical negatives that may appear after FFT round‑off
    np.maximum(c, 0.0, out=c)
    return c

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal – return the uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric centred Gaussian on a discrete grid."""
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    return np.exp(-((x - mu) ** 2) / (2.0 * sigma ** 2))

def generate_laplace(n: int, scale: float) -> np.ndarray:
    """Symmetric centred Laplace distribution."""
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    return np.exp(-np.abs(x - mu) / scale)

def generate_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    """Beta‑type shape a_i = (i+1)^{α‑1} (n‑i)^{β‑1} (non‑negative)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta_ - 1.0))

def generate_power(n: int, alpha: float) -> np.ndarray:
    """Power‑law decreasing shape a_i = (i+1)^{-α}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-alpha)

def compute_score(seq: np.ndarray) -> float:
    """Return the value that evaluate_sequence would return for `seq`."""
    n = seq.size
    conv = convolve_fft(seq, seq)
    maxb = np.max(conv)
    # After scaling to √(2 n) the denominator becomes 2 n,
    # therefore the score equals maxb.
    # We keep the exact expression to avoid any rounding surprise.
    s = np.sum(seq)
    return float(2.0 * n * maxb / (s * s))

def refine_adam(w: np.ndarray,
                n_iters: int,
                eta0: float = 0.5,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                mu_factor: float = 0.5,
                mu_decay: float = 0.9999,
                time_limit: float = None) -> np.ndarray:
    """
    Adam‑style optimisation of the smooth max of the self‑convolution.
    The vector w is assumed to lie on the probability simplex.
    """
    n = w.size
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    start = time.monotonic()
    for t in range(1, n_iters + 1):
        if time_limit is not None and time.monotonic() - start > time_limit:
            break
        # Convolution and smooth‑max weights
        conv = convolve_fft(w, w)
        maxc = np.max(conv)
        mu = maxc * mu_factor
        if mu <= 0.0:
            mu = 1e-12
        # Prevent overflow – shift by the maximum value before exponentiation
        shifted = (conv - maxc) / mu
        exp_term = np.exp(shifted - np.max(shifted))
        p = exp_term / np.sum(exp_term)
        # Gradient of the smooth max (see description)
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # Adam update on the simplex – we perform the update in the
        # ambient space and project back onto the simplex afterwards.
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        w = w - step
        w = simplex_projection(w)

        # Decay the smoothing parameter slowly
        mu_factor = max(mu_factor * mu_decay, 0.01)

    return w

    """

    `evaluate_sequence` score is expected to be < 1.5025.
    The routine tries a handful of parametric families (Gaussian,
    Laplace, Beta, power‑law) for several lengths, keeps the best,
    and finally runs a short Adam optimisation.
    """

    start_time = time.monotonic()
    best_seq = None
    best_score = np.inf

    # 1.  Try a few lengths.  We keep the total work modest – the
    #     convolution of a 8000‑point vector with itself costs < 2 ms.
    # --------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    # --------------------------------------------------------------
    # 2.  For each length generate a few candidates from the families.
    # --------------------------------------------------------------
    for n in lengths:
        target_sum = np.sqrt(2.0 * n)

        # Gaussian family
        for _ in range(25):
            if time.monotonic() - start_time > budget_s - 5:
                break

            seq = generate_laplace(n, scale)
            seq_sum = np.sum(seq)
            if seq_sum <= 0.0:
                continue
            seq = seq * (target_sum / seq_sum)
            score = compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[laplace] n={n}, scale={scale:.1f}, score={score:.6f}")

        # Beta‑type family (both parameters drawn from a log‑uniform range)
        for _ in range(20):

                print(f"[power] n={n}, alpha={alpha:.3f}, score={score:.6f}")

        # Early exit if we already exceed the target by comfortable margin
        if best_score < 1.30:   # we have already a comfortable win
            break

    # --------------------------------------------------------------
    # 3.  If a sequence was found, run a short Adam refinement on the
    #     simplex.  The refinement works directly on the probability
    #     vector – we rescale only at the very end.
    # --------------------------------------------------------------
    if best_seq is not None:

        remaining = max(0.0, budget_s - (time.monotonic() - start_time) - 2.0)
        # Allocate at most ~200 000 Adam steps (about a few seconds per 10 000‑pt

            refined_score = compute_score(refined_seq)

    #     and has a reasonable size.  The harness will cap values at

        best_seq = np.array([np.sqrt(2.0), np.sqrt(2.0)], dtype=np.float64)

    # --------------------------------------------------------------

    print(f"[final] length={best_seq.size}, best score={best_score:.6f}, final score={final_score:.6f}")

"""
Optimization of the L∞ norm of a self‑convolution using simple
parametric families.

The routine builds candidates from a few low‑dimensional families
(Gaussian, Laplace, Beta‑type, power‑law), evaluates them with a fast
FFT‑based convolution, keeps the best and finally refines the best
candidate with a short Adam‑style optimisation on the probability
simplex.  The returned sequence is rescaled to the conventional
√(2 n) ℓ₁‑norm, which makes the value returned by `evaluate_sequence`
equal to the maximal convolution entry.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required by the harness)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n up to a few 10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negative values that can appear because of rounding.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric centred Gaussian on a discrete grid."""
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    return np.exp(-((x - mu) ** 2) / (2.0 * sigma ** 2))


def generate_laplace(n: int, scale: float) -> np.ndarray:
    """Symmetric centred Laplace distribution."""
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    return np.exp(-np.abs(x - mu) / scale)


def generate_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    """
    Beta‑type shape a_i = (i+1)^{α‑1}·(n‑i)^{β‑1}.
    Both parameters must be > 0.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta_ - 1.0))


def generate_power(n: int, alpha: float) -> np.ndarray:
    """Power‑law decreasing shape a_i = (i+1)^{‑α}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-alpha)


def compute_score(seq: np.ndarray) -> float:
    """Score identical to what `evaluate_sequence` would return."""
    n = seq.size
    conv = convolve_fft(seq, seq)
    maxb = np.max(conv)
    s = np.sum(seq)
    # The denominator 2·n equals s² after our normalisation,
    # but we keep the exact formula for safety.
    return float(2.0 * n * maxb / (s * s))


def refine_adam(w: np.ndarray,
                n_iters: int,
                eta0: float = 0.6,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8,
                mu_factor: float = 0.5,
                mu_decay: float = 0.9998,
                time_limit: float = None) -> np.ndarray:
    """
    Adam‑style optimisation of the smooth‑max of the self‑convolution.
    The vector w lives on the probability simplex (Σw = 1, w ≥ 0).
    """
    n = w.size
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    start = time.monotonic()
    for t in range(1, n_iters + 1):
        if time_limit is not None and time.monotonic() - start > time_limit:
            break

        conv = convolve_fft(w, w)
        maxc = np.max(conv)
        mu = maxc * mu_factor
        if mu <= 0.0:
            mu = 1e-12

        # Soft‑max over convolution entries (shifted by the max for stability)
        shifted = (conv - maxc) / mu
        exp_term = np.exp(shifted - np.max(shifted))
        p = exp_term / np.sum(exp_term)

        # Gradient of the smooth max: 2·conv(p, w_rev) (slice of length n)
        grad = 2.0 * convolve_fft(p, w[::-1])[n - 1:n - 1 + n]

        # Adam update in ambient space, followed by projection onto simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        w = w - step
        w = simplex_projection(w)

        # Reduce the smoothing factor slowly – this makes the surrogate
        # problem tighter as optimisation proceeds.
        mu_factor = max(mu_factor * mu_decay, 0.01)

    return w
