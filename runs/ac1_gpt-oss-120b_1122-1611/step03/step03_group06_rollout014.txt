"""
Optimisation of the max‑self‑convolution bound.

The objective (scale invariant) is
        score(w) = 2·n·max_k (w∗w)[k]   ,   w_i ≥ 0 , Σ w_i = 1 .
A lower score gives a better value for ``evaluate_sequence`` after the
usual √(2 n) rescaling.

We use a two‑stage scheme:
  1. optimise a modest sized problem (n≈2000) with Adam on the
     smooth “soft‑max” surrogate of the max.
  2. interpolate the best small‑scale vector to a larger size
     (n≈8000) and continue optimisation, finally polishing the
     solution with an exact‑max gradient step.

The optimiser repeatedly projects onto the probability simplex,
applies occasional random block perturbations to escape plateaus
and increases the soft‑max temperature β so that the surrogate
converges to the true max.  The whole routine respects the 1000 s
budget and returns the best vector found (scaled to the required
√(2 n) L¹‑norm).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan‑type coefficients a_i = C(2i,i)/4**i (≈1/√i)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # eliminate tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation: multiply a random contiguous block
    by a log‑normal factor and optionally symmetrise.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 250)
    hi = max(2, n // 10)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.15)))
    cand[start:stop] *= factor
    if rng.random() < 0.30:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def upscale_vector(v_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    x_small = np.linspace(0.0, 1.0, v_small.size)
    x_large = np.linspace(0.0, 1.0, n_large)
    v_large = np.interp(x_large, x_small, v_small)
    v_large = np.maximum(v_large, 0.0)
    if v_large.sum() == 0.0:
        v_large = np.full_like(v_large, 1.0 / n_large)
    else:
        v_large = v_large / v_large.sum()
    return v_large


def score_of(w: np.ndarray) -> float:
    """Return the scale‑invariant objective 2·n·max(w∗w)."""
    n = w.size
    b = convolve_fft(w, w)
    return 2.0 * n * np.max(b)


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a non‑negative step function (list of floats) whose
    `evaluate_sequence` value is as small as possible.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Phase 0 – pick a starting vector.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            # down‑sample to a convenient size for the cheap phase
            w_small = upscale_vector(seq0 / np.sum(seq0), 2000)
        except Exception:                     # fall back to Catalan
            w_small = catalan_sequence(2000)
            w_small = w_small / np.sum(w_small)
    else:
        w_small = catalan_sequence(2000)
        w_small = w_small / np.sum(w_small)

    # ------------------------------------------------------------------
    # Optimisation parameters (shared between phases)
    # ------------------------------------------------------------------
    beta = 1.0                     # soft‑max temperature
    beta_max = 800.0
    beta_growth = 1.025
    beta_inc_interval = 2500

    # Adam hyper‑parameters
    eta0 = 0.15                    # base step size
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # ------------------------------------------------------------------
    # Helper to run a (soft‑max) Adam optimisation on a given vector.
    # ------------------------------------------------------------------
    def optimise_vector(w: np.ndarray, time_remaining: float) -> tuple[np.ndarray, float]:
        n = w.size
        best_w = w.copy()
        best_score = score_of(w)

        m = np.zeros_like(w)
        v = np.zeros_like(w)
        it = 0
        no_improve = 0

        start = time.monotonic()
        while True:
            if time.monotonic() - start > time_remaining:
                break

            # convolution and soft‑max weights
            b = convolve_fft(w, w)
            max_b = np.max(b)
            cur_score = 2.0 * n * max_b
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_w = w.copy()
                no_improve = 0
                print(f"[improve] n={n:5d} it={it:7d} score={best_score:.6f}")
            else:
                no_improve += 1

            # soft‑max distribution over convolution entries
            exp_arg = beta * (b - max_b)        # ≤ 0, stabilises exponentials
            p = np.exp(exp_arg)
            sum_p = np.sum(p)
            if sum_p == 0.0:                    # underflow guard
                p = np.full_like(b, 1.0 / b.size)
                sum_p = 1.0
            p /= sum_p

            # gradient of the smoothed objective
            grad = 2.0 * convolve_fft(p, w)[:n]

            # Adam update (learning‑rate decays with √it)
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = (eta0 / np.sqrt(it + 1.0)) * m_hat / (np.sqrt(v_hat) + eps)

            w = simplex_projection(w - step)

            # occasional random block jump to shake the system
            if it % 6000 == 0 and it > 0:
                w = block_perturbation(best_w, rng)
                w = w / np.sum(w)
                m.fill(0.0)
                v.fill(0.0)
                print(f"[jump] n={n:5d} it={it:7d}")

            # schedule for the soft‑max temperature
            if it % beta_inc_interval == 0 and beta < beta_max:
                beta = min(beta_max, beta * beta_growth)

            # long stagnation → reset from current best
            if no_improve > 50000:
                w = block_perturbation(best_w, rng)
                w = w / np.sum(w)
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[reset] n={n:5d} it={it:7d}")

        return best_w, best_score

    # ------------------------------------------------------------------
    # Phase 1 – cheap optimisation on a small problem (≈2 k entries).
    # ------------------------------------------------------------------
    start = time.monotonic()
    time_allowed = max(0.0, budget_s * 0.20 - 2.0)   # 20 % of budget, keep headroom
    w_small, score_small = optimise_vector(w_small, time_allowed)
    print(f"[phase‑1] finished, score={score_small:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to a larger length and continue optimisation.
    # ------------------------------------------------------------------
    n_large = 8000
    w_large = upscale_vector(w_small, n_large)

    # reset optimiser state for the larger problem
    beta = 1.0
    m = np.zeros_like(w_large)
    v = np.zeros_like(w_large)

    # remaining time (≈75 % of the budget)
    time_used = time.monotonic() - start
    remaining = max(0.0, budget_s - time_used - 5.0)

    w_large, score_large = optimise_vector(w_large, remaining)
    print(f"[phase‑2] finished, score={score_large:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – switch to the exact max gradient (β → ∞) for a
    # short burst.  This often squeezes a little extra improvement.
    # ------------------------------------------------------------------
    n = w_large.size
    start = time.monotonic()
    while time.monotonic() - start < min(5.0, remaining * 0.05):
        b = convolve_fft(w_large, w_large)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b
        if cur_score < score_large - 1e-12:
            score_large = cur_score
            best_w = w_large.copy()
            print(f"[polish] score={score_large:.6f}")

        # index of the (first) maximal entry
        k_max = int(np.argmax(b))

        # exact gradient of (w*w)[k_max]
        grad = np.zeros_like(w_large)
        # contribution w_{k_max-i}
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=int)
        grad[i_range] += w_large[k_max - i_range]
        # contribution w_{i-k_max}
        lo2 = max(0, -k_max)
        hi2 = min(n - 1, n - 1 - k_max)
        j_range = np.arange(lo2, hi2 + 1, dtype=int)
        grad[j_range] += w_large[j_range + k_max] if k_max >= 0 else 0.0
        grad *= 2.0

        # simple step (no momentum) and project
        step = 0.04 / np.sqrt(time.monotonic() - start + 1.0) * grad
        w_large = simplex_projection(w_large - step)

    # ------------------------------------------------------------------
    # Final scaling to the conventional L¹‑norm √(2 n) required by the
    # harness.  The value returned by ``evaluate_sequence`` equals
    # 2·n·max(b) for the normalised vector, i.e. our best scores.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = w_large * target_sum
    final_value = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score={score_large:.6f}, "
          f"evaluate={final_value:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
