def _project_simplex(v):
    """
    Projects a vector onto the probability simplex (non‑negative, sum = 1).
    If the input sum is zero or all entries are non‑positive, returns the
    uniform distribution.
    """
    n = v.shape[0]
    # shift to ensure financing numeric stability
    # Use standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    # due to numerical errors may not sum exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    # renormalize to exactly 1 to avoid drift
    w = w / w_sum
    return w

"""
Propose candidate via subgradient descent on max ‑norm of the self‑convolution.
We treat the score (2·n·max(conv)) as the objective (the sum is fixed to 1).
At each step we
* compute the full linear convolution with an FFT,
* pick the K biggest convolution entries,
* accumulate the sub‑gradient of those max​-terms,
* take a projected gradient step onto the probability simplex,
* occasionally jitter the vector and restart if progress stalls.
We cycle over a few problem sizes (including the supplied height_sequence_1)
and keep the best ever seen.
"""
import time
import numpy as np

def _fft_len_for_n(n: int) -> int:
    """Return smallest power‑of‑two length that can hold a linear convolution of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w

Sub‑gradient descent on the max‑norm of the self‑convolution.

We minimise   score = 2·n·max(conv(a,a))   under a_i≥0 , Σa_i=1.
The objective is the maximum of a family of convex quadratic forms, so a
sub‑gradient is simply the sum of the “partner” vectors that generate the
largest convolution entries.  The algorithm iterates

    * full self‑convolution (FFT, O(n log n)),
    * selection of the K largest entries,
    * accumulation of the corresponding sub‑gradient,
    * a projected gradient step onto the probability simplex,

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of size n."""
    size = 2 * n - 1

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a non‑negative real vector using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

    {x ≥ 0, Σx = 1}.  If the projection would be empty, return the uniform
    distribution.
