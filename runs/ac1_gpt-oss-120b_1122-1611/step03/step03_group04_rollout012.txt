"""
propose_candidate
-----------------

Hybrid search for a low‑peak self‑convolution sequence.

The objective for a probability vector w (∑w=1) is
    score(w) = 2·n·max_k (w*w)[k] .
This is exactly the value returned by ``evaluate_sequence`` after the
usual √(2 n) ℓ¹‑normalisation, hence minimising the maximal entry of the
linear convolution directly minimises the final score.

The algorithm consists of three stages:

1. **Parametric grid search** –  
   For several lengths n we generate symmetric “V‑shaped’’ distributions

        v_i(p) = (i+1)^{-p} + (n-i)^{-p}     (0≤i<n)

   and mix them with a uniform component:
        w = λ·uniform + (1‑λ)·v(p) .
   The two parameters p (shape exponent) and λ (mix with uniform) are
   scanned on a coarse grid.  For each candidate the convolution is
   computed via a real FFT and the raw score 2·n·max(b) is recorded.

2. **Local refinement** –  
   The best candidate from the grid is further optimised with a
   *hard‑max* sub‑gradient descent.  At each iteration the index k of the
   maximal convolution entry is identified, the exact gradient
   (∇max = 2·corr(w,δ_k)) is formed analytically (O(n)), and an
   entropic mirror‑descent update
        w ← w·exp(−η·∇) ,  η ∝ 1/√t
   is applied, followed by projection back onto the simplex and a
   symmetrisation step w←½(w+w[::-1]).  Occasional block‑perturbations
   escape plateaus.

3. **Fallback to the previous best** –  
   If a global variable ``height_sequence_1`` exists it is taken,
   normalised to a probability vector and refined with the same
   sub‑gradient routine; the final answer is the best of the two
   refinement tracks.

The whole search is limited to the supplied time budget (default
≈ 960 s).  The returned list is scaled to the conventional
√(2 n) ℓ¹‑norm; this scaling is immaterial for the score because the
evaluation function is scale‑invariant.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (FFT convolution, block perturbation, gradient)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT, result length = len(a)+len(b)-1."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)            # eliminate tiny negatives
    return c


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random jump: multiply a random block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)      # 0.5 % of length
    hi = max(2, n // 12)       # up to ~8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ≈ 13 % multiplicative noise
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])          # optional symmetric smoothing
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def raw_score(w: np.ndarray) -> float:
    """Score for a probability vector (∑w = 1).  Returns 2·n·max(w*w)."""
    n = w.size
    b = convolve_fft(w, w)
    return float(2.0 * n * np.max(b))


def hard_max_gradient(w: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the hard maximum a*a at index k.
    For linear convolution the gradient is
        g_j = 2·w_{k-j}   if 0 ≤ k-j < n, else 0 .
    Returned as a vector of length n.
    """
    n = w.size
    grad = np.zeros_like(w)
    if k < n:
        # indices 0 … k receive the reversed prefix of w
        grad[: k + 1] = 2.0 * w[k :: -1]
    else:
        # indices k-n+1 … n-1 receive a reversed suffix of w
        start = k - n + 1
        # slice w[n-1 : start-1 : -1] yields w[n-1], …, w[start]
        rev = w[n - 1 : start - 1 : -1]
        grad[start:] = 2.0 * rev
    return grad


# ----------------------------------------------------------------------
# Parametric shape generation
# ----------------------------------------------------------------------


def generate_v_shape(n: int, p: float, lam: float) -> np.ndarray:
    """
    Produce a symmetric “V’’ shape on {0,…,n‑1}.

        base_i = (i+1)^{-p} + (n-i)^{-p}
        w    = λ·uniform + (1‑λ)·base

    The resulting vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    # power‑law backbone (decreasing from ends towards centre)
    base = (i + 1) ** (-p) + (n - i) ** (-p)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    w = lam * uniform + (1.0 - lam) * base
    s = w.sum()
    if s == 0.0 or not np.isfinite(s):
        w = uniform
        s = w.sum()
    w /= s
    return w


# ----------------------------------------------------------------------
# Local optimisation using hard‑max sub‑gradient (mirror descent)
# ----------------------------------------------------------------------


def refine_candidate(
    w0: np.ndarray,
    max_iters: int,
    time_limit: float,
    rng: np.random.Generator,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the hard‑max objective.
    Returns the best vector seen and its raw score.
    """
    w = w0.copy()
    n = w.size
    best_w = w.copy()
    best_sc = raw_score(w)
    start = time.monotonic()
    no_improve = 0
    eta0 = 0.5            # initial step size for entropic update

    for it in range(1, max_iters + 1):
        if time.monotonic() - start > time_limit:
            break

        # 1) current convolution, locate the maximal entry
        b = convolve_fft(w, w)
        k = int(np.argmax(b))

        # 2) exact hard‑max gradient
        grad = hard_max_gradient(w, k)

        # 3) multiplicative (entropic) update + projection onto simplex
        eta = eta0 / np.sqrt(it)
        w = w * np.exp(-eta * grad)
        s = w.sum()
        if s == 0.0 or not np.isfinite(s):
            w = best_w.copy()
            s = w.sum()
        w /= s

        # 4) enforce symmetry (the optimum is symmetric)
        w = 0.5 * (w + w[::-1])
        w /= w.sum()                     # re‑normalise, just in case

        # 5) evaluate
        sc = raw_score(w)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_w = w.copy()
            no_improve = 0
            if verbose:
                print(f"[refine improve] it={it:5d}, score={best_sc:.6f}")
        else:
            no_improve += 1

        # 6) occasional random jump if stuck
        if no_improve > 1200:
            w = block_perturbation(best_w, rng)
            w /= w.sum()
            no_improve = 0
            if verbose:
                print(f"[refine reset] it={it:5d}, reset to perturbed best")

    return best_w, best_sc

                w = generate_v_shape(n, float(p), float(lam))
                sc = raw_score(w)
                if sc < best_sc:
                    best_sc = sc
                    best_w = w.copy()
                    best_n = n
                    # simple progress output
                    print(
                        f"[grid] n={n:5d}, p={p:4.2f}, λ={lam:4.2f}, score={sc:.6f}"
                    )

    # ------------------------------------------------------------------
    # 2) Local refinement of the best grid candidate
    # ------------------------------------------------------------------
    if best_w is None:
        raise RuntimeError("Grid search failed to produce a candidate.")

    # Give the refinement about 70 % of the remaining budget
    remaining = max(0.0, budget_s - (time.monotonic() - overall_start))
    refine_time = remaining * 0.70
    max_iters = 8000

    print(f"[refine] start refinement of grid best (n={best_n})")
    refined_w, refined_sc = refine_candidate(
        best_w, max_iters, refine_time, rng, verbose=True
    )
    if refined_sc < best_sc:
        best_sc = refined_sc
        best_w = refined_w.copy()
        best_n = best_w.size

    # ------------------------------------------------------------------
    # 3) Optional fallback: improve the previously supplied sequence,
    #    if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= 2:
                w_prev = prev / np.sum(prev)                     # normalise to prob.
                # Ensure symmetry (the final optimum is symmetric)
                w_prev = 0.5 * (w_prev + w_prev[::-1])
                w_prev /= w_prev.sum()
                fallback_time = remaining * 0.30
                fallback_iters = 6000
                print("[fallback] refining previous best sequence")
                w_fbk, sc_fbk = refine_candidate(
                    w_prev, fallback_iters, fallback_time, rng, verbose=False
                )
                if sc_fbk < best_sc:
                    best_sc = sc_fbk
                    best_w = w_fbk.copy()
                    best_n = best_w.size
        except Exception as exc:
            print("[fallback] could not use height_sequence_1:", exc)

    # ------------------------------------------------------------------
    # 4) Final scaling to √(2 n) ℓ¹‑norm (the evaluation is scale‑invariant)
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * best_n)
    final_seq = best_w * (target_sum / best_w.sum())

    # sanity check (optional)
    try:
        final_eval = float(evaluate_sequence(final_seq.tolist()))
        print(
            f"[final] length={best_n}, best raw score={best_sc:.6f}, "
            f"evaluate={final_eval:.6f}"
        )
    except Exception:
        print("[final] evaluation failed – will be performed by the harness")

    # Convert to plain Python floats for the harness
    return [float(x) for x in final_seq]
