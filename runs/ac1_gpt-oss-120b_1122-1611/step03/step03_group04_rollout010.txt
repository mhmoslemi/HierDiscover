def symmetric_block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.12))
    # apply to block and symmetric counterpart
    cand[start:stop] *= factor
    # symmetric part:
    sym_start = n - stop
    sym_stop = n - start
    cand[sym_start:sym_stop] *= factor
    # optional smoothing maybe not needed
    np.clip(cand, 0.0, 1000.0, out=cand)
    # enforce symmetry
    cand = 0.5 * (cand + cand[::-1])
    return cand
```

    rng = np.random.default_rng(seed)
    best_score = np.inf
    best_w = None
    for p in np.linspace(1.3, 1.8, 26):  # step 0.02

        w_vec = np.maximum(w_vec, 0.0)
        w_norm = w_vec / np.sum(w_vec)

"""
Propose a low‑∞ norm step‑function.

The algorithm consists of three stages:

1. **Parameter scan** – For the chosen length *n* a few one‑parameter
   symmetric families (powers of the index, binomial‑coefficients, etc.) are
   generated.  The one with the smallest hard‑max of the self‑convolution is
   kept as the starting point.

2. **Entropic mirror descent** –  The objective
   ``f(w)=max_k (w*w)[k]`` under the simplex constraint is convex.
   We minimise a smooth approximation with a soft‑max (log‑sum‑exp) using
   an exponential (multiplicative) update.  A momentum term and a modest
   schedule for the temperature (``β``) accelerate convergence.
   The vector is kept symmetric at all times.

3. **Symmetric perturbations** –  If progress stalls, a cheap symmetric
   block perturbation (multiply one random block on both sides by a
   log‑normal factor) is applied; the candidate is accepted when it improves
   the hard maximum.  This helps escaping plateaus.

The whole search respects the 1000 s time budget, constantly tracking the
best sequence.  The final answer is scaled to the conventional
ℓ¹‑norm ``sqrt(2 n)`` (the evaluation is scale‑invariant) and returned as a
plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Clip tiny negatives caused by round‑off.
    np.maximum(c, 0.0, out=c)
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max:   g = 2·corr(w, α)
    ``α`` has length 2·n‑1 and sums to 1.
    """
    rev = w[::-1]
    conv = convolve_fft(rev, alpha)          # length = 3·n‑2
    start = w.shape[0] - 1
    return 2.0 * conv[start:start + w.shape[0]]


def symmetric_block_perturbation(base: np.ndarray,
                                 rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) symmetric jump: pick a random block and multiply both the block
    and its mirror image by the same log‑normal factor.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # roughly 0.5 %
    hi = max(2, n // 12)           # up to ~8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ≈ 1 ± 13 %
    cand[start:stop] *= factor
    # mirror the block
    sym_start = n - stop
    sym_stop = n - start
    cand[sym_start:sym_stop] *= factor
    # keep within the allowed range (the optimiser never creates >1000)
    np.clip(cand, 0.0, 1000.0, out=cand)
    # enforce exact symmetry
    cand = 0.5 * (cand + cand[::-1])
    return cand


def power_family_sequence(n: int, p: float) -> np.ndarray:
    """
    Symmetric sequence w_i ∝ (i+1)^(-p) + (n-i)^(-p)   (i = 0…n‑1)
    Returns a probability vector (sum = 1).
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p) + (n - idx + 1) ** (-p)
    seq = np.maximum(seq, 0.0)
    seq /= np.sum(seq)
    return seq


def binomial_sequence(n: int) -> np.ndarray:
    """
    a_i = binom(2i,i) / 4**i   (i = 0 … n‑1).  Normalised to sum = 1.
    This is the “Catalan” sequence used in the reference code.
    """
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        # recurrence a_{i} = a_{i‑1} * (2i‑1) / (2i)
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    seq /= np.sum(seq)
    return seq


def scan_power_family(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Scan a modest grid of exponents around p=0.5 (the binomial case) and
    return the vector with the smallest ``max(w*w)``.
    """
    best_score = np.inf
    best_w = None
    best_p = None

    # grid covering a plausible interval; step size 0.02 → 21 candidates.
    for p in np.linspace(0.35, 0.70, 18):
        w = power_family_sequence(n, p)
        b = convolve_fft(w, w)
        score = 2.0 * n * np.max(b)    # corresponds to evaluate_sequence with sum=1
        if score < best_score:
            best_score = score
            best_w = w
            best_p = p

    # also try the exact binomial sequence (p≈0.5) for safety
    w_binom = binomial_sequence(n)
    b = convolve_fft(w_binom, w_binom)
    score = 2.0 * n * np.max(b)
    if score < best_score:
        best_score = score
        best_w = w_binom
        best_p = "binomial"

    print(f"[scan] best exponent p={best_p}, score={best_score:.6f}")
    return best_w


# --------------------------------------------------------------
# Main optimisation routine
# --------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):
    """
    Find a non‑negative step function (list of floats) with a low
    `evaluate_sequence` score inside the allotted CPU budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total wall‑clock budget in seconds.  The function tries to finish
        earlier and returns the best candidate discovered.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------
    # Choose a problem size.
    # ----------------------------------------------------------
    # If the arena already provided a sequence, it is usually already
    # near‑optimal for its length.  We deliberately ignore it and try a
    # larger length, because the asymptotic optimum is approached as n grows.
    if budget_s > 800:
        n = 20000
    elif budget_s > 500:
        n = 14000
    elif budget_s > 250:
        n = 10000
    else:
        n = 8000

    print(f"[setup] chosen length n={n}")

    # ----------------------------------------------------------
    # 0. Initialisation – scan a simple parametric family.
    # ----------------------------------------------------------
    w = scan_power_family(n, rng)            # w sums to 1 and is symmetric
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = float(2 * n * np.max(b))
    print(f"[init] initial score={best_score:.6f}")

    # ----------------------------------------------------------
    # 1. Entropic mirror‑descent on the smoothed max.
    # ----------------------------------------------------------
    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # hyper‑parameters (tuned heuristically)
    eta0 = 4.5                 # base step size for the multiplicative update
    beta0 = 1.0                # temperature at iteration 1
    beta_max = 250.0           # safety cap – large enough to approximate the hard max
    momentum = 0.85            # exponential moving‑average of the gradient
    block_every = 2500         # frequency of symmetric block jumps
    reset_patience = 30000     # after this many non‑improving iters we reset

    # store previous gradient for momentum
    prev_grad = np.zeros_like(w)

    while True:
        # ------------------------------------------------------------------
        # 1.1 Time check – break if we are about to exceed the budget.
        # ------------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        if elapsed > max(0.0, float(budget_s) - 5.0):
            break

        # ------------------------------------------------------------------
        # 1.2 Compute the hard objective and keep the best solution.
        # ------------------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # 1.3 Build the soft‑max probability vector α over the convolution entries.
        # ------------------------------------------------------------------
        beta = beta0 * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        # shift by max for numerical stability
        diff = b - max_b
        exp_term = np.exp(beta * diff)
        alpha = exp_term / np.sum(exp_term)   # length 2·n‑1, sums to 1

        # ------------------------------------------------------------------
        # 1.4 Gradient of the smoothed objective.
        # ------------------------------------------------------------------
        grad = correlation_grad(w, alpha)     # shape (n,)

        # momentum smoothing
        if iteration == 0:
            smoothed = grad
        else:
            smoothed = (1.0 - momentum) * grad + momentum * prev_grad
        prev_grad = grad

        # ------------------------------------------------------------------
        # 1.5 Multiplicative (entropic) update.
        # ------------------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1)
        w = w * np.exp(-eta * smoothed)

        # renormalise to a probability vector
        w_sum = w.sum()
        if w_sum <= 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # enforce exact symmetry (the gradient already respects symmetry,
        # but rounding errors accumulate)
        w = 0.5 * (w + w[::-1])
        w /= w.sum()

        # ------------------------------------------------------------------
        # 1.6 Symmetric block perturbation (helps escape plateaus).
        # ------------------------------------------------------------------
        if (iteration % block_every == 0) and (iteration > 0):
            cand = symmetric_block_perturbation(best_w, rng)
            cand /= np.sum(cand)
            cand_score = float(2 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()          # continue from this improved point
                no_improve = 0
                print(f"[block] iter={iteration:7d}, new best={best_score:.6f}")

        # ------------------------------------------------------------------
        # 1.7 Reset if stuck for a long time.
        # ------------------------------------------------------------------
        if no_improve > reset_patience:
            # Jump away from current state – start from a fresh symmetric
            # perturbation of the best known solution.
            w = symmetric_block_perturbation(best_w, rng)
            w /= np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ----------------------------------------------------------
    # 2. Final scaling – the evaluator is scale‑invariant, but we use the
    #    conventional ℓ¹‑norm √(2 n) for a pretty representation.
    # ----------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum          # sum(final_seq) = √(2 n)

    # optional sanity check – the harness will evaluate it anyway
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best raw score={best_score:.6f}, "
              f"scaled evaluate={final_score:.6f}")
    except Exception:   # evaluate_sequence may not be available in the local namespace
        print(f"[final] n={n}, best raw score={best_score:.6f} (no sanity check)")

    # return a plain Python list of floats
    return [float(x) for x in final_seq]
