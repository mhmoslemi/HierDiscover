def simplex_projection(v):
    # v: array of real numbers
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Optimisation of a non‑negative step function for the evaluation

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

using a smooth (soft‑max) approximation of the max‑norm and a
projected gradient descent on the probability simplex.  The
gradient of the log‑sum‑exp of the convolution is computed by a
single FFT‑based convolution, which is much cheaper than the
previous per‑entry accumulation.  A linear (or exponential)
schedule drives the temperature β from a low value (smooth)
to a high value (sharp max), while the step size η decays.
Occasional log‑normal block perturbations and a simple
restart‑on‑stagnation policy help escape local minima.  The
best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold a linear convolution
    of length *n*.
    """
    size = n - 1
    return 1 << (size.bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative vector *a* using FFT.
    Result length is 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  Σ_i w_i * (a * a)_i  w.r.t. *a*.
    Equivalent to 2 * (w ⊛ a_rev) restricted to the original length.
    """
    n = a.size
    rev_a = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    af = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(wf * af, n=fft_len)[: (3 * n - 2)]
    # slice the middle part: indices n-1 … n-1+n-1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (the best found) that
    minimises ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # 1) Choose an initial vector (prefer an existing solution)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                if a.sum() == 0.0:
                    a = np.full_like(a, 1.0 / a.size)
                else:
                    a = a / a.sum()
                n = a.size
            else:
                raise ValueError
        except Exception:
            n = 256
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 256
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # If the starting vector is very short we can safely enlarge it to obtain
    # more optimisation freedom.  Choose a length that is a power of two
    # but not larger than 5000 (to keep each iteration cheap).
    target_len = max(n, 1024)
    if target_len != n:
        # enlarge by padding zeros and renormalising
        a = np.concatenate([a, np.zeros(target_len - n)])
        a = _project_simplex(a)
        n = target_len

    # FFT lengths needed repeatedly
    fft_len_conv = _fft_len_for_n(2 * n - 1)          # for self‑convolution
    fft_len_grad = _fft_len_for_n(3 * n - 2)          # for gradient convolution

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len_conv)
    best_score = 2.0 * n * np.max(conv_best) / (np.sum(best) ** 2)

    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 0.8        # initial additive step size
    eta_end = 0.02         # final step size
    beta_start = 0.5       # start temperature (smooth max)
    beta_end = 2000.0      # final temperature (sharp max)
    block_perturb_prob = 0.04
    restart_gap = 25.0     # seconds without improvement before a restart
    momentum = 0.9         # exponential smoothing of the gradient

    # momentum buffer
    grad_mom = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution and score -----
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        # sum(a) == 1 after each projection, but recompute for safety
        s = a.sum()
        cur_score = 2.0 * n * cur_max / (s * s)

        # ----- keep the best solution -----
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedules for η and β -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # linear interpolation for η
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        # exponential schedule for β (more aggressive sharpening)
        beta = beta_start * (beta_end / beta_start) ** (elapsed / total)

        # ----- soft‑max weights (log‑sum‑exp gradient) -----
        # stability: shift by max
        shift = conv.max()
        w_raw = np.exp(beta * (conv - shift))
        w_sum = w_raw.sum()
        w = w_raw / w_sum               # normalised weights, sum to 1

        # ----- gradient of the smoothed objective (full) -----
        grad = _gradient_from_weights(a, w, fft_len_grad)

        # momentum smoothing
        grad_mom = momentum * grad_mom + (1.0 - momentum) * grad

        # ----- projected additive gradient step -----
        a_tmp = a - eta * grad_mom
        a = _project_simplex(a_tmp)

        # ----- occasional random block perturbation -----
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- soft restart when progress stalls -----
        if now - last_improve > restart_gap:
            # start from the current best but add a modest log‑normal jitter
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = best * noise
            a = _project_simplex(a)
            grad_mom.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (safety clipping)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
