def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (non‑circular)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power‑of‑two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max objective.
    g = 2 * corr(w, alpha),   corr(w,α)[j] = Σ_i w[i]·α[i+j].
    Implemented by a second FFT convolution with the reversed vector.
    """

    conv = convolve_fft(w_rev, alpha)      # length = 3·n‑2
    start = w.shape[0] - 1
    return 2.0 * conv[start:start + w.shape[0]]

    alphas = np.linspace(0.30, 0.95, 7)               # α<1 gives heavier ends
    eps = 1e-6
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a in alphas:
        # w_i ∝ (x·(1−x) + eps)^{a‑1}
        cand = np.power(x * (1.0 - x) + eps, a - 1.0)
        cand /= cand.sum()
        try_candidate(cand, f"beta α={a:.3f}")

    # (c) mixture of Catalan and uniform
    uniform = np.full(n, 1.0 / n)
    catalan = catalan_sequence(n)
    catalan /= catalan.sum()
    mixes = np.linspace(0.0, 1.0, 11)                 # λ for Catalan
    for lam in mixes:
        cand = lam * catalan + (1.0 - lam) * uniform
        cand /= cand.sum()
        try_candidate(cand, f"mix λ={lam:.2f}")

    # --------------------------------------------------------------
    # 3. Entropic / Adam mirror descent from the best warm‑start
    # --------------------------------------------------------------
    w = best_w.copy()
    best_w = w.copy()
    best_score = score_from_weights(best_w)

    # Adam‑like moments (all‑zero initialisation)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-8

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    eta0 = 5.0               # base step size for the multiplicative Adam update
    beta0 = 0.35             # soft‑max temperature coefficient
    beta_max = 250.0
    symmetrize_every = 200
    block_every = 1200
    reset_patience = 200000

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ----- convolution and hard score -----
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ----- soft‑max (smooth max) -----
        beta = beta0 * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b                     # shift for numerical stability
        exp_term = np.exp(beta * diff)
        alpha = exp_term / np.sum(exp_term)   # probability vector, length 2·n‑1

        # ----- gradient of the smooth objective -----
        grad = correlation_grad(w, alpha)

        # ----- Adam‑style multiplicative update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        # bias‑correction
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        eta = eta0 / np.sqrt(iteration + 1)
        w *= np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps_adam))

        # ----- projection onto the simplex (renormalise) -----
        w_sum = w.sum()
        if w_sum <= 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # ----- enforce symmetry periodically -----
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= w.sum()

        # ----- cheap random block perturbation (helps escape plateaux) -----
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand /= cand.sum()
            cand_score = score_from_weights(cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best score={best_score:.6f}")

        # ----- reset if stuck for a long time -----
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w /= w.sum()
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final scaling – the evaluation is scale‑invariant; we use the
    #    conventional ℓ¹‑norm √(2 n) for readability.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum   # sum(best_w)=1, so simply scale

    # Optional sanity check (the harness will also call evaluate_sequence)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:
        final_score = float('nan')
    print(f"[final] length={n}, best raw score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
```

This implementation builds a solid initial distribution by exploring a variety of low‑dimensional families, then refines it with a fast Adam‑style mirror‑descent that directly minimises a smooth approximation of the maximum self‑convolution.  Symmetry enforcement, occasional block perturbations and a reset mechanism help escape plateaus.  All operations stay well within the 1000 s budget, and the final sequence respects the non‑negativity, sum‑normalisation and magnitude limits required by `evaluate_sequence`.
