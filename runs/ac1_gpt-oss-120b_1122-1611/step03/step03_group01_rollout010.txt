    conv = convolve_fft(a, b[::-1])
    start = b.size - 1

def simplex_projection(v: np.ndarray) -> np.ndarray:

    n = v.shape[0]

        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    return w
