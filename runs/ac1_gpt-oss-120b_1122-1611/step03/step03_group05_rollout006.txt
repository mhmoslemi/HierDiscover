"""
Propose a candidate step‑function (non‑negative heights) that minimizes the
self‑convolution peak used by ``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – Use the globally supplied best construction
   ``height_sequence_1`` if available, otherwise fall back to a Catalan‑type
   sequence.  The vector is projected onto the probability simplex
   (non‑negative, sum = 1).

2. **Smooth max‑approximation phase** – Optimise the smooth surrogate
   ``L_β(w) = (1/β)·log Σ exp(β·(w*w)[k])`` with Adam.  The gradient of the
   surrogate is a weighted sum of correlation vectors; the weights are the
   soft‑max over the convolution values.  β is increased linearly (homotopy)
   from 2 to a large final value (≈ 400) so the landscape gradually becomes
   the true ``max``.  The iterate is projected onto the simplex after each
   Adam update.

3. **Sharp sub‑gradient phase** – Once β reaches its final value we switch
   to the exact sub‑gradient of the max (soft‑max essentially becomes a uniform
   distribution over the active constraints).  Adam continues with a smaller
   learning‑rate.

4. **Local hill‑climbing** – With the best vector found so far we perform a
   large number of cheap pairwise mass‑transfer moves.  A move takes a small
   amount of weight from an index participating in the current maximal
   convolution entry and moves it to a random location.  The move is kept
   only if it strictly reduces the maximal convolution value.  This fine‑tunes
   the solution beyond the gradient descent and typically yields the extra
   few percent needed to beat the target.

5. **Rescaling** – The best probability vector is finally scaled to the
   conventional ℓ¹‑norm `√(2·n)` required by the benchmark.  The returned list
   can be handed directly to ``evaluate_sequence``.

The whole routine respects the 1000 s budget, runs on a single thread and
uses only NumPy (no external solvers).  All helper functions are defined at
module level as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level as required)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers divided by 4**i, i.e. a_i = C(2i,i)/4**i.
    The recurrence a_{i+1}=a_i*(2i+1)/(2i+2) is numerically stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    # Duchi et al. (2008) – O(n log n) implementation
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random perturbation – useful as an occasional jump."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)            # 0.5 % of length
    hi = max(2, n // 12)             # up to ~8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ~±12 % multiplicative noise
    cand[start:stop] *= factor
    if rng.random() < 0.35:          # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Return the next power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """
    Linear self‑convolution w*w using an FFT of length N.
    Returns the full (2*len(w)-1) result.
    """
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # remove tiny negatives
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth max‑approximation L_β at point w.
    ``probs`` are the soft‑max weights (sum 1, length 2*len(w)-1).
    Returns a vector of length len(w).
    """
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = probs.size                 # = 2*len(w)-1
    offset = m - 1
    grad = 2.0 * corr_full[offset : offset + w.size]
    return grad


# ----------------------------------------------------------------------
# Main function required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function with a small self‑convolution peak.
    Returns a plain Python list of floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size and initial vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=np.float64)
            if init_seq.ndim != 1 or init_seq.size < 2:
                raise ValueError
            w = init_seq.copy()
            w = np.maximum(w, 0.0)
            total = np.sum(w)
            if total < 1e-12:
                raise ValueError
            w = w / total                     # project onto simplex (sum = 1)
        except Exception:
            n = 8000
            w = catalan_sequence(n)
            w = w / np.sum(w)
    else:
        n = 8000
        w = catalan_sequence(n)
        w = w / np.sum(w)

    n = w.size
    N_self = _next_pow2(2 * n - 1)          # FFT length for self‑convolution
    N_corr = _next_pow2(3 * n - 2)          # FFT length for gradient correlation

    # --------------------------------------------------------------
    # 2. Optimisation state.
    # --------------------------------------------------------------
    b = self_convolve_fft(w, N_self)
    best_w = w.copy()
    best_score = np.max(b)                 # this is max‑conv for sum=1
    cur_iter = 0
    no_improve = 0
    start_time = time.monotonic()
    print(f"[init] n={n}, initial max‑conv={best_score:.9f}")

    # Adam hyper‑parameters (smooth phase)
    lr = 0.045
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # β‑homotopy
    beta_start = 2.0
    beta_end = 400.0               # a fairly sharp approximation of max
    beta_ramp_iters = 250_000      # number of iterations over which β grows

    # --------------------------------------------------------------
    # 3. Smooth + sub‑gradient optimisation (≈ 80 % of the budget)
    # --------------------------------------------------------------
    phase_end = start_time + budget_s * 0.80
    while time.monotonic() < phase_end:
        # a) current self‑convolution and its max
        b = self_convolve_fft(w, N_self)
        cur_max = np.max(b)

        # keep best ever seen
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={cur_iter:7d}, max‑conv={best_score:.9f}")
        else:
            no_improve += 1

        # b) β schedule
        if cur_iter < beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * (cur_iter / beta_ramp_iters)
        else:
            beta = beta_end

        # c) soft‑max weights over convolution values
        shift = beta * (b - cur_max)          # ≤ 0, stable
        exp_shift = np.exp(shift)            # max weight == 1
        probs = exp_shift / np.sum(exp_shift)  # shape (2n-1,), sum 1

        # d) gradient of the smooth surrogate
        grad = gradient_smooth(w, probs, N_corr)

        # e) Adam update with simplex projection
        cur_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** cur_iter)
        v_hat = v / (1.0 - beta2 ** cur_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional clipping to avoid too large jumps
        max_step = 0.25
        step = np.clip(step, -max_step, max_step)

        w_new = w - step
        w = simplex_projection(w_new)

        # f) occasional random jump to escape plateaus
        if cur_iter % 6000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_b = self_convolve_fft(cand, N_self)
            cand_max = np.max(cand_b)
            if cand_max < best_score - 1e-12:
                best_w = cand.copy()
                best_score = cand_max
                w = cand.copy()
                print(f"[block] iter={cur_iter:7d}, new best max‑conv={best_score:.9f}")

        # g) reset when stuck for a long while
        if no_improve > 30_000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={cur_iter:7d}, reset to perturbed best")

    # --------------------------------------------------------------
    # 4. Fine‑grained hill‑climbing (remaining ~20 % of the budget)
    # --------------------------------------------------------------
    hill_end = start_time + budget_s
    # Pre‑compute a fresh convolution for the current best vector
    w = best_w.copy()
    b = self_convolve_fft(w, N_self)
    cur_max = np.max(b)
    print(f"[hill‑start] max‑conv={cur_max:.9f}")

    # parameters for the local moves
    eps_factor = 0.30          # portion of the donor weight to move
    min_eps = 1e-6
    max_eps = 0.003            # absolute ceiling for a single transfer

    while time.monotonic() < hill_end:
        # locate the (possibly multiple) peak indices
        k_max = int(np.argmax(b))
        cur_max = b[k_max]

        # try to pick a donor i that participates in this peak
        # i must satisfy 0 ≤ i < n and 0 ≤ j = k_max - i < n
        i = rng.integers(0, n)
        j = k_max - i
        if not (0 <= j < n):
            continue
        # both positions must hold some mass to be able to donate
        if w[i] <= min_eps or w[j] <= min_eps:
            continue

        # amount to move from each donor (we move from both i and j)
        eps_i = min(w[i] * eps_factor, max_eps)
        eps_j = min(w[j] * eps_factor, max_eps)
        if eps_i < min_eps or eps_j < min_eps:
            continue

        # pick two random recipients (could coincide)
        p = rng.integers(0, n)
        q = rng.integers(0, n)

        # construct candidate vector
        cand = w.copy()
        cand[i] -= eps_i
        cand[j] -= eps_j
        cand[p] += eps_i
        cand[q] += eps_j

        # ensure feasibility (non‑negativity)
        if cand[i] < -1e-12 or cand[j] < -1e-12:
            continue

        # evaluate candidate
        cand_b = self_convolve_fft(cand, N_self)
        cand_max = np.max(cand_b)

        # accept only if strictly better
        if cand_max < cur_max - 1e-12:
            w = cand
            b = cand_b
            cur_max = cand_max
            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_w = w.copy()
                print(f"[hill‑improve] iter={cur_iter:7d}, max‑conv={best_score:.9f}")

        cur_iter += 1

    # --------------------------------------------------------------
    # 5. Final scaling – bring the best probability vector to the
    #    conventional √(2 n) ℓ¹‑norm.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # safeguard sum==1

    # optional sanity check (cheap)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best max‑conv={best_score:.9f}, "
              f"evaluate={final_score:.9f}")
    except Exception:
        pass

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
