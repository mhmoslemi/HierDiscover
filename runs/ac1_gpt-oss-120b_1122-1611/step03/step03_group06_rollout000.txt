def optimize_w(w, n, rng, budget):
    start = time.monotonic()
    # track best
    best_w = w.copy()
    b = convolve_fft(w,w)
    best_score = 2.0 * n * np.max(b) # evaluation metric
    print(f"[opt] start len {n} score {best_score}")
    # Phase parameters
    # boundaries for phases:
    phase1_time = budget * 0.6
    phase2_time = budget * 0.35
    # Phase1: similar to previous but with slower beta growth etc.
    # We'll keep same code but maybe more aggressive eta0.
    # Use gradient descent with momentum; no special block perturbation maybe minimal.
    # Phase1 loop
    phase1_end = start + phase1_time
    eta0 = 1.0
    beta = 1.0
    beta_max = 200.0
    beta_growth = 1.01
    momentum = 0.9
    velocity = np.zeros_like(w)
    iteration = 0
    while time.monotonic() < phase1_end:
        # compute convolution b
        b = convolve_fft(w,w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            print(f"[phase1 improve] iter {iteration} score {best_score}")
        # softmax weights
        exp_arg = beta * (b - max_b)
        p = np.exp(exp_arg)
        sum_p = np.sum(p)
        if sum_p == 0.0:
            p = np.full_like(b, 1.0 / b.size)
            sum_p = 1.0
        p /= sum_p
        grad = 2.0 * convolve_fft(p, w)[:n]
        eta = eta0 / np.sqrt(iteration + 1.0)
        velocity = momentum * velocity + eta * grad
        w = simplex_projection(w - velocity)
        # symmetry enforcement optional:
        if iteration % 1000 == 0:
            w = (w + w[::-1]) * 0.5
        # increase beta each 2000 iter
        if iteration % 2000 == 0:
            beta = min(beta_max, beta * beta_growth)
        iteration += 1
    # Phase2: exact max gradient descent with Adam.
    # compute gradient using p indicator of max positions.
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 0
    phase2_end = phase1_end + phase2_time
    while time.monotonic() < phase2_end:
        b = convolve_fft(w,w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            print(f"[phase2 improve] iter {iteration} score {best_score}")
        # indicator of indices near max
        # tolerance to allow multiple peaks; use eps relative
        tol = max_b * 1e-12 + 1e-15
        K = np.where(b >= max_b - tol)[0]
        p = np.zeros_like(b)
        if K.size > 0:
            p[K] = 1.0 / K.size
        # gradient
        grad = 2.0 * convolve_fft(p, w)[:n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        eta = 0.2 / (np.sqrt(iteration + 1) + 1)  # some schedule
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        w = simplex_projection(w - step)
        # Symmetrize occasionally
        if iteration % 500 == 0:
            w = (w + w[::-1]) * 0.5
        iteration += 1
        # occasional random small perturbation if no improvement for many
        if iteration % 2000 == 0 and cur_score - best_score > 1e-9:
            # random jitter
            perturb = block_perturbation(w, rng)
            w = perturb / np.sum(perturb)
    # final best_seq scaling
    final_score = 2.0 * n * np.max(convolve_fft(best_w,best_w))
    print(f"[opt final] len {n} best_score {final_score}")
    return best_w, final_score

"""
Hybrid sub‑gradient optimiser for the “max self‑convolution” problem.

The evaluation metric for a vector a (non‑negative, Σa = √(2 n))
is 2 n · max_k (a*a)[k].  Because the objective is scale‑invariant we
optimise a probability vector w ( Σw = 1 ) and later rescale it.

Algorithm
---------
1. Initialise w with a known good construction (``height_sequence_1``) or
   with a Catalan‑type seed.
2. **Phase 1** – soft‑max gradient descent (same idea as the previous
   baseline but with a more aggressive step‑size schedule and optional
   symmetrisation).  This quickly reduces the objective from the start.
3. **Phase 2** – exact max‑subgradient descent using the true arg‑max
   positions of the convolution.  The sub‑gradient is
        g = 2·(p * w)[:n]   with  p_k = 1/|K|  for k∈K,  K = {indices of max(b)}.
   An Adam‑style adaptive step is applied, followed by Euclidean projection
   onto the probability simplex.
4. Periodic random block perturbations and an optional symmetrisation keep
   the search from stagnating.
5. The best vector encountered is finally scaled to Σ = √(2 n) and returned.

The optimiser runs for the allotted time budget (default ≈ 1000 s) and
prints intermediate improvements.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level, no closures.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Generate the first ``length`` Catalan‑type coefficients:
    a_i = C(2i,i)/4**i  (the sequence used in many earlier attempts)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (full) convolution via real FFT.  Returns a non‑negative array."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()            # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # suppress tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries equal (rare)
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation – a contiguous block is scaled by a
    log‑normal factor; optionally the whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                 # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Core optimiser (two‑phase hybrid gradient descent).
# ----------------------------------------------------------------------


def _optimise_w(initial_w: np.ndarray,
                n: int,
                rng: np.random.Generator,
                time_budget: float) -> tuple[np.ndarray, float]:
    """
    Perform the two‑phase optimisation on probability vector ``initial_w``.
    Returns the best vector (Σ=1) found and its corresponding score
    ``2·n·max(convolution)``.
    """
    start_time = time.monotonic()
    deadline = start_time + time_budget

    # ------------------------------------------------------------------
    # State.
    # ------------------------------------------------------------------
    w = initial_w.copy()
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)

    print(f"[optimiser] start n={n}, init_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent.
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * time_budget   # ~55 % of time
    eta0 = 0.9
    beta = 1.0
    beta_max = 300.0
    beta_growth = 1.015
    momentum = 0.85
    velocity = np.zeros_like(w)
    it = 0

    while time.monotonic() < phase1_deadline:
        b = convolve_fft(w, w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            print(f"[phase1 improve] it={it:06d}, score={best_score:.6f}")

        # soft‑max weights (log‑sum‑exp trick)
        exp_arg = beta * (b - max_b)            # ≤ 0
        p = np.exp(exp_arg)
        sum_p = np.sum(p)
        if sum_p == 0.0:                        # under‑flow → uniform fallback
            p = np.full_like(b, 1.0 / b.size)
            sum_p = 1.0
        p /= sum_p

        # gradient of the smoothed objective
        grad = 2.0 * convolve_fft(p, w)[:n]

        # heavy‑ball momentum with diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)
        velocity = momentum * velocity + eta * grad
        w = simplex_projection(w - velocity)

        # occasional symmetrisation (helps keep the shape balanced)
        if it % 1000 == 0:
            w = 0.5 * (w + w[::-1])

        # beta schedule
        if it % 3000 == 0:
            beta = min(beta_max, beta * beta_growth)

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – exact max‑subgradient descent with Adam‑style adaptation.
    # ------------------------------------------------------------------
    phase2_deadline = deadline
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it2 = 0
    no_imp_iters = 0

    while time.monotonic() < phase2_deadline:
        b = convolve_fft(w, w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_imp_iters = 0
            print(f"[phase2 improve] it={it2:06d}, score={best_score:.6f}")
        else:
            no_imp_iters += 1

        # Identify all (near) maximisers – a tiny tolerance guards against
        # floating‑point ties.
        tol = max_b * 1e-13 + 1e-15
        K = np.where(b >= max_b - tol)[0]
        p = np.zeros_like(b)
        if K.size:
            p[K] = 1.0 / K.size

        # sub‑gradient of the true max (no smoothing)
        grad = 2.0 * convolve_fft(p, w)[:n]

        # Adam update (learning‑rate schedule decays with √t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it2 + 1))
        v_hat = v / (1 - beta2 ** (it2 + 1))
        eta = 0.25 / (np.sqrt(it2 + 1) + 1.0)   # modest step size
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        w = simplex_projection(w - step)

        # keep shape roughly symmetric (does not hurt feasibility)
        if it2 % 500 == 0:
            w = 0.5 * (w + w[::-1])

        # to avoid long stagnation, perform a cheap random jump
        if no_imp_iters > 30000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_imp_iters = 0
            print(f"[reset] it={it2}")

        it2 += 1

    # ------------------------------------------------------------------
    # Return the best vector (still on the simplex) and its 2·n·max(b) score.
    # ------------------------------------------------------------------
    final_b = convolve_fft(best_w, best_w)
    final_score = 2.0 * n * np.max(final_b)
    print(f"[optimiser] finished n={n}, best_score={final_score:.6f}")
    return best_w, final_score


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a non‑negative step function (list of floats) with a low
    ``evaluate_sequence`` value.  The routine runs a hybrid optimiser
    (soft‑max warm‑up + exact max sub‑gradient descent) for the allotted
    time budget and finally rescales the best probability vector to the
    required ℓ₁‑norm √(2 n).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.  Prefer the length of the previously best
    # construction (if present); otherwise start with ~8 k entries.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            # Normalise to a probability vector.
            w0 = prev / np.sum(prev)
        except Exception:
            n = 8000
            w0 = catalan_sequence(n)
            w0 = w0 / np.sum(w0)
    else:
        n = 8000
        w0 = catalan_sequence(n)
        w0 = w0 / np.sum(w0)

    # Reserve a couple of seconds for final scaling/printing.
    time_budget = max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Run the optimiser.
    # ------------------------------------------------------------------
    best_w, best_score = _optimise_w(w0, n, rng, time_budget)

    # ------------------------------------------------------------------
    # Rescale to the required ℓ₁‑norm √(2 n) and return as a plain list.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = (best_w * target_sum).tolist()

    # Store the result globally for possible reuse in subsequent runs.
    global height_sequence_1
    height_sequence_1 = final_seq

    # (Optionally) verify via the provided evaluator – this is cheap.

    return final_seq
