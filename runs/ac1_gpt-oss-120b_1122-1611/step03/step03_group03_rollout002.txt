"""
Propose a non‑negative step‑function that minimizes
    evaluate_sequence(sequence) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm explores several sensible initial shapes (uniform,
U‑shaped “arcsine”, random) and improves them with a projected Adam
optimizer on a smooth‑max surrogate of the L∞‑norm of the
self‑convolution.  Symmetry is enforced explicitly (the optimum is
symmetric), and occasional block‑perturbations are used to escape
plateaus.  The best sequence found within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT.  Negative round‑off errors are
    clipped to zero for numerical stability."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}
    (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor; with some
    probability also enforce symmetry.  The result stays on the simplex."""
    cand = base.copy()
    n = cand.size
    # block width is between 0.5 % and 8 % of n (tunable)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:          # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    # keep on simplex (simplex_projection will also enforce non‑negativity)
    cand = simplex_projection(cand)
    return cand


def arcsine_init(n: int) -> np.ndarray:
    """U‑shaped “arcsine” distribution:
       a_i ∝ 1/√((i+½)(n‑½‑i))   (discrete analogue of Beta(½,½))."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((idx + 0.5) * (n - 0.5 - idx))
    vals = vals / np.sum(vals)    # Σ = 1
    return vals


def uniform_init(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def random_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential random vector, projected onto the simplex."""
    v = rng.exponential(scale=1.0, size=n)
    v = v / np.sum(v)
    return v.astype(np.float64)


# ----------------------------------------------------------------------
# Core optimiser for a single length and a single initialisation
# ----------------------------------------------------------------------


def optimise_one_start(w0: np.ndarray,
                       n: int,
                       time_budget: float,
                       rng: np.random.Generator,
                       target_score: float = 1.5025) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser with symmetric enforcement.
    Returns the best vector (sum = 1) and its evaluation score.
    """
    w = w0.copy()
    # enforce exact simplex & symmetry before starting
    w = simplex_projection(w)
    w = 0.5 * (w + w[::-1])
    w = simplex_projection(w)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.18                # base learning rate
    alpha = 10.0              # start temperature for soft‑max
    max_alpha = 2000.0

    start = time.monotonic()
    best_w = w.copy()
    conv = convolve_fft(w, w)
    best_score = float(2.0 * n * np.max(conv))
    no_improve = 0
    iteration = 0

    while time.monotonic() - start < time_budget:
        # ----- convolution -----
        conv = convolve_fft(w, w)               # length 2n‑1
        maxc = np.max(conv)

        # ----- smooth‑max weights -----
        # temperature α; shift by maxc for numerical stability
        exp_vals = np.exp(alpha * (conv - maxc))
        p = exp_vals / np.sum(exp_vals)        # soft‑max over convolution indices

        # ----- gradient of the smoothed objective -----
        # grad = 2 * (p * w) convolution (correlation)
        grad_full = convolve_fft(p, w[::-1])    # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]  # central slice = true gradient

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        w_new = w - step
        w_new = simplex_projection(w_new)
        # enforce symmetry (optimal solution is symmetric)
        w_new = 0.5 * (w_new + w_new[::-1])
        w = simplex_projection(w_new)

        # ----- evaluate true objective (cheap: reuse conv) -----
        conv = convolve_fft(w, w)
        score = float(2.0 * n * np.max(conv))

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            # optional logging
            # print(f"[improve] n={n}, iter={iteration}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ----- schedule temperature growth -----
        if iteration % 2500 == 0 and iteration > 0:
            alpha = min(alpha * 1.07, max_alpha)

        # ----- occasional large perturbation of the current best -----
        if iteration % 1200 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            # keep symmetry
            cand = 0.5 * (cand + cand[::-1])
            cand = simplex_projection(cand)
            cand_score = float(2.0 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                w = cand.copy()
                # print(f"[block] n={n}, iter={iteration}, new best={best_score:.6f}")

        # ----- restart if stuck -----
        if no_improve > 30000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            w = 0.5 * (w + w[::-1])
            w = simplex_projection(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            # print(f"[reset] n={n}, iter={iteration}")

        iteration += 1

        # early exit if we have already reached the target
        if best_score <= target_score:
            break

    return best_w, best_score


# ----------------------------------------------------------------------
# Orchestrator – runs several lengths and several initialisers
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative list) with low
    evaluate_sequence() value.  The routine respects the supplied time
    budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine which lengths to try.  Re‑use the length of a previously
    # stored construction if available, otherwise test a few sensible
    # sizes.
    # ------------------------------------------------------------------
    lengths = []
    if "height_sequence_1" in globals():
        try:
            lengths.append(len(height_sequence_1))
        except Exception:
            pass
    # a few lengths that are known to be tractable
    lengths.extend([6000, 6500, 7000])
    # deduplicate while preserving order
    lengths = list(dict.fromkeys(lengths))

    total_start = time.monotonic()
    safety_margin = 5.0
    total_budget = max(0.0, float(budget_s) - safety_margin)

    best_overall_seq = None
    best_overall_score = np.inf

    # Allocate budget proportionally to the number of remaining lengths
    for idx, n in enumerate(lengths):
        elapsed = time.monotonic() - total_start
        remaining = total_budget - elapsed
        if remaining <= 0:
            break
        # Evenly split the remainder among the lengths we have not processed yet
        per_len_budget = remaining / (len(lengths) - idx)

        # ------------------------------------------------------------------
        # Prepare a set of initialisers for this length
        # ------------------------------------------------------------------
        init_fns = [
            arcsine_init,
            uniform_init,
            lambda nn: random_init(nn, rng),
        ]

        best_local_seq = None
        best_local_score = np.inf

        for init_fn in init_fns:
            # evaluate remaining time before starting a new run
            elapsed = time.monotonic() - total_start
            if elapsed >= total_budget:
                break

            # Give each initialiser a thin slice of the per‑length budget
            init_budget = per_len_budget * 0.3   # 30 % of the per‑length budget
            # ensure we have some time left for the optimisation
            remaining_for_opt = per_len_budget - init_budget
            if remaining_for_opt <= 0:
                remaining_for_opt = per_len_budget * 0.7

            w0 = init_fn(n)
            # run the optimizer (it will respect the time limit)
            seq, score = optimise_one_start(
                w0,
                n,
                time_budget=remaining_for_opt,
                rng=rng,
                target_score=1.5025,
            )
            if score < best_local_score:
                best_local_score = score
                best_local_seq = seq.copy()
                # optional log
                # print(f"[len={n}] new local best with init {init_fn.__name__}: {score:.6f}")

            # Early exit if we already beat the global target
            if best_local_score <= 1.5025:
                break

        # Update global best
        if best_local_score < best_overall_score:
            best_overall_score = best_local_score
            best_overall_seq = best_local_seq.copy()
            # print(f"[global] length={n}, best_score={best_overall_score:.6f}")

        # Stop completely if target reached
        if best_overall_score <= 1.5025:
            break

    # ------------------------------------------------------------------
    # Safety fallback: if nothing was discovered, fall back to a simple
    # arcsine shape (guaranteed to be a valid candidate).
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        n_fallback = 6000
        best_overall_seq = arcsine_init(n_fallback)

    # Convert to plain Python list of floats for the harness
    return [float(x) for x in best_overall_seq]
