        b = convolve_fft(w,w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1

        p = np.exp(exp_arg)

Algorithm
---------

* Phase 1 – modest length (≈ 4 k).  
  Initialise from the previously found sequence if it exists,
  otherwise from the Catalan‐type distribution.  Run a fast
  exponentiated‑gradient descent on a smooth (log‑sum‑exp) surrogate
  of the maximum self‑convolution value.  A tiny amount of explicit
  smoothing (3‑point moving average) is applied each iteration to
  keep the shape regular.

* Phase 2 – larger length (≈ 8 k).  
  Upsample the best vector from phase 1 by linear interpolation,
  renormalise and continue the same optimisation.

* Throughout the optimisation a soft‑max temperature β is slowly
  increased so that the surrogate converges to the true max.
  Every few thousand iterations a “block perturbation’’ (multiplication
  of a random contiguous block by a log‑normal factor) is applied to
  escape plateaus.  If no improvement is observed for a long stretch
  the current iterate is re‑initialised from the best known point.

* After the time budget is exhausted the best vector is rescaled to
  the required ℓ¹‑norm √(2 n) – after this scaling the evaluation
  function simply returns the recorded best score.

The optimiser uses exponentiated‑gradient updates (no sorting‑based
projection) which makes each iteration cheap and lets us perform many
hundreds of thousands of iterations within the ~1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan‑type coefficients a_i = C(2i,i)/4**i for i=0…length‑1."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using a real FFT.
    Equivalent to np.convolve(a, b) but O(n log n) for large n.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # remove tiny negatives
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Random contiguous‑block perturbation used as a “jump’’.
    A block is multiplied by a log‑normal factor; optionally the
    whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def init_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial non‑negative probability vector of length n.
    Preference is given to the global `height_sequence_1` if it exists.
    """
    global_seq = globals().get('height_sequence_1')
    if global_seq is not None:
        seq0 = np.asarray(global_seq, dtype=np.float64)
        # Down‑sample / up‑sample to the wanted length.
        if seq0.size != n:
            idx = np.linspace(0, seq0.size - 1, n)
            w = np.interp(idx, np.arange(seq0.size), seq0)
        else:
            w = seq0.copy()
        w = np.maximum(w, 0.0)
        if np.sum(w) == 0.0:
            w = catalan_sequence(n)
    else:
        w = catalan_sequence(n)
    w = np.maximum(w, 0.0)
    w = w / np.sum(w)
    return w


def run_opt(w: np.ndarray,
            n: int,
            time_limit: float,
            rng: np.random.Generator,
            *,
            beta_start: float = 1.0,
            beta_max: float = 400.0,
            beta_growth: float = 1.01,
            beta_interval: int = 2000,
            base_lr: float = 0.5,
            smooth_gamma: float = 0.03,
            perturb_interval: int = 4000,
            stagnation_limit: int = 50000) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector `w` (sum = 1, non‑negative) to minimise
    2·n·max(conv(w,w)).  Returns the best vector encountered and its score.
    """
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)

    # smoothing kernel for the cheap low‑pass filter
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    # temperature for the soft‑max approximation of the max
    beta = beta_start

    iteration = 0
    no_improve = 0
    start = time.monotonic()

    while time.monotonic() - start < time_limit:
        # --------------------------------------------------------------
        # Current objective
        # --------------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Soft‑max weights (smooth approximation of the max)
        # --------------------------------------------------------------
        exp_arg = beta * (b - max_b)          # ≤ 0, stabilised by subtraction
        p = np.exp(exp_arg)
        sum_p = np.sum(p)
        if sum_p == 0.0:                      # underflow guard
            p.fill(1.0 / p.size)
        else:
            p /= sum_p

        # --------------------------------------------------------------
        # Gradient of the smoothed objective:
        #   ∇S = 2·(p * w)[:n]
        # --------------------------------------------------------------
        grad = 2.0 * convolve_fft(p, w)[:n]

        # --------------------------------------------------------------
        # Exponentiated‑gradient update (no sorting‐based projection)
        # --------------------------------------------------------------
        lr = base_lr / np.sqrt(iteration + 1.0)
        w = w * np.exp(-lr * grad)           # multiplicative update
        w = np.maximum(w, 1e-15)             # avoid exact zeros
        w = w / np.sum(w)                    # re‑normalise to sum = 1

        # --------------------------------------------------------------
        # Light smoothing to keep the shape regular
        # --------------------------------------------------------------
        w = (1.0 - smooth_gamma) * w + smooth_gamma * np.convolve(w, kernel, mode='same')
        w = np.maximum(w, 0.0)
        w = w / np.sum(w)

        # --------------------------------------------------------------
        # Occasional random block perturbation (escape from plateaus)
        # --------------------------------------------------------------
        if iteration % perturb_interval == 0 and iteration > 0:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)

        # --------------------------------------------------------------
        # Temperature schedule for the soft‑max
        # --------------------------------------------------------------
        if iteration % beta_interval == 0:
            beta = min(beta_max, beta * beta_growth)

        # --------------------------------------------------------------
        # Reset when stagnating for a long time
        # --------------------------------------------------------------
        if no_improve > stagnation_limit:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration}")

        iteration += 1

    return best_w, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) whose
    `evaluate_sequence` value is as low as possible.
    The routine runs a two‑phase optimisation (medium → large size)
    using exponentiated‑gradient descent on a smooth surrogate of the
    max self‑convolution.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    total_time = max(0.0, float(budget_s) - 5.0)   # keep a few seconds for final scaling
    start_time = time.monotonic()
    phase1_ratio = 0.35
    phase1_time = total_time * phase1_ratio
    phase2_time = total_time - phase1_time

    # ------------------------------------------------------------------
    # Phase 1 : moderate length, quick exploration
    # ------------------------------------------------------------------
    n1 = 4096
    w1 = init_sequence(n1, rng)

    best_w1, best_score1 = run_opt(
        w1,
        n1,
        phase1_time,
        rng,
        beta_start=1.0,
        beta_max=300.0,
        beta_growth=1.02,
        base_lr=0.6,
        smooth_gamma=0.025,
        perturb_interval=3500,
        stagnation_limit=40000,
    )
    elapsed = time.monotonic() - start_time
    # ------------------------------------------------------------------
    # If time permits run Phase 2 (larger problem)
    # ------------------------------------------------------------------
    final_w = best_w1
    final_n = n1
    final_score = best_score1

    time_left = total_time - elapsed
    if time_left > 5.0:                     # need at least a few seconds for Phase 2
        n2 = 8192
        # Linear up‑sampling of the best vector from Phase 1
        idx = np.linspace(0, n1 - 1, n2)
        w2 = np.interp(idx, np.arange(n1), best_w1)
        w2 = np.maximum(w2, 0.0)
        w2 = w2 / np.sum(w2)

        best_w2, best_score2 = run_opt(
            w2,
            n2,
            time_left,
            rng,
            beta_start=2.0,
            beta_max=400.0,
            beta_growth=1.015,
            base_lr=0.5,
            smooth_gamma=0.03,
            perturb_interval=4000,
            stagnation_limit=50000,
        )

        if best_score2 < best_score1:
            final_w = best_w2
            final_n = n2
            final_score = best_score2

    # ------------------------------------------------------------------
    # Final scaling: the harness expects Σ a_i = √(2 n)
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * final_n)
    final_seq = final_w * target_sum   # Σ final_seq = √(2 n)

    # ------------------------------------------------------------------
    # (Optional) sanity check – the harness would do this anyway.
    # ------------------------------------------------------------------
    try:
        # evaluate_sequence is provided by the surrounding framework.
        eval_val = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={final_n}, best_score={final_score:.6f}, evaluate={eval_val:.6f}")
    except Exception:
        # In the unlikely event the harness does not provide it here,
        # we simply report the internally tracked best score.
        print(f"[final] length={final_n}, best_score={final_score:.6f}")

    # Return a plain Python list as required by the evaluation harness.
    return [float(x) for x in final_seq]
