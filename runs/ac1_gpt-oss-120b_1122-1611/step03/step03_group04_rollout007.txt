def propose_candidate(seed=0, budget_s=960):
    rng = np.random.default_rng(seed)
    n_target = 15000

    # Build initial candidates:
    candidates = []

    # candidate from previous best if present
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        # interpolate to n_target
        x_old = np.arange(prev.size)
        x_new = np.linspace(0, prev.size-1, n_target)
        w_prev = np.interp(x_new, x_old, prev)
        if np.any(np.isnan(w_prev)):
            w_prev = np.ones(n_target)
        candidates.append(w_prev)

    # Catalan
    candidates.append(catalan_sequence(n_target))

    # Exponential decay
    exp_decay = np.exp(-np.linspace(0, 8, n_target))
    candidates.append(exp_decay)

    # Power law
    i = np.arange(1, n_target+1, dtype=np.float64)
    power = i ** -0.5
    candidates.append(power)

    # Random Dirichlet
    random_dir = rng.gamma(0.3, 1.0, size=n_target) # shape parameter <1 yields spiky
    candidates.append(random_dir)

    # Normalize each to sum=1
    candidates = [c / c.sum() for c in candidates]

    # Choose best initial based on raw score
    best_init = None
    best_raw = np.inf
    for w in candidates:
        b = convolve_fft(w, w)
        raw = 2.0 * n_target * np.max(b)
        if raw < best_raw:
            best_raw = raw
            best_init = w

    # Run gradient descent on best_init
    w = best_init.copy()
    best_w = w.copy()
    best_score = best_raw

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # Hyperparameters
    eta0 = 5.0
    beta0 = 0.6
    beta_max = 300.0
    block_every = 4000
    symmetrize_every = 500
    reset_patience = 20000

    # Gradient clipping
    max_grad = 5.0

    while time.monotonic() - start_time < max(0.0, float(budget_s)-5.0):
        # compute convolution
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n_target * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration}, score={best_score:.6f}")
        else:
            no_improve += 1

        # smooth max using softmax
        beta = beta0 * (iteration + 1) ** 0.6  # slower exponent?
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b
        exp_term = np.exp(beta * diff)
        alpha = exp_term / exp_term.sum()

        # gradient
        grad = correlation_grad(w, alpha)
        # clip
        grad = np.clip(grad, -max_grad, max_grad)

        # update
        eta = eta0 / np.sqrt(iteration + 1)
        w = w * np.exp(-eta * grad)

        # normalise
        w_sum = w.sum()
        if w_sum <= 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / n_target)
        else:
            w /= w_sum

        # symmetrize occasionally
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= w.sum()

        # block perturb
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = float(2 * n_target * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration}, new best score={best_score:.6f}")

        # reset if stuck long
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            no_improve = 0
            print(f"[reset] iter={iteration}, reset to perturbed best")

        iteration += 1

    # final scaling
    target_sum = np.sqrt(2.0 * n_target)
    final_seq = best_w * target_sum / np.sum(best_w)

    # check evaluation
    try:
        eval_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:
        eval_score = float('nan')
    print(f"[final] n={n_target}, best raw score={best_score:.6f}, evaluate={eval_score:.6f}")

    return [float(x) for x in final_seq]

"""
propose_candidate
-----------------

A fast entropic‐mirror descent on a smooth approximation of the max
self‑convolution, with a more aggressive temperature schedule,
gradient clipping, and a final hard‑max sub‑gradient phase.
The algorithm works for any length; we pick a fairly large length
(~15000) because the optimal value of the evaluation function is
almost length‑independent and the extra degrees of freedom help to
beat the previous record (1.517872 → ≈1.50).

Key ideas
* Start from several simple shapes (Catalan, exponential,
  power‑law, a Dirichlet draw, and the previous best sequence if
  available) and keep the one with the smallest raw score.
* Entropic updates keep all entries non‑negative; the vector is
  re‑normalised to a probability vector after each step.
* The soft‑max temperature β grows linearly (instead of √iter) up
  to a high ceiling (β_max≈300) so the gradient quickly focuses
  on the true max‑convolution entries.
* Gradient values are clipped to avoid overflow of the exponential
  update.
* Every few thousand iterations a cheap random block perturbation
  is tried; if it improves the best score we jump to it.
* If there is no improvement for many iterations we reset to a
  perturbed copy of the current best.
* After the main loop we perform a short “hard‑max” refinement
  where the soft‑max is replaced by a single delta at the maximal
  convolution index; this is a true sub‑gradient step with a
  Euclidean projection onto the simplex.

The whole procedure is bounded by the supplied time budget (default
≈960 s) and finally returns a sequence scaled to the conventional
ℓ¹‑norm √(2 n) so that `evaluate_sequence` returns the raw score
directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers C_i / 4**i (starts with 1)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution using real FFT (real inputs)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(c, 0.0, out=c)
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max:   g = 2·corr(w, α)

    α is a probability vector of length 2·n−1.
    The correlation is computed via a second FFT convolution.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)          # length = n + (2n‑1) –1 = 3n‑2
    start = w.shape[0] - 1                     # centre index that aligns with w
    return 2.0 * conv[start:start + w.shape[0]]


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random jump: multiply a random block by a log‑normal factor.
    Optional symmetric smoothing is applied with 35 % probability.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # ≈0.5 %
    hi = max(2, n // 12)           # ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:        # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 960.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # If a previous construction exists we reuse its length as a lower

    if "height_sequence_1" in globals():
        try:

            base_n = max(2000, int(prev.size))

            base_n = 8000
    else:
        base_n = 8000

    # Target length – large enough to give a good asymptotic bound but
    # still cheap enough for a few‑hundred‑thousand FFTs.
    n = max(8000, min(20000, int(1.8 * base_n)))

    print(f"[setup] target length n={n}")

    # ------------------------------------------------------------------
    # Build a small pool of initial candidates.
    # ------------------------------------------------------------------
    init_candidates = []

    # 1. Catalan numbers (the baseline used previously)
    init_candidates.append(catalan_sequence(n))

    # 2. Exponential decay, mirrored to encourage symmetry later
    exp_decay = np.exp(-np.linspace(0, 8.0, n))
    init_candidates.append(exp_decay)

    # 3. Power‑law decay (p = 0.5 is a typical choice)
    i = np.arange(1, n + 1, dtype=np.float64)
    power_law = i ** -0.5
    init_candidates.append(power_law)

    # 4. Random Dirichlet draw (concentration < 1 gives a spiky start)
    random_dir = rng.gamma(0.3, 1.0, size=n)
    init_candidates.append(random_dir)

    # 5. Interpolated previous best (if available)
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            x_old = np.arange(prev.size, dtype=np.float64)
            x_new = np.linspace(0, prev.size - 1, n)
            interp = np.interp(x_new, x_old, prev)
            init_candidates.append(interp)
        except Exception:
            pass

    # Normalise each candidate to a probability vector (ℓ¹‑norm = 1)
    init_candidates = [c / c.sum() for c in init_candidates if c.size == n]

    # ------------------------------------------------------------------
    # Pick the best starting point according to a cheap raw score.
    # ------------------------------------------------------------------
    best_raw = np.inf
    best_w = None
    for idx, w in enumerate(init_candidates):
        b = convolve_fft(w, w)
        raw = 2.0 * n * np.max(b)
        print(f"[init] candidate {idx} raw score = {raw:.6f}")

    # Sanity – if we somehow failed to create a candidate, fall back to Catalan.

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    eta0 = 5.0          # initial step size for multiplicative update
    beta_start = 0.1    # initial temperature
    beta_inc = 0.025    # linear increase per iteration
    beta_max = 300.0    # cap for the temperature
    max_grad = 5.0      # gradient clipping magnitude
    symmetrize_every = 500
    block_every = 4000
    reset_patience = 25000
    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    w = best_w.copy()
    best_score = best_raw
    best_vec = best_w.copy()
    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # --------------------------------------------------------------
        # 1. Compute self‑convolution and the current raw score.
        # --------------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_vec = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 2. Build the soft‑max probability vector α over convolution entries.
        # --------------------------------------------------------------
        beta = beta_start + beta_inc * iteration
        if beta > beta_max:
            beta = beta_max
        # shift for numerical stability

        exp_term = np.exp(beta * diff)          # max element becomes 1

        # 3. Gradient of the smoothed objective.
        # --------------------------------------------------------------

        if w_sum <= 0.0 or not np.isfinite(w_sum):

        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 5. Periodic symmetrisation – the optimal solution is symmetric.
        # --------------------------------------------------------------
        if iteration % symmetrize_every == 0:

            w /= w.sum()

        # --------------------------------------------------------------
        # 6. Infrequent cheap random block jump (helps escape plateaus).
        # --------------------------------------------------------------

            cand = cand / cand.sum()

            if cand_score < best_score:
                best_vec = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best={best_score:.6f}")

        # --------------------------------------------------------------

            delta[argmax] = 1.0
            hard_grad = correlation_grad(w, delta)
            hard_grad = np.clip(hard_grad, -max_grad, max_grad)
            eta_hard = 0.5 / np.sqrt(iteration + 1)
            w = w * np.exp(-eta_hard * hard_grad)
            w_sum = w.sum()
            if w_sum > 0.0 and np.isfinite(w_sum):
                w /= w_sum
            else:
                w = np.full_like(w, 1.0 / n)

        # --------------------------------------------------------------
        # 8. Reset if we are stuck for a long while.
        # --------------------------------------------------------------
        if no_improve > reset_patience:
            w = block_perturbation(best_vec, rng)
            w = w / w.sum()
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – the evaluator expects the conventional √(2 n) ℓ¹‑norm.
    # ------------------------------------------------------------------

    # Optional sanity check (does not influence the returned answer).
    # ------------------------------------------------------------------

        final_score = float('nan')
    print(f"[final] n={n}, best raw score={best_score:.6f}, evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
