"""
Gradient descent with smoothed max (log-sum-exp) and Adam optimizer.
Optimises the sequence on the simplex to minimise max convolution.
"""

import time, math, numpy as np

def convolve_fft(a,b):
    n = a.size + b.size - 1
    N = 1 << (n-1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    return c

def simplex_projection(v):
    # Duchi et al. algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def block_perturbation(base, rng, scale=0.15):
    # similar as previous, but with random factor and optional symmetry
    cand = base.copy()
    n = cand.size
    min_width = max(1, n // 300)
    max_width = max(2, n // 12)
    width = int(rng.integers(min_width, max_width+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.10)))  # moderate stretch/shrink
    cand[start:stop] *= factor
    if rng.random() < 0.4:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= np.sum(cand)
    return cand

def score_of_w(w):
    # w is probability vector sum=1
    n = w.size
    conv = convolve_fft(w, w)
    return float(2.0 * n * np.max(conv))

        except Exception:

    # Choose the best initial point according to true score

    best_score = np.inf

        sc = score_of_w(w)
        print(f"[init {idx}] score={sc:.6f}")

            best_score = sc
            best_w = w.copy()

    # Adam moments

The evaluation metric after normalising the ‟mass’’ to `√(2·n)` is simply  

```
score = max( w * w )
```

where `*` denotes the linear convolution.  
Thus we want a non‑negative vector on the probability simplex (`∑w = 1`) that makes the
convolution as flat as possible.

The new search combines three ideas:

1. **Smooth approximation of the `max`** – we minimise a *log‑sum‑exp* of the
   convolution (`L(w) = 1/λ·log∑exp(λ·cₖ)`) whose gradient is a weighted sum of the
   gradients of all convolution entries.  With a small λ the objective is very smooth
   (roughly the average of the convolution), and λ is increased gradually so the
   algorithm finally focuses on the true maximum.

2. **Adam optimiser on the simplex** – after each gradient step we project back onto
   the probability simplex (Euclidean projection by Duchi et al.). Adam’s adaptive
   step‑size handles the different scales of the gradient components.

3. **Escaping local minima** – every few thousand iterations a cheap block‑perturbation
   (random stretch of a contiguous block) is tried, and symmetry (`w ← (w+wᵀ)/2`) is
   enforced periodically.  A short simulated‑annealing polish finishes the run.

All heavy work (convolution and the gradient) is performed with a real FFT
(`numpy.fft.rfft`), so even a few‑ten‑thousand‑dimensional problem fits easily
into the 1000 s budget.

"""
Gradient‑descent on a smoothed max‑convolution objective.
- Uses a log‑sum‑exp approximation of the max to obtain a smooth gradient.
- Adam optimiser with Euclidean projection onto the probability simplex.
- Periodic symmetric averaging, block perturbations and a final simulated‑annealing
  polish to escape local minima.
The returned list is scaled to √(2·n) so that evaluate_sequence() can be called
directly.
"""

import time
import math
import numpy as np


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT. Returns a+b‑1 length array."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()            # next power‑of‑two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negatives from rounding
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}
       (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       scale: float = 0.12) -> np.ndarray:
    """Randomly stretch a contiguous block, optionally symmetrise."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 300)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, scale)))
    cand[start:stop] *= factor
    if rng.random() < 0.4:                     # occasional symmetry
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= np.sum(cand)
    return cand

    """Raw score = 2·n·max(conv(w,w)) – w is assumed to sum to 1."""
    n = w.size
    conv = convolve_fft(w, w)
    return float(2.0 * n * np.max(conv))
