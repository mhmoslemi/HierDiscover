"""
Propose a step‑function (non‑negative list) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is scale‑invariant, so we work with a probability vector
w (sum‑to‑one) and minimise 2·n·max(conv(w,w)).  The algorithm is a
hybrid of:

* a soft‑max Adam optimiser (smooth approximation of the ℓ∞‑norm);
* occasional sub‑gradient steps that directly attack the current peak;
* random block perturbations to escape local minima;
* several random restarts using diverse initialisations
  (uniform, cosine‑squared, Gaussian, triangular and Dirichlet draws).

The best probability vector found is finally rescaled to the conventional
√(2 n) ℓ¹‑norm before returning.  The routine respects the 1000 s time
budget and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution using a real FFT.
    Small negative round‑off errors are clipped to zero.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random jump – multiply a random contiguous block by a
    log‑normal factor and optionally symmetrise.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def compute_score(w: np.ndarray) -> float:
    """
    Return the exact value of the objective for a probability vector w:
        score = 2 * n * max(conv(w, w))
    """
    n = w.size
    conv = convolve_fft(w, w)
    return float(2.0 * n * np.max(conv))

        # --------------------------------------------------------------
        for it in range(max_iters):
            # time check – abort the inner loop if we are close to the deadline
            if time.monotonic() - start_time > budget_s - safety_margin:
                break

            # --- (1) Soft‑max Adam step --------------------------------
            conv = convolve_fft(w, w)
            maxc = float(np.max(conv))
            # soft‑max over the convolution entries
            exp_vals = np.exp(alpha * (conv - maxc))  # stabilised
            p = exp_vals / np.sum(exp_vals)           # sums to 1

            # gradient of the smooth objective:
            # ∂/∂w Σ_i p_i (w*w)[i] = 2 * corr(p, w)  (corr = convolution with reversed w)
            w_rev = w[::-1]
            grad_full = convolve_fft(p, w_rev)        # length 2n‑1
            start = n - 1
            grad = 2.0 * grad_full[start:start + n]

            # Adam update & projection onto the simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            t = it + 1
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / np.sqrt(t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            w = simplex_projection(w - step)

            # --- (2) Periodic direct sub‑gradient on the current peak ---
            if it % 500 == 0 and it > 0:
                conv = convolve_fft(w, w)
                k_max = int(np.argmax(conv))
                lo = max(0, k_max - (n - 1))
                hi = min(k_max, n - 1)
                idx = np.arange(lo, hi + 1, dtype=np.int64)
                grad_sub = np.zeros_like(w)
                grad_sub[idx] = 2.0 * w[k_max - idx]
                # decreasing step size for the sub‑gradient
                step_sub = 0.5 / np.sqrt(it + 1)
                w = simplex_projection(w - step_sub * grad_sub)

            # --- (3) Occasional random block perturbation ---------------
            if it % 1200 == 0 and it > 0:
                cand = block_perturbation(w, rng)
                cand = cand / cand.sum()
                cand_score = compute_score(cand)
                cur_score = compute_score(w)
                if cand_score < cur_score:
                    w = cand
                    m.fill(0.0)
                    v.fill(0.0)

            # --- (4) Update the global best from time to time ----------
            if it % 300 == 0:
                maybe_update(w, n, tag="inner")

            # --- (5) Anneal the soft‑max temperature --------------------
            if it % 1000 == 0 and it > 0:
                alpha = min(alpha * 1.07, 2000.0)

        # End of this restart – final evaluation.
        maybe_update(w, n, tag="restart")

    # ------------------------------------------------------------------
    # If nothing was ever found (unlikely) fall back to a uniform vector.
    # ------------------------------------------------------------------
    if best_w is None:
        best_n = 6000
        best_w = np.ones(best_n) / best_n
        best_score = compute_score(best_w)

    # ------------------------------------------------------------------
    # Rescale to the conventional √(2 n) ℓ¹‑norm before returning.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * best_n)
    final_seq = best_w * target_sum / best_w.sum()
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    # Diagnostic print – the harness will also compute the official score.
    print(f"[final] length={best_n}, best_score≈{best_score:.6f}")

    return [float(x) for x in final_seq]
