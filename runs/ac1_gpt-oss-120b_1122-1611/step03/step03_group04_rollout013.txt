import time, numpy as np

def convolve_fft(a,b):
    n = a.size + b.size - 1
    N = 1 << (n-1).bit_length()
    Fa = np.fft.rfft(a,N)
    Fb = np.fft.rfft(b,N)
    Fc = Fa*Fb
    c = np.fft.irfft(Fc,N)[:n]
    # round errors
    np.maximum(c,0,c)
    return c

def correlation_grad(w, alpha):
    # w_rev convolution with alpha (alpha length 2n-1)
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)
    start = w.size - 1
    return 2.0 * conv[start:start + w.size]

def softmax_weights(b, beta):
    maxb = np.max(b)
    diff = b - maxb
    exp_term = np.exp(beta * diff)
    alpha = exp_term / np.sum(exp_term)
    return alpha

def project_simplex(v):
    # Project onto simplex sum=1, v>=0
    # using standard algorithm
    # v may be negative; we shift.
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def random_block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renorm later
    return cand

def compute_score(w):
    conv = convolve_fft(w, w)
    return 2.0 * w.size * np.max(conv) / (np.sum(w) ** 2)

def propose_candidate(seed=42, budget_s=960.0, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine length n
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        n = prev.size
        if n < 2:
            raise Exception
        # Normalise to sum 1 (probability)
        w = prev.copy()
        w = np.maximum(w, 0)
        far = w.sum()
        if far == 0 or not np.isfinite(far):
            raise Exception
        w = w / far
    except Exception:
        n = 8000
        # We'll later pick a good shape via param sweep
        w = None

    # Parameter sweep for simple families (if w not set)
    best_w = None
    best_score = np.inf

    def evaluate_candidate(vec):
        # vec is nonnegative, not necessarily sum 1; we will compute score normalized.
        return compute_score(vec)

    if w is None:
        # Build candidate families
        # 1. Normal distribution family
        mu = (n-1)/2.0
        sigma_vals = np.logspace(np.log10(n/200), np.log10(n/5), num=30)
        for sigma in sigma_vals:
            i = np.arange(n)
            vals = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
            vals = np.maximum(vals, 0.0)
            vals_sum = vals.sum()
            if vals_sum == 0:
                continue
            vals = vals / vals_sum
            score = compute_score(vals)
            if score < best_score:
                best_score = score
                best_w = vals.copy()
        # 2. Power family
        # product distance
        i = np.arange(n)
        d = (i + 0.5) * (n - 0.5 - i)  # positive
        p_vals = np.linspace(-2.0, 2.0, 41)
        for p in p_vals:
            vals = d ** p
            # handle extreme large numbers
            if np.any(np.isinf(vals)) or np.any(np.isnan(vals)):
                # clip extreme
                vals = np.clip(vals, 0, np.max(vals[np.isfinite(vals)]))
            vals_sum = vals.sum()
            if vals_sum == 0 or not np.isfinite(vals_sum):
                continue
            vals = vals / vals_sum
            score = compute_score(vals)
            if score < best_score:
                best_score = score
                best_w = vals.copy()
        # 3. Catalan (baseline)
        # compute catalan-like sequence similar to provided
        def catalan_seq(L):
            seq = np.empty(L, dtype=np.float64)
            seq[0] = 1.0
            for ii in range(1, L):
                seq[ii] = seq[ii-1] * (2*ii-1) / (2*ii)
            return seq

        cat = catalan_seq(n)
        cat = np.maximum(cat, 0.0)
        cat = cat / cat.sum()
        score = compute_score(cat)
        if score < best_score:
            best_score = score
            best_w = cat.copy()
        # If all fails, fallback uniform
        if best_w is None:
            best_w = np.full(n, 1.0 / n)
            best_score = compute_score(best_w)

        w = best_w.copy()
        print(f"[init] n={n}, initial best score from sweep={best_score:.6f}")
    else:
        # Use given w as starting
        w = w / w.sum()
        best_w = w.copy()
        best_score = compute_score(w)
        print(f"[init] n={n}, starting from height_sequence_1, score={best_score:.6f}")

    # -------------------------------------------
    # Stage 1: Adaptive entropic gradient descent (Adam)
    # -------------------------------------------
    start_time = time.monotonic()
    # Allocate moment vectors
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.8  # base step size
    beta_base = 0.5
    beta_max = 200.0
    block_every = 3000
    sym_every = 500
    iteration = 0
    max_iter = 500000  # as fallback

    while True:
        elapsed = time.monotonic() - start_time
        if elapsed > max(0.0, float(budget_s) - 30.0):
            break
        if iteration >= max_iter:
            break

        # convolution and score
        conv = convolve_fft(w, w)
        max_conv = np.max(conv)
        score = 2.0 * n * max_conv  # sum=1

        # update best
        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            # print improvement
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")

        # softmax weights
        beta = beta_base * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        alpha = softmax_weights(conv, beta)

        # gradient
        grad = correlation_grad(w, alpha)

        # Adam update (multiplicative)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        # bias correction
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))

        # effective learning rate
        eta = eta0 / (np.sqrt(iteration + 1) + 1.0)

        # multiplicative factor
        update_factor = np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        w = w * update_factor

        # renormalise and enforce nonnegativity
        w_sum = w.sum()
        if w_sum <= 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / n)
        else:
            w = w / w_sum

        # Symmetrisation
        if iteration % sym_every == 0:
            w = 0.5 * (w + w[::-1])
            w = w / w.sum()

        # Random block perturbation on best (exploration)
        if iteration % block_every == 0 and iteration > 0:
            cand = random_block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best score={best_score:.6f}")

        iteration += 1

    # -------------------------------------------
    # Stage 2: focused local search around best_w
    # -------------------------------------------
    # Use a simple hill-climbing: try many perturbations
    start_local = time.monotonic()
    local_iter = 0
    while time.monotonic() - start_local < max(0.0, float(budget_s) - elapsed - 10.0):
        cand = random_block_perturbation(best_w, rng)
        cand = cand / cand.sum()
        cand_score = compute_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_w = cand.copy()
            print(f"[local] iter={local_iter:7d}, best score={best_score:.6f}")
        local_iter += 1

    # Final scaling
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    # ensure within bounds
    np.clip(final_seq, 0.0, 1000.0, out=final_seq)
    # Convert to list
    final_list = [float(x) for x in final_seq]
    # Optionally evaluate via provided evaluate_sequence
    try:
        final_eval = float(evaluate_sequence(final_list))
        print(f"[final] n={n}, best raw score={best_score:.6f}, evaluate={final_eval:.6f}")
    except Exception:
        print(f"[final] n={n}, best raw score={best_score:.6f}")

    return final_list

"""
Propose a step‑function (non‑negative sequence) whose ``evaluate_sequence`` score
is as low as possible.

Algorithm
---------
1. Choose a length n (reuse the previous solution if available, otherwise 8000).
2. Perform a cheap parametric sweep over a few simple families
   (Gaussian, distance‑product power, Catalan) that are normalized to sum = 1.
   Keep the best shape as the starting point.
3. Run an adaptive entropic mirror‑descent (Adam‑style) on the smoothed
   max‑convolution objective.  The smoothed max is obtained by a soft‑max
   over the linear convolution ``w*w`` with a temperature β that grows
   like √iteration (capped at 200).  The update is multiplicative:
        w ← w·exp(−η·m̂/(√v̂+ε)),
   where m̂,v̂ are Adam‑corrected moments of the gradient.
   After each update the vector is renormalised to a probability vector
   and symmetrised (w←½(w+w[::-1])).
4. Every few thousand iterations a cheap random block perturbation of the
   current best solution is tried; if it improves the score it becomes the new
   current iterate.
5. When the time budget is almost exhausted a short hill‑climbing phase
   (many random block moves) is performed to polish the result.
6. Finally the probability vector is scaled to the conventional ℓ¹‑norm
   √(2 n) (the evaluation is scale‑invariant) and returned as a plain list.

The whole routine respects the 1000 s time limit, prints progress
information, and returns a list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear (non‑circular) convolution via real FFT
# ----------------------------------------------------------------------
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (both inputs are 1‑D)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()            # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # clip tiny negatives
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max :  g = 2·corr(w, α)
    where α is a probability vector of length 2·n−1.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)       # length = 3·n−2
    start = w.size - 1
    return 2.0 * conv[start:start + w.size]


def softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution entries (β≥0)."""
    maxb = np.max(b)
    diff = b - maxb
    # all diffs ≤ 0 ⇒ exp does not overflow
    exp_term = np.exp(beta * diff)
    return exp_term / np.sum(exp_term)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v≥0)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def random_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random block by a log‑normal factor (cheap O(n) move)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # 0.5 % of the length
    hi = max(2, n // 12)           # up to ≈8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # ≈12 % log‑std
    cand[start:stop] *= factor
    if rng.random() < 0.35:       # occasional symmetric smoothing
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def compute_score(w: np.ndarray) -> float:
    """Raw score = 2·n·max(w*w) (sum(w) is assumed to be 1)."""
    conv = convolve_fft(w, w)
    return 2.0 * w.size * np.max(conv)

        cat = np.maximum(cat, 0.0)
        cat /= cat.sum()
        try_candidate(cat)

        # (d) Uniform (fallback)
        if best_w is None:
            uni = np.full(n, 1.0 / n)
            try_candidate(uni)

        w = best_w.copy()
        print(f"[init] n={n}, score after sweep={best_score:.6f}")
    else:
        # Use the provided sequence (already normalised)
        best_w = w.copy()
        best_score = compute_score(w)
        print(f"[init] n={n}, score from height_sequence_1={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Adaptive entropic (Adam) descent on the smoothed max.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    m = np.zeros_like(w)                # first moment
    v = np.zeros_like(w)                # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # hyper‑parameters (tuned empirically)
    eta0 = 1.2               # base learning‑rate for the multiplicative step
    beta_base = 0.5
    beta_max = 200.0
    sym_every = 200          # enforce symmetry frequently
    block_every = 2000       # try a cheap random jump occasionally

    iteration = 0
    max_iter = 500_000       # safety cap

    while True:
        elapsed = time.monotonic() - start_time
        if elapsed > max(0.0, float(budget_s) - 5.0):
            break
        if iteration >= max_iter:
            break

        # ----- (a) current convolution and score -----
        conv = convolve_fft(w, w)
        max_conv = np.max(conv)
        score = 2.0 * n * max_conv          # sum(w) == 1

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")

        # ----- (b) soft‑max over the convolution entries -----
        beta = beta_base * np.sqrt(iteration + 1)
        if beta > beta_max:
            beta = beta_max
        alpha = softmax_weights(conv, beta)   # length = 2·n−1, sums to 1

        # ----- (c) gradient of the smoothed objective -----
        grad = correlation_grad(w, alpha)

        # ----- (d) Adam moment updates -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # ----- (e) multiplicative (entropic) update -----
        eta = eta0 / (np.sqrt(iteration + 1) + 1.0)
        factor = np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))
        w = w * factor

        # project back onto the simplex
        s = w.sum()
        if s <= 0.0 or not np.isfinite(s):
            w = np.full_like(w, 1.0 / n)
        else:
            w = w / s

        # ----- (f) periodic symmetrisation -----
        if iteration % sym_every == 0:
            w = 0.5 * (w + w[::-1])
            w = w / w.sum()

        # ----- (g) cheap random block jump (escape plateaus) -----
        if iteration % block_every == 0 and iteration > 0:
            cand = random_block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best score={best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 2. Local hill‑climbing (many cheap random moves) while time permits.
    # ------------------------------------------------------------------
    time_remaining = max(0.0, float(budget_s) - (time.monotonic() - start_time) - 2.0)
    if time_remaining > 0.5:
        start_local = time.monotonic()
        local_iter = 0
        while time.monotonic() - start_local < time_remaining:
            cand = random_block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_w = cand.copy()
                print(f"[local] iter={local_iter:7d}, best score={best_score:.6f}")
            local_iter += 1

    # ------------------------------------------------------------------
    # 3. Final scaling – the evaluation is scale‑invariant, but we keep the
    #    conventional √(2 n) ℓ¹‑norm for readability.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    np.clip(final_seq, 0.0, 1000.0, out=final_seq)

    # optional sanity check (the harness will call evaluate_sequence itself)
    try:
        final_eval = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, raw best score={best_score:.6f}, evaluate={final_eval:.6f}")
    except Exception:
        print(f"[final] n={n}, raw best score={best_score:.6f}")

    return [float(x) for x in final_seq]
```

This implementation augments the earlier entropic gradient descent with:

* a quick parametric sweep to obtain a better starting distribution,
* Adam‑style adaptive step‑sizes,
* more aggressive symmetrisation and block perturbations,
* a final short hill‑climbing phase.

All operations respect the non‑negative, ℓ¹‑normalised constraints, run within the 1000 s budget, and return a list ready for ``evaluate_sequence``.
