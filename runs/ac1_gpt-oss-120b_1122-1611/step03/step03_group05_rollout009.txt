"""
propose_candidate
-----------------

A two‑phase optimizer for the self‑convolution peak problem.

Phase 1 – smooth log‑sum‑exp surrogate.
    A temperature β is increased from 2 to 200 (homotopy) while an
    Adam optimizer minimizes the smooth approximation
        L_β(w) = (1/β)·log∑_k exp(β·(w*w)[k]).
    The gradient of L_β is obtained by an FFT‑based correlation.
    The iterate stays on the probability simplex (∑w=1, w≥0) by
    Euclidean projection; after each step we symmetrise the vector
    (w←(w+w[::-1])/2) because the optimal shape is expected to be
    symmetric.

Phase 2 – exact sub‑gradient polishing.
    After β has reached its maximal value we switch to the true
    objective f(w)=max_k (w*w)[k].  The sub‑gradient of f at w is
    2·w mirrored around each active (maximal) convolution index.
    Adam with a smaller learning rate is used again; occasional
    random block perturbations (multiplying a contiguous block by a
    factor ≈±12 %) are applied to escape stagnation.  The best vector
    (smallest peak) seen so far is kept and finally scaled to the
    conventional ℓ¹‑norm √(2 n) required by ``evaluate_sequence``.

The optimizer runs until the time budget (≈1000 s) is exhausted and
returns a plain Python list suitable for the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan numbers divided by 4**i.
    Recurrence a_{i+1}=a_i·(2i+1)/(2i+2) is numerically stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    # Duchi et al. (2008) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random perturbation – multiply a random block by a
    log‑normal factor (≈±12 %) and (with prob. 0.35) symmetrise."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)           # 0.5 % of length
    hi = max(2, n // 12)            # up to ~8 %
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # exp(N(0,0.12²))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def self_convolve_fft(w: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution w*w using an FFT of length N.
    Returns the (2·len(w)‑1) full result, with tiny negatives clipped."""
    Fw = np.fft.rfft(w, N)
    conv_full = np.fft.irfft(Fw * Fw, N)[: 2 * w.size - 1]
    np.maximum(conv_full, 0.0, out=conv_full)   # remove numeric noise
    return conv_full


def gradient_smooth(w: np.ndarray, probs: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate L_β at w.
    ``probs`` are the soft‑max weights (sum = 1, length = 2·len(w)‑1).
    Returned gradient has the same length as w.
    """
    # correlation w ⋆ probs (shifted) → 2·∑ probs_i·w_{i‑j}
    Fw = np.fft.rfft(w, N)
    Fs = np.fft.rfft(probs, N)
    corr_full = np.fft.irfft(Fw * np.conj(Fs), N)
    m = probs.size                # = 2·len(w)‑1
    offset = m - 1
    grad = 2.0 * corr_full[offset: offset + w.size]
    return grad


def subgradient_max(w: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of f(w)=max_k (w*w)[k] at w.
    ``max_idx`` contains all indices where the maximum is attained.
    The sub‑gradient returned is the average of the individual
    gradients 2·w shifted by each active index.
    """
    n = w.size
    g = np.zeros_like(w)
    for k in max_idx:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=np.intp)
        # contribution w_{k‑i}
        g[idx] += w[k - idx]
    if max_idx.size > 0:
        g = 2.0 * g / max_idx.size
    return g

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function with a small self‑convolution
    peak.  Returns a plain Python list of floats ready for evaluate_sequence.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size and initial vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init_seq = np.asarray(height_sequence_1, dtype=np.float64)
            if init_seq.ndim != 1 or init_seq.size < 2:
                raise ValueError
            n = int(init_seq.size)
            w = np.maximum(init_seq, 0.0)
            if np.sum(w) < 1e-12:
                raise ValueError
            w = w / np.sum(w)                     # normalise to ℓ¹‑norm = 1
        except Exception:
            n = 10000
            w = catalan_sequence(n)
            w = w / np.sum(w)
    else:
        n = 10000
        w = catalan_sequence(n)
        w = w / np.sum(w)

    # enforce symmetry – the optimum is expected to be symmetric
    w = 0.5 * (w + w[::-1])

    # --------------------------------------------------------------
    # Pre‑computed FFT lengths for the two convolutions we need.
    # --------------------------------------------------------------
    N_self = _next_pow2(2 * n - 1)          # for self‑convolution
    N_corr = _next_pow2(3 * n - 2)          # for correlation (gradient)

    # --------------------------------------------------------------
    # Optimisation state.
    # --------------------------------------------------------------
    best_w = w.copy()
    b = self_convolve_fft(w, N_self)
    best_score = float(np.max(b))          # max convolution when Σw = 1
    print(f"[init] n={n}, initial max‑conv={best_score:.9f}")

    # Adam hyper‑parameters
    lr_smooth = 0.07
    lr_subgrad = 0.02
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # β‑homotopy parameters for the smooth phase
    beta_start = 2.0
    beta_end = 200.0            # larger than in the reference code
    # timing allocation
    total_budget = max(0.0, float(budget_s) - 5.0)   # keep a few seconds for finalisation
    start_time = time.monotonic()
    smooth_time_frac = 0.60                         # 60 % of time for phase 1
    smooth_time_limit = start_time + smooth_time_frac * total_budget

    iteration = 0
    no_improve = 0
    block_interval = 5000
    reset_patience = 30000

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < total_budget:
        # ----- common work: self‑convolution and current peak -----
        b = self_convolve_fft(w, N_self)
        cur_max = float(np.max(b))

        # keep the best ever seen
        if cur_max < best_score - 1e-14:
            best_score = cur_max
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, max‑conv={best_score:.9f}")
        else:
            no_improve += 1

        # ----- decide which phase we are in -----
        now = time.monotonic()
        if now < smooth_time_limit:                     # ---------- Phase 1 ----------
            # homotopic β
            beta = beta_start + (beta_end - beta_start) * (
                (now - start_time) / (smooth_time_limit - start_time)
            )
            # soft‑max over convolution values
            shift = beta * (b - cur_max)                # ≤ 0
            exp_shift = np.exp(shift)                  # max = 1
            probs = exp_shift / np.sum(exp_shift)      # length = 2n‑1, sum = 1
            # gradient of the smooth surrogate
            grad = gradient_smooth(w, probs, N_corr)
            lr = lr_smooth
        else:                                            # ---------- Phase 2 ----------
            # exact sub‑gradient of the max
            max_idx = np.flatnonzero(np.abs(b - cur_max) <= 1e-12)
            grad = subgradient_max(w, max_idx)
            lr = lr_subgrad

        # ----- Adam update with projection onto the simplex -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_new = w - step
        w = simplex_projection(w_new)

        # enforce symmetry (helps keep the search in a smaller subspace)
        w = 0.5 * (w + w[::-1])

        # ----- occasional random block perturbation (jump) -----
        if iteration % block_interval == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_b = self_convolve_fft(cand, N_self)
            cand_max = float(np.max(cand_b))
            if cand_max < best_score:
                best_w = cand.copy()
                best_score = cand_max
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best max‑conv={best_score:.9f}")

        # ----- reset if we have been stuck for a long time -----
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

    # --------------------------------------------------------------
    # Final scaling – convert the best probability vector to the
    # conventional √(2 n) ℓ¹‑norm.  After this scaling evaluate_sequence
    # returns exactly 2·n·max_b, i.e. the value printed as “best_score”.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)   # sum should already be 1

    # Optional sanity check (cheap)
    try:
        final_val = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] n={n}, best max‑conv={best_score:.9f}, "
              f"after scaling evaluate={final_val:.9f}")
    except Exception:
        # the evaluator may not be importable in the offline setting
        pass

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
