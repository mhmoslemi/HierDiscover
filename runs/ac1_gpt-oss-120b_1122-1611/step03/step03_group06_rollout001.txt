"""
Propose a candidate step function using a two‑phase Adam optimiser.

Phase 1 (warm‑up): minimise the L2 distance between the self‑convolution
and the uniform distribution (the “perfect” flat convolution”).  This
produces a probability vector whose autocorrelation is already close to
uniform, giving a good starting point for the subsequent phase.

Phase 2 (refinement): minimise a smooth approximation of the
∞‑norm ‑ the log‑sum‑exp (soft‑max) of the convolution values – using
the Adam optimiser and periodic symmetry enforcement.  The soft‑max
parameter β is slowly increased so that the surrogate becomes the true
max.  Occasional block perturbations and resets help escape local minima.

All vectors are kept on the probability simplex (non‑negative,
Σ w = 1) by Euclidean projection.  At the end the best probability vector
found is scaled to the required L¹‑norm √(2 n) and returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – defined at top level.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan‑type coefficients a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using real FFT.
    Equivalent to np.convolve(a, b) but O(n log n) for large n.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negative round‑off errors
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Random perturbation used as a “jump’’.
    A random contiguous block is multiplied by a log‑normal factor and,
    with some probability, the whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def enforce_symmetry(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry: v = (v + rev(v))/2."""
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Main function required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step function (list of floats) whose
    `evaluate_sequence` value is as small as possible.
    The routine performs a two‑phase Adam optimiser:
      * Phase 1 – L2‑norm minimisation of the convolution w.r.t. a uniform target.
      * Phase 2 – Log‑sum‑exp (soft‑max) minimisation of the max convolution value.
    Periodic symmetry enforcement, random block perturbations and resets
    are used to escape shallow minima.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose the length of the sequence.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(seq0.size)
            if n < 2:
                raise ValueError
            w = seq0 / np.sum(seq0)                     # normalise to a probability vector
        except Exception:
            # fallback to a longer Catalan‑type start
            n = 8000
            a = catalan_sequence(n)
            w = a / np.sum(a)
    else:
        n = 8000
        a = catalan_sequence(n)
        w = a / np.sum(a)

    # make sure we start symmetric (many good constructions are symmetric)
    w = enforce_symmetry(w)

    # --------------------------------------------------------------
    # Optimiser state
    # --------------------------------------------------------------
    best_w = w.copy()
    b_full = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b_full)               # this equals evaluate_sequence when scaled
    print(f"[init] length={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    max_time = max(0.0, float(budget_s) - 5.0)          # keep a few seconds for finalisation

    # Adam hyper‑parameters (common for both phases)
    lr = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    # Phase‑1 / Phase‑2 split (by elapsed time)
    phase1_fraction = 0.22                               # roughly 22 % of the budget for warm‑up
    phase1_deadline = start_time + phase1_fraction * max_time

    # Parameters for the soft‑max surrogate (Phase 2)
    beta = 5.0                                            # start with a moderate smoothing
    beta_max = 250.0
    beta_growth = 1.035
    beta_inc_interval = 3000

    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < max_time:
        # --------------------------------------------------------------
        # Full self‑convolution of the current probability vector
        # --------------------------------------------------------------
        b_full = convolve_fft(w, w)
        max_b = np.max(b_full)
        current_score = 2.0 * n * max_b

        # --------------------------------------------------------------
        # Record best solution so far
        # --------------------------------------------------------------
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Choose the phase (warm‑up vs refinement)
        # --------------------------------------------------------------
        t_now = time.monotonic()
        if t_now < phase1_deadline:
            # ----- Phase 1 : L2 loss against a uniform convolution -----
            target = np.full_like(b_full, 1.0 / (2 * n - 1), dtype=np.float64)
            residual = b_full - target
            loss = np.sum(residual * residual)                # not used for anything but monitoring
            # gradient of ½‖residual‖² = (residual) * (∂b/∂w) ; ∂b/∂w = 2*(residual * w) convolution
            grad = 4.0 * convolve_fft(residual, w)[:n]

        else:
            # ----- Phase 2 : soft‑max approximation of the max -----
            max_b = np.max(b_full)                           # recompute for numerical stability
            # soft‑max weights
            exp_arg = beta * (b_full - max_b)                # ≤ 0
            p = np.exp(exp_arg)
            sum_p = np.sum(p)
            if sum_p == 0.0:                                 # underflow protection
                p = np.full_like(b_full, 1.0 / b_full.size)
                sum_p = 1.0
            p /= sum_p
            # gradient of the smooth max: 2·(p * w)[:n]
            grad = 2.0 * convolve_fft(p, w)[:n]

            # slowly increase the soft‑max temperature
            if iteration % beta_inc_interval == 0 and beta < beta_max:
                beta = min(beta_max, beta * beta_growth)

        # --------------------------------------------------------------
        # Adam update (projected onto the simplex)
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        w_candidate = w - step
        # enforce simplex constraints
        w = simplex_projection(w_candidate)

        # --------------------------------------------------------------
        # Encourage symmetry – many good constructions are symmetric.
        # --------------------------------------------------------------
        w = enforce_symmetry(w)

        # --------------------------------------------------------------
        # Random block perturbation to escape stagnation
        # --------------------------------------------------------------
        if iteration % 4000 == 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            w = cand.copy()
            m.fill(0.0)
            v.fill(0.0)
            print(f"[perturb] iter={iteration}")

        # --------------------------------------------------------------
        # Hard reset if no improvement for a long while
        # --------------------------------------------------------------
        if no_improve > 120000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration}")

    # ------------------------------------------------------------------
    # Final scaling – convert the best probability vector to the required
    # L¹‑norm √(2 n).  The value of `evaluate_sequence` after this scaling
    # equals `best_score`, but we invoke it once for safety.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum                     # Σ final_seq = √(2 n)

    # Safety call – will clamp negatives / infinities inside evaluate_sequence
    final_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"[final] length={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Store the best found sequence globally for possible future restarts.
    global height_sequence_1
    height_sequence_1 = final_seq.tolist()

    # Return a plain Python list.
    return [float(x) for x in final_seq]
