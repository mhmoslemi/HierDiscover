"""
propose_candidate – soft‑max mirror descent with full gradient

The score that must be minimised is  

    S(a) = 2·n·max(conv(a,a)) / (∑a)² ,

and the evaluation routine normalises the vector to a simplex
(sum = 1) before the computation.  Consequently we may keep the
candidate always on the probability simplex and minimise  

    f(a) = 2·n·max(conv(a,a))     (∑a = 1).

The algorithm works with a *temperature* β that slowly grows.
For a given β the soft‑max weights  

    w_k ∝ exp(β·(c_k – max(c))) ,   c = a∗a ,

smoothly approximate the indicator of the maximal convolution
entry.  The gradient of the smoothed objective is

    ∇f(a) ≈ 2·∑_k w_k·∂conv_k/∂a
          = 2·( w  *  a_rev )[n‑1 : n‑1+n] ,

where “*” is a linear convolution and a_rev is a reversed copy of a.
Both the self‑convolution and the gradient are obtained with a
single FFT each, i.e. O(n log n).  An exponentiated‑gradient (mirror
descent on the simplex with the negative entropy regulariser) updates
the probabilities:

    a ← a ⊙ exp( –η·∇f(a) )      (⊙ is element‑wise product)

followed by a re‑normalisation.  The stepsize η is linearly
decreased, while β is linearly increased, so the method first shapes
the whole distribution and later concentrates on the current peak.
To avoid stagnation we occasionally

* apply a multiplicative log‑normal jitter on a random contiguous
  block,
* swap two random coordinates,
* perform a soft‑restart from the best solution seen so far with a
  small log‑normal jitter when no improvement occurs for ≈30 s.

All operations are vectorised; with n≈4 k a single iteration costs
only a few milliseconds, so tens of thousands of iterations comfortably
fit into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a using an FFT."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[: 2 * n - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:n]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initial vector – use the provided construction if it looks sane.
    # ------------------------------------------------------------------
    default_n = 512
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                a = np.clip(cand, 0.0, None)
                s = a.sum()
                if s > 0.0:
                    a = a / s
                else:
                    raise ValueError
            else:
                raise ValueError
        except Exception:
            n = default_n
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = default_n
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size  # final length

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best)
    best_score = 2.0 * n * conv_best.max()
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 1.0          # initial mirror‑descent stepsize
    eta_end = 0.001          # final stepsize (fine‑tuning)
    beta_start = 0.1         # start temperature (very smooth)
    beta_end = 800.0         # final temperature (near hard max)
    momentum = 0.85          # exponential smoothing of the gradient
    block_perturb_prob = 0.04
    swap_prob = 0.05
    restart_gap = 30.0       # seconds without improvement → restart
    jitter_sigma = 0.13      # log‑normal jitter during restart

    grad_prev = np.zeros_like(a)

    # Pre‑compute FFT lengths (only depend on n)
    fft_len_self = _next_pow2(2 * n - 1)          # for a∗a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w∗a_rev

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- self‑convolution and current score -----------------------
        conv = _conv_self(a)                      # length 2n‑1
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max             # sum(a) == 1 by construction

        # ---- keep the best solution ----------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- schedule (linear interpolation) -------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- soft‑max weights for the current temperature ------------
        # Shift by the maximum to avoid overflow.
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            # Very large beta could under‑flow everything; fall back to uniform.
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smoothed objective (full soft‑max) -------
        a_rev = a[::-1]                           # reversed copy
        # convolution w * a_rev  (length 3n‑2)
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_rev = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: (len(w) + len(a_rev) - 1)]
        # slice the part that corresponds to the gradient
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # length n

        # ---- momentum smoothing ---------------------------------------
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- exponentiated‑gradient (mirror‑descent) update -----------
        # Clip the exponent to avoid overflow/underflow.
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -50.0, 50.0)
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- occasional random perturbations ---------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            a[i], a[j] = a[j], a[i]   # simple coordinate swap

        # ---- soft restart when progress stalls -------------------------
        if now - last_improve > restart_gap:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best solution (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
