"""
Propose_candidate uses an exponentiated‑gradient (mirror descent) method
with a soft‑max approximation of the maximum self‑convolution.
The full convolution and the gradient are computed with FFTs.
A temperature (β) is increased and the stepsize decreased linearly,
random block perturbations are occasionally applied and a soft‑restart
jittering is performed when progress stalls.
The algorithm works for any length; it starts from the provided
``height_sequence_1`` if present, otherwise from a uniform distribution.
"""

import time
import numpy as np

def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start+width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _score(a: np.ndarray) -> float:
    n = a.size
    conv = _conv_self(a, _fft_len_for_n(n))
    return float(2.0 * n * np.max(conv) / (np.sum(a) ** 2))

def _grad_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute grad_i = sum_k w_k * a_{k-i}
    using convolution of w and the reversed a via FFT.
    The result slice corresponds to indices i = 0 .. n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    full = np.fft.irfft(fw * fr, n=fft_len)
    n = a.size
    start = n - 1
    return full[start:start + n]

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Return a non‑negative step‑function that (hopefully) improves the best score.
    
    The routine runs for at most *budget_s* seconds, performing a mirror‑descent
    optimisation on a soft‑max approximation of the maximal self‑convolution.
    """
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1) initialise the sequence (use supplied one if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if init.ndim == 1 and init.size > 0 and np.isfinite(init).all():
            a = np.clip(init, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(init.size, 1.0 / init.size)
        else:
            a = None
    else:
        a = None

    if a is None:
        # fall back to a uniform vector of moderate size
        n0 = 4096
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size
    fft_len_self = _fft_len_for_n(n)
    # for the gradient we need the convolution length (2n‑1) + n‑1 = 3n‑2
    fft_len_grad = 1 << ((3 * n - 2 - 1).bit_length())

    # ------------------------------------------------------------------
    # 2) bookkeeping of the best solution seen so far
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = _score(best_a)
    print(f"[Init] n={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 0.9          # initial stepsize
    eta_end = 0.001          # final stepsize
    beta_start = 0.5         # temperature start (soft‑max softness)
    beta_end = 600.0         # temperature end (sharper focus)
    momentum = 0.9
    block_perturb_prob = 0.04
    restart_gap = 45.0       # seconds without improvement before a soft restart

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ----- current convolution and score ---------------------------------
        conv = _conv_self(a, fft_len_self)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv / (np.sum(a) ** 2)   # sum(a) stays 1

        # ----- keep the best solution ----------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- schedules ------------------------------------------------------
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ----- soft‑max weights ------------------------------------------------
        # all entries are used; the exponential automatically down‑weights the
        # non‑maximal ones.
        w = np.exp(beta * (conv - max_conv))

        # ----- gradient (via FFT) --------------------------------------------
        grad = _grad_from_weights(a, w, fft_len_grad)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- mirror‑descent update (exponentiated gradient) -----------------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random block perturbation --------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----- soft restart when progress stalls -------------------------------
        if now - last_improve > restart_gap:
            # jitter the current best and continue from there
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]

    # 3) hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 0.9          # initial mirror‑descent stepsize
    eta_end = 0.001          # final stepsize (fine tuning)
    beta_start = 0.5         # temperature start – very soft soft‑max
    beta_end = 600.0         # temperature end – almost a hard max
    momentum = 0.9           # exponential smoothing of the gradient
    block_perturb_prob = 0.04
    restart_gap = 45.0       # seconds without improvement → soft restart

    grad_prev = np.zeros_like(a)

    # ------------------------------------------------------------------
    # 4) main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ---- current convolution and score ---------------------------------
        conv = _conv_self(a, fft_len_self)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv / (np.sum(a) ** 2)   # sum(a) stays 1

        # ---- keep the best solution ----------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- linear schedules for η and β ----------------------------------
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ---- soft‑max weights (normalised) ---------------------------------
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:                # should never happen, but be safe
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ---- gradient via FFT (convolution w * rev(a)) --------------------
        grad = _grad_from_weights(a, w, fft_len_grad)

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- mirror‑descent (exponentiated‑gradient) update ---------------
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ---- occasional random block perturbation -------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ---- soft restart when progress stalls -----------------------------
        if now - last_improve > restart_gap:
            # jitter the current best and continue from there
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            grad_prev = np.zeros_like(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
