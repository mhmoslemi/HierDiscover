def correlation_fft(weights, w):
    # w length n, weights length 2n-1
    n = w.shape[0]
    m = weights.shape[0]  # 2n-1
    # want r[k] = sum_j weights[j] * w[j + k], for k=0..n-1
    # Equivalent to conv of weights and w_rev, then take indices (n-1 - k)
    w_rev = w[::-1]
    N = 1 << ((m + n - 1).bit_length())   # next power of two >= len(weights)+len(w_rev)-1
    Fw = np.fft.rfft(w_rev, N)
    Fwt = np.fft.rfft(weights, N)
    conv = np.fft.irfft(Fwt * Fw, N)
    # conv length = m + n - 1 = 3n - 2
    # retrieve indices:
    start = n - 1  # conv index for k=0?
    # Let's verify:
    # For k=0, r[0] = sum_j weights[j] * w[j]; conv index i = (n-1)?? compute conv[i] = sum_j weights[j] * w_rev[i - j].
    # w_rev[i - j] = w[n-1 - (i - j)] = w[n-1-i + j].
    # To get w[j], we need n-1-i + j = j => n-1-i = 0 => i = n-1. So conv[n-1] yields sum_j weights[j] * w[j].
    # So r[0] = conv[n-1].
    # For k, need w[j+k]. Condition: w_rev[i - j] = w[n-1 - (i - j)] = w[j + k] => n-1-i+j = j + k => n-1-i = k => i = n-1 - k.
    # So r[k] = conv[n-1 - k].
    # So need conv indices from n-1 down to 0.
    grad = 2.0 * conv[(n-1) - np.arange(n)]
    return grad

"""
Propose a step‑function candidate using a projected Adam optimiser
on a smooth Lp–approximation of the L∞‐norm of the self‑convolution.
The algorithm works on the probability simplex (∑w = 1, w ≥ 0);
after optimisation the vector is scaled to √(2·n) – the evaluation
function is invariant to that scaling.  The optimiser uses a
temperature‑controlled power weighting of the convolution entries,
symmetrises the vector every few iterations and occasionally applies
large random block perturbations to escape local minima.  A short
simulated‑annealing polishing phase finishes the run.

The implementation respects the 1000 s time budget, prints
intermediate scores and returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan numbers
    a_i = C(2i,i)/4**i using the stable recurrence."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2.0 * (i - 1) + 1.0) / (2.0 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # tiny negatives from round‑off are harmless for the max, but we clip
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0,∑x=1}
    (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def correlation_fft(weights: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute r[k] = Σ_j weights[j] * w[j+k]   (k = 0 … n‑1)
    using an FFT‑based convolution.  The returned gradient is
    2·r (the sub‑gradient of Σ_j w_j·w_{k‑j} at the active
    convolution entry).
    """
    n = w.shape[0]
    # reversed copy of w
    w_rev = w[::-1]
    m = weights.shape[0]               # = 2·n‑1
    N = 1 << ((m + n - 1).bit_length())  # power‑of‑two size for FFT
    Fw_rev = np.fft.rfft(w_rev, N)
    Fwt = np.fft.rfft(weights, N)
    conv = np.fft.irfft(Fw_rev * Fwt, N)[: m + n - 1]   # length 3·n‑2
    # gradient entry k is 2·conv[n‑1‑k]  (see derivation in analysis)
    idx = (n - 1) - np.arange(n)
    grad = 2.0 * conv[idx]
    # eliminate possible tiny negatives from round‑off
    np.maximum(grad, 0.0, out=grad)
    return grad


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Large cheap O(n) random jump – used for occasional restarts."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
