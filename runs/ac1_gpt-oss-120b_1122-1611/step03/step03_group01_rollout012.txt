class Objective:
    def __init__(self, n, lam):
        self.n = n
        self.lam = lam

    def __call__(self, u):
        # compute w
        max_u = np.max(u)  # for numerical stability
        exp_u = np.exp(u - max_u)
        w = exp_u / exp_u.sum()
        # compute conv = convolve_fft(w,w)
        conv = convolve_fft(w, w)
        # compute logsumexp
        lam_conv = self.lam * conv
        max_lc = np.max(lam_conv)
        # To avoid overflow, subtract max
        exp_terms = np.exp(lam_conv - max_lc)
        sum_exp = np.sum(exp_terms)
        logsumexp = max_lc + np.log(sum_exp)
        f = logsumexp / self.lam  # smooth max

        # compute p = exp(lam_conv - logsumexp)  = exp(lam_conv - max_lc) / sum_exp
        p = exp_terms / sum_exp

        # gradient
        # w_rev = w[::-1]
        # conv_pw_rev = convolve_fft(p, w[::-1]) -> length len(p)+len(w)-1 = (2n-1)+n-1 = 3n-2
        # compute slice start = self.n - 1, end = start + self.n
        gp = convolve_fft(p, w[::-1])
        grad_w = gp[self.n-1:self.n-1+self.n]

        # compute mu = (w * grad_w).sum()
        mu = np.dot(w, grad_w)
        grad_u = w * (grad_w - mu)  # vector

        return f, grad_u

"""
Algorithm
---------

Search for a non‑negative step function that minimises
`evaluate_sequence`.  The idea is:

1.  Choose a length `n` (taken from `height_sequence_1` if present,
    otherwise a default of 8000).
2.  Produce several diverse probability vectors (Catalan, arcsine law,
    random Dirichlet, optionally the previous best).
3.  Refine each start with a smooth‑max L‑BFGS optimisation.
    The true objective `max_i (w*w)[i]` is replaced by a
    log‑sum‑exp approximation `logsumexp(λ·conv)/λ`.  A soft‑max
    parametrisation (`w = softmax(u)`) guarantees the simplex constraints.
    The gradient of the smoothed objective is obtained analytically by
    a second convolution, so each function/gradient evaluation costs two
    FFT‑based convolutions.
4.  Run the L‑BFGS optimisation with an increasing schedule for `λ`
    (5 → 10 → 20 → 40 → 80) which gradually sharpens the approximation to
    the true max.
5.  Keep the best probability vector found; enforce symmetry
    (the optimum is typically symmetric).
6.  Perform a short projected sub‑gradient polishing phase (the same
    scheme that was used in the previous baseline) to exploit any
    remaining improvement opportunities.
7.  Finally scale the vector to ℓ¹‑norm `√(2·n)` and return it as a plain
    Python list.

The whole search respects the 1000 s time budget; intermediate progress
(and current best scores) is printed for diagnostics.

# Basic linear‑convolution routine (real FFT, fast for n≈10⁴)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (returns non‑negative result)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # drop tiny negatives
    return c

# Analytic seed generators
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First `length` Catalan numbers a_i = C(2i,i)/4**i."""

    seq[0] = 1.0

# ----------------------------------------------------------------------
# Smooth‑max objective (log‑sum‑exp of the convolution) and its gradient
# ----------------------------------------------------------------------


class SmoothMaxObjective:
    """
    Callable class used by SciPy's L‑BFGS.  For a given λ it returns
    a smooth approximation of `max_i (w*w)[i]` together with its gradient
    w.r.t. the soft‑max parameters `u` (where w = softmax(u)).
    """

    def __init__(self, n: int, lam: float):
        self.n = n
        self.lam = lam

    def __call__(self, u: np.ndarray):
        # ----- soft‑max to obtain a probability vector -----------------
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        w = exp_u / exp_u.sum()                     # w ≥ 0, ∑w = 1

        # ----- convolution w * w ------------------------------------
        conv = convolve_fft(w, w)                   # length 2n‑1

        # ----- smooth max via log‑sum‑exp ----------------------------
        lam_conv = self.lam * conv
        lse = logsumexp(lam_conv)                   # log(∑ exp(λ·conv))
        f = lse / self.lam                          # smooth max

        # ----- gradient of the smooth max w.r.t. the convolution -----
        # p_i = exp(λ conv_i) / ∑exp(λ conv)
        p = np.exp(lam_conv - lse)                  # already normalised

        # ----- chain rule: ∂f/∂w = ∑_i p_i ∂conv_i/∂w ---------------
        # ∂conv_i/∂w_k = w_{i‑k}, i.e. a correlation.
        # Compute via convolution with the reversed w.
        w_rev = w[::-1]
        corr = convolve_fft(p, w_rev)              # length 3n‑2
        grad_w = corr[self.n - 1:self.n - 1 + self.n]  # size n

        # ----- chain rule through the soft‑max parametrisation -------
        mu = np.dot(w, grad_w)                      # scalar ⟨w,∂f/∂w⟩
        grad_u = w * (grad_w - mu)                  # ∂f/∂u (same shape as u)

        return f, grad_u

    w = best_w.copy()
    w_prev = w.copy()
    eta0 = 0.5
    momentum = 0.8
    it = 0
    stagn = 0

    while time.monotonic() - start_time < max_time:
        conv = convolve_fft(w, w)
        max_val = np.max(conv)
        idxs = np.flatnonzero(conv >= max_val - 1e-15)
        i_max = int(rng.choice(idxs))

        # sub‑gradient of the active quadratic term
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        idx = i_max - j
        grad[j] = 2.0 * w[idx]

        eta = eta0 / np.sqrt(it + 1)
        w_tmp = w - eta * grad + momentum * (w - w_prev)
        w_new = simplex_projection(w_tmp)

        # occasional mirror‑descent variant
        if rng.random() < 0.3:
            factor = np.exp(-eta * grad)
            w_mir = w * factor
            w_mir = w_mir / w_mir.sum()
            if np.max(convolve_fft(w_mir, w_mir)) < np.max(convolve_fft(w_new, w_new)):
                w_new = w_mir

        w_prev = w
        w = w_new

        s = raw_score(w)
        if s < best_score - 1e-12:
            best_score = s
            best_w = w.copy()
            stagn = 0
            print(f"[polish] iter={it} new best raw score = {best_score:.6f}")
        else:
            stagn += 1

        # periodic large random jump (helps escape shallow plateaus)
        if it % 3000 == 0 and it > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / cand.sum()
            sc = raw_score(cand)
            if sc < best_score:
                best_score = sc
                best_w = cand.copy()
                w = cand.copy()
                w_prev = w.copy()
                print(f"[jump] iter={it} new best raw score = {best_score:.6f}")

        # a long streak without improvement → reset to a perturbed best
        if stagn > 150_000:
            w = block_perturbation(best_w, rng)
            w = w / w.sum()
            w_prev = w.copy()
            stagn = 0
            print(f"[reset] iter={it}")

        it += 1

    # ------------------------------------------------------------------
    # Scale to the norm required by `evaluate_sequence` (ℓ¹ = √(2·n))
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best raw score = {best_score:.6f}, "
          f"evaluate_sequence = {final_score:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in final_seq]
