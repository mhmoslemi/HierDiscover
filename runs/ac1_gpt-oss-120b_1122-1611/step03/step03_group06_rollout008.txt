    best_w = initial_w.copy()
    best_score = compute_score(initial_w)
    w = initial_w.copy()

    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta_smooth = start_beta
    for iter in range(max_iter):
        # compute b, p, grad
        b = convolve_fft(w, w)
        max_b = np.max(b)
        # softmax weights
        exp_arg = beta_smooth * (b - max_b)
        p = np.exp(exp_arg)
        p_sum = np.sum(p)
        if p_sum == 0:
            p = np.ones_like(p) / p.size
        else:
            p = p / p_sum
        grad = 2.0 * convolve_fft(p, w)[:w.size]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**(iter+1))
        v_hat = v / (1 - beta2**(iter+1))

"""
Propose a step‑function sequence that (approximately) minimizes
    2·n·max_k (w*w)[k]   with   w_i ≥ 0 , Σ w_i = 1

The algorithm is a two‑phase projected Adam optimiser:

* Phase 1 – “soft‑max” smoothing.
  A smooth surrogate  Sβ(w) = (1/β)·log∑exp(β·(w*w)[k])  is minimised.
  The gradient is   ∇S = 2·(p*w)[:n]  where p_k are the soft‑max
  weights of the convolution values.  Adam updates are applied and
  the iterate is projected onto the probability simplex.

* Phase 2 – “hard‑max” sub‑gradient.
  When the time budget is almost exhausted the surrogate is replaced
  by the true max: p is the uniform distribution on the indices where
  (w*w) attains its maximum.  This refines the solution towards the
  exact objective.

Additional tricks that helped to beat the previous 1.517874 bound:
    – automatic schedule for the smoothing parameter β (fast increase
      and a final hard‑max phase);
    – Adam’s adaptive step‑size (instead of fixed momentum);
    – occasional block‑perturbations of the current best vector to
      escape shallow minima;
    – a simple “no‑improvement” reset that jumps back to the best
      seen solution;
    – all updates keep the vector on the simplex, guaranteeing
      non‑negativity and a well‑defined objective.

The routine runs for at most ``budget_s`` seconds (default ~1100 s)
and finally rescales the best probability vector to the required
ℓ¹‑norm √(2 n) before returning it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan‑type coefficients a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using a real FFT.
    Equivalent to ``np.convolve(a, b)`` but O(n log n).
    Tiny negative round‑off errors are clipped to zero.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)               # eliminate -0‑noise
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation: a random contiguous block is multiplied
    by a log‑normal factor.  The whole vector is optionally symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
