"""Propose a step‐function sequence with a low max‐convolution score.

The algorithm is a refined version of the exponentiated‑gradient / soft‑max
method:

* Initialise the probability vector w (∑w = 1) from several sensible seeds
  (previous best, Catalan numbers, power‑law decays, a few Dirichlet draws)
  and keep the one with the smallest current score.
* Repeatedly:
    – compute the self‑convolution b = w*w,
    – compute a temperature γ that is annealed from a relatively large value
      down to a tiny one (γ = γ_factor·max(b) with γ_factor linearly
      decreasing),
    – build a weight vector p over the convolution entries:
        p = (1‑λ)·softmax(b/γ) + λ·uniform over the active set
        (indices where b is within a tiny tolerance of its maximum);
      λ is increased from 0 to 1 during the run, so the update moves from a
      smooth average to a true sub‑gradient of the max‑operator.
    – obtain the gradient of the smoothed max:
        grad = 2·( w * reverse(p) )   (the first n components after a final
        reversal),
    – perform an exponentiated‑gradient step with a decaying step‑size,
    – project back onto the probability simplex.
    – every few hundred steps enforce symmetry w ← (w+w[::-1])/2,
      and occasionally try a cheap random block perturbation; if it improves
      the best score we accept it.
    – after a long stretch without improvement we restart from a perturbed
      version of the best vector seen so far.

The loop runs until the allotted time budget is exhausted (≈1000 s).  The
final vector is scaled to the conventional √(2 n) ℓ¹‑norm before returning,
so that ``evaluate_sequence`` simply returns the maximal convolution entry.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """Return the first ``length`` Catalan numbers C(2i,i)/4**i as float64."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for moderate sizes)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    conv = np.fft.irfft(Fa * Fb, N)[:n]
    # eliminate tiny negatives from rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation (used as a jump)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:          # occasional extra symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Return soft‑max weights p_k ∝ exp(conv_k / γ), sum(p)=1."""
    shift = conv - np.max(conv)                # avoid overflow
    expv = np.exp(shift / gamma)
    total = np.sum(expv)
    if total == 0.0:
        # fallback – uniform (should never happen)
        return np.full_like(conv, 1.0 / conv.size)
    return expv / total


def compute_score(w: np.ndarray, n: int) -> float:
    """Score = 2·n·max(w*w) for a probability vector w (∑w = 1)."""
    b = convolve_fft(w, w)
    return 2.0 * n * np.max(b)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights whose `evaluate_sequence`
    score is as low as possible.  The routine runs for roughly the given
    time budget (seconds) and employs a temperature‑annealed exponentiated‑
    gradient optimisation with occasional random jumps.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # Determine the length n and create a set of initial candidates.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(prev.size)
        w_prev = np.maximum(prev, 0.0)
        if np.sum(w_prev) == 0.0:
            w_prev = np.full(n, 1.0 / n)
        else:
            w_prev = w_prev / np.sum(w_prev)
        candidates = [w_prev]
    else:
        n = 8000
        candidates = []

    # deterministic seeds based on n
    # 1) Catalan numbers (classic start)
    cat = catalan_sequence(n)
    cat = cat / np.sum(cat)
    candidates.append(cat)

    # 2) Power‑law decays (i+1)^(-α) for a few α
    alphas = [0.45, 0.55, 0.65, 0.75, 0.85]
    for a in alphas:
        idx = np.arange(1, n + 1, dtype=np.float64)
        vec = idx ** (-a)
        vec = vec / np.sum(vec)
        # enforce (optional) symmetry – often helpful
        vec = 0.5 * (vec + vec[::-1])
        candidates.append(vec)

    # 3) A few random Dirichlet draws
    for _ in range(3):
        vec = rng.dirichlet(np.ones(n))
        candidates.append(vec)

    # --------------------------------------------------------------
    # Pick the best among the cheap initial candidates.
    # --------------------------------------------------------------
    best_w = None
    best_score = np.inf
    for cand in candidates:
        sc = compute_score(cand, n)
        if sc < best_score:
            best_score = sc
            best_w = cand.copy()
    # Ensure symmetry of the starting point (helps flatten the convolution)
    best_w = 0.5 * (best_w + best_w[::-1])
    w = best_w.copy()

    print(f"[init] length={n}, initial best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for the 1000 s budget).
    # --------------------------------------------------------------
    eta0 = 1.2                     # base step‑size for exponentiated gradient
    gamma_start = 0.20            # temperature factor at iteration 0
    gamma_end   = 0.01            # temperature factor after the annealing horizon
    gamma_decay_iters = 300000    # #iterations over which γ is linearly reduced
    lambda_ramp   = 50000         # iterations to ramp λ from 0 to 1
    sym_interval   = 400          # enforce symmetry every this many steps
    block_interval = 3000         # try a random block jump every this many steps
    reset_limit    = 12000        # iterations without improvement before a reset
    max_exponent   = 60.0         # clipping for the exponent in EG updates
    tol_active     = 1e-12        # tolerance to decide the active set for max

    iteration = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # (a) current self‑convolution and score
        b = convolve_fft(w, w)
        max_b = np.max(b)
        cur_score = 2.0 * n * max_b

        # (b) record best solution
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # (c) annealed temperature γ
        frac = iteration / gamma_decay_iters
        if frac > 1.0:
            frac = 1.0
        gamma_factor = gamma_start * (1.0 - frac) + gamma_end * frac
        gamma = max(gamma_factor * max_b, 1e-14)

        # (d) soft‑max weights
        p_soft = _softmax_weights(b, gamma)

        # (e) active‑set (hard) weights
        mask = b >= max_b - tol_active
        if np.any(mask):
            lambda_val = iteration / lambda_ramp
            if lambda_val > 1.0:
                lambda_val = 1.0
            p_hard = mask.astype(float) / np.sum(mask)
            p = (1.0 - lambda_val) * p_soft + lambda_val * p_hard
        else:
            p = p_soft

        # (f) gradient of the smoothed max‑convolution
        p_rev = p[::-1]
        conv_grad = convolve_fft(w, p_rev)
        grad = 2.0 * conv_grad[::-1][:n]

        # (g) exponentiated‑gradient update
        eta = eta0 / np.sqrt(iteration + 1.0)
        exponent = -eta * grad
        np.clip(exponent, -max_exponent, max_exponent, out=exponent)
        w = w * np.exp(exponent)

        # (h) project back onto the simplex
        w = simplex_projection(w)

        # (i) periodic symmetrisation
        if iteration % sym_interval == 0 and iteration > 0:
            w = 0.5 * (w + w[::-1])

        # (j) occasional random block jump – keep it if it improves best
        if iteration % block_interval == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = compute_score(cand, n)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best score={best_score:.6f}")

        # (k) long‑run stagnation handling – reset from a perturbed best
        if no_improve > reset_limit:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation routine expects √(2 n) ℓ¹‑norm.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum   # best_w already sums to 1

    # sanity check (silent in the final judge)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"[final] length={n}, best score={best_score:.6f}, "
              f"evaluate={final_score:.6f}")
    except Exception:
        print(f"[final] length={n}, best score={best_score:.6f}")

    return [float(x) for x in final_seq]
