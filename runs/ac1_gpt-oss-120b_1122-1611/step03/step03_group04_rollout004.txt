Algorithm from Duchi:

```
def proj_simplex(v):
    # returns projection onto simplex {x >=0, sum=1}
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_k (a*a)[k]}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

The expression is *scale‑invariant*: we may normalise the vector to a probability
distribution \(\sum_i a_i=1\) during the optimisation and only rescale it at the
very end.  
We therefore minimise  

\[
F(a)=\max_k (a*a)[k]
\]

subject to \(a_i\ge0\) and \(\sum a_i = 1\).

The optimisation uses three ideas:

1. **Entropic (mirror) descent** – a multiplicative update on the probability
   simplex.
2. **A smooth approximation of the max** – a soft‑max (log‑sum‑exp) with a
   temperature that grows with the iteration count.
3. **Escaping local minima** – cheap random block perturbations, occasional
   symmetrisation (the optimum is symmetric) and a restart mechanism.

All heavy work (convolution and the gradient) is performed with FFTs,
so each iteration costs only \(O(n\log n)\).  
The algorithm is deliberately written to be generic – it works for any
length `n`.  A larger `n` gives a better theoretical bound (the average of the
convolution is \(1/(2n-1)\)), therefore we start with a fairly large
sequence (`n≈20000`) while staying comfortably inside the time budget.

The code below follows the outline above, keeps the best vector found,
scales it to the conventional \(\|a\|_1=\sqrt{2n}\) and returns it as a plain
Python list.

```python
"""
Propose a non‑negative step sequence that minimises the score
2·n·max(a*a)/(sum a)².

The algorithm:
  * normalises the vector to a probability distribution (∑a = 1);
  * repeatedly performs an entropic mirror‑descent step on a
    smooth approximation of the max‑convolution;
  * every few thousand iterations applies a cheap random block perturbation;
  * enforces symmetry regularly (the optimal solution is symmetric);
  * restarts from the best‑so‑far after long periods without improvement;
  * finally rescales the best vector to the conventional √(2n) ℓ¹‑norm.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear (non‑circular) convolution via a real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)              # clip tiny negatives
    return c


def correlation_grad(w: np.ndarray, alpha: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max:   g = 2·corr(w, α)

    α is a probability vector (len = 2·n‑1).  The correlation is implemented
    with one additional FFT convolution.
    """
    w_rev = w[::-1]
    conv = convolve_fft(w_rev, alpha)      # length = 3·n‑2
    start = w.shape[0] - 1                 # index that aligns with j = 0
    return 2.0 * conv[start:start + w.shape[0]]


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random jump: multiply a random block by a log‑normal factor,
    optionally symmetrise, and clip to the allowed value range.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # between 0.5 % and ~8 % of the length
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:        # occasional symmetric smoothing
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, sum x = 1}.  Implements the algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """
    Initialise a symmetric, smooth probability vector.
    The shape follows a half‑circle (arcsine) density,
    i.e. w_i ∝ √(1‑x_i²) with x_i∈[‑1,1].
    """
    x = np.linspace(-1.0, 1.0, n)
    w = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    w /= np.sum(w)
    return w


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 960.0, **kwargs):
    """
    Search for a low‑score step function.
    Returns a list of floats (the heights) ready for evaluate_sequence().
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    # Prefer the length of a previously discovered construction,
    # otherwise fall back to a reasonably large default.
    default_n = 20000
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size >= 2:
                n = int(prev.size)
                w = prev / np.sum(prev)                     # already a probability vector
            else:
                raise ValueError
        except Exception:
            n = default_n
            w = arcsine_initial(n)
    else:
        n = default_n
        w = arcsine_initial(n)

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best_w = w.copy()
    b = convolve_fft(w, w)
    best_score = float(2 * n * np.max(b))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned experimentally)
    # ------------------------------------------------------------------
    eta0 = 5.0                # base step size for the multiplicative update
    beta0 = 1.0               # temperature growth factor for the soft‑max
    beta_max = 300.0          # cap on temperature
    symmetrize_every = 500    # enforce symmetry regularly
    block_every = 2500        # cheap random jump frequency
    reset_patience = 30000    # restart after many non‑improving steps

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # --------------------------------------------------------------
        # 1. Current convolution and hard score
        # --------------------------------------------------------------
        b = convolve_fft(w, w)
        max_b = np.max(b)
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:7d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 2. Smooth‑max (soft‑max) weights α
        # --------------------------------------------------------------
        beta = beta0 * (iteration + 1) ** 0.6
        if beta > beta_max:
            beta = beta_max
        diff = b - max_b                     # shift for numerical stability
        exp_term = np.exp(beta * diff)       # largest entry becomes 1.0
        alpha = exp_term / np.sum(exp_term)  # probability vector length 2·n‑1

        # --------------------------------------------------------------
        # 3. Gradient of the smoothed objective
        # --------------------------------------------------------------
        grad = correlation_grad(w, alpha)    # shape (n,)

        # --------------------------------------------------------------
        # 4. Entropic (multiplicative) mirror‑descent update
        # --------------------------------------------------------------
        eta = eta0 / np.sqrt(iteration + 1)
        w = w * np.exp(-eta * grad)          # element‑wise multiplication
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fall back to a uniform distribution if numerical disaster occurs
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 5. Periodic symmetrisation (the optimum is symmetric)
        # --------------------------------------------------------------
        if iteration % symmetrize_every == 0:
            w = 0.5 * (w + w[::-1])
            w /= w.sum()

        # --------------------------------------------------------------
        # 6. Occasional cheap random block perturbation
        # --------------------------------------------------------------
        if iteration % block_every == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = float(2 * n * np.max(convolve_fft(cand, cand)))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:7d}, new best score={best_score:.6f}")

        # --------------------------------------------------------------
        # 7. Restart if stuck for a long time
        # --------------------------------------------------------------
        if no_improve > reset_patience:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:7d}, reset to perturbed best")

        iteration += 1

    # ------------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant,
    # but we output the conventional √(2 n) ℓ¹‑norm.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / best_w.sum()

    # Optional sanity check (does not affect the harness)
    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:
        final_score = float('nan')
    print(f"[final] length={n}, best raw score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    return [float(x) for x in final_seq]
```
