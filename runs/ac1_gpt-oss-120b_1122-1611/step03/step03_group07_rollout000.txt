            cand_score = 2 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:

                best_score = cand_score

"""
Propose a candidate step function whose `evaluate_sequence` value is low.

Algorithm
---------
* Work on the probability simplex (Σ w_i = 1, w_i ≥ 0).  The objective
  after scaling the vector to have L¹‑norm √(2 n) is simply
          score = 2·n·max_k (w*w)[k] .
* Use a *soft‑max* approximation of the max (log‑sum‑exp) to obtain a smooth
  gradient of the objective.
* Apply an Adam‑style *multiplicative* (exponentiated) mirror‑descent update
  w_i ← w_i·exp(−η·g_i) and re‑normalise onto the simplex.
* The temperature γ of the soft‑max is annealed: we start with a relatively
  large γ (smooth landscape) and shrink it slowly to focus on the true peak.
* Every few hundred iterations we enforce (approximate) symmetry
  w ← (w + w[::-1]) / 2, and periodically try a random block perturbation;
  if the perturbation improves the best score we keep it.
* If the search stagnates for many iterations we reset to a perturbed copy
  of the best solution found so far.
* The loop runs until the time budget expires (≈1000 s).  The best vector
  is finally scaled to L¹‑norm √(2 n) and returned as a plain Python list.

The code is deliberately self‑contained, uses only NumPy, and respects the
required interface `propose_candidate`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers divided by 4**i (a_i = C(2i,i)/4**i)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for n up to ~2·10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negatives caused by round‑off
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) random block perturbation (multiplicative factor on a sub‑array)."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈ log‑normal factor ≈ 1±12%
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max probabilities over convolution entries."""
    shift = conv - np.max(conv)               # shift for numerical stability
    exp = np.exp(shift / gamma)
    total = np.sum(exp)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp / total


def _smoothed_gradient(w: np.ndarray, gamma: float):
    """
    Return (gradient, max_conv) where the gradient is the derivative of the
    log‑sum‑exp smoothed max of the self‑convolution (w*w).
    """
    # self‑convolution
    b = convolve_fft(w, w)                     # length 2n‑1
    # soft‑max over its entries
    p = _softmax_weights(b, gamma)             # same length
    # reverse to obtain shift pattern
    p_rev = p[::-1]
    # w * p_rev convolution (length 3n‑2)
    conv = convolve_fft(w, p_rev)
    # align and take first n entries
    grad = 2.0 * conv[::-1][: w.size]
    return grad, np.max(b)

    """

    gamma_factor = 0.5                  # initial temperature multiplier
