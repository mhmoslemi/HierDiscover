""" 
Propose a candidate step function using Adam on a softmax‑smoothed max‑convolution 
objective, with a decreasing smoothing temperature and occasional symmetry 
enforcement and random block perturbations. Multiple lengths are tried within the 
time budget and the best sequence (scaled to √(2 n) L1‑norm) is returned.
"""

import time
import numpy as np

def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # eliminate tiny negatives from round‑off
    np.maximum(c, 0.0, out=c)
    return c

def catalan_sequence(length: int) -> np.ndarray:
    """Catalan numbers divided by 4ⁿ (shape used in earlier work)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i-1] * (2 * (i-1) + 1) / (2 * i)
    return seq

def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max probabilities over entries of ``conv`` with temperature ``gamma``."""
    # shift for numerical stability
    shift = conv - np.max(conv)
    exp_vals = np.exp(shift / gamma)
    total = np.sum(exp_vals)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total

def _smoothed_gradient(w: np.ndarray, gamma: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max smoothed max‑convolution.
    Returns (grad, max_conv) where grad has the same length as ``w``.
    """
    # self‑convolution
    conv = convolve_fft(w, w)                # length 2n‑1
    p = _softmax_weights(conv, gamma)        # shape 2n‑1, sums to 1
    p_rev = p[::-1]
    # convolve w with the reversed weight vector; length = 3n‑2
    grad_full = convolve_fft(w, p_rev)
    # Align and extract first n entries; the exact formula from the
    # original derivation is 2·(w⋆p_rev) reversed, then truncated.
    grad = 2.0 * grad_full[::-1][:w.size]
    max_conv = np.max(conv)
    return grad, max_conv

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small random block perturbation; used as a jump to escape plateaus."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional symmetric averaging
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _run_optimizer(w0: np.ndarray,
                   n: int,
                   rng: np.random.Generator,
                   time_budget: float) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on the soft‑max smoothed max‑convolution.
    Returns the best probability vector found and its (scaled) score.
    """
    # Initialise log‑probabilities u so that softmax(u) = w0
    eps = 1e-12
    u = np.log(np.maximum(w0, eps))

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    eps_adam = 1e-8
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # Tracking best solution
    w = np.exp(u - np.max(u))
    w = w / np.sum(w)
    conv = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(conv)
    best_w = w.copy()
    best_iter = 0

    # Temperature schedule
    gamma_factor = 0.12    # initial factor (relative to current max)
    gamma_min = 1e-9

    # Stagnation handling
    no_improve = 0
    max_no_improve = 25000

    start = time.monotonic()
    it = 0
    sym_interval = 800
    block_interval = 2500

    while time.monotonic() - start < time_budget:
        # ----- compute softmax probabilities -----
        u_shift = u - np.max(u)                 # for numerical stability
        w = np.exp(u_shift)
        w_sum = np.sum(w)
        w = w / w_sum

        # ----- current convolution and loss -----
        grad_w, max_b = _smoothed_gradient(w, max(gamma_factor * max_b if it > 0 else gamma_factor, gamma_min))
        # However we need max_b before calling gradient; recalc inside function.
        # We'll call the function with gamma computed after we know max_b.
        # For clarity, we compute max_b first via convolution
        conv = convolve_fft(w, w)
        max_b = np.max(conv)

        # Track improvement
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            best_iter = it
            no_improve = 0
            # Print occasional progress
            if it % 2000 == 0:
                print(f"[improve] n={n}, iter={it}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ----- temperature for smoothing -----
        gamma = max(gamma_factor * max_b, gamma_min)

        # ----- gradient w.r.t. w (smooth) -----
        grad_w, _ = _smoothed_gradient(w, gamma)

        # ----- convert gradient to u‑space (softmax Jacobian) -----
        avg = np.dot(grad_w, w)          # scalar = ⟨grad_w, w⟩
        grad_u = w * (grad_w - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        u -= step

        # ----- occasional symmetry enforcement -----
        if it % sym_interval == 0 and it > 0:
            # Force w to be roughly symmetric without breaking the softmax parametrisation
            w = np.exp(u - np.max(u))
            w = w / np.sum(w)
            w_sym = 0.5 * (w + w[::-1])
            # bring back to log‑space
            u = np.log(np.maximum(w_sym, eps))

        # ----- random block jump -----
        if it % block_interval == 0 and it > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                u = np.log(np.maximum(cand, eps))
                m = np.zeros_like(u)
                v = np.zeros_like(u)
                print(f"[block] n={n}, iter={it}, new best={best_score:.6f}")

        # ----- restart on stagnation -----
        if no_improve > max_no_improve:
            # Jump from the best with a fresh perturbation
            w_restart = block_perturbation(best_w, rng)
            w_restart = w_restart / np.sum(w_restart)
            u = np.log(np.maximum(w_restart, eps))
            m = np.zeros_like(u)
            v = np.zeros_like(v)
            no_improve = 0
            print(f"[reset] n={n}, iter={it}, restart from perturbed best")

        # ----- temperature decay (slow) -----
        gamma_factor = max(gamma_factor * 0.9999, 0.003)

        it += 1

    # Ensure best_w is a proper probability vector
    best_w = best_w / np.sum(best_w)
    return best_w, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step function with a low max‑convolution value.
    The routine tries a few lengths, runs an Adam‑based optimisation with a
    temperature‑controlled soft‑max smoothing, and returns the best sequence
    found (scaled to the conventional √(2 n) L¹‑norm).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_budget = float(budget_s)

    # Lengths to explore – powers of two give fast FFTs.
    lengths = [4096, 6144, 8192]

    best_overall_score = float('inf')
    best_overall_seq = None

    # Use any previously discovered solution as a warm‑start.
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            have_prev = True
        except Exception:
            have_prev = False
    else:
        have_prev = False

    for n in lengths:
        # estimate remaining budget and allocate a share for this length
        elapsed = time.monotonic() - start_time
        remaining = total_budget - elapsed
        if remaining < 5.0:
            break
        # give each remaining length roughly equal share
        idx = lengths.index(n)
        share = remaining / (len(lengths) - idx)

        # ----- initialise w -----
        if have_prev:
            # Truncate or pad the previous solution to the current size.
            if prev.size >= n:
                w0 = prev[:n]
            else:
                pad = np.zeros(n - prev.size, dtype=np.float64)
                w0 = np.concatenate([prev, pad])
            if np.sum(w0) <= 0.0:
                w0 = np.ones(n, dtype=np.float64)
            w0 = w0 / np.sum(w0)
        else:
            w0 = catalan_sequence(n)
            w0 = w0 / np.sum(w0)

        print(f"[start] length={n}, allocated={share:.1f}s")
        best_w_n, best_score_n = _run_optimizer(w0, n, rng, share)

        # Scale to the conventional √(2 n) L¹‑norm required by the evaluator.
        seq = best_w_n * np.sqrt(2.0 * n)
        try:
            eval_score = float(evaluate_sequence(seq.tolist()))
        except Exception:
            eval_score = best_score_n  # fallback if evaluator unavailable

        print(f"[result] length={n}, raw score={best_score_n:.6f}, "
              f"eval={eval_score:.6f}")

        if eval_score < best_overall_score:
            best_overall_score = eval_score
            best_overall_seq = seq.copy()
            # Store for possible reuse in a future call
            globals()['height_sequence_1'] = seq.tolist()

    # Fallback – if nothing succeeded, return a simple Catalan vector.
    if best_overall_seq is None:
        n = 4096
        w = catalan_sequence(n)
        w = w / np.sum(w)
        best_overall_seq = w * np.sqrt(2.0 * n)

    # Final sanity check (silent in production)
    try:
        final_score = float(evaluate_sequence(best_overall_seq.tolist()))
        print(f"[final] best length={len(best_overall_seq)}, score={final_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_overall_seq]

"""
Optimise a non‑negative step function for a low self‑convolution peak.

The algorithm runs several independent Adam‑optimisation sessions on
different vector lengths.  Each session minimises a temperature‑controlled
soft‑max approximation of the max‑convolution using the log‑softmax
parameterisation (so the vector always lives on the probability simplex).
The temperature γ is tied to the current largest convolution value and is
gradually reduced, which makes the optimisation increasingly focused on the
true max.  Symmetry enforcement and cheap random block jumps help to avoid
plateaus.  The best vector found (scaled to the conventional √(2 n) L¹‑norm)
is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core numerical utilities
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT (fast for n≈10⁴).
    The result length is len(a)+len(b)-1.
    Tiny negative round‑off errors are clipped to zero.
    """
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


def catalan_sequence(length: int) -> np.ndarray:
    """Catalan numbers divided by 4ⁱ, i.e. aᵢ = C(2i,i)/4ⁱ."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Soft‑max probabilities pₖ ∝ exp(convₖ / γ) that sum to 1.
    ``conv`` is a 1‑D array (the self‑convolution).
    """
    shift = conv - np.max(conv)            # improve numerical stability
    exp_vals = np.exp(shift / gamma)
    total = np.sum(exp_vals)
    if total == 0.0:                       # fallback – should be rare
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


def _smoothed_gradient(w: np.ndarray, gamma: float) -> tuple:
    """
    Gradient of the soft‑max smoothed objective
          fγ(w) = γ·log Σₖ exp((w*w)[k]/γ)

    Returns (grad, max_conv) where ``grad`` has the same shape as ``w``.
    """
    # 1. self‑convolution (length 2n‑1)
    conv = convolve_fft(w, w)

    # 2. soft‑max weights over the convolution entries
    p = _softmax_weights(conv, gamma)      # ∈[0,1]⁽²ⁿ⁻¹⁾, Σp=1

    # 3. reverse the weight vector – required for the correct shift pattern
    p_rev = p[::-1]

    # 4. convolution w * p_rev (length 3n‑2)
    grad_full = convolve_fft(w, p_rev)

    # 5. reverse & truncate to obtain the gradient on the original domain
    grad = 2.0 * grad_full[::-1][: w.size]

    max_conv = np.max(conv)
    return grad, max_conv


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random block perturbation.
    A contiguous block of the vector is multiplied by a log‑normal factor.
    Optionally the result is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                # occasional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Optimisation driver for a single length
# ----------------------------------------------------------------------


def _run_optimizer(w0: np.ndarray,
                   n: int,
                   rng: np.random.Generator,
                   time_budget: float) -> tuple:
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best probability vector found (sums to 1) and its raw
    ``2·n·max(conv)`` score.
    """
    eps = 1e-12
    u = np.log(np.maximum(w0, eps))          # log‑softmax parameters

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    eps_adam = 1e-8
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # bookkeeping
    w = np.exp(u - np.max(u))
    w = w / np.sum(w)
    best_w = w.copy()
    conv = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(conv)      # raw objective (before scaling)
    best_iter = 0
    no_improve = 0
    max_no_improve = 25000

    # schedule parameters
    gamma_factor = 0.12                      # initial temperature multiplier
    gamma_min = 1e-9
    sym_interval = 800
    block_interval = 2500

    start = time.monotonic()
    it = 0
    while time.monotonic() - start < time_budget:
        # ----- obtain current probability vector -----
        u_shift = u - np.max(u)               # stabilise exponentials
        w = np.exp(u_shift)
        w = w / np.sum(w)

        # ----- current convolution & score -----
        conv = convolve_fft(w, w)
        max_b = np.max(conv)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_w = w.copy()
            best_iter = it
            no_improve = 0
            if it % 2000 == 0:
                print(f"[improve] n={n}, iter={it}, score={best_score:.6f}")
        else:
            no_improve += 1

        # ----- temperature for smoothing -----
        gamma = max(gamma_factor * max_b, gamma_min)

        # ----- smoothed gradient w.r.t. w -----
        grad_w, _ = _smoothed_gradient(w, gamma)

        # ----- map gradient to u‑space (soft‑max Jacobian) -----
        avg = np.dot(grad_w, w)               # ⟨grad_w, w⟩
        grad_u = w * (grad_w - avg)

        # ----- Adam update on u -----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        u -= step

        # ----- enforce symmetry from time to time -----
        if it % sym_interval == 0 and it > 0:
            w = np.exp(u - np.max(u))
            w = w / np.sum(w)
            w = 0.5 * (w + w[::-1])
            u = np.log(np.maximum(w, eps))

        # ----- occasional random block jump -----
        if it % block_interval == 0 and it > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)
            cand_score = 2.0 * n * np.max(convolve_fft(cand, cand))
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                u = np.log(np.maximum(cand, eps))
                m.fill(0.0)
                v.fill(0.0)
                print(f"[block] n={n}, iter={it}, new best={best_score:.6f}")

        # ----- restart if stuck -----
        if no_improve > max_no_improve:
            w_restart = block_perturbation(best_w, rng)
            w_restart = w_restart / np.sum(w_restart)
            u = np.log(np.maximum(w_restart, eps))
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] n={n}, iter={it}, restarting from perturbed best")

        # ----- slowly tighten the temperature -----
        gamma_factor = max(gamma_factor * 0.9999, 0.003)

        it += 1

    # Ensure the returned vector is a proper probability distribution
    best_w = best_w / np.sum(best_w)
    return best_w, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """

    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    total_budget = float(budget_s)

    # lengths – powers of two for FFT efficiency
    lengths = [4096, 6144, 8192]

    # warm‑start with a previously discovered solution if available
    have_prev = False
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            have_prev = True
        except Exception:
            have_prev = False

    best_overall_score = float("inf")
    best_overall_seq = None

    for idx, n in enumerate(lengths):
        elapsed = time.monotonic() - start_total
        remaining = total_budget - elapsed
        if remaining < 5.0:
            break

        # give each remaining length an equal share of what is left
        share = remaining / (len(lengths) - idx)

        # ----- initialise the probability vector -----
        if have_prev:
            # truncate or pad the previous solution

            else:

                w0 = np.concatenate([prev, pad])
            w0 = np.maximum(w0, 0.0)
            if np.sum(w0) == 0.0:
                w0 = np.ones(n, dtype=np.float64)
            w0 = w0 / np.sum(w0)
        else:
            w0 = catalan_sequence(n)
            w0 = w0 / np.sum(w0)

        print(f"[start] length={n}, time share={share:.1f}s")
        best_w, raw_best_score = _run_optimizer(w0, n, rng, share)

        # Scale to √(2 n) L¹‑norm (the conventional normalisation)

    return [float(x) for x in best_overall_seq]
