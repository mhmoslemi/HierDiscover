    eps_vals = [0.0, 0.5, 1.0, 2.0, 5.0]
    p_vals = np.linspace(0.35, 0.65, 7)  #0.35..0.65
    for p in p_vals:
        for eps in eps_vals:
            idx = np.arange(n)
            w = (idx + eps + 1e-12) ** (-p)
            w = w / np.sum(w)
            score = compute_score(w)
            if score < best_score:
                best_score = score
                best_w = w.copy()
                print(f"[grid] p={p:.3f} eps={eps} score={score:.6f}")

    # also try uniform
    w_uniform = np.full(n, 1.0 / n)
    score_uni = compute_score(w_uniform)
    if score_uni < best_score:
        best_w = w_uniform
        best_score = score_uni

    # Use best_w as start
    w = best_w.copy()
    # Adam variables
    m = np.zeros(n); v = np.zeros(n)
    beta1 = 0.9; beta2 = 0.999; eps_adam = 1e-8
    lr0 = 0.12; # tuned
    alpha = 10.0
    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - start_time < max(0.0, budget_s - 5.0):
        # compute conv
        conv = convolve_fft(w, w)
        # compute softmax weights
        maxc = np.max(conv)
        exp_vals = np.exp(alpha * (conv - maxc))
        p = exp_vals / np.sum(exp_vals)

        # gradient corr
        grad_full = convolve_fft(p, w[::-1])
        grad = 2.0 * grad_full[n-1: n-1 + n]  # slice of length n

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        w_new = w - step
        w = simplex_projection(w_new)

        # occasional evaluation
        if iteration % 300 == 0:
            curr_score = compute_score(w)
            if curr_score < best_score - 1e-9:
                best_score = curr_score
                best_w = w.copy()
                no_improve = 0
                print(f"[improve] iter={iteration}, score={best_score:.6f}")
            else:
                no_improve += 1

        # block perturbation with larger amplitude less frequently
        if iteration % 1200 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)  # use previous block_perturbation
            cand = cand / np.sum(cand)
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration}, new best={best_score:.6f}")

        # increase temperature slower
        if iteration % 3000 == 0 and iteration > 0:
            alpha = min(alpha * 1.05, 2000.0)

        # reset if stuck
        if no_improve > 20000:
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            m.fill(0); v.fill(0)
            no_improve = 0
            print(f"[reset] iter={iteration}, reset to perturbed best")
        iteration += 1

    # final scaling
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")
    return [float(x) for x in final_seq]

"""
propose_candidate
-----------------
A hybrid search for the minimum of ``evaluate_sequence``.
The algorithm combines a cheap two‑parameter family (power–law with a
small shift) with a projected‑Adam optimiser that works on the
probability simplex.  Several random restarts and occasional large
perturbations help to escape local minima.  The final vector is
scaled to the conventional √(2 n) ℓ¹‑norm before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution (non‑circular) – used everywhere
# ----------------------------------------------------------------------
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT.  Small negative round‑off errors
    are clipped to zero (the objective never needs negative values)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()               # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    c = np.fft.irfft(Fa * Fb, N)[:n]
    np.maximum(c, 0.0, out=c)
    return c


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex using the O(n) algorithm of
    Duchi et al. (2008).  Returns a new array, never modifies ``v``."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small random block perturbation – useful as a “restart” move
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Select a random contiguous block, multiply it by a log‑normal factor,
    optionally symmetrise, then clip to the allowed range."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.13)))
    cand[start:stop] *= factor
    # occasional symmetric averaging – helps a lot for the convolution objective
    if rng.random() < 0.4:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Helper: exact objective (scale‑invariant, works with probability vectors)
# ----------------------------------------------------------------------
def compute_score(w: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(w,w)).  ``w`` is assumed to sum to 1."""
    conv = convolve_fft(w, w)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the
    current best bound.  The routine runs for at most ``budget_s`` seconds,
    prints progress information and finally returns the best sequence
    discovered."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.  Re‑use a previously discovered construction if
    # possible, otherwise fall back to a few thousand entries.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n0 = int(np.asarray(height_sequence_1).size)
            if n0 >= 2:
                n = n0
            else:
                raise ValueError
        except Exception:
            n = 6000
    else:
        n = 6000

    print(f"[setup] length={n}")

    # --------------------------------------------------------------
    # Initialise a few promising seeds (Catalan, power‑law variants,
    # random Dirichlet).  The best of them will become the start point.
    # --------------------------------------------------------------
    candidates = []

    # 1) Uniform – useful baseline
    candidates.append(np.full(n, 1.0 / n))

    # 2) Catalan sequence – the classic starting point
    cat = np.empty(n, dtype=np.float64)
    cat[0] = 1.0
    for i in range(1, n):
        cat[i] = cat[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    cat = cat / np.sum(cat)
    candidates.append(cat)

    # 3) Power‑law family a_i = (i+shift)^{-p}
    p_vals = np.linspace(0.40, 0.60, 9)          # centred on p=0.5
    shift_vals = [0.0, 0.5, 1.0, 2.0, 5.0]
    idx = np.arange(n, dtype=np.float64)
    for p in p_vals:
        for sh in shift_vals:
            w = (idx + sh + 1e-12) ** (-p)
            w = w / np.sum(w)
            candidates.append(w)

    # 4) A few random Dirichlet draws (α=0.5 gives a sparse start)
    for _ in range(6):
        g = rng.gamma(0.5, 1.0, size=n)
        candidates.append(g / np.sum(g))

    # --------------------------------------------------------------
    # Evaluate the coarse candidates and keep the best one.
    # --------------------------------------------------------------
    best_w = None
    best_score = np.inf
    for i, w0 in enumerate(candidates):
        score = compute_score(w0, n)
        if score < best_score:
            best_score = score
            best_w = w0.copy()
        print(f"[init {i:02d}] score={score:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser state (shared across restarts)
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.14                # a touch smaller – helps stability for large n
    alpha = 10.0              # initial soft‑max temperature

    # --------------------------------------------------------------
    # Main optimisation loop – runs until the time budget is (almost)
    # exhausted.  The loop occasionally performs a large random restart
    # if progress stalls.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    iteration = 0
    no_improve = 0

    # keep a copy of the very best vector seen so far
    best_vector = best_w.copy()
    best_score = compute_score(best_vector, n)

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # ---- 1. convolution (linear) ---------------------------------
        conv = convolve_fft(best_vector, best_vector)

        # ---- 2. soft‑max over the convolution -------------------------
        maxc = np.max(conv)
        exp_vals = np.exp(alpha * (conv - maxc))
        soft = exp_vals / np.sum(exp_vals)          # sum_i soft_i = 1

        # ---- 3. gradient of the smoothed objective --------------------
        # ∂/∂w Σ_i soft_i (w*w)[i] = 2 * correlation(soft, w)
        grad_full = convolve_fft(soft, best_vector[::-1])
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # ---- 4. Adam update (projected onto simplex) ------------------
        # we maintain a separate Adam state for the current vector
        if iteration == 0:
            m = np.zeros_like(best_vector)
            v = np.zeros_like(best_vector)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        w_new = best_vector - step
        best_vector = simplex_projection(w_new)

        # ---- 5. occasional evaluation of the true objective ----------
        if iteration % 350 == 0:
            cur_score = compute_score(best_vector, n)
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_vector = best_vector.copy()
                no_improve = 0
                print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")
            else:
                no_improve += 1

        # ---- 6. random block perturbation (large jump) ----------------
        if iteration % 1300 == 0 and iteration > 0:
            cand = block_perturbation(best_vector, rng)
            cand = cand / np.sum(cand)
            cand_score = compute_score(cand, n)
            if cand_score < best_score:
                best_score = cand_score
                best_vector = cand.copy()
                print(f"[block] iter={iteration:6d}, new best={best_score:.6f}")

        # ---- 7. increase temperature slowly ---------------------------
        if iteration % 2600 == 0 and iteration > 0:
            alpha = min(alpha * 1.07, 2500.0)

        # ---- 8. restart if stagnant -----------------------------------
        if no_improve > 25000:
            # jump to a fresh random Dirichlet point, but keep the best so far
            cand = rng.gamma(0.6, 1.0, size=n)
            cand = cand / np.sum(cand)
            best_vector = cand.copy()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[reset] iter={iteration:6d}, reset to random Dirichlet")

        iteration += 1

    # --------------------------------------------------------------
    # Scale the probability vector to the conventional √(2 n) L¹‑norm.
    # This does not change the objective value (scale‑invariant).
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_vector * target_sum / np.sum(best_vector)

    # Final official score (the harness will compute the same value)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best_score={best_score:.6f}, evaluate={final_score:.6f}")

    # Return plain Python list as required by the harness.
    return [float(x) for x in final_seq]
