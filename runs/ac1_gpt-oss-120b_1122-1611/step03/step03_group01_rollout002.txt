def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` Catalan numbers a_i = C(2i,i)/4**i."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def beta_shape_sequence(length: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape on a discrete grid.
    a_i ∝ (x_i·(1‑x_i))**(α‑1) where x_i = (i+0.5)/length.
    The distribution is symmetric, diverges at the edges for α<1.
    """
    eps = 1e-12
    i = np.arange(length, dtype=np.float64)
    x = (i + 0.5) / length
    # avoid exact 0/1 to keep powers finite
    x = np.clip(x, eps, 1.0 - eps)
    shape = np.power(x * (1.0 - x), alpha - 1.0)
    # guard against infinities for very small alpha
    shape[np.isinf(shape)] = 0.0
    shape = np.clip(shape, 0.0, 1e12)
    shape /= shape.sum()
    return shape


def sin_power_sequence(length: int, power: float = 1.0) -> np.ndarray:
    """Sinusoidal shape: a_i ∝ sin(π·(i+0.5)/length) ** power."""
    i = np.arange(length, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / length) ** power
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # small negative values from floating point noise are removed
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x ≥ 0, Σ x = 1 } (Duchi et al., 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Large cheap jump: multiply a random contiguous block by a log‑normal factor.
    Occasionally enforces symmetry.
    """
    cand = base.copy()
    n = cand.size
    # block length between 0.5 % and 15 % of the vector
    lo = max(2, n // 200)
    hi = max(3, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= cand.sum()
    return cand


def compute_score(w: np.ndarray, n: int) -> float:
    """Return the objective 2·n·max(w★w) (scale‑invariant)."""
    conv = convolve_fft(w, w)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main candidate proposal – entry point for the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a non‑negative step‑function (list of floats) with a low
    `evaluate_sequence` score.  The routine respects the supplied time
    budget (≈1000 s), combines analytical initialisation, smoothed
    gradient descent on a soft‑max approximation of the max‑convolution,
    and a final simulated‑annealing polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # --------------------------------------------------------------
    # Determine a suitable length n.
    # --------------------------------------------------------------
    try:
        # If the harness supplies a previous construction, reuse its length.
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(seq0.size)
        if n < 2000:
            raise ValueError
    except Exception:
        n = 6000                     # comfortable default size

    print(f"[info] chosen length n = {n}")

    # --------------------------------------------------------------
    # Build a diverse set of initial candidates.
    # --------------------------------------------------------------
    candidates = []

    # 1. Catalan baseline
    cat = catalan_sequence(n)
    candidates.append(cat / cat.sum())

    # 2. Simple arcsine (Beta with α=0.5)
    arcs = beta_shape_sequence(n, alpha=0.5)
    candidates.append(arcs)

    # 3. Sweep over a few Beta‑shape exponents (α∈[0.1,0.9])
    for alpha in np.linspace(0.15, 0.85, 7):
        shape = beta_shape_sequence(n, alpha=alpha)
        candidates.append(shape)

    # 4. Sin‑power shape (just for extra diversity)
    sinp = sin_power_sequence(n, power=0.7)
    candidates.append(sinp)

    # 5. Random Dirichlet start (α=0.5 gives a spiky distribution)
    rand = rng.gamma(0.5, 1.0, size=n)
    candidates.append(rand / rand.sum())

    # 6. If a previous solution exists, reuse it (after simplex normalisation)
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size == n:
                prev = prev.copy()
                prev[prev < 0] = 0.0
                prev /= prev.sum()
                candidates.append(prev)
        except Exception:
            pass

    # --------------------------------------------------------------
    # Pick the best starting point according to the true objective.
    # --------------------------------------------------------------
    best_w = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        sc = compute_score(cand, n)
        print(f"[init] cand {idx:02d} – score = {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_w = cand.copy()
    print(f"[init] selected initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Smoothed gradient descent (soft‑max of the convolution).
    # --------------------------------------------------------------
    w = best_w.copy()
    w_rev = w[::-1]
    eta0 = 0.15            # base learning rate
    alpha_start = 2.0
    alpha_max = 30.0
    alpha_inc = (alpha_max - alpha_start) / 8000.0   # linear increase
    iteration = 0
    max_iter = 200_000
    sym_interval = 500            # enforce symmetry every few steps
    restart_interval = 20_000     # re‑centre momentum if stuck

    # Simple momentum implementation (Nesterov‑style)
    w_prev = w.copy()

    while True:
        now = time.monotonic()
        if now - start_time > budget_s - 8.0:       # leave a safety margin for polishing
            break
        if iteration >= max_iter:
            break

        # current convolution & smoothed weights
        conv = convolve_fft(w, w)                    # length 2n‑1
        max_conv = conv.max()

        # soft‑max temperature (α) – grows slowly
        alpha = min(alpha_max, alpha_start + iteration * alpha_inc)
        # shift before exponentiation to keep numbers ≤1
        shifted = conv - max_conv
        exp_term = np.exp(alpha * shifted)          # all ≤1
        s = exp_term / exp_term.sum()               # soft‑max distribution (length 2n‑1)

        # gradient of the smoothed objective:
        #   grad = 2 * (s ★ w_rev) restricted to n central entries
        temp = convolve_fft(s, w_rev)               # length 3n‑2
        grad = 2.0 * temp[n - 1:n - 1 + n]

        # step size (decreasing with sqrt(iteration))
        eta = eta0 / np.sqrt(iteration + 1)

        # Nesterov‑style momentum update
        w_temp = w - eta * grad + 0.9 * (w - w_prev)
        w_temp = np.maximum(w_temp, 0.0)            # stay non‑negative before projection
        w_new = simplex_projection(w_temp)

        # occasional symmetry enforcement
        if iteration % sym_interval == 0 and iteration > 0:
            w_new = 0.5 * (w_new + w_new[::-1])
            w_new /= w_new.sum()

        w_prev = w
        w = w_new

        # evaluate current score
        cur_score = compute_score(w, n)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_w = w.copy()
            print(f"[sgd improve] iter={iteration:06d}, α={alpha:.2f}, score={best_score:.6f}")

        # occasional large random jump (helps escape plateaus)
        if iteration % 3500 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand_score = compute_score(cand, n)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                w_prev = w.copy()
                print(f"[sgd block] iter={iteration:06d}, new best={best_score:.6f}")

        # restart momentum if progress stalls for a long time
        if iteration % restart_interval == 0 and iteration > 0:
            w = best_w.copy()
            w_prev = w.copy()
            print(f"[sgd restart] iter={iteration:06d}")

        iteration += 1

    print(f"[sgd] finished after {iteration} iterations – best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Simulated‑annealing polishing (local random moves).
    # --------------------------------------------------------------
    remaining = budget_s - (time.monotonic() - start_time)
    if remaining > 12.0:
        temperature = 0.025
        anneal_start = time.monotonic()
        cur_w = best_w.copy()
        cur_score = best_score
        cur_iter = 0
        # we keep a simple exponential cooling schedule
        while time.monotonic() - anneal_start < remaining - 4.0:
            # propose a move: either tiny mass transfer or a block scaling
            if rng.random() < 0.6:
                # tiny transfer
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                delta = rng.uniform(0.0, min(cur_w[i], 0.012))
                if delta <= 0.0:
                    continue
                cand = cur_w.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                # block perturbation (larger jump)
                cand = block_perturbation(cur_w, rng)

            # enforce symmetry (helps keep the structure)
            if rng.random() < 0.5:
                cand = 0.5 * (cand + cand[::-1])

            cand /= cand.sum()
            cand_score = compute_score(cand, n)

            # Metropolis acceptance
            if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / temperature):
                cur_w = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_w = cur_w.copy()
                    best_score = cur_score
                    print(f"[anneal] iter={cur_iter:06d}, new best={best_score:.6f}")

            # cool down
            temperature *= 0.99995
            if temperature < 1e-6:
                temperature = 0.001
            cur_iter += 1

    print(f"[final polishing] best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Scale to √(2·n) ℓ¹‑norm as required by the evaluation harness.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * (target_sum / best_w.sum())
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length = {n}, score after scaling = {final_score:.6f}")

    # Return a plain Python list.
    return [float(x) for x in final_seq]
```

This implementation blends analytic initial shapes, a soft‑max gradient descent on the max‑convolution, and a temperature‑controlled simulated‑annealing finish while always staying on the probability simplex and exploiting symmetry.  
The routine respects the 1000 s budget, prints progress, and returns the final step‑function as a list of floats.
