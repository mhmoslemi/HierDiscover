def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, cvxpy as cp

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Initialize f from existing sequence or uniform
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # Ensure valid shape
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            base = np.ones(256, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    # Clip to allowed range
    base = np.clip(base, 0.0, 1000.0)

    # Desired L1 norm (sum) – chosen sqrt(2n)
    n = base.size
    target_sum = np.sqrt(2.0 * n)

    # Scale to target sum
    if np.sum(base) > 0:
        f = base * (target_sum / np.sum(base))
    else:
        f = np.ones_like(base) * (target_sum / base.size)

    f = np.clip(f, 0.0, 1000.0)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # mixing factor
    t = 0.3   # can be tuned

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # compute current bound
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))

        # solve LP for g
        g = cp.Variable(n, nonneg=True)
        constraints = [ cp.conv(f, g) <= M ]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=5000)
        except Exception as e:
            # fall back to numeric solve failure: break
            print(f"LP solve failed at iter {iteration}: {e}")
            break

        if g.value is None:
            # infeasible
            print(f"LP infeasible at iteration {iteration}")
            break

        g0 = np.asarray(g.value, dtype=np.float64)
        # Clip to allowed range before scaling (still nonneg)
        g0 = np.clip(g0, 0.0, 1000.0)

        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            # cannot scale, fallback to small perturbation
            perturb = (rng.normal(size=n) * 0.05).astype(np.float64)
            f = np.clip(f + perturb, 0.0, 1000.0)
            continue

        # rescale g to have same L1 as target sum
        scale = target_sum / sum_g0
        g = g0 * scale
        g = np.clip(g, 0.0, 1000.0)

        # convex combination update
        f = (1.0 - t) * f + t * g

        # keep within bounds
        f = np.clip(f, 0.0, 1000.0)

        # normalise again to keep exact target sum (helps avoid drift)
        cur_sum = np.sum(f)
        if cur_sum > 0:
            f = f * (target_sum / cur_sum)
        else:
            f = np.ones_like(f) * (target_sum / n)

        # evaluate
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f.copy()
            best_score = score
            print(f"Iter {iteration}: improved score {best_score:.6f}")

        # occasional random restart if stuck
        if iteration % 50 == 0:
            # small random perturbation to explore
            mask = rng.integers(0, n, size=max(1, n // 100))
            f[mask] *= np.exp(rng.normal(0, 0.2, size=mask.shape))
            f = np.clip(f, 0.0, 1000.0)

    # final best
    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    return [float(x) for x in best]

"""
Propose a candidate sequence using an LP‑based “flatten‑convolution” update.

The idea follows the sketch from the statement:
* Keep the ℓ₁‑norm fixed (chosen as √(2·n)).
* Compute M = max_k (f * f)[k].
* Solve a linear program that finds a non‑negative vector g with the largest possible
  ℓ₁‑norm under the linear constraints (f * g) ≤ M (element‑wise).  Because the
  convolution with the *fixed* f is linear in g, this is an LP.
* Rescale g back to the target ℓ₁‑norm and replace f by a convex combination
  f ← (1‑t)·f + t·g.  The new f has the same total mass but a flatter self‑convolution,
  so its score 2·n·max(f*f)/‖f‖₁² tends to decrease.
* The loop runs until the time budget (≈1000 s) expires, keeping the best sequence
  seen.  Small random block perturbations are interleaved to avoid stagnation.

The algorithm improves on the naive block‑perturbation search and often pushes the
score below the uniform‑sequence bound of 2.0.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper: cheap block perturbation (used for occasional exploration)
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # pick a block whose length is a modest fraction of n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise on the block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation (helps flatten the convolution)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The method runs a projected‑LP local search for as long
    as the allotted time budget allows and keeps the best candidate found.
    """
    rng = np.random.default_rng(seed)
    # give a tiny safety margin for the surrounding harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise f from the global construction (if present) or fall back
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            base = np.ones(256, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)

    # enforce the evaluator's hard limits
    base = np.clip(base, 0.0, 1000.0)
    n = base.size
    target_sum = np.sqrt(2.0 * n)                     # the classic normalisation

    # break the perfect symmetry of the uniform vector with a tiny random kick
    base *= np.exp(rng.normal(0.0, 0.04, size=n))
    base = np.clip(base, 0.0, 1000.0)

    if np.sum(base) > 0:
        f = base * (target_sum / np.sum(base))
    else:
        f = np.full_like(base, target_sum / n)

    f = np.clip(f, 0.0, 1000.0)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # mixing factor (size of each update step)
    t = 0.3

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # current bound from the self‑convolution of f
        # --------------------------------------------------------------
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))

        # --------------------------------------------------------------
        # Linear program: maximise sum(g)  s.t.  (f * g) ≤ M  (element‑wise)
        # --------------------------------------------------------------
        g_var = cp.Variable(n, nonneg=True)
        # cp.conv treats the numpy array as a constant expression
        conv_expr = cp.conv(f, g_var)        # length 2n‑1
        constraints = [conv_expr <= M]

        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

        try:

            f = _make_block_perturbation(f, rng)

        # --------------------------------------------------------------
