def reduce_max_step(a):
    c = np.convolve(a, a)
    k_max = int(np.argmax(c))
    n = len(a)
    # compute derivative weights
    w = np.zeros(n)
    # shift forward
    if k_max < n:
        w[:n - k_max] += a[k_max:]   # a_{i+k} for i from 0 to n-1-k
        w[k_max:] += a[:n - k_max]   # a_{i-k} for i from k_max to n-1
    # pick indices
    # We may want to reduce a_i where w_i large and a_i positive
    idx_high = np.argmax(w * a)  # high contribution
    idx_low = np.argmin(w * (1000 - a))  # where increase has small effect
    eps = min(a[idx_high] * 0.2, (1000 - a[idx_low]) * 0.2, 0.1)
    if eps <= 0:
        return a
    a_new = a.copy()
    a_new[idx_high] -= eps
    a_new[idx_low] += eps
    return np.clip(a_new, 0.0, 1000.0)

    # compute convolution
    L = 1 << ((2*n - 1).bit_length())

    # reflect to <= n-1
    if k_max > n-1:
        k = 2*n - 2 - k_max
    else:
        k = k_max
    # compute derivative weights
    w = np.zeros(n, dtype=np.float64)
    # forward shift
    if k < n:
        w[:n - k] += a[k:]    # a_{i+k}
        w[k:] += a[:n - k]    # a_{i-k}
    # choose indices
    # weight * a large indicates high contribution to c_k
    # index for reduction: high weighted a_i
    weighted = w * a
    if np.all(weighted == 0):

    idx_high = int(np.argmax(weighted))

    # compute potential increase weight: low w, and low a currently

"""
Propose a candidate step‑function sequence.

Algorithm
---------
* Start from the provided `height_sequence_1` (or a uniform fallback).
* Use a simulated‑annealing loop that runs until the given time budget expires.
* In each iteration one of several mutation operators is applied:
    - multiplicative block perturbation (as in the reference code)
    - additive block perturbation
    - swapping two blocks
    - optional mirroring (average with its reverse)
    - a *targeted* move that directly reduces the current maximal
      convolution value by moving a small amount of mass from a high‑impact
      index to a low‑impact index (derived from the sub‑gradient of the max).
* The new sequence is accepted if it improves the objective or,
  with a temperature dependent probability, if it is worse (simulated annealing).
* The best sequence seen so far is stored and reported at the end.
* Scoring is performed locally with a fast FFT based convolution that
  reproduces `evaluate_sequence` (clipping to [0,1000] and guarding
  against near‑zero sums).

The method is deliberately simple, fast and completely self‑contained;
it does not rely on external solvers and respects the 2‑second per
iteration budget by using O(n log n) convolution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – fast replica of the public evaluate_sequence (without I/O)
def _score(seq: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(seq,seq)) / (sum(seq)**2)
    after clipping to the [0,1000] interval required by `evaluate_sequence`.
    """
    a = np.clip(seq, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.size
    # next power of two for an efficient FFT
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    m = conv.max()
    return float(2.0 * n * m / (s * s))


# ----------------------------------------------------------------------
# Helper – deterministic step that tries to lower the current maximum
def _reduce_max_step(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify the offset where the auto‑convolution is maximal,
    compute its sub‑gradient with respect to each coefficient and move a
    small amount of mass from a high‑impact index to a low‑impact index.
    The move respects the [0,1000] bounds.
    """
    n = a.size
    # compute convolution once
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

    k_max = int(np.argmax(conv))
    # because the convolution is symmetric we can fold the offset to [0,n-1]
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    # sub‑gradient of the max‑offset w.r.t. a_i:
    #   w_i = a_{i+k} (if i+k < n) + a_{i-k} (if i-k >= 0)
    w = np.zeros(n, dtype=np.float64)
    if k < n:
        w[: n - k] += a[k:]          # a_{i+k}
        w[k:] += a[: n - k]          # a_{i-k}

    # pick a donor (large w_i * a_i) and a receiver (small w_i, room to grow)
    impact = w * a
    if not np.any(impact):
        return a
    donor = int(np.argmax(impact))

    # choose a receiver that is far from the max‑impact region
    # we minimise w_i while also favouring positions that are far from the upper bound.
    receiver = int(np.argmin(w + a))

    # shift a small, adaptive amount of mass
    eps = min(
        a[donor] * 0.3,
        (1000.0 - a[receiver]) * 0.3,
        rng.uniform(0.05, 0.5),
    )
    if eps <= 0.0:
        return a

    b = a.copy()
    b[donor] -= eps
    b[receiver] += eps
    return b


# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: float = 1100.0,
    **kwargs,
) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    (minus a small safety margin) and returns the best sequence found.

    It uses simulated annealing with several mutation operators:
        * block scaling (log‑normal multiplicative noise)
        * block additive noise
        * swapping two blocks
        * optional mirroring (average with reverse)
        * a targeted gradient‑based reduction of the maximal convolution.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # initialise the parent sequence
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:  # height_sequence_1 not defined or invalid
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    n = base.size

    # best / current solutions
    best = base.copy()
    best_score = _score(best)
    current = best.copy()
    current_score = best_score

    # timers
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)  # tiny safety margin

    # pre‑compute constants for the schedule
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # progress‑based temperature (simple linear cooling)
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / total_time)
        temperature = max(1e-4, (1.0 - progress) * 0.15)  # start ~0.15, cool to ~0

        # pick a mutation
        move_type = rng.integers(0, 5)
        cand = current.copy()

        if move_type == 0:  # multiplicative block perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start : start + width] *= factor

        elif move_type == 1:  # additive block perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            delta = float(rng.normal(0.0, 0.05))
            cand[start : start + width] += delta

        elif move_type == 2:  # swap two blocks of equal size
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            i1 = int(rng.integers(0, max(1, n - width + 1)))
            i2 = int(rng.integers(0, max(1, n - width + 1)))
            # perform the swap
            tmp = cand[i1 : i1 + width].copy()
            cand[i1 : i1 + width] = cand[i2 : i2 + width]
            cand[i2 : i2 + width] = tmp

        elif move_type == 3:  # mirror the sequence (averaging with its reverse)
            cand = 0.5 * (cand + cand[::-1])

        else:  # move_type == 4 – deterministic reduction of the current max
            cand = _reduce_max_step(cand, rng)

        # enforce the global bounds
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate candidate (fast local replica of evaluate_sequence)
        cand_score = _score(cand)
        if not np.isfinite(cand_score):
            continue

        # acceptance rule – classic Metropolis
        delta = cand_score - current_score
        if delta < 0.0 or rng.random() < np.exp(-delta / temperature):
            current = cand
            current_score = cand_score
            # keep the global best
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"[{elapsed:6.2f}s] New best score: {best_score:.6f}"
                )
        # occasional reset to the best solution to avoid drifting forever
        if rng.random() < 0.001:
            current = best.copy()
            current_score = best_score

    # ------------------------------------------------------------------
    # Return the best sequence found as a plain list of floats
    return [float(x) for x in best]
