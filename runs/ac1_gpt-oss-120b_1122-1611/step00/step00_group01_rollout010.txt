def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Simple optimizer that analytically finds the optimal sequence: uniform heights.
    The evaluation function's lower bound for non‑negative entries is 2, attained
    when all entries are equal.  The function therefore builds a uniform
    sequence of the same length as ``height_sequence_1`` (or 256 if unavailable)
    and returns it.  A tiny random‑search loop is kept only to honour the
    allocated time budget and to demonstrate that no improvement below the
    theoretical optimum can be found.
    """
    import numpy as np, time

    # Determine desired length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 256

    # Uniform constant (any positive value works, use 1.0)
    uniform_val = 1.0
    candidate = np.full(n, uniform_val, dtype=np.float64)

    # Clip to allowed range (redundant for 1.0)
    candidate = np.clip(candidate, 0.0, 1000.0)

    best = candidate.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, uniform value={uniform_val}, score={best_score}")

    # Short stochastic refinement – will never beat the optimum, but respects the time‐budget.
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    while time.monotonic() < deadline:
        # Randomly pick a block and apply a tiny multiplicative perturbation.
        width = int(rng.integers(1, max(2, n // 20)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        pert = best.copy()
        factor = np.exp(rng.normal(0.0, 0.02))
        pert[start:stop] *= factor
        pert = np.clip(pert, 0.0, 1000.0)

        score = float(evaluate_sequence(pert.tolist()))
        if np.isfinite(score) and score < best_score - 1e-12:
            best = pert
            best_score = score
            print(f"[improve] new best score {best_score:.12f}")

    print(f"[final] best score = {best_score}")
    return [float(x) for x in best]

def project_simplex(v):
    # ensure nonnegative sum to 1
    n = v.shape[0]
    u = np.sort(v)[::-1]

    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Proposed candidate generator – projected gradient descent on a smooth
approximation of the worst‐case convolution entry.

The evaluation function is
    2 * n * max_k (∑_i a_i a_{k-i}) / (∑_i a_i)^2 .
For a fixed ℓ₁ norm the denominator is constant, so we only need to
minimise the largest convolution coefficient.  This routine works with a
fixed length (default 1024) and repeatedly:

1. computes the convolution,
2. identifies the index of the current worst entry,
3. builds a sub‑gradient of that entry (∂/∂a_i max ≈ 2·a_{k‑i}),
4. takes a small step opposite the gradient,
5. projects back onto the simplex {a_i ≥ 0, ∑ a_i = 1}.

Occasionally a random multiplicative perturbation is added to
avoid getting stuck in plateaus.  The best sequence found during the
run is returned.  All operations respect the security checks in
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper: projection onto the probability simplex (non‑negative, unit sum)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a step‑function height sequence with a low evaluation score.

    The algorithm performs a projected sub‑gradient descent on a smooth
    surrogate of the maximum convolution entry, maintaining a unit ℓ₁
    norm (the scale does not affect the final score).  Random perturb
    ates are mixed in to provide exploration.
    """
    rng = np.random.default_rng(seed)

    # total time allowed for the optimisation loop
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # choose a modest length – large enough to give a good theoretical
    # bound, small enough for many fast iterations.
    n = 1024
    # start from the uniform distribution (guarantees a feasible point)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # best solution seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # parameters for the gradient descent schedule
    initial_eta = 0.5          # initial step size
    alpha = 5.0                # smooth‑max temperature (larger → sharper)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # convolution and identification of the current worst entry
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))

        # --------------------------------------------------------------
        # sub‑gradient of the (non‑smooth) max – only the peak contributes
        grad = np.zeros_like(a)
        # i runs over indices that appear in conv[max_idx] = Σ a_i a_{max_idx-i}
        i_min = max(0, max_idx - (n - 1))
        i_max = min(n - 1, max_idx)
        # fill gradient: ∂/∂a_i conv[max_idx] = 2 * a_{max_idx-i}
        for i in range(i_min, i_max + 1):
            j = max_idx - i
            grad[i] = 2.0 * a[j]

        # --------------------------------------------------------------
        # step size – simple diminishing schedule
        eta = initial_eta / np.sqrt(iteration)

        # gradient descent step and projection onto the simplex
        a_candidate = a - eta * grad
        a_candidate = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        # occasional random shake‑up to increase exploration
        if rng.random() < 0.07:
            # pick a random position and apply a small log‑normal factor
            pos = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.15)))
            a_candidate[pos] *= factor
            # re‑project to keep a valid distribution
            a_candidate = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        # evaluate the candidate – the evaluate_sequence routine enforces
        # clipping and positivity, but our construction already satisfies
        # those requirements.
        score = float(evaluate_sequence(a_candidate.tolist()))

        # keep the best ever found
        if np.isfinite(score) and score < best_score:
            best = a_candidate.copy()
            best_score = score
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # for the next iteration we stay with the current candidate
        a = a_candidate

        # a quick safeguard: if the score is already close to the theoretical
        # lower bound (≈ 2 * n / (2*n-1)), we can break early.
        theoretical_lb = 2.0 * n / (2 * n - 1)
        if best_score <= theoretical_lb * 1.01:
            print("[early stop] near‑optimal bound reached")
            break

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # Return as a plain Python list of floats as required by the framework.
    return [float(x) for x in best]
