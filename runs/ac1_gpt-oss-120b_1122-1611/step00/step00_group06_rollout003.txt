"""
Propose a step‑function heights sequence that minimizes the provided
evaluation function.

The construction is based on the *discrete arcsine distribution*
    a_i ∝ 1 / √(i·(n‑1‑i)) .
In the continuous setting the arcsine density convolved with itself is
perfectly flat, which makes the maximum entry of the self‑convolution
as close as possible to its average.  Consequently the evaluation
score 2·n·max(b)/(sum(a)²) approaches the theoretical lower bound
≈2·n/(2·n‑1)≈1.  The sequence is scaled so that its largest entry is
exactly 1000 (the clipping limit of ``evaluate_sequence``).  After the
initial analytically‑derived candidate a short time‑bounded local search
performs random multiplicative block perturbations to try to improve
the score further.

The function respects the competition constraints:
* only top‑level functions, no closures or lambdas,
* uses only standard scientific libraries,
* runs for at most ``budget_s`` seconds (default ≈1100 s),
* returns a list of non‑negative floats.
"""

import time
import numpy as np


def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Return a length‑n array whose entries follow the discrete arcsine shape
    a_i = C / sqrt((i+0.5)*(n-0.5-i)), clipped to the range [0,1000].
    The constant C is chosen so that the maximum entry equals 1000.
    """
    i = np.arange(n, dtype=np.float64)
    # half‑step avoids division by zero at the ends
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    # Scale so that the largest value becomes exactly 1000
    scale = 1000.0 / np.max(w)
    a = w * scale
    # Ensure hard clipping (though it should already be ≤1000)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Generate a candidate sequence and improve it within the allotted time.

    Steps
    -----
    1. Determine the required length ``n`` from the global ``height_sequence_1`
       (fallback to 256 if unavailable).
    2. Build the initial arcsine‑shaped candidate.
    3. Evaluate it once with ``evaluate_sequence``.
    4. While time permits, apply random block multiplicative perturbations
       (log‑normal factor) and occasionally symmetrise the perturbed block.
       Accept a perturbation only if it strictly improves the score.
    5. Return the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # Reserve a few seconds for finalisation / printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1. Length of the target sequence
    # ------------------------------------------------------------------
    try:
        n = int(len(globals().get("height_sequence_1", [])))
        if n == 0:
            raise ValueError
    except Exception:
        n = 256  # safe fallback

    # ------------------------------------------------------------------
    # 2. Initial analytic candidate (discrete arcsine)
    # ------------------------------------------------------------------
    best = _arcsine_sequence(n)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] arcsine candidate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Simple local search – random block perturbations
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Choose a block width roughly between 0.5% and 8% of n
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        candidate = best.copy()

        # Multiplicative jitter drawn from a log‑normal distribution
        factor = float(np.exp(rng.normal(0.0, 0.05)))     # σ≈0.05 → modest change
        candidate[start:stop] *= factor

        # Occasionally enforce a mirrored update to keep the shape roughly symmetric
        if rng.random() < 0.2:
            rev_start = n - stop
            rev_stop = n - start
            # Ensure the mirrored slice has the same length
            block = candidate[start:stop]
            rev_block = candidate[rev_start:rev_stop]
            avg = 0.5 * (block + rev_block[::-1])
            candidate[start:stop] = avg
            candidate[rev_start:rev_stop] = avg[::-1]

        # Hard clipping to satisfy the evaluator's limits
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        # ------------------------------------------------------------------
        # 4. Evaluate and accept if better
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[improve] new best score = {best_score:.6f}")

    print(f"[final] best score after search = {best_score:.6f}")
    return [float(x) for x in best]
