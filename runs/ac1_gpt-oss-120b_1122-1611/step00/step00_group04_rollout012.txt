     max_idx = int(conv.argmax())
     # Find pairs i where 0 <= i < n, 0 <= j = max_idx - i < n
     # compute products
     # Since concurrency pattern is large, but we can compute vectorized:
     i_vals = np.arange(max(0, max_idx - (n-1)), min(n-1, max_idx) + 1)  # inclusive

"""
Greedy Sidon‑like construction for a low‑evaluation step function.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For a binary vector `a` (0/1) the convolution entry `c[k]` is exactly the
number of *ordered* pairs (i, j) with i + j = k and a[i] = a[j] = 1.
If we can keep this count below a small constant `L` while having many
ones, the numerator stays tiny while the denominator (sum = #ones) grows,
giving a small score.

The algorithm repeatedly builds such binary vectors by a greedy
procedure:

*   Choose a random order of the indices 0…n‑1.
*   Keep a table `cnt` of the current ordered‑pair counts for each
    possible sum (size 2 n‑1).
*   Scan the indices in that random order and add an index `p` iff
    - `cnt[2 p] + 1 ≤ L`  (the self‑pair)
    - for every already‑chosen index `q`, `cnt[p+q] + 2 ≤ L`
      (the two ordered pairs (p,q) and (q,p)).
*   When an index is added we update `cnt` accordingly.

A single greedy run gives a vector with many ones and a guaranteed
upper bound `max(conv) ≤ L`.  The theoretical maximum number of ones is
≈ √(2 n L); with `L = 8` the bound is about 250 for n = 3809, enough to
push the score below the target 1.5025.  The routine performs many
random‑restart greedy runs inside the allocated time, keeps the best
vector seen, and occasionally applies the original block‑perturbation
fallback to escape local optima.

The returned list consists of floats (0.0 or 1.0) and satisfies all
security checks in `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Greedy construction of a binary sequence with bounded convolution count
# ----------------------------------------------------------------------
def greedy_binary_sequence(max_cnt: int, rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Build a 0/1 sequence of length ``n`` such that every entry of its
    self‑convolution is at most ``max_cnt``.  The algorithm is greedy
    and uses the given random generator to randomise the order in which
    positions are tried.

    Parameters
    ----------
    max_cnt : int
        Upper bound for any convolution entry (ordered pair count).
    rng : np.random.Generator
        Random generator for shuffling the indices.
    n : int
        Length of the desired sequence.

    Returns
    -------
    np.ndarray
        Float array of shape (n,) containing only 0.0 or 1.0.
    """
    # counts for each possible sum 0 … 2*n‑2 (ordered pairs)
    cnt = np.zeros(2 * n - 1, dtype=np.int16)
    chosen = []                     # list of already accepted positions

    # a random permutation gives different greedy outcomes on each run
    order = rng.permutation(n)

    for p in order:
        self_idx = p + p                       # index for the (p,p) pair
        if cnt[self_idx] + 1 > max_cnt:
            continue

        # Check that adding p would not make any sum exceed max_cnt
        ok = True
        for q in chosen:
            s = p + q
            if cnt[s] + 2 > max_cnt:   # (p,q) and (q,p) are added together
                ok = False
                break
        if not ok:
            continue

        # Accept p: update the count table
        for q in chosen:
            s = p + q
            cnt[s] += 2
        cnt[self_idx] += 1
        chosen.append(p)

    # Build the final 0/1 vector
    seq = np.zeros(n, dtype=np.float64)
    if chosen:
        seq[chosen] = 1.0
    return seq


# ----------------------------------------------------------------------
# Small perturbation helper (kept from the reference implementation)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

# Main entry point used by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a sequence with a low evaluation score.
    ``budget_s`` is the total wall‑clock time allowed (seconds).  The
    function returns the best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Starting point: use the global ``height_sequence_1`` if it exists,
    # otherwise fall back to a uniform vector of ones.
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = np.ones(256, dtype=np.float64)

    n = base_seq.size
    base_seq = np.clip(base_seq, 0.0, 1000.0)
    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # Main optimisation loop: many random‑restart greedy constructions,
    # occasional block perturbations, keep the best result.
    # ------------------------------------------------------------------
    attempts = 0
    while time.monotonic() < deadline:
        # 1) greedy binary construction with a random bound L
        L = int(rng.integers(6, 13))          # try L in {6,…,12}
        candidate = greedy_binary_sequence(L, rng, n)

        # 2) evaluate and possibly keep it
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[{time.monotonic():.2f}] New best: {best_score:.6f} "
                  f"(L={L}, ones={int(candidate.sum())})")

        attempts += 1

        # Occasionally try a small random perturbation of the current best
        if rng.random() < 0.05:
            pert = make_block_perturbation(best, rng)
            pscore = float(evaluate_sequence(pert.tolist()))
            if np.isfinite(pscore) and pscore < best_score:
                best = pert
                best_score = pscore
                print(f"[{time.monotonic():.2f}] Improved by perturbation: "
                      f"{best_score:.6f}")

    print(f"Finished after {attempts} greedy attempts. Best score = {best_score:.6f}")
    return [float(x) for x in best]
