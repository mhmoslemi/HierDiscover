def lp_refine(seq, rng, deadline):
    # Compute current conv and bound M
    conv = np.convolve(seq, seq)
    M = float(conv.max())
    n = seq.size
    # initial active set: index where M attained
    active = {int(np.argmax(conv))}
    # Solve LP iteratively
    # We'll cap iterations to e.g., 12 or time left.
    max_iter = 12
    for _ in range(max_iter):
        # Build LP variables
        b = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active:
            expr = 0.0
            i_start = max(0, k - (n-1))
            i_end = min(k, n-1)
            for i in range(i_start, i_end+1):
                j = k - i
                expr = expr + seq[i] * b[j]
            constraints.append(expr <= M)
        problem = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            problem.solve(solver='ECOS', warm_start=True, max_iters=5000, verbose=False)
        except Exception as e:
            # solver failure, break
            break
        if b.value is None:
            break
        b_val = np.maximum(0.0, np.minimum(1000.0, np.asarray(b.value, dtype=np.float64)))
        # evaluate max conv of seq with b
        conv_fb = np.convolve(seq, b_val)
        new_max = float(conv_fb.max())
        if new_max <= M + 1e-8:
            # feasible solution found, stop
            # Rescale to match original sum (optional)
            sum_seq = seq.sum()
            sum_b = b_val.sum()
            if sum_b > 0:
                scale = sum_seq / sum_b
                b_val = b_val * scale
                b_val = np.clip(b_val, 0.0, 1000.0)
            return b_val
        # otherwise add the violating index (the argmax)
        viol = int(np.argmax(conv_fb))
        if viol in active:
            # already active; break to avoid infinite loop
            break
        active.add(viol)
        # Check time
        if time.monotonic() > deadline - 1.0:
            break
    # If we exit without feasible, just return original seq
    return seq

def block_perturb(base, rng):
    candidate = np.array(base, copy=True, dtype=np.float64)
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    candidate = np.clip(candidate, 0.0, 1000.0)
    return candidate

import numpy as np
import time
import cvxpy as cp

def block_perturb(base, rng):
    """
    Generate a small local perturbation of *base*.
    A random contiguous block is multiplied by a log‑normal factor,
    optionally symmetrised, and finally clipped to the allowed range.
    """
    candidate = np.array(base, copy=True, dtype=np.float64)
    n = candidate.size
    # block width proportional to length but at least a few elements
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # with some probability mirror the whole sequence to exploit symmetry
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # keep values inside the limits enforced by ``evaluate_sequence``
    candidate = np.clip(candidate, 0.0, 1000.0)
    return candidate


def lp_refine(seq, rng, deadline, max_cuts=12):
    """
    Cutting‑plane linear‑programming refinement.
    Starting from ``seq`` we keep the current maximal convolution value ``M``.
    We iteratively grow a set of indices where the convolution of ``seq`` with the
    unknown vector ``b`` must stay below ``M`` and solve a small LP that maximises
    the total sum of ``b``.  The resulting feasible ``b`` has at least the same sum
    as ``seq`` while respecting the original bound, so a convex combination of the
    two yields a strictly better step function.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    M = float(conv.max())
    # start with the index that actually attains the bound
    active = {int(np.argmax(conv))}
    # keep a reference to the variable for warm‑starting
    b_var = cp.Variable(n, nonneg=True)

    for _ in range(max_cuts):
        # stop early if we are out of time
        if time.monotonic() > deadline - 1.0:
            break

        # build the (tiny) linear system
        constraints = []
        for k in active:
            expr = 0.0
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # sum_i seq[i] * b[k-i]
            for i in range(i_start, i_end + 1):
                j = k - i
                expr = expr + seq[i] * b_var[j]
            constraints.append(expr <= M)

        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            # solver failed – give up the refinement
            break

        if b_var.value is None or prob.status not in ["optimal", "optimal_inaccurate"]:
            # infeasible or unbounded – abort the refinement
            break

        b_val = np.asarray(b_var.value, dtype=np.float64)
        # keep the vector inside the domain that ``evaluate_sequence`` will later clip
        b_val = np.clip(b_val, 0.0, 1000.0)

        # check whether the candidate respects the bound on *all* convolution indices
        conv_fb = np.convolve(seq, b_val)
        new_max = float(conv_fb.max())
        if new_max <= M + 1e-9:
            # feasible – upscale/downscale to keep the total sum unchanged
            sum_seq = float(seq.sum())
            sum_b = float(b_val.sum())
            if sum_b > 0:
                scale = sum_seq / sum_b
                b_val = np.clip(b_val * scale, 0.0, 1000.0)
            return b_val

        # otherwise add the most violating index to the active set and repeat
        viol = int(np.argmax(conv_fb))
        # if the same index is already active we stop to avoid endless loops
        if viol in active:
            break
        active.add(viol)

    # If we exit the loop without a feasible point, fall back to the original sequence
    return seq

"""
Hybrid search for a low‑autocorrelation step function.

The routine mixes cheap block‑wise multiplicative perturbations (the
strategy used in the reference code) with a cutting‑plane linear‑program
refinement that follows the idea from the statement:

1.  From the current sequence `f` we compute the current maximal convolution
    value `M = max(f * f)`.
2.  We maximise `sum(b)` subject to `b ≥ 0` and `f * b ≤ M` only on a
    small *active* set of convolution indices.
3.  If the solution violates the bound on some previously ignored index we
    add that index to the active set and resolve (cut‑plane loop).
4.  The feasible `b` has at least as large a total sum as `f` while never
    increasing the maximal convolution, therefore a convex combination
    `f←(1‑t)f + t b` strictly improves the evaluation score.
5.  The outer loop repeatedly decides between a cheap block‑perturbation
    and an LP refinement, keeps the best sequence seen and restarts
    occasionally if progress stalls.

The implementation uses only `numpy`, `cvxpy` (solver ECOS) and the provided
`evaluate_sequence` function.  All numbers are kept in the allowed range
[0, 1000] and the function returns the best sequence found before the
time budget expires.
"""

    # occasional symmetrisation (helps for the reversed convolution)
