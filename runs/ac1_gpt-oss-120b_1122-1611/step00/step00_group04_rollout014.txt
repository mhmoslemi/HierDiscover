def project_simplex(v):
    """Project non-negative vector v onto simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv/ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

```
def max_conv_fft(a):
    N = a.shape[0]
    size = 2*N-1
    # compute FFT length as power of two for speed
    m = 1 << (size-1).bit_length()
    A = np.fft.rfft(a, n=m)
    B = A * A
    conv = np.fft.irfft(B, n=m)[:size]
    return conv
```

Write function simplex_projection(v):
    # Based on algorithm from Wu et al. (Michelot) to project onto simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

```
def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    n = len(height_sequence_1) if 'height_sequence_1' in globals() and hasattr(height_sequence_1,'__len__') else 256
    n = max(2, n)  # ensure at least length 2

    a = a / a.sum()
    # also consider mixing with previous best
    try:

        if prev.shape[0] == n and np.isfinite(prev).all():

            prev = np.clip(prev, 0.0, 1000.0)
            prev = prev / prev.sum()
            a = 0.5 * a + 0.5 * prev

    # ensure a is simplex
    a = _project_simplex(a)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    iter = 0

    while time.monotonic() < deadline:

        conv = _conv_self(a)
```

    """Euclidean projection of v onto the probability simplex {x>=0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
