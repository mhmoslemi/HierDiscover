def attempt_move(a):
    conv = np.convolve(a, a)
    k_max = np.argmax(conv)
    # compute contributions
    idxs = np.arange(len(a))
    j = k_max - idxs
    valid = (j >= 0) & (j < len(a))
    contrib = np.zeros_like(a, dtype=np.float64)
    contrib[valid] = a[valid] * a[j[valid]]
    # Source (largest contribution)
    i_src = np.argmax(contrib)
    i_dst = np.argmin(contrib)  # minimal
    # Transfer fraction
    delta = a[i_src] * 0.1  # reduce by 10%
    if delta < 1e-6:
        return a  # nothing
    b = a.copy()
    b[i_src] = max(0.0, b[i_src] - delta)
    b[i_dst] = min(1000.0, b[i_dst] + delta)  # maintain sum approx same
    return b

def greedy_move(seq, rng, alpha=0.2):
    n = seq.size
    conv = np.convolve(seq, seq)
    k_max = int(conv.argmax())
    # contributions
    idx = np.arange(n)
    j = k_max - idx
    valid = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[valid] = seq[valid] * seq[j[valid]]
    src = int(np.argmax(contrib))
    # choose destination as index with smallest contribution
    dst = int(np.argmin(contrib))
    if src == dst:
        # choose random from low contribution set
        low_candidates = np.where(contrib <= np.percentile(contrib, 5))[0]
        if low_candidates.size > 0:
            dst = int(rng.choice(low_candidates))
        else:
            dst = (src + n // 2) % n  # fallback

    # transfer amount
    amount = seq[src] * alpha
    if amount < 1e-8:
        # fallback random perturbation
        return make_block_perturbation(seq, rng)
    cand = seq.copy()
    cand[src] = max(0.0, cand[src] - amount)
    cand[dst] = min(1000.0, cand[dst] + amount)
    # ensure minimal sum: maybe add small epsilon to avoid sum too low
    return cand

Now LP function:

def solve_lp_max_sum(f, M, active_idx):
    import numpy as np, scipy.optimize as opt
    n = f.size
    K = active_idx.size
    if K == 0:
        return None
    # Build A matrix dense
    A = np.zeros((K, n), dtype=np.float64)
    for row_i, k in enumerate(active_idx):
        i_start = max(0, int(k - (n - 1)))
        i_end = min(n - 1, int(k))
        # slice f from i_start to i_end inclusive
        A[row_i, i_start:i_end + 1] = f[i_start:i_end + 1]

    b = np.full(K, M, dtype=np.float64)
    # maximize sum of g => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None) for _ in range(n)]
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x, 0.0, 1000.0)
        else:
            return None
    except Exception as e:
        return None

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step-height sequence minimizing evaluate_sequence.

    Uses a combination of greedy max-convolution reduction moves and occasional
    linear‑programming refinement of the sum under the current upper bound.
    """
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # initialize from the global variable if possible
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    # clip to allowed range
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    # start with parent as current best
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # also keep a separate current that we mutate more aggressively
    current = best.copy()
    current_score = best_score

    iter_cnt = 0
    last_improve_time = time.monotonic()

    while time.monotonic() < deadline:
        iter_cnt += 1

        # Greedy move aimed at the current maximal convolution entry
        cand = greedy_move(current, rng, alpha=0.2)
        cand_score = float(evaluate_sequence(cand.tolist()))

        # Accept if better than current (not just best)
        if np.isfinite(cand_score) and cand_score < current_score:
            current = cand
            current_score = cand_score

        # Track best overall
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            last_improve_time = time.monotonic()
            print(f"Iter {iter_cnt}: new best score = {best_score:.6f}")

        # Occasionally run a linear‑programming refinement (every few seconds)
        if (time.monotonic() - last_improve_time) > 5.0:
            # recompute with best so far to get a more reliable bound
            conv = np.convolve(best, best)
            M = float(conv.max())
            # take the top 2% of shifts as active constraints
            thresh = M * 0.99
            active_idx = np.where(conv >= thresh)[0]
            if active_idx.size > 0:
                g = solve_lp_max_sum(best, M, active_idx)
                if g is not None:
                    # blend a fraction of the LP solution into the current sequence
                    t = 0.1  # step size
                    blended = (1.0 - t) * best + t * g
                    blended = np.clip(blended, 0.0, 1000.0)
                    blended_score = float(evaluate_sequence(blended.tolist()))
                    if np.isfinite(blended_score) and blended_score < best_score:
                        best = blended
                        best_score = blended_score
                        current = blended
                        current_score = blended_score
                        last_improve_time = time.monotonic()
                        print(f"Iter {iter_cnt} (LP): improved score = {best_score:.6f}")

        # If we haven't improved for a while, do a random block perturbation to escape local minima
        if (time.monotonic() - last_improve_time) > 30.0:
            # random block
            cand2 = make_block_perturbation(best, rng)
            cand2_score = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(cand2_score) and cand2_score < best_score:
                best = cand2
                best_score = cand2_score
                current = cand2
                current_score = cand2_score
                last_improve_time = time.monotonic()
                print(f"Iter {iter_cnt} (random): new best = {best_score:.6f}")

    # Return the best sequence found (as plain Python list of floats)
    return [float(x) for x in best]

"""
search.py – a fast local‑search for the step‑height optimisation problem.

The evaluator

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

is minimized by making the self‑convolution as flat as possible.
The algorithm mixes two complementary ideas:

1. **Greedy bottleneck reduction** – at each iteration we look at the shift
   `k_max` where the convolution attains its maximum.  The product
   `a[i]·a[k_max‑i]` tells how much each coefficient participates in that
   bottleneck.  A small amount of mass is moved from a high‑contributing
   index to a low‑contributing one.  This monotonically lowers the peak
   while keeping the total sum (and therefore the denominator) unchanged.

2. **Periodic linear‑programming refinement** – after a few seconds of
   greedy moves we solve a tiny LP that maximises the total sum under the
   current bound `M = max(conv(f,f))`.  Only the *active* constraints
   (shifts whose convolution is within 99 % of `M`) are kept, so the LP
   is tiny (≤ ~100 rows) even for thousands of coefficients.  The LP solution
   `g` is blended into the current sequence with a small mixing factor.
   This mirrors the state‑of‑the‑art iteration described in the reference
   material but is far cheaper because the constraint matrix is tiny.

Both phases are cheap (O(n) for the greedy step, O(K·n) for the LP with
K≪n) and fit easily into the 1000 s budget.  Random block perturbations are
used as a fallback when the search stalls.

The function `propose_candidate` returns the best sequence found as a plain
list of floats.
"""

import time
import numpy as np
from scipy import optimize


def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def greedy_move(seq, rng, alpha=0.15):
    """
    One greedy step: identify the convolution peak, move a small fraction
    of mass from a high‑contributing index to a low‑contributing one.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k_max = int(conv.argmax())

    # contributions of each index to the peak shift
    idx = np.arange(n)
    partner = k_max - idx
    valid = (partner >= 0) & (partner < n)
    contrib = np.zeros_like(seq)
    contrib[valid] = seq[valid] * seq[partner[valid]]

    # choose source among the top 5 % contributors
    high_thr = np.percentile(contrib, 95)
    high = np.where(contrib >= high_thr)[0]
    if high.size == 0:
        src = int(np.argmax(contrib))
    else:
        src = int(rng.choice(high))

    # choose destination among the bottom 5 % contributors
    low_thr = np.percentile(contrib, 5)
    low = np.where(contrib <= low_thr)[0]
    if low.size == 0:
        dst = int(np.argmin(contrib))
        if dst == src:
            dst = (src + n // 2) % n
    else:
        dst = int(rng.choice(low))
        if dst == src:
            dst = (src + n // 2) % n

    # amount to transfer (a fraction of the source value)
    amount = seq[src] * alpha
    if amount < 1e-9:               # nothing useful to move – fall back
        return make_block_perturbation(seq, rng)

    cand = seq.copy()
    cand[src] = max(0.0, cand[src] - amount)
    cand[dst] = min(1000.0, cand[dst] + amount)
    return cand


def solve_lp_max_sum(f, M, active_idx):
    """
    Solve   max  Σ g_i
    s.t.    (f * g)[k] ≤ M   for every k in ``active_idx``,
            g_i ≥ 0.
    Returns the optimal g (clipped to the allowed range) or None on failure.
    """
    n = f.size
    K = active_idx.size
    if K == 0:
        return None

    # Build a dense matrix – it stays tiny because K ≪ n.
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        i_start = max(0, int(k - (n - 1)))
        i_end = min(n - 1, int(k))
        # The coefficient of g_i in the convolution row is simply f_i
        A[row, i_start:i_end + 1] = f[i_start:i_end + 1]

    b = np.full(K, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                 # maximise Σg  ↔ minimise –Σg
    bounds = [(0, None) for _ in range(n)]

    try:
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                               method='highs', options={'presolve': True})
        if res.success:
            return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        pass
    return None
