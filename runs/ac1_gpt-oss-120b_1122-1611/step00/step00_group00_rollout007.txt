def project_simplex(v, s=1):
    """Projects vector v onto the simplex {x: x>=0, sum x = s}."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # Find rho = max i s.t. cond true
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

```python
def project_simplex(v, s=1.0):
    """Project onto simplex {x >=0, sum(x) = s}."""
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not cond.any():
        # all zero case
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

implemented with an exponentiated‑gradient update followed by a projection
onto the probability simplex.  Random jitter is injected periodically to
avoid stagnation at symmetric points (e.g. the uniform distribution).

The routine runs until the allotted time expires and returns the best
sequence (scaled to the conventional ‑√(2n)‑norm) it has seen.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σ x = s}.

    Uses the O(n log n) algorithm of Duchi et al.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform point
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a step‑function height sequence that (heuristically) minimises the
    given `evaluate_sequence` value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds (the function will stop a few
        seconds before this limit).

    Returns
    -------
    list[float]
        The best height sequence found within the time limit.
    """
    rng = np.random.default_rng(seed)
    # give ourselves a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1. Determine the length of the sequence.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = base.size
    except Exception:
        # fallback – a power‑of‑two length is convenient for FFT‑based conv
        n = 4096
        base = None

    # ------------------------------------------------------------------
    # 2. Initialise a probability vector (sum = 1).  A modest random
    #    perturbation of the uniform distribution breaks symmetry.
    # ------------------------------------------------------------------
    p = np.ones(n, dtype=np.float64) / n
    jitter = np.exp(rng.normal(0.0, 0.15, size=n))
    p *= jitter
    p = p / p.sum()                     # ensure exact sum = 1

    # Keep the best sequence seen so far
    best = p.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Hyper‑parameters for the mirror‑descent loop
    # ------------------------------------------------------------------
    eta = 0.5                # initial step size
    eta_min = 1e-4
    eta_decay = 0.9995       # exponential decay per iteration

    alpha = 20.0             # soft‑max concentration (higher → tighter on the max)
    alpha_max = 5000.0
    alpha_inc = 1.001        # slowly increase to focus on the peak

    iteration = 0

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (runs until time runs out)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- convolution (p * p) -----
        conv_full = np.convolve(p, p)          # length 2n-1

        # ----- soft‑max weighting of the lag‑values -----
        max_c = np.max(conv_full)
        # numerical stability: shift by max
        w = np.exp(alpha * (conv_full - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            # degenerate case – fall back to uniform weighting
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the weighted sum w.r.t. p -----
        # grad = 2 * (w * p_rev)   (where * denotes convolution)
        p_rev = p[::-1]
        grad_full = np.convolve(w, p_rev)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # ----- mirror‑descent update on the simplex -----
        p = p * np.exp(-eta * grad)             # multiplicative step
        p = _project_simplex(p, s=1.0)          # enforce Σp = 1, p ≥ 0

        # ----- occasional random jitter to avoid dead‑ends -----
        if iteration % 500 == 0 and iteration > 0:
            eps = rng.normal(0.0, 0.05, size=n)
            p = p * np.exp(eps)
            p = _project_simplex(p, s=1.0)

        # ----- periodic evaluation of the true objective -----
        if iteration % 20 == 0:
            cur_score = float(evaluate_sequence(p.tolist()))
            if np.isfinite(cur_score) and cur_score < best_score:
                best = p.copy()
                best_score = cur_score
                print(f"[iter {iteration:5d}] new best: {best_score:.6f}")

        # ----- update schedules -----
        eta = max(eta_min, eta * eta_decay)
        alpha = min(alpha_max, alpha * alpha_inc)

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Transform the best probability vector back to a concrete height
    #    sequence.  Scaling does not affect the objective, but we adopt the
    #    common normalisation ∑a = √(2n) used in the reference solution.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    best_heights = best * (target_sum / best.sum())
    best_heights = np.clip(best_heights, 0.0, 1000.0)   # safety clip

    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return [float(v) for v in best_heights]
