"""
Propose a better step‑function sequence.

The algorithm performs a projected gradient descent on a smooth
approximation of the quantity  max(conv(a,a)).
The max is approximated with a soft‑max (log‑sum‑exp) using a temperature
parameter β that is slowly increased (annealed).  The gradient of the
smooth objective can be written analytically and is evaluated with FFT
convolutions, which makes each iteration O(n log n) even for n≈7000.

The iterate is kept non‑negative and re‑scaled after every step (the
objective is scale‑invariant).  A simple line‑search adapts the step
size: it is increased when an improvement is seen and decreased otherwise.
Occasional random perturbations help to escape local minima.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution values, stabilised by subtracting max."""
    # subtract the maximum to avoid overflow
    max_c = conv.max()
    # exponentials; clip the exponent to avoid underflow to zero
    exp_arg = beta * (conv - max_c)
    exp_arg = np.clip(exp_arg, -50, None)          # ~1e‑22 is already zero for float64
    w_raw = np.exp(exp_arg)
    total = w_raw.sum()
    if total == 0.0:
        # fallback to uniform weights (should not happen)
        return np.full_like(w_raw, 1.0 / w_raw.size)
    return w_raw / total


def _gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed max–convolution objective.

    grad_i = 2 * sum_k w_k * a_{k-i}
    This is the (full) convolution of w and a, taken at indices 0…n‑1.
    """
    FA = np.fft.rfft(a, n=fft_len)
    FW = np.fft.rfft(w, n=fft_len)
    conv = np.fft.irfft(FW * FA, n=fft_len)[: a.size]   # first n entries
    return 2.0 * conv


def _convolution_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a*a (full convolution) using FFT (length 2n‑1)."""
    FA = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(FA * FA, n=fft_len)[: 2 * a.size - 1]
    return conv


def _initial_sequence(seed: int, base_seq: np.ndarray) -> np.ndarray:
    """
    Produce a starting sequence.
    - If a valid global `height_sequence_1` is available, use it.
    - Otherwise fall back to a uniform vector.
    - Add a small random perturbation to break symmetry.
    """

    if base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
        n = 256
        a = np.ones(n, dtype=np.float64)
    else:
        a = base_seq.astype(np.float64).copy()
        n = a.size

    # enforce the safety checks that evaluate_sequence applies internally
    a = np.clip(a, 0.0, 1000.0)

    # normalise to a probability vector (the score is scale invariant)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s

    # tiny random noise to avoid the symmetric stationary point
    a += 0.01 * rng.normal(size=n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_new.copy()
            best_score = score
            # if we made progress we become a bit more aggressive
            step *= 1.05
        else:
            # otherwise retreat a little
            step *= 0.70

        a = a_new

        # ------------------------------------------------------------------
        # anneal β (increase focus on the true max) every few iterations
        # ------------------------------------------------------------------
        if it % 20 == 0:
            beta = min(beta * 1.05, 200.0)                 # cap to keep exponents stable

        # ------------------------------------------------------------------
        # occasional random jitter to escape shallow local minima
        # ------------------------------------------------------------------
        if it % 200 == 0:
            a += rng.normal(scale=0.01, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # ------------------------------------------------------------------
        # logging (once every 100 iterations)
        # ------------------------------------------------------------------
        if it % 100 == 0:
            print(
                f"Iter {it:5d} | best_score={best_score:.6f} | "
                f"step={step:.4f} | beta={beta:.2f}"
            )

        # early stop if we already beat the target (optional)
        if best_score < 1.5025:
            print("Target reached – exiting early.")
            break

    # final sanity check – ensure the returned list is acceptable
    result = [float(x) for x in best]
    return result
