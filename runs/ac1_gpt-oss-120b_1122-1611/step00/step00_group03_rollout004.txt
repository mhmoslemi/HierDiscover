def lp_step(f, M):
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for s in range(rows):
        j_start = max(0, s - (n - 1))
        j_end = min(s, n - 1)
        # fill columns j_start..j_end
        # compute f[s - j] for each j

"""
Propose a candidate step‑function sequence.

The algorithm starts from the provided global sequence `height_sequence_1`
(or a simple uniform fallback).  It then performs a few iterations of a
linear‑programming based “active‑set” improvement described in the
statement: for the current vector f we maximise the L1‑mass of a vector g
subject to the convolution bound ‖f∗g‖∞≤M (M = max(f∗f)).
The LP solution g can carry more total mass without increasing the
critical convolution value.  After scaling g back to the original L1‑mass
we mix it with f (t = 0.5) – this provably never increases
‖f∗f‖∞.  A few LP steps quickly drive the objective
2·n·max(f∗f) / (∑f)² below the trivial value 2.

When the remaining time budget is small we fall back to the fast
block‑perturbation local search (the method already used in the starter
code) to polish the solution.

The whole routine respects the 1000 s budget, prints progress and
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _conv_max(seq: np.ndarray) -> float:
    """Return the maximum value of the full convolution seq*seq."""
    conv = np.convolve(seq, seq, mode='full')
    return float(conv.max())


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One O(n) non‑negative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # width roughly between n/200 and n/12 (as in the starter code)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # Apply the same clipping that `evaluate_sequence` does
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2n‑1)×n matrix A such that (A @ g) = f * g (full convolution)."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # Fill row by row – each row is a shifted copy of f (zero‑padded)
    # Row s contains f[s‑j] for columns j where 0≤s‑j<n.
    for s in range(rows):
        j_start = max(0, s - (n - 1))
        j_end = min(s, n - 1)
        # Columns j_start … j_end are the only non‑zero entries.
        # The corresponding coefficients are f[s‑j].
        col_range = slice(j_start, j_end + 1)
        A[s, col_range] = f[s - np.arange(j_start, j_end + 1)]
    return A


def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(g)
        subject to A @ g ≤ M   (A is the Toeplitz convolution matrix for f)
                    g ≥ 0
    Returns the optimal g or None if the solver fails.
    """
    n = f.size
    A = _build_toeplitz_matrix(f)

    # Linprog minimises; maximise sum(g) → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)          # objective coefficients
    b_ub = np.full(A.shape[0], M, dtype=np.float64)  # RHS = M for every row
    bounds = [(0, None)] * n                   # g_j ≥ 0

    # Use HiGHS (fast interior‑point / simplex implementation)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 30.0}  # safety cap
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    evaluate_sequence.  The routine combines a few LP‑based refinement
    steps with a fast block‑perturbation local search for the remaining
    time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 3.0)   # keep a small safety margin

    # ------------------------------------------------------------------
    # Initialise the parent vector (fallback to a uniform vector if needed)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 should exist globally; make a safe copy.
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        parent = np.ones(256, dtype=np.float64)          # fallback

    # Sanity checks – clip to the domain allowed by evaluate_sequence
    np.clip(parent, 0.0, 1000.0, out=parent)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    best = parent
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={best.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – a few LP improvement steps (expensive but powerful)
    # ------------------------------------------------------------------
    max_lp_iters = 6            # a handful; each may take a few seconds
    lp_iter = 0
    while lp_iter < max_lp_iters and time.monotonic() < deadline:
        n = best.size
        M = _conv_max(best)

        # Solve the LP
        g = _lp_step(best, M)
        if g is None:
            print("[LP] solver failed – exiting LP phase")
            break

        sum_f = best.sum()
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] trivial solution – exiting LP phase")
            break

        # Scale g back to the original L1‑mass so that mixing keeps the sum constant
        scale = sum_f / sum_g
        g_scaled = g * scale

        # Mix (1‑t)f + t·g_scaled.  t=0.5 works well and respects the monotonicity proof.
        t = 0.5
        candidate = (1.0 - t) * best + t * g_scaled
        np.clip(candidate, 0.0, 1000.0, out=candidate)

        cand_score = float(evaluate_sequence(candidate.tolist()))
        print(f"[LP] iter={lp_iter+1}, n={n}, M={M:.3e}, score={cand_score:.6f}")

        if cand_score < best_score:
            best = candidate
            best_score = cand_score
            print(f"   -> improvement! new best={best_score:.6f}")
        else:
            # No improvement – stop LP refinements; further LP steps are unlikely to help.
            print("[LP] no improvement – stop LP phase")
            break

        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – fast block perturbations for the remaining time
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = _make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[perturb] improved to {best_score:.6f}")

    print(f"[final] length={best.size}, best_score={best_score:.6f}")
    # Return a plain Python list as required by the evaluator
    return [float(x) for x in best.tolist()]
