    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # Choose sequence length n. We could use existing height_sequence_1 length or a new.
    n = 512  # or 1024
    # Initialize sequence with uniform over n.
    seq = np.full(n, 1.0 / n, dtype=np.float64)  # sum to 1
    best_seq = seq.copy()
    best_ratio = evaluate_sequence(seq.tolist())
    current_seq = seq.copy()
    current_ratio = best_ratio
    # Simulated annealing parameters
    temperature = 0.1
    while time.monotonic() < deadline:
        # move type: transfer from i to j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # delta = random fraction up to 0.5 of current value at i
        max_delta = current_seq[i] * 0.5
        if max_delta <= 1e-12:
            continue
        delta = rng.random() * max_delta
        # create new seq
        new_seq = current_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # evaluate
        new_ratio = 2 * n * np.max(np.convolve(new_seq, new_seq))  # since sum=1
        # acceptance
        if new_ratio < current_ratio or rng.random() < np.exp((current_ratio - new_ratio) / temperature):
            current_seq = new_seq
            current_ratio = new_ratio
            if new_ratio < best_ratio:
                best_seq = new_seq.copy()
                best_ratio = new_ratio
        # decrease temperature
        temperature *= 0.999999  # small decay
    # Final conversion to list, ensure nonnegative and clip
    best_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return best_list

"""
Propose candidate using simulated‑annealing style stochastic search.

The algorithm works on a probability vector (the coefficients are kept
normalised to sum = 1, which does not change the evaluation because the
objective is scale‑invariant).  Starting from a uniform distribution we
repeatedly perform small “transfer” moves – moving a random amount of
mass from one index to another – and accept them if they improve the
evaluation or with a temperature‑dependent probability.  The temperature
is slowly cooled.  The best sequence found before the time budget
expires is returned (clipped to the allowed range).
"""

import time
import numpy as np

# Helper: compute the evaluation value without the overhead of the
# provided evaluate_sequence (the result is identical for non‑negative
# inputs that sum to 1).
def _fast_score(seq: np.ndarray) -> float:
    """Return 2 * n * max(conv(seq, seq)) for a sequence with sum == 1."""
    conv = np.convolve(seq, seq)
    return 2.0 * seq.size * float(conv.max())

def _transfer_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Propose a new sequence by moving a random fraction of the mass from
    one index to another.  The total sum stays unchanged.
    """
    n = seq.size
    i = rng.integers(0, n)
    # ensure j != i
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1

    if seq[i] <= 1e-12:                     # nothing to move
        return None
    # move at most 50 % of the source value
    delta = rng.random() * 0.5 * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return new_seq

def _scale_block_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Like the baseline block perturbation but works on a normalised
    sequence (sum = 1).  A random contiguous block is multiplied by a
    log‑normal factor and the whole vector is re‑normalised.
    """
    n = seq.size
    # width is a small fraction of n (avoid degenerate blocks)
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = np.exp(rng.normal(0.0, 0.12))
    new_seq = seq.copy()
    new_seq[start:stop] *= factor
    # optional mirroring – mirrors the vector with 35 % probability
    if rng.random() < 0.35:
        new_seq = 0.5 * (new_seq + new_seq[::-1])
    # re‑normalise to keep sum == 1 (scale invariance)
    total = new_seq.sum()
    if total <= 0:
        return None
    new_seq /= total
    return new_seq

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a coefficient vector with a low evaluation value.

    The routine runs for roughly `budget_s` seconds (default 1100 s)
    using a simulated‑annealing loop that mixes two simple moves:
    – a mass transfer between two positions,
    – a random block scaling (similar to the reference implementation).
    The best vector found is returned as a plain list of Python floats.
    """
    rng = np.random.default_rng(seed)
    # ---------------------------------------------------------------------------------
    # Choose the length of the vector.
    # If a previous construction exists we keep its length, otherwise fall back to 512.
    # ---------------------------------------------------------------------------------
    try:
        base_len = len(height_sequence_1)          # type: ignore[name-defined]
        n = max(256, base_len)                    # at least a few hundred entries
    except Exception:
        n = 512

    # Initialise a uniform probability vector (sum == 1).  This is the reference
    # point that yields a score of exactly 2.0.
    cur_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = cur_seq.copy()
    best_score = _fast_score(best_seq)

    # Simulated‑annealing parameters
    temperature = 0.02                           # initial temperature
    decay = 0.999995                             # exponential decay per iteration
    min_temp = 1e-6

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iter_cnt = 0
    last_report = time.monotonic()

    while time.monotonic() < deadline:
        # Choose a move type (transfer ≈ 70 %, block ≈ 30 %)
        if rng.random() < 0.7:
            cand = _transfer_move(cur_seq, rng)
        else:
            cand = _scale_block_move(cur_seq, rng)

        if cand is None:
            continue

        # Fast evaluation (sum is 1, so the score is 2 · n · max(conv))
        cand_score = _fast_score(cand)

        # Acceptance rule (Metropolis)
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score

        delta = cand_score - _fast_score(cur_seq)
        accept = False
        if delta < 0:
            accept = True
        else:
            # probability exp(-Δ/T); avoid under‑flow for tiny temperatures
            prob = np.exp(-delta / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = cand

        # Temperature schedule
        temperature = max(min_temp, temperature * decay)

        iter_cnt += 1

        # Periodic progress report (every ~2 s)
        if time.monotonic() - last_report > 2.0:
            print(f"Iter {iter_cnt:8d} – best score {best_score:.6f} – "
                  f"temp {temperature:.2e}")
            last_report = time.monotonic()

    # Final post‑processing: clip to allowed range and ensure positivity.
    # The evaluator will apply the same clipping, but we do it here
    # for safety.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Convert to plain Python floats (required return type)
    result = [float(x) for x in best_seq.tolist()]
    print(f"Search finished after {iter_cnt} iterations, best score = {best_score:.6f}")
    return result
