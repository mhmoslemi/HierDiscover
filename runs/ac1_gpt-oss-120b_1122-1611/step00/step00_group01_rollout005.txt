"""
Propose a candidate step function that minimizes the evaluation score.

The algorithm is based on a simple analytical insight:
if we choose a monotone polynomial shape
    a_i = M * ((i+1)/N) ** p      (M=1000, N = length, p > 0)
then the maximum of the self‑convolution is attained at the centre and
its value scales like M²·N·Beta(p+1,p+1), while the total sum scales like
M·N/(p+1).  Consequently the evaluation score tends to

    score(p) = 2 * (p+1)² * Beta(p+1,p+1),

which is < 2 for any p>0 and drops quickly as p grows (p=4 → ~0.08).
Thus a short polynomial‑shaped sequence already beats the target 1.5025.

The function performs a quick deterministic grid search over several
lengths and exponents, then continues with random sampling of (N,p)
pairs until the allocated time runs out, always keeping the best
candidate seen.  Small Gaussian perturbations of the current best are
also tried to escape possible plateaus.

The final best sequence (as a plain Python list of floats) is returned.
"""

import time
import numpy as np

# The global variable ``height_sequence_1`` may exist; we keep it as a
# fallback but it is not required for the main strategy.
# It will be injected by the judge environment.

def _poly_sequence(length: int, exponent: float, max_val: float = 1000.0) -> np.ndarray:
    """
    Return a monotone polynomial sequence a_i = max_val * ((i+1)/length) ** exponent.
    The values are already within [0, max_val].
    """
    idx = np.arange(1, length + 1, dtype=np.float64)
    seq = max_val * (idx / length) ** exponent
    # Numerical safety – clip to the exact limits used by evaluate_sequence.
    return np.clip(seq, 0.0, max_val)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence that yields a low evaluation score.
    The search is mostly analytical (polynomial shapes) with a small
    stochastic component to fine‑tune the exponent and length.
    Returns a list of floats that can be passed directly to evaluate_sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------------
    # 1.  Initialise best candidate with a simple fallback (uniform ones)
    # --------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        fallback = np.asarray(height_sequence_1, dtype=np.float64)
        fallback = np.clip(fallback, 0.0, 1000.0)
    else:
        fallback = np.ones(256, dtype=np.float64)

    best_seq = fallback.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] baseline score = {best_score:.6f} (length={best_seq.size})")

    # --------------------------------------------------------------------
    # 2.  Deterministic grid search (covers the most promising region)
    # --------------------------------------------------------------------
    lengths = [32, 64, 128, 256, 512, 1024]      # modest sizes, keep within 1000 ms
    exponents = [1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]

    for N in lengths:
        for p in exponents:
            cand = _poly_sequence(N, p)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_seq, best_score = cand, score
                print(f"[grid] new best: score={score:.6f}, N={N}, p={p}")

    # --------------------------------------------------------------------
    # 3.  Random exploration + tiny perturbations until budget expires
    # --------------------------------------------------------------------
    # Parameter ranges for the random phase
    min_len, max_len = 20, 2000          # keeps evaluation cheap
    min_exp, max_exp = 0.5, 6.0          # covers a wide family

    while time.monotonic() < deadline:
        # Occasionally (≈10 % of the time) perturb the current best
        if rng.random() < 0.09 and best_seq.size > 0:
            # Gaussian perturbation with relative std ~2 %
            noise = rng.normal(0.0, 0.02, size=best_seq.shape)
            cand = best_seq * (1.0 + noise)
            cand = np.clip(cand, 0.0, 1000.0)
        else:
            # Sample a fresh polynomial shape
            N = int(rng.integers(min_len, max_len + 1))
            p = float(rng.uniform(min_exp, max_exp))
            cand = _poly_sequence(N, p)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq, best_score = cand, score
            print(f"[new]  best updated: score={score:.6f}, N={cand.size}, p={p:.3f}")

    # --------------------------------------------------------------------
    # 4.  Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}, length = {best_seq.size}")
    return [float(x) for x in best_seq]
