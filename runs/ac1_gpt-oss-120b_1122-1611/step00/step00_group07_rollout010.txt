"""
Propose a best step‑function sequence.

The key idea is to use a simple parametric family
    a_i = ((i+1)/n)**p          (i = 0 … n‑1)
which interpolates between the uniform sequence (p=0) that gives the
baseline score 2 and increasingly “tilted’’ sequences for p>0.
For large n the evaluation function reduces to

    score(p) = 2·(p+1)²·Beta(p+1, p+1)   (Beta – Euler beta function)

and this expression falls below the target 1.5025 when p≈0.75…0.80.
We therefore scan a set of exponents, keep the best candidate,
and afterwards perform a short stochastic refinement using the
block‑perturbation routine from the reference code.

All functions are top‑level – no closures – and the search respects the
time budget supplied via ``budget_s`` (default 1100 s).  The final
sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – the same perturbation routine that was used in the reference.
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width proportional to the current length
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiply the whole block by a log‑normal factor
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring (adds a little symmetry)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)
