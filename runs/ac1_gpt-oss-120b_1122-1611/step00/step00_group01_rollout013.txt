def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 5000
    # start from given sequence if valid.
    best = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if best.ndim != 1 or best.size == 0 or not np.isfinite(best).all():
        best = np.ones(n, dtype=np.float64) * 0.1
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"initial score: {best_score}")

    # predefine families generator functions
    def make_power(beta, offset=0.0):
        seq = (np.arange(1, n+1, dtype=np.float64) ** beta) + offset
        return np.clip(seq, 0.0, 1000.0)

    def make_sine(gamma=1.0, offset=0.0):
        angles = np.pi * (np.arange(n) + 0.5) / n
        seq = np.sin(angles) ** gamma + offset
        return np.clip(seq, 0.0, 1000.0)

    def make_triangle():
        half = np.arange(1, n//2+1)
        if n%2==0:
            seq = np.concatenate([half, half[::-1]])
        else:
            seq = np.concatenate([half, half[-2::-1]])
        return np.clip(seq.astype(np.float64), 0.0, 1000.0)

    # grid search
    betas = np.arange(-0.7, -0.2, 0.05)
    offsets = np.linspace(0.0, 1.0, 5)
    for beta in betas:
        for off in offsets:
            cand = make_power(beta, off)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                print(f"new best beta={beta:.3f} offset={off:.3f} score={score:.6f}")

    # sine family
    gammas = [0.5, 1.0, 2.0]
    for gamma in gammas:
        for off in offsets:
            cand = make_sine(gamma, off)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                print(f"new best sine gamma={gamma} offset={off:.3f} score={score:.6f}")

    # try triangle shape
    cand = make_triangle()
    score = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(score) and score < best_score:
        best = cand
        best_score = score
        print(f"new best triangle score={score}")

    # local refinement: small block perturbations around best for remaining time
    while time.monotonic() < deadline:
        # similar to original perturbation but with smaller variance
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"refined score={score}")

    print(f"final best score: {best_score}")
    return [float(x) for x in best]

"""
Propose a high‑quality coefficient sequence.

The search builds on the well‑known optimal continuous shape a_i ≈ 1/√i
(which gives a theoretical limit score π/2 ≈ 1.5708) and then
explores a small family of variants (different shifts, exponents,
and a sinusoidal envelope).  The best candidate among these is further
refined by cheap block‑wise multiplicative perturbations.
The whole procedure respects the 1000 s budget and the required
non‑negative, ≤1000 bounds.
"""
import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (all at top‑level – no closures)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width roughly 0.5–5 % of the sequence length
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    # multiplicative log‑normal tweak of the whole block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[st:en] *= factor
    # very occasional symmetrisation – helps especially for symmetric shapes
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the bounds required by evaluate_sequence
    return np.clip(cand, 0.0, 1000.0)


def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift)^beta  (beta can be negative)."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_sine(gamma: float, n: int) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n)^gamma   – symmetric hump shape."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    return np.clip(seq, 0.0, 1000.0)


def generate_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape – rises then falls linearly."""
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        seq = np.concatenate([half, half[::-1]])
    else:
        seq = np.concatenate([half, half[-2::-1]])
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the test harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimizes the
    evaluation function.  The routine:
      1. builds a handful of analytical families (power law, sine hump,
         triangular);
      2. evaluates them on the given length (taken from the provided
         ``height_sequence_1`` if available);
      3. remembers the best one;
      4. spends the remaining time running cheap block‑perturbations
         around the best candidate.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine the working length.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 6000                     # a safe default size
    print(f"search length = {n}")

    # ------------------------------------------------------------------
    # Start from a reasonable deterministic seed (uniform ones) so we
    # always have a valid fallback.
    # ------------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"initial uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1)  Power‑law family   a_i = (i+shift)^beta
    # ------------------------------------------------------------------
    betas = np.arange(-0.7, -0.4 + 1e-9, 0.05)   # explore around -0.5
    shifts = np.linspace(0.5, 5.0, 6)           # avoid the singularity at i=0
    for beta in betas:

            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
