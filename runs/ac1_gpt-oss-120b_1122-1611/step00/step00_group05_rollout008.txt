```python
def euclidean_proj_simplex(v, s=1):
    """Projection of a vector v onto the simplex {w >=0, sum w = s}."""
    if s < 0:
        raise ValueError...
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho = max {j | u_j - cssv_j/j > 0}
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v, s):

"""
Gradient + simple‑annealing search for a low‑max‑convolution step function.

The evaluation metric (with sum normalized to sqrt(2·n)) equals the maximum
value of the self‑convolution.  The algorithm keeps a feasible vector
(a_i ≥ 0, Σ a_i = √(2n)) and repeatedly:
    * computes the convolution by FFT,
    * finds the shift that attains the current maximum,
    * takes a sub‑gradient step that reduces the overlap for this shift,
    * projects back onto the simplex (exact sum and non‑negativity),
    * occasionally applies a random block perturbation to escape plateaus,
    * accepts the new vector if it improves the score (simulated‑annealing
      acceptance of worse moves is also used early on).

The routine runs until the time budget expires and returns the best
sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x >= 0, sum x = s}
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    if s < 0:
        raise ValueError("Simplex target sum must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)          # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast linear convolution using FFT (real input → real output)
def _full_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a using FFT; returns length 2·n‑1 array."""
    n = a.shape[0]
    # length for zero‑padding: next power‑of‑two >= 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# The perturbation used by the reference implementation (copied here)
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.array(base, dtype=np.float64).copy()
    n = cand.size
    # width of the block (between n/200 and n/12)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimizes
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and basic normalization target
    try:
        # height_sequence_1 is provided globally by the platform
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = None

    if base_seq is None or base_seq.ndim != 1 or base_seq.size == 0:
        n = 256
        a = np.ones(n, dtype=np.float64)
    else:
        n = base_seq.size
        a = np.clip(base_seq, 0.0, 1000.0)

    # Target sum Σa_i = sqrt(2·n) (the usual normalisation)
    target_sum = np.sqrt(2.0 * n)

    # Initialise with a random Dirichlet‑like vector if the start is uniform
    if np.allclose(a, a[0]):                 # uniform or all‑ones → need randomness
        a = rng.exponential(scale=1.0, size=n)
    # Scale to the exact target sum and enforce simplex constraints
    a = a * (target_sum / a.sum())
    a = _project_onto_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Keep track of the best solution we have seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, target_sum={target_sum:.4f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Search parameters
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    it = 0
    # Base learning rate (relative to the typical magnitude of entries)
    base_lr = 0.5 * target_sum                     # about 5 % of the total mass
    # Annealing temperature for occasional uphill moves
    temp_start = 1.0
    temp_end = 0.001

    while time.monotonic() < deadline:
        it += 1
        # ----- compute convolution and the active shift -----------------
        conv = _full_convolve(a)
        max_b = conv.max()
        k_max = int(conv.argmax())

        # ----- sub‑gradient of the active quadratic term ---------------
        # grad_i = 2 * a_{k_max - i}   (zero when index out of range)
        idx = np.arange(n)
        rev_idx = k_max - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[rev_idx[mask]]

        # ----- learning‑rate schedule ---------------------------------
        # simple decay; also shrink as we approach the deadline
        elapsed = it
        lr = base_lr / np.sqrt(elapsed + 1.0)

        # ----- propose a gradient step ---------------------------------
        cand = a - lr * grad
        # enforce the hard bounds that the evaluator will clip anyway
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _project_onto_simplex(cand, target_sum)

        # ----- occasional random block perturbation --------------------
        if rng.random() < 0.25:                     # 25 % of iterations
            cand = _make_block_perturbation(cand, rng)
            cand = _project_onto_simplex(cand, target_sum)

        # ----- evaluate candidate ---------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        # ----- simulated‑annealing acceptance ---------------------------
        # Temperature decays linearly with iteration count
        t = temp_start + (temp_end - temp_start) * (it / (budget_s * 20))  # arbitrary scaling
        accept = False
        if np.isfinite(score):
            if score < best_score:
                accept = True
            else:
                # allow uphill moves early on
                if rng.random() < np.exp(-(score - best_score) / max(t, 1e-12)):
                    accept = True

        if accept:
            a = cand
            if score < best_score:
                best = cand.copy()
                best_score = score
                print(f"[{it:6d}] new best score = {best_score:.6f}")

        # periodic progress report (without hurting timing)
        if it % 5000 == 0:
            print(f"[{it:6d}] current best = {best_score:.6f}, current max_b = {max_b:.6f}")

    # ------------------------------------------------------------------
    # final reporting
    print(f"[done] iterations={it}, best_score={best_score:.6f}")
    # Convert to plain Python floats for the return value
    return [float(x) for x in best]
