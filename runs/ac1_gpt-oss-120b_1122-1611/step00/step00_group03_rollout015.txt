```
def flatten_step(seq, rng):
    n = seq.size
    b = np.convolve(seq, seq)
    k = int(np.argmax(b))
    # indices i that pair with j = k - i
    # Create bool mask for i in [0, n)
    # We'll reduce a subset of i where corresponding product large.
    # Compute product contributions = seq * shifted_seq (where shifted_seq[i] = seq[k - i] if within bounds else 0)
    shift = np.zeros_like(seq)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            shift[i] = seq[j]
    contributions = seq * shift
    # We'll select a fraction (e.g., top 30%) of indices by contributions to reduce
    num = max(1, int(0.2 * n))
    idxs = np.argpartition(contributions, -num)[-num:]
    # reduction factor
    factor = 0.9 + 0.09 * rng.random()  # random factor between 0.9 and 0.99
    seq_new = seq.copy()
    seq_new[idxs] *= factor
    # Now compute scaling to keep total sum equal to original sum
    orig_sum = seq.sum()
    new_sum = seq_new.sum()
    if new_sum > 1e-12:
        scale = orig_sum / new_sum
        seq_new *= scale
    # clip values
    seq_new = np.clip(seq_new, 0.0, 1000.0)
    return seq_new
```

"""
Simple subgradient flattening for step function minimization.
The algorithm alternates a gradient step that reduces the current
maximum convolution entry with occasional random block
perturbations for escaping local minima. It works entirely with
numpy and stays within the memory limits.
"""

import numpy as np
import time

def subgradient_step(seq, lr):
    n = seq.size
    b = np.convolve(seq, seq)
    k = int(np.argmax(b))
    # gradient g_i = seq[k-i] if 0<=k-i<n else 0
    # This is a reversed slice of seq
    g = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    stop = min(n, k + 1)
    # indices i in [start, stop)
    # For i in this range, j = k - i in range
    i_range = np.arange(start, stop)
    j_range = k - i_range
    g[i_range] = seq[j_range]
    # gradient descent step
    new_seq = seq - lr * g
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    # renormalize to keep total mass (optional, scaling does not affect the score)
    total = new_seq.sum()
    if total > 0:
        new_seq = new_seq * (seq.sum() / total)
    return new_seq

def make_block_perturbation(base, rng):
    """Return one O(n)-memory nonnegative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step sequence with low ``evaluate_sequence`` value.

    The search combines a sub‑gradient descent step that directly targets the
    currently largest entry of the self‑convolution with occasional random
    block perturbations (the strategy used in the baseline). The best sequence
    found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # initialise starting point
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    # clip to permissible range
    parent = np.clip(parent, 0.0, 1000.0)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"initial score = {best_score:.6f}")

    # parameters for sub‑gradient descent
    lr = 0.5          # initial learning rate (in the same units as a_i)
    decay = 0.9995    # multiplicative decay per iteration
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # choose the type of move
        if rng.random() < 0.6:
            # sub‑gradient flattening move
            candidate = subgradient_step(best, lr)
        else:

        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"iteration {iteration}: new best {best_score:.6f}")

        # decay learning rate
        lr *= decay
        # enforce a lower bound on learning rate to keep updates meaningful
        if lr < 1e-4:
            lr = 1e-4

    print(f"finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(x) for x in best]

"""
search for a low‑value step sequence.

The method mixes two ideas:

1. **Sub‑gradient flattening** – the current maximal entry of the
   self‑convolution is identified and a gradient step is taken that
   directly reduces the contribution of the indices that form this
   maximum.  After the step the whole vector is re‑normalised (the
   objective 2·n·max(b)/(sum(a)**2) is scale‑invariant, so the
   renormalisation does not hurt the score).

2. **Exploratory block perturbations** – a cheap O(n) random
   multiplicative block change (the same building block that the
   reference solution uses) is applied from time to time to escape
   local minima.

Both moves are combined in a simple simulated‑annealing loop: a move
is always accepted if it improves the score, otherwise it may be
accepted with a small probability that decays with a temperature
parameter.  The best sequence seen before the time budget expires is
returned.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# small helper: evaluate the objective without the heavy python‑list
# conversion that the original `evaluate_sequence` does.
def _fast_score(seq: np.ndarray) -> float:
    """Return the same value as `evaluate_sequence` for a numpy array."""
    n = seq.size
    # convolution (full) of the non‑negative sequence with itself
    b = np.convolve(seq, seq)
    max_b = float(b.max())
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def _subgradient_step(seq: np.ndarray, lr: float) -> np.ndarray:
    """
    One deterministic step that reduces the current maximal convolution entry.
    `lr` is a learning‑rate‑like factor (in the same magnitude as the
    sequence values).  The step is projected onto the feasible box
    [0, 1000] and the total mass is re‑normalised to keep the scale
    unchanged (the objective is scale‑invariant).
    """
    n = seq.size
    # current full convolution
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))                     # index of the current max

    # gradient of the max‑entry w.r.t. the sequence:
    #   ∂conv_k / ∂a_i = a_{k‑i}   (zero when the partner index is out of range)
    grad = np.zeros_like(seq)
    i_start = max(0, k - (n - 1))
    i_end = min(n, k + 1)                        # exclusive
    i_idx = np.arange(i_start, i_end)
    j_idx = k - i_idx
    grad[i_idx] = seq[j_idx]

    # take a small step opposite to the gradient
    new_seq = seq - lr * grad
    new_seq = np.clip(new_seq, 0.0, 1000.0)

    # re‑normalise the total mass (the objective does not depend on the scale)
    tot = new_seq.sum()
    if tot > 0.0:
        new_seq = new_seq * (seq.sum() / tot)
    return new_seq


# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Same block‑perturbation used in the reference solution."""
    candidate = base.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence` function.  The routine runs until the
    time budget expires (default ~1100 s, a few seconds are kept as a
    safety margin) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise the population
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)

    # Normalise to a convenient total mass; the exact scale does not affect
    # the objective because it is homogeneous of degree 0.
    if parent.sum() == 0.0:
        parent = np.ones_like(parent)
    parent = parent / parent.sum()

    best = parent.copy()
    best_score = _fast_score(best)
    print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # hyper‑parameters
    lr = 0.5               # step size for the sub‑gradient move
    lr_decay = 0.9995
    temperature = 1.0      # annealing temperature
    temp_decay = 0.999
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # ----- decide which move to apply --------------------------------
        if rng.random() < 0.62:               # majority are deterministic gradient steps
            cand = _subgradient_step(best, lr)
        else:
            cand = make_block_perturbation(best, rng)

        # occasional full random restart (helps to explore far away)
        if rng.random() < 0.004:
            cand = rng.random(best.size) * 1000.0

        # ----- evaluate --------------------------------------------------
        cand_score = _fast_score(cand)

        # ----- acceptance (simulated annealing) -------------------------
        accept = False
        if np.isfinite(cand_score):
            if cand_score < best_score:
                accept = True
            else:
                # Accept worse solutions with a probability that decays with temperature
                prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            best = cand
            best_score = cand_score
            if cand_score < best_score:
                print(f"[{it:06d}] new best = {best_score:.6f}")
            else:
                # occasional log of accepted worse moves (useful for debugging)
                if it % 5000 == 0:
                    print(f"[{it:06d}] accepted worse, score = {cand_score:.6f}")

        # ----- decay hyper‑parameters ------------------------------------
        lr = max(lr * lr_decay, 1e-6)
        temperature = max(temperature * temp_decay, 1e-8)

        # lightweight optional early exit if we already reach an extremely good bound
        if best_score < 1.51:          # the target is 1.5025; stop early if we beat it
            break

    print(f"[done] iterations={it}, best_score={best_score:.6f}")
    # Convert back to plain Python floats as required by the runner
    return [float(x) for x in best]
