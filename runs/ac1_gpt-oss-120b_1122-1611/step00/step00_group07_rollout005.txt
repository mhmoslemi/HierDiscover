"""
Propose a candidate step‑function height sequence.

The objective is to minimise
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

For non‑negative a this value is bounded below by 2 (obtained by the uniform
distribution).  The optimiser therefore focuses on finding the uniform
distribution (or something numerically indistinguishable) while exploring a
variety of cheap perturbations so that the search never gets stuck in a poor
local minimum.

The search algorithm combines several cheap moves:
* multiplicative block scaling   (the original “block_perturbation”)
* additive Gaussian noise on a random block
* a small mass‑transfer between two random indices
* a simple “gradient‑like” move that reduces the contribution of the current
  maximal convolution coefficient.

All moves respect the security checks performed by ``evaluate_sequence``:
values are clipped to the interval [0, 1000] and the sum is forced to stay
well above the 0.01 safety threshold.  The best sequence found before the
time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper functions – they are top‑level (no closures) as required
# ----------------------------------------------------------------------


def _clip(seq: np.ndarray) -> np.ndarray:
    """Apply the same clipping that evaluate_sequence does."""
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # convert to plain Python list – evaluate_sequence expects a list
    return float(evaluate_sequence(seq.tolist()))


def _initial_sequence(seed: int = 42) -> np.ndarray:
    """Return a valid starting sequence.

    Preference order:
      1. ``height_sequence_1`` if it exists and is a finite 1‑D array.
      2. a uniform vector of length 256 (the fallback used by the reference code).
    """
    rng = np.random.default_rng(seed)
    # try to use the global variable supplied by the runner
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.ndim == 1 and seq.size > 0 and np.isfinite(seq).all():
            return _clip(seq)
    except Exception:
        pass
    # fallback – a uniform vector (any length works, 256 is a reasonable size)
    return _clip(np.ones(256, dtype=np.float64))


def _block_multiply(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block (the original idea)."""
    cand = base.copy()
    n = cand.size
    # block width between n//200 and n//12 (as in the reference code)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = np.exp(rng.normal(0.0, 0.12))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return _clip(cand)


def _block_add_noise(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Additive Gaussian noise on a random block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 250)
    hi = max(2, n // 10)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # noise scale is a small fraction of the current values
    scale = 0.05 * np.maximum(1.0, np.abs(cand[start:stop]))
    noise = rng.normal(0.0, scale)
    cand[start:stop] += noise
    return _clip(cand)


def _mass_transfer(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a tiny amount of mass from i to j (i ≠ j)."""
    cand = base.copy()
    n = cand.size
    i, j = rng.choice(n, size=2, replace=False)
    # only transfer if the source has a non‑zero amount
    if cand[i] > 1e-9:
        delta = rng.random() * 0.02 * cand[i]          # up to 2 % transfer
        cand[i] -= delta
        cand[j] += delta
    return _clip(cand)


def _gradient_step(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Decrease the contribution of the currently maximal convolution coefficient."""
    cand = base.copy()
    n = cand.size
    # compute the full convolution once (O(n log n) under the hood)
    conv = np.convolve(cand, cand)
    k_max = int(np.argmax(conv))
    # the indices i that actually contribute to conv[k_max] satisfy
    #   0 ≤ i < n  and  0 ≤ k_max - i < n
    # i.e. i ∈ [max(0, k_max-(n-1)), min(n-1, k_max)]
    i_lo = max(0, k_max - (n - 1))
    i_hi = min(n - 1, k_max)
    if i_lo > i_hi:
        return cand  # should never happen
    # pick a random contributing index and move a bit of its mass elsewhere
    i = int(rng.integers(i_lo, i_hi + 1))
    if cand[i] > 1e-9:
        delta = rng.random() * 0.01 * cand[i]          # up to 1 % transfer
        cand[i] -= delta
        # destination is any index *outside* the contributing window –
        # this reduces the product for the active lag while preserving total sum.
        other_candidates = np.r_[np.arange(0, i_lo), np.arange(i_hi + 1, n)]
        if other_candidates.size > 0:
            j = int(rng.choice(other_candidates))
            cand[j] += delta
    return _clip(cand)

def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs) -> list[float]:
    """
    Search for a (near)‑optimal step‑function height sequence.

    The function runs a time‑bounded stochastic local search using several cheap
    perturbations (multiplicative block scaling, additive noise, mass transfer,
    and a simple gradient‑like step).  The best sequence discovered before the
    deadline is returned as a plain list of floats.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total wall‑clock time budget in seconds.  The implementation reserves
        a few seconds for safety and returns the current best before the limit.
    **kwargs
        Ignored – kept for compatibility with the surrounding infrastructure.
    """
    rng = np.random.default_rng(seed)
    # give ourselves a small safety margin (the caller often adds a few seconds)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.5)

    # ------------------------------------------------------------------
    # initialise the incumbent
    # ------------------------------------------------------------------
    parent = _initial_sequence(seed)
    best = parent.copy()
    best_score = _evaluate(best)
    if not np.isfinite(best_score):
        # fallback to a safe uniform vector if the provided one is broken
        best = np.ones(256, dtype=np.float64)
        best_score = _evaluate(best)

    print(f"[init] initial score = {best_score:.6f}, length = {best.size}")

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    # we sample which move to apply with roughly equal probability
    move_fns = [_block_multiply, _block_add_noise, _mass_transfer, _gradient_step]

    while time.monotonic() < deadline:
        iteration += 1
        # pick a move at random
        move = rng.choice(move_fns)
        candidate = move(best, rng)

        # quick discard: sum must be > 0.01, otherwise evaluate_sequence returns inf
        if candidate.sum() < 0.011:
            continue

        score = _evaluate(candidate)
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[{iteration:05d}] new best = {best_score:.6f} (move: {move.__name__})")

        # occasional housekeeping – report progress every ~2 seconds
        if iteration % 200 == 0:
            elapsed = time.monotonic() + (float(budget_s) - deadline) - (float(budget_s) - 2.5)
            print(f"[status] {elapsed:.1f}s elapsed, best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    total_time = time.monotonic() + (float(budget_s) - deadline) - (float(budget_s) - 2.5)
    print(f"[done] finished after {total_time:.1f}s, best score = {best_score:.6f}")
    # convert to plain Python floats as required by the evaluator
    return [float(x) for x in best]
