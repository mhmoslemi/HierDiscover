def find_sidon_set(n, target, max_trials=50):
    rng = np.random.default_rng(seed)
    for _ in range(max_trials):
        order = rng.permutation(n)
        sums = set()
        sel = []
        for i in order:
            conflict=False
            # compute prospective sums:
            if i+i in sums:
                continue
            for s in sel:
                if i+s in sums:
                    conflict=True
                    break
            if not conflict:
                # add i
                sel.append(i)
                sums.add(i+i)
                for s in sel[:-1]:   # previous ones only to avoid double addition of i+i
                    sums.add(i+s)
                if len(sel)>=target:
                    return sel
    # fallback: return whatever we have (maybe smaller)
    return sel

def project_simplex(v):
    # v non-negative not required.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Simple hill‑climbing optimizer for the convolution‑based evaluation problem.

The algorithm works in three stages:

1. **Geometric warm‑start** – a sequence a_i = r**i (r≈exp(-c/n)) gives a good
   initial bound (≈1.8 for the given length).  The optimal ratio r is searched
   quickly on a 1‑D grid because the objective is cheap to evaluate.

2. **Random perturbation** – the geometric sequence is slightly disturbed so that
   the convolution is not perfectly symmetric (otherwise the gradient is zero).

3. **Iterative max‑convolution reduction** – at each step the index `k_max` where
   the full convolution `a * a` attains its maximum is identified.  The
   contribution of each coordinate to that maximum is
        v_i = a_i * a_{k_max-i}
   (zero when the partner index is out of range).  Mass is transferred from a
   coordinate with large `v_i` to a random coordinate, preserving the total sum.
   Only moves that strictly improve the evaluation score are kept (with a tiny
   probability of accepting a worse move to escape plateaus).  The process runs
   until the time budget expires.

The whole routine respects the limits of `evaluate_sequence` (non‑negative,
clipped at 1000) and never exceeds the given time budget.
"""

import numpy as np
import time


def _project_to_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    # Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen for non‑negative inputs
        return np.maximum(v, 0)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score.

    The search uses a geometric warm‑start followed by a simple hill‑climbing
    that repeatedly reduces the peak of the self‑convolution.
    """
    rng = np.random.default_rng(seed)

    # 1. Determine length of the sequence (use the existing global if present)
    # ------------------------------------------------------------------

    except Exception:

    # ------------------------------------------------------------------
    # 2. Geometric warm‑start: a_i = r**i
    # ------------------------------------------------------------------
    # The optimal geometric ratio r satisfies  e^{x}=1+2x  with x≈1.256,
    # where x = -n*log(r).  Hence r = exp(-x/n).  We search a small neighbourhood.
    x_opt = 1.256
    r_center = np.exp(-x_opt / n)

    # Grid around the centre (log‑space) – enough to locate the best on this family.
    grid = np.logspace(np.log10(r_center) - 0.3,
                       np.log10(r_center) + 0.0,
                       num=30)
    best_seq = None
    best_score = float("inf")
    for r in grid:
        seq = np.power(r, np.arange(n, dtype=np.float64))
        # small random jitter to avoid perfect symmetry (helps later steps)
        jitter = rng.normal(0.0, 1e-6, size=n)
        seq = np.clip(seq + jitter, 0.0, None)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
    if best_seq is None:
        # very unlikely, fall back to a uniform start
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 3. Hill‑climbing: repeatedly move mass away from the most active convolution index
    # ------------------------------------------------------------------
    a = best_seq.copy()
    # keep the best ever seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    iteration = 0
    # temperature for occasional uphill moves (simulated‑annealing flavour)
    T0 = 1.0

    while time.monotonic() < deadline:
        iteration += 1
        # compute full convolution
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # partner indices for each position i
        i = np.arange(n, dtype=np.int64)
        j = k_max - i
        mask = (j >= 0) & (j < n)

        # contribution of each coordinate to the current maximum
        contrib = np.zeros_like(a)
        contrib[mask] = a[i[mask]] * a[j[mask]]
        total_contrib = contrib.sum()
        if total_contrib == 0.0:
            # degenerate case – pick a random move
            src = rng.integers(0, n)
        else:
            # pick a source index proportionally to its contribution
            prob = contrib / total_contrib
            src = rng.choice(n, p=prob)

        # ensure we can take a little mass from src
        if a[src] <= 1e-12:
            continue

        # pick a random destination different from src
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1

        # transfer a small fraction of the source mass
        frac = rng.random() * 0.5          # up to 50 % of the source value
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # Re‑project onto the simplex to keep the sum exactly 1 (helps numerical stability)
        a_new = _project_to_simplex(a_new)

        # evaluate the new candidate
        try:
            new_score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            new_score = float("inf")

        # accept if better, otherwise maybe accept with a tiny probability
        accept = False
        if np.isfinite(new_score) and new_score < best_score:
            accept = True
        else:
            # simple annealing acceptance
            T = T0 * max(0.0, 1.0 - iteration / 200000.0)
            if T > 0 and np.isfinite(new_score):
                prob = np.exp((best_score - new_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
                print(f"Iter {iteration:6d}  new best = {best_score:.6f}")

        # Occasionally restart from a fresh random Dirichlet point (helps escape deep plateaus)
        if iteration % 5000 == 0:
            rand = rng.random(n)
            rand = rand / rand.sum()
            cand_score = float(evaluate_sequence(rand.tolist()))
            if cand_score < best_score:
                a = rand
                best = rand.copy()
                best_score = cand_score
                print(f"Iter {iteration:6d}  random restart improved best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score = {best_score:.6f}")
    return [float(x) for x in best.tolist()]
