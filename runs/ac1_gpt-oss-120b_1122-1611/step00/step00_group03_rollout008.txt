""" 
Search for a step function minimizing evaluate_sequence.
The algorithm:
1. Choose the length n of the sequence (from the provided global `height_sequence_1`,
   falling back to 256).
2. Build a family of “U‑shaped’’ sequences a_i = exp(-beta * min(i, n-1-i)).
   This shape makes the anti‑diagonal sums of the outer‑product matrix almost
   constant, which dramatically reduces the maximum convolution value.
3. Scan beta on a dense grid (log‑ and linear‑scale) while respecting the time budget.
   The best sequence from this scan is kept.
4. Starting from the best candidate, run a cheap block‑perturbation local‑search
   (the same primitive that the reference solution uses) for the remaining time.
5. Return the best sequence found.
The whole procedure runs in O(n·#beta) time for the grid search plus a few
hundred thousand O(n) convolution evaluations, comfortably within the 1000‑second
budget.
"""

import numpy as np
import time

def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # width of the block is chosen on a log‑uniform scale between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation of the selected block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – it does not hurt and sometimes helps
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # respect the hard limits imposed by ``evaluate_sequence``
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) improves on the
    current best bound.  The function respects a time budget and may return the
    uniform baseline if the budget is exhausted.
    """
    rng = np.random.default_rng(seed)
    # Give the search a safe margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine the length of the sequence we are allowed to work with.
    # ``height_sequence_1`` is provided by the environment; fall back to a
    # reasonable default if it is missing or malformed.
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = None
    if base_seq is None or base_seq.ndim != 1 or base_seq.size == 0:
        n = 256
    else:
        n = int(base_seq.size)

    # ------------------------------------------------------------------
    # 0.  Initialise the best solution with the uniform vector – its score is
    #     exactly 2.0 and therefore a safe upper bound.
    # ------------------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # 1.  Scan a one‑parameter family a_i(beta) = exp(-beta * min(i,n‑1‑i)).
    #     For each beta we evaluate the score, keep the best.  The scan uses
    #     a mixture of linear and logarithmic steps to cover the region where
    #     the optimum is expected (beta ≈ 1/n … 5·10⁻³).
    # ------------------------------------------------------------------
    # Pre‑compute the distance to the nearest edge for every index.
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)

    # Build a grid of beta values.
    beta_grid = np.concatenate((
        np.linspace(1.0 / n, 5e-3, 200),                     # linear part
        np.logspace(np.log10(1.0 / n), np.log10(2e-2), 200)  # log part
    ))
    # Remove duplicates and enforce a small lower bound (beta cannot be zero).
    beta_grid = np.unique(np.clip(beta_grid, 1e-6, 0.05))

    # Iterate over the grid while we still have time.
    for beta in beta_grid:
        if time.monotonic() >= deadline:
            break
        # Build the U‑shape: a_i = exp(-beta * distance_to_edge)
        cand = np.exp(-beta * dist)
        # ``evaluate_sequence`` already clips values, so we can pass the raw
        # array directly.
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand
            # Log progress – the harness will capture stdout.
            print(f"beta={beta:.6g}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refine the current best solution with cheap stochastic block
    #     perturbations (the same primitive that the reference solution used).
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        candidate = make_block_perturbation(best_seq, rng)
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = candidate
            best_score = sc
            print(f"local improvement: score={sc:.6f}")

    # Return a plain Python list – the outer harness expects ``list[float]``.
    return [float(x) for x in best_seq]

4. Return the best sequence found as a plain Python list of floats.

The whole search fits easily into the 1000 s budget: the grid scan needs
only a few hundred convolution evaluations (each `np.convolve` on a
≈2500‑element vector takes ≈ 1 ms), after which the remaining time is
spent on fast O(n) perturbations.
"""

import numpy as np
import time


def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # Width of the block is chosen between n/200 and n/12.
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Random multiplicative log‑normal perturbation of the selected block.
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # Occasionally symmetrise – harmless and sometimes beneficial.
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # Respect the hard limits used inside ``evaluate_sequence``.
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a non‑negative height sequence that (hopefully) improves on the
    current upper bound.  The function respects a time budget; if the budget
    expires the best sequence found so far is returned.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------ #
    # Determine length n.  ``height_sequence_1`` is supplied by the runner.
    # ------------------------------------------------------------------ #
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base_seq = None
    if base_seq is None or base_seq.ndim != 1 or base_seq.size == 0:
        n = 256
    else:
        n = int(base_seq.size)

    # ------------------------------------------------------------------ #
    # 0.  Start from the uniform vector – its score is exactly 2.0.
    # ------------------------------------------------------------------ #
    best_seq = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------ #
    # 1.  Scan the U‑shaped family a_i(β) = exp(-β * dist_to_edge).
    # ------------------------------------------------------------------ #
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)               # distance to the nearest edge

    # Create a mixed linear / logarithmic grid for β.
    beta_grid = np.concatenate((
        np.linspace(1.0 / n, 5e-3, 200),               # linear part (≈ 1/n … 5·10⁻³)
        np.logspace(np.log10(1.0 / n), np.log10(2e-2), 200)  # log part up to 0.02
    ))
    beta_grid = np.unique(np.clip(beta_grid, 1e-6, 0.05))

    for beta in beta_grid:
        if time.monotonic() >= deadline:
            break
        cand = np.exp(-beta * dist)                   # raw U‑shape (max entry = 1)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand
            print(f"beta={beta:.6g}  score={sc:.6f}")

    # ------------------------------------------------------------------ #
    # 2.  Local refinement using cheap block perturbations.
    # ------------------------------------------------------------------ #
    while time.monotonic() < deadline:
        candidate = make_block_perturbation(best_seq, rng)
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate
            print(f"local improvement: score={sc:.6f}")

    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq]
