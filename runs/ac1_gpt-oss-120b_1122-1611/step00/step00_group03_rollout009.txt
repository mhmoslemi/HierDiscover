"""
Propose a step‑function sequence with a low evaluation score.

The algorithm combines two complementary local search moves:

1. **Block perturbation** – the same cheap O(n) move used in the reference
   implementation.  A random sub‑interval is multiplied by a log‑normal factor
   and optionally symmetrised.  This gives fast exploration of the large
   neighbourhood of the current best vector.

2. **Maximum‑convolution reduction** – the current autocorrelation peak
   `max_b = max(a*a)` is identified.  For one of the indices attaining this
   maximum we locate the pair `(i , j)` that contributes the largest product
   `a[i]*a[j]` to the peak and shrink both entries by a random factor
   (≈0.85‑0.95).  This directly attacks the quantity we want to minimise
   (the evaluation function is proportional to `max_b` after the standard
   normalisation `sum(a)=sqrt(2*n)`).  The move is cheap (O(n) for the pair
   search) and usually lowers the score.

The two moves are chosen probabilistically (70 % reduction, 30 % block
perturbation).  Occasionally a random permutation is applied to avoid
symmetry stagnation.  All candidates are clipped to the admissible range
[0, 1000] before evaluation.

The search runs until the supplied time budget (≈ 1000 s) expires, constantly
keeping the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: block perturbation (identical to the reference code)
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return an O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # width of the perturbed block (between n/200 and n/12)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

# --------------------------------------------------------------
# Helper: reduce the current convolution maximum
def reduce_max_convolution(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify one maximal convolution index and attenuate the pair of
    entries that contributes the largest product to that peak.
    """
    n = seq.size
    conv = np.convolve(seq, seq)          # length 2n‑1
    max_val = conv.max()
    # indices where the peak occurs (normally just one)
    peak_idxs = np.flatnonzero(conv >= max_val * (1.0 - 1e-12))
    if peak_idxs.size == 0:
        return seq
    k = int(rng.choice(peak_idxs))        # choose a maximal index
    # range of i such that both i and k‑i lie in [0, n‑1]
    i_start = max(0, k - (n - 1))
    i_end   = min(n - 1, k)
    i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
    partner = k - i_vals
    # product contributed by each pair (i, k‑i)
    products = seq[i_vals] * seq[partner]
    # pick the pair with the largest contribution
    max_pair = int(np.argmax(products))
    i = int(i_vals[max_pair])
    j = int(partner[max_pair])
    # shrink the two entries by a random factor in [0.85, 0.95]
    factor = 0.85 + 0.10 * rng.random()
    seq[i] *= factor
    if j != i:
        seq[j] *= factor
    return seq
