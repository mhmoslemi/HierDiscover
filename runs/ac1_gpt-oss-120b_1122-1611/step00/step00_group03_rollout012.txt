"""
Iterative minimisation of `evaluate_sequence`.

The evaluator computes (2·n·max(b))/sum(a)^2 where b = a∗a (full convolution).
For a fixed length n the quantity is homogeneous: scaling `a` does not change
the score, therefore the only freedom is the *shape* of the step function.

The algorithm combines two complementary ideas:

1. **Random block perturbations** (the baseline method) to explore the space.
2. **Linear‑programming refinement**:  for a current candidate `f` we solve
   the LP

        maximise   Σ g_j
        subject to   f ∗ g ≤ M          (M = max(f∗f))
                     g ≥ 0

   The constraints are linear in `g` because `f` is fixed.
   The optimal `g` has many zero entries and tends to spread the mass of `f`
   more evenly, thereby lowering the maximal convolution value.
   After solving we renormalise `g` so that Σg = √(2·n) (the normalisation used
   in the reference paper) and mix it back with `f`:
        f ← (1‑t)·f + t·g   (t ≈ 0.5)
   If the mixed vector improves the score we keep it.

The loop runs until the time budget (≈ 1000 s) is exhausted, constantly
printing progress.  If `cvxpy` (or `scipy`) is unavailable the code falls back
to pure random search.

The returned list is the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: cheap block‑wise perturbation (baseline)
def make_block_perturbation(base, rng):
    """Return a cheap O(n)‑memory perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between n/200 and n/12
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

# The main entry point required by the evaluation harness
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine mixes random block perturbations with
    a linear‑programming refinement step.
    """
    # --------------------------------------------------------------
    # Imports that may be optional in the execution environment
    try:
        import cvxpy as cp
        from scipy import sparse
        HAVE_LP = True
    except Exception:            # pragma: no cover – fall back if missing
        HAVE_LP = False

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise a starting sequence ------------------------------------------------
    try:                     # height_sequence_1 is supplied by the harness
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    n = base.size

    # Normalise to the standard L1‑norm √(2·n) – the value does not affect the
    # objective but keeps the numbers in a moderate range.
    target_sum = math.sqrt(2.0 * n)
    s = np.sum(base)
    if s == 0:
        base = np.full_like(base, target_sum / n)
    else:
        base = base * (target_sum / s)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] score={best_score:.6f}   sum={np.sum(best):.3f}")

    # ------------------------------------------------------------------
    # Helper to compute max convolution and the whole convolution vector
    def max_convolution(vec):
        """Return (max value, full convolution) for a non‑negative vector."""
        conv = np.convolve(vec, vec)
        return float(conv.max()), conv

    # ------------------------------------------------------------------
    # Main optimisation loop
    t_mix = 0.5               # mixing ratio for the LP step
    last_improve = time.monotonic()
    lp_cooldown = 0.0         # seconds to wait before next LP attempt

    while time.monotonic() < deadline:
        # ---------------------------------------------------------- random walk
        candidate = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            last_improve = time.monotonic()
            print(f"[rand] improved score={best_score:.6f}")

        # ------------------------------------------------------- LP refinement
        # try an LP step only if enough time passed since last attempt
        if HAVE_LP and (time.monotonic() - last_improve > 5.0) and (time.monotonic() > lp_cooldown):
            # current maximal convolution M
            M, _ = max_convolution(best)

            # Build the (2n‑1) × n constraint matrix A where
            # (A @ g)[k] = Σ_i best[i] * g[k‑i]   (out‑of‑range indices give 0)
            m_len = 2 * n - 1
            rows = []
            cols = []
            data = []
            # each column i contributes best[i] to rows i … i+n‑1
            for i in range(n):
                rows.extend(range(i, i + n))
                cols.extend([i] * n)
                data.extend([best[i]] * n)
            A = sparse.coo_matrix((data, (rows, cols)), shape=(m_len, n)).tocsr()

            # Linear program: maximise Σ g  subject to A·g ≤ M , g ≥ 0
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
            except Exception:                         # pragma: no cover
                prob_status = None
            else:
                prob_status = prob.status

            if prob_status == cp.OPTIMAL and g.value is not None:
                g0 = np.asarray(g.value, dtype=np.float64)
                # renormalise to the prescribed L1‑norm √(2·n)
                sum_g0 = g0.sum()
                if sum_g0 > 1e-12:
                    g_scaled = g0 * (target_sum / sum_g0)

                    # convex combination with the current best
                    mixed = (1.0 - t_mix) * best + t_mix * g_scaled
                    mixed = np.clip(mixed, 0.0, 1000.0)

                    mixed_score = float(evaluate_sequence(mixed.tolist()))
                    if np.isfinite(mixed_score) and mixed_score < best_score:
                        best = mixed
                        best_score = mixed_score
                        last_improve = time.monotonic()
                        print(f"[LP]   improved score={best_score:.6f}")
                        # after a successful LP step we give the algorithm a little
                        # breather before the next LP call
                        lp_cooldown = time.monotonic() + 2.0
                        continue
            # if we reach here the LP step did not help; fall back to random moves
            lp_cooldown = time.monotonic() + 1.0

        # ---------------------------------------------------------- simple restart
        # occasional complete restart if we have been idle for a while
        if time.monotonic() - last_improve > 30.0:
            # create a fresh random vector with the correct L1‑norm
            rnd = rng.random(n)
            rnd = np.clip(rnd, 0.0, 1.0)
            rnd = rnd * (target_sum / rnd.sum())
            best = rnd
            best_score = float(evaluate_sequence(best.tolist()))
            last_improve = time.monotonic()
            print(f"[restart] new random start score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}")
    return [float(x) for x in best]
```
