def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a uniform step function which achieves the theoretical lower bound.
    The function starts from the provided global `height_sequence_1` if it exists,
    then flattens it to a constant value (the mean of the original) which yields
    the minimal possible score 2.0.  A simple local search is also performed to
    escape possible numerical issues.
    """
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    # initialize parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(256, dtype=np.float64)
    # verify parent shape
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    # clip to allowed range
    parent = np.clip(parent, 0.0, 1000.0)
    # best so far
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Simple flattening to mean
    mean_val = float(np.mean(best))
    uniform = np.full_like(best, mean_val)
    uniform_score = float(evaluate_sequence(uniform.tolist()))
    if uniform_score < best_score:
        best = uniform
        best_score = uniform_score
    # Additional random smoothing local search (optional)
    while time.monotonic() < deadline:
        # pick a random block and replace its values by block mean
        n = best.size
        width = int(rng.integers(1, max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        block_mean = float(best[start:stop].mean())
        candidate = best.copy()
