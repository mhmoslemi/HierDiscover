def block_perturb(seq, rng, std=0.12):
    n = seq.size
    candidate = seq.copy()
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = np.exp(rng.normal(0.0, std))
    candidate[start:stop] *= factor
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)
```

- swap_perturb: randomly pick two indices i,j; transfer portion:

```
def transfer_perturb(seq, rng, max_fraction=0.2):
    candidate = seq.copy()
    i, j = rng.choice(seq.size, size=2, replace=False)
    transfer = candidate[i] * rng.uniform(0, max_fraction)
    candidate[i] -= transfer
    candidate[j] += transfer
    return np.clip(candidate, 0.0, 1000.0)

"""
Propose a near‑optimal step‑height sequence.

The evaluation metric is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For a non‑negative sequence the best possible behaviour is achieved when the
self‑convolution is (almost) flat: every entry of conv(a, a) has the same value.
A classical continuous construction with this property is the *arcsine*
distribution:
    f(x) = 1 / (π·sqrt(x·(1‑x))) ,   x∈(0,1)
If X and Y are independent with density f, then X+Y is uniformly distributed on
(0,2).  Discretising f on a regular grid yields a vector `a` whose discrete
convolution is almost constant, so the ratio `max(conv)/sum²` is close to the
theoretical lower bound `1/(2n)` and the score is close to 1.

The algorithm:

1.  Build the arcsine‑type sequence of length n (taken from the global
    `height_sequence_1`).  The raw values are
        a_i = 1 / sqrt((i+0.5)*(n-i-0.5))
    and are then scaled so that the total sum equals √(2 n) – this scaling
    does not change the score but keeps the numbers comfortably away from the
    clipping limits.

2.  Use a small stochastic local‑search loop (≈1000 s) that repeatedly
    perturbs the current best vector either by scaling a random block
    (`block_perturb`) or by transferring a fraction of mass from one index to
    another (`transfer_perturb`).  Whenever a perturbation yields a lower
    score we keep it.

3.  Return the best sequence found (as a plain list of Python floats).

The whole procedure runs without any file or network access, uses only
`numpy`, `time` and the already‑imported `evaluate_sequence`, and respects the
required API.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level (no closures)
# ----------------------------------------------------------------------


def arcsine_initial_sequence(length: int, target_sum: float) -> np.ndarray:
    """Return a discretised arcsine distribution of given length.

    The raw values are 1 / sqrt((i+0.5)*(length-i-0.5)).
    They are scaled so that the total sum equals ``target_sum``.
    """
    i = np.arange(length, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (length - i - 0.5))
    # Scale to a convenient total sum (does not affect the score)
    scale = target_sum / np.sum(a)
    a *= scale
    # Clip to the limits enforced by ``evaluate_sequence``.
    return np.clip(a, 0.0, 1000.0)


def block_perturb(seq: np.ndarray, rng: np.random.Generator,
                  std: float = 0.12) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block of ``seq``."""
    n = seq.size
    # block width between n/200 and n/12 (at least 1)
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    candidate = seq.copy()
    factor = float(np.exp(rng.normal(0.0, std)))
    candidate[start:stop] *= factor

    # occasional symmetrisation – helps flatten the convolution
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])

    return np.clip(candidate, 0.0, 1000.0)


def transfer_perturb(seq: np.ndarray, rng: np.random.Generator,
                     max_fraction: float = 0.20) -> np.ndarray:
    """Move a random fraction of mass from one index to a different index."""
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    candidate = seq.copy()

    # amount to move – never exceeds the source value
    transfer = candidate[i] * rng.uniform(0.0, max_fraction)
    candidate[i] -= transfer
    candidate[j] += transfer

    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a step‑function sequence with a very low evaluation score.

    The construction is based on a discretised arcsine distribution,
    which gives an (almost) flat self‑convolution, i.e. a score close to 1.
    A short stochastic local search is performed for the remaining time
    to squeeze out any further improvement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (use the length of the provided construction)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        # fall back to a moderate size if the global variable is missing/corrupt
        n = 1024

    # ------------------------------------------------------------------
    # 1. Initialise with the arcsine distribution (scaled to sqrt(2n))
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    current = arcsine_initial_sequence(n, target_sum)

    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Local search – keep improving while time permits
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # Choose a perturbation type (70 % block, 30 % transfer)
        if rng.random() < 0.7:
            cand = block_perturb(best, rng)
        else:
            cand = transfer_perturb(best, rng)

        # Occasionally apply a tiny global jitter to avoid stagnation
        if rng.random() < 0.05:
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            cand *= jitter
            cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[{iteration:06d}] new best = {best_score:.6f}")

    print(f"[final] best score after search = {best_score:.6f}")
    # Return a plain Python list of floats as required by the evaluator
    return [float(x) for x in best]
