def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # leave margin for final return

    base = np.array(height_sequence_1, dtype=np.float64)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        n = 1024
        base = np.ones(n, dtype=np.float64)
    # Clip and ensure positive sum
    base = np.clip(base, 0.0, 1000.0)
    if np.sum(base) < 0.01:
        base = np.ones_like(base) * 0.01

    # Keep target sum for rescaling
    target_sum = np.sum(base)

    # Evaluate initial
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Init score {best_score:.6f}')
    # Precompute length
    n = len(base)

    # Gradient descent parameters
    step_scale = 0.1   # base step size factor
    # Keep best.
    while time.monotonic() < deadline:
        # Compute convolution via FFT for speed
        # Determine power of two size
        m = int(2**np.ceil(np.log2(2*n-1)))
        fft_a = np.fft.rfft(best_seq, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[:2*n-1]
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        # Compute gradient
        grad = np.zeros_like(best_seq)
        # For each i
        # contributions for max_idx
        # Use vectorized approach: compute reversed index
        # At indices i where 0 <= max_idx - i < n
        j = max_idx - np.arange(n)  # array of possible partners
        mask = (j >= 0) & (j < n)
        # For i where 2*i == max_idx
        diag_mask = (2 * np.arange(n) == max_idx)
        grad[diag_mask] = 2 * best_seq[diag_mask]
        # Off-diagonal contributions
        off = mask & (~diag_mask)
        grad[off] = best_seq[j[off]]
        # Gradient of full objective
        S = np.sum(best_seq)
        # scale factor
        coeff = 2 * n / (S**2)
        grad_obj = coeff * grad - (4 * n * max_val) / (S**3) * np.ones_like(best_seq)
        # Step size adapt. Use step_scale * S maybe.
        # Choose alpha decreasing with iteration count maybe random.
        # We'll just use relative step: max change of each element ~ step_scale * best_seq.mean()
        step = step_scale * S / n  # typical magnitude per element
        # Random sign? We move opposite to gradient.
        new_seq = best_seq - step * grad_obj
        # Add small random jitter to avoid stick.
        jitter = rng.normal(scale=0.01 * step, size=n)
        new_seq += jitter
        # Clip and enforce nonneg, apply scaling to retain target sum
        new_seq = np.clip(new_seq, 0.0, 1000.0)
        # If sum zero, rescale small.
        new_sum = np.sum(new_seq)
        if new_sum < 1e-6:
            # reinitialize random
            new_seq = np.full_like(best_seq, target_sum / n)
            new_sum = target_sum
        # Rescale to target_sum (maintain scale invariance)
        new_seq = new_seq * (target_sum / new_sum)
        # Evaluate
        score = evaluate_sequence(new_seq.tolist())
        if np.isfinite(score) and score < best_score:
            best_seq = new_seq
            best_score = score
            print(f'[{time.monotonic():.2f}] New best {best_score:.6f}')
            # adapt step size downwards for fine search
            step_scale *= 0.9
        else:
            # maybe increase step scale little for exploration
            step_scale *= 1.02

    return best_seq.tolist()

def gradient_step(seq, rng, step_scale, target_sum):
    n = len(seq)
    # compute convolution using FFT
    m = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fft_a = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)[:2 * n - 1]
    max_idx = np.argmax(conv)
    max_val = conv[max_idx]
    grad = gradient_of_max(seq, max_idx)
    S = np.sum(seq)
    coeff = 2 * n / (S * S)
    grad_obj = coeff * grad - (4 * n * max_val) / (S ** 3) * np.ones_like(seq)
    # step size: relative to average amplitude
    step = step_scale * S / n
    new_seq = seq - step * grad_obj
    # add small random noise to avoid stagnation
    new_seq += rng.normal(scale=0.001 * step, size=n)
    # clip
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    total = np.sum(new_seq)
    if total > 0:
        new_seq = new_seq * (target_sum / total)
    else:
        # fallback: uniform
        new_seq = np.full_like(seq, target_sum / n)
    return new_seq

The algorithm mixes a cheap random block perturbation with a
gradient‑descent step on the L∞‑norm of the autocorrelation.
A simple simulated‑annealing rule occasionally accepts uphill
moves to avoid local minima.  The best sequence found before
the time budget expires is returned.

import numpy as np

    n = seq.size
    cand = seq.copy()
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

def _grad_of_max(seq, max_idx):
    """Gradient of the convolution value at position max_idx."""
    n = seq.size
    i = np.arange(n)
    j = max_idx - i
    mask = (j >= 0) & (j < n)
    diag = mask & (i == j)
    off = mask & (~diag)
    grad = np.zeros_like(seq)
    grad[diag] = 2.0 * seq[diag]
    grad[off] = seq[j[off]]
    return grad

def _gradient_step(seq, rng, step_scale, target_sum, fft_len):
    """One projected gradient step aimed at reducing max(conv(seq,seq))."""
    n = seq.size
    # convolution via FFT
    fft_a = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fft_a * fft_a, n=fft_len)[:2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])

    # gradient of the active convolution entry
    grad = _grad_of_max(seq, max_idx)

    S = float(np.sum(seq))
    # coefficient of the numerator term
    coeff = 2.0 * n / (S * S)

    # sub‑gradient of the full objective
    grad_obj = coeff * grad - (4.0 * n * max_val) / (S ** 3) * np.ones_like(seq)

    # step magnitude – roughly a small fraction of the average entry
    step = step_scale * (S / n)
    new_seq = seq - step * grad_obj

    # tiny random jitter to keep the dynamics from freezing
    new_seq += rng.normal(scale=0.001 * step, size=n)

    # clip into the admissible box
    new_seq = np.clip(new_seq, 0.0, 1000.0)

    # rescale to keep the total mass constant (scale invariance of the objective)
    total = float(np.sum(new_seq))
    if total > 0.0:
        new_seq *= (target_sum / total)
    else:
        new_seq = np.full_like(seq, target_sum / n)

    return new_seq

    sum_a = float(np.sum(seq))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
