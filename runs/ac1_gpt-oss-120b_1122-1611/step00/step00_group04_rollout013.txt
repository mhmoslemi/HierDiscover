"""
Propose a candidate sequence that improves the evaluation score.

The algorithm works in two stages:

1. **Deterministic initialization** – we start from the analytically
   motivated shape a_i = 1 / sqrt(i+1).  This sequence makes the
   self‑convolution almost flat (its continuous analogue gives a
   constant π) and already beats the trivial uniform construction
   (score ≈ 2).

2. **Stochastic local search** – a simple simulated‑annealing style
   optimisation moves a small amount of mass from one index to another
   while keeping the total sum unchanged.  After each move we evaluate
   the exact score with ``evaluate_sequence``; moves that improve the
   score are always accepted and worsening moves are accepted with a
   temperature‑dependent probability.  The temperature decays linearly
   during the allotted time budget.

The search runs until the time budget (default 990 s, leaving a small
margin for the runner) expires and returns the best sequence found.
All numbers are kept in the range [0, 1000] as required by the
evaluator.
"""

import time
import numpy as np
import math

def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Return a candidate obtained by scaling a random contiguous block
    of ``base`` by a log‑normal factor and optionally symmetrising it.
    The operation is performed in O(n) time and respects the
    [0,1000] clipping required by ``evaluate_sequence``.
    """
    cand = base.copy()
    n = cand.size
    # width of the block: between n/200 and n/12 elements
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation to keep the shape balanced
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights (floats) that minimises the
    provided ``evaluate_sequence`` function.

    The search starts from the analytically motivated 1/√i shape and
    then performs a timed stochastic local search using block
    perturbations (as in the reference code) together with a simple
    simulated‑annealing acceptance rule.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length of the sequence we are allowed to output.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))          # use the global if present
    except Exception:
        n = 256                                    # fallback length

    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 2. Deterministic starting point: a_i = 1 / sqrt(i+1)
    # ------------------------------------------------------------------
    base = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    base = np.clip(base, 0.0, 1000.0)

    # Ensure the initial candidate is valid for the evaluator.
    if not np.isfinite(base).all() or base.size == 0:
        base = np.ones(n, dtype=np.float64)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = base.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 3. Simulated annealing / local search loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # leave a tiny margin

    # annealing schedule parameters
    init_temp = 0.1 * best_score          # start temperature
    final_temp = 1e-6
    max_iters = 200_000                   # hard cap – will stop earlier by time

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # linear cooling
        temperature = init_temp + (final_temp - init_temp) * (it / max_iters)

        # ------------------------------------------------------------------
        # 3a. Propose a new candidate.
        # ------------------------------------------------------------------
        # Two complementary move generators:
        #   • a block‑wise multiplicative perturbation (large‑scale change)
        #   • a small pairwise mass transfer (fine‑grained adjustment)
        if rng.random() < 0.7:   # most steps: block perturbation
            cand = _make_block_perturbation(cur, rng)
        else:                   # occasional pairwise transfer
            cand = cur.copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n - 1))
            if j >= i:
                j += 1
            if cand[i] > 0.0:
                # move at most 8 % of the source entry
                frac = rng.random() * 0.08
                delta = cand[i] * frac
                cand[i] -= delta
                cand[j] += delta
                cand = np.clip(cand, 0.0, 1000.0)

        # ------------------------------------------------------------------
        # 3b. Evaluate the proposal.
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if not math.isfinite(score):
            continue

        # ------------------------------------------------------------------
        # 3c. Acceptance rule.
        # ------------------------------------------------------------------
        if score < cur_score:
            accept = True
        else:
            # Metropolis probability
            delta = score - cur_score
            accept = rng.random() < math.exp(-delta / max(temperature, 1e-12))

        if accept:
            cur = cand
            cur_score = score
            if score < best_score:
                best = cand.copy()
                best_score = score
                print(f"Iter {it:6d} – new best score: {best_score:.8f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"Search finished after {it} iterations. Best score = {best_score:.8f}")
    return [float(x) for x in best]
