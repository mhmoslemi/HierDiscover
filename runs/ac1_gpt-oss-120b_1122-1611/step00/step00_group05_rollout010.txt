    deadline = time.monotonic() + max(0., budget_s - 2.)

    if a.sum() < 0.01:
        a += 0.01
    # Normalise sum to something - not necessary but keep sum stable
    # We may keep original sum as target_sum
    target_sum = a.sum()
    # Evaluate initial
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"Initial score: {best_score:.6f}")
    # Parameters
    eta = 0.2  # gradient step size
    decay = 0.9995  # gradually reduce eta
    # main loop
    while time.monotonic() < deadline:
        # compute convolution
        b = np.convolve(a, a)
        k = int(np.argmax(b))
        # region indices that contribute to this k
        start = max(0, k - (n - 1))

        g = np.zeros_like(a)

"""
search.py – Simple hill‑climbing search for a step‑function that lowers
evaluate_sequence(sequence).

The algorithm starts from a random (non‑uniform) sequence and repeatedly
focuses on the shift where the autocorrelation a*a attains its maximum.
All entries that contribute to this peak are scaled down; the whole
vector is then renormalised to keep its ℓ1‑norm constant (the evaluation
function is scale‑invariant).  Occasionally a random block perturbation
or a tiny Gaussian jitter is applied to escape local minima.

The method runs until the allocated time budget is exhausted and returns
the best sequence found.
"""

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights (floats) that tries to minimise
    evaluate_sequence.  The function respects the time budget (seconds)
    passed via ``budget_s`` and prints occasional progress messages.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Approximate wall‑clock seconds available for the search.
    **kwargs
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the allotted time.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Choose a problem size.
    # --------------------------------------------------------------
    # A moderate length balances exploration (larger n gives a finer
    # objective) and speed (convolution is O(n^2) with np.convolve).
    n = 3000                     # can be tuned; 2‑3k works well in ~0.02 s per step

    # --------------------------------------------------------------
    # 2) Initialise a non‑uniform starting vector.
    # --------------------------------------------------------------
    # Try to use the global ``height_sequence_1`` if it exists; otherwise
    # fall back to a random Dirichlet‑like start (positive entries, random
    # distribution, constant ℓ1‑norm).
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        # keep only the first ``n`` entries (or pad if too short)
        if base.size < n:
            extra = rng.random(n - base.size)
            base = np.concatenate([base, extra])
        else:
            base = base[:n]
        a = np.clip(base, 0.0, 1000.0)
        # inject a tiny disturbance so we are not already at a local optimum
        a += rng.normal(0.0, 0.01, size=n)
    except Exception:
        # uniform random numbers in (0,1) and then normalise
        a = rng.random(n)
    # make sure the sum is safely away from 0 (evaluate_sequence returns inf for sum<0.01)
    if a.sum() < 0.01:
        a += 0.01
    target_sum = a.sum()
    a = np.clip(a, 0.0, 1000.0)

    # --------------------------------------------------------------
    # 3) Evaluate the initial candidate and initialise bookkeeping.
    # --------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Main optimisation loop.
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    iteration = 0
    # hyper‑parameters (can be tweaked on the fly)
    factor_range = (0.5, 0.9)          # how much we shrink the offending region
    block_prob   = 0.03               # probability of a generic block perturbation
    jitter_prob  = 0.02               # probability of a tiny Gaussian jitter
    reset_prob   = 0.001              # probability of a full random restart

    while time.monotonic() < deadline:
        # ------------------------------------------------------
        # a) compute the full convolution and locate the worst shift
        # ------------------------------------------------------
        b = np.convolve(a, a)               # length 2*n-1
        k_max = int(b.argmax())             # shift with maximal autocorrelation

        # ------------------------------------------------------

        i_start = max(0, k_max - (n - 1))

        # ------------------------------------------------------
        if region_len > 1 and rng.random() > block_prob:

            blk_start = i_start + offset
            blk_stop  = blk_start + width   # exclusive

        # ------------------------------------------------------

        iteration += 1

    # --------------------------------------------------------------
    # 5) Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]
