def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np, cvxpy as cp
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine initial length n
    if 'height_sequence_1' in globals():
        base = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
        else:
            n = base.size
    else:
        n = 256
    # initialize uniform normalized to sqrt(2n)
    total_norm = np.sqrt(2.0 * n)
    f = np.full(n, total_norm / n, dtype=np.float64)

    # Prepare CVXPY problem
    f_param = cp.Parameter(n, value=f, nonneg=True)
    M_param = cp.Parameter(nonneg=True)
    b = cp.Variable(n, nonneg=True)
    conv_expr = cp.conv(f_param, b)      # length 2n-1
    constraints = [conv_expr <= M_param]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # best tracking
    best_seq = f.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial score: {best_score}")

    # iteration parameters
    t = 0.3
    iteration = 0

    while time.monotonic() < deadline:
        # compute current M
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())
        # update cvx parameters
        f_param.value = f
        M_param.value = M

        # solve LP
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
        if b.value is None:
            print("LP solver failed, breaking.")
            break
        b_opt = np.asarray(b.value, dtype=np.float64)
        # ensure non-negative numeric
        b_opt = np.maximum(b_opt, 0.0)

        sum_b = b_opt.sum()
        if sum_b <= 0:
            print("LP produced zero sum, breaking.")
            break

        # scale to have same L1 norm as f (sqrt(2n))
        scale = total_norm / sum_b
        g = np.clip(b_opt * scale, 0.0, 1000.0)

        # mixture
        f_new = (1.0 - t) * f + t * g

        # optional: clip to allowed range
        f_new = np.clip(f_new, 0.0, 1000.0)

        # evaluate
        score = float(evaluate_sequence(f_new.tolist()))
        print(f"Iter {iteration}: M={M:.6f}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            print(f"  New best score: {best_score:.6f}")

        # Check if improvement
        if score < best_score - 1e-9:
            f = f_new
        else:
            # if not improving, maybe reduce step size or break
            t *= 0.5
            if t < 1e-3:
                print("Step size too small, terminating.")
                break
        iteration += 1

    return [float(x) for x in best_seq]

"""
Propose a candidate step‑function sequence.

The algorithm starts from the uniform distribution (which gives the
known score 2.0) and performs a targeted hill‑climbing search that
tries to lower the maximal entry of the self‑convolution.
At each iteration we:

1. Compute the full convolution a*a and locate its current maximum
   index k_max.
2. Identify the pair (i, j = k_max‑i) that contributes most to this
   peak.
3. Move a small amount of mass ε from that high‑contributing pair
   to a randomly chosen pair (p, q) whose convolution value is
   currently low.
4. Accept the move only if it improves the evaluation score.
5. Occasionally restart from the uniform distribution if no improvement
   is seen for many steps.

The search runs until the time budget expires (≈ 1000 s) and returns the
best sequence found.  All values are kept non‑negative and clipped to the
allowed range [0, 1000].
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    # --------------------------------------------------------------
    # Setup
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Length of the sequence – use the provided construction if possible
    try:
        base_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                           0.0, 1000.0)
        if base_seq.ndim == 1 and base_seq.size > 0 and np.isfinite(base_seq).all():
            n = int(base_seq.size)
        else:
            raise Exception
    except Exception:
        n = 256                       # fallback size

    # Normalise to the same L1‑norm that the original code uses
    total_norm = np.sqrt(2.0 * n)     # sum of the sequence = sqrt(2 n)
    a = np.full(n, total_norm / n, dtype=np.float64)   # uniform start

    # Keep the best sequence seen so far
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:                # safety net
        best_score = float('inf')
    best_seq = a.copy()

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main search loop
    # --------------------------------------------------------------
    iteration = 0
    no_improve = 0                     # counter for stagnation
    while time.monotonic() < deadline:
        # 1. convolution and the index of its current maximum
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # 2. find the pair (i, j) that contributes most to this peak
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)           # inclusive
        i_candidates = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_candidates = k_max - i_candidates
        contrib = a[i_candidates] * a[j_candidates]
        if contrib.size == 0:
            # should never happen, but guard against it
            break
        idx = int(i_candidates[np.argmax(contrib)])
        jdx = int(k_max - idx)

        # 3. amount of mass to move – a small fraction of the smaller entry
        eps = min(a[idx], a[jdx]) * rng.uniform(0.01, 0.05)
        if eps <= 0.0:
            # if one of them is already zero fall back to a tiny step
            eps = min(a[idx], a[jdx]) * 0.01

        # 4. pick a target shift that currently has a low convolution value
        low_shifts = np.where(conv < max_val * 0.5)[0]
        if low_shifts.size > 0:
            target_shift = int(rng.choice(low_shifts))
        else:
            target_shift = int(rng.integers(0, 2 * n - 1))

        # pick two indices that realise the target shift (if possible)
        p = int(rng.integers(0, n))
        q = target_shift - p
        if q < 0 or q >= n:
            q = int(rng.integers(0, n))

        # 5. build the candidate by moving ε from (idx, jdx) to (p, q)
        cand = a.copy()
        cand[idx] -= eps
        cand[jdx] -= eps
        cand[p]   += eps
        cand[q]   += eps
        np.clip(cand, 0.0, 1000.0, out=cand)   # enforce problem limits

        # 6. evaluate
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            # any numerical trouble – just skip this move
            continue

        # 7. accept if we made progress
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score
            a = cand                     # continue searching from the improved point
            no_improve = 0
            print(f"[iter {iteration:5d}] improved score = {best_score:.6f}")
        else:
            # no improvement – keep the current a
            no_improve += 1

        # occasional random restart to escape possible plateaus
        if no_improve > 2500:
            a = np.full(n, total_norm / n, dtype=np.float64)
            no_improve = 0
            print(f"[iter {iteration:5d}] restart (no improvement)")

        iteration += 1

    print(f"[final] best score = {best_score:.6f}")
    return [float(x) for x in best_seq]
