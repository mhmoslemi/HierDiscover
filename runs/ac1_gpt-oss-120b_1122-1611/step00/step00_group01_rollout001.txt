    best_score = float(evaluate_sequence(best.tolist()))

    while time.monotonic() < deadline:
        # Random block perturbation (log‑normal multiplier).
        width = int(rng.integers(1, max(2, length // 10)))
        start = int(rng.integers(0, max(1, length - width + 1)))
        stop = start + width
        factor = float(np.exp(rng.normal(0.0, 0.05)))
        trial = best.copy()
        trial[start:stop] *= factor
        trial = np.clip(trial, 0.0, 1000.0)

        score = float(evaluate_sequence(trial.tolist()))
        if np.isfinite(score) and score < best_score:
            best = trial
            best_score = score
            print(f"Found better score: {best_score:.12f}")

    # ------------------------------------------------------------
    # 4. Return the best candidate (the uniform vector).
    # ------------------------------------------------------------
    return [float(x) for x in best]
