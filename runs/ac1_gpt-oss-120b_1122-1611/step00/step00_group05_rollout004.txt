    """Return sub‑gradient of the max‑convolution term for a non‑negative vector a."""
    conv = np.convolve(a, a)
    max_val = conv.max()

        j_end = min(k, n - 1)
        idx = np.arange(j_start, j_end + 1, dtype=np.int64)

            _maybe_update(_norm(beta), f"beta‑a{a:.1f}‑b{b:.1f}", best, start)

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the max‑convolution
    a = best["seq"].copy()
    a = _norm(a)  # ensure unit sum for the gradient loop

    eta = 0.5          # initial step size
    decay = 0.99       # geometric decay per iteration
    min_eta = 1e-4
    it = 0

    while time.monotonic() < deadline:
        grad, _ = _subgradient_max_conv(a)
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)
        a_new = _norm(a_new)

        _maybe_update(a_new, f"grad‑step‑{it}", best, start)

        a = a_new
        eta = max(min_eta, eta * decay)
        it += 1

        # occasional random block perturbation to escape plateaus
        if it % 20 == 0 and rng.random() < 0.3:
            cand = _block_perturbation(a, rng)
            cand = _norm(cand)
            _maybe_update(cand, f"rand‑block‑{it}", best, start)
            a = cand

    # ------------------------------------------------------------------
    # Phase 3 – final cheap hill climbing (block perturbations around best)
    while time.monotonic() < deadline:
        cand = _block_perturbation(best["seq"], rng)
        _maybe_update(cand, "final‑hill", best, start)

    # Return as plain Python list
    return [float(x) for x in best["seq"]]
```
