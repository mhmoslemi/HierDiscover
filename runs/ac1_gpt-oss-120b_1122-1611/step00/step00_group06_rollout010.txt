def project_simplex(v):
    # v is array-like
    # Implementation: ensure sum(v) = 1 and v >=0
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform?
        tau = 0
    else:
        rho = ind[cond][-1]
        tau = cssv[cond][-1] / rho
    w = np.maximum(v - tau, 0)
    return w

def project_simplex(v):
    # ensure v is 1-d array
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # Find rho
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non-positive; return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]  # zero-indexed
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a better step‑function sequence.

The algorithm starts from a random point on the probability simplex
(and optionally from the provided `height_sequence_1`).  It then
minimises a smooth approximation of the objective

    2·n·max(conv(a,a)) / (sum(a)^2)

by projected gradient descent.  The smooth max is built with a
log‑sum‑exp (parameter `temp`), whose temperature is increased during
the run so that the optimisation first sees a softened objective and
later focuses on the true maximum.  After each gradient step the
vector is projected back onto the simplex (non‑negative, sums to 1).
The best sequence seen so far (according to the official
`evaluate_sequence` function) is kept and finally returned.

The whole search respects the time limit given by `budget_s` and
periodically prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x >= 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]

    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0          # cumulative sum minus 1
    ind = np.arange(1, n + 1)          # 1‑based index
    # find the last position where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive -> return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1            # zero‑based index of rho
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function (non‑negative list of heights) that
    minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds (real execution time is
        truncated a few seconds before the deadline to avoid timeout).

    Returns
    -------
    list[float]
        The best sequence found (converted to a plain Python list).
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin so the host can still terminate cleanly
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Choose a working length.
    # Use the length of the provided construction if it exists,
    # otherwise fall back to a moderate size (2‑k) for speed.
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size >= 2:
            # limit size to keep each iteration fast (≈2 k)
            n = min(base.size, 2000)
            # down‑sample the existing sequence if it is longer
            if base.size > n:
                step = max(1, base.size // n)
                a0 = base[::step][:n]
                if a0.size < n:               # pad if needed
                    a0 = np.pad(a0, (0, n - a0.size), mode='wrap')
            else:
                a0 = base.copy()
        else:
            n = 2000
            a0 = None
    else:
        n = 2000
        a0 = None

    # ------------------------------------------------------------------
    # Initialise a point on the simplex.
    if a0 is not None:
        a = np.clip(a0, 1e-12, None)       # avoid exact zeros for stability
        if a.sum() == 0:
            a = rng.dirichlet(np.ones(n))
        else:
            a = a / a.sum()
    else:
        a = rng.dirichlet(np.ones(n))

    # Keep the best sequence seen so far.
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    # Parameters for the optimisation
    it = 0
    temp = 0.5                # temperature for the smooth‑max
    lr = 0.6                  # base step‑size (will be normalised)
    last_improve = 0          # iteration of the last improvement
    # Main optimisation loop
    while time.monotonic() < deadline:
        # ----- convolution and smooth‑max weights -----
        b = np.convolve(a, a)                     # full convolution
        max_b = b.max()
        # slowly increase the temperature so that we start smooth and end sharp
        if it % 50 == 0 and temp < 30.0:
            temp *= 1.07

        # log‑sum‑exp trick for numerical stability
        b_shift = b - max_b                     # max becomes zero
        w = np.exp(temp * b_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                            # normalised weights

        # ----- gradient of the smooth objective -----
        # gradient = 2 * (w * a)_{i+j}  = 2 * sum_j w_{i+j} * a_j
        conv = np.convolve(a[::-1], w, mode='full')
        # slice the part that corresponds to i = 0 … n‑1
        grad = 2.0 * conv[n - 1:n - 1 + n]

        # ----- take a step and project back onto the simplex -----
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = (lr / norm) * grad          # normalise step length
        else:
            step = np.zeros_like(a)
        a_candidate = a - step

        # occasional random jitter to escape shallow plateaus
        if rng.random() < 0.03:
            jitter = rng.normal(scale=0.02, size=n)
            a_candidate = a_candidate + jitter

        a = _project_onto_simplex(a_candidate)

        # ----- evaluate periodically -----
        if it % 10 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if score < best_score:
                best_score = score
                best_a = a.copy()
                last_improve = it
                print(f'Iter {it:5d} – score {best_score:.6f} – max_b {max_b:.8f}')
        # restart if stuck for a long time
        if it - last_improve > 2500:
            a = rng.dirichlet(np.ones(n))
            last_improve = it

        it += 1

    # Final log entry
    print(f'Finished after {it} iterations – best score {best_score:.6f}')
    # Convert to plain Python list of floats, as required by the evaluator.
    return [float(x) for x in best_a]
