"""Propose a candidate step‑function sequence.

The evaluation function is  

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

with all entries forced to be in [0, 1000] and non‑negative.
Mathematically, for any non‑negative vector `a`

    max(conv(a, a)) ≥ Σ a_i² ≥ (Σ a_i)² / n

so the score is always at least 2.  The uniform (constant) sequence
reaches this bound exactly, giving the optimal possible score
(independent of the length of the vector).

The routine therefore builds a uniform sequence of the same length as
the provided `height_sequence_1` (or a default of 256) and returns it.
A very cheap random local‑search loop is kept as a fallback – it may
occasionally find a numerically slightly better score because of
floating‑point rounding, but the uniform construction guarantees the
theoretical optimum of 2.0.
"""
import time
import numpy as np

def uniform_sequence(length: int, value: float = 1.0) -> np.ndarray:
    """Return a constant non‑negative sequence of the given length."""
    return np.full(length, float(value), dtype=np.float64)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a simple O(n) perturbation of ``base``.
    The perturbation keeps the sequence non‑negative and respects the
    clipping limits (0, 1000) used by ``evaluate_sequence``.
    """
    candidate = base.copy()
    n = candidate.size
    # Choose a block width; keep it at least 1 and at most n.
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    # Apply a log‑normal multiplicative factor to the whole block.
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor

    # Occasionally symmetrize the candidate to keep it “smooth”.
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])

    # Enforce the hard bounds that ``evaluate_sequence`` also enforces.
    candidate = np.clip(candidate, 0.0, 1000.0)
    return candidate

    Return a sequence of non‑negative heights that (theoretically) attains
    the minimal possible score of 2.0.  The function spends a small part
    of the time budget performing random perturbations starting from the
    uniform sequence – this may occasionally discover a marginally lower
    score due to floating‑point rounding, but the uniform sequence is a
    guaranteed optimum.

    Parameters
    ----------
    seed : int
        Seed for the NumPy random generator.
    budget_s : float
        Approximate time budget (seconds) for the whole search.
    **kwargs : dict
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    rng = np.random.default_rng(seed)

    # Determine the desired length.  If the global variable ``height_sequence_1``
    # exists and looks like a 1‑D sequence, use its length; otherwise fall back
    # to a modest default of 256.
    try:
        base_len = int(len(height_sequence_1))
        if base_len <= 0:
            raise Exception
    except Exception:
        base_len = 256

    # Start from the mathematically optimal uniform sequence.
    best = uniform_sequence(base_len, value=1.0)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={base_len}, initial uniform score={best_score:.12f}")

    # Allocate a small portion of the budget for random local search.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Even though we know the uniform sequence is optimal, perform a cheap
    # random walk – it occasionally yields a tiny numerical improvement.
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        candidate = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[improve] iter={iteration}, new best score={best_score:.12f}")

    print(f"[final] returning sequence of length {best.size} with score {best_score:.12f}")
    # Return a plain Python list of floats as required.
    return [float(x) for x in best]
