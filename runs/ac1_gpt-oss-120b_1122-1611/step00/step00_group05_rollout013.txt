def project_simplex(v, s=1.0):
    # v is numpy array
    # ensure sum(v) = s, nonnegative
    # Implementation based on "Efficient Projections onto the l1-Ball for Learning in High Dimensions" or simple.

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function by starting from a simple geometric (exponential)
profile and then applying a fast greedy “mass‑transfer” local search.
The search repeatedly moves a small amount of weight from the index that
contributes most to the current maximal convolution value to an index that
contributes the least, always keeping the total sum unchanged.
Occasional random block perturbations (the strategy used in the reference
implementation) are applied to escape shallow local minima.
The whole routine stops after the allotted time budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast score computation (exactly the same as evaluate_sequence)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Return the evaluation score of a non‑negative sequence."""
    # Apply the same safety clipping as the official evaluator.
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf
    # convolution (order‑N * order‑N → order‑2N‑1)
    conv = np.convolve(seq, seq)
    m = conv.max()
    return float(2.0 * len(seq) * m / (s * s))


# ----------------------------------------------------------------------
# Greedy mass‑transfer step
# ----------------------------------------------------------------------
def _transfer_step(a: np.ndarray, k_max: int, lr: float, rng: np.random.Generator) -> np.ndarray:
    """
    Perform a single greedy transfer:
      * find the index i with the largest contribution to the peak at k_max,
      * find the index j with the smallest partner weight,
      * move a fraction lr of a[i] to a[j].

    The sum of the vector stays unchanged.
    """
    n = a.shape[0]
    idx = np.arange(n)
    partner = k_max - idx                     # partner index for each i
    mask = (partner >= 0) & (partner < n)     # valid positions
    w = np.zeros_like(a)
    w[mask] = a[partner[mask]]                # partner values (a_{k_max-i})

    # contribution of each i to the peak value
    contrib = a * w

    # source = index with largest contribution (most harmful)
    src = int(np.argmax(contrib))

    # destination = index whose partner is smallest (least harmful)
    # break ties randomly to avoid deterministic dead‑ends
    min_w = w.min()
    cand_dst = np.where(w == min_w)[0]
    if cand_dst.size > 1:
        dst = int(rng.choice(cand_dst))
    else:
        dst = int(cand_dst[0])

    if src == dst:
        # pick a different destination at random
        dst = int(rng.integers(0, n))
        while dst == src:
            dst = int(rng.integers(0, n))

    eps = a[src] * lr
    if eps <= 0.0:
        return a   # nothing to move

    b = a.copy()
    b[src] -= eps
    b[dst] += eps
    # numerical safety – never let a value become negative
    if b[src] < 0.0:
        b[dst] += b[src]   # b[src] is negative here
        b[src] = 0.0
    return b
