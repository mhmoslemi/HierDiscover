    np.clip(seq, 0.0, 1000.0, out=seq)

def move_block_scale(seq, rng):
    n = seq.size
    candidate = seq.copy()
    # random block width between n//200 and n//12
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    width = int(rng.integers(min_width, max_width+1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= scale
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def move_reallocate(seq, rng):
    n = seq.size
    candidate = seq.copy()
    if np.all(candidate == 0):
        return candidate
    # source weighted by value
    total = np.sum(candidate)
    if total == 0:
        src = rng.integers(0, n)
    else:
        probs = candidate / total
        src = rng.choice(n, p=probs)
    dst = rng.integers(0, n)
    while dst == src:
        dst = rng.integers(0, n)
    # amount to move: up to 30% of source
    max_move = candidate[src] * 0.5
    move = float(rng.random()) * max_move
    candidate[src] -= move
    candidate[dst] += move
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def move_reduce_peak(seq, rng):
    n = seq.size
    candidate = seq.copy()
    # Compute convolution
    b = np.convolve(candidate, candidate)
    k = int(np.argmax(b))
    # determine range of i where both i and j=k-i are valid
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    # pick a random i
    i = rng.integers(i_min, i_max + 1)
    j = k - i
    # reduction factor less than 1
    factor = float(np.exp(rng.normal(0.0, 0.07)))  # typical <1
    factor = min(factor, 1.0)  # ensure not increase
    # apply reduction to both indices
    delta_i = candidate[i] * (1 - factor)
    candidate[i] *= factor
    if j != i:
        delta_j = candidate[j] * (1 - factor)
        candidate[j] *= factor
    else:
        delta_j = 0.0
    removed = delta_i + delta_j
    # redistribute removed mass to two random positions
    for _ in range(2):
        idx = rng.integers(0, n)
        add = removed * rng.random()
        candidate[idx] += add
        removed -= add
    if removed > 0:
        candidate[0] += removed
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a step-height sequence that minimizes evaluate_sequence.

    Uses a simulated‑annealing style hill‑climbing with several
    randomized perturbations (block scaling, redistribution,
    smoothing, single‑element scaling and a specialised move that
    targets the current maximal convolution entry).  The algorithm
    runs for the allotted time budget and returns the best sequence
    seen.
    """
    import time, numpy as np

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # keep 5 s safety margin

    # initialise with the globally supplied solution if it looks valid.
    base = globals().get('height_sequence_1')
    if isinstance(base, (list, np.ndarray)):
        seq0 = np.asarray(base, dtype=np.float64)
    else:
        seq0 = np.ones(256, dtype=np.float64)

    seq0 = np.clip(seq0, 0.0, 1000.0)
    if seq0.ndim != 1 or seq0.size == 0 or not np.isfinite(seq0).all():
        seq0 = np.ones(256, dtype=np.float64)

    best = seq0.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    cur_score = best_score

    n = len(best)

    # temperature schedule parameters
    init_temp = 2.0   # generous at start
    final_temp = 1e-4

    iteration = 0
    # main loop
    while time.monotonic() < deadline:
        # linearly decreasing temperature (logarithmic schedule also works)
        elapsed = time.monotonic() - start
        progress = elapsed / (deadline - start)
        temp = init_temp * (final_temp / init_temp) ** progress

        # pick a move style (0…4)
        move_type = rng.integers(0, 5)
        if move_type == 0:
            cand = move_block_scale(cur, rng)
        elif move_type == 1:
            cand = move_reallocate(cur, rng)
        elif move_type == 2:
            cand = move_smooth(cur, rng)
        elif move_type == 3:
            cand = move_random_scale(cur, rng)
        else:
            cand = move_reduce_peak(cur, rng)

        # evaluate candidate
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # deterministic improvement for the global best
        if score < best_score:
            best = cand
            best_score = score
            print(f"[{time.monotonic():.1f}] new best: {best_score:.6f} (iter {iteration})")
            # keep the current point at the new best as well
            cur = cand
            cur_score = score
            continue

        # simulated‑annealing acceptance for the current walk
        if score < cur_score:
            accept = True
        else:
            # probability of uphill move
            prob = np.exp(-(score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = score

        # occasional restart to avoid drifting too far from the best point
        if iteration % 1000 == 0 and iteration > 0:
            # with a small probability we return to the best solution and add a tiny jitter
            if rng.random() < 0.2:
                jitter = move_block_scale(best, rng)  # reuse block perturb as jitter
                cur = jitter
                cur_score = float(evaluate_sequence(cur.tolist()))

        iteration += 1

    # safety fallback: if no finite score was ever found return a simple constant vector
    if not np.isfinite(best_score):

"""
Simulated‑annealing optimisation for the step‑height sequence.

The routine starts from the globally supplied `height_sequence_1`
(if it exists) and then performs a time‑bounded stochastic search.
Five different perturbation operators are used:

0. block scaling – multiplies a random contiguous block by a log‑normal factor,
   optionally mirrors the whole sequence;
1. mass reallocation – moves a fraction of mass from a high‑value index to a
   randomly chosen low‑value index;
2. local smoothing – applies a short moving‑average filter;
3. single‑element scaling – rescales one random entry;
4. peak reduction – identifies the current maximal convolution index,
   reduces the two entries that realise this peak and redistributes the
   removed mass elsewhere.  This operator directly attacks the objective.

A simple simulated‑annealing acceptance rule (always accept improvements,
accept worsening moves with probability exp(−Δ/temperature)) allows the
search to escape local minima.  The temperature decays logarithmically from
`init_temp` to `final_temp`.  The best sequence seen so far is kept and
returned after the allocated time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper perturbation operators
# ----------------------------------------------------------------------


def move_block_scale(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random block and optionally symmetrise."""
    n = seq.size
    cand = seq.copy()
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))          # log‑normal factor
    cand[start:stop] *= scale
    if rng.random() < 0.35:                              # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def move_reallocate(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a fraction of mass from a large entry to another random index."""
    n = seq.size
    cand = seq.copy()
    total = np.sum(cand)
    if total == 0.0:
        src = rng.integers(0, n)
    else:
        probs = cand / total
        src = rng.choice(n, p=probs)                     # source weighted by size
    dst = rng.integers(0, n)
    while dst == src:
        dst = rng.integers(0, n)
    max_move = cand[src] * 0.5
    move = float(rng.random()) * max_move
    cand[src] -= move
    cand[dst] += move
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def move_smooth(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a short moving‑average filter (low‑pass smoothing)."""
    n = seq.size
    cand = seq.copy()
    k = int(rng.integers(3, 8))                         # kernel size 3‑7
    kernel = np.full(k, 1.0 / k, dtype=np.float64)
    cand = np.convolve(cand, kernel, mode='same')
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def move_random_scale(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a single random coefficient."""
    n = seq.size
    cand = seq.copy()
    idx = rng.integers(0, n)
    factor = float(np.exp(rng.normal(0.0, 0.09)))       # log‑normal, usually near 1
    cand[idx] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def move_reduce_peak(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Target the current maximal convolution entry and dampen it."""
    n = seq.size
    cand = seq.copy()
    # locate the peak of the full convolution
    conv = np.convolve(cand, cand)
    k = int(np.argmax(conv))

    # indices that participate in this peak
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    i = rng.integers(i_min, i_max + 1)
    j = k - i

    # shrink both contributors (factor <= 1)
    factor = float(np.exp(rng.normal(0.0, 0.07)))       # typical < 1
    factor = min(factor, 1.0)

    delta_i = cand[i] * (1.0 - factor)
    cand[i] *= factor
    if j != i:
        delta_j = cand[j] * (1.0 - factor)
        cand[j] *= factor
    else:
        delta_j = 0.0
    removed = delta_i + delta_j

    # redistribute the removed mass to two random positions
    for _ in range(2):
        idx = rng.integers(0, n)
        add = removed * rng.random()
        cand[idx] += add
        removed -= add
    if removed > 0.0:                                   # any leftovers go to index 0
        cand[0] += removed

    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function is given ~1000 s and must return the
    best sequence it discovered within that time.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin (the harness may cut us off after budget_s)
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # initialise from the global construction if possible
    # ------------------------------------------------------------------
    base = globals().get('height_sequence_1')
    if isinstance(base, (list, np.ndarray)):
        cur = np.asarray(base, dtype=np.float64)
    else:
        cur = np.ones(256, dtype=np.float64)            # simple fallback
    cur = np.clip(cur, 0.0, 1000.0)

    if cur.ndim != 1 or cur.size == 0 or not np.isfinite(cur).all():
        cur = np.ones(256, dtype=np.float64)

    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score

    n = best.size

    # simulated‑annealing schedule
    init_temp = 2.0
    final_temp = 1e-4

    # weighted choice of move types (peak reduction gets higher weight)
    move_weights = np.array([1, 1, 1, 1, 2])
    move_probs = move_weights / move_weights.sum()

    iteration = 0
    while time.monotonic() < deadline:
        # temperature decays logarithmically
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)
        temp = init_temp * (final_temp / init_temp) ** progress

        # pick a move
        mv = rng.choice(len(move_probs), p=move_probs)
        if mv == 0:
            cand = move_block_scale(cur, rng)
        elif mv == 1:
            cand = move_reallocate(cur, rng)
        elif mv == 2:
            cand = move_smooth(cur, rng)
        elif mv == 3:
            cand = move_random_scale(cur, rng)
        else:
            cand = move_reduce_peak(cur, rng)

        # evaluate – this is the authoritative, possibly expensive, call
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # keep the global best
        if score < best_score:
            best = cand
            best_score = score
            print(f"[{elapsed:.1f}s] new best: {best_score:.6f} (iter {iteration})")
            # also move the current point to the best (helps acceptance)
            cur = cand
            cur_score = score
            iteration += 1
            continue

        # simulated‑annealing acceptance for the current walk
        if score < cur_score:
            accept = True
        else:
            prob = np.exp(-(score - cur_score) / max(temp, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = score

        # occasional gentle restart to the best solution (adds jitter)
        if iteration % 1000 == 0 and iteration > 0:
            if rng.random() < 0.2:
                cur = move_block_scale(best, rng)
                cur_score = float(evaluate_sequence(cur.tolist()))

        iteration += 1

    # safety fallback – should never happen because `best` starts valid
    if not np.isfinite(best_score):
        print("Fallback: returning uniform sequence.")
        best = np.ones(256, dtype=np.float64)

    return [float(x) for x in best]
