"""
Propose a step‑function (list of non‑negative heights) that minimises the
security‑evaluation function.

The algorithm works in three stages:

1.  **Parameter sweep** – tries simple analytically motivated families:
    *   power‑law decay a_i = (i+1)^(-p)   (p in [0.2, 1.0])
    *   shifted square‑root decay a_i = 1/√(i+α) (α in [0,10])
    *   a combination of the two.
    The best candidate from this cheap sweep is kept.

2.  **Local random search** – starting from the best candidate, repeatedly
    applies a random move (block multiplicative perturbation, a small
    mass‑swap, or a smoothing/average step) and keeps any improvement.
    The search runs until the allotted time expires.

3.  Returns the best sequence found as a plain ``list[float]`` – exactly the
    format expected by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: block perturbation (the same idea used by the reference code)
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # width is a modest fraction of the length (between 0.5% and 8%)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring – helps to reduce the peak of the convolution
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

    """

    # Determine the problem size (normally the global ``height_sequence_1``)
    base_seq = None

        try:

            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                base_seq = arr.copy()
        except Exception:
            base_seq = None

        base_seq = np.ones(256, dtype=np.float64)

    n = base_seq.size

    # ------------------------------------------------------------------
    # Initial candidate (just the provided sequence)
    best = np.clip(base_seq, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) PARAMETER SWEEP ------------------------------------------------
    # power‑law decay a_i = (i+1)^(-p)
    for p in np.linspace(0.20, 1.00, 9):

        if np.isfinite(score) and score < best_score:

    # 2) LOCAL RANDOM SEARCH -------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # spare a couple of seconds
    while time.monotonic() < deadline:
        # pick a cheap perturbation type
        move = rng.choice(["block", "swap", "smooth"])

        if move == "block":
            candidate = make_block_perturbation(best, rng)

        elif move == "swap":
            # move a tiny amount of mass from i to j
            candidate = best.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # amount proportional to the source height, never negative
                delta = rng.uniform(0.0, min(candidate[i], 0.5 * candidate[i] + 1e-12))
                candidate[i] -= delta
                candidate[j] += delta
            candidate = np.clip(candidate, 0.0, 1000.0)

        else:  # "smooth"
            # a soft blend with its reversal plus a tiny random jitter
            candidate = 0.5 * (best + best[::-1])
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            candidate *= jitter
            candidate = np.clip(candidate, 0.0, 1000.0)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = candidate, score
            print(f"[search] new best {score:.6f} via {move}")

    # ------------------------------------------------------------------
    print(f"[done] best_score={best_score:.6f}")
    # Convert to plain Python list as required by the evaluator
    return [float(x) for x in best]
