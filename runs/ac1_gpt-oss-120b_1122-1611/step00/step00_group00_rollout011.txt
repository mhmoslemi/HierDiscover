def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, budget_s - 3.0)

    best_seq = None
    best_score = float('inf')
    
    # Determine candidate lengths to try
    lengths = [int(5000), int(8000), int(12000)]
    # maybe also use length of height_sequence_1 as fallback.
    
    # Try geometric sequences with ratios r = 1 - c/ n for c in [0.5, 1, 1.5, 2, 3, 4]
    for n in lengths:
        for c in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
            if time.monotonic() >= deadline:
                break
            r = 1.0 - c / n
            # Ensure r positive
            if r <= 0:
                continue
            # construct geometric shape
            i = np.arange(n, dtype=np.float64)
            a = np.exp(i * np.log(r))  # r**i
            S = a.sum()
            a_norm = a / S
            # convert to list for evaluate
            seq = a_norm.tolist()
            score = float(evaluate_sequence(seq))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_norm.copy()
                print(f"New best: n={n} c={c} score={score}")
    
    # If still none (should not), fallback to uniform or previous best.
    if best_seq is None:
        # fallback uniform of same length as previous sequence
        n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("Fallback uniform sequence", best_score)
    
    # Additional local search after initial best
    while time.monotonic() < deadline:
        # random small perturbation with block scaling or log-normal
        candidate = np.array(best_seq, copy=True)
        # random block scaling similar to earlier
        width = int(rng.integers(max(1, len(candidate) // 200), max(2, len(candidate) // 12)))
        start = int(rng.integers(0, max(1, len(candidate) - width + 1)))
        stop = min(len(candidate), start + width)
        factor = float(np.exp(rng.normal(0, 0.05)))  # smaller perturb
        candidate[start:stop] *= factor
        # optional smoothing
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        # re-normalize to sum 1
        candidate = np.clip(candidate, 0, 1000)
        s = candidate.sum()
        if s > 0:
            candidate /= s
        else:
            continue
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"Improved to {best_score}")
    
    # return list of float
    return [float(x) for x in best_seq]

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Gradient‑descent / stochastic search for a non‑negative step function
    that minimises ``evaluate_sequence``.

    The method starts from a gently decreasing geometric profile (which is
    already better than the uniform step function) and repeatedly flattens the
    auto‑convolution by moving the sequence in the direction that reduces the
    squared L2‑error between the convolution and its average value.
    Random block perturbations and occasional element swaps are mixed in to
    escape local minima.  The best sequence encountered before the time budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)
    # time budget (few seconds are left for the final return)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise a sequence of length n
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            n = 256
        else:
            n = parent.size
    else:
        n = 256

    # geometric start: a_i ∝ r^i, r = 1 - c/n  (c≈1.2 gives a good base)
    c_start = 1.2
    r = 1.0 - c_start / n
    if r <= 0.0:
        r = 0.5
    base = r ** np.arange(n, dtype=np.float64)
    a = base / base.sum()                     # normalise to sum = 1.0

    # safety – guarantee non‑negative and finite
    a = np.clip(a, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # helper to compute the same objective that evaluate_sequence uses
    # (much faster because we stay in NumPy land)
    # ------------------------------------------------------------------
    def score(seq: np.ndarray) -> float:
        s = seq.sum()
        if s < 0.01:
            return np.inf
        conv = np.convolve(seq, seq)          # full self‑convolution
        return float(2.0 * seq.size * conv.max() / (s * s))

    best = a.copy()
    best_score = score(best)
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    # parameters for the gradient step
    step_size = 0.12          # controls how far we move each iteration
    stagnation = 0           # counts iterations without improvement

    while time.monotonic() < deadline:
        # 1. compute current convolution and the residual w.r.t. the flat target
        conv = np.convolve(a, a)                       # length 2n‑1
        target = (a.sum() ** 2) / (2 * n - 1)          # the constant we would like to reach
        resid = conv - target

        # 2. gradient of the squared‑error loss L = ||conv‑target||²
        #    ∂L/∂a_i = 4 * Σ_j a_j * (conv_{i+j} - target)
        grad = 4.0 * np.correlate(resid, a, mode='valid')

        # 3. take a small step in the negative gradient direction
        #    The step length η is chosen so that the L2‑norm of the update
        #    equals ``step_size`` (this makes the magnitude of each step
        #    roughly independent of n).
        eta = step_size / (np.linalg.norm(grad) + 1e-12)
        cand = a - eta * grad

        # 4. enforce constraints (non‑negative, sum = 1, clip to allowed range)
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total > 0:
            cand /= total
        else:
            cand = a.copy()                 # fall back to the previous point

        cand_score = score(cand)

        # ------------------------------------------------------------------
        # accept if we improved, otherwise perform a small random perturbation
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            a = cand
            stagnation = 0
            # slowly shrink the step as we get closer to a good solution
            step_size = max(0.005, step_size * 0.99)
            print(f"[improve] score={best_score:.6f}  step={step_size:.4f}")
        else:
            stagnation += 1

            # every few unsuccessful steps try a cheap random move
            if stagnation % 15 == 0:
                # ---- random block scaling (similar to the baseline method) ----
                width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)
                tmp = a.copy()
                factor = float(np.exp(rng.normal(0.0, 0.30)))
                tmp[start:stop] *= factor
                # occasional mirror‑averaging (helps symmetry)
                if rng.random() < 0.25:
                    tmp = 0.5 * (tmp + tmp[::-1])
                tmp = np.clip(tmp, 0.0, 1000.0)
                tmp_sum = tmp.sum()
                if tmp_sum > 0:
                    tmp /= tmp_sum
                tmp_score = score(tmp)
                if tmp_score < best_score:
                    best = tmp.copy()
                    best_score = tmp_score
                    a = tmp
                    stagnation = 0
                    step_size = 0.12
                    print(f"[block]   score={best_score:.6f}")

            # if we are stuck for a long time, restart from a fresh random point
            if stagnation > 200:
                a = rng.exponential(scale=1.0, size=n)
                a = np.clip(a, 0.0, 1000.0)
                a /= a.sum()
                stagnation = 0
                step_size = 0.12
                print("[restart] new random start")

        # early exit if we already beat the target bound
        if best_score < 1.5025:
            print(f"[target reached] score={best_score:.6f}")
            break

    # return as plain Python list (the runner will feed it to evaluate_sequence)
    return [float(x) for x in best]
