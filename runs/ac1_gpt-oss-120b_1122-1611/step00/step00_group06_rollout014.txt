import cvxpy as cp
import numpy as np
import time

def propose_candidate(seed=123, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # initial sequence – use height_sequence_1 if valid else uniform
    base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        n = 256
        base = np.ones(n, dtype=np.float64)
    n = base.size

    # normalize sum to sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    base = base * (target_sum / np.sum(base))

    f = np.clip(base, 0.0, 1000.0)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # main loop: alternate LP step and local perturbations
    t = 0.02  # step size for convex combination
    while time.monotonic() < deadline:
        # compute convolution M
        conv = np.convolve(f, f)
        M = np.max(conv)

        # formulate LP: maximize sum(g) subject g >= 0 and conv_fg <= M
        g = cp.Variable(n)
        # build Toeplitz matrix for convolution
        # conv_fg[k] = sum_{i} f[i] * g[k-i] , with g out-of-range = 0
        # We can implement constraints by iterating over k
        constraints = [g >= 0]
        # Efficiently compute convolutions using Toeplitz
        # For each k from 0 to 2n-2:
        for k in range(2*n - 1):
            # compute indices i where both f[i] and g[k-i] are defined
            # i ranges max(0, k-(n-1)) .. min(n-1, k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start > i_end:
                # conv value = 0 <= M automatically
                continue
            # vector of f[i] values
            f_slice = f[i_start:i_end+1]
            # g indices: k-i correspond to indices in that slice from (k-i_start) downwards
            # Let's write linear expression sum_i f[i] * g[k-i]
            expr = cp.sum(cp.multiply(f_slice, g[i_start:i_end+1][::-1]))  # reversed slice
            constraints.append(expr <= M)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:
            # fallback if solver fails
            break

        if g.value is None:
            break
        g0 = np.maximum(0.0, np.minimum(1000.0, np.array(g.value, dtype=np.float64)))
        if np.isfinite(g0).all():
            # rescale to target sum
            sum_g0 = np.sum(g0)
            if sum_g0 > 1e-12:
                g_scaled = g0 * (target_sum / sum_g0)
                # convex combination
                f_new = (1-t) * f + t * g_scaled
                f_new = np.clip(f_new, 0.0, 1000.0)
                # evaluate
                score = float(evaluate_sequence(f_new.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = f_new
                    best_score = score
                    print(f"Improved score: {best_score:.6f}")
                f = f_new
        # occasional random perturbation to escape local optimum
        if rng.random() < 0.1:
            f = make_block_perturbation(f, rng)
    # return best list
    return [float(x) for x in best]
```

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

    c = 1.0

    # Build the three‑block pattern: block – zeros – block.
    block = np.full(L, c, dtype=np.float64)
    zeros = np.zeros(L, dtype=np.float64)
    seq = np.concatenate([block, zeros, block])

    # If the original length was not a multiple of three, pad with zeros.
    # Padding only adds zeros, so `max(conv)` does not change.
    if len(seq) < base_len:
        seq = np.concatenate([seq, np.zeros(base_len - len(seq), dtype=np.float64)])

    # Clip to the bounds enforced by `evaluate_sequence` (just in case).
    seq = np.clip(seq, 0.0, 1000.0)

    # Optional: compute and display the expected score (one exact evaluation).
    # This call costs O(n²) but is done only once.
    try:
        score = float(evaluate_sequence(seq.tolist()))
        print(f"[propose_candidate] constructed length={len(seq)}, block={L}, "
              f"expected score≈{score:.6f}")
    except Exception:
        # If evaluation fails for any reason we still return the sequence.
        print("[propose_candidate] constructed sequence (evaluation skipped)")

    # Return as a plain Python list of floats.
    return [float(x) for x in seq]
