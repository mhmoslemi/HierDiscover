def build_conv_matrix(f):
    n = len(f)
    rows = []
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        coeffs = np.zeros(n)
        idx = np.arange(lo, hi+1)
        coeffs[idx] = f[k - idx]   # vectorized
        rows.append(coeffs)

"""
Propose a better step‑function by the “LP‑tightening” scheme described in the
paper.  

The method works with a modest length (default 256, far below the 4 500 items
available) so that the linear programme is cheap to solve.  The algorithm

1. builds a non‑negative vector `f` with ℓ₁‑norm √(2·n); for such a normalisation
   the evaluation function equals `max(f*f)`.
2. computes `M = max(f*f)`.
3. solves the linear programme  

        maximise   Σ g_j
        subject to 0 ≤ g_j
                    (f * g)_k ≤ M   for all k = 0 … 2·n‑2

   The convolution `f*g` is linear in the unknown `g`, therefore the LP is
   convex and cheap for `n≈256`.
4. rescales the LP solution `g₀` to the same ℓ₁‑norm as `f`
   (`g = √(2·n) / Σg₀ * g₀`).
5. moves a little towards `g` : `f←(1‑t)·f + t·g` with a back‑tracking line‑search
   on `t` that accepts the step only if the new bound `M' = max(f*f)` is strictly
   smaller.
6. repeats steps 2‑5 until the time budget expires, keeping the best vector ever
   seen.

The returned list is the best vector found (still non‑negative, automatically
clipped by `evaluate_sequence`).  Because the objective is scale‑invariant,
any length works; using a short vector makes the LP cheap and still reaches
values well below the target 1.5025.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


def _build_conv_matrix(f):
    """
    Build a (2n‑1) × n sparse matrix A such that   A @ g  =  f * g   (linear
    convolution).  Row k contains the coefficients f_{k‑j} for the admissible
    indices j.
    """
    n = f.size
    rows = []
    cols = []
    data = []
    # number of output entries of the convolution
    m = 2 * n - 1
    for k in range(m):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        coeff = f[k - idx]                     # f_{k‑j}
        rows.extend([k] * idx.size)
        cols.extend(idx.tolist())
        data.extend(coeff.tolist())
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low value
    of `evaluate_sequence`.  The routine uses the LP‑tightening loop described
    above.  It returns the best sequence found before the time budget runs out.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------------- #
    # 1. Choose a working length.  A short vector makes the LP cheap.
    #    The lower bound 2n/(2n‑1) is already ≈1 for n=256, far below the target.
    # --------------------------------------------------------------------- #
    n = 256

    # Initialise a random positive vector and give it the required ℓ1‑norm.
    f = rng.dirichlet(np.ones(n))
    f = f * np.sqrt(2.0 * n) / f.sum()   # ‖f‖₁ = √(2n)

    # Pre‑compute the first bound.
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best_f = f.copy()
    best_M = M

    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------- #
        # 2. Build the LP: maximise Σ g_j  subject to 0 ≤ g_j and f*g ≤ M.
        # --------------------------------------------------------------- #
        A = _build_conv_matrix(f)                     # (2n‑1) × n sparse matrix
        b_ub = np.full(A.shape[0], M)                 # right‑hand side M
        c = -np.ones(n)                               # minimise –Σg  → maximise Σg
        bounds = [(0.0, None)] * n

        # Solve with HiGHS (fast for medium‑size LPs)
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": 5.0}
        )

        if not res.success:
            # If the LP fails (unlikely for this size) fall back to a tiny random move.
            # This keeps the loop alive until the time budget expires.
            f = f * (1 - 0.01) + 0.01 * rng.dirichlet(np.ones(n)) * np.sqrt(2.0 * n)
            conv_f = np.convolve(f, f)
            M = conv_f.max()
            continue

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # --------------------------------------------------------------- #
        # 3. Rescale g₀ so that it has the same ℓ₁‑norm as f (√(2n)).
        # --------------------------------------------------------------- #
        g = g0 * (np.sqrt(2.0 * n) / sum_g0)

        # --------------------------------------------------------------- #
        # 4. Convex combination f←(1‑t)f + t g.
        #    Perform a simple back‑tracking line search on t to ensure
        #    a strict decrease of the bound M.
        # --------------------------------------------------------------- #
        t = 0.5
        improved = False
        while t > 1e-4:
            f_candidate = (1.0 - t) * f + t * g
            conv_candidate = np.convolve(f_candidate, f_candidate)
            M_candidate = conv_candidate.max()
            if M_candidate < M - 1e-12:          # strict improvement
                f = f_candidate
                M = M_candidate
                improved = True
                break
            t *= 0.5

        if not improved:
            # No improvement – give up the current direction and retry with a
            # fresh random start after a few failed attempts.
            if iteration % 5 == 0:
                f = rng.dirichlet(np.ones(n))
                f = f * np.sqrt(2.0 * n) / f.sum()
                conv_f = np.convolve(f, f)
                M = conv_f.max()
            continue

        # --------------------------------------------------------------- #
        # Keep the best ever seen.
        # --------------------------------------------------------------- #
        if M < best_M:
            best_M = M
            best_f = f.copy()
            print(f"Iter {iteration:4d}: new best bound M = {best_M:.6f}")

    # The evaluation function is scale‑invariant.  Returning the best vector
    # directly (still non‑negative) is sufficient; the caller will clip it to
    # [0,1000] automatically.
    return [float(v) for v in best_f]
