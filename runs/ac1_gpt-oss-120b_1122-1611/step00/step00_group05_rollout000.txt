"""
Propose a candidate sequence for the step‑function optimization problem.

Algorithm
---------
1. **Initialisation** – Start from a uniform non‑negative sequence.
   A uniform sequence (all entries equal) attains the theoretical lower
   bound `evaluate_sequence == 2.0` because for any non‑negative vector `a`

        max(a*a) ≥ sum_i a_i^2 ≥ (sum_i a_i)^2 / n

   and thus `2*n*max_b/(sum_a**2) ≥ 2`.  Uniformity makes the first
   inequality tight, so the best possible finite score is exactly `2.0`.

2. **Exploration (simulated‑annealing style)** – While time permits,
   repeatedly perturb a randomly chosen contiguous block of the current
   best vector by a log‑normal factor, optionally symmetrising it.  The
   perturbation is kept within `[0, 1000]` (the same clipping that
   ``evaluate_sequence`` applies).  A new candidate is accepted if it
   improves the score, otherwise it may be accepted with a small
   temperature‑dependent probability to avoid getting stuck.

3. **Return** – After the time budget expires, the best sequence found is
   returned as a plain ``list[float]``.  In practice the uniform start
   already yields the optimal score `2.0`; the search is a safety net
   that can discover any accidental improvement caused by floating‑point
   quirks.

The function complies with all competition constraints: no file I/O,
no closures, deterministic given the seed, and it respects the
``budget_s`` time limit.
"""

import time
import numpy as np

def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative O(n) perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # Choose a block whose width is a reasonable fraction of the sequence.
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # Log‑normal multiplicative factor (median 1) – small variance.
    factor = float(np.exp(rng.normal(0.0, 0.08)))
    candidate[start:stop] *= factor
    # Occasionally enforce symmetry – helps explore the space.
    if rng.random() < 0.20:
        candidate = 0.5 * (candidate + candidate[::-1])
    # Apply the same clipping that ``evaluate_sequence`` does.
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that aims to minimise
    ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds.  The function stops a few
        seconds before the limit to guarantee a return value.
    **kwargs
        Ignored – present for compatibility with the competition harness.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # Time management – stop a couple of seconds early to avoid a forced kill.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1. Determine a sensible starting length.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is provided by the framework.
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if base_arr.ndim != 1 or base_arr.size == 0 or not np.isfinite(base_arr).all():
            raise ValueError
        n = base_arr.size
    except Exception:
        # Fallback – a modest length that evaluates instantly.
        n = 256
        base_arr = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2. Initialise with a uniform sequence – this attains the theoretical
    #    lower bound of 2.0.
    # ------------------------------------------------------------------
    uniform_val = 1.0                     # any positive constant works
    best = np.full(n, uniform_val, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing style local search.
    # ------------------------------------------------------------------
    temperature = 1.0          # starting temperature
    iter_count = 0

    while time.monotonic() < deadline:
        candidate = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(candidate.tolist()))
        iter_count += 1

        if np.isfinite(score):
            # Accept improvements outright.
            if score < best_score:
                best = candidate
                best_score = score
                # Reset temperature a little to encourage finer moves.
                temperature = max(temperature * 0.95, 1e-4)

            else:
                # Probabilistic acceptance of worse solutions.
                if temperature > 0:
                    prob = np.exp(-(score - best_score) / temperature)
                    if rng.random() < prob:
                        best = candidate
                        best_score = score

        # Cool the temperature slowly.
        temperature *= 0.9999

    # ------------------------------------------------------------------
    # 4. Logging for transparency.
    # ------------------------------------------------------------------
    print(f"Finished after {iter_count} iterations.")
    print(f"Best score obtained: {best_score:.12f} (theoretical optimum = 2.0)")

    # Return a plain list of Python floats as required.
    return [float(v) for v in best]
