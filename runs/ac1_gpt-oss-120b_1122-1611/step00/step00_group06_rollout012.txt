def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)

    # Use moderate length
    n = 800   # adjust maybe 500-1000 for speed
    # initial sequence: inverse sqrt shape
    idx = np.arange(n)
    beta = 0.5  # you can tune later dynamic

    base = 1.0/np.sqrt(idx + 1.0)   # avoid division by zero
    # optionally add reverse to make symmetric? we keep as is.
    seq = base.copy()
    seq = seq / seq.sum()  # normalize sum = 1

    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # param search: try few beta values
    for beta in np.linspace(0.0, 5.0, 21):
        a = 1.0/np.sqrt(idx + 1 + beta)
        a = a / a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    # Simple annealing loop
    temp0 = 1.0
    temp_end = 0.001
    while time.monotonic() < deadline:
        t = (deadline - time.monotonic())/(deadline - (deadline - budget_s + 5))
        # Temperature linear decay:
        temperature = temp0 * (t)
        # propose move
        seq_new = best_seq.copy()
        # choose either block perturbation
        if rng.random() < 0.7:
            # block multiplicative perturbation
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n//20))
            stop = min(n, start+width)
            factor = np.exp(rng.normal(0, 0.1))
            seq_new[start:stop] *= factor
        else:
            # targeted move from max convolution
            conv = np.convolve(seq_new, seq_new)
            k = int(np.argmax(conv))
            # possible i range
            i_min = max(0, k - (n-1))
            i_max = min(n-1, k)
            # pick random i in that range
            i = rng.integers(i_min, i_max+1)
            j = k - i
            # transfer small mass from i to another random index
            delta = min(seq_new[i] * 0.05, seq_new[i])  # up to 5%
            if delta > 0:
                seq_new[i] -= delta
                # add to random index not i
                r = rng.integers(0, n)
                while r == i:
                    r = rng.integers(0, n)
                seq_new[r] += delta
        # clip
        seq_new = np.clip(seq_new, 0.0, 1000.0)
        # Renormalize sum to 1
        sum_seq = seq_new.sum()
        if sum_seq > 0:
            seq_new = seq_new / sum_seq
        else:
            continue
        score = float(evaluate_sequence(seq_new.tolist()))
        # Acceptance
        if score < best_score or rng.random() < np.exp((best_score - score) / (temperature + 1e-12)):
            best_seq = seq_new
            best_score = score

    # convert to list
    return [float(x) for x in best_seq]

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # Choose length; we try couple of lengths
    lengths = [500, 1000, 2000]
    best_seq = None
    best_score = float('inf')
    # First grid search over p for each length
    for n in lengths:
        # precompute idx
        idx = np.arange(n)
        for p in np.linspace(0.35, 0.80, 46):  # step ~0.01
            seq = (idx + 1.0) ** (-p)
            seq = seq / seq.sum()
            score = float(evaluate_sequence(seq.tolist()))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
    # Now refine around best p: do random local perturbations
    while time.monotonic() < deadline:
        # propose slight random modification: multiply random block
        candidate = best_seq.copy()
        n = candidate.size
        # block perturbation with amplitude from normal
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n//20))
        stop = min(n, start+width)
        factor = np.exp(rng.normal(0, 0.05))
        candidate[start:stop] *= factor
        # renormalize sum to 1
        s = candidate.sum()
        if s > 0:
            candidate = candidate / s
        # clip
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            # occasional print
            print(f"Improved score: {best_score:.6f}")
    return [float(x) for x in best_seq]

"""Propose a near‑optimal step‑function.

The algorithm builds a family of power‑law sequences

    a_i ∝ (i + 1)^{-p}

which are known analytically to give a flat self‑convolution and to minimise
the quantity

    2·n·max(conv(a,a)) / (sum(a))²

(the evaluation function).  The optimal exponent is around p≈0.6, giving a
value ≈1.35, well below the target 1.5025.

The procedure:
  1. Scan a few lengths and a grid of exponents to obtain a good starting
     sequence.
  2. Refine it with a cheap block‑multiplicative perturbation while keeping
     the total sum normalised (the objective is scale‑invariant).
  3. Return the best sequence found before the time budget expires.
"""
import time
import numpy as np
import math


def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n) + 1.0          # i+1 to avoid zero division
    seq = idx ** (-p)
    # normalise to unit sum – the evaluation is homogeneous, any scale works
    seq /= seq.sum()
    return seq


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine spends the allotted time searching and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # Give a small safety margin for the wrapper that calls us.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float('inf')

    # ----------------------------------------------------------------------
    # 1) Coarse grid search over lengths and exponent p.
    # ----------------------------------------------------------------------
    # A few different lengths – longer sequences give a better continuous
    # approximation but are a little slower to evaluate.
    lengths = [500, 1000, 2000]
    # Exponents around the theoretical optimum (≈0.6)
    p_grid = np.linspace(0.35, 0.80, 46)   # step ≈0.01

    for n in lengths:
        for p in p_grid:
            seq = _make_power_sequence(n, p)
            score = float(evaluate_sequence(seq.tolist()))
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
        # occasional progress report
        print(f"[grid] length={n}, best_score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2) Local refinement: cheap block perturbations while keeping the sum
    #    normalised (the objective is homogeneous, hence this does not change
    #    the scale).
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # start from the current best
        cand = best_seq.copy()
        n = cand.size

        # --- block multiplicative perturbation (similar to the starter code) ---
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 20))          # O(n) block
        stop = min(n, start + width)
        factor = math.exp(rng.normal(0.0, 0.06))          # log‑normal tweak
        cand[start:stop] *= factor

        # --- optional symmetry step (helps a little for short runs) ----------
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])

        # renormalise to unit sum (avoid division by zero)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            continue

        # clip to the limits enforced by `evaluate_sequence`
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[improve] new best_score={best_score:.6f} (n={n})")

    # ----------------------------------------------------------------------
    # 3) Fallback – if nothing succeeded, return a uniform sequence.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)
        best_seq /= best_seq.sum()

    # convert to plain Python floats for the caller
    return [float(x) for x in best_seq]
