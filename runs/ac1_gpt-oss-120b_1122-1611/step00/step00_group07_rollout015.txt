def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 5000
    # initial sequence: random positive numbers
    a = rng.random(n) + 1e-3  # avoid zeros
    # normalize sum to sqrt(2n) (not needed)
    a = a / a.sum() * math.sqrt(2 * n)   # scale to typical sum
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    temperature = 1.0
    iteration = 0
    while time.monotonic() < deadline:
        # random perturbation
        # choose operation type
        op = rng.integers(0,3)
        cand = best_seq.copy()
        if op == 0:
            # perturb a random block multiplicatively
            width = int(rng.integers(1, max(2, n // 100)))  # small width
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.1))
            cand[start:start+width] *= factor
        elif op == 1:
            # transfer mass between two indices
            i, j = rng.choice(n, size=2, replace=False)
            amount = rng.random() * cand[i] * 0.5
            cand[i] -= amount
            cand[j] += amount
        else:
            # add Gaussian noise
            noise = rng.normal(0.0, 0.05, size=n)
            cand = np.maximum(cand + noise, 0.0)
        # renormalize to keep sum similar
        s = cand.sum()
        if s < 1e-6:
            continue
        cand = cand * (math.sqrt(2 * n) / s)  # keep same sum as original base
        # clip bounds
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        # accept if better or with probability exp((best_score - score)/temperature)
        if score < best_score or rng.random() < math.exp((best_score - score) / max(temperature, 1e-8)):
            best_seq = cand
            best_score = score
            temperature *= 0.9999  # cool down slowly
        iteration += 1
        if iteration % 50 ==0:
            print(f"Iter {iteration}, best score {best_score:.6f}")
    return [float(x) for x in best_seq]

"""
Propose a candidate sequence for the step‐function inequality.

The algorithm starts from a random positive sequence (scaled to a typical
total sum) and then performs a simple simulated‑annealing hill‑climb.
At each iteration a random local modification is applied:
  * multiplicative scaling of a short contiguous block,
  * transfer of mass between two positions,
  * addition of small Gaussian noise.
The new sequence is re‑normalised (the evaluation function is
scale‑invariant) and clipped to the allowed range.
If the modification improves the evaluation score it is kept;
otherwise it may be accepted with a temperature‑dependent probability.
The best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a low‑evaluation height sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget (seconds).  The function returns the best
        sequence found before `budget_s - 3` seconds have elapsed.
    kwargs : dict
        Unused – required by the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered (list of non‑negative floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine the length of the sequence.  Use the provided global if
    # available, otherwise fall back to a reasonable size.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5000

    # ------------------------------------------------------------------
    # Initialise a random positive sequence and scale it to a convenient
    # total sum (the evaluation is homogeneous, any non‑zero sum works).
    # Using sqrt(2*n) mirrors the normalisation used in the reference
    # algorithm but is not required for correctness.
    # ------------------------------------------------------------------
    a = rng.random(n) + 1e-4                # avoid exact zeros
    target_sum = math.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())

    # Keep the current state and the best ever state.

    cur_score = float(evaluate_sequence(cur.tolist()))

    best_score = cur_score

    temperature = 1.0
    iteration = 0

    # Helper: evaluate a sequence (clipped inside evaluate_sequence).
    def eval_seq(seq):
        return float(evaluate_sequence(seq.tolist()))

    # ------------------------------------------------------------------
    # Main optimisation loop – stops when the time budget is exhausted.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- choose a random modification ---------------------------------
        op = rng.integers(0, 3)          # three types of moves
        cand = cur.copy()

        if op == 0:
            # 0 – multiplicative scaling of a short block
            max_width = max(2, n // 150)               # keep O(n) memory
            width = int(rng.integers(1, max_width + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.12))   # modest log‑normal change
            cand[start:start + width] *= factor
        elif op == 1:
            # 1 – transfer a fraction of mass from i to j
            i, j = rng.choice(n, size=2, replace=False)
            if cand[i] > 0.0:
                delta = cand[i] * rng.random() * 0.5   # up to 50 % of i
                cand[i] -= delta
                cand[j] += delta
        else:
            # 2 – add small zero‑mean Gaussian noise (ensures exploration)
            noise = rng.normal(0.0, 0.05, size=n)
            cand = np.maximum(cand + noise, 0.0)

        # ----- re‑normalise and enforce bounds -------------------------------
        total = cand.sum()
        if total < 1e-12:                # degenerate – reject immediately
            continue
        # Keep the same total sum (scale invariance of the objective)
        cand = cand * (target_sum / total)
        # Clip according to the problem's security checks
        np.clip(cand, 0.0, 1000.0, out=cand)

        # ----- evaluate -------------------------------------------------------
        cand_score = eval_seq(cand)

        # ----- acceptance logic -----------------------------------------------
        # Always keep improvements.  Occasionally accept worse candidates
        # to escape local minima (simulated‑annealing style).
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            # Update the global best if we improved.
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                # Cool down faster when we find a new record.
                temperature *= 0.95
            else:
                # Gentle cooling on normal moves.
                temperature *= 0.9995

        # ----- occasional status output ---------------------------------------
        if iteration % 100 == 0:
            print(f"Iter {iteration:6d} | cur_score={cur_score: .6f} "
                  f"| best_score={best_score: .6f} | temp={temperature:.4f}")

    # Return the best sequence found (as plain Python floats).
    return [float(x) for x in best]
