def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Generate a monotone power‐law height sequence.
    
    The evaluation function is 
        2*n*max(conv(a,a))/(sum(a)^2)
    which is minimised when the sequence is increasing and concentrates
    its mass at the end.  For a fixed length the optimal empirical shape
    is a power law a_i = (i/(n-1))**p * 1000.  The routine tries a few
    exponents, keeps the best and then performs a short random refinement.
    """
    import numpy as np, time, math

    # Determine a reasonable length.  If a previous construction is present
    # we keep its size, otherwise we fall back to 2000 elements.
    base = globals().get('height_sequence_1')
    if isinstance(base, (list, np.ndarray)) and len(base) > 0:
        n = len(base)
    else:
        n = 2000

    # ---------- initial grid search over a few exponents ----------
    best_seq = None
    best_score = float('inf')
    # exponents to test – the sweet spot for n≈2500 is around p=5
    exponents = [0.5, 1, 2, 3, 4, 5, 6, 7, 8]
    for p in exponents:
        # monotone power‑law on [0,1000]
        x = np.arange(n, dtype=np.float64) / (n - 1)
        seq = (x ** p) * 1000.0
        seq = np.clip(seq, 0.0, 1000.0)
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception as e:
            # any unexpected error – treat as infinite
            score = float('inf')
        print(f"p={p:.1f} → score={score}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # ---------- optional random refinement until the time budget expires ----------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    rng = np.random.default_rng(seed + 9999)
    while time.monotonic() < deadline:
        # probe a exponent around the current best (which is usually p≈5)
        p = 5.0 + rng.normal(0.0, 0.5)
        if p <= 0.0:
            p = 0.1
        x = np.arange(n, dtype=np.float64) / (n - 1)
        cand = (x ** p) * 1000.0
        cand = np.clip(cand, 0.0, 1000.0)
        try:
            score = evaluate_sequence(cand.tolist())

            continue
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()

        best_seq = np.full(n, 1.0, dtype=np.float64)

    a_i = (i/(n-1))**p * 1000   for i = 0,…,n‑1

It evaluates each candidate with `evaluate_sequence`, keeps the best,
and then performs a quick random refinement while the time budget allows.
All values are clipped to the allowed interval [0, 1000].

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Generate a monotone power‑law height sequence that minimises
    `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for any stochastic refinement.

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
