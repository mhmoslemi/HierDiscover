import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a step‑function (list of non‑negative heights) that minimizes
    `evaluate_sequence`.  By the Cauchy‑Schwarz inequality the theoretical
    minimum of the objective is 2.0 and it is achieved by any uniform
    non‑zero sequence (all entries equal).  The function therefore returns
    a uniform sequence of the same length as the current best construction
    (`height_sequence_1` if available, otherwise a default length).  A short
    local‑search loop is run for the allotted time; any improvement over the
    uniform candidate would be extremely rare, so the uniform solution is
    returned as the final answer.
    """
    # initialise RNG and timing
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine a starting length: try to keep the same length as the
    # previously discovered sequence, falling back to a modest default.
    try:
        # height_sequence_1 is expected to be defined globally
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = base.size
    except Exception:
        n = 256  # fallback length

    # Uniform candidate – each entry is the maximal allowed value (1000)
    # to maximise the denominator (sum) while keeping the numerator minimal.
    uniform_val = 1000.0
    best = np.full(n, uniform_val, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial uniform candidate (n={n}) score = {best_score:.6f}")

    # If for any reason the uniform candidate is not the best, we perform a
    # very cheap random‑block perturbation search.  This loop respects the
    # time budget and may be effectively a no‑op when the uniform sequence
    # is already optimal.
    while time.monotonic() < deadline:
        # generate a perturbation: pick a block and scale it slightly
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = best.copy()
        # multiply the block by a log‑normal factor centred at 1
        factor = np.exp(rng.normal(0.0, 0.12))
        cand[start:stop] *= factor
        # optional symmetric blending
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])

        # Clip to the allowed interval [0,1000] (exactly as evaluate_sequence does)

"""
Propose a step‑function that minimizes `evaluate_sequence`.

The objective
    f(a) = 2 * n * max(a * a) / (sum(a)^2)
is bounded from below by 2.  This follows from the Cauchy–Schwarz inequality:
    sum_i a_i^2 >= (sum_i a_i)^2 / n,
and because the zero‑lag convolution equals sum_i a_i^2.  Equality holds
iff all entries are equal (a uniform sequence).  Consequently the
theoretical optimum is exactly 2.0, attained by any non‑zero constant
sequence.

The function therefore builds a uniform sequence of the same length as
the current best construction (`height_sequence_1` if it exists, otherwise a
fallback length).  A short random‑block perturbation loop is run for the
remaining time budget; any improvement over the uniform candidate would be
exceptional, so the uniform sequence is returned as the final answer.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (almost) minimises the
    evaluation function.  The returned sequence is uniform, which is
    provably optimal (score = 2.0).  A lightweight local search is performed
    only to respect the allotted time budget; it will keep the uniform
    sequence unless a rare numerical improvement appears.
    """
    rng = np.random.default_rng(seed)
    # Allocate time for the search (a few seconds are saved for safe exit)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine the length of the candidate.
    # Prefer the length of the previously discovered construction.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = int(base.size)
    except Exception:
        n = 256        # fallback length if the global variable is absent

    # ------------------------------------------------------------------
    # Uniform candidate – the provable optimum.
    # Each entry is set to the maximal allowed value (1000.0) to maximise
    # the denominator while keeping the numerator minimal.
    # ------------------------------------------------------------------
    uniform_val = 1000.0
    best = np.full(n, uniform_val, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] uniform length={n}, score={best_score:.12f}")

    # ------------------------------------------------------------------
    # Optional cheap random‑block perturbation search.
    # The loop respects the time budget; any improvement is kept.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # Choose a random block width and location.
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        # Propose a candidate by scaling the chosen block.
        cand = best.copy()
        factor = np.exp(rng.normal(0.0, 0.12))   # log‑normal perturbation
        cand[start:stop] *= factor

        # Occasionally symmetrise the candidate.
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])

        # Clip to the admissible interval [0, 1000] (exactly as evaluate_sequence does).
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Evaluate the proposal.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[improve] new best score={best_score:.12f}")

        # The uniform solution already achieves the lower bound 2.0.
        # If we have reached that bound (up to floating‑point tolerance),
        # break early – no better solution exists.
        if best_score <= 2.0 + 1e-12:
            break

    print(f"[final] returning length={n}, final score={best_score:.12f}")
    # Convert to plain Python floats as required by the evaluator.
    return [float(x) for x in best]
