def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Generate a power-law decreasing step sequence that flattens the self-convolution.
    The exponent p>0.5 yields a smaller L∞ norm of the convolution, giving a lower evaluation score.
    The function tries several exponents and returns the best found within the time budget.
    """
    import numpy as np, time, math

    # Determine length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            n = 256
        else:
            n = base.size
    except Exception:
        n = 256

    # Candidate exponents to test
    exponents = [0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95]

    best_seq = None
    best_score = float('inf')
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)

    idx = np.arange(1, n+1, dtype=np.float64)  # 1..n

    for p in exponents:
        if time.monotonic() >= deadline:
            break
        seq = idx ** (-p)
        # Clip to evaluator limits
        seq = np.clip(seq, 0.0, 1000.0)
        # The evaluator is scaling invariant; keep as is.
        score = float(evaluate_sequence(seq.tolist()))
        print(f"Test exponent p={p:.3f}, score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq

    # If still none (should not happen), fallback to uniform
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("Fallback to uniform sequence, score:", best_score)

    print(f"Chosen exponent yields score {best_score:.6f}")

    # Return as plain python list of floats
    return [float(x) for x in best_seq]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate a power‑law decreasing step sequence that flattens the self‑convolution.
    For an exponent p > 0.5 the L∞ norm of the convolution becomes much smaller than
    that of a uniform sequence, yielding a lower value of ``evaluate_sequence``.
    The routine tests several exponents (while respecting the time budget) and
    returns the sequence with the lowest score found.
    """
    import numpy as np, time

    # ----------------------------------------------------------------------
    # Determine the length of the problem (use the provided global if possible)
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            n = int(base.size)
        else:
            n = 256
    except Exception:
        n = 256

    # --------------------------------------------------------------
    # Candidate exponents (p>0.5 gives a flattened convolution)
    # --------------------------------------------------------------
    exponents = [0.55, 0.60, 0.65, 0.70, 0.75, 0.80, 0.85, 0.90, 0.95]

    # --------------------------------------------------------------
    # Time‑limited search over the exponent list
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 3.0)   # leave a few seconds margin

    best_seq = None
    best_score = float('inf')
    idx = np.arange(1, n + 1, dtype=np.float64)   # i = 1 … n

    for p in exponents:
        if time.monotonic() >= deadline:
            break

        # a_i = i^{-p}
        seq = idx ** (-p)
        # Clip to the limits enforced inside evaluate_sequence
        seq = np.clip(seq, 0.0, 1000.0)

        # The metric is scale‑invariant, no additional normalisation required
        score = float(evaluate_sequence(seq.tolist()))

        print(f"Test exponent p={p:.3f} → score={score:.6f}")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()

    # -----------------------------------------------------------------
    # In the unlikely event that none of the candidates succeeded,
    # fall back to a uniform sequence (the previous best known bound).
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("Fallback to uniform sequence, score:", best_score)

    print(f"Chosen exponent yields final score {best_score:.6f}")

    # Return a plain Python list of floats, as required by the runner.
    return [float(x) for x in best_seq]
