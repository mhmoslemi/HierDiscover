"""
propose_candidate
-----------------

A stochastic hill‑climbing / simulated‑annealing optimizer that tries to lower the
value returned by ``evaluate_sequence``.  The algorithm works with the
following ideas:

*   The objective 2·n·max(conv)/sum² is homogeneous, therefore we keep the
    total L¹‑norm fixed (≈√(2·n)) – this removes the trivial scaling degree of
    freedom and makes the landscape easier to explore.

*   Starting from the uniform distribution (which gives a score of exactly 2),
    we repeatedly apply two complementary perturbations:

    - **Block perturbation** – a contiguous block of the sequence is multiplied
      by a log‑normal factor.  This creates large‑scale shape changes.

    - **Pairwise transfer** – a random amount of mass is moved from one index
      to another while keeping the total sum constant.  This is a cheap way
      to re‑distribute weight locally.

*   After each perturbation the sequence is projected back onto the feasible
    set (non‑negative, sum = √(2·n) and clipped to the evaluator’s hard limits
    0 ≤ value ≤ 1000).

*   The new sequence is accepted if it improves the score, otherwise it may
    be accepted with a probability that follows a simple annealing schedule.
    The best sequence seen so far is always kept.

*   The loop runs until the time budget (≈ 997 s, leaving a few seconds for the
    final return) is exhausted, printing occasional progress reports.

The implementation avoids closures and lambda‑expressions, uses only the
standard library together with ``numpy`` and respects the required function
signature.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projects the vector ``v`` onto the simplex
        { x >= 0 , sum(x) = target_sum }.
    This is the classic O(n) algorithm based on sorting.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # shift to make the sum equal to target_sum after projection
    v = v.copy()
    # ensure non‑negative before the projection step
    v = np.maximum(v, 0.0)
    # if the current sum already equals the target, we are done
    cur = v.sum()
    if np.isclose(cur, target_sum):
        return v
    # if the vector is all zeros, just return a uniform distribution
    if cur == 0.0:
        return np.full_like(v, target_sum / v.size)

    # ----  projection onto the simplex  ----
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42,
                      budget_s: float = 1100,
                      **kwargs) -> list[float]:
    """Search for a sequence with a low value of ``evaluate_sequence``.

    The function performs a simple simulated‑annealing search.  It returns the
    best sequence found before the time budget expires.  The global variable
    ``height_sequence_1`` (if present) is used only as a fallback – the search
    starts from a uniform distribution because that is a known feasible point
    with a score of exactly 2.0.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock budget in seconds (the function stops a few
        seconds before the limit to avoid a forced termination).

    Returns
    -------
    list[float]
        The best sequence discovered, ready to be fed into the evaluator.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and a convenient normalisation (sum = √(2·n)).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        # a reasonable default – the tests always provide height_sequence_1
        n = 4806
    target_sum = np.sqrt(2.0 * n)          # the normalisation used in the paper

    # start from the uniform distribution (this yields a score of exactly 2)
    a = np.full(n, target_sum / n, dtype=np.float64)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score
    cur = a.copy()

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)   # leave a margin
    # initial temperature (controls acceptance of uphill moves)
    temperature = 0.5
    # cooling factor – after each outer iteration we shrink the temperature
    cooling = 0.9995

    iteration = 0
    report_every = 2000          # progress report interval

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Choose a random move type
        # --------------------------------------------------------------
        if rng.random() < 0.6:
            # ----- block perturbation (same idea as the reference code) -----
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = np.exp(rng.normal(0.0, 0.12))
            candidate = cur.copy()
            candidate[start:stop] *= factor
            # optional mirroring to keep the shape symmetric
            if rng.random() < 0.35:
                candidate = 0.5 * (candidate + candidate[::-1])
        else:
            # ----- pairwise transfer move --------------------------------
            i, j = rng.choice(n, size=2, replace=False)
            if cur[i] > 0.0:
                delta = cur[i] * rng.random() * 0.5    # move at most 50 % of a_i
                candidate = cur.copy()
                candidate[i] -= delta
                candidate[j] += delta
            else:
                candidate = cur.copy()

        # --------------------------------------------------------------
        # Enforce feasibility: non‑negative, clip to [0,1000] and fix the sum.
        # --------------------------------------------------------------
        candidate = np.clip(candidate, 0.0, 1000.0)
        # project onto the simplex with the required L¹‑norm
        candidate = _project_onto_simplex(candidate, target_sum)

        # --------------------------------------------------------------
        # Evaluate the new candidate
        # --------------------------------------------------------------
        score = float(evaluate_sequence(candidate.tolist()))

        # --------------------------------------------------------------
        # Acceptance rule (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if np.isfinite(score):
            if score < cur_score:
                accept = True
            else:
                # occasional uphill moves according to the current temperature
                diff = score - cur_score
                prob = np.exp(-diff / max(temperature, 1e-12))
                accept = rng.random() < prob

        if accept:
            cur = candidate
            cur_score = score
            # keep the global best ever seen
            if score < best_score:
                best = candidate.copy()
                best_score = score
                print(f"[{iteration:08d}] NEW BEST: {best_score:.6f}")

        # --------------------------------------------------------------
        # Temperature schedule
        # --------------------------------------------------------------
        temperature *= cooling

        # --------------------------------------------------------------
        # Periodic progress report
        # --------------------------------------------------------------
        if iteration % report_every == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{iteration:08d}] elapsed={elapsed:.1f}s "
                  f"cur={cur_score:.6f} best={best_score:.6f}")

    # final report
    total_time = time.monotonic() - start_time
    print(f"Search finished after {total_time:.1f}s – best score = {best_score:.6f}")

    # return a plain Python list as required by the evaluator
    return [float(x) for x in best]
