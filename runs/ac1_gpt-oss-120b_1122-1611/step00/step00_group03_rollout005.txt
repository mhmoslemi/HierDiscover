def build_convolution_matrix(f):
    n = len(f)
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        # i from max(0, k-(n-1)) to min(k, n-1) inclusive.
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for each i, j = k - i
        # weight f_i is coefficient with g_j.
        # Thus A[k, j] = f_i where i = k - j.
        # Instead of nested loops, we can directly assign slice.
        # The j indexes are from k - i_end to k - i_start.
        # Let's compute j_start = k - i_end, j_end = k - i_start (inclusive)
        # Number of elements = i_end - i_start + 1
        # The corresponding f_i slice = f[i_start:i_end+1]
        # The g indices slice = j_start:j_end+1 (the same length).
        js = slice(k - i_end, k - i_start + 1)  # +1 because slice end exclusive
        A[k, js] = f[i_start:i_end+1]
    return A

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Find a step function with low max convolution ratio using random sampling
    and simple hill climbing.

    The search runs for up to `budget_s` seconds. It starts from the provided
    global `height_sequence_1` (if any) and repeatedly draws random perturbations,
    evaluates them with a fast hand‑crafted score function and keeps the best.
    Occasionally a simulated‑annealing hill‑climb is performed to fine‑tune the
    current best sequence.
    """
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # Determine length n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise Exception
        n = base.size
        # Clip and ensure non‑negative
        base = np.clip(base, 0.0, 1000.0)
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # fast score without calling evaluate_sequence
    def score(seq):
        seq = np.clip(seq, 0.0, 1000.0)
        s = np.sum(seq)
        if s < 0.01:
            return np.inf
        b = np.convolve(seq, seq)
        return 2.0 * n * np.max(b) / (s * s)

    best = base.copy()
    best_score = score(best)

    # Keep track of iteration counts
    it = 0
    # Parameters for random generation
    # probability of using a fresh random candidate vs local perturbation
    prob_random = 0.3
    # magnitude for multiplicative block perturbation similar to previous code
    while time.monotonic() < deadline:
        it += 1
        if rng.random() < prob_random:
            # fresh random candidate
            # Use exponential distribution to get many small values and a few big ones
            cand = rng.exponential(scale=5.0, size=n)  # mean 5, adjust later
            # optionally apply a random permutation
            cand = rng.permuted(cand)
            # Scale total sum to a reasonable magnitude to avoid hitting clip limits
            # Ensure at least one element hits the cap 1000 occasionally
            scale_factor = rng.uniform(0.5, 2.0)
            cand = cand * scale_factor
        else:
            # local block perturbation of current best
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = best.copy()
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:stop] *= factor
            # occasional mirroring
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])

        # Clip to allowed range
        cand = np.clip(cand, 0.0, 1000.0)

        # Compute score
        sc = score(cand)

        # Accept improvement
        if sc < best_score:
            best = cand
            best_score = sc
            # decrease prob_random to focus on hill‑climbing when we are doing well
            prob_random = max(0.05, prob_random * 0.99)
            # log progress
            print(f"Iter {it}: new best score {best_score:.6f}")

        # Occasionally perform a simulated‑annealing style move that swaps two entries
        if it % 1000 == 0:
            # try a swap move on the current best
            i, j = rng.choice(n, size=2, replace=False)
            cand2 = best.copy()
            cand2[i], cand2[j] = cand2[j], cand2[i]
            sc2 = score(cand2)
            if sc2 < best_score:
                best = cand2
                best_score = sc2
                print(f"Iter {it} swap improvement: {best_score:.6f}")

        # Time check
        if it % 5000 == 0 and time.monotonic() > deadline - 5.0:
            # final stretch: tighten by small Gaussian noise per element
            cand = best + rng.normal(0.0, 0.05, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = score(cand)
            if sc < best_score:
                best = cand
                best_score = sc
                print(f"Iter {it} final noise improvement: {best_score:.6f}")

    # Return best as python list of floats
    return [float(x) for x in best]

import numpy as np, time, math

def compute_score(seq: np.ndarray) -> float:
    """Return the evaluate_sequence score for a 1‑D array of non‑negative numbers.

    The implementation mirrors evaluate_sequence but avoids the Python
    overhead of the provided function.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(np.sum(seq))
    if s < 0.01:
        return np.inf
    b = np.convolve(seq, seq)
    return 2.0 * seq.size * float(np.max(b)) / (s * s)

def random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a fresh random candidate of length n.

    Values are drawn from an exponential distribution, then optionally scaled
    and permuted to increase diversity.
    """
    # Exponential with mean ~5 (adjustable)
    cand = rng.exponential(scale=5.0, size=n)
    # Random permutation
    cand = rng.permuted(cand)
    # Random global scaling between 0.3 and 3.0 to explore different total sums
    scale = rng.uniform(0.3, 3.0)
    cand = cand * scale
    # Clip to the problem limits
    return np.clip(cand, 0.0, 1000.0)

def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a block‑wise multiplicative perturbation of ``seq``."""
    n = seq.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:

    return np.clip(cand, 0.0, 1000.0)

"""
Fast random‑search + tiny local hill‑climbing for the convolution‑ratio
problem.

The objective is   2·n·max(a*a) / (sum(a)^2),  where a is a non‑negative
vector (the step‑function heights).  The code repeatedly creates random
candidates (exponential draws, global scaling and permutation) and mixes in
small block‑wise multiplicative perturbations of the current best solution.
A very cheap “swap” and a tiny Gaussian jitter are also tried from time to
time to escape plateaus.  The best sequence seen before the time budget
expires is returned as a plain list of Python floats.
"""

# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------
def compute_score(seq: np.ndarray) -> float:
    """
    Compute the same score that ``evaluate_sequence`` returns, but without
    the extra Python checks.  The sequence is first clipped to the allowed
    range [0, 1000] and the sum is required to be at least 0.01.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(np.sum(seq))
    if s < 0.01:
        return np.inf
    # full convolution (length 2·n‑1)
    b = np.convolve(seq, seq)
    return 2.0 * seq.size * float(np.max(b)) / (s * s)


def random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a fresh random candidate of length ``n``.
    Values are drawn from an exponential distribution (mean ≈5), globally
    scaled by a random factor and randomly permuted.
    """
    # Exponential draws – they give a heavy tail and a reasonable variance.
    cand = rng.exponential(scale=5.0, size=n)

    # Random permutation to break any accidental structure.
    cand = rng.permuted(cand)

    # Random global scaling – allows the total L1 norm to explore a wide
    # region while staying inside the [0,1000] limits after clipping.
    scale = rng.uniform(0.3, 3.0)
    cand = cand * scale

    # Enforce the hard limits required by ``evaluate_sequence``.
    return np.clip(cand, 0.0, 1000.0)


def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:

    return np.clip(cand, 0.0, 1000.0)
