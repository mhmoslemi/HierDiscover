def gradient_move(seq, sum_seq, rng):
    n = seq.shape[0]
    # compute convolution only needed at max index
    b = np.convolve(seq, seq)
    k = int(np.argmax(b))
    # compute gradient contributions
    start = max(0, k - (n - 1))
    stop = min(k, n - 1)
    # we can compute g as zeros then fill
    g = np.zeros_like(seq)
    # i from start to stop inclusive
    # g[i] = 2 * seq[k - i]
    # we can vectorize:
    i_range = np.arange(start, stop+1)
    g[i_range] = 2 * seq[k - i_range]
    # choose index i weighted by g (higher weight)
    # to avoid zero sum, add epsilon
    weights = g + 1e-12
    i = int(rng.choice(n, p=weights / np.sum(weights)))
    # choose j weighted by inverse g (or lower)
    inv_weights = (np.max(g) - g) + 1e-12
    j = int(rng.choice(n, p=inv_weights / np.sum(inv_weights)))
    # transfer small amount
    # max transfer limited by seq[i]
    max_transfer = seq[i]
    if max_transfer <= 1e-12:
        return seq  # nothing to transfer
    # choose delta: random fraction up to maybe 0.2 of max_transfer
    delta = float(max_transfer) * rng.random() * 0.2
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # ensure within [0,1000]
    new_seq[i] = max(0.0, min(new_seq[i], 1000.0))
    new_seq[j] = max(0.0, min(new_seq[j], 1000.0))
    # due to clipping, sum might change a tiny bit, renormalize to original sum
    total = np.sum(new_seq)
    if total > 0:
        new_seq = new_seq * (sum_seq / total)
    else:
        # if sum zero (unlikely), just keep existing
        pass
    return new_seq

def gradient_move(seq, sum_target, rng):
    n = seq.size
    # compute convolution for max
    b = np.convolve(seq, seq)
    k = int(np.argmax(b))
    # compute gradient contributions
    start = max(0, k - (n - 1))
    stop = min(k, n - 1)
    # indices i where both seq[i] and seq[k-i] exist
    i_range = np.arange(start, stop + 1, dtype=np.int64)
    # counterpart indices
    j_range = k - i_range
    # gradient = 2 * seq[j_range]
    g = np.zeros_like(seq)
    g[i_range] = 2.0 * seq[j_range]
    # probabilities for picking source index (high gradient)
    # add small eps to avoid zero prob
    src_weights = g + 1e-12
    src_idx = int(rng.choice(n, p=src_weights / src_weights.sum()))
    # probabilities for picking destination index (low gradient)
    max_g = g.max()
    dst_weights = (max_g - g) + 1e-12
    dst_idx = int(rng.choice(n, p=dst_weights / dst_weights.sum()))
    # transfer mass
    amount = float(seq[src_idx]) * rng.random() * 0.5  # up to 50% of source
    if amount <= 0:
        return seq
    new_seq = seq.copy()
    new_seq[src_idx] -= amount
    new_seq[dst_idx] += amount
    # clip [0,1000]
    new_seq[src_idx] = max(0.0, min(new_seq[src_idx], 1000.0))
    new_seq[dst_idx] = max(0.0, min(new_seq[dst_idx], 1000.0))
    # maintain sum (will renorm later)
    return new_seq

"""
Propose a step‑function sequence whose self‑convolution is (asymptotically) constant.

The continuous identity
    f(t) = 1 / sqrt(t)      (t>0)   →   (f * f)(s) = π   (for all s>0)
suggests the discrete analogue
    a_i ∝ 1 / sqrt(min(i+1, n-i))

gives a self‑convolution that is almost flat, so the ratio
    2·n·max(b) / (sum(a)^2)  →  π/4 ≈ 0.785

which is far below the target 1.5025.  The function builds this analytic
shape for the required length, optionally applies a few tiny random
log‑normal perturbations while time allows, and returns the best sequence
found.
"""

import time
import numpy as np

def _analytic_u_shape(n: int) -> np.ndarray:
    """Return a symmetric U‑shaped array a_i = 1/√(min(i+1, n-i))."""
    idx = np.arange(n)
    dist = np.minimum(idx + 1, n - idx)          # distance to the nearest edge (≥1)
    seq = 1.0 / np.sqrt(dist.astype(np.float64))
    # values are already in [0,1] → safe for the evaluator
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Generate a high‑quality step‑function sequence.

    The core construction is the analytic U‑shape described above.
    While time remains (budget_s seconds) we perform a tiny stochastic
    refinement: each entry is multiplied by an independent log‑normal factor
    (σ≈0.02) and the whole vector is rescaled to keep its total sum unchanged.
    Because the evaluation metric is scale‑invariant, this refinement can only
    improve the score by reducing the effect of clipping.

    Returns
    -------
    list[float]
        The best sequence discovered (as plain Python floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine the required length.
    # Prefer the length of the global reference sequence if it exists.
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        n = 1024                                   # fallback length

    # ------------------------------------------------------------------
    # Analytic start point.
    seq = _analytic_u_shape(n)
    sum_target = seq.sum() if seq.sum() > 0 else 1.0

    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] analytic score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Light stochastic refinement (log‑normal multiplicative noise).
    # The refinement respects the sum constraint – the metric is invariant
    # under uniform scaling, but keeping the sum helps the clipping step.
    while time.monotonic() < deadline:
        # Propose a new candidate by perturbing each entry multiplicatively.
        factor = np.exp(rng.normal(0.0, 0.02, size=n))   # σ≈0.02 → ≤~20 % change
        cand = best_seq * factor
        cand = np.clip(cand, 0.0, 1000.0)

        # Rescale to the original sum (prevent drift caused by clipping).
        total = cand.sum()
        if total > 0:
            cand = cand * (sum_target / total)

        # Evaluate the candidate.
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand
            print(f"[improve] new best score = {best_score:.6f}")

    # Return a plain list of floats as required by the harness.
    return [float(x) for x in best_seq]
