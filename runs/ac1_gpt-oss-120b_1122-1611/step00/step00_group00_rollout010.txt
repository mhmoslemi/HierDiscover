def propose_candidate(seed=42, budget_s=1100, ...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # start point
    if 'height_sequence_1' in globals():
        parent = np.array(height_sequence_1, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    best = parent.copy()
    best_score = evaluate_sequence(best.tolist())
    # Ensure score finite
    if not np.isfinite(best_score):
        best_score = float('inf')
    # Pre compute length
    n = best.size

    # main loop
    while time.monotonic() < deadline:
        # Use targeted move
        candidate = best.copy()
        # Compute convolution
        b = np.convolve(candidate, candidate)
        k_max = int(np.argmax(b))

        # Get list of indices i that contribute to k_max
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        # Pick random i in this interval
        i = int(rng.integers(i_min, i_max + 1))
        j = k_max - i

        # Random factor for reduction
        factor = float(np.exp(rng.normal(0.0, 0.1)))  # factor around 1
        # decide to increase or decrease? We'll attempt decreasing product for those indices
        # Ensure factor <=1
        if factor > 1.0:
            factor = 1.0 / factor

        # Change values
        old_i = candidate[i]
        candidate[i] = max(0.0, old_i * factor)
        if i != j:
            old_j = candidate[j]
            candidate[j] = max(0.0, old_j * factor)
        else:
            old_j = old_i

        # Compute amount removed from sum
        delta_sum = (old_i - candidate[i]) + (old_j - candidate[j]) if i != j else (old_i - candidate[i])

        # Add mass to two random indices not in {i,j}
        # Choose two random positions (could be same)
        p = int(rng.integers(0, n))
        q = int(rng.integers(0, n))
        # ensure not i/j? not needed as adding mass may increase max but we hope not
        candidate[p] = np.clip(candidate[p] + delta_sum/2.0, 0.0, 1000.0)
        candidate[q] = np.clip(candidate[q] + delta_sum/2.0, 0.0, 1000.0)

        # Optionally occasionally symmetrize
        if rng.random() < 0.1:
            candidate = 0.5 * (candidate + candidate[::-1])

        # Clip again
        candidate = np.clip(candidate, 0.0, 1000.0)

        # Evaluate
        score = evaluate_sequence(candidate.tolist())
        if np.isfinite(score) and score < best_score:

"""
A simple guided local‑search for minimizing the evaluation function

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm repeatedly attacks the current worst convolution index.
For the index k where the autocorrelation attains its maximum we pick a pair
(i,j) with i+j=k and reduce the product a_i·a_j by scaling both entries
down. The removed mass is redistributed to random positions that are far
from k, which tends to increase the denominator without creating a new
large peak.  The search is complemented by occasional random block
perturbations (the original heuristic) to provide diversification.

The routine runs for the allocated time budget (default 1100 s, 3 s safety
margin) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) perturbation of ``base`` by scaling a random block."""
    cand = base.copy()
    n = cand.size
    # block width roughly between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _score(seq: np.ndarray) -> float:
    """Exact score matching evaluate_sequence (without its safety checks)."""
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.size
    max_b = np.max(np.convolve(seq, seq))
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _best_convolution_info(seq: np.ndarray):
    """
    Return (max_value, argmax_index, full_convolution) for the given sequence.
    """
    conv = np.convolve(seq, seq)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimizes the
    evaluation function.  The function runs for at most ``budget_s`` seconds
    (a small safety margin is subtracted) and returns the best list found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise starting point
    # ------------------------------------------------------------------
    # ``height_sequence_1`` may be provided by the surrounding environment.
    init = globals().get('height_sequence_1')
    if isinstance(init, (list, tuple, np.ndarray)):
        parent = np.asarray(init, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)          # fallback

    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = _score(best)
    best_max, best_kmax, best_conv = _best_convolution_info(best)

    print(f"[init] length={best.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----------------------------------------------------------------
        # 1) directed move: attack the current worst convolution index
        # ----------------------------------------------------------------
        n = best.size
        kmax = best_kmax                     # index where current max occurs
        i_min = max(0, kmax - (n - 1))
        i_max = min(n - 1, kmax)

        idx_range = np.arange(i_min, i_max + 1, dtype=np.int64)
        j_range = kmax - idx_range

        # contribution of each (i,j) pair to the offending peak
        contrib = best[idx_range] * best[j_range]
        total = np.sum(contrib)

        if total > 0.0:
            # sample a pair proportionally to its contribution
            probs = contrib / total
            i = rng.choice(idx_range, p=probs)
        else:
            # fallback: random i inside the feasible range
            i = int(rng.integers(i_min, i_max + 1))

        j = kmax - i

        # ----------------------------------------------------------------
        # 2) propose a new candidate by reducing the chosen pair
        # ----------------------------------------------------------------
        cand = best.copy()

        # scaling factor < 1 (usually a reduction); drawn from log‑normal
        factor = float(np.exp(rng.normal(0.0, 0.10)))   # ≈ 0.9 … 1.1
        # to bias towards reduction we mirror values >1
        if factor > 1.0:
            factor = 1.0 / factor

        old_i = cand[i]
        cand[i] = max(0.0, old_i * factor)

        if i != j:
            old_j = cand[j]
            cand[j] = max(0.0, old_j * factor)
        else:
            old_j = old_i

        # mass that disappeared from the sum
        delta_sum = (old_i - cand[i]) + (old_j - cand[j]) if i != j else (old_i - cand[i])

        # redistribute the removed mass to two random locations far from kmax
        # (they are unlikely to create a new dominant peak)
        for _ in range(2):
            p = int(rng.integers(0, n))
            # ensure we do not write back onto i or j by a tiny offset
            if p == i or p == j:
                continue
            cand[p] = np.clip(cand[p] + delta_sum / 2.0, 0.0, 1000.0)

        # occasional symmetrisation (helps sometimes)
        if rng.random() < 0.07:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)

        # ----------------------------------------------------------------
        # 3) evaluate the candidate
        # ----------------------------------------------------------------
        cand_score = _score(cand)

        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand
            best_score = cand_score
            best_max, best_kmax, best_conv = _best_convolution_info(best)
            print(f"[{it:05d}] improved score={best_score:.6f}  (kmax={best_kmax})")
            continue   # go to next iteration with the fresh best

        # ----------------------------------------------------------------
        # 4) occasional random block perturbation for diversification
        # ----------------------------------------------------------------
        if rng.random() < 0.04:
            cand2 = _make_block_perturbation(best, rng)
            cand2_score = _score(cand2)
            if np.isfinite(cand2_score) and cand2_score < best_score:
                best = cand2
                best_score = cand2_score
                best_max, best_kmax, best_conv = _best_convolution_info(best)
                print(f"[{it:05d}] block‑perturb improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final report
    # ------------------------------------------------------------------
    print(f"[done] iterations={it}, final score={best_score:.6f}, length={best.size}")
    return [float(x) for x in best]
