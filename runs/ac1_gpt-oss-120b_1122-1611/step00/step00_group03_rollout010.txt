    f = f / f.sum()  # sum = 1
    best_f = f.copy()
    best_score = float(evaluate_sequence(f.tolist()))
    # Initial M
    M = np.max(np.convolve(f, f))
    t = 0.2  # step size for mixing; may reduce over time

    iter = 0
    while time.monotonic() < deadline:
        # Solve LP to get g
        g = cp.Variable(n, nonneg=True)
        # Use param for f
        f_param = cp.Parameter(n, value=f)
        conv_fg = cp.conv(f_param, g)
        constraints = [conv_fg <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        if g.value is None:
            # solver failed
            break
        g_val = np.maximum(0, np.array(g.value).flatten())
        sum_g = g_val.sum()
        if sum_g < 1e-12:
            break
        # normalize g to sum=1
        g_norm = g_val / sum_g
        # Mix f and g_norm
        f = (1 - t) * f + t * g_norm
        # Renormalize to sum=1
        f = f / f.sum()
        # compute new M
        M = np.max(np.convolve(f, f))
        # evaluate
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f.copy()
            print(f"Iter {iter}: new best {best_score:.6f}")
        iter += 1
        # maybe reduce t over time
        if iter % 10 == 0:
            t = max(0.01, t * 0.9)
    # finalize
    return [float(x) for x in best_f]

"""
Improved construction for the step–function minimisation problem.

The algorithm follows a Frank–Wolfe style scheme:
1. Initialise a uniform distribution `f` (∑f = 1, f_i ≥ 0).
2. Let M = max_k (f * f)[k] (the current worst convolution value).
3. Solve the linear programme  

        maximise   Σ g_i
        subject to conv(f , g) ≤ M   (element‑wise)
                  g_i ≥ 0

   The convolution constraint is linear in the unknown vector `g`,
   therefore a cheap LP can be solved with CVXPY/ECOS.
   The optimiser returns a partner sequence `g` with the largest possible
   ℓ₁‑norm while respecting the bound `M`.

4. Normalise `g` to the same ℓ₁‑norm as `f` (here we keep the sum equal to 1)
   and blend the two vectors  

        f ← (1‑t)·f + t·g_norm

   with a decreasing step size `t`.  Each blend reduces the peak of the
   self‑convolution and consequently lowers the evaluation score.
5. The loop runs until the wall‑clock budget expires, keeping the best
   sequence ever observed.  If CVXPY is unavailable (or a solve fails) the
   code falls back to the simple block‑perturbation local search used in the
   reference implementation.

The returned list can be fed directly to ``evaluate_sequence``.
"""

import numpy as np
import time


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) beats the
    current upper bound 2.0.  The function is time‑limited; after ``budget_s``
    seconds it returns the best sequence found so far.
    """
    # -----------------------------------------------------------------------
    # 1.  Imports and optional CVXPY availability test
    # -----------------------------------------------------------------------
    try:
        import cvxpy as cp
        HAVE_CVX = True
    except Exception:          # pragma: no cover
        HAVE_CVX = False

    # -----------------------------------------------------------------------
    # 2.  Determine the length of the target sequence.
    #    ``height_sequence_1`` is supplied by the evaluation harness.
    # -----------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:                      # pragma: no cover
        n = 256                              # safe fallback

    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------------
    # 3.  Initialise a uniform distribution (sum = 1) – this is scale‑invariant.
    # -----------------------------------------------------------------------
    f = np.full(n, 1.0 / n, dtype=np.float64)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))   # evaluate_sequence is global

    # mixing factor – start fairly aggressive and decay slowly
    t = 0.3

    # -----------------------------------------------------------------------
    # 4.  Prepare CVXPY objects once (if possible) so that only parameters are
    #     updated each iteration.
    # -----------------------------------------------------------------------
    if HAVE_CVX:
        f_param = cp.Parameter(n, nonneg=True)          # current f
        g = cp.Variable(n, nonneg=True)                # partner vector
        conv_fg = cp.conv(f_param, g)                  # full convolution (2n‑1)
        M_param = cp.Parameter(nonneg=True)            # current bound M
        constraints = [conv_fg <= M_param]             # element‑wise ≤
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # -----------------------------------------------------------------------
    # 5.  Main optimisation loop – run until we run out of the time budget.
    # -----------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    it = 0

    while time.monotonic() < deadline:
        # current peak of the self‑convolution
        M = np.max(np.convolve(f, f))

        if HAVE_CVX:
            # Update CVXPY parameters
            f_param.value = f
            M_param.value = float(M)

            # Solve the LP (ECOS works well for moderate sized LPs)
            try:
                prob.solve(solver=cp.ECOS, max_iters=2000,
                           warm_start=True, verbose=False)
            except Exception:            # pragma: no cover
                HAVE_CVX = False
                # fall through to the fallback routine on the next iteration
                continue

            if g.value is None:          # pragma: no cover
                # Solver failed – switch to the fallback method
                HAVE_CVX = False
                continue

            # Extract the partner vector, enforce non‑negativity explicitly
            g_val = np.maximum(0.0, np.asarray(g.value, dtype=np.float64).ravel())
            sum_g = g_val.sum()
            if sum_g < 1e-12:            # safety – should not happen
                break

            # Normalise to the same ℓ₁‑norm as `f` (here the sum is 1)
            g_norm = g_val / sum_g

            # Frank–Wolfe blending
            f = (1.0 - t) * f + t * g_norm
            # Renormalise to combat numerical drift
            total = f.sum()
            if total > 0:
                f = f / total

            # evaluate the blended candidate
            score = float(evaluate_sequence(f.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = f.copy()
                print(f"Iter {it:4d}: improved score = {best_score:.8f}")

        else:
            # ----------------------------------------------------------------
            # 5b. Fallback local‑search step (identical to the reference code)
            # ----------------------------------------------------------------
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)

            cand = f.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])

            # keep numbers inside the evaluator's admissible range
            cand = np.clip(cand, 0.0, 1000.0)

            # renormalise to keep the ℓ₁‑norm = 1  (any positive scale works)
            s = cand.sum()
            if s > 0:
                cand = cand / s

            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = cand.copy()
                f = cand.copy()
                print(f"Iter {it:4d} (fallback): improved score = {best_score:.8f}")

        # Decay the mixing coefficient – smaller steps as we approach a plateau
        if it % 10 == 0 and t > 0.01:
            t *= 0.9

        it += 1

    # -----------------------------------------------------------------------
    # 6.  Return the best sequence discovered (as plain Python floats)
    # -----------------------------------------------------------------------
    return [float(x) for x in best]
