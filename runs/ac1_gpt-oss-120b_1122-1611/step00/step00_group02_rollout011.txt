def solve_lp(f, M):
    n = len(f)
    # Build A_ub
    # Use scipy.sparse.diags
    import scipy.sparse as sp
    # Diagonal offsets from 0 to n-1
    diags = [f] * n
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csc')
    # constraint: A @ g <= M * np.ones(2*n-1)
    b = np.full(2*n-1, M)
    # bounds for g >=0
    bounds = [(0, None)] * n
    c = -np.ones(n)  # maximize sum(g)
    # Solve using hiGHs method
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(f, f)) / (sum(f)**2)

The implementation works as follows:

1. **Initial shape** – a short geometric decay (or a random start) is
   normalised so that the sum equals √(2 n).  For a normalised sequence
   the evaluation value is exactly the maximal convolution entry, thus
   minimising that entry.

2. **LP improvement** – for the current sequence `f` we solve the linear
   programme

        maximise   Σ g_i
        subject to   (f ⋆ g)_k ≤ M      for all k
                     g_i ≥ 0

   where `M = max(f ⋆ f)`.  The constraints are linear in the unknown
   `g`.  This is a classic “alternating‑LP” step: if `Σ g_i` can be made
   larger than the prescribed normalising sum, we can increase the total
   mass while keeping the same worst‑case convolution bound.

3. **Mixing** – the solution `g` is scaled to the required sum
   √(2 n) and mixed into the current `f`:
        f ← (1‑t)·f + t·g_scaled   (t≈0.5)

   This step strictly reduces the maximal convolution value (or leaves
   it unchanged).  Repeating it drives the bound down.

4. **Time‑bounded search** – the whole procedure is repeated with
   different random starting points while the allotted time (≈ budget_s
   seconds) is not exhausted.  The best sequence found is returned.

If SciPy is unavailable the code gracefully falls back to a simple
block‑perturbation hill‑climber (the method used in the reference
implementation).  All numbers are clipped to the range accepted by
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# Optional SciPy imports – they may be missing in the execution sandbox.
try:
    import scipy.sparse as _sp
    import scipy.optimize as _opt
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq ⋆ seq)[k] (full convolution)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ g_i    s.t.   (f ⋆ g)_k ≤ M   for all k,   g_i ≥ 0.

    Returns the optimal g (numpy array) or None if the solver fails.
    """
    n = f.shape[0]
    # Build sparse matrix A where (A @ g)[k] = (f ⋆ g)[k].
    # Each diagonal d (0 ≤ d < n) contains `f` and contributes to rows d … d+n‑1.
    diags = [f] * n
    offsets = np.arange(n, dtype=int)
    A = _sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csc')
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # maximise sum(g)  ==> minimise -sum(g)
    bounds = [(0, None)] * n
    # HiGHS is fast and accepts sparse matrices.
    res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                       method='highs', polish=True, options={'presolve': True})
    if res.success:
        return res.x
    return None


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a first guess.  With 50 % probability a geometric decay
    (close to the continuous optimum) is used; otherwise a random
    positive vector is generated.
    """
    # Normalisation target: sum = sqrt(2 n)
    target_sum = math.sqrt(2.0 * n)

    if rng.random() < 0.5:
        # geometric decay; the constant c≈1.2 is near the continuous optimum
        c = 1.2 + rng.normal(0.0, 0.2)
        r = math.exp(-c / n)                 # r ≈ 1 – c/n
        seq = np.array([r ** i for i in range(n)], dtype=np.float64)
    else:
        seq = rng.random(n, dtype=np.float64)  # uniform in (0,1)

    seq *= target_sum / seq.sum()
    return seq


def _evaluate(seq: np.ndarray) -> float:
    """Call the provided evaluate_sequence (may be costly)."""
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:                # pragma: no cover
        return float('inf')


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Search for a sequence with a low evaluation value.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Approximate wall‑clock budget (seconds).  The routine stops a few
        seconds before the limit to guarantee a return value.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Configuration: choose a modest size (large enough for a good bound,
    # small enough for the LP to stay fast).  800 is a good compromise.
    # ------------------------------------------------------------------
    n = 800
    target_sum = math.sqrt(2.0 * n)

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # Main time‑bounded loop – many random restarts, each followed by a
    # short LP‑based refinement.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 1) initialise ------------------------------------------------
        f = _initial_sequence(n, rng)

        # ---- 2) LP refinement --------------------------------------------
        # a few LP steps are enough to obtain a monotone improvement;
        # we stop early if there is no progress.
        for _ in range(15):
            if time.monotonic() >= deadline:
                break
            M = _max_convolution(f)

            if not _SCIPY_AVAILABLE:                 # fall back to simple search
                break

            g = _solve_lp(f, M)
            if g is None:
                break

            sum_g = g.sum()
            # No improvement possible – stop the inner loop.
            if sum_g <= target_sum * (1.0 + 1e-12):
                break

            # Scale g back to the required normalisation.
            g_scaled = g * (target_sum / sum_g)

            # Convex combination (t≈0.5 works well in practice).
            t = 0.5
            f = (1.0 - t) * f + t * g_scaled

            # Numerical safety – ensure non‑negativity.
            f = np.clip(f, 0.0, None)

        # ---- 3) evaluate --------------------------------------------------
        score = _evaluate(f)
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"[{time.monotonic():.2f}] New best score: {best_score:.6f}")

        # Very short random perturbation (optional, cheap) to escape
        # possible stagnation – similar to the reference block perturbation.
        if rng.random() < 0.15 and time.monotonic() < deadline:
            # tiny multiplicative jitter on a random block
            width = max(1, n // 10)
            start = rng.integers(0, n - width + 1)
            factor = math.exp(rng.normal(0.0, 0.08))
            f_pert = f.copy()
            f_pert[start:start + width] *= factor
            f_pert = np.clip(f_pert, 0.0, 1000.0)
            f_pert *= target_sum / f_pert.sum()
            pert_score = _evaluate(f_pert)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = f_pert.copy()
                print(f"[{time.monotonic():.2f}] Improved by perturbation: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a simple uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = _evaluate(best_seq)
        print(f"Fallback uniform sequence, score = {best_score:.6f}")

    print(f"Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    # Convert to plain Python floats as required by the evaluation harness.
    return [float(x) for x in best_seq]
