Implementation of projection onto simplex:

```
def project_to_simplex(v):
    # v: np.ndarray (any real numbers)
    # return w with sum(w)=1, w>=0, and Euclidean distance minimal.
    # handle zero sum
    # algorithm from https://arxiv.org/pdf/1101.6081.pdf
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all nonpositive -> return uniform distribution?
        w = np.full_like(v, 1.0/n)
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # Ensure sum ~ 1 (due to rounding)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0/n)
    else:
        w = w / w_sum
    return w
```

"""
Propose a candidate sequence for the step‐function minimisation problem.

The algorithm is a hybrid of a simple stochastic block perturbation
and a small linear‑programming (LP) routine that attempts to increase the
total mass of the sequence while keeping the self‑convolution bounded.
The LP is built only from the (few) “active’’ convolution constraints
that are close to the current maximum, so it is fast even for a few‑
thousand coefficients.

The search works as follows
  * initialise a non‑negative vector (taken from the global
    ``height_sequence_1`` or a random Dirichlet if that is invalid);
  * repeatedly
        – compute the convolution and its maximum M,
        – collect the index set where the convolution is within a tiny
          fraction of M (the active set),
        – formulate the LP   maximise   Σ g_j   subject to   (f∗g)_k ≤ M
          for every active k  and   g_j ≥ 0,
        – if the LP succeeds, scale the solution to the same ℓ₁‑norm as
          the current vector and mix it with the current vector;
        – otherwise fall back to a cheap block‑wise random perturbation,
        – keep the best vector seen so far.
The whole routine stops after the allotted time budget (≈ 1000 s) and
returns the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def make_block_perturbation(base, rng):
    """
    Return a simple O(n) memory perturbation of ``base``:
    a contiguous block is multiplied by a log‑normal factor,
    optionally symmetrised, and finally clipped to the allowed range.
    """
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # width roughly between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def project_to_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0 , Σ w = 1 }.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w = w / w_sum
    return w


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Return a list of non‑negative heights with a low evaluation score.

    The function runs for at most ``budget_s`` seconds (a few seconds are
    subtracted for safety) and returns the best sequence it discovered.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise the starting sequence (global variable may be present)
    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        init = np.empty(0, dtype=np.float64)

    if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
        # fallback: random Dirichlet – guarantees a positive sum
        init = rng.random(512) + 0.01

    init = np.clip(init, 0.0, 1000.0)

    # normalise to ℓ₁ = 1  (the evaluation is scale‑invariant below the clip)
    if init.sum() == 0.0:
        init[:] = 1.0
    a = init / init.sum()
    n = a.size

    # store the overall best we have seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # ------------------------------------------------------------------
    # hyper‑parameters
    # ------------------------------------------------------------------
    mix_factor = 0.2          # how much of the LP solution is mixed in
    active_thr = 0.999        # keep constraints whose value ≥ M·thr
    max_active = 12           # cap on the number of active constraints
    iteration = 0

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # --- convolution of the current vector ---------------------------------
        conv = np.convolve(a, a)
        M = float(conv.max())

        # --- determine the “active’’ indices where the bound is almost tight ----
        active_idx = np.where(conv >= M * active_thr)[0]
        if active_idx.size > max_active:
            # keep the highest ones only
            active_idx = np.argsort(-conv)[:max_active]

        # Build the sparse A_ub matrix for the LP:
        #  (f * g)_k = Σ_j f_{k-j} * g_j  ≤  M   for each active k.
        rows, cols, data = [], [], []
        for row_id, k in enumerate(active_idx):
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1, dtype=np.int64)
            coeffs = a[k - js]                     # f_{k-j}
            rows.extend([row_id] * len(js))
            cols.extend(js.tolist())
            data.extend(coeffs.tolist())

        if rows:   # at least one active constraint – solve the LP
            A_ub = csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
            b_ub = np.full(len(active_idx), M, dtype=np.float64)

            # maximise Σ g_j  ⇔ minimise –Σ g_j
            c = -np.ones(n, dtype=np.float64)

            # Use HiGHS interior point (fast for moderate sized problems)
            lp_res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                             bounds=(0, None),
                             method='highs',
                             options={'presolve': True})

            if lp_res.success and lp_res.x.sum() > 0:
                g = lp_res.x
                # rescale g so that Σ g = Σ a   (the LP does not enforce this)
                scale = a.sum() / g.sum()
                g_scaled = g * scale
                # convex combination with the current vector
                candidate = (1.0 - mix_factor) * a + mix_factor * g_scaled
                candidate = np.clip(candidate, 0.0, 1000.0)
                # force exact ℓ₁ normalisation again (tiny numerical drift)
                if candidate.sum() == 0.0:
                    candidate = np.full_like(candidate, 1.0 / n)
                else:
                    candidate = candidate / candidate.sum()
            else:
                # LP failed – fall back to cheap block perturbation
                candidate = make_block_perturbation(a, rng)
                candidate = np.clip(candidate, 0.0, 1000.0)
                if candidate.sum() == 0.0:
                    candidate = np.full_like(candidate, 1.0 / n)
                else:
                    candidate = candidate / candidate.sum()
        else:
            # no active constraints (should not happen) – use a perturbation
            candidate = make_block_perturbation(a, rng)
            candidate = np.clip(candidate, 0.0, 1000.0)
            if candidate.sum() == 0.0:
                candidate = np.full_like(candidate, 1.0 / n)
            else:
                candidate = candidate / candidate.sum()

        # ------------------------------------------------------------------
        # evaluate the new candidate (the official evaluator does the clipping)
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            best_score = score
            print(f"Iter {iteration:5d}: new best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # replace the working vector – a small amount of random “re‑shake’’
        # every now and then helps to escape plateaus.
        # ------------------------------------------------------------------
        a = candidate
        if rng.random() < 0.03:
            # inject a fresh random perturbation of the current best
            a = make_block_perturbation(best, rng)
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a.sum()

    # final return – the harness expects a plain Python list of floats
    return [float(x) for x in best]
