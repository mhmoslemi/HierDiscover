    def try_seq(seq):
        nonlocal best_seq, best_score
        # Clip to [0,1000] and convert to list for evaluation
        score = evaluate_sequence(seq.tolist())
        if np.isfinite(score) and score < best_score:

            best_seq = seq.copy()
    # Family: exponential
    for r in np.linspace(0.998, 0.99999, 200):
        raw = r ** np.arange(n)
        # scale to 1000
        raw = raw * (1000.0 / raw.max())
        try_seq(raw)
    # Family: exponential + uniform background
    for r in np.linspace(0.998, 0.99999, 100):
        for eps in [0.0, 0.1, 1.0, 5.0]:
            raw = r ** np.arange(n) + eps
            raw = raw * (1000.0 / raw.max())
            try_seq(raw)
    # Family: sinusoidal
    x = np.arange(n)
    for p in [1.0, 2.0, 3.0]:
        raw = np.sin(np.pi * (x+1)/(n+1)) ** p
        # to avoid zero values, maybe add small epsilon
        raw = raw + 1e-12
        raw = raw * (1000.0 / raw.max())
        try_seq(raw)
    # Family: polynomial (i/n)^alpha
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        raw = (x/(n-1)) ** alpha
        raw = raw * (1000.0 / raw.max())
        try_seq(raw)
    # Family: inverse power, (i+1)^{-beta}
    for beta in [0.5, 1.0, 1.5, 2.0]:
        raw = (x+1) ** (-beta)
        raw = raw * (1000.0 / raw.max())
        try_seq(raw)
    # Family: arithmetic progression (linear) (maybe not good)
    raw = x.astype(np.float64)
    raw = raw * (1000.0 / raw.max())
    try_seq(raw)
    # Now local refinement: use hill climbing from best_seq
    rng = np.random.default_rng(12345)
    deadline = start + 950.0  # seconds
    parent = best_seq.copy()
    while time.monotonic() < deadline:
        # block perturbation similar to earlier but smaller magnitude for fine tuning
        candidate = make_block_perturbation(parent, rng)
        # Maybe also random scaling of entire sequence
        if rng.random() < 0.1:
            factor = np.exp(rng.normal(0, 0.02))

        try_seq(candidate)

            try_seq(cand2)

"""
Propose a candidate sequence minimizing the objective.
The search proceeds in two phases:
1. Coarse parametric search over simple families (exponential, power‑exponential, sinusoid, polynomial)
   to obtain a good initial sequence.
2. Stochastic local refinement using block perturbations (multiplicative noise) and occasional
   global scaling.  The best sequence found before the time budget expires is returned.
The method is deterministic given the fixed RNG seeds, but explores enough variety
to beat the uniform benchmark (score 2.0) and aims for the target < 1.5025.
"""

import numpy as np, time, math, random

def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width roughly between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal factor
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # with some probability mirror to keep symmetry
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # clip to the limits used by ``evaluate_sequence``
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def _evaluate(seq):
    """Wrapper for evaluate_sequence that tolerates Python lists/arrays."""
    # the external evaluate_sequence is already imported in the execution environment
    return float(evaluate_sequence(seq.tolist()))

    Returns a list of floats (the best sequence found before the deadline).
    """

    # --------------------------------------------------------------------

        n = int(len(height_sequence_1))

    # --------------------------------------------------------------------

    best_score = _evaluate(best_seq)

    def maybe_update(candidate):
        nonlocal best_seq, best_score
        sc = _evaluate(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            # occasional progress output
            print(f"New best score {best_score:.6f}")

    # --------------------------------------------------------------------
    # 1. Coarse parametric exploration
    # --------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    N = n - 1
    # families to explore -------------------------------------------------
    # --------------------------------------------------------------
    # (a) Pure exponential a_i = r**i   (r = exp(-c/N))
    # --------------------------------------------------------------
    c_vals = np.linspace(0.5, 3.0, 26)   # 0.5 … 3.0
    for c in c_vals:
        r = math.exp(-c / N)
        seq = r ** idx
        # normalise – maximum is 1, no clipping needed
        maybe_update(seq)

        # also try a tiny uniform background (helps a little in some cases)
        for eps in (0.0, 0.02, 0.05, 0.1):
            cand = seq + eps
            # normalise to keep the peak at 1 before scaling
            cand = cand / (cand.max())
            maybe_update(cand)

    # --------------------------------------------------------------
    # (b) Power‑exponential a_i = (i+1)**α * r**i
    # --------------------------------------------------------------
    alphas = [-1.0, -0.5, 0.0, 0.5, 1.0]
    for alpha in alphas:
        for c in np.linspace(0.8, 2.2, 15):

            # avoid 0**negative – start from 1

            # shift/scale to keep everything non‑negative and bounded

            if seq.max() == 0:

            seq = seq / seq.max()          # normalise peak to 1
            maybe_update(seq)

    # --------------------------------------------------------------
    # (c) Simple sinusoidal window  a_i = sin(π * (i+1)/(n+1))
    # --------------------------------------------------------------
    sin_seq = np.sin(np.pi * (idx + 1.0) / (n + 1.0))
    sin_seq = sin_seq / sin_seq.max()

    maybe_update(hann_seq)

    # (d) Polynomial decay a_i = (1 - i/N)^p
    # --------------------------------------------------------------

        maybe_update(seq)

    # --------------------------------------------------------------
    for beta in (0.5, 1.0, 1.5, 2.0):
        seq = 1.0 / ((idx + 1.0) ** beta)
        seq = seq / seq.max()
        maybe_update(seq)

"""
Propose a candidate sequence minimizing the evaluation metric.

The routine works in two stages:

1. **Parametric sweep** – simple analytic families (exponential,
   power‑exponential, sinusoidal, polynomial, inverse‑power) are sampled.
   Each candidate is normalised so its maximum is 1 (scaling does not affect the
   score because the objective is homogeneous).  The best candidate from this
   coarse sweep becomes the starting point for the next stage.

2. **Stochastic refinement** – a block‑perturbation operator (multiplicative
   log‑normal noise on a random contiguous segment) is repeatedly applied.
   Small global scalings and per‑element log‑normal jitter are also tried.
   Accepted candidates replace the current best if they improve the score.
   The loop runs until the time budget (≈ 1000 s) expires.

The algorithm is deterministic for a given seed, uses only NumPy and the
provided `evaluate_sequence`, and is designed to beat the uniform benchmark
(score 2.0) and aim for the target < 1.5025.
"""

import numpy as np
import time
import math

def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # choose a block size between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal factor on the block
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # occasionally enforce a mild symmetry (helps stability)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def _evaluate(seq):
    """Convenient wrapper around the external evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function sequence with low evaluation score.
    Returns a Python list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # spare a few seconds

    # --------------------------------------------------------------------
    # Problem size – fall back to 4806 if the global variable is missing
    # --------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 4806
        base_seq = np.ones(n, dtype=np.float64)

    # --------------------------------------------------------------------
    # Keep track of the best candidate seen so far
    # --------------------------------------------------------------------
    best_seq = np.clip(base_seq, 0.0, 1000.0)
    best_score = _evaluate(best_seq)

    def maybe_update(candidate):
        nonlocal best_seq, best_score
        sc = _evaluate(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"New best score {best_score:.6f}")

    # --------------------------------------------------------------------
    # 1) Coarse parametric exploration
    # --------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    N = n - 1

    # --- pure exponential a_i = r**i ------------------------------------
    for c in np.linspace(0.5, 3.0, 26):          # 0.5 … 3.0
        r = math.exp(-c / N)
        seq = r ** idx
        maybe_update(seq)

        # tiny uniform background may help in some regimes
        for eps in (0.0, 0.02, 0.05, 0.1):
            cand = seq + eps
            cand = cand / cand.max()          # renormalise peak to 1
            maybe_update(cand)

    # --- power‑exponential a_i = (i+1)**α * r**i -----------------------
    alphas = [-1.0, -0.5, 0.0, 0.5, 1.0]
    for alpha in alphas:
        for c in np.linspace(0.8, 2.2, 15):
            r = math.exp(-c / N)
            seq = (idx + 1.0) ** alpha * (r ** idx)
            seq = np.clip(seq, 0.0, None)
            if seq.max() == 0:
                continue
            seq = seq / seq.max()
            maybe_update(seq)

    # --- simple sinusoidal window ----------------------------------------
    sin_seq = np.sin(np.pi * (idx + 1.0) / (n + 1.0))
    maybe_update(sin_seq / sin_seq.max())

    # --- Hann window ------------------------------------------------------
    hann_seq = 0.5 * (1 - np.cos(2.0 * np.pi * idx / N))
    maybe_update(hann_seq / hann_seq.max())

    # --- polynomial decay a_i = (1 - i/N)^p -------------------------------
    for p in (0.5, 1.0, 2.0, 4.0):
        seq = (1.0 - idx / N) ** p
        maybe_update(seq / seq.max())

    # --- inverse power law a_i = 1/(i+1)^β -------------------------------
    for beta in (0.5, 1.0, 1.5, 2.0):
        seq = 1.0 / ((idx + 1.0) ** beta)
        maybe_update(seq / seq.max())

    # --------------------------------------------------------------------
    # 2) Stochastic local refinement using block perturbations
    # --------------------------------------------------------------------
    parent = best_seq.copy()
    while time.monotonic() < deadline:
        # a) multiplicative block perturbation
        cand = make_block_perturbation(parent, rng)
        maybe_update(cand)

        # b) occasional small global scaling
        if rng.random() < 0.07:

            cand2 = np.clip(parent * factor, 0.0, 1000.0)
            maybe_update(cand2)

        if rng.random() < 0.03:

            cand3 = np.clip(parent * noise, 0.0, 1000.0)
            maybe_update(cand3)

            st = int(rng.integers(0, max(1, n - w + 1)))
            en = min(n, st + w)
            parent = best_seq.copy()
            parent[st:en] *= float(np.exp(rng.normal(0.0, 0.25)))
            np.clip(parent, 0.0, 1000.0, out=parent)

        # simple momentum towards the best known point
        parent = 0.9 * parent + 0.1 * best_seq

    # --------------------------------------------------------------------
    # Return the best sequence (clipped to the allowed range)
    # --------------------------------------------------------------------
    print(f"Search completed – best score {best_score:.6f}")
    final = np.clip(best_seq, 0.0, 1000.0)
    return final.tolist()
