def targeted_move(seq, rng):
    n = seq.size
    # compute convolution
    conv = np.convolve(seq, seq)
    max_idx = int(np.argmax(conv))
    # contributions
    contrib = np.zeros(n)
    for i in range(n):
        j = max_idx - i
        if 0 <= j < n:
            contrib[i] = seq[i] * seq[j]
    # choose i to decrease: index weighted by contribution (prefer high)
    if contrib.sum() == 0:
        # fallback random
        i = rng.integers(0, n)
    else:
        # probabilities proportional to contribution (plus epsilon)
        probs = contrib / contrib.sum()
        i = rng.choice(np.arange(n), p=probs)
    # choose j to increase: prefer low contribution, also not same as i.
    low_mask = np.arange(n) != i
    # compute distinct contributions for other indices
    contrib_other = contrib.copy()
    contrib_other[i] = np.inf  # exclude
    # pick j with minimal contribution
    # Could randomly sample among few lowest.
    # Get array sorted
    idxs = np.argsort(contrib_other)
    # pick j among first few (e.g., first 5)
    candidates = idxs[:min(10, len(idxs))]
    j = rng.choice(candidates)
    # Determine delta as a fraction of a_i, not too large.
    delta = seq[i] * rng.uniform(0.05, 0.2)  # move 5-20% of its mass
    # apply move
    new_seq = seq.copy()
    new_seq[i] = max(0.0, new_seq[i] - delta)
    new_seq[j] = new_seq[j] + delta
    return new_seq

from scipy.optimize import linprog

def solve_lp_for_g(f, M):
    n = len(f)
    L = 2*n - 1
    # Build A_ub as dense 2D array maybe using np.zeros((L, n))
    rows = np.arange(L)[:, None] - np.arange(n)[None, :]  # compute k - j
    # rows is shape L x n
    # we need entries where 0 <= rows < n, then coefficient = f[rows]
    mask = (rows >= 0) & (rows < n)
    A = np.zeros((L, n), dtype=np.float64)
    A[mask] = f[rows[mask]]
    b = np.full(L, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
```

def build_constraint_matrix(f):
    n = len(f)
    L = 2*n -1
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    idx = 0
    for d in range(n):
        coeff = f[d]
        # For each j in 0..n-1: row = j + d, col=j
        rows[idx:idx+n] = np.arange(d, d+n)
        cols[idx:idx+n] = np.arange(n)
        data[idx:idx+n] = coeff
        idx += n
    A = sp.csr_matrix((data, (rows, cols)), shape=(L, n))
    return A
```

def targeted_move_pair(seq, rng):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_idx = int(np.argmax(conv))
    # choose i uniformly among indices that contribute.
    i_candidates = np.arange(max(0, max_idx - (n-1)), min(max_idx, n-1) + 1)
    i = rng.choice(i_candidates)
    j = max_idx - i
    # choose removal amount:
    delta_i = seq[i] * rng.uniform(0.02, 0.1)

"""
Hill‑climbing optimizer for the self‑convolution bound.

The algorithm starts from a feasible seed (the provided `height_sequence_1`
or a uniform vector) and repeatedly performs two kinds of moves:

* **Block perturbations** – multiplicative scaling of a random contiguous
  block (the same heuristic used in the baseline).
* **Targeted pair moves** – the current convolution is inspected,
  the index `k*` where the convolution attains its maximum is found,
  a contributing pair `(i, j = k*–i)` is selected and a small amount of
  mass is transferred from that pair to two unrelated positions.
  This directly attacks the peak of the convolution.

All candidates are renormalised to unit ℓ₁‑norm (the evaluation function
is scale‑invariant) and clipped to the allowed range [0, 1000].  The best
sequence found before the time budget expires is returned.

The search runs for the allotted wall‑clock time (`budget_s` seconds) and
periodically prints progress information.
"""

import numpy as np
import time
from scipy.optimize import linprog  # optional – not used in the final loop

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply the original block‑multiplication perturbation."""
    candidate = base.copy()
    n = candidate.size
    # block width: between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # respect the global bounds enforced later
    return candidate


def _targeted_pair_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the current maximum of the self‑convolution by moving a small
    amount of mass away from the pair of indices that contribute most
    to the peak.
    """
    n = seq.size
    # full convolution (linear, not cyclic)
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # all i that participate in this sum
    i_low = max(0, k_max - (n - 1))
    i_high = min(k_max, n - 1)
    i_vals = np.arange(i_low, i_high + 1, dtype=int)
    j_vals = k_max - i_vals

    # contribution of each pair (ordered, one per i)
    contrib = seq[i_vals] * seq[j_vals]

    # pick the pair with the largest contribution
    idx = int(np.argmax(contrib))
    i = int(i_vals[idx])
    j = int(j_vals[idx])

    # if either entry is essentially zero, bail out
    if seq[i] < 1e-12 or seq[j] < 1e-12:
        return seq.copy()

    # amount to withdraw – a modest fraction of the smaller of the two
    frac = rng.uniform(0.05, 0.20)
    delta = min(seq[i], seq[j]) * frac

    # pick two destination indices far from (i, j)
    possible = np.arange(n)
    mask = (possible != i) & (possible != j)
    if mask.sum() < 2:
        # fallback: just return the original sequence
        return seq.copy()
    dest = rng.choice(possible[mask], size=2, replace=False)

    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] -= delta
    # distribute the withdrawn mass equally to the two destinations
    new_seq[dest[0]] += delta
    new_seq[dest[1]] += delta

    # Numerical safety
    new_seq = np.clip(new_seq, 0.0, None)
    return new_seq


def _normalise(v: np.ndarray) -> np.ndarray:
    """Scale the vector to have ℓ₁‑norm equal to 1 (avoids scaling issues)."""
    total = v.sum()
    if total <= 0:
        # fallback to a uniform distribution
        return np.ones_like(v) / v.size
    return v / total

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) improves
    on the current bound.  The function runs for at most `budget_s`
    seconds (the last 3 s are reserved for a graceful shutdown) and
    uses a combination of block perturbations and targeted pair moves.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise the starting vector
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is expected to be defined globally
        start_vec = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        start_vec = None

    if (
        start_vec is None
        or start_vec.ndim != 1
        or start_vec.size == 0
        or not np.isfinite(start_vec).all()
    ):
        # fall back to a modest uniform vector (length 256)
        start_vec = np.ones(256, dtype=np.float64)

    # enforce the problem's clipping limits and normalise
    start_vec = np.clip(start_vec, 0.0, 1000.0)
    start_vec = _normalise(start_vec)

    best = start_vec.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] score = {best_score:.6f}   (length={best.size})")

    # use the best vector as the current state for hill‑climbing
    current = best.copy()
    iteration = 0
    # statistics for reporting
    last_report = time.monotonic()

    while time.monotonic() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # Choose a move type: block perturbation (exploration) or
        # targeted pair move (exploitation).  The majority of steps are
        # targeted because they directly aim at the current peak.
        # ------------------------------------------------------------------
        if rng.random() < 0.35:
            # block perturbation – large, cheap random change
            cand = _make_block_perturbation(current, rng)
        else:
            # targeted pair move – tries to shrink the peak of the convolution
            cand = _targeted_pair_move(current, rng)

        # post‑process candidate: clip, normalise (preserves scale invariance)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalise(cand)

        # evaluate – the heavy part (convolution) is performed inside
        # evaluate_sequence; we keep the raw list to avoid extra copies
        score = float(evaluate_sequence(cand.tolist()))

        # accept only if we improve (simple greedy hill‑climbing)
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            current = cand  # continue from the improved point
            print(f"[{iteration:06d}] new best = {best_score:.6f}")

        else:
            # occasionally replace the current point with the best found so far
            # to avoid drifting away after many rejected moves
            if rng.random() < 0.02:
                current = best.copy()

        # periodic progress report (every few seconds)
        if time.monotonic() - last_report > 4.0:
            elapsed = time.monotonic() - (deadline - budget_s + 3.0)
            print(
                f"[{elapsed:6.1f}s] iter={iteration}, best={best_score:.6f}"
            )
            last_report = time.monotonic()

    print(f"[done] best score after {iteration} iterations: {best_score:.6f}")
    # return a plain Python list of floats
    return [float(x) for x in best]
