def hill_climb(a, rng, time_end):
    best = a.copy()
    best_score = evaluate
    cur = a.copy()
    cur_score = best_score
    temperature = init_T
    iter = 0

    while time < deadline:
        # compute conv, idx_max and gradient contributions
        conv = np.convolve(cur, cur)
        idx_max = np.argmax(conv)
        # gradient contributions
        g = np.zeros(len(cur))
        for i in range(len(cur)):
            j = idx_max - i
            if 0 <= j < len(cur):
                g[i] = cur[j]  # note omitted factor 2
        # source selection:
        src_candidates = np.where(cur > 1e-12)[0]
        if src_candidates.size == 0:
            break
        # source weighted by g
        if g.sum() > 0:
            probs = g[src_candidates] / g[src_candidates].sum()
            src = rng.choice(src_candidates, p=probs)
        else:
            src = rng.choice(src_candidates)
        # target selection: want low g
        target_candidates = np.arange(len(cur))
        # remove src
        # compute inverse weights: w = 1/(g+eps)
        inv_weights = 1.0/(g+1e-12)
        inv_weights[src] = 0
        probs_t = inv_weights / inv_weights.sum()
        tgt = rng.choice(len(cur), p=probs_t)
        # delta
        max_delta = min(cur[src], step_frac * cur[src] + 1e-5, 1000 - cur[tgt])
        if max_delta <= 0:
            continue
        delta = rng.uniform(0, max_delta)
        new_cur = cur.copy()
        new_cur[src] -= delta
        new_cur[tgt] += delta
        # No scaling needed; sum preserved.
        # Evaluate
        new_score = float(evaluate_sequence(new_cur.tolist()))
        if np.isfinite(new_score):
            # accept criteria
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score)/temperature):
                cur = new_cur
                cur_score = new_score
                if new_score < best_score:
                    best = new_cur.copy()
                    best_score = new_score
                    print(f"Iter {iter}, new best {best_score}")
        # temperature schedule
        temperature = max(temperature * 0.99999, 1e-6)
        # optionally if time left we do large perturbation occasionally
        if iter % 5000 == 0 and iter>0:
            # large block perturbation using previous make_block_perturbation function
            block = make_block_perturbation(best, rng)
            # ensure sum normalized again:
            block = block * (np.sqrt(2*len(block)) / np.sum(block))
            block_score = float(evaluate_sequence(block.tolist()))
            if block_score < best_score:
                best = block.copy()
                best_score = block_score
                cur = block.copy()
                cur_score = block_score
                print(f"Large perturbation improved best to {best_score}")
        iter += 1

"""
Simulated‑annealing + gradient‑guided mass transfers.

The score to minimise is
    2·n·max(conv(a,a)) / (sum(a))²

Because the score is invariant under global scaling, we keep the
ℓ₁‑norm fixed to √(2 n).  Starting from the provided `height_sequence_1`
(or a uniform fallback) the algorithm repeatedly

1. computes the autocorrelation,
2. finds the index where the convolution attains its maximum,
3. builds a simple “gradient” – the contribution of each component to
   that peak,
4. moves a small amount of mass from a high‑gradient position to a
   low‑gradient one,
5. accepts the move if it improves the score or with a temperature‑dependent
   probability (simulated annealing).

Occasionally we try larger block perturbations (the previous heuristic)
and pure random restarts, keeping the best solution found before the
deadline expires.  The routine returns the best sequence as a list of
Python floats.
"""

import time
import numpy as np

def _compute_score(seq: np.ndarray) -> float:
    """Return the same value as evaluate_sequence for a valid, clipped seq."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _gradient_for_max(seq: np.ndarray, idx_max: int) -> np.ndarray:
    """
    For a given index `idx_max` of the maximal convolution entry,
    compute a simple gradient g_i = a_{idx_max-i} (0 if out of range).
    """
    n = seq.shape[0]
    g = np.zeros_like(seq)
    i_min = max(0, idx_max - (n - 1))
    i_max_excl = min(n, idx_max + 1)
    i_range = np.arange(i_min, i_max_excl)
    j_range = idx_max - i_range
    g[i_range] = seq[j_range]
    return g


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Return one O(n)‑memory non‑negative perturbation of ``base``.
    This function is the same as the one used in the reference code.
    """
    candidate = base.astype(np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.

    The function runs until `budget_s` seconds have elapsed (minus a tiny
    safety margin) and returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise the parent sequence (fallback to uniform if needed)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        base = None

    if (
        base is None
        or base.ndim != 1
        or base.size == 0
        or not np.isfinite(base).all()
    ):
        n = 256
        base = np.ones(n, dtype=np.float64)
    else:
        n = base.shape[0]

    # clip to the allowed interval and normalise ℓ₁‑norm to √(2 n)
    base = np.clip(base, 0.0, 1000.0)
    target_sum = np.sqrt(2.0 * n)
    cur_sum = base.sum()
    if cur_sum == 0.0:
        base = np.full_like(base, target_sum / n)
    else:
        base = base * (target_sum / cur_sum)
    base = np.clip(base, 0.0, 1000.0)

    best = base.copy()
    best_score = _compute_score(best)
    cur = base.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    temperature = 1.0                # initial temperature
    decay = 0.99995                  # exponential cooling factor
    min_temp = 1e-8
    step_frac = 0.25                 # fraction of source mass that can be moved
    iteration = 0

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --- autocorrelation and gradient ----------------------------------------------------
        conv = np.convolve(cur, cur)
        idx_max = int(np.argmax(conv))

        nonzero = np.where(cur > 1e-12)[0]

            if new_score < cur_score:

            if accept:

                cur_score = new_score
                if new_score < best_score:
                    best = new_seq.copy()
                    best_score = new_score
                    print(f"[iter {iteration}] new best score = {best_score:.6f}")

        # --- temperature schedule -------------------------------------------------------------
        temperature = max(temperature * decay, min_temp)

        # --- occasional large perturbation / random restart ---------------------------------
        if iteration % 5000 == 0 and iteration > 0:

    # Return a plain Python list as required by the evaluation harness.
