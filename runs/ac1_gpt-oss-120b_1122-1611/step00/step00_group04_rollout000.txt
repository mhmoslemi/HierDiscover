"""
Simulated annealing / multi-start search for minimizing evaluate_sequence.

The algorithm:
- Start from a valid reference sequence (height_sequence_1) if possible,
  otherwise from a uniform sequence of length 256.
- Compute its score.
- Perform a time‑bounded loop producing random block‑wise multiplicative
  perturbations (the same strategy as before) but also accept non‑improving
  moves with probability exp(-Δ/T) to escape local minima.
- Temperature T decays linearly from an initial value to a small final value.
- Keep the best sequence seen.
- Return the best at the deadline.
"""

import numpy as np
import time

def make_block_perturbation(base, rng, max_width_factor=12):
    """Return a non‑negative block perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # Choose a block width between n//200 and n//max_width_factor (both at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // max_width_factor)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal factor
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # occasional symmetrisation
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # Clip to the evaluator’s admissible range
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Search for a coefficient sequence that minimizes ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Wall‑clock time budget (seconds). The function stops a few seconds
        before the limit to guarantee a return.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # -----------------------------------------------------------------
    # 1. Initialise a feasible starting point.
    # -----------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the evaluation environment.
        start = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        start = None

    if start is None or start.ndim != 1 or start.size == 0 or not np.isfinite(start).all():
        # Fallback: a uniform vector of moderate length.
        length = 256
        start = np.full(length, 1.0, dtype=np.float64)
        print("[init] Using uniform fallback of length", length)
    else:
        print("[init] Using provided height_sequence_1 of length", start.size)

    # Clamp to the allowed interval – this mirrors the security checks.
    np.clip(start, 0.0, 1000.0, out=start)

    best = start.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print("[init] Initial score =", best_score)

    # -----------------------------------------------------------------
    # 2. Simulated‑annealing loop.
    # -----------------------------------------------------------------
    # Temperature schedule.
    T0 = 1.0      # initial temperature (acceptance probability scale)
    Tf = 1e-4     # final temperature
    # Linear cooling: T(t) = T0 + (Tf - T0) * (elapsed / budget)
    # We will recompute T each iteration from the elapsed fraction.
    iteration = 0
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        # Fraction of the allotted time that has elapsed (0‑1)
        frac = (now - (deadline - float(budget_s) + 1.0)) / float(budget_s - 1.0)
        # Protect against division by zero / negative values.
        if frac < 0.0:
            frac = 0.0
        elif frac > 1.0:
            frac = 1.0
        T = T0 + (Tf - T0) * frac

        # Produce a candidate perturbation.
        cand = make_block_perturbation(best, rng)
        cand_score = float(evaluate_sequence(cand.tolist()))

        # Accept if better, otherwise accept with probability exp(-Δ/T)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[{iteration:06d}] Improved score: {best_score:.6f}")
        else:
            # Δ = cand_score - best_score > 0
            delta = cand_score - best_score
            # Metropolis criterion
            if T > 0.0 and rng.random() < np.exp(-delta / T):
                best = cand
                best_score = cand_score
                print(f"[{iteration:06d}] Accepted worse score: {best_score:.6f} (Δ={delta:.6f}, T={T:.6f})")

        iteration += 1

    print("[final] Best score after", iteration, "iterations:", best_score)
    # Convert to plain Python list of floats as required.
    return [float(x) for x in best]

def make_block_perturbation(base, rng, max_width_factor=12):
    """
    Generate a cheap O(n) non‑negative perturbation of ``base``.

    A random contiguous block is multiplied by a log‑normal factor.
    With probability 0.35 the whole vector is symmetrised.

    print("[init] Initial score =", best_score)

    # -----------------------------------------------------------------
    # 2. Simulated‑annealing loop (time‑bounded).
    # -----------------------------------------------------------------
    # Temperature schedule: linear cooling from T0 to Tf.
    T0 = 1.0          # initial temperature
    Tf = 1e-4         # final temperature (near‑zero)
    iteration = 0

    # Pre‑compute total allotted time for convenient fraction calculation.
    total_time = float(budget_s) - 1.0

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # Fraction of time elapsed (0 → 1). Guard against numerical glitches.
        elapsed = now - (deadline - total_time)
        frac = min(max(elapsed / total_time, 0.0), 1.0)

        # Linear interpolation of temperature.
        T = T0 + (Tf - T0) * frac

        # Propose a new candidate by perturbing the current best.
        cand = make_block_perturbation(best, rng)
        cand_score = float(evaluate_sequence(cand.tolist()))

        if cand_score < best_score:
            # Strict improvement.
            best = cand
            best_score = cand_score
            print(f"[{iteration:07d}] Improved score: {best_score:.6f}")
        else:
            # Metropolis acceptance of a worse move.
            delta = cand_score - best_score
            if T > 0.0 and rng.random() < np.exp(-delta / T):
                best = cand
                best_score = cand_score
                print(
                    f"[{iteration:07d}] Accepted worse score: {best_score:.6f} "
                    f"(Δ={delta:.6f}, T={T:.6f})"
                )

        iteration += 1

    print("[final] Best score after", iteration, "iterations:", best_score)
    return [float(x) for x in best]
