    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    n = 256 maybe; set if height_sequence_1 length > n use its length? Use maybe 256 to keep LP small.

    # Initialize f
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 10:
        # choose random
        f = rng.random(256)
    else:
        # resize or use a resized version?
        # Use first n elements?
        n = parent.size
        f = parent.copy()

    # Normalize sum sqrt(2n)
    target_sum = np.sqrt(2 * n)
    f = f / (np.sum(f) + 1e-12) * target_sum

    best_seq = None
    best_score = float('inf')
    # compute initial score
    seq_to_eval = [float(x) for x in (f * (1000.0 / np.max(f)))]
    best_score = evaluate_sequence(seq_to_eval)
    best_seq = seq_to_eval

    t = 0.1   # step size
    iteration = 0
    while time.monotonic() < deadline and iteration < 30:
        # compute convolution and M
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # Build LP constraints
        # build matrix A (2n-1 x n)
        # Use for each shift k, row = f_shifted such that
        # row[j] = f[k - j] if 0 <= k - j < n else 0
        # Could precompute using Toeplitz
        rows = 2 * n - 1
        A_ub = np.zeros((rows, n))
        for k in range(rows):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # i corresponds to f_i, g_{k - i}
            # loop over i
            # row[j] = f_i where j = k - i
            # Equivalent: for i in i_start..i_end:
            j = k - np.arange(i_start, i_end + 1)
            A_ub[k, j] = f[i_start:i_end+1]
        b_ub = np.full(rows, M)

        # Objective: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)

        # bounds: g_j >= 0
        bounds = [(0, None)] * n

        # solve LP
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fall back may break
            break

        g = res.x
        sum_g = np.sum(g)
        if sum_g == 0:
            break

        # rescale g to have same sum sqrt(2n)
        g = g * (target_sum / sum_g)

        # update f
        f = (1 - t) * f + t * g

        # evaluate new sequence after scaling to [0,1000]
        # scaling to max 1000
        max_f = np.max(f)
        if max_f <= 0:
            break
        scale = 1000.0 / max_f
        seq = (f * scale).tolist()
        score = evaluate_sequence(seq)
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_seq = seq

        iteration += 1

    # Return best_seq, else fallback uniform maybe
    if best_seq is None:
        # fallback uniform
        n = 256
        seq = [1000.0] * n
    else:
        # Ensure non-negative and clip
        best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return best_seq

Propose a candidate step‑function using the iterative linear‑programming scheme
described in the statement.  Starting from a random (or fallback) sequence we
normalise its ℓ¹‑norm to sqrt(2·n).  At each iteration we:

1. Compute the current maximum of the self‑convolution M = max(a*a).
2. Solve the linear programme
        maximise   Σ g_j
        subject to  a * g ≤ M   (every shift of the convolution)
                    g_j ≥ 0.
   For a fixed a the constraints are linear in g.
3. Rescale g to the same ℓ¹‑norm as a and move a a little towards g:
        a ← (1‑t)·a + t·g   (t is a small step size).

The process reduces the maximal convolution value while keeping the total
mass (the denominator of the score) large, driving the evaluation score
2·n·max(a*a)/‖a‖₁² down.  After the optimisation we scale the sequence so
that its largest entry equals 1000 (the evaluator caps values at 1000) – this
does not change the score because the objective is homogeneous of degree 0.

The routine runs until the time budget expires (≈ 1000 s) or no further
improvement is observed.  If SciPy is unavailable a simple random block
perturbation fallback is used.
"""

import time
import numpy as np

# Try to import the LP solver; if unavailable we fall back to a simple heuristic.

def _build_ub_matrix(f):
    """
    Build the matrix A_ub for the constraints  (f * g)[k] ≤ M.
    Row k contains the coefficients f_i for all i such that g_{k‑i} appears
    in the convolution sum.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # Vectorised construction
    ks = np.arange(rows)[:, None]                 # shape (rows,1)
    is_ = np.arange(n)                           # shape (n,)
    # Compute indices of the opponent variable: j = k - i
    js = ks - is_                                 # shape (rows,n)

    # Valid positions where 0 ≤ j < n
    valid = (js >= 0) & (js < n)

    # For valid entries put f_i
    A[valid] = f[is_][np.newaxis, :][valid]

    return A
