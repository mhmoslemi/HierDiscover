def solve_lp(f, M):
    n = len(f)
    # Build convolution matrix T where T[k,i] = f[k-i] if 0 <= k-i < n else 0.
    T = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k) + 1
        # for i in [i_start, i_end):
        for i in range(i_start, i_end):
            T[k,i] = f[k-i]
    # Define variable g
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
    return np.array(g.value).flatten()

"""
Propose a sequence that tries to beat the baseline score (≈2.0).

Algorithm
---------
1.  Initialise a moderate‑length (≈200) non‑negative sequence.
    The sequence is normalised so that its ℓ₁‑norm equals
    √(2 n) – the same normalisation used by the reference method.

2.  Iteratively improve the sequence by solving a linear programme.
    For the current sequence `f` we compute
        M = max(conv(f, f))                # the current worst convolution value
    and then solve
        maximise   Σ g_i
        subject to  conv(f, g) ≤ M   (component‑wise)
                    g_i ≥ 0 .
    The convolution constraint is linear in `g`, so a LP can be used.
    The optimal `g` tends to spread the mass more uniformly across the
    convolution lags and therefore reduces the peak `M`.

3.  Rescale the LP solution `g` to the required normalisation
       Σ g_i = √(2 n) .
    Mix it with the current sequence:
        f ← (1‑t)·f + t·g   (t = 0.2)
    The normalisation guarantees that the denominator of the
    evaluation function stays constant while the numerator (the peak
    convolution) is non‑increasing.

4.  Keep the best sequence seen (according to `evaluate_sequence`) and
    stop when the allotted time is exhausted.

If cvxpy is unavailable or the LP fails, the function falls back to the
simple block‑perturbation local search used in the original baseline.

The whole search runs within the provided `budget_s` (≈1100 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: build the convolution matrix T such that (f * g) = T @ g
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.size
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    # each entry f[i] contributes to rows i … i+n‑1
    for i in range(n):
        T[i : i + n, i] = f[i]
    return T


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence().  The routine uses an LP‑based iterative refinement
    (as described in the statement) and falls back to a simple random
    block perturbation if the LP solver cannot be used.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1.  Initialise a base sequence.
    # Use a modest size (200‑600) – large enough to give a good bound,
    # small enough for the LP to finish quickly.
    n = 256
    # start from a uniform vector, then add a tiny random perturbation
    f = np.ones(n, dtype=np.float64)
    f += rng.normal(scale=0.01, size=n)          # break strict symmetry
    f = np.clip(f, 0.0, 1000.0)

    # normalise so that Σ f = √(2n)  (the reference normalisation)
    target_sum = math.sqrt(2.0 * n)
    f *= target_sum / np.sum(f)

    best_seq = f.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Try the LP‑based refinement if cvxpy is available.
    try:
        import cvxpy as cp

        # mixing weight (how aggressive the update is)
        t_mix = 0.2

        # pre‑allocate the variable once and reuse it (warm start)
        g = cp.Variable(n, nonneg=True)

        iteration = 0
        while time.monotonic() < deadline:
            iteration += 1
            # current max convolution (scalar)
            conv_fg = np.convolve(f, f)
            M = np.max(conv_fg)

            # Build the convolution matrix for the current f
            T = _conv_matrix(f)

            # Linear programme: maximise sum(g)  s.t.  T @ g ≤ M
            prob = cp.Problem(cp.Maximize(cp.sum(g)),
                              [T @ g <= M, g >= 0])
            # solve – ECOS is fast for dense medium‑sized LPs
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)

            if g.value is None:
                print("[lp]  solver failed – breaking out of LP loop")
                break

            g_val = np.asarray(g.value, dtype=np.float64)
            sum_g = np.sum(g_val)
            if sum_g <= 0.0:
                print("[lp]  trivial solution – breaking")
                break

            # Rescale to the required ℓ₁‑norm
            scale = target_sum / sum_g
            g_scaled = g_val * scale

            # Convex combination with the previous iterate
            f_new = (1.0 - t_mix) * f + t_mix * g_scaled

            # Clip to the allowed range (the evaluator also clips, but we stay safe)
            f_new = np.clip(f_new, 0.0, 1000.0)

            # Evaluate
            score = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = f_new.copy()
                print(f"[lp] iter {iteration:3d}  improved score = {best_score:.6f}")

            # Prepare for next iteration
            f = f_new

        print("[lp] finished LP refinement")

    except Exception as exc:  # cvxpy not installed or any other error
        print("[lp]  not available or failed:", exc)

    # --------------------------------------------------------------
    # 3.  Simple block‑perturbation fallback (as a safety net).
    #    It runs only if there is still time left.
    def make_block_perturbation(base, rng):
        cand = np.asarray(base, dtype=np.float64).copy()
        n_ = cand.size
        # perturb a random contiguous block
        width = int(rng.integers(max(1, n_ // 200), max(2, n_ // 12)))
        start = int(rng.integers(0, max(1, n_ - width + 1)))
        stop = min(n_, start + width)
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        # occasional symmetrisation
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        return np.clip(cand, 0.0, 1000.0)

    # fallback local search
    cand = best_seq.copy()
    while time.monotonic() < deadline:
        cand2 = make_block_perturbation(cand, rng)
        score2 = float(evaluate_sequence(cand2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = cand2.copy()
            cand = cand2
            print(f"[fallback] improved score = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return [float(x) for x in best_seq.tolist()]
