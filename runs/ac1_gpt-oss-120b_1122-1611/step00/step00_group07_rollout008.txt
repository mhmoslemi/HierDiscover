For a sequence a_i = C·C(2i,i)/4**i (C(i) are central binomial coefficients)
the discrete convolution a*a is constant.  After a simple normalisation
the evaluation value becomes

    2·n·max(a*a) / (sum(a)**2) = 2·n / (sum_i a_i)**2  →  π/2 ≈ 1.571

which is well below the previous bound 2.0.  The routine builds this
Catalan‑type sequence for the requested length (taken from the global
`height_sequence_1` if present) and optionally performs a tiny amount of
local random search to polish the result.  The function returns the best
sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    Return the first ``length`` terms of the sequence
        a_i = binom(2*i, i) / 4**i
    which satisfies (a * a)[k] = 1 for 0 <= k < length.
    The recurrence a_{i+1} = a_i * (2*i+1) / (2*i+2) is used for stability.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        # i in code is the new index; previous index is i-1
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale ``seq`` so that its L1 norm equals ``target_sum``.
    The scaling factor is positive and therefore respects the non‑negative
    requirement of the problem.
    """
    current_sum = np.sum(seq)
    if current_sum == 0:
        return seq
    factor = target_sum / current_sum
    return seq * factor


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a cheap O(n) perturbation of ``base``.  A random contiguous block
    is multiplied by a log‑normal factor; optional mirroring adds a little
    extra exploration.
    """
    cand = base.copy()
    n = cand.size
    # block width roughly between n/200 and n/12 (as in the reference code)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetric averaging – keeps the shape nicely behaved
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # guarantee all entries stay in the allowed interval
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main function required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Build a Catalan‑type sequence (optimal for the convolution problem) and,
    while time permits, perform a tiny stochastic local search to possibly
    improve it further.  The returned list of floats is guaranteed to be
    non‑negative and complies with the security checks of ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the candidate.  Use the length of the
    # previously discovered construction if it exists, otherwise fall back
    # to a modest size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 0:
                raise ValueError
        except Exception:
            n = 256
    else:
        n = 256

    # ------------------------------------------------------------------
    # Build the baseline Catalan sequence and normalise it.  The natural
    # normalisation used in the reference papers is sum = sqrt(2*n);
    # that puts the sequence comfortably inside the 0…1000 range.
    # ------------------------------------------------------------------
    base_seq = catalan_sequence(n)
    target_sum = np.sqrt(2.0 * n)
    base_seq = normalise_to_target_sum(base_seq, target_sum)

    # Initial evaluation (for logging)
    try:
        best_score = float(evaluate_sequence(base_seq.tolist()))
    except Exception:      # safety – should never happen
        best_score = float('inf')
    best_seq = base_seq.copy()

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Small optional local search: keep trying cheap block perturbations
    # and keep any improvement.  The time budget is deliberately reduced
    # by a few seconds to guarantee we return before the hard deadline.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    while time.monotonic() < deadline:
        cand = block_perturbation(best_seq, rng)
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            print(f"[improve] new best score={best_score:.6f}")

    print(f"[final] returning sequence of length {n} with score {best_score:.6f}")
    # Convert to a plain Python list of floats as required.
    return [float(x) for x in best_seq]
