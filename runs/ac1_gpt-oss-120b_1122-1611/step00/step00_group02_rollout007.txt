def generate_sidons(n, rng, max_iter):
    best_set = []
    best_len = 0
    for iter in range(max_iter):
        current_set = []
        sums_used = np.zeros(2*n, dtype=bool) # index from 0 to 2n-2 inclusive.
        order = rng.permutation(n) # random order
        for idx in order:
            # Check if insertion allowed
            conflict = False
            # self sum 2*idx
            if sums_used[idx+idx]:
                conflict = True
                continue
            # For each existing element j in current_set:
            for j in current_set:
                s = idx + j
                if sums_used[s]:
                    conflict = True
                    break
            if not conflict:
                # add
                current_set.append(idx)
                # mark sums
                sums_used[idx+idx] = True
                for j in current_set[:-1]: # all previous elements in set before we added this
                    sums_used[j+idx] = True
        if len(current_set) > best_len:
            best_set = current_set.copy()
            best_len = len(current_set)
            print(f'New best size: {best_len}')
        # early break: if best_len close to theoretical bound maybe stop
    return best_set

"""
Sidon‑set based construction.

The convolution b = a * a contains, at position k, all unordered pairs
(i , j) with i + j = k.  If the set of indices where a_i > 0 is a
Sidon (B₂) set, each sum i + j is unique, therefore each b_k consists of
at most one product a_i·a_j.  With a uniform height H (the maximal allowed
value 1000) the maximum entry of b is H² and the score becomes

    score = 2·n·H² / (m·H)² = 2·n / m²

where n = len(a) and m = |{i : a_i > 0}|.  The largest possible Sidon set
inside {0,…,n‑1} has size ≈ √(2·n); for n≈7000 this is about 115.
A classical greedy construction (the Mian‑Chowla sequence) reaches the
theoretical limit, giving a score ≈ 1.0 – well below the required 1.5025.

The function below builds this Sidon set, puts height 1000 on every
selected index and returns the full sequence.
"""

import time
import numpy as np

def _mian_chowla(limit: int) -> list[int]:
    """
    Greedy construction of a B₂ (Sidon) set up to ``limit``.
    Returns the list of chosen indices (including 0).
    """
    # start with the element 0 – its self‑sum is 0
    chosen = [0]
    used_sums = {0}               # sums i+j that already occurred
    last = 0

    while True:
        cand = last + 1
        # find the smallest integer that does not create a duplicated sum
        while True:
            conflict = False
            # check the self‑sum
            if cand + cand in used_sums:
                conflict = True
            else:
                # check sums with all previously chosen elements
                for p in chosen:
                    if cand + p in used_sums:
                        conflict = True
                        break
            if not conflict:
                break
            cand += 1

        if cand > limit:
            break

        # record all new sums produced by this element
        for p in chosen:
            used_sums.add(cand + p)
        used_sums.add(cand + cand)

        chosen.append(cand)
        last = cand

    return chosen
