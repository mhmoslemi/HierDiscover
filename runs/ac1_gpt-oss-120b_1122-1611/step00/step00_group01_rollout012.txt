def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Get initial sequence as global variable 'height_sequence_1'
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        seq0 = None

    if seq0 is None or seq0.ndim != 1 or seq0.size == 0 or not np.isfinite(seq0).all():
        seq = np.ones(512, dtype=np.float64)  # fallback
    else:
        seq = np.clip(seq0, 0.0, 1000.0)

    # Normalize: scaling irrelevant but ensure sum not <0.01; but values okay.

    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial score: {best_score}")

    n = len(seq)

    # parameters
    block_width_min = max(1, n // 400)   # narrower
    block_width_max = max(2, n // 10)

    # Step size for targeted modifications
    step_factor = 0.1

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        mode = rng.integers(0, 3)  # 0 block, 1 target, 2 swap

        if mode == 0:
            # block perturbation similar but with varying sigma
            width = int(rng.integers(block_width_min, block_width_max+1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            cand = best_seq.copy()
            factor = np.exp(rng.normal(0.0, 0.08))  # smaller sigma
            cand[start:stop] *= factor
            # random symmetry flip
            if rng.random() < 0.3:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)
        elif mode == 1:
            # Targeted reduction of max convolution contributions
            conv = np.convolve(best_seq, best_seq)
            k_max = int(np.argmax(conv))
            # Identify top contributing indices
            j = k_max - np.arange(n)
            mask = (j >= 0) & (j < n)
            contrib = np.zeros_like(best_seq)
            contrib[mask] = best_seq[mask] * best_seq[j[mask]]
            # choose an index with high contribution
            if mask.any():
                # probability weighted by contribution
                probs = contrib / (contrib.sum() + 1e-12)
                i = int(rng.choice(np.arange(n), p=probs))
                # partner index
                p = k_max - i
                # reduction factor
                f = 1.0 - rng.uniform(0.05, 0.2)
                cand = best_seq.copy()
                # apply reduction to both i and p (if distinct)
                cand[i] = max(0.0, cand[i] * f)
                if p != i:
                    cand[p] = max(0.0, cand[p] * f)
                # redistribute removed mass to random low-contrib indices
                removed = best_seq[i] - cand[i]
                if p != i:
                    removed += best_seq[p] - cand[p]
                # select some recipient indices (2 to 5)
                num_rcpt = rng.integers(1, 5)
                rcpts = rng.choice(np.arange(n), size=num_rcpt, replace=False)
                addition = removed / num_rcpt
                cand[rcpts] += addition
                cand = np.clip(cand, 0.0, 1000.0)
            else:
                # fallback to block
                cand = best_seq.copy()
        else:  # mode == 2 pair swap
            # Choose two indices and move mass
            i, j = rng.choice(np.arange(n), size=2, replace=False)
            amount = rng.uniform(0.0, min(best_seq[i], best_seq[i] * 0.3))
            cand = best_seq.copy()
            cand[i] -= amount
            cand[j] += amount
            cand = np.clip(cand, 0.0, 1000.0)

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            print(f"Iter {iteration}, mode {mode}, new best {best_score}")
            # optionally adapt step sizes etc.

    return [float(x) for x in best_seq]

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    # get base sequence length
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base_seq = None

    if base_seq is None or base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
        n = 4096  # default length
        base_seq = np.ones(n, dtype=np.float64) * (1.0 / n)  # uniform with sum 1
    else:
        n = base_seq.size

    # generate arcsine distribution
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i + 1) / n
    arcsine = (2.0 / np.pi) * (np.arcsin(np.sqrt(x1)) - np.arcsin(np.sqrt(x0)))
    # Ensure no negative, clip
    arcsine = np.clip(arcsine, 0.0, 1000.0)

    # starting candidate is either arcsine or fallback uniform or given base
    best_seq = arcsine.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Init arcsine candidate score: {best_score}")

    # also consider the provided height_sequence_1 (maybe already good)
    parent = np.clip(base_seq, 0.0, 1000.0)
    parent_score = float(evaluate_sequence(parent.tolist()))
    if parent_score < best_score:
        best_seq = parent.copy()
        best_score = parent_score
        print(f"Using provided parent score: {parent_score}")

    # time loop
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    iteration = 0
    # Pre-calc parameters for block perturbation
    min_width = max(1, n // 400)
    max_width = max(2, n // 12)

    while time.monotonic() < deadline:
        iteration += 1
        # Choose mode with probabilities
        mode = rng.integers(0, 4)  # 0 block,1 target,2 swap,3 random_scale

        if mode == 0:
            # block perturbation (smaller sigma)
            width = int(rng.integers(min_width, max_width+1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand = best_seq.copy()
            factor = np.exp(rng.normal(0.0, 0.09))  # multiplicative noise
            cand[start:start+width] *= factor
            if rng.random() < 0.33:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)

        elif mode == 1:
            # target reduction of max convolution entry
            conv = np.convolve(best_seq, best_seq)
            k_max = int(np.argmax(conv))
            # Compute contributions
            ind = np.arange(n)
            partner = k_max - ind
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros(n, dtype=np.float64)
            contrib[mask] = best_seq[mask] * best_seq[partner[mask]]
            # Choose index with probability proportional to contribution
            if contrib.sum() > 0:
                probs = contrib / contrib.sum()
                i = int(rng.choice(ind, p=probs))
                j = int(k_max - i)
                # Propose reduction
                reduction = rng.uniform(0.05, 0.2)  # fraction to reduce
                cand = best_seq.copy()
                new_i = max(0.0, cand[i] * (1.0 - reduction))
                delta_i = cand[i] - new_i
                cand[i] = new_i
                if j != i:
                    new_j = max(0.0, cand[j] * (1.0 - reduction))
                    delta_j = cand[j] - new_j
                    cand[j] = new_j
                else:
                    delta_j = 0.0
                # total removed mass
                removed = delta_i + delta_j
                # distribute removed to random low-contrib indices
                if removed > 0:
                    # choose a few receivers
                    rcpts = rng.choice(ind, size=rng.integers(1, 4), replace=False)
                    per = removed / len(rcpts)
                    cand[rcpts] += per
                cand = np.clip(cand, 0.0, 1000.0)
            else:
                # fallback
                cand = best_seq.copy()

        elif mode == 2:
            # swap mass between two positions
            i, j = rng.choice(np.arange(n), size=2, replace=False)
            # compute amount to move
            max_move = best_seq[i] * rng.random()
            amount = max_move * rng.uniform(0.0, 0.5)
            cand = best_seq.copy()
            cand[i] -= amount
            cand[j] += amount
            cand = np.clip(cand, 0.0, 1000.0)

        else:
            # random global scaling of a small set
            cand = best_seq.copy()
            # pick few indices
            idxs = rng.integers(0, n, size=rng.integers(1, 10))
            factor = np.exp(rng.normal(0.0, 0.1, size=idxs.shape))
            cand[idxs] *= factor
            cand = np.clip(cand, 0.0, 1000.0)

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            print(f"Iter {iteration}, mode {mode}, score {best_score:.6f}")

    # final logging
    print(f"Finished after {iteration} iterations, best score {best_score}")

    return [float(x) for x in best_seq]

"""
propose_candidate
-----------------

A fast heuristic optimizer that builds a non‑negative “U‑shaped” step
function and then iteratively reduces the worst convolution entry.
The algorithm works on a fixed length (taken from the global variable
`height_sequence_1`).  It starts from the analytically simple sequence

    a[i] = 1 + cos(π * i / (n‑1))

which already has a relatively small maximal self‑convolution.
During the search the algorithm repeatedly

* finds the lag k where the convolution a * a is maximal,
* identifies the pair (i, k‑i) that contributes most to this lag,
* transfers a small amount of mass from that pair towards the outer
  parts of the sequence (edges), and
* occasionally applies a random block perturbation to escape local
  minima.

All changes keep the coefficients non‑negative and bounded by 1000.
The loop stops when the allotted time expires or a score below the
target 1.5025 is reached.  The best sequence found is returned as a
plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas, top‑level only)
# ----------------------------------------------------------------------


def _initial_u_shape(n: int) -> np.ndarray:
    """
    Return a deterministic U‑shaped sequence of length n:
    a[i] = 1 + cos(π * i / (n‑1)).
    The sequence satisfies sum(a) = n, values are in [0, 2].
    """
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(np.pi * idx / (n - 1))
    # Clip to the limits enforced later – here just a safety net.
    return np.clip(a, 0.0, 1000.0)


def _block_perturb(seq: np.ndarray, rng: np.random.Generator,
                   min_width: int, max_width: int) -> np.ndarray:
    """Create a cheap O(n) perturbation by scaling a random contiguous block."""
    n = seq.size
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = seq.copy()
    # multiplicative noise – a modest sigma keeps us close to the current point
    factor = float(np.exp(rng.normal(0.0, 0.07)))
    cand[start:stop] *= factor
    # occasional mirroring (helps when the optimal shape is almost symmetric)
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _center_to_edge_move(seq: np.ndarray, rng: np.random.Generator,
                         edge_frac: float = 0.12,
                         transfer_frac: float = 0.15) -> np.ndarray:
    """
    Move a modest amount of mass from a randomly chosen centre index
    to a random edge index.
    - edge_frac  : fraction of the whole array that is considered “edge”.
    - transfer_frac : maximal fraction of the source value that may be moved.
    """
    n = seq.size
    # define edge and centre index sets
    edge_len = max(1, int(n * edge_frac))
    centre_start = edge_len
    centre_end = n - edge_len

    # pick source (centre) and destination (edge)
    src = int(rng.integers(centre_start, centre_end))
    # choose one side of the edge uniformly
    if rng.random() < 0.5:
        dst = int(rng.integers(0, edge_len))                     # left edge
    else:
        dst = int(rng.integers(n - edge_len, n))                 # right edge

    # amount to transfer – a fraction of the source value
    max_amount = seq[src] * transfer_frac
    if max_amount <= 0.0:
        return seq.copy()          # nothing to move
    amount = float(rng.uniform(0.0, max_amount))

    cand = seq.copy()
    cand[src] -= amount
    cand[dst] += amount
    return np.clip(cand, 0.0, 1000.0)


def _targeted_move(seq: np.ndarray, rng: np.random.Generator,
                   reduction_frac_low: float = 0.05,
                   reduction_frac_high: float = 0.20,
                   redistribute: bool = True) -> np.ndarray:
    """
    Reduce the two entries that contribute the most to the current maximal
    convolution lag and redistribute the removed mass to a few low‑impact
    positions.  This directly attacks the current bottleneck.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # contributions of each index to the lag k_max  (a[i] * a[k_max-i])
    i = np.arange(n, dtype=np.int64)
    j = k_max - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[mask] * seq[j[mask]]

    if not np.any(mask):
        return seq.copy()

    # pick the index with the largest contribution
    i_star = int(np.argmax(contrib))
    j_star = int(k_max - i_star)

    # decide by how much to shrink the two entries
    red_frac = float(rng.uniform(reduction_frac_low, reduction_frac_high))
    cand = seq.copy()

    removed = 0.0
    # shrink i_star
    old = cand[i_star]
    cand[i_star] = max(0.0, old * (1.0 - red_frac))
    removed += old - cand[i_star]

    # shrink the partner (if different)
    if j_star != i_star:
        old = cand[j_star]
        cand[j_star] = max(0.0, old * (1.0 - red_frac))
        removed += old - cand[j_star]

    if redistribute and removed > 0.0:
        # choose a few indices where the contribution to k_max is small
        low_mask = (contrib < (np.percentile(contrib[mask], 20) if np.any(mask) else 0))
        low_idxs = np.where(low_mask)[0]
        if low_idxs.size == 0:
            low_idxs = np.arange(n)                 # fall back to any index
        rcpt_cnt = int(rng.integers(1, 5))
        rcpts = rng.choice(low_idxs, size=min(rcpt_cnt, low_idxs.size), replace=False)
        share = removed / rcpts.size
        cand[rcpts] += share

    return np.clip(cand, 0.0, 1000.0)

# Main entry point required by the competition environment
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds
    (a small safety margin is subtracted) and uses three simple
    mutation operators:
      * a block multiplicative perturbation,
      * a centre‑to‑edge mass transfer,
      * a direct reduction of the most harmful pair.
    The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise the sequence: try to use the provided construction,
    # fall back to the analytic U‑shape otherwise
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
        seq = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback length – a few thousand entries work well for the time budget
        n = 6000
        seq = _initial_u_shape(n)

    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n}, score={best_score:.6f}")

    # parameters for the three mutation operators
    min_width = max(1, n // 250)
    max_width = max(2, n // 15)
    iter_cnt = 0

    # Main optimisation loop
    while time.monotonic() < deadline:
        iter_cnt += 1

        # pick a mutation operator
        op = rng.integers(0, 4)   # 0: block, 1: centre→edge, 2: targeted, 3: small random tweak

        if op == 0:
            cand = _block_perturb(best_seq, rng, min_width, max_width)

        elif op == 1:
            cand = _center_to_edge_move(best_seq, rng)

        elif op == 2:
            cand = _targeted_move(best_seq, rng)

        else:   # a tiny random tweak – useful to shake out of tiny plateaus
            idx = int(rng.integers(0, n))
            factor = float(np.exp(rng.normal(0.0, 0.05)))
            cand = best_seq.copy()
            cand[idx] *= factor
            cand = np.clip(cand, 0.0, 1000.0)

        # evaluate the candidate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            print(f"[iter {iter_cnt:5d}] new best {best_score:.6f} (op={op})")
            # exit early if we have already beaten the target bound
            if best_score < 1.5025:
                print("[info] target achieved – terminating early.")
                break

    print(f"[finished] iterations={iter_cnt}, best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
