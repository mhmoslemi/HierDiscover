def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.size
    # convolution via FFT of length 2n-1: use size next_pow2.
    m = 1 << (2*n-1).bit_length()
    A = np.fft.rfft(a, n=m)
    B = A * A
    conv = np.fft.irfft(B)[:2*n-1]  # length at least 2n-1
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a**2)

def block_perturb(seq, rng):
    n = len(seq)
    # pick width between 1% and 10% of n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    scale = np.exp(rng.normal(0, 0.15))
    out = seq.copy()
    out[start:start+width] *= scale
    # maybe also random additive noise:
    if rng.random() < 0.3:
        out += rng.normal(0, 0.05, size=n)
    out = np.clip(out, 0, None)
    return out

def constant_magnitude_phase_seq(n, rng):
    m = 1 << (2*n-1).bit_length()
    # choose random phases for half of rfft output
    # rfft length = m//2+1.
    phases = rng.uniform(0, 2*np.pi, size=m//2+1)
    # magnitude constant = 1
    coeff = np.exp(1j*phases)
    # ensure DC and Nyquist real for conjugate symmetry?
    coeff[0] = coeff[0].real  # make real.
    if m%2==0:
        coeff[-1] = coeff[-1].real
    # Inverse rfft
    seq = np.fft.irfft(coeff, n=m)[:n]
    # shift to positive:

    best = base.copy()

    # parameters

        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)
        temp = init_temp * (1 - frac) + final_temp * frac

        # propose candidate
        proposal_type = rng.choice(['block', 'transfer', 'noise', 'restart'], p=[0.5,0.2,0.2,0.1])

        if proposal_type == 'block':

        # normalize sum to 1
        s = cand.sum()

def fast_score(a, n=None):
    if n is None:
        n = a.shape[0]
    # assume sum(a) = 1
    m = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    B = A * A
    conv = np.fft.irfft(B, n=m)[:2*n-1]
    # small negative due to numeric issues
    max_b = np.max(conv)
    return 2.0 * n * max_b

def generate_sidon_set(n, s, rng):
    selected = []
    sums = set()
    attempts = 0
    while len(selected) < s and attempts < 10000:
        x = int(rng.integers(0, n))
        # Ensure not already selected
        if x in selected:
            attempts += 1
            continue
        # Check sums
        conflict = False
        for y in selected:
            if (x + y) in sums:
                conflict = True
                break
        if conflict:
            attempts += 1
            continue
        # also check square sum (x+x)
        if (x + x) in sums:
            attempts += 1
            continue
        # add
        for y in selected:
            sums.add(x + y)
        sums.add(x + x)
        selected.append(x)
        attempts = 0
    return selected

But note the sums set includes both x+y for distinct y ( unordered) and x+x. This ensures no collisions.

"""
Propose a candidate step sequence with a very low evaluation score.

The idea:
* Use a **Sidon set** (a set of indices whose pairwise sums are all distinct).
  If we put the maximum allowed value 1000 at those positions the
  self‑convolution has at most one term 1000·1000 per index, i.e.
  max(b) ≈ 10⁶.
* Fill the remaining positions with a uniform small value `L`.  This
  increases the total sum (the denominator) while only adding
  a modest amount to the convolution, because the extra contribution
  per index is at most `s·1000·L + (n‑s)·L²` where `s` is the Sidon set size.
* By trying a few Sidon sets (different random seeds) and a range of
  low values `L` we can find a construction whose score is well below
  the previous bound `2.0`.  The code evaluates candidates with a fast
  FFT based surrogate of `evaluate_sequence` and finally returns the
  best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Fast surrogate for evaluate_sequence (uses FFT, much quicker)
def _fast_score(seq: np.ndarray) -> float:
    """Return the same objective value as evaluate_sequence but using FFT."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    # length for linear convolution (next power of two is fast)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    B = A * A                     # convolution in frequency domain
    conv = np.fft.irfft(B, n=m)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def _generate_sidon_set(n: int, target_sz: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy construction of a Sidon set (all pairwise sums distinct) inside [0, n).

    Parameters
    ----------
    n : int
        Upper bound for the indices.
    target_sz : int
        Desired size of the set (the greedy algorithm may stop earlier).
    rng : np.random.Generator
        Random number generator.

    Returns
    -------
    np.ndarray
        Sorted array of indices belonging to the Sidon set.
    """
    chosen = []
    sums = set()            # all i+j values seen so far (including i=j)
    attempts = 0
    # we allow a generous number of attempts; if we cannot reach the
    # target we simply return the best we found.
    while len(chosen) < target_sz and attempts < target_sz * 500:
        cand = int(rng.integers(0, n))
        if cand in chosen:
            attempts += 1
            continue
        # check if adding cand would duplicate a sum
        bad = False
        # self‑sum
        if cand + cand in sums:
            bad = True
        else:
            for x in chosen:
                s = cand + x
                if s in sums:
                    bad = True
                    break
        if bad:
            attempts += 1
            continue
        # accept
        for x in chosen:
            sums.add(cand + x)
        sums.add(cand + cand)
        chosen.append(cand)
        attempts = 0  # reset after a successful insertion
    return np.array(sorted(chosen), dtype=np.int64)


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Build a sequence using a Sidon set of indices set to 1000 and the rest
    filled with a small constant `L`.  Several Sidon constructions and several
    low‑value candidates are examined; the best sequence (according to the
    fast surrogate) is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Length of the sequence – use the provided global if available
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).size)
    else:
        n = 256  # fallback
    # ------------------------------------------------------------------

    # Upper bound on the size of a Sidon set inside [0, n).  Theory gives
    # roughly sqrt(2*n); we try a little below that to make the greedy work.
    max_sid_size = int(np.sqrt(2 * n))          # ~110 for n≈6000
    target_sid_size = max_sid_size - 5          # a safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Try a few different Sidon sets (different random seeds)
    for outer in range(5):
        sid_set = _generate_sidon_set(n, target_sid_size, rng)
        s = sid_set.size
        if s == 0:
            continue

        # ------------------------------------------------------------------
        # Scan a reasonable range of low values.
        # The low value must be <= 1000; we look at integers from 1 to 30
        # (larger values quickly increase the diagonal from low‑low terms).
        for L in range(1, 31):
            # Build the candidate sequence
            seq = np.full(n, float(L), dtype=np.float64)
            seq[sid_set] = 1000.0

            # Fast surrogate score
            sc = _fast_score(seq)

            # Keep the best overall
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

        # Occasionally print progress
        if time.monotonic() > deadline:
            break
        print(f"[{time.monotonic():.1f}] Sidon size {s}, best score so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # If we still have no sequence (unlikely) fall back to a uniform one.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Final sanity: ensure values are within the allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain Python list of floats
    return [float(x) for x in best_seq]
