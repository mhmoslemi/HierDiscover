def compute_eval(seq):
    n = seq.size
    # Ensure sum not zero
    sum_a = seq.sum()
    if sum_a <= 0:
        return np.inf
    # Normalize to sum = 1 to simplify
    normalized = seq / sum_a
    # Compute convolution via FFT
    n2 = 2 * n - 1
    A = np.fft.rfft(normalized, n2)
    conv = np.fft.irfft(A * A, n2)
    max_b = conv.max()
    # evaluation = 2 * n * max_b / (sum_a**2) = 2 * n * max_b  because sum_a = 1 after normalization
    return float(2.0 * n * max_b)
```

Explanation: conv = (normalized_seq) * (normalized_seq) = (seq / sum) * (seq / sum) = conv_original / sum^2. So max_b_norm = max_b_original/(sum^2). def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    end_time = start_time + max(0.0, float(budget_s) - 2.0)

    # Determine sequence length n
    try:
        n = len(height_sequence_1)
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            raise ValueError
    except Exception:
        n = 256
        base_seq = np.ones(n, dtype=np.float64)

    # Helper functions
    def normalize(v):
        s = v.sum()
        if s == 0:
            return v
        return v / s

    def eval_seq(v):
        # v assumed non-negative, not necessarily normalized
        v_norm = normalize(v)
        n_local = v_norm.size
        n2 = 2 * n_local - 1
        A = np.fft.rfft(v_norm, n2)
        conv = np.fft.irfft(A * A, n2)
        max_b = conv.max()
        return 2.0 * n_local * max_b

    def conv_full(v):
        v_norm = normalize(v)
        n2 = 2 * v_norm.size - 1
        A = np.fft.rfft(v_norm, n2)
        return np.fft.irfft(A * A, n2)

    # Initialize candidate pool
    candidates = []

    # Uniform
    candidates.append(np.ones(n, dtype=np.float64))

    # Random Dirichlet
    candidates.append(rng.random(n))

    # V-shaped (high at ends)
    idx = np.arange(n)
    vshape = 1.0 - np.minimum(idx, n - 1 - idx) / (n // 2 + 1)  # linear decreasing to centre
    candidates.append(vshape)

    # Prior best if available
    if 'height_sequence_1' in globals():
        candidates.append(base_seq)

    # Evaluate initial pool
    best_seq = None
    best_score = np.inf
    for cand in candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        cand = normalize(cand)
        sc = eval_seq(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"Initial candidate score {sc:.6f}")

    # Current state for SA
    cur_seq = best_seq.copy()
    cur_score = best_score

    iter_count = 0
    # SA temperature schedule
    T0 = 0.02
    T_end = 0.0005

    while time.monotonic() < end_time:
        # Compute fraction remaining
        frac = (end_time - time.monotonic()) / (end_time - start_time)
        # temperature linearly decays
        temp = T0 * frac + T_end * (1 - frac)

        # Choose move
        move_type = rng.choice(['block', 'swap', 'smooth', 'grad'])
        if move_type == 'block':
            # block perturbation similar to provided but using normalized seq
            width = int(rng.integers(max(2, n // 200), max(3, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = np.exp(rng.normal(0.0, 0.12))
            cand = cur_seq.copy()
            cand[start:stop] *= factor
            if rng.random() < 0.3:
                cand = 0.5 * (cand + cand[::-1])
        elif move_type == 'swap':
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
            eps = rng.random() * 0.2  # fraction at most 20%
            delta = eps * cur_seq[i]
            cand = cur_seq.copy()
            cand[i] = max(0.0, cand[i] - delta)
            cand[j] += delta
        elif move_type == 'smooth':
            # moving average smoothing with small window
            w = int(rng.integers(3, min(15, n // 10)))
            # pad with edge value
            pad = np.pad(cur_seq, (w, w), mode='edge')
            kernel = np.ones(2*w+1) / (2*w+1)
            cand = np.convolve(pad, kernel, mode='valid')
        elif move_type == 'grad':
            # gradient step based on current max shift
            conv = conv_full(cur_seq)   # length 2n-1
            kmax = int(np.argmax(conv))
            # compute gradient
            idx = kmax - np.arange(n)
            grad = np.zeros_like(cur_seq)
            valid = (idx >= 0) & (idx < n)
            grad[valid] = cur_seq[idx[valid]]
            # step size proportional to temp
            alpha = 0.5 * temp  # small
            cand = cur_seq - alpha * grad
            cand = np.maximum(cand, 0.0)
        else:
            cand = cur_seq.copy()
        # clip and normalize
        cand = np.clip(cand, 0.0, 1000.0)
        cand = normalize(cand)
        cand_score = eval_seq(cand)

        # Acceptance
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {iter_count}: new best {best_score:.6f}")
                if best_score < 1.5025:
                    print("Reached target threshold.")
                    break

        iter_count += 1

    # ensure final output is list of floats
    if best_seq is None:
        best_seq = np.ones(n) / n
    return [float(x) for x in best_seq.tolist()]

"""
Simple LP‑guided optimizer for the convolution bound.

The algorithm works with a normalized (sum‑to‑one) non‑negative sequence `f`.
At each iteration we

1. compute the self‑convolution `f*f` (via FFT) and its current maximum `M`,
2. pick a small set of shifts where this maximum is attained (the “active”
   constraints),
3. solve the linear program

        max   Σ g_i
        s.t.  (f * g)[k] ≤ M    for every active shift k
              g_i ≥ 0

   which is a tiny LP (a few constraints, many variables).  The optimal `g`
   concentrates mass on indices where `f` is smallest (i.e. where the
   coefficient in the active constraint is smallest).
4. mix the current sequence with the new direction `g`

        f ← (1‑t)·f + t·g ,   t slowly decays with time,

   re‑normalize, and enforce mirror symmetry `f = (f+f[::-1])/2`.

The best sequence seen during the run is kept and returned.
All heavy work (FFT, small LP) runs in O(n log n) per iteration,
so many thousands of iterations fit easily in the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top‑level (no closures, no lambdas)
# ----------------------------------------------------------------------


def _normalize(vec: np.ndarray) -> np.ndarray:
    """Return a copy of ``vec`` scaled so that its L1‑norm equals 1."""
    s = vec.sum()
    if s == 0:
        return vec.copy()
    return vec / s


def _conv_fft(vec: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``vec`` with itself using a real FFT.
    The result has length 2*len(vec)-1.
    """
    n = vec.size
    n2 = 2 * n - 1
    # rfft works with real input and returns the half‑complex spectrum
    spec = np.fft.rfft(vec, n=n2)
    conv = np.fft.irfft(spec * spec, n=n2)
    return conv


def _eval_score(vec: np.ndarray) -> float:
    """
    Fast approximation of ``evaluate_sequence`` for a normalized vector.
    ``vec`` is assumed non‑negative; it is internally normalised.
    Returns the same value that ``evaluate_sequence`` would produce.
    """
    v = _normalize(vec)
    n = v.size
    conv = _conv_fft(v)
    max_b = conv.max()
    # because the vector is normalised, sum(v)=1, so
    # evaluate = 2 * n * max_b / (1**2) = 2 * n * max_b
    return float(2.0 * n * max_b)


def _build_active_constraints(f: np.ndarray, conv: np.ndarray, k_max: int,
                             max_k: int) -> np.ndarray:
    """
    Return a list of shift indices that are used as active constraints.
    We take:
      * the global maximum shift,
      * a few other largest shifts (parameterised by ``max_k``).
    """
    # number of active constraints (including the global max)
    K = max(5, max_k)
    # take the K largest values of the convolution
    # argpartition is O(n) and works well for large arrays
    idx = np.argpartition(conv, -K)[-K:]
    # guarantee that the global maximum is present
    if k_max not in idx:
        # replace the smallest of the chosen indices
        smallest = idx[np.argmin(conv[idx])]
        idx[np.argmin(conv[idx])] = k_max
    return idx


def _solve_lp(f: np.ndarray, active_idx: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the tiny LP:

        max Σ g_i
        s.t. Σ f_{k-i} g_i ≤ M   for every k in ``active_idx``
             g_i ≥ 0

    Returns the optimal vector ``g`` (not normalised).  If the LP fails
    for any reason the caller should fall back to a simple perturbation.
    """
    from scipy.optimize import linprog

    n = f.size
    rows = active_idx.size
    # Build the constraint matrix A_ub (rows × n)
    A_ub = np.zeros((rows, n), dtype=np.float64)
    for r, k in enumerate(active_idx):
        # coefficient for g_i is f_{k-i} when the index is valid
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        A_ub[r, mask] = f[idx[mask]]
    b_ub = np.full(rows, M, dtype=np.float64)

    # Objective: maximize sum(g) → minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use the high‑performance HiGHS interior‑point/simplex solver
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP solver failed")
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step‑function height sequence.

    The routine respects the time budget, repeatedly improves the
    sequence by the LP‑guided scheme described above and finally returns
    the best sequence found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length to work with.  If the global ``height_sequence_1``
    # exists we reuse its length; otherwise fall back to a modest size.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
        n = base.size
    except Exception:
        n = 256
        base = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initial sequence: uniform (any non‑negative start works)
    # ------------------------------------------------------------------
    f = np.ones(n, dtype=np.float64)
    f = _normalize(f)

    best_f = f.copy()
    best_score = _eval_score(f)
    print(f"Initial score: {best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    it = 0
    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # 1) compute convolution, max value, and a few active constraints
        # ------------------------------------------------------------------
        conv = _conv_fft(f)
        M = conv.max()
        k_max = int(np.argmax(conv))

        # ------------------------------------------------------------------
        # 2) build a modest active set (the few largest convolution entries)
        # ------------------------------------------------------------------
        active = _build_active_constraints(f, conv, k_max, max_k=5)

        # ------------------------------------------------------------------
        # 3) solve the LP; fall back to a tiny random perturbation on failure
        # ------------------------------------------------------------------
        try:
            g_raw = _solve_lp(f, active, M)
        except Exception:
            # a simple multiplicative perturbation (always feasible)
            perturb = np.exp(rng.normal(0.0, 0.08, size=n))
            g_raw = f * perturb

        # ------------------------------------------------------------------
        # 4) normalise the direction, mix with the current vector, symmetrise
        # ------------------------------------------------------------------
        g = _normalize(g_raw)

        # mixing coefficient – linearly decays from ~0.2 to ~0.01
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        mix = 0.01 + 0.19 * (1.0 - elapsed / total)

        f = (1.0 - mix) * f + mix * g
        # enforce mirror symmetry (helps keep the central convolution peak
        # well behaved) and numerical positivity
        f = 0.5 * (f + f[::-1])
        f = np.maximum(f, 0.0)
        f = _normalize(f)

        # ------------------------------------------------------------------
        # 5) evaluate and keep the best sequence seen so far
        # ------------------------------------------------------------------
        cur_score = _eval_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"Iter {it:06d}: new best score {best_score:.6f}")

        # early exit if we already beat the target bound
        if best_score < 1.5025:
            print("Target bound reached – terminating early.")
            break

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python list of floats)
    # ------------------------------------------------------------------
    if best_f is None or best_f.size == 0:
        best_f = np.ones(n) / n
    return [float(x) for x in best_f.tolist()]
