def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-3.0)
    # start length
    n = 500
    # initial sequence f: uniform or random
    f = np.ones(n, dtype=np.float64) # uniform
    # could also combine with height_sequence_1 truncated to n
    # clip
    f = np.clip(f, 0, 1000)
    # compute best
    best_seq = f.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    t = 0.2
    while time.monotonic() < deadline:
        # compute convolution and M
        conv = np.convolve(f, f)
        M = conv.max()
        # build linear programming problem: maximize sum(g)
        # constraints: conv_f_g <= M for each index
        # g >= 0, g <= 1000 (due to clipping)
        # We'll use scipy.optimize.linprog
        c = -np.ones(n)  # maximize sum => minimize negative sum
        # Build constraint matrix A_ub * g <= b_ub
        # A_ub dimension (2*n-1) x n
        # each row k: sum_i f[i] * g[k-i] <= M
        # Represent as linear combination of g variables: A[k, j] = f[k-j] if 0 <= k-j < n else 0.
        rows = 2*n - 1
        A_ub = np.zeros((rows, n))
        for k in range(rows):
            # i ranges over indices where 0 <= i < n and 0 <= k-i < n -> i in [max(0,k-(n-1)), min(n-1,k)]
            start_i = max(0, k-(n-1))
            end_i = min(n-1, k)
            # for each i, the variable index j = k-i
            # coefficient is f[i]
            # we can fill row:
            # for j from k - end_i to k - start_i ??? Let's fill by iterating i.
            for i in range(start_i, end_i+1):
                j = k - i
                A_ub[k, j] = f[i]
        b_ub = np.full(rows, M)
        # Add upper bound g <= 1000 as inequality
        # also we can add bound (0, 1000) using bounds argument
        bounds = [(0, 1000) for _ in range(n)]
        # Solve
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.monotonic())})
        if not res.success:
            # fallback: break or continue
            break
        g = res.x
        # Scale g to have same L1 sum as f
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g * (sum_f / sum_g)
        # Update f via convex combination
        f = (1 - t) * f + t * g
        # Clip
        f = np.clip(f, 0, 1000)
        # Evaluate
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = f.copy()
            best_score = score
            print(f'New best score: {best_score:.6f}')
    # return best sequence as list
    return [float(x) for x in best_seq]

"""
Propose a candidate sequence that minimizes the evaluation function.

The algorithm is an alternating linear‑programming (LP) scheme inspired by the
method described in the prompt.

* A target L1‑norm S = sqrt(2·n) is imposed.  When the sum equals S the
  evaluation reduces to the maximal convolution coefficient, i.e.
  evaluate_sequence(f) = max_k (f*f)[k].

* Starting from a random (or a truncated copy of the global `height_sequence_1`)
  we repeatedly:
    1. Compute M = max convolution of the current sequence f.
    2. Solve an LP that maximises Σ g_i subject to the *linear* constraints
       (f * g)[k] ≤ M for every index k (the convolution with a fixed f is
       linear in g) and g_i ≥ 0.
    3. Rescale g so that its L1‑norm equals the target S.
    4. Update f ← (1‑t)·f + t·g with a mixing factor t∈(0,1) (t=0.25 works
       well).  The sum stays S, the sequence stays non‑negative, and the
       maximal convolution coefficient tends to decrease.
    5. Keep the best sequence seen according to `evaluate_sequence`.

The LP has (2·n‑1) inequality rows and n variables.  For n≈800 the dense
coefficient matrix is ~10 MiB and solves in well under a second with SciPy’s
high‑performance HiGHS solver, allowing many iterations inside the 1000 s
budget.

If SciPy is unavailable the routine falls back to a simple random block
perturbation search (the previous baseline) so it always returns a valid
sequence.

The function respects the required signature and returns a list of floats.
"""

import numpy as np
import time
import math

# Try to import the LP solver; if unavailable we will fall back to the
# baseline block‑perturbation method.
try:
    from scipy.optimize import linprog
except Exception:          # pragma: no cover
    linprog = None


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A of shape (2*n-1, n) such that
    (A @ g)[k] = (f * g)[k] for every k.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    col = np.arange(n)
    # For each i, f[i] adds to rows i+col at column col.
    for i in range(n):
        rows_i = i + col          # indices i+j for j = 0..n-1
        A[rows_i, col] += f[i]
    return A


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Return a sequence of non‑negative heights (list of floats) that
    minimises `evaluate_sequence`.  The algorithm uses an alternating
    LP refinement; if the LP solver is not available it falls back to the
    block‑perturbation local search from the reference implementation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Choose a working length n.
    # Prefer a slice of the provided global construction if it is long.
    # ------------------------------------------------------------------
    global height_sequence_1   # may be defined by the runner
    if isinstance(globals().get('height_sequence_1'), (list, np.ndarray)):
        base = np.asarray(height_sequence_1, dtype=np.float64)
        # we want a moderate size for the LP; 800 works well.
        n = min(800, base.size)
        if n < base.size:
            start = rng.integers(0, base.size - n + 1)
            f = base[start:start + n].copy()
        else:
            f = base.copy()
    else:
        # No global sequence – generate a fresh random one.
        n = 800
        f = rng.exponential(scale=1.0, size=n)

    # Clip to the allowed value range and normalise the L1 norm.
    f = np.clip(f, 0.0, 1000.0)
    target_sum = math.sqrt(2.0 * n)          # we enforce Σf_i = sqrt(2·n)
    s = f.sum()
    if s > 0:
        f = f * (target_sum / s)
    else:
        f = np.full(n, target_sum / n, dtype=np.float64)

    best_seq = f.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the alternating scheme.
    # ------------------------------------------------------------------
    t_mix = 0.25                     # convex‑mixing factor
    iteration = 0

    # Use the LP if possible, otherwise fall back to the simple baseline.
    use_lp = linprog is not None

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # 1. Compute the current maximal convolution value M.
        # ------------------------------------------------------------------
        conv_f_f = np.convolve(f, f)
        M = float(conv_f_f.max())

        # ------------------------------------------------------------------
        # 2. LP step – maximise Σ g_i under (f * g)[k] ≤ M.
        # ------------------------------------------------------------------
        if use_lp:
            # Build the (2·n‑1) × n matrix for the constraints.
            A = _build_convolution_matrix(f)
            rows = A.shape[0]

            # The LP is:
            #   minimise   -1·g
            #   subject to A @ g ≤ M·1, 0 ≤ g ≤ 1000.
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(rows, M, dtype=np.float64)
            bounds = [(0.0, 1000.0) for _ in range(n)]

            # Solve.  HiGHS (the default backend for method='highs') is fast.
            # No explicit time‑limit is supplied – the outer loop enforces the
            # overall budget.
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                # If the LP fails we abort the alternating loop and fall back.
                print("[lp] solver failed, exiting LP loop")
                use_lp = False
                continue
            g = res.x
        else:
            # ------------------------------------------------------------------
            # Fallback: a cheap random block perturbation (same as the reference).
            # ------------------------------------------------------------------
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            g = f.copy()
            g[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:
                g = 0.5 * (g + g[::-1])
            g = np.clip(g, 0.0, 1000.0)

        # ------------------------------------------------------------------
        # 3. Rescale g so that its L1 norm equals the target sum.
        # ------------------------------------------------------------------
        sum_g = g.sum()
        if sum_g == 0.0:
            # cannot rescale a zero vector – abort the LP loop.
            break
        g = g * (target_sum / sum_g)

        # ------------------------------------------------------------------
        # 4. Convex combination update.
        # ------------------------------------------------------------------
        f = (1.0 - t_mix) * f + t_mix * g
        f = np.clip(f, 0.0, 1000.0)
        # Re‑normalise exactly (tiny numerical drift may appear).
        f = f * (target_sum / f.sum())

        # ------------------------------------------------------------------
        # 5. Evaluate and keep the best sequence.
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(f.tolist()))
        iteration += 1
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"[iter {iteration:4d}] new best score = {best_score:.6f}")

        # Occasionally (every ~20 iterations) add a tiny random jitter to
        # escape possible stagnation when using the LP approach.
        if use_lp and iteration % 20 == 0:
            jitter = rng.normal(0.0, 0.001, size=n)
            f = np.clip(f + jitter, 0.0, 1000.0)
            f = f * (target_sum / f.sum())

    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return [float(v) for v in best_seq]
