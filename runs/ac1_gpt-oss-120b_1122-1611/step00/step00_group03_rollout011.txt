def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    # Determine length n: use length of height_sequence_1 if available else default 256
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = base_seq.size
        if n == 0:
            raise Exception
    except Exception:
        n = 256
        base_seq = np.ones(n, dtype=np.float64)
    best = base_seq.copy()
    best = np.clip(best, 0.0, 1000.0)
    best_score = evaluate_sequence(best.tolist())
    # If invalid, reinitialize
    if not np.isfinite(best_score):
        best = np.ones(n, dtype=np.float64)
        best_score = evaluate_sequence(best.tolist())
    # Helper: evaluate
    def eval_seq(seq):
        return float(evaluate_sequence(seq.tolist()))

    # We'll store candidates and track.
    # Parameter search ranges
    # Exponential families: a_i = exp(-alpha * i) with normalization not needed.
    # Use alpha in [0, 5 * log(2) / n maybe]. Actually we can set scaling factor such that a_i approx near 1 and decays.
    # Variation: scale exponent factor and shift exponent.

    # We generate many candidate sequences as per parameter distributions
    while time.monotonic() < deadline:
        # randomly choose a family
        family = rng.choice(['exp', 'pow', 'dual_exp', 'linear', 'random'])
        if family == 'exp':
            # a_i = exp(-alpha * i / n) ; alpha in [0, 10]
            alpha = rng.uniform(0.0, 10.0)
            seq = np.exp(-alpha * np.arange(n) / n)
        elif family == 'pow':
            # a_i = (i+1)^(-p) ; p in [0.0, 2.0]
            p = rng.uniform(0.0, 2.0)
            seq = 1.0 / (np.arange(1, n+1) ** p)
        elif family == 'dual_exp':
            # two-stage exponential: early part decays fast, later part shallow
            split = int(rng.integers(1, n-1))
            alpha1 = rng.uniform(0.0, 20.0)
            alpha2 = rng.uniform(0.0, 5.0)
            part1 = np.exp(-alpha1 * np.arange(split) / split)
            part2 = np.exp(-alpha2 * np.arange(n - split) / (n - split))
            seq = np.concatenate([part1, part2])
        elif family == 'linear':
            # a_i = max(0, 1 - beta * i/n) ; beta in [0,2]
            beta = rng.uniform(0.0, 2.0)
            seq = 1.0 - beta * np.arange(n) / n
            seq = np.maximum(seq, 0.0)
        elif family == 'random':
            # random values with smoothing
            seq = rng.random(n)  # uniform [0,1)
            # optionally apply smoothing via moving average
            if rng.random() < 0.5:
                w = rng.integers(1, 10)
                seq = np.convolve(seq, np.ones(w)/w, mode='same')
        # apply optional scaling factor and clipping
        seq = np.clip(seq, 0.0, 1000.0)
        # evaluate
        score = eval_seq(seq)
        if np.isfinite(score) and score < best_score:
            best = seq
            best_score = score
            print(f"New best score {best_score:.6f} at time {time.monotonic():.2f}")
            # attempt local refinement: small perturbations around best
            # maybe break inner loops later

        # occasional local hill climb from best
        if rng.random() < 0.1:
            # generate a small perturbation of best
            candidate = best.copy()
            # pick a block to perturb
            width = max(1, int(n // rng.integers(20, 200)))  # block size
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.08))
            candidate[start:start+width] *= mult
            # optional smoothing
            if rng.random() < 0.2:
                candidate = 0.5 * (candidate + candidate[::-1])
            candidate = np.clip(candidate, 0.0, 1000.0)
            score2 = eval_seq(candidate)
            if np.isfinite(score2) and score2 < best_score:
                best = candidate
                best_score = score2
                print(f"Local improve to {best_score:.6f}")

def generate_shape(n, rng):
    kind = rng.choice(['exp', 'power', 'stretched_exp', 'piecewise'])
    if kind == 'exp':
        alpha = rng.uniform(0.1, 10.0)
        seq = np.exp(-alpha * np.arange(n) / n)
    elif kind == 'power':
        p = rng.uniform(0.1, 2.0)
        seq = 1.0 / (np.arange(1, n+1) ** p)
    elif kind == 'stretched_exp':
        beta = rng.uniform(0.5, 3.0)
        seq = np.exp(-beta * np.sqrt(np.arange(n)))
    else:
        # piecewise: two exponentials
        split = rng.integers(1, n-1)
        a1 = rng.uniform(0.1, 10.0)
        a2 = rng.uniform(0.1, 5.0)
        seq = np.empty(n)
        seq[:split] = np.exp(-a1 * np.arange(split) / split)
        seq[split:] = np.exp(-a2 * np.arange(n-split) / (n-split))
    # ensure monotonic decreasing
    seq = np.sort(seq)[::-1]   # descending
    return seq
```

def alternate_extremes(arr):
    lo = 0
    hi = len(arr)-1
    out = []
    toggle = True
    while lo <= hi:
        if toggle:
            out.append(arr[lo])
            lo += 1
        else:
            out.append(arr[hi])
            hi -= 1
        toggle = not toggle
    return np.array(out, dtype=np.float64)

def project_simplex(v):
    """Project vector v onto the probability simplex (nonnegative sum 1)."""
    # standard algorithm: sort descending, compute threshold.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```
