def generate_alpha(N):
    alphas = np.empty(N, dtype=np.float64)

        alphas[i] = comb(2*i, i) / (4**i)

def propose_candidate(seed=42, budget_s=990, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # starting length maybe len(height_sequence_1) if available else 3000

        iter += 1
        # Randomly choose move

            i = rng.choice(n, p=best / best.sum())
            j = rng.integers(0, n)

            candidate[j] += delta
        else:

        s = candidate.sum()
        if s < 1e-9:
            continue
        candidate /= s

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"iter {iter}, new best {best_score:.6f}")

import scipy.sparse as sp

def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

"""
Simple evolutionary search using the optimal “alpha‑sequence” as a seed.
The alpha sequence a_i = C(2i,i)/4^i (i=0..n-1) has the property
that its self‑convolution is constant (=1) on the first n positions,
giving a score close to π/2 ≈ 1.57.  Random local perturbations,
mass‑transfer moves and a cheap gradient step are used to push the
maximum convolution value down while keeping the L1‐norm fixed.
The algorithm runs until the time budget expires and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """
    Return the first n coefficients of the series
        a_i = C(2i,i) / 4^i .
    The recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1)) is used.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def convolution_max(a: np.ndarray) -> float:
    """Return the maximum value of the linear convolution a * a."""
    b = np.convolve(a, a)
    return float(b.max())

    Search for a sequence with a low value of the provided objective.
    The routine uses the alpha‑sequence as a sophisticated start point
    and then performs a mixture of random perturbations, block scaling
    and a simple gradient‑based step.  It returns the best sequence found
    before the time budget expires.

    # ------------------------------------------------------------------

    it = 0

    sigma = 0.25
    # probability of using the cheap gradient step
    grad_prob = 0.07

    while time.monotonic() < deadline:
        it += 1

        # ----------------------------------------------------------------
        # Pick a proposal type
        # ----------------------------------------------------------------
        move_type = rng.random()
        if move_type < 0.4:
            # ---- Gaussian multiplicative noise (global) -----------------
            cand = best * (1.0 + sigma * rng.standard_normal(n))
        elif move_type < 0.8:
            # ---- Small mass transfer between two positions -------------
            i = rng.choice(n, p=best / best.sum())
            j = rng.integers(0, n)
            if i == j:

            delta = best[i] * rng.uniform(0.0, 0.5)
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
        else:

            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))

            cand = best.copy()
