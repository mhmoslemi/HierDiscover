def generate_sidon_set(target_size, n, rng):

    # Use a random permutation of range(n) to consider candidates.
    attempts = 0

"""...
Generate Sidon set based step function with many non-zero entries,
giving low convolution peak and thus low evaluation score.
...
"""
import numpy as np
import time

def generate_sidon_set(target_size, n, rng):
    """Return a list of indices forming a Sidon set of given size.
    The set satisfies that all i+j are distinct (including i+i).
    If construction fails after a few attempts, return None.
    """
    attempts = 0
    max_attempts = 20
    while attempts < max_attempts:
        indices = []
        sums = set()
        # random order improves chances to find large set
        candidates = rng.permutation(n)
        for i in candidates:
            # Early exit - if remaining candidates insufficient to reach target
            if len(indices) + (n - np.where(candidates == i)[0][0]) < target_size:
                break
            # test collision
            conflict = False
            # self sum
            if (i + i) in sums:
                conflict = True
            else:
                for j in indices:
                    if (i + j) in sums:
                        conflict = True
                        break
            if not conflict:
                # add new sums
                for j in indices:
                    sums.add(i + j)
                sums.add(i + i)
                indices.append(i)
                if len(indices) >= target_size:
                    return np.array(indices, dtype=int)
        attempts += 1
    return None

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """Search for a low‑score step function using Sidon sets.
    
    The algorithm repeatedly builds a sparse Sidon set of indices, assigns equal
    weight to those positions, and keeps the best sequence found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine length n from provided global sequence if present
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 6733  # fallback to known length

    best_seq = None
    best_score = float('inf')
    # Pre‑compute a simple fallback uniform sequence for safety
    uniform_seq = np.full(n, 1.0 / n, dtype=np.float64)
    uniform_score = float(evaluate_sequence(uniform_seq.tolist()))
    best_seq = uniform_seq.copy()
    best_score = uniform_score
    print(f"Fallback uniform score {best_score:.6f}")

    while time.monotonic() < deadline:
        # Choose a target size; larger size gives lower theoretical bound.
        target_m = int(rng.integers(140, 165))   # try aggressive sizes

The algorithm builds a sparse step function whose non‑zero positions form a
Sidon set (all pairwise sums are distinct, including i+i).  For such a set,
the convolution's maximal entry is at most 2/(m²) where m is the number of
non‑zero entries (weights are equal).  The evaluation function equals
2·n·max_b/(∑a)², so the score is bounded by 4·n/m².  By choosing
m≈150 for n≈6733 we obtain a score well below the target 1.5025.

The search repeatedly draws random Sidon sets of size between 140 and 164,
evaluates them with ``evaluate_sequence`` and keeps the best one before the
time budget expires.
"""

import numpy as np
import time

def _sidon_set(target_size: int, n: int, rng: np.random.Generator):
    """Return an array of `target_size` indices forming a Sidon set.

    The set satisfies that all unordered sums i+j (including i+i) are distinct.
    A simple greedy construction with random ordering is used.  If the
    construction fails after several attempts, ``None`` is returned.
    """
    max_attempts = 15
    for _ in range(max_attempts):
        indices = []
        sums = set()
        # Random ordering gives a chance to reach larger sizes.
        for i in rng.permutation(n):
