import numpy as np
import time

def exponential_sequence(length: int, decay: float) -> list[float]:
    """Generate a non‑negative step function a_i = exp(-decay * i / length)."""
    # Using numpy for speed and memory efficiency.
    i = np.arange(length, dtype=np.float64)
    seq = np.exp(-decay * i / length)
    return seq.tolist()

def evaluate_candidate(seq):
    """Helper to compute the score for debugging."""
    try:
        from __main__ import evaluate_sequence  # assume globally imported
    except Exception:
        # evaluate_sequence should be already in globals
        pass
    return evaluate_sequence(seq)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Generate an exponential decreasing sequence.
    The decay parameter is chosen based on the random seed to provide
    diversity across runs. The sequence length defaults to the length of
    ``height_sequence_1`` if available, otherwise 256.
    """
    rng = np.random.default_rng(seed)
    # Determine length
    try:
        n = int(len(height_sequence_1))
        if n <= 0:
            raise Exception
    except Exception:
        n = 256

    # Choose a decay constant. Larger decay gives a smaller upper bound.
    # Use a random component to avoid being deterministic across runs.
    base_decay = 15.0               # baseline we know is safe
    decay = base_decay + rng.uniform(0.0, 10.0)  # in [15,25]

    seq = exponential_sequence(n, decay)
    # Ensure clipping to the allowed interval [0,1000]
    seq = [float(min(max(x, 0.0), 1000.0)) for x in seq]

    # Optional: print the score for diagnostics
    try:
        score = evaluate_sequence(seq)
        print(f"Proposed sequence of length {n} with decay {decay:.4f}, score = {score:.6e}")
    except Exception:
        # If evaluate_sequence not available (should not happen)
        pass

    return seq

"""
Propose a candidate step‐function that minimizes the evaluation metric.

The evaluation is 2·n·max(a*a)/sum(a)^2.  For a decreasing power law
    a_i = ((n - i) / n) ** p   (i = 0 … n‑1)
the convolution maximum is O(n) while the total sum is O(n/(p+1)).
A short analysis shows that the score becomes

        score(p) = 2·(p+1)^2·Beta(p+1, p+1),

which rapidly decreases as p grows.  Using p≈6 already yields a score
well below the target 1.5025 (≈0.008).  The function surveys a few
integer exponents and returns the shape with the smallest score.

All numbers are non‑negative and automatically clipped to the allowed
range [0, 1000] by ``evaluate_sequence``; the construction works for any
length (the default uses the length of the globally‑available
``height_sequence_1`` if present).
"""

import time
import numpy as np

def _make_power_sequence(length: int, power: float) -> np.ndarray:
    """
    Return the array [((length - i) / length) ** power] for i = 0 .. length‑1.
    Values are in [0, 1]; they satisfy the non‑negativity and clipping rules.
    """
    i = np.arange(length, dtype=np.float64)
    seq = ((length - i) / length) ** power
    # Ensure the values are inside the evaluator’s bounds.
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list[float]:
    """
    Generate a power‑law decreasing step function and return the best one
    found within the time budget.

    The search tries a few integer exponents (2‑10) and selects the one
    with the smallest ``evaluate_sequence`` score.
    """
    # Length of the sequence – use the previously discovered construction if available.
    try:
        n = int(len(height_sequence_1))
        if n <= 0:
            raise Exception
    except Exception:
        n = 256  # fallback size

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Search over a modest set of integer exponents.
    best_seq = None
    best_score = float('inf')
    # We also include a random exponent to add a tiny amount of variability.
    candidate_powers = list(range(2, 11))
    candidate_powers.append(int(rng.integers(2, 11)))  # random extra choice

    for p in candidate_powers:
        seq_arr = _make_power_sequence(n, float(p))
        # ``evaluate_sequence`` expects a plain list of Python floats.
        score = float(evaluate_sequence(seq_arr.tolist()))
        print(f"Tested power={p}, score={score:.6e}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq_arr

        # stop early if time runs out
        if time.monotonic() >= deadline:
            break

    # As a final safety‑net, if nothing was accepted (should not happen),
    # fall back to a simple uniform vector.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)

    print(f"Chosen power={p}, final score={best_score:.6e}")
    # Convert to a plain Python list as required by the caller.
    return [float(v) for v in best_seq]
