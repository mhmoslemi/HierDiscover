def fast_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.size
    size = 1 << (2 * n - 1).bit_length()  # next power of two >= 2n-1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a * sum_a)

"""
Propose a step‑function (non‑negative sequence) that tries to minimise
evaluate_sequence.  The algorithm works with a binary (0/1) sequence of a
chosen length n (taken from the provided height_sequence_1 if it exists,
otherwise a power‑of‑two length).  It starts from a random placement of K
ones (density about 15‑20 %) and performs a simulated‑annealing hill‑climb
in the space of swaps “one → empty”.  Each swap is accepted if it
strictly improves the objective or with a temperature–controlled
probability if it is slightly worse.  The convolution needed for the
objective is evaluated by FFT (O(n log n)) so many thousands of swaps can
be tried within the time budget.  The best sequence found is returned
as a plain list of floats.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective using FFT (identical to the
# provided evaluate_sequence for non‑negative inputs).
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return the same value as evaluate_sequence(seq.tolist()), but fast.
    seq must be a 1‑D np.ndarray of non‑negative floats.
    """
    n = seq.size
    if n == 0:
        return np.inf
    s = seq.sum()
    if s < 0.01:
        return np.inf

    # next power of two for a linear convolution of length 2*n-1
    m = int(2 ** np.ceil(np.log2(2 * n - 1)))
    # FFT of the zero‑padded vector
    f = np.fft.rfft(seq, m)
    conv = np.fft.irfft(f * f, m)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Main entry point – the function called by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a low‑score step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds (the function will stop a few
        seconds before this value to avoid timeout).

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length n.  Prefer the length of the supplied
    # construction (height_sequence_1), otherwise fall back to 4096.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 4096                         # power of two – friendly for FFT

    # ------------------------------------------------------------------
    # Choose the number of ones K (density 15‑20 %).  This gives a reasonable
    # average value of the convolution while keeping the search space
    # manageable.
    # ------------------------------------------------------------------
    density = 0.18                       # tweakable
    K = max(2, int(round(density * n)))  # at least two ones to avoid inf

    # ------------------------------------------------------------------
    # Initialise a random binary pattern.
    # ------------------------------------------------------------------
    a = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=K, replace=False)
    a[ones] = 1.0

    best_seq = a.copy()
    best_score = fast_score(best_seq)
    # keep a reference score from the official evaluator for safety
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        # fallback to the fast version if the external evaluator is unavailable
        pass

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters.
    # ------------------------------------------------------------------
    start_temp = 0.2
    end_temp = 0.001
    total_iters = 0
    iters_without_improve = 0
    max_no_improve = 2000               # trigger occasional restart

    # time handling – stop a little before the budget to be safe
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Pre‑compute list of empty positions for fast sampling
    def empty_positions(arr):
        return np.where(arr == 0.0)[0]

    while time.monotonic() < deadline:
        # linear temperature decay
        elapsed = time.monotonic() - (deadline - float(budget_s) + 3.0)
        progress = min(1.0, elapsed / (float(budget_s) - 3.0))
        temp = start_temp * (1.0 - progress) + end_temp * progress

        # pick a random occupied index and a random empty index
        occ_idx = rng.integers(0, n)
        while best_seq[occ_idx] == 0.0:
            occ_idx = rng.integers(0, n)

        emp_idx = rng.integers(0, n)
        while best_seq[emp_idx] != 0.0:
            emp_idx = rng.integers(0, n)

        # propose the swap
        cand = best_seq.copy()
        cand[occ_idx] = 0.0
        cand[emp_idx] = 1.0

        # fast evaluation
        cand_score = fast_score(cand)

        # acceptance test (Metropolis criterion)
        delta = cand_score - best_score
        accept = False
        if delta < 0:
            accept = True
        elif temp > 0 and rng.random() < np.exp(-delta / temp):
            accept = True

        if accept:
            best_seq = cand
            best_score = cand_score
            iters_without_improve = 0
        else:
            iters_without_improve += 1

        total_iters += 1

        # occasional restart to escape deep local minima
        if iters_without_improve >= max_no_improve:
            # re‑initialise a fresh random pattern (same K)
            a = np.zeros(n, dtype=np.float64)
            ones = rng.choice(n, size=K, replace=False)
            a[ones] = 1.0
            cand_score = fast_score(a)
            if cand_score < best_score:
                best_seq = a
                best_score = cand_score
            iters_without_improve = 0

        # occasional progress print (lightweight)
        if total_iters % 2000 == 0:
            print(f"Iter {total_iters:,d} – best score ≈ {best_score:.6f}")

    # Finally convert to plain Python floats (the grader expects a list)
    result = [float(x) for x in best_seq]
    print(f"Search finished after {total_iters:,d} iterations – final score ≈ {best_score:.6f}")
    return result
