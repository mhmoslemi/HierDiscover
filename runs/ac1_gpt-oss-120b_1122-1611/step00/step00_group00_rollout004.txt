"""
Propose a candidate sequence based on the discrete arcsine distribution.
We start from the analytic solution that makes the self‑convolution
approximately uniform, which gives a theoretical lower bound of 1 for the
objective.  The algorithm then performs a short random block‑multiplicative
local search to possibly improve the maximum convolution value further.
"""

import numpy as np
import time

def arcsine_distribution(n):
    """Return a non‑negative array of length n whose self‑convolution
    is close to uniform (discrete arcsine law)."""
    i = np.arange(n, dtype=np.float64)
    denom = np.sqrt((i + 0.5) * (n - 0.5 - i))
    # guard against zero denominator (should never happen)
    denom = np.where(denom == 0, 1.0, denom)
    a = 1.0 / denom
    # Any positive scaling does not affect the objective, keep the raw values
    # and clip to the allowed interval.
    return np.clip(a, 0.0, 1000.0)

def make_block_perturbation(base, rng, sigma=0.12):
    """Create a candidate from ``base`` by multiplying a random block
    by a log‑normal factor.  The block length is chosen proportionally to
    ``n``.  Returns a new ``np.ndarray``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width at least 1 and at most about n/5
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, sigma)))
    candidate[start:stop] *= factor
    # occasional symmetrisation – helps to keep the convolution shape balanced
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

"""
Step‑function optimizer for the `evaluate_sequence` objective.

The continuous arcsine law has the property that the sum of two
independent arcsine‑distributed variables is *uniform*.
Discretising this law gives a sequence whose self‑convolution is almost
flat, which drives the objective
    2·n·max(conv) / (sum²)
close to its theoretical lower bound of 1.
The optimizer therefore starts from the discrete arcsine distribution,
tunes the exponent of the singularity (a free parameter that changes the
shape) and finally performs a cheap stochastic block‑wise local search.
All operations respect the non‑negative, ≤1000 constraints required by
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: discrete arcsine‑type distribution
# ----------------------------------------------------------------------
def arcsine_like(n: int, exponent: float = 0.5) -> np.ndarray:
    """
    Return a length‑n non‑negative array whose autocorrelation is close to
    uniform.  The shape follows
        a_i ∝ 1 / ((i+0.5)*(n-0.5-i))^{exponent/2}
    for i = 0 … n‑1.
    Any positive scaling leaves the objective unchanged; we keep the raw
    values and clip them to the allowed interval.
    """
    i = np.arange(n, dtype=np.float64)
    # add 0.5 to avoid division by zero at the ends
    left = i + 0.5
    right = n - 0.5 - i
    denom = np.sqrt(left * right) ** exponent            # (left*right)^{exponent/2}
    # Guard against numerical zero (should not happen with the 0.5 offset)
    denom = np.where(denom == 0, 1.0, denom)
    a = 1.0 / denom
    return np.clip(a, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: small multiplicative block perturbation (local search)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       sigma: float = 0.12) -> np.ndarray:
    """
    Produce a new candidate by multiplying a random contiguous block of
    ``base`` with a log‑normal factor (exp(N(0, sigma²))).  Occasionally
    the whole vector is symmetrised to keep the convolution balanced.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # minimal block width
    hi = max(2, n // 12)           # maximal block width
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:        # occasional mirroring improves symmetry
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The algorithm proceeds as follows:

    1. Determine the problem length ``n`` (taken from the global
       ``height_sequence_1`` if present, otherwise fall back to 256).
    2. Initialise with the discrete arcsine distribution (exponent = 0.5).
    3. Randomly explore a small neighbourhood of exponent values
       (0.30 – 0.80) and keep the best among them.
    4. Run a cheap block‑wise stochastic local search until the time budget
       expires or the score reaches ≈ 1 (the theoretical optimum).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine the length of the sequence we must output
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))      # global provided by the runner
    except Exception:
        n = 256                               # safe fallback

    # ------------------------------------------------------------------
    # 1) Start with the canonical arcsine shape (exponent = 0.5)
    # ------------------------------------------------------------------
    best = arcsine_like(n, exponent=0.5)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Randomly tweak the exponent – this is essentially a 1‑D
    #    optimisation that costs O(n) per trial.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # sample a new exponent uniformly in a sensible interval
        exp_candidate = float(rng.uniform(0.30, 0.80))
        cand = arcsine_like(n, exponent=exp_candidate)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[exp] new exponent={exp_candidate:.3f}, score={best_score:.6f}")
            # break early if we are already very close to the lower bound
            if best_score <= 1.02:
                break

    # ------------------------------------------------------------------
    # 3) Stochastic block perturbations (hill‑climbing)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[perturb] improved score={best_score:.6f}")
            if best_score <= 1.02:         # theoretical optimum ≈ 1
                break

    print(f"[final] best score={best_score:.6f}")
    # Convert to plain Python list as required by the external framework
    return [float(x) for x in best]
