    best_seq = None ; best_score = np.inf

    while time.monotonic() < deadline:
        positions = np.arange(n)
        rng.shuffle(positions) # works if using numpy's rng

        S = []
        sum_counts = {}
        # Could store as dict int->int

        for pos in positions:
            # fast check: will need to test all existing s in S. If S huge, this inner loop O(k) which may be up to ~200, fine.
            feasible = True
            # self sum
            t_self = pos + pos
            if sum_counts.get(t_self, 0) + 1 > max_occ:
                feasible = False
            else:
                for s in S:
                    t = s + pos

                        break

                    t = s + pos
                    sum_counts[t] = sum_counts.get(t, 0) + 2
                sum_counts[t_self] = sum_counts.get(t_self, 0) + 1
                S.append(pos)
                if len(S) >= target_k:
                    break
        k = len(S)
        if k > best_k:
            # Build sequence
            seq = np.zeros(n, dtype=np.float64)

import numpy as np, time, math, random, itertools
from collections import defaultdict

def greedy_sidonic_set(n, max_occ, rng):
    positions = np.arange(n)
    rng.shuffle(positions)
    S = []
    sum_counts = defaultdict(int)
    for pos in positions:
        # Check self
        t_self = pos + pos
        if sum_counts[t_self] + 1 > max_occ:
            continue
        feasible = True
        for s in S:
            t = s + pos
            if sum_counts[t] + 2 > max_occ:
                feasible = False
                break
        if feasible:
            # add contributions
            for s in S:
                sum_counts[s + pos] += 2
            sum_counts[t_self] += 1
            S.append(int(pos))
    return S, sum_counts
```

"""
Propose a step–function sequence with a low evaluation score.

The algorithm builds a sparse set of positions where the convolution
(i.e. the number of ordered pairs i,j with i+j = t) is kept small.
A greedy construction respects a user‑defined bound *max_occ* on the
maximum number of ordered pairs per sum (the “collision” limit).
Repeated random orders give many candidates; the largest set found
has k spikes (value 1) and a maximal convolution value M≤max_occ.

After a set S is obtained the whole vector is filled with a tiny
baseline ε (added to every entry).  This increases the total sum
while only slightly raising the maximal convolution, typically moving
the score well below the target 1.5025.

The search runs until the allotted time expires (≈ 1000 s) and always
returns the best sequence found.
"""

import time
import math
import numpy as np
from collections import defaultdict

# ----------------------------------------------------------------------
# Helper: greedy construction of a low‑collision set
# ----------------------------------------------------------------------
def greedy_low_collision_set(n: int, max_occ: int, rng) -> tuple[list[int], dict[int, int]]:
    """
    Return a set of positions S (as a Python list) such that for the
    binary sequence a_i = 1_{i∈S} every ordered pair (i,j) contributes
    to the convolution count at t=i+j and no t receives more than
    ``max_occ`` contributions.  The construction proceeds in a random
    order and adds a position when it does not violate the bound.
    """
    positions = list(range(n))
    rng.shuffle(positions)
    S = []                     # selected positions
    count = defaultdict(int)   # ordered‑pair count per sum t

    for pos in positions:
        # self‑pair contribution (i=j=pos) counts as 1
        if count[pos + pos] + 1 > max_occ:
            continue

        feasible = True
        for s in S:
            if count[s + pos] + 2 > max_occ:      # (s,pos) and (pos,s)
                feasible = False
                break
        if not feasible:
            continue

        # accept the position – update the counts
        for s in S:
            count[s + pos] += 2
        count[pos + pos] += 1
        S.append(pos)

    return S, count


def extend_set(S: list[int], count: dict[int, int],
               max_occ: int, n: int) -> tuple[list[int], dict[int, int]]:
    """
    Try to add any further positions to an already constructed set S.
    This is a simple deterministic pass; it is iterated until no more
    positions can be added.
    """
    in_set = np.zeros(n, dtype=bool)
    in_set[S] = True
    improved = True

    while improved:
        improved = False
        for pos in range(n):
            if in_set[pos]:
                continue
            if count[pos + pos] + 1 > max_occ:
                continue
            ok = True
            # test every existing element – cheap because S stays relatively small
            for s in S:
                if count[s + pos] + 2 > max_occ:
                    ok = False
                    break
            if not ok:
                continue
            # add the position
            for s in S:
                count[s + pos] += 2
            count[pos + pos] += 1
            S.append(pos)
            in_set[pos] = True
            improved = True
    return S, count
