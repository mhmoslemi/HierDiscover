"""
Hybrid optimizer for the step‑function problem.

The algorithm starts from a decreasing exponential profile (which already
gives a better bound than the uniform sequence) and then performs a
projected sub‑gradient descent that directly reduces the maximal entry of
the self‑convolution.  To avoid being trapped in a symmetric configuration,
random block perturbations are interleaved with the gradient steps.
All iterates are projected onto the probability simplex (non‑negative and
summing to one); the final vector is rescaled so that its largest entry is
exactly 1000 – this does not change the evaluation because the objective is
scale‑invariant (the evaluator clips to [0,1000] before computing).

The routine runs for the given time budget (default 1100 s, a few seconds
reserved for housekeeping) and returns the best sequence found.
"""

import numpy as np
import time

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("simplex projection works only on 1‑D arrays")
    n = v.size
    # shift to make the algorithm numerically stable
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_vector(a: np.ndarray, n: int) -> float:
    """
    Compute the objective value 2·n·max(conv(a,a)) / (sum(a))**2.
    The vector a is assumed to be non‑negative.
    """
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    s = np.sum(a)
    if s < 0.01:               # guard – should never happen in our code
        return np.inf
    return 2.0 * n * max_conv / (s * s)
