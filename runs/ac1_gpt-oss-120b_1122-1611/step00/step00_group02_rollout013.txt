def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    # Determine n based on height_sequence_1
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    # Build base shape
    base = 1.0 / np.sqrt(np.arange(1, n+1, dtype=np.float64))
    # Compute initial best score
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # compute max convolution of base
    conv = np.convolve(base, base)
    M0 = float(conv.max())
    # high weight
    high_w = np.sqrt(M0 / 2.0) * 0.99  # safety margin
    # generate candidate indices (powers of two)
    pow_idx = []
    p = 1
    while p < n:
        pow_idx.append(int(p))
        p <<= 1
    # Ensure we have at least 1
    if not pow_idx:
        pow_idx = [0]
    # Main loop
    while time.monotonic() < deadline:
        cand = base.copy()
        # random number of indices to upgrade
        k = rng.integers(0, len(pow_idx)+1)
        if k:
            chosen = rng.choice(pow_idx, size=k, replace=False)
            cand[chosen] = high_w
            # optionally also set symmetric positions
            if rng.random() < 0.5:
                sym = n - 1 - chosen
                # keep within bounds
                sym = np.clip(sym, 0, n-1)
                cand[sym] = high_w
        # maybe small random perturbations to all entries to explore
        if rng.random() < 0.2:
            # multiply by random factor per entry with small lognormal noise
            factor = np.exp(rng.normal(0.0, 0.05, size=n))
            cand *= factor
        # clip and evaluate
        cand = np.clip(cand, 0.0, 1000.0)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"New best score: {best_score:.6f}, time: {time.monotonic():.2f}")
    # return best as list
    return [float(x) for x in best]

import time
import numpy as np

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    Construct a non‑negative step‑function sequence with a 1/√i profile and a few
    high peaks placed at power‑of‑two positions. The profile makes the
    self‑convolution almost flat (peak ≈ π) while the extra peaks increase the
    total sum without raising the convolution maximum, thereby lowering the
    evaluation score below the target 1.5025.  Random variations around this
    construction are explored until the time budget expires.
    """
    rng = np.random.default_rng(seed)
    # Small safety margin before the time budget ends
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # Length of the problem – fall back to 256 if the global variable is missing
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # ---- deterministic core construction ---------------------------------
    # a_i = 1 / sqrt(i+1)  (i from 0 to n‑1)
    base = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))

    # peak of the convolution for the pure 1/√i profile
    base_conv = np.convolve(base, base)
    M0 = float(base_conv.max())

    # a safe value for the extra peaks:
    # two ordered copies of such a peak give at most 2 * w^2,
    # we keep it slightly below M0 to guarantee the global maximum does not grow.
    high_w = np.sqrt(M0 / 2.0) * 0.99   # ~0.99 · √(M0/2)

    # indices where we will insert the peaks – powers of two (1,2,4,…)
    pow_idx = []
    p = 1
    while p < n:
        pow_idx.append(p)
        p <<= 1   # multiply by two

    # initial best sequence = the pure 1/√i profile
    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial score (1/√i): {best_score:.6f}")

    # --------------------------------------------------------------
    # Try the deterministic variant: raise every power‑of‑two position
    cand = base.copy()
    cand[pow_idx] = np.maximum(cand[pow_idx], high_w)
    cand = np.clip(cand, 0.0, 1000.0)
    score = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(score) and score < best_score:
        best_seq = cand.copy()
        best_score = score
        print(f"Deterministic boost score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Randomised search that perturbs the construction
    while time.monotonic() < deadline:
        # start from the pure profile again
        cand = base.copy()

        # pick a random non‑empty subset of the power‑of‑two indices
        k = rng.integers(1, len(pow_idx) + 1)
        chosen = rng.choice(pow_idx, size=k, replace=False)
        cand[chosen] = high_w

        # with some probability also raise the mirrored positions near the end
        if rng.random() < 0.3:
            mirrored = n - 1 - chosen
            # keep indices inside the array bounds
            mirrored = np.clip(mirrored, 0, n - 1)
            cand[mirrored] = high_w

        # small multiplicative noise to explore a neighbourhood
        if rng.random() < 0.2:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cand *= noise

        # enforce the global limits required by evaluate_sequence
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score
            print(f"New best: {best_score:.6f} (time {time.monotonic():.2f})")

    # Return the best sequence found as a plain Python list of floats
    return [float(x) for x in best_seq]
