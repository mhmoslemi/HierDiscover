"""
Propose a high‑quality step‑function sequence.

The evaluation metric is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the full (non‑circular) convolution.
For a fixed length `n` the best possible score is 2, obtained by a uniform
sequence.  A classical continuous construction shows that a sequence of the
form

    a[i] ∝ 1 / sqrt(i + 1)

has a convolution that is almost constant; its theoretical score tends to
π/2 ≈ 1.5708, well below the current bound of 2.0.  Starting from this
analytic shape we perform a simple stochastic hill‑climb (block‑wise
multiplicative perturbations) keeping the best sequence found within the
allowed time budget.

The algorithm:
* Initialise `a[i] = 1 / sqrt(i + 1)` (length taken from the global
  `height_sequence_1`, otherwise a default of 3809).
* Repeatedly pick a random contiguous block, scale it by `exp(N(0, σ))`
  (σ slowly decays), and keep the change if it reduces the score.
* Occasionally symmetrise the sequence (`a = (a + a[::-1]) / 2`) – this
  often reduces the peak of the convolution.
* Stop after the time budget and return the best sequence found.

The method respects all security checks in `evaluate_sequence` (all entries
are non‑negative, finite, and bounded by 1000) and makes no filesystem or
network accesses.
"""

import time
import numpy as np


def _initial_sequence(length: int) -> np.ndarray:
    """
    Return the analytic 1/sqrt(i+1) start vector of the given length.
    """
    idx = np.arange(length, dtype=np.float64)
    # avoid zero division; the smallest index is 0 -> 1/sqrt(1) = 1.0
    seq = 1.0 / np.sqrt(idx + 1.0)
    return seq


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator,
                             sigma: float) -> np.ndarray:
    """
    Produce a copy of ``base`` where a random contiguous block of entries is
    multiplied by a log‑normal factor with standard deviation ``sigma``.
    With probability 0.35 the whole vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # block width: between n/200 and n/12 (as in the reference code)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:stop] *= factor
    # optional symmetrisation step
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the hard limits of the evaluator
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Produce a candidate sequence for the step‑function problem.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Approximate time budget in seconds (the function stops a few seconds
        before this to guarantee a return value).

    Returns
    -------
    list[float]
        The best sequence encountered before the time limit.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size.
    # If the global reference sequence exists use its length, otherwise fall
    # back to the length mentioned in the statement.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is supplied by the environment
        length = int(np.asarray(height_sequence_1).size)
    except Exception:
        length = 3809  # default length from the description

    # ------------------------------------------------------------------
    # Initial candidate – analytic 1/sqrt(i+1) shape.
    # ------------------------------------------------------------------
    parent = _initial_sequence(length)
    # Clip to respect the evaluator's bounds (just a safety net)
    np.clip(parent, 0.0, 1000.0, out=parent)

    best = parent
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={length}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hill‑climbing loop.  The schedule for sigma gradually shrinks so that
    # early iterations explore widely while later ones fine‑tune the vector.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.5)

    iteration = 0
    sigma = 0.30                       # initial standard deviation
    sigma_min = 0.02
    decay_rate = 0.9995                # exponential decay per iteration

    while time.monotonic() < deadline:
        iteration += 1
        # gradually reduce sigma
        sigma = max(sigma_min, sigma * decay_rate)

        cand = _make_block_perturbation(best, rng, sigma)
        score = float(evaluate_sequence(cand.tolist()))

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[{iteration:>6}] new best score: {best_score:.6f}")

        # every few hundred iterations force a cheap symmetrisation
        if iteration % 400 == 0:
            sym = 0.5 * (best + best[::-1])
            np.clip(sym, 0.0, 1000.0, out=sym)
            sym_score = float(evaluate_sequence(sym.tolist()))
            if np.isfinite(sym_score) and sym_score < best_score:
                best = sym
                best_score = sym_score
                print(f"[{iteration:>6}] symmetrised improved score: {best_score:.6f}")

    total_time = time.monotonic() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.2f}s")
    # Convert to a plain Python list of floats as required by the harness.
    return [float(x) for x in best]
