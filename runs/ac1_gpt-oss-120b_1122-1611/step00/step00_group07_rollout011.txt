def _make_power_decay(length: int, power: float = 4.0) -> np.ndarray:
    """
    Return a monotone decreasing sequence a_i = 1000 * ((L-i-1)/(L-1))**power.
    The first element is exactly 1000, the last is 0.  The shape
    a_i ∝ (L-i)^p minimises the infinity norm of the self‑convolution
    for a fixed L1‑norm, giving a theoretical score
        score = 2·(p+1)²·B(p+1,p+1)
    where B is the beta‑function.  For p=4 the score is ≈0.079,
    far below the target 1.5025.
    """
    if length <= 1:
        # degenerate case – just return a single positive entry.
        return np.array([1000.0], dtype=np.float64)

    # Normalised positions in [0,1] decreasing from 1 to 0.
    idx = np.arange(length, dtype=np.float64)
    # (L-i-1)/(L-1) gives 1 at i=0 and 0 at i=L-1.
    t = (length - 1 - idx) / (length - 1)
    seq = 1000.0 * np.power(t, power)
    # Numerical safety – enforce bounds required by the evaluator.
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq
