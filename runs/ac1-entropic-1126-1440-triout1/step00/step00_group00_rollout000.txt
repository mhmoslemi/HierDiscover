def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    try: n = len(height_sequence_1)
    except: n = 2000
    # initialize uniform distribution sum=1
    a = np.full(n, 1.0 / n)
    # compute conv via FFT:
    def conv_of(arr):
        m = 1 << ((2*n - 1).bit_length())
        # compute real FFT
        A = np.fft.rfft(arr, m)
        conv = np.fft.irfft(A * A, m)[:2*n - 1]
        return conv
    conv = conv_of(a)
    best_a = a.copy()
    best_score = 2 * n * np.max(conv) / (np.sum(a) ** 2)  # should be 2
    start = time.time()
    deadline = start + budget_s - 5.0
    step_frac = 0.1
    # adaptive step size
    while time.time() < deadline:
        # find index of max conv
        k = np.argmax(conv)
        # compute contribution array for indices that affect k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k) # inclusive
        # slice a_i and a_j = a[k - i]
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        contrib = a[idx_i] * a[idx_j]
        # choose i with probability proportional to contrib
        if np.sum(contrib) == 0:
            # fallback random
            i = np.random.randint(n)
        else:
            probs = contrib / np.sum(contrib)
            i = np.random.choice(idx_i, p=probs)
        # pick j outside the contributing set (so that it does NOT affect k)
        # choose j randomly until not in idx_i range
        # we can sample j by random and if j in idx_i continue
        while True:
            j = np.random.randint(n)
            if j < i_start or j > i_end:
                break
        # compute delta
        delta = step_frac * a[i]
        if delta < 1e-12:
            continue
        # enforce a[i] - delta >= 0
        delta = min(delta, a[i])
        # apply move
        a[i] -= delta
        a[j] += delta
        # recompute conv
        conv = conv_of(a)
        # compute score
        score = 2 * n * np.max(conv) / (np.sum(a) ** 2)
        # Accept if improved
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # print progress
            print(f"Iter {int(time.time() - start)} best {best_score:.6f}")
        else:
            # revert move
            a[i] += delta
            a[j] -= delta
        # optionally adapt step_frac based on acceptance ratio
        # we can slowly decay step_frac
        step_frac *= 0.99999
    # after loop, scale up to max 1000
    max_val = np.max(best_a)
    if max_val > 1000:
        best_a = best_a * (1000.0 / max_val)
    # convert to list of float
    return best_a.tolist()

"""
Propose a step‑function (list of non‑negative heights) that improves over the
baseline uniform sequence.  The algorithm starts from a uniform distribution,
repeatedly identifies the index `k` where the convolution `a*a` attains its
maximum and moves a small amount of mass from a contributing position to a
position that does **not** contribute to that index.  The move is kept only if
the new maximum decreases, otherwise it is reverted.  A simple annealing
schedule (decreasing step size) is used.  The routine works for any length
`n`; if the global variable ``height_sequence_1`` is present its length is used,
otherwise a fallback length of 2000 is taken.  All values are kept well below
the clipping limit 1000, so the security checks in ``evaluate_sequence`` do not
alter the vector.

The method runs for the allotted ``budget_s`` seconds (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a power‑of‑two FFT."""
    n = a.shape[0]
    # Length of the full linear convolution
    conv_len = 2 * n - 1
    # Next power of two for efficient FFT
    m = 1 << (conv_len - 1).bit_length()
    # Real FFT, element‑wise multiplication, inverse FFT
    A = np.fft.rfft(a, m)
    conv = np.fft.irfft(A * A, m)[:conv_len]
    return conv


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low value of ``evaluate_sequence``.
    The algorithm is a simple projected gradient / simulated‑annealing style
    optimiser that moves probability mass away from the currently most loaded
    convolution index.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Problem size and initial uniform distribution
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # use the existing construction length
    except NameError:
        n = 2000                            # fallback size if the variable is missing

    # uniform distribution with total mass 1  (scale does not matter for the metric)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # initial convolution and score
    conv = _conv_fft(a)
    current_score = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)   # should be exactly 2
    best_a = a.copy()
    best_score = current_score

    # ------------------------------------------------------------------
    # 2.  Pre‑compute helper structures
    # ------------------------------------------------------------------
    all_indices = np.arange(n)
    conv_len = 2 * n - 1

    # annealing / step‑size schedule
    step_frac = 0.5           # initial fraction of the selected entry to move
    min_step = 1e-9           # do not try to move less than this
    decay = 0.99999           # geometric decay per iteration

    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # a few seconds margin for finalisation

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify the convolution index with the current largest value
        # --------------------------------------------------------------
        k_max = int(np.argmax(conv))

        # Indices i that actually contribute to conv[k_max]
        i_start = max(0, k_max - (n - 1))
        i_end   = min(n - 1, k_max)                     # inclusive
        idx_i   = np.arange(i_start, i_end + 1, dtype=np.int64)
        idx_j   = k_max - idx_i                         # partner indices (always in range)

        # Contribution of each i to the maximal entry
        contrib = a[idx_i] * a[idx_j]
        total_contrib = np.sum(contrib)

        if total_contrib == 0.0:            # degenerate (should not happen)
            i = np.random.randint(n)
        else:
            # pick i proportionally to its contribution – a greedy choice
            probs = contrib / total_contrib
            i = np.random.choice(idx_i, p=probs)

        # Choose a destination index j that does NOT affect the current maximal
        # convolution entry (i.e. j is outside [i_start, i_end]).
        # Sampling until a suitable j is found is cheap because the excluded
        # interval is at most n long and the complement is also large.
        while True:
            j = np.random.randint(n)
            if not (i_start <= j <= i_end):
                break

        # Amount of probability mass to move
        delta = step_frac * a[i]
        if delta < min_step:
            # step became too small – stop early
            break
        delta = min(delta, a[i])          # keep a[i] non‑negative

        # --------------------------------------------------------------
        # Apply the tentative move
        # --------------------------------------------------------------
        a[i] -= delta
        a[j] += delta

        # Re‑compute the convolution (full FFT – still very fast for n≈7k)
        conv_new = _conv_fft(a)
        new_score = 2.0 * n * np.max(conv_new) / (np.sum(a) ** 2)

        # --------------------------------------------------------------
        # Accept / reject the move
        # --------------------------------------------------------------
        if new_score < current_score:
            # improvement – keep the move
            current_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                # occasional progress report
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.1f}s] new best score = {best_score:.6f}")
        else:
            # reject – restore the previous state
            a[i] += delta
            a[j] -= delta

        # decay the step size a little (helps to converge)
        step_frac *= decay

    # ------------------------------------------------------------------
    # 4.  Final post‑processing
    # ------------------------------------------------------------------
    # Ensure that no entry exceeds the security limit 1000.
    max_val = np.max(best_a)
    if max_val > 1000.0:
        best_a = best_a * (1000.0 / max_val)

    # Return a plain Python list of floats as required by the checker
    return best_a.tolist()
