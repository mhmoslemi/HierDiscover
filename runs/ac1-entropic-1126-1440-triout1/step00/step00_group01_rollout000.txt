"""
Gradient‑descent + occasional LP refinement for the step‑function problem.

The evaluation metric is scale‑invariant, therefore the algorithm works with
unit‑L1 sequences.  Starting from the currently known best construction
(`height_sequence_1`) or a random one, the routine repeatedly flattens the
self‑convolution by minimising the variance of the convolution values.
Because a flatter convolution reduces the maximal entry while keeping the
total sum fixed, the evaluation value drops.

To avoid getting stuck in a local minimum a tiny random “jitter” is added
when no improvement is seen for a while, and, every few hundred iterations,
a linear‑programming step (the same idea as the reference SOTA) is used to
increase the L1‑norm under the current convolution bound.  A short line‑search
over the mixing proportion chooses the best step.

The function runs until the supplied time budget expires and returns the
best sequence found (as a plain Python list of floats).
"""

import time
import numpy as np
from scipy import optimize


def solve_convolution_lp(f_sequence, rhs):
    """
    Solve the linear program:
        maximise   sum(b)
        subject to  conv(f, b) <= rhs   (for all shift positions)
                    b_i >= 0
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f_sequence)
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # Build convolution constraints conv(f, b) <= rhs.
    # Each row corresponds to one shift k = 0 .. 2n-2.
    rows = []
    b_vals = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv_k = sum_i f_i * b_{k-i}
        # only add when the index of b lies inside the vector.
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for i from i_min..i_max, j = k - i is a valid index
        row[k - np.arange(i_min, i_max + 1)] = f_sequence[i_min:i_max + 1]
        rows.append(row)
        b_vals.append(rhs)

    # Non‑negativity constraints: -b_i <= 0
    rows.append(-np.eye(n))
    b_vals.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.concatenate(b_vals)

    # Use the high‑performance HiGHS solver.
    res = optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        method="highs",
        options={"disp": False, "maxiter": 2000},
    )
    if res.success:
        return res.x
    return None
