"""
Simple analytical construction for a very low evaluation score.

The evaluation function is invariant under scaling of the whole sequence
(i.e. multiplying all entries by a constant does not change the score).
Therefore we can freely choose any shape and then scale it to avoid the
built‑in clipping to the interval [0, 1000].

For a monotone polynomial shape a_i = (i+1)^p the normalized score can be
computed in closed form:

    score(p) = 2 * (p+1)^2 * (p!)^2 / (2p+1)!   (independent of n)

This value decreases rapidly with p, e.g.
    p=1 → 1.333…, p=2 → 0.600…, p=3 → 0.228…, p=4 → 0.0793, …
Even p=2 already beats the target 1.503, and higher exponents give
scores two orders of magnitude smaller.

The algorithm simply tries several small integer exponents (p=1…10),
scales the resulting vector so that its maximum entry is 0.5
(to stay far from the clipping threshold), evaluates each candidate,
and returns the best one found.
The whole procedure runs in a fraction of a second.
"""

import numpy as np

def propose_candidate(seed=42, **kwargs):
    """
    Generate a non‑negative step‑function sequence with a very small
    evaluation score by using a polynomially increasing shape.
    
    The length of the sequence is taken from the globally provided
    `height_sequence_1` (default length 2000 if unavailable).  Several
    exponents p are tried and the one with the lowest score is returned.
    """
    # the random seed is kept for reproducibility (not used directly)
    np.random.seed(seed)

    # Determine the length of the sequence to produce.
    try:
        n = len(height_sequence_1)      # use the reference length if present
    except NameError:
        n = 2000                        # fallback length

    best_seq = None
    best_score = float('inf')
    best_p = None

    # Try a modest range of exponents; higher p gives lower score.
    for p in range(1, 11):
        # Build the raw polynomial shape (float64 to avoid integer overflow)
        raw = np.arange(1, n + 1, dtype=np.float64) ** p
        if np.isinf(raw).any() or np.isnan(raw).any():
            # overflow – stop trying larger exponents
            continue

        # Scale to keep values comfortably inside the security bounds.
        # Any positive scaling leaves the score unchanged, but we avoid
        # the clipping to 1000 by limiting the maximum to 0.5.
        scale = 0.5 / raw.max()
        seq = (raw * scale).tolist()

        # Evaluate the candidate (the function is already imported in the
        # execution environment).  Invalid sequences raise an exception;
        # we simply skip those.
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue

        print(f"p = {p:2d} → score = {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq
            best_p = p

    if best_seq is None:
        # As a safety fallback return a uniform sequence (the trivial solution)
        uniform = [1.0] * n
        print("All attempts failed; returning a uniform sequence.")
        return uniform

    print(f"Best exponent p = {best_p} with evaluation score {best_score:.6f}")
    return best_seq
