@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute batch id
    pid_batch = tl.program_id(axis=2)  # 0..batch-1
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # offsets for block start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)  # for inner loop

    # pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_abatch
    b_batch_ptr = b_ptr + pid_batch * stride_bbatch
    c_batch_ptr = c_ptr + pid_batch * stride_cbatch

    # load a tiles
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

"""
custom_kernel: high‑performance forward implementation of the
TriMul “outgoing” update (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension (dim).
2. Linear projections for left/right tensors and three gating tensors.
   - left / right are masked (if a mask is supplied) and multiplied by
     their respective sigmoid gates.
3. Compute the pairwise “tri‑multiplicative” product:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM where for each hidden channel d we need
   left_d @ right_dᵀ.  We fuse this with a custom Triton kernel that
   processes all hidden channels in a single launch (batch = B×H).
4. Layer‑norm the result over the hidden dimension, multiply by the
   output gate and apply a final linear projection back to `dim`.
All heavy arithmetic (step 3) runs in FP16 inside Triton with an
FP32 accumulator; everything else stays in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_abatch, stride_bbatch, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C = A @ Bᵀ  (batched, A: (M,K), B: (N,K) → C: (M,N))"""

    pid_batch = tl.program_id(axis=2)          # index of the (B*H) batch
    pid_m = tl.program_id(axis=0)              # tile row
    pid_n = tl.program_id(axis=1)              # tile column

    # Pointers to the start of the current batch
    a_batch = a_ptr + pid_batch * stride_abatch
    b_batch = b_ptr + pid_batch * stride_bbatch
    c_batch = c_ptr + pid_batch * stride_cbatch

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current tile
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)   accumulator stays FP32

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using a custom Triton kernel for the
    O(N³) pairwise multiplication.  Returns a tensor of shape
    [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # Layer‑norm on the input (float32)
    # ------------------------------------------------------------------ #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )
    # Cast to half for the heavy work
    x_norm = x_norm_fp32.half()

    # ------------------------------------------------------------------ #
    # Linear projections (FP16) and masks
    # ------------------------------------------------------------------ #
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left = torch.nn.functional.linear(x_norm, left_proj_w)          # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # Apply mask if present (mask may be omitted → treat as all ones)
    if mask is None or config.get("nomask", False):
        mask_fp = None
    else:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------ #
    # Gating tensors (FP16)
    # ------------------------------------------------------------------ #
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # Prepare tensors for batched GEMM: (B*H, N, N)  where each matrix is
    # left[b,h,:,:] and right_T[b,h,:,:] = right[b,h,:,:].transpose(-2,-1)
    # ------------------------------------------------------------------ #
    B, N, _, _ = input_tensor.shape
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right for the A·Bᵀ product
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()

    batch_h = B * hidden_dim                     # total number of “matrices”

    # Flatten the batch & hidden dimensions
    A = left_perm.view(batch_h, N, N)            # (batch_h, M=N, K=N)
    B_T = right_T.view(batch_h, N, N)            # (batch_h, K=N, N=N)

    # Output placeholder
    C = torch.empty((batch_h, N, N), dtype=torch.float16, device=A.device)

    # Strides in element units (not bytes)
    stride_am = A.stride(1)          # distance between rows of A (K)
    stride_ak = A.stride(2)          # distance between columns of A (1)
    stride_abatch = A.stride(0)      # batch stride = M*K

    stride_bk = B_T.stride(1)        # distance between rows of B_T (N)
    stride_bn = B_T.stride(2)        # distance between columns of B_T (1)
    stride_bbatch = B_T.stride(0)    # batch stride = K*N

    stride_cm = C.stride(1)          # distance between rows of C (N)
    stride_cn = C.stride(2)          # distance between columns of C (1)
    stride_cbatch = C.stride(0)      # batch stride = M*N

    # ------------------------------------------------------------------ #
    # Triton GEMM launch
    # ------------------------------------------------------------------ #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_h,
    )
    batch_matmul_kernel[grid](
        A, B_T, C,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_abatch, stride_bbatch, stride_cbatch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # Reshape back to (B,N,N,hidden) and continue in FP32
    # ------------------------------------------------------------------ #
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().float()

    # ------------------------------------------------------------------ #
    # Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------ #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # out_gate was computed in FP16 – cast to FP32 before multiplication
    out = out * out_gate.float()

    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP32

    return out
```
