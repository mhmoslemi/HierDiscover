"""
TriMul (outgoing) – fast Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (B, N, N, C).
2. Linear projections: left_proj, right_proj (→ hidden_dim) and three gates.
3. Apply mask (if present) and element‑wise gate (sigmoid) to the projected tensors.
4. Compute the core “tri‑multiplicative” update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is equivalent to a batch of H independent matrix‑multiplications
   (left_h @ right_hᵀ) for each batch element.  It is fused into a single
   Triton kernel that processes blocks of the i‑ and j‑dimensions while
   iterating over k in tiles.
5. Layer‑norm the hidden result, apply the output gate and a final linear
   projection back to the original channel dimension.

The kernel works for any batch size (≤2), sequence length (≤1024) and hidden
dimension (typically 128).  All tensors are assumed to be on the GPU.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the reduction over the sequence dimension.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # *const float*
    right_ptr,         # *const float*
    out_ptr,           # *float*
    B, N, H,           # tensor sizes (int32)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[...,k,:] * right[...,k,:] for each hidden slice."""
    pid_i = tl.program_id(0)          # block index for i dimension
    pid_j = tl.program_id(1)          # block index for j dimension
    pid_bh = tl.program_id(2)         # combined batch & hidden index

    # decode batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # i / j coordinates for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left block: shape (M, K) --------
        left_ptrs = left_ptr \
            + b * stride_left_b + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # -------- load right block transposed: shape (K, N) --------
        # we need right[k, j] (k as first dim) for the dot product.
        right_ptrs = right_ptr \
            + b * stride_right_b + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0).to(tl.float32)

        # -------- accumulate dot product over k --------
        # left: (M, K), right_T: (K, N) → result (M, N)
        acc += tl.dot(left, right_T)

    # -------- store result --------
    out_ptrs = out_ptr \
        + b * stride_out_b + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based implementation of the TriMul reduction.
    left/right: [B, N, N, H] (float32, contiguous)
    Returns: out of the same shape and dtype.
    """
    assert left.shape == right.shape, "left and right must have identical shape"
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # Tunable block sizes – 64 works well for most N ≤ 1024 on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=s_left[0],
        stride_left_i=s_left[1],
        stride_left_k=s_left[2],
        stride_left_h=s_left[3],
        stride_right_b=s_right[0],
        stride_right_j=s_right[1],   # same as i‑stride for the right tensor
        stride_right_k=s_right[2],
        stride_right_h=s_right[3],
        stride_out_b=s_out[0],
        stride_out_i=s_out[1],
        stride_out_j=s_out[2],
        stride_out_h=s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

# ----------------------------------------------------------------------
# Entry‑point expected by the autograder
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module with a fused Triton kernel.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32) or None
            - weights: dict of all model weights (float32 tensors)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm over the last channel dim (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # -> [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) – also linear, no bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] → broadcast over hidden dim
        mask_exp = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp * left_gate
        right = right * mask_exp * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # Ensure contiguous layout for Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction (batched matmul per hidden slice)
    # ------------------------------------------------------------------
    out_hidden = _trimul_triton(left, right)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate  # element‑wise gating
    out = F.linear(out, to_out_weight)   # back to [B, N, N, dim]

    return out
