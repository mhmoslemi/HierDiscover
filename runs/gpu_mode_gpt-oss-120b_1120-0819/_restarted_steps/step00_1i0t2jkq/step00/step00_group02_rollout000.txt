def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # Get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. Norm layer
    # Use torch.nn.functional.layer_norm on the last dim
    # for 4d tensor: shape [bs, seq_len, seq_len, dim]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # 2. Projection
    # Linear: out = x @ weight.T (no bias)
    # Using torch.nn.functional.linear
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

        right = right * mask

import triton
import triton.language as tl

@triton.jit
def outgoing_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # compute number of blocks in M and N dimensions
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # total batches = B * H
    total_batch = B * H

    # Determine batch index and block indices within matrix
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)

    # Batch coordinates
    b = batch_idx // H
    h = batch_idx % H

    # matrix tile coordinates
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # Starting coordinates of the tile.
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # Create range vectors.
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks for out-of-bounds (for final storage)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the current batch slice.
    left_ptr_batch = left_ptr + b * stride_left_b + h * stride_left_h
    right_ptr_batch = right_ptr + b * stride_right_b + h * stride_right_h
    out_ptr_batch = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_blocks_k):
        k_offset = kb * BLOCK_K
        # Offset the k dimension.
        offs_k_cur = k_offset + offs_k
        mask_k = offs_k_cur < N

        # Pointers for the current tile.
        left_tile_ptr = left_ptr_batch + (offs_i[:, None] * stride_left_i) + (offs_k_cur[None, :] * stride_left_k)
        right_tile_ptr = right_ptr_batch + (offs_k_cur[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_j)

        # Load tiles with mask to avoid reading OOB.
        left_tile = tl.load(left_tile_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_tile_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Core dot product (sum over k).
        acc += tl.dot(left_tile, right_tile)

    # Write back results.
    out_tile_ptr = out_ptr_batch + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_tile_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    # param extraction
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # Norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if not nomask
    if not config.get("nomask", True):
        mask = mask_tensor.unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Permute to [bs, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Allocate output
    out_perm = torch.empty_like(left_perm)

    # Compute core einsum via Triton
    _launch_trimul_kernel(left_perm, right_perm, out_perm, bs, hidden_dim, N)

    # Permute back
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # Out normalization
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul “outgoing” forward pass (AlphaFold3).

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C}:
1) Layer‑norm over the last channel dimension C.
2) Two linear projections → left/right with shape (B,N,N,H) where H = hidden_dim.
   Optionally a binary mask (B,N,N,1) is applied to the projected tensors.
3) Three gate linear layers (sigmoid) computed from the normalized X.
   left  = left  * left_gate
   right = right * right_gate
4) Core “tri‑mul” contraction:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is exactly a batched matrix‑multiply L_h @ R_hᵀ for each hidden
   slice h.  The heavy O(B·H·N³) work is performed by a custom Triton
   kernel that tiles the matrices (M×K)·(K×N) → (M×N) using BLOCK_M,
   BLOCK_N and BLOCK_K.
5) Layer‑norm over the hidden dimension (H) → to_out_norm.
6) Element‑wise gating with out_gate.
7) Final linear projection H → C → to_out.

All steps except the heavy contraction are done with PyTorch (still
GPU‑resident).  The kernel loads tiles in FP32 (or FP16 if the input is
half‑precision) and accumulates in FP32, producing an FP32 result.
The implementation supports both masked / un‑masked inputs and any
combination of the test shapes (B∈{1,2}, N∈{128,…,1024},
C∈{128,384}, H=128).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the “k” dimension of the TriMul.
# ----------------------------------------------------------------------
@triton.jit
def outgoing_trimul_kernel(
    left_ptr,               # const float*  [B, H, N, N]
    right_ptr,              # const float*  [B, H, N, N]
    out_ptr,                #       float*  [B, H, N, N]
    B, H, N,                # int32 scalars
    # strides for left (B,H,i,k)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    # strides for right (B,H,j,k) – note j is the 3rd dim, k the last
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    # strides for out (B,H,i,j)
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    # --------------------------------------------------------------
    BLOCK_M: tl.constexpr,  # tile size in i (rows of left)
    BLOCK_N: tl.constexpr,  # tile size in j (rows of right / cols of out)
    BLOCK_K: tl.constexpr,  # reduction tile size over k
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Grid decomposition: one program per (batch, hidden, tile_i, tile_j)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(N, BLOCK_M)          # tiles along i
    num_blocks_n = tl.cdiv(N, BLOCK_N)          # tiles along j
    total_batch = B * H

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    tile_idx  = pid %  (num_blocks_m * num_blocks_n)

    # batch dimensions
    b = batch_idx // H
    h = batch_idx % H

    # tile coordinates inside the (N,N) matrix
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx %  num_blocks_n

    # Global offsets for the current tile
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # ------------------------------------------------------------------
    # Offsets vectors
    # ------------------------------------------------------------------
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this (b, h) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(N, BLOCK_K)

    for kb in range(0, num_blocks_k):
        k_offset = kb * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < N

        # --------------------------------------------------------------
        # Load a tile of L (i,k) and R (k,j)   →  shapes (M,K) and (K,N)
        # --------------------------------------------------------------
        left_tile_ptr  = left_base  + (offs_i[:, None] * stride_left_i) \
                                    + (cur_k[None, :] * stride_left_k)
        right_tile_ptr = right_base + (cur_k[:, None] * stride_right_k) \
                                    + (offs_j[None, :] * stride_right_j)

        left_tile  = tl.load(left_tile_ptr,
                             mask=mask_i[:, None] & mask_k[None, :],
                             other=0.0)
        right_tile = tl.load(right_tile_ptr,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # --------------------------------------------------------------
        # Accumulate the dot product over the K‑dimension.
        # tl.dot supports (M×K)·(K×N) → (M×N)
        # --------------------------------------------------------------
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Write the result tile back to global memory
    # ------------------------------------------------------------------
    out_tile_ptr = out_base + (offs_i[:, None] * stride_out_i) \
                             + (offs_j[None, :] * stride_out_j)
    tl.store(out_tile_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel with sensible tiling.
# ----------------------------------------------------------------------
def _launch_trimul_kernel(left: torch.Tensor,
                          right: torch.Tensor,
                          out: torch.Tensor,
                          B: int, H: int, N: int):
    """Launch `outgoing_trimul_kernel` for tensors of shape (B,H,N,N)."""
    # Tile sizes – chosen to give good occupancy on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid dimensions.
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * num_blocks_m * num_blocks_n,)

    # Strides in element units (not bytes).
    s_lb, s_lh, s_li, s_lk = left.stride()
    s_rb, s_rh, s_rj, s_rk = right.stride()
    s_ob, s_oh, s_oi, s_oj = out.stride()

    outgoing_trimul_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,
        s_lb,
        s_lh,
        s_li,
        s_lk,
        s_rb,
        s_rh,
        s_rj,
        s_rk,
        s_ob,
        s_oh,
        s_oi,
        s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,      # good default for these tile sizes on H100
    )


# ----------------------------------------------------------------------
# Entry point required by the problem statement.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module using a Triton kernel.

    Arguments:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (C = dim)
            - mask_tensor  : torch.Tensor [B, N, N]  (ignored if config["nomask"]=True)
            - weights_dict : mapping of string → torch.nn.Parameter
            - config_dict  : must contain "dim" and "hidden_dim".
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Extract shapes & config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Layer‑norm over the feature dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # Optional mask – broadcast on the last dim.
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # Ensure mask is floating point for multiplication.
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left  = left  * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid of a linear projection of the normalized input)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates.
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton‑accelerated pairwise contraction.
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for easier indexing inside the kernel.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm   = torch.empty_like(left_perm)   # (B, H, N, N)

    _launch_trimul_kernel(left_perm, right_perm, out_perm, B, hidden_dim, N)

    # Back to the original layout (B,N,N,H).
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Hidden‑dim layer‑norm, gating by out_gate, final linear projection.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
