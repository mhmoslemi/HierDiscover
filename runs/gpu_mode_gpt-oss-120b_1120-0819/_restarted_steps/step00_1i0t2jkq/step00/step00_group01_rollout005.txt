```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bd,
    stride_rl, stride_ri, stride_rk, stride_rd,
    stride_ol, stride_oi, stride_ok, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, HIDDEN: tl.constexpr
):
    pid = tl.program_id(0)

    # compute batch and block_i, block_j
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_i * num_j)
    block_idx = pid % (num_i * num_j)
    pid_i = block_idx // (num_j)
    pid_j = block_idx % (num_j)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute actual block sizes (handles tails)
    cur_m = tl.minimum(BLOCK_M, N - i_start)
    cur_n = tl.minimum(BLOCK_N, N - j_start)

    # range indices
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_d = tl.arange(0, HIDDEN)

    # mask for out-of-bound i/j dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, HIDDEN), dtype=tl.float32)

    # Loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        # k range
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (M, K, H)
        # compute pointer offsets:
        left_ptrs = left_ptr + \
            batch * stride_bl + \
            offs_i[:, None, None] * stride_bi + \
            offs_k[None, :, None] * stride_bk + \
            offs_d[None, None, :] * stride_bd
        left_mask = mask_i[:, None, None] & mask_k[None, :, None]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)  # (M, K, H)

        # Load right tile: shape (N, K, H)
        right_ptrs = right_ptr + \
            batch * stride_rl + \
            offs_j[:, None, None] * stride_ri + \
            offs_k[None, :, None] * stride_rk + \
            offs_d[None, None, :] * stride_rd
        right_mask = mask_j[:, None, None] & mask_k[None, :, None]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)  # (N, K, H)

        # Now compute contribution for each k in the tile
        # iterate over k dimension (unrolled)
        for kk in range(0, BLOCK_K):
            # check if this k index is valid
            if kk >= BLOCK_K:
                break
            # optional mask for k tail
            k_valid = (k_start + kk) < N
            if not k_valid:
                continue

            # left_vec shape (M, H)
            left_vec = left_tile[:, kk, :]   # (M, H)
            # right_vec shape (N, H)
            right_vec = right_tile[:, kk, :] # (N, H)

            # Broadcast and accumulate
            # left_vec[:, None, :] shape (M, 1, H)
            # right_vec[None, :, :] shape (1, N, H)
            # Multiply
            prod = left_vec[:, None, :] * right_vec[None, :, :]  # (M, N, H)
            acc += prod
    # Write results
    # output offsets
    out_ptrs = out_ptr + \
        batch * stride_ol + \
        offs_i[:, None, None] * stride_oi + \
        offs_j[None, :, None] * stride_ok + \
        offs_d[None, None, :] * stride_od
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & (offs_d[None, None, :] < HIDDEN)
    tl.store(out_ptrs, acc, mask=out_mask)
```

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # layernorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1) # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
```

"""
TriMul (outgoing) – forward pass.

The heavy part of the module is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which we implement with a fused Triton kernel.
All other operations (LayerNorm, linear projections, gates and the final
output‑norm/linear) are performed with PyTorch for simplicity.
The kernel works on half‑precision tensors (to save memory/bandwidth) but
accumulates in float32 and finally returns a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel for the pairwise multiplication (the O(N³·D) part)
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, D,                                      # runtime sizes
    stride_l0, stride_l1, stride_l2, stride_l3,  # left strides
    stride_r0, stride_r1, stride_r2, stride_r3,  # right strides
    stride_o0, stride_o1, stride_o2, stride_o3,  # out strides
    BLOCK_M: tl.constexpr,                       # tile size in i‑dimension
    BLOCK_N: tl.constexpr,                       # tile size in j‑dimension
    BLOCK_K: tl.constexpr,                       # tile size in k‑dimension
    HIDDEN_DIM: tl.constexpr                      # compile‑time hidden dim (=D)
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    for a block of i (size BLOCK_M) and j (size BLOCK_N).

    Each program instance processes a (BLOCK_M × BLOCK_N) output tile.
    Accumulation is done in float32, while the inputs/outputs are half‑precision.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) Map 1‑D program id to (batch, block_i, block_j)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)          # number of i‑tiles
    num_j = tl.cdiv(N, BLOCK_N)          # number of j‑tiles
    batch = pid // (num_i * num_j)       # which batch element
    block_idx = pid % (num_i * num_j)
    pid_i = block_idx // num_j           # i‑tile id
    pid_j = block_idx % num_j            # j‑tile id

    # start indices of the tile (could be a tail)
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # -----------------------------------------------------------------
    # 2) Index ranges (including possible tails)
    # -----------------------------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_M)          # (M,)
    offs_j = j_start + tl.arange(0, BLOCK_N)          # (N,)
    offs_d = tl.arange(0, HIDDEN_DIM)                 # (D,)

    # masks for out‑of‑bounds (tails)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]       # (M,N)

    # -----------------------------------------------------------------
    # 3) Accumulator: one hidden‑dim vector per output element (i,j)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, HIDDEN_DIM), dtype=tl.float32)

    # -----------------------------------------------------------------
    # 4) Loop over the reduction dimension k
    # -----------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        # k‑range (may be a tail)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile : shape (M, K, D) --------------------
        left_ptrs = (left_ptr + batch * stride_l0
                     + offs_i[:, None, None] * stride_l1
                     + offs_k[None, :, None] * stride_l2
                     + offs_d[None, None, :] * stride_l3)
        left_mask = mask_i[:, None, None] & mask_k[None, :, None]
        left_tile = tl.load(left_ptrs, mask=left_mask,
                            other=tl.float16(0)).to(tl.float32)   # (M,K,D)

        # ----- load right tile : shape (N, K, D) -------------------
        right_ptrs = (right_ptr + batch * stride_r0
                      + offs_j[:, None, None] * stride_r1
                      + offs_k[None, :, None] * stride_r2
                      + offs_d[None, None, :] * stride_r3)
        right_mask = mask_j[:, None, None] & mask_k[None, :, None]
        right_tile = tl.load(right_ptrs, mask=right_mask,
                             other=tl.float16(0)).to(tl.float32)  # (N,K,D)

        # ----- accumulate contribution of this k‑tile ---------------
        # Unroll the small K‑loop (BLOCK_K is a constexpr)
        for kk in range(0, BLOCK_K):
            # left_vec : (M, D)   right_vec : (N, D)
            left_vec = left_tile[:, kk, :]         # (M,D)
            right_vec = right_tile[:, kk, :]       # (N,D)

            # outer product across i and j, keep hidden dim
            # (M,1,D) * (1,N,D)  -> (M,N,D)
            acc += left_vec[:, None, :] * right_vec[None, :, :]

    # -----------------------------------------------------------------
    # 5) Write the result (still half‑precision)
    # -----------------------------------------------------------------
    out_ptrs = (out_ptr + batch * stride_o0
                + offs_i[:, None, None] * stride_o1
                + offs_j[None, :, None] * stride_o2
                + offs_d[None, None, :] * stride_o3)

    # only store valid (i,j) positions
    store_mask = mask_ij[:, :, None]                     # (M,N,1) -> broadcast to (M,N,D)
    tl.store(out_ptrs, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------------- #
# Public entry‑point – called by the evaluation harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # input channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension (D)

    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B,N,N,dim)  float32

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Apply mask if it exists (mask shape: [B,N,N])
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(x.dtype).unsqueeze(-1)            # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating values (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Cast to half‑precision and allocate output buffer
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 6) Triton launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 16          # tuned for register pressure (you may change)
    BLOCK_N = 16
    BLOCK_K = 32

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * num_i * num_j
    grid = (total_blocks,)

    # launch
    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, hidden_dim,
        left_h.stride(0), left_h.stride(1), left_h.stride(2), left_h.stride(3),
        right_h.stride(0), right_h.stride(1), right_h.stride(2), right_h.stride(3),
        out_h.stride(0), out_h.stride(1), out_h.stride(2), out_h.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
    )

    # back to float32 for the remaining ops
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 7) Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
