    left_ptr, right_ptr,

    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,

    idx = idx // H

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    """
    Triton‑accelerated TriMul (outgoing) forward pass.
    1. Layer‑normalize the input.
    2. Project to hidden dimension and compute three sigmoidal gates.
    3. Fuse mask, left‑gate and right‑gate with a tiny Triton kernel (elementwise).
    4. Batched matrix‑multiply across the sequence dimension:
       out_{i,j} = Σ_k left_{i,k} · right_{j,k}   (per batch & hidden channel).
    5. Apply a second LayerNorm, the output gate and a final linear projection.
    All heavy arithmetic is performed in FP16; the final result is returned in FP32.
    """

    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # If mask is provided as None, replace with ones
    if mask_tensor is None:
        mask_tensor = torch.ones((B, N, N), dtype=input_tensor.dtype, device=input_tensor.device)

    # -------------------------------------------------------------------------
    # Helper to cast weights once
    # -------------------------------------------------------------------------
    def to_fp16(t):
        return t.to(torch.float16)

    # LayerNorm parameters
    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias = to_fp16(weights["norm.bias"])

    # Projection and gate weights
    left_proj_weight = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])
    left_gate_weight = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight = to_fp16(weights["out_gate.weight"])

    # Output LayerNorm and projection
    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias = to_fp16(weights["to_out_norm.bias"])
    to_out_weight = to_fp16(weights["to_out.weight"])

    # -------------------------------------------------------------------------
    # 1. First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # shape [B,N,N,dim], dtype float32

    # Cast to FP16 for the rest of the pipeline
    x_norm_h = x_norm.to(torch.float16)

    # -------------------------------------------------------------------------
    # 2. Projections and gates (all in FP16)
    # -------------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_norm_h, left_proj_weight)            # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm_h, right_proj_weight)          # [B,N,N,hidden_dim]

    left_gate = torch.nn.functional.linear(x_norm_h, left_gate_weight).sigmoid()  # [B,N,N,hidden_dim]
    right_gate = torch.nn.functional.linear(x_norm_h, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_h, out_gate_weight).sigmoid()    # [B,N,N,hidden_dim]

    # -------------------------------------------------------------------------
    # 3. Fuse mask + left/right gates with a tiny Triton kernel
    # -------------------------------------------------------------------------
    mask_h = mask_tensor.to(torch.float16)   # [B,N,N]

    left = torch.empty_like(left_proj, dtype=torch.float16, device=left_proj.device)
    right = torch.empty_like(right_proj, dtype=torch.float16, device=right_proj.device)

    total_elems = B * N * N * hidden_dim
    BLOCK = 1024
    grid = ( (total_elems + BLOCK - 1) // BLOCK, )

    @triton.jit

        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
        stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
        stride_m_b, stride_m_i, stride_m_k,

        start = pid * BLOCK_SIZE
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < (B * N * N * H)

        h = idx % H

        k = idx % N
        idx = idx // N

        off_r = b * stride_r_b + i * stride_r_i + k * stride_r_k + h * stride_r_h
        off_lg = b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + h * stride_lg_h
        off_rg = b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + h * stride_rg_h
        off_m = b * stride_m_b + i * stride_m_i + k * stride_m_k

        right_val = tl.load(right_ptr + off_r, mask=mask, other=tl.float16(0.0))
        left_gate_val = tl.load(left_gate_ptr + off_lg, mask=mask, other=tl.float16(0.0))
        right_gate_val = tl.load(right_gate_ptr + off_rg, mask=mask, other=tl.float16(0.0))
        m_val = tl.load(mask_ptr + off_m, mask=mask, other=tl.float16(0.0))

        out_l = left_val * m_val * left_gate_val
        out_r = right_val * m_val * right_gate_val

        tl.store(out_left_ptr + off_l, out_l, mask=mask)
        tl.store(out_right_ptr + off_r, out_r, mask=mask)

    _mask_gate_kernel[grid](
        left_proj, right_proj,
        left_gate, right_gate,
        mask_h,
        left, right,
        B, N, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_h.stride(0), mask_h.stride(1), mask_h.stride(2),
        BLOCK_SIZE=BLOCK
    )

    # -------------------------------------------------------------------------
    # 4. Multiplicative update via batched GEMM (left @ rightᵀ)
    # -------------------------------------------------------------------------
    # Rearrange to [B, hidden_dim, N, N] and make them contiguous for cuBLAS
    left_bc = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bc = right.permute(0, 3, 1, 2).contiguous()

    B_, H_, N_, _ = left_bc.shape
    left_mat = left_bc.view(B_ * H_, N_, N_)
    right_mat = right_bc.view(B_ * H_, N_, N_)

    # rightᵀ on the second dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # -------------------------------------------------------------------------
    # 5. Output LayerNorm, output gate and final projection
    # -------------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias
    )  # still FP16

    out_gated = out_norm * out_gate   # FP16 elementwise

    out_final = torch.nn.functional.linear(out_gated, to_out_weight)   # [B,N,N,dim] FP16

    # Return in Float32 as required by the interface
    return out_final.to(torch.float32)

def custom_kernel(data):
    """
    Triton‑accelerated “outgoing” TriMul forward.

    Steps
    -----
    1. Layer‑norm the 4‑D input (float32 → stable statistics).
    2. FP16 linear projections to the hidden dimension and three sigmoid gates.
    3. Fuse mask, left‑gate and right‑gate with a tiny Triton kernel (elementwise).
    4. Perform the multiplicative update:
           out[i,j,h] = Σ_k left[i,k,h] · right[j,k,h]
       implemented as a batched GEMM (left @ rightᵀ) on the (B·H) matrices.
    5. Second LayerNorm, output‑gate, and final linear projection back to `dim`.
       All heavy arithmetic stays in FP16; the result is returned in FP32.
    """
    import torch
    import triton
    import triton.language as tl

    # --------------------------------------------------------------------- #
    # 0. unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # mask may be omitted – treat as all‑ones
    if mask_tensor is None:
        mask_tensor = torch.ones((B, N, N), dtype=input_tensor.dtype,
                                 device=input_tensor.device)

    # --------------------------------------------------------------------- #
    # 1. cast weights once to FP16 (the GPU kernels work in half precision)
    # --------------------------------------------------------------------- #
    def to_fp16(t): return t.to(torch.float16)

    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])

    left_proj_weight  = to_fp16(weights["left_proj.weight"])
    right_proj_weight = to_fp16(weights["right_proj.weight"])

    left_gate_weight  = to_fp16(weights["left_gate.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    out_gate_weight   = to_fp16(weights["out_gate.weight"])

    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])
    to_out_weight      = to_fp16(weights["to_out.weight"])

    # --------------------------------------------------------------------- #
    # 2. First LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                                 # [B, N, N, dim]  (float32)
    x_norm_h = x_norm.to(torch.float16)   # cast once for the rest

    # --------------------------------------------------------------------- #
    # 3. Linear projections + sigmoid gates (all FP16)
    # --------------------------------------------------------------------- #
    left_proj  = torch.nn.functional.linear(x_norm_h, left_proj_weight)   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x_norm_h, right_proj_weight)

    left_gate  = torch.nn.functional.linear(x_norm_h, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_h, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm_h, out_gate_weight).sigmoid()

    # --------------------------------------------------------------------- #
    # 4. Fuse mask + gates with a Triton kernel (elementwise)
    # --------------------------------------------------------------------- #
    mask_h = mask_tensor.to(torch.float16)   # [B,N,N]

    left  = torch.empty_like(left_proj,  dtype=torch.float16, device=left_proj.device)
    right = torch.empty_like(right_proj, dtype=torch.float16, device=right_proj.device)

    total = B * N * N * hidden_dim
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    @triton.jit
    def _mask_gate_kernel(
        # pointers
        left_ptr, right_ptr,
        left_gate_ptr, right_gate_ptr,
        mask_ptr,
        out_left_ptr, out_right_ptr,
        # run‑time scalars
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
        stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
        stride_m_b, stride_m_i, stride_m_k,
        # compile‑time
        BLOCK_SIZE: tl.constexpr
    ):
        pid = tl.program_id(0)
        start = pid * BLOCK_SIZE
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < (B * N * N * H)

        # -------- decode linear index --------
        idx = offs
        h = idx % H
        idx = idx // H
        k = idx % N
        idx = idx // N
        i = idx % N
        b = idx // N

        # -------- address computation --------
        off_l  = b * stride_l_b  + i * stride_l_i  + k * stride_l_k  + h * stride_l_h
        off_r  = b * stride_r_b  + i * stride_r_i  + k * stride_r_k  + h * stride_r_h
        off_lg = b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + h * stride_lg_h
        off_rg = b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + h * stride_rg_h
        off_m  = b * stride_m_b  + i * stride_m_i  + k * stride_m_k

        # -------- load, apply mask & gates --------
        lv = tl.load(left_ptr      + off_l,  mask=mask, other=tl.float16(0.0))
        rv = tl.load(right_ptr     + off_r,  mask=mask, other=tl.float16(0.0))
        lgv = tl.load(left_gate_ptr + off_lg, mask=mask, other=tl.float16(0.0))
        rgv = tl.load(right_gate_ptr+ off_rg, mask=mask, other=tl.float16(0.0))
        mv = tl.load(mask_ptr      + off_m,  mask=mask, other=tl.float16(0.0))

        out_l = lv * mv * lgv
        out_r = rv * mv * rgv

        tl.store(out_left_ptr  + off_l, out_l, mask=mask)
        tl.store(out_right_ptr + off_r, out_r, mask=mask)

    _mask_gate_kernel[grid](
        left_proj, right_proj,
        left_gate, right_gate,
        mask_h,
        left, right,
        B, N, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_h.stride(0), mask_h.stride(1), mask_h.stride(2),
        BLOCK_SIZE=BLOCK
    )

    # --------------------------------------------------------------------- #
    # 5. Multiplicative update: batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------------- #
    # bring the hidden dimension to the batch axis for cuBLAS
    left_bc  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bc = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH, Hdim, _, _ = left_bc.shape   # BH = B*H
    left_mat  = left_bc.view(B * hidden_dim, N, N)          # [B*H, N, N]
    right_mat = right_bc.view(B * hidden_dim, N, N)         # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm, output gate, final linear (all FP16)
    # --------------------------------------------------------------------- #
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias
    )   # FP16

    out_gated = out_norm * out_gate   # elementwise FP16

    out_final = torch.nn.functional.linear(out_gated, to_out_weight)   # [B,N,N,dim] FP16

    # --------------------------------------------------------------------- #
    # 7. Return FP32 tensor as required by the interface
    # --------------------------------------------------------------------- #
    return out_final.to(torch.float32)
