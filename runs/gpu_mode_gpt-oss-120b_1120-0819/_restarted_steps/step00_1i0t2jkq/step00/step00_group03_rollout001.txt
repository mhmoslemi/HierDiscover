    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']

    x_norm = torch.nn.functional.layer_norm(
        input_tensor,

    )

    stride_br, stride_jr, stride_kr, stride_hr,

"""
custom_kernel - Outgoing TriMul (AlphaFold3)

This implementation follows the PyTorch reference of the TriMul module but
off‑loads the cubic reduction (the “einsum” over the sequence dimension) to a
hand‑written Triton kernel.  The surrounding LayerNorms, linear projections,
and sigmoid gates are kept in PyTorch for simplicity.  The kernel computes

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for every batch `b`, output channel `h` (the hidden dimension), and pair of
sequence positions `i, j`.  The reduction is tiled over the sequence dimension
(`K`) and over the hidden dimension to keep register pressure low.  The kernel
operates on `float16` inputs (cast from the original `float32`) and accumulates
in `float32` for numerical stability.  The final result is normalised,
gated and projected back to the original `dim` using the provided weights.

The kernel is launched with a 1‑D grid covering:
    batch × i‑tile × j‑tile × hidden‑tile
so every program instance computes a sub‑block of the output tensor
`[B, N, N, hidden_dim]`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel for the big O(N³) reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,          # *half  [B,N,N,H]
    right_ptr,         # *half  [B,N,N,H]
    out_ptr,           # *float [B,N,N,H]
    B, N, H,           # problem size
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_jr, stride_kr, stride_hr,  # right strides
    stride_bo, stride_io, stride_jo, stride_ho,  # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    NUM_I: tl.constexpr,
    NUM_J: tl.constexpr,
    NUM_HB: tl.constexpr,
):
    """
    Each program instance computes a tile of size
        (BLOCK_I, BLOCK_J, BLOCK_H)   within the output tensor.
    The hidden dimension is also tiled; NUM_HB = ceil(H / BLOCK_H).
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decode the 4‑D grid index: batch, i‑tile, j‑tile, hidden‑tile
    # ------------------------------------------------------------------
    b = pid // (NUM_I * NUM_J * NUM_HB)
    tmp = pid % (NUM_I * NUM_J * NUM_HB)
    i_tile = tmp // (NUM_J * NUM_HB)
    j_tile = (tmp // NUM_HB) % NUM_J
    h_tile = tmp % NUM_HB

    i0 = i_tile * BLOCK_I
    j0 = j_tile * BLOCK_J
    h0 = h_tile * BLOCK_H

    # ------------------------------------------------------------------
    # Accumulator (float32) for the tile
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Iterate over the K dimension in blocks
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k0 = k_start

        # --------------------------------------------------------------
        # Load a block from left: shape (I, K, H)
        # --------------------------------------------------------------
        i = tl.arange(0, BLOCK_I)
        k = tl.arange(0, BLOCK_K)
        h = tl.arange(0, BLOCK_H)

        offs_i = i0 + i[:, None, None]          # (I,1,1)
        offs_k = k0 + k[None, :, None]          # (1,K,1)
        offs_h = h0 + h[None, None, :]          # (1,1,H)

        mask_i = offs_i < N
        mask_k = offs_k < N
        mask_h = offs_h < H
        mask_ik = mask_i & mask_k                # (I,K,1)
        mask_ik_h = mask_ik & mask_h              # (I,K,H)

        left_ptrs = (left_ptr
                     + b * stride_bl
                     + offs_i * stride_il
                     + offs_k * stride_kl
                     + offs_h * stride_hl)
        left_tile = tl.load(left_ptrs, mask=mask_ik_h, other=0.0)   # half

        # --------------------------------------------------------------
        # Load a block from right: shape (J, K, H)
        # --------------------------------------------------------------
        j = tl.arange(0, BLOCK_J)
        offs_j = j0 + j[:, None, None]          # (J,1,1)

        mask_j = offs_j < N
        mask_jk = mask_j & mask_k                # (J,K,1)
        mask_jk_h = mask_jk & mask_h              # (J,K,H)

        right_ptrs = (right_ptr
                      + b * stride_br
                      + offs_j * stride_jr
                      + offs_k * stride_kr
                      + offs_h * stride_hr)
        right_tile = tl.load(right_ptrs, mask=mask_jk_h, other=0.0)  # half

        # --------------------------------------------------------------
        # Compute dot‑products for each hidden sub‑block
        #   acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]
        # --------------------------------------------------------------
        for hh in range(0, BLOCK_H):
            # slice (I,K) and (J,K) for this hidden channel
            l = left_tile[:, :, hh]      # (I,K)
            r = right_tile[:, :, hh]     # (J,K)
            # dot with transposed right → (I,J)
            acc[:, :, hh] += tl.dot(l, r, trans_b=True)

    # ------------------------------------------------------------------
    # Store the accumulated tile back to global memory
    # ------------------------------------------------------------------
    i = tl.arange(0, BLOCK_I)
    j = tl.arange(0, BLOCK_J)
    h = tl.arange(0, BLOCK_H)

    offs_i = i0 + i[:, None, None]
    offs_j = j0 + j[None, :, None]
    offs_h = h0 + h[None, None, :]

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H
    store_mask = mask_i & mask_j & mask_h   # (I,J,H)

    out_ptrs = (out_ptr
                + b * stride_bo
                + offs_i * stride_io
                + offs_j * stride_jo
                + offs_h * stride_ho)

    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Python entry point – performs the full forward pass
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (x, mask, weights, config)
            x      : [B, N, N, dim]      (float32)
            mask   : [B, N, N]           (bool/int or None)
            weights: dict of torch tensors containing the model parameters
            config : dict with keys "dim" and "hidden_dim"
    Returns:
        out    : [B, N, N, dim] (float32)
    """
    # ------------------- unpack -------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ---------- LayerNorm on the input ----------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- Linear projections ----------
    # (B,N,N,hidden_dim) = x_norm @ W.T
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])   # bias=False
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ---------- Gates (sigmoid) ----------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # ---------- Apply mask if present ----------
    if mask is not None:
        # make mask broadcastable: [B,N,N,1]
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ---------- Apply element‑wise gates ----------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy reduction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output of the reduction (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # ------------------------------------------------------------------
    # Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 16                     # hidden‑dim is tiled

    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    num_hb = (hidden_dim + BLOCK_H - 1) // BLOCK_H

    total_programs = B * num_i * num_j * num_hb
    grid = (total_programs,)

    # Strides in *elements* (not bytes)
    stride_bl = left_h.stride(0)
    stride_il = left_h.stride(1)
    stride_kl = left_h.stride(2)
    stride_hl = left_h.stride(3)

    stride_br = right_h.stride(0)
    stride_jr = right_h.stride(1)
    stride_kr = right_h.stride(2)
    stride_hr = right_h.stride(3)

    stride_bo = out.stride(0)
    stride_io = out.stride(1)
    stride_jo = out.stride(2)
    stride_ho = out.stride(3)

    # Launch the kernel
    tri_mul_einsum_kernel[grid](
        left_h, right_h, out,
        B, N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_jr, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        NUM_I=num_i,
        NUM_J=num_j,
        NUM_HB=num_hb,
        num_warps=8,   # H100 prefers more warps
    )

    # ------------------- post‑processing -------------------
    # LayerNorm on the reduced tensor (hidden_dim)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the final gating (broadcast over the last dim)
    out_gated = out_norm * out_gate

    # Final linear projection back to `dim`
    final = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return final
