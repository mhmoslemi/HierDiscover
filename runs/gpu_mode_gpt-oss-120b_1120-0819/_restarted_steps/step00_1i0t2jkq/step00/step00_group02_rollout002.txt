    device = input_tensor.device
    dtype = input_tensor.dtype

    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (D,), weight=norm_weight, bias=norm_bias)

    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

        mask_tensor = torch.ones((B,N,N), dtype=x.dtype, device=device)
    else:

    mask_tensor = mask_tensor.unsqueeze(-1) # shape B,N,N,1

    # fuse gating and mask using Triton (returns new tensors)
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # define helper to launch kernel
    fused_gate_mask(left, left_gate, mask_tensor, left_fused)
    fused_gate_mask(right, right_gate, mask_tensor, right_fused)

    # cast to half for matmul (optional)
    left_fused_h = left_fused.to(torch.float16)
    right_fused_h = right_fused.to(torch.float16)

    out_b = torch.bmm(left_b, right_b.transpose(1,2))  # (B*H, N, N) in float16

    out = out_b.view(B, hidden, N, N).permute(0,2,3,1).to(torch.float32) # back to float32

def custom_kernel(data):
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    B, N, _, D = x.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(x, (D,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # mask handling
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        # ensuring mask dtype matches
        mask_tensor = mask.to(dtype)
    # unsqueeze to broadcast over hidden dim
    mask_tensor = mask_tensor.unsqueeze(-1)  # (B,N,N,1)

    # Fuse mask and gating using triton
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    fused_gate_mask(left, left_gate, mask_tensor, left_fused)
    fused_gate_mask(right, right_gate, mask_tensor, right_fused)

    # half precision for matmul
    left_h = left_fused.to(torch.float16)
    right_h = right_fused.to(torch.float16)

    # Reshape for batched matmul
    left_b = left_h.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, N)
    right_b = right_h.permute(0, 3, 1, 2).contiguous().view(-1, N, N) # (B*H, N, N)

    # Compute C = A @ B^T for each batch channel
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))  # (B*H, N, N), dtype float16

    # Restore shape
    out = out_b.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)  # (B,N,N,H)

    # Output normalization
    out = torch.nn.functional.layer_norm(out,
                                         (hidden,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"])

    # Apply out gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) – fast forward implementation.

The module receives a 4‑D tensor `x` of shape [B, N, N, D] and a mask
[B, N, N].  After a LayerNorm, linear projections and sigmoid gates
the core computation is

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]          (1)

which is exactly a batched matrix multiplication
    left_h @ right_hᵀ  for each hidden channel h.
The heavy O(N³·H) work is performed with an efficient
`torch.bmm` (cuBLAS) after fusing the element‑wise mask & gate
operations with a small Triton kernel.  The kernel loads a scalar
from the input tensor, its gate and the binary mask, multiplies them
and stores the result, thus reducing memory passes.

The overall steps are:
1. Layer‑norm over the input channel dimension.
2. Linear projections (`left_proj`, `right_proj`) and gate projections
   (`left_gate`, `right_gate`, `out_gate`).
3. Fuse (`tensor * gate * mask`) with a Triton kernel.
4. Batched GEMM `C = A @ Bᵀ` (A=left, B=right) using `torch.bmm`.
5. Output‑norm, out‑gate multiplication and final linear projection.

The implementation works for all test configurations (different B, N,
D, hidden_dim and optional mask) and runs in < 1 ms for the listed
sizes on an H100.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F
from typing import Tuple, Dict

# -------------------------------------------------------------------------
# Triton kernel: fuse element‑wise mask & gate into the projected tensor.
# -------------------------------------------------------------------------
@triton.jit
def _fuse_mask_gate_kernel(
    tensor_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_t0, stride_t1, stride_t2, stride_t3,   # tensor strides
    stride_g0, stride_g1, stride_g2, stride_g3,   # gate strides
    stride_m0, stride_m1, stride_m2, stride_m3,   # mask strides (last stride = 0 after unsqueeze)
    BLOCK_SIZE: tl.constexpr,
):
    """Compute out = tensor * gate * mask  (mask broadcast over hidden dim)."""
    pid = tl.program_id(0)
    total = B * N * N * H
    idx = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)  # linear indices
    mask = idx < total

    # decompose linear index into (b, i, j, h)
    i0 = idx % H
    idx = idx // H
    j0 = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # pointer offsets
    off_t = (
        b * stride_t0 + i * stride_t1 + j0 * stride_t2 + i0 * stride_t3
    )
    off_g = (
        b * stride_g0 + i * stride_g1 + j0 * stride_g2 + i0 * stride_g3
    )
    off_m = (
        b * stride_m0 + i * stride_m1 + j0 * stride_m2 + i0 * stride_m3
    )

    # load, multiply, store
    t = tl.load(tensor_ptr + off_t, mask=mask, other=0.0)
    g = tl.load(gate_ptr + off_g, mask=mask, other=0.0)
    m = tl.load(mask_ptr + off_m, mask=mask, other=1.0)  # default mask=1 (no masking)

    tl.store(out_ptr + off_t, t * g * m, mask=mask)


def _fuse_mask_gate(tensor: torch.Tensor,
                    gate: torch.Tensor,
                    mask: torch.Tensor,
                    out: torch.Tensor,
                    block_size: int = 256) -> None:
    """
    Wrapper for the Triton kernel.  All tensors must be on the same CUDA device
    and have matching shapes: tensor/ gate = [B,N,N,H], mask = [B,N,N,1].
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    grid = (triton.cdiv(total, block_size),)

    # strides are python ints – Triton expects them as arguments
    _fuse_mask_gate_kernel[grid](
        tensor,
        gate,
        mask,
        out,
        B,
        N,
        H,
        *tensor.stride(),
        *gate.stride(),
        *mask.stride(),
        BLOCK_SIZE=block_size,
    )

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3. Mask handling – broadcast to hidden dim and fuse with gates
    # -----------------------------------------------------------------
    if mask is None or nomask:
        mask_tensor = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)

    # unsqueeze to make shape [B, N, N, 1]; the hidden‑dim stride becomes 0 => broadcasting
    mask_tensor = mask_tensor.unsqueeze(-1)

    # Allocate fused tensors
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Triton kernel fuses (tensor * gate * mask) in one pass
    _fuse_mask_gate(left, left_gate, mask_tensor, left_fused)
    _fuse_mask_gate(right, right_gate, mask_tensor, right_fused)

    # -----------------------------------------------------------------
    # 4. Core contraction: out = Σ_k left[...,k] * right[...,k]ᵀ
    #    Implemented as batched GEMM using torch.bmm (cuBLAS)
    # -----------------------------------------------------------------
    # Use half‑precision for the heavy GEMM (Tensor Cores)
    left_h = left_fused.to(torch.float16)
    right_h = right_fused.to(torch.float16)

    # reshape to (B*hidden, N, N) → each row‑matrix multiplies with its transpose
    left_b = left_h.permute(0, 3, 1, 2).contiguous().view(-1, N, N)    # (B*H, N, N)
    right_b = right_h.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # (B*H, N, N)

    # C = A @ Bᵀ  (shape (B*H, N, N))
    out_b = torch.bmm(left_b, right_b.transpose(1, 2))

    # Restore original layout [B, N, N, hidden] (back to float32)
    out = out_b.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 5. Output normalisation, out‑gate and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
