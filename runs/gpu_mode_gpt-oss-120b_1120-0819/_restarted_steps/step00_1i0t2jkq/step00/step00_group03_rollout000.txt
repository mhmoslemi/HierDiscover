def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    # pid for batch dimension?
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_rem = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_rem // tl.cdiv(N, BLOCK_N)
    pid_n = pid_rem % tl.cdiv(N, BLOCK_N)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

"""
TriMul forward (outgoing) implementation using Triton.
----------------------------------------------------

The operation works on a 4‑D tensor `x` of shape [B, N, N, D] (D = dim).
1. Layer‑Norm over the last dimension.
2. Linear projections to a hidden dimension H (no bias):
   left = x·W_left ,   right = x·W_right
3. Optional pair‑wise mask applied to both projections.
4. Gating: left *= sigmoid(x·W_left_gate), right *= sigmoid(x·W_right_gate)
5. Core “triangle‑multiplicative” update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel:
          left_h (N×N)  @  right_hᵀ (N×N)  →  out_h (N×N)
   We implement this reduction with a Triton kernel that processes
   all (B·H) batches in parallel.
6. Layer‑Norm over the hidden dimension, gated by `out_gate`,
   and a final linear projection back to the original dimension.

The heavy reduction (step 5) is executed in FP16 for bandwidth while
accumulating in FP32 for numerical stability. All other steps are
performed with native PyTorch ops (they are cheap compared to the
N³ reduction). The function expects:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
and returns the transformed tensor of shape [B, N, N, D].
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_am, stride_ak,          # A: (batch, M, K)
    stride_batch_b, stride_bk, stride_bn,          # B: (batch, K, N)
    stride_batch_c, stride_cm, stride_cn,          # C: (batch, M, N)
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B  for a batch of matrices.
    A: (batch, M, K)   B: (batch, K, N)   C: (batch, M, N)
    All tensors are stored in row‑major order.
    """
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # ----- block offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- pointers for the current block (including batch offset) -----
    a_ptrs = a_ptr + batch_id * stride_batch_a \
                   + offs_m[:, None] * stride_am \
                   + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_batch_b \
                   + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension -----
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)          # absolute k indices
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (cur_k < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(cur_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # advance pointers to next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----- write result -----
    c_ptrs = c_ptr + batch_id * stride_batch_c \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    TriMul outgoing forward pass.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, H]

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; cast to same dtype as x and unsqueeze
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core triangle multiplicative update:
    #   out[h] = left[h] @ right[h].T   for each hidden channel h
    # We rearrange tensors so that each (batch, h) pair becomes a
    # separate matrix‑multiply and execute it with a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, N, N, H) -> (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # note the transpose on the last two dims

    # flatten batch and hidden dimensions
    batch_h = left_perm.shape[0] * left_perm.shape[1]   # B * H
    left_flat  = left_perm.view(batch_h, N, N).to(torch.float16)
    right_flat = right_perm.view(batch_h, N, N).to(torch.float16)

    # output buffer (float16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H] (float32 for the following ops)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # Output layer‑norm over hidden dimension, gating and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
