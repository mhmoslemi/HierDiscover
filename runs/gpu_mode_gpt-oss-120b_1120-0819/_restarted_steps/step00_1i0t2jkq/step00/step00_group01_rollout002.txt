   pid_n = tl.program_id(2)

import triton
import triton.language as tl

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bi, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    MASK_VAL: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create masks for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile: shape (BLOCK_M, BLOCK_K)

def custom_kernel(data):
    # Unpack
    x, mask, weights, config = data

    dim = config["dim"]; hidden = config["hidden_dim"]; maybe also config specify "nomask".

    # Use torch functions for LayerNorm:
    # Equivalent to nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)
    # but careful: shape includes [B,N,N,dim]; per last dimension.

    # Linear projections:
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape B,N,N,hidden
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Mask: shape [B,N,N]; unsqueeze to (...,1) for broadcasting.
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1) # shape B,N,N,1
        left = left * mask
        right = right * mask

    # Gates:
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Optionally cast to fp16
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute for gemm: (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right_T = right.permute(0, 3, 2, 1).contiguous()  # transpose inner dims

    # Flatten batch and hidden: shape (B*hidden, N, N)
    B, H, N, _ = left.shape
    left_flat = left.view(B*H, N, N)
    right_T_flat = right_T.view(B*H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16) # same dtype

    # Call triton kernel
    tri_mul_matmul_kernel[(B*H, (N+BLOCK_M-1)//BLOCK_M, (N+BLOCK_N-1)//BLOCK_N)](
        left_flat, right_T_flat, out_flat,
        B*H, N, N, N,
        stride of left..., etc.
    )
    # reshape back: (B, hidden, N, N)
    out = out_flat.view(B, H, N, N)

    # Permute back to (B,N,N,hidden)
    out = out.permute(0, 2, 3, 1)

    # Convert to float32 for following LayerNorm and linear. We can cast out to float32 for final steps.
    out = out.to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate

    # Linear to output dimension
    out = torch.nn.functional.linear(out, to_out_weight) # shape B,N,N,dim

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bi, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_idx * stride_ah + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile: (BLOCK_K, BLOCK_N) where B is already transposed (K, N)
        b_ptrs = b_ptr + batch_idx * stride_bh + offs_k[:, None] * stride_bi + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result tile
    c_ptrs = c_ptr + batch_idx * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated outgoing TriMul implementation.
    Operates on input [B, N, N, dim] and returns same shape.
    Heavy tensor contraction (einsum) is performed in a fused Triton kernel.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Optional mask presence
    use_mask = not config.get("nomask", False)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --- LayerNorm on the last dimension ---
    # torch.nn.functional.layer_norm expects normalized_shape = (dim,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # --- Linear projections (no bias) ---
    # shape: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_weight)   # (out_features, in_features) = (hidden_dim, dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # --- Mask handling ---
    if use_mask and mask_tensor is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # shape [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --- Gating ---
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Cast gates to same dtype as projected tensors for elementwise multiplication
    left_gate = left_gate.to(left.dtype)
    right_gate = right_gate.to(right.dtype)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --- Cast to half precision for the heavy contraction ---
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden_dim in front: [B, hidden, N, N]
    # left:  (B, N, N, H) -> (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    # right needs a transpose on the last two dims before permute
    right = right.permute(0, 3, 2, 1).contiguous()   # now shape (B, H, N, N) where inner dim is transposed

    B, H, N, _ = left.shape
    batch = B * H

    # Flatten batch and hidden into a single dimension for the kernel
    left_flat = left.view(batch, N, N)
    right_flat = right.view(batch, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # output in fp16

    # Strides (in element units)
    stride_ah = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bi = right_flat.stride(1)   # K dimension after transpose
    stride_bj = right_flat.stride(2)   # N dimension

    stride_ch = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Block sizes – tuned for H100 (FP16, up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 32 works well with FP16 on H100

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Launch Triton kernel
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch, N, N, N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bi, stride_bj,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B, N, N, hidden]

    # Cast back to float32 for subsequent layers
    out = out.to(torch.float32)

    # --- to_out_norm (LayerNorm over hidden_dim) ---
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Multiply by out_gate (already float32)
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out
```

"""
Outgoing TriMul (TriMul v2) – Triton‐accelerated core contraction.

The heavy part of the module is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is mathematically a batched matrix multiplication
    out = left @ rightᵀ
over the (N × N) slices for every hidden dimension d.
The kernel below computes this contraction in FP16 with a
float‑32 accumulator, using a classic tiling strategy
(BLOCK_M‑BLOCK_N tiles, inner reduction BLOCK_K).  The rest
of the TriMul forward pass (LayerNorms, linear projections,
gates, masking) is performed in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    a_ptr,                # left  [batch, M, K]   (fp16)
    b_ptr,                # rightᵀ[batch, K, N]   (fp16)
    c_ptr,                # out   [batch, M, N]   (fp16)
    batch, M, N, K,       # problem sizes
    stride_ah, stride_ai, stride_ak,   # a strides: batch, M, K
    stride_bh, stride_bi, stride_bj,   # b strides: batch, K, N
    stride_ch, stride_ci, stride_cj,   # c strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N) for many batches."""
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------ tile offsets ------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (better precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------ reduction over K ------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + batch_idx * stride_ah \
                 + offs_m[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile [BLOCK_K, BLOCK_N]  (already transposed)
        b_ptrs = b_ptr + batch_idx * stride_bh \
                 + offs_k[:, None] * stride_bi \
                 + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # ------------ write back ------------
    c_ptrs = c_ptr + batch_idx * stride_ch \
             + offs_m[:, None] * stride_ci \
             + offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc.to(tl.float16),        # store in FP16 (output later cast to FP32)
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ---------------- unpack ----------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False)

    # ---------------- load weights ----------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------- LayerNorm (dim) -------------------
    # Normalizes over the last dim = `dim`
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------- Linear projections (no bias) -------------------
    # Output shape: [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_weight)   # (hidden_dim, dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------- Optional masking -------------------
    if use_mask and mask_tensor is not None:
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------- Gating (sigmoid) -------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Cast gating to the same dtype as projections (later fp16)
    left_gate = left_gate.to(left.dtype)
    right_gate = right_gate.to(right.dtype)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------- Cast to FP16 for the heavy contraction -------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Bring hidden dimension in front: [B, hidden, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()            # (B, H, N, N)
    # Transpose inner N‑N for the matmul: rightᵀ → (B, H, K, N) where K == N
    right = right.permute(0, 3, 2, 1).contiguous()          # (B, H, N, N)

    B, H, N, _ = left.shape
    batch = B * H

    # Flatten batch & hidden into a single batch dimension for the kernel
    left_flat = left.view(batch, N, N)
    right_flat = right.view(batch, N, N)

    # Output buffer (FP16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------- Strides (in elements) -------------------
    stride_ah = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bi = right_flat.stride(1)   # K‑dimension after transpose
    stride_bj = right_flat.stride(2)   # N‑dimension

    stride_ch = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # ------------------- Triton launch parameters -------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # empirically good for FP16 on H100

    grid = (
        batch,                                      # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,               # M‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,               # N‑tiles
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch, N, N, N,
        stride_ah, stride_ai, stride_ak,
        stride_bh, stride_bi, stride_bj,
        stride_ch, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------- Reshape back to original layout -------------------
    # out_flat: [batch, N, N] → [B, H, N, N] → [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to FP32 for the rest of the pipeline

    # ------------------- to_out_norm (LayerNorm over hidden_dim) -------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Multiply by the output gate (already float32)
    out = out * out_gate

    # ------------------- Final linear projection back to `dim` -------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # (dim, hidden_dim)

    return out
