                   stride_left_b, stride_left_i, stride_left_k, stride_left_d,
                   stride_right_b, stride_right_j, stride_right_k, stride_right_d,

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # combined batch and channel

    d = pid_z % D

    # compute block start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets in each dimension
    offs_m = i0 + tl.arange(0, BLOCK_M)
    offs_n = j0 + tl.arange(0, BLOCK_N)

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    # Get arguments
    input_tensor, mask, weights, config = data
    # config contains dim and hidden_dim, maybe others.
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # (optional spec for epsilon)
    eps = config.get("eps", 1e-5)

    # Weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Step 1: layer norm on input x
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Step 2: linear projections
    left = torch.nn.functional.linear(x, left_proj_weight) # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Step 3: optional mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Step 4: gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Step 5: Triangular multiplicative update using Triton
    B, N, _, _ = left.shape  # B,N,N,hidden_dim

    out = torch.empty_like(left)  # allocate output [B,N,N,hidden_dim]

    # define Triton kernel
    # We'll need to compile the kernel here
    # We'll use trailing code to launch

    # Compute strides (in elements)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()

    # Define block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Compute grid
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * hidden_dim)

    # Launch kernel
    tri_mul_kernel[grid](
        out_ptr=out,
        left_ptr=left,
        right_ptr=right,
        B=B,
        N=N,
        D=hidden_dim,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,

    final = torch.nn.functional.linear(out, to_out_weight)

This implementation follows the reference PyTorch code but accelerates the

    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    Grid:
        axis 0 -> i‑tile (rows)
        axis 1 -> j‑tile (cols)
        axis 2 -> combined (batch * hidden) dimension.
    """
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch * hidden index

    # decode batch and hidden channel from the merged dimension
    b = pid_z // D                    # batch index
    d = pid_z % D                     # hidden channel index

    # starting offsets of the current tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range vectors within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks for the i and j dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,d) slice
    base_out   = out_ptr   + b * stride_out_b   + d * stride_out_d
    base_left  = left_ptr  + b * stride_left_b  + d * stride_left_d
    base_right = right_ptr + b * stride_right_b + d * stride_right_d

    # accumulator in fp32 (regardless of input dtype)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k in tiles of size BLOCK_K
    for k_offset in range(0, N, BLOCK_K):
        # ---- left[b,i,k,d] -------------------------------------------------
        # pointer to the (i,k) tile
        left_ptrs = base_left + (offs_i[:, None]) * stride_left_i \
                                 + (k_offset + offs_k)[None, :] * stride_left_k
        # mask for out‑of‑bounds elements in this tile
        mask_left = mask_i[:, None] & ((k_offset + offs_k) < N)[None, :]
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)

        # ---- right[b,j,k,d] ------------------------------------------------
        # we need the (k,j) tile because the contraction is over k.
        right_ptrs = base_right + (k_offset + offs_k)[:, None] * stride_right_k \
                                   + (offs_j[None, :]) * stride_right_j
        mask_right = ((k_offset + offs_k) < N)[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # matrix‑multiply the two tiles and accumulate
        acc += tl.dot(left_tile, right_tile)

    # ---- write the result -------------------------------------------------
    out_ptrs = base_out + (offs_i[:, None]) * stride_out_i \
                        + (offs_j[None, :]) * stride_out_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters (see code)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    import torch
    import triton

    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # retrieve weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (left/right)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the channel dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        m = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core triangular multiplication (Triton)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                      # B, N, N, hidden_dim
    out = torch.empty_like(left)                  # allocate output

    # compute strides (in number of elements) for the three tensors
    s_out_b, s_out_i, s_out_j, s_out_d = out.stride()
    s_l_b,   s_l_i,   s_l_k,   s_l_d   = left.stride()
    s_r_b,   s_r_i,   s_r_k,   s_r_d   = right.stride()

    # block sizes – chosen to give good occupancy on H100
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # launch the Triton kernel
    tri_mul_kernel[grid](
        out_ptr=out,
        left_ptr=left,
        right_ptr=right,
        B=B,
        N=N,
        D=hidden_dim,
        stride_out_b=s_out_b,
        stride_out_i=s_out_i,
        stride_out_j=s_out_j,
        stride_out_d=s_out_d,
        stride_left_b=s_l_b,
        stride_left_i=s_l_i,
        stride_left_k=s_l_k,
        stride_left_d=s_l_d,
        stride_right_b=s_r_b,
        stride_right_j=s_r_i,          # original second axis corresponds to “j” in the contraction
        stride_right_k=s_r_k,
        stride_right_d=s_r_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                  # H100 can sustain many warps
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight,
        bias=to_out_norm_bias, eps=eps
    )

    # ------------------------------------------------------------------
    # 7. Output gating
    # ------------------------------------------------------------------
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
