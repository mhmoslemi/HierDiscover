def triton_batched_matmul(left: torch.Tensor, right: torch.Tensor, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    # left shape [B,N,N,D]
    # We'll permute to [B,D,N,N] and reshape.
    B, N, _, D = left.shape
    left_perm = left.permute(0,3,1,2).contiguous()  # B, D, N, N
    right_perm = right.permute(0,3,1,2).contiguous()
    # transpose last two dims of right
    right_perm_t = right_perm.transpose(-2, -1).contiguous()
    # flatten batch and D
    left_flat = left_perm.view(B*D, N, N)
    right_flat = right_perm_t.view(B*D, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=left.dtype)  # same dtype (half)
    # compute strides:
    stride_a = left_flat.stride()

    # left_flat is contiguous with shape (batch, M, K)
    # Right shape (batch, K, N)
    # So we pass stride across batch? Actually we will launch kernel with each batch as independent in axis0.
    # Strides:

    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bj = right_flat.stride(2)
    stride_cba = out_flat.stride(0)
    stride_cbi = out_flat.stride(1)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:
        # mask should be [B,N,N]
        mask = mask_tensor.unsqueeze(-1).type_as(left)  # cast to same dtype
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Compute out via Triton batched GEMM
    out_h = batched_matmul_triton(left_h, right_h, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)  # shape [B,N,N,hidden_dim]

    out = out_h.to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    assert A.shape == B.shape  # both [B,N,N,D]
    Bsize, N, _, D = A.shape
    # Permute
    A_perm = A.permute(0,3,1,2).contiguous()
    B_perm = B.permute(0,3,1,2).contiguous()
    # B perm transposed for inner dimension
    B_perm_t = B_perm.transpose(-2, -1).contiguous()
    # Flatten
    A_flat = A_perm.view(Bsize*D, N, N)
    B_flat = B_perm_t.view(Bsize*D, N, N)
    # Allocate output
    C_flat = torch.empty_like(A_flat, dtype=A.dtype)   # half
    # Strides
    stride_a_batch, stride_a_m, stride_a_k = A_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = C_flat.stride()
    # Launch kernel
    grid = (Bsize*D,
            (N + BLOCK_M - 1)//BLOCK_M,
            (N + BLOCK_N - 1)//BLOCK_N)
    batched_gemm_kernel[grid](
        A_flat, B_flat, C_flat,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Reshape to original shape
    C = C_flat.view(Bsize, D, N, N).permute(0,2,3,1).contiguous()
    return C
```

Now define batched_gemm_kernel:

```
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # (M,N)-tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (M, K)
        a_ptrs = A_ptr + batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0).to(tl.float16)

        # Load B tile (K, N)
        b_ptrs = B_ptr + batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0).to(tl.float16)

        # Compute dot product: (M,K) x (K,N)
        acc += tl.dot(a, b)

    # Write back C tile
    c_ptrs = C_ptr + batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

"""
custom_kernel: Triton‑accelerated "outgoing" TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor x of shape (B, N, N, C).
2. Linear projections without bias:
      left  = x @ W_left   (B,N,N,H)
      right = x @ W_right  (B,N,N,H)
3. Optional pairwise mask of shape (B,N,N) is applied (unsqueezed to last dim).
4. Gating values from three sigmoided linear layers:
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
   Apply left/right gates: left  *= left_gate;  right *= right_gate.
5. Cast left/right to float16 and compute the heavy contraction
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is a batch‑wise matrix multiplication
      (B*H, N, N) @ (B*H, N, N)^T.
   The contraction is implemented in Triton (batched GEMM) with
   float16 arithmetic and float32 accumulation.
6. Convert the result back to float32, apply a second LayerNorm over
   the hidden dimension H, multiply by out_gate, and finish with a linear
   projection back to C.
7. Return a tensor of shape (B, N, N, C).

Only the O(N³·H) contraction (step 5) is performed in Triton; all other
operations use native PyTorch which is negligible in comparison.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM for matrices A (M×K) and B (K×N) per batch.
# Input tensors are contiguous in (batch, M, K) and (batch, K, N) order.
# ---------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_Ab, stride_Am, stride_Ak,          # A strides (batch, M, K)
    stride_Bb, stride_Bk, stride_Bn,          # B strides (batch, K, N)
    stride_Cb, stride_Cm, stride_Cn,          # C strides (batch, M, N)
    M, N, K,                                  # matrix dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)      # batch index (B*H)
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Compute absolute coordinates of the block.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute a mask that is true for valid positions (handles tails).
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointer arithmetic.
        a_ptr = A_ptr + batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptr = B_ptr + batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        # Load tiles with masking. Cast to FP16 for compute.
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Matrix multiply the tile.
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Write back C tile (cast accumulator to FP16).
    c_ptr = C_ptr + batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptr,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ---------------------------------------------------------------------------
# Helper that wraps the Triton kernel for the specific shape needed by TriMul.
# ---------------------------------------------------------------------------
def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor,
                         BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    Batched GEMM:
        out_{b,i,j} = Σ_k left_{b,i,k} * right_{b,k,j}
    Input tensors are (B, N, N, H) in FP16.
    Returns (B, N, N, H) in FP16.
    """
    B, N, _, H = left.shape

    # Move hidden dimension to batch axis and make right transposed.
    # left_perm: (B, H, N, N)   -> row‑major matrix A (M=N, K=N)
    # right_perm_t: (B, H, N, N) where last two axes are (K, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    right_perm_t = right_perm.transpose(-2, -1).contiguous()

    # Collapse (B, H) into a single batch dimension.
    A = left_perm.view(B * H, N, N)
    B_ = right_perm_t.view(B * H, N, N)

    # Output buffer.
    C = torch.empty_like(A)

    # Strides for Triton (int64).
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B_.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    # Grid dimensions.
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_gemm_kernel[grid](
        A, B_, C,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # typical choice for FP16 matmul
    )

    # Reshape back to (B, N, N, H).
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ---------------------------------------------------------------------------
# Main entry point expected by the evaluator.
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton‑accelerated
    contraction.  Arguments:

        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape (batch, seq_len, seq_len, dim)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------- LayerNorm (first) -------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # shape: (B, N, N, dim)

    # ------------------------ Linear projections ------------------------
    left = F.linear(x, weights["left_proj.weight"])      # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])    # (B,N,N,H)

    # ---------------------------- Masking ----------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: (B,N,N) -> (B,N,N,1) and cast to same dtype.
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # -------------------------- Gating values -------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------- Heavy contraction (Triton) --------------------
    # Cast to fp16 for the kernel.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Out shape: (B,N,N,H) in fp16 after contraction.
    out_h = _batched_gemm_triton(left_h, right_h,
                                 BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Back to fp32 for the remaining layers.
    out = out_h.to(torch.float32)

    # ------------------------ Second LayerNorm ------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------------------------- Gate -------------------------------
    out = out * out_gate

    # -------------------------- Final Linear -------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
