@triton.jit
def trmul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D_h,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_bd = tl.program_id(2)  # combined batch and hidden dimension
    # Compute batch index and hidden index
    d = pid_bd % D_h
    b = pid_bd // D_h
    # Compute start offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Create masks for out-of-bound i,j
    mask_i = i_offset < N
    mask_j = j_offset < N
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left block A: shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr + b*stride_lb + d*stride_ld + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        # load with mask
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load right block B: shape (BLOCK_K, BLOCK_N)
        b_ptr = right_ptr + b*stride_rb + d*stride_rd + k_offset[:, None] * stride_rk + j_offset[None, :] * stride_rj
        b_ = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate

    # Compute final offsets for output
    out_ptrs = out_ptr + b*stride_ob + d*stride_od + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj
    # Store result
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

# Triton kernel for the N‑by‑N “outer‑product” contraction
@triton.jit
def trmul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,  # batch, seq_len, hidden_dim
    # strides for left = (b,i,k,d)
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides for right = (b,i,j,d)  (we will access as (b,j,k,d))
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides for out   = (b,i,j,d)
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 2‑D tiling over the i and j axes, 1‑D tiling over the (batch, hidden) pair

==============

Highly‑optimized forward implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The expensive N³ contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

of independent matrix multiplications (one per hidden dimension).  The rest
of the module (LayerNorms, linear projections and sigmoid gates) is kept in
PyTorch for clarity, but the dominant work is executed on the GPU with
FP16 arithmetic while the final result is accumulated in FP32.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024,
and hidden dimension hidden_dim (typically 128).  It supports optional
masking and works with both normal and heavy‑tailed input distributions.

import triton.language as tl


@triton.jit
def trmul_matmul_kernel(
    left_ptr,   # pointer to left  [B, N, N, D]  (FP16)
    right_ptr,  # pointer to right [B, N, N, D]  (FP16)
    out_ptr,    # pointer to out   [B, N, N, D]  (FP32)
    B, N, D,    # runtime values
    # strides for left  = (b, i, k, d)
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides for right = (b, i, j, d) – we will read as (b, j, k, d)
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides for out   = (b, i, j, d)
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    Each program instance handles a tile (BLOCK_M × BLOCK_N) of the output
    for a particular (batch, hidden) pair.

    pid_j = tl.program_id(1)          # tile index over the j‑axis

    d = pid_bd % D
    b = pid_bd // D

    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
