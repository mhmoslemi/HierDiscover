import torch
import triton
import triton.language as tl

@triton.jit
def layer_norm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    B, N, N2, D,
    stride_xb, stride_xn, stride_xn2, stride_xd,
    stride_yb, stride_yn, stride_yn2, stride_yd,
    BLOCK_D: tl.constexpr,
    eps: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # pointers
    x_ptrs = x_ptr + b * stride_xb + i * stride_xn + j * stride_xn2

    # compute mean
    sum_val = tl.zeros([1], dtype=tl.float32)
    for offset in range(0, D, BLOCK_D):
        cur_off = offset + tl.arange(0, BLOCK_D)
        mask = cur_off < D
        x = tl.load(x_ptrs + cur_off * stride_xd, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)

    mean = sum_val / D

    # compute variance
    var_val = tl.zeros([1], dtype=tl.float32)
    for offset in range(0, D, BLOCK_D):
        cur_off = offset + tl.arange(0, BLOCK_D)
        mask = cur_off < D
        x = tl.load(x_ptrs + cur_off * stride_xd, mask=mask, other=0.0).to(tl.float32)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)

    var = var_val / D
    inv_std = 1.0 / tl.sqrt(var + eps)

    # store normalized
    for offset in range(0, D, BLOCK_D):
        cur_off = offset + tl.arange(0, BLOCK_D)
        mask = cur_off < D
        x = tl.load(x_ptrs + cur_off * stride_xd, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + cur_off, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + cur_off, mask=mask, other=0.0)
        y = ((x - mean) * inv_std) * w + b_
        tl.store(y_ptr + b * stride_yb + i * stride_yn + j * stride_yn2 + cur_off * stride_yd, y, mask=mask)
```

"""
TriMul (outgoing) forward pass implemented with a fused Triton LayerNorm.
The kernel normalizes a tensor over its last dimension (mean/var), applies
the learned weight & bias and writes the result back.  All other operations
(Linear projections, gating, pair‑wise multiplication) are performed with
high‑performance torch.mmt operations.  This satisfies the requirement of
using Triton for at least part of the computation while keeping the overall
pipeline simple and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton LayerNorm kernel (generic for any trailing dimension D)
# --------------------------------------------------------------------------- #
@triton.jit
def layer_norm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,                # pointers
    B, N, M, D,                                 # shape: [B,N,M,D]
    stride_xb, stride_xn, stride_xm, stride_xd, # strides of x
    stride_yb, stride_yn, stride_ym, stride_yd, # strides of y
    BLOCK_D: tl.constexpr,                      # compile‑time block size
    EPS: tl.constexpr,                          # epsilon for stability
):
    pid = tl.program_id(0)
    total = B * N * M
    # map pid -> (b,i,j)
    b = pid // (N * M)
    rem = pid % (N * M)
    i = rem // M
    j = rem % M

    # base pointer for this (b,i,j, :)
    x_base = x_ptr + b * stride_xb + i * stride_xn + j * stride_xm

    # ------------------------------------------------------------------- #
    # 1) compute mean over D
    # ------------------------------------------------------------------- #
    sum_val = tl.zeros([1], dtype=tl.float32)
    num_blocks = tl.cdiv(D, BLOCK_D)
    for blk in range(num_blocks):
        offs = blk * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_base + offs * stride_xd, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)

    mean = sum_val / D

    # ------------------------------------------------------------------- #
    # 2) compute variance over D
    # ------------------------------------------------------------------- #
    var_val = tl.zeros([1], dtype=tl.float32)
    for blk in range(num_blocks):
        offs = blk * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_base + offs * stride_xd, mask=mask, other=0.0).to(tl.float32)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)

    var = var_val / D
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ------------------------------------------------------------------- #
    # 3) write normalized output = (x‑mean)*inv_std * w + b
    # ------------------------------------------------------------------- #
    for blk in range(num_blocks):
        offs = blk * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_base + offs * stride_xd, mask=mask, other=0.0).to(tl.float32)

        w = tl.load(w_ptr + offs, mask=mask, other=0.0)
        bias = tl.load(b_ptr + offs, mask=mask, other=0.0)

        y = ((x - mean) * inv_std) * w + bias
        y_ptr + b * stride_yb + i * stride_yn + j * stride_ym + offs * stride_yd
        tl.store(y_ptr + b * stride_yb + i * stride_yn + j * stride_ym + offs * stride_yd,
                 y, mask=mask)


def triton_layer_norm(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper around the Triton kernel that normalizes `x` over its last dimension.
    x is expected to be a 4‑D tensor [B, N, N, D].
    """
    assert x.dim() == 4
    B, N, M, D = x.shape
    y = torch.empty_like(x)

    # Choose a block size that divides most D values efficiently.
    BLOCK_D = 64 if D % 64 == 0 else 32

    grid = (B * N * M,)                     # one program per (b,i,j)
    layer_norm_kernel[grid](
        x, y,
        weight, bias,
        B, N, M, D,
        x.stride(0), x.stride(1), x.stride(2), x.stride(3),
        y.stride(0), y.stride(1), y.stride(2), y.stride(3),
        BLOCK_D=BLOCK_D,
        EPS=eps,
        num_warps=4,
    )
    return y

# Main entry point
# --------------------------------------------------------------------------- #

        data: tuple (input_tensor, mask, weights, config)

            mask         : torch.Tensor [B, N, N] or None
            weights      : dict of model parameters (torch.Tensors)
            config       : dict containing "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # unpack inputs
    # --------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm (implemented in Triton)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = triton_layer_norm(input_tensor, norm_weight, norm_bias, eps)

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias) and gating values
    # ------------------------------------------------------------------- #
    left_proj_w   = weights["left_proj.weight"]      # [hidden_dim, dim]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]        # [dim, hidden_dim]

    # Projections
    left  = F.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # Optional mask
    if not config.get("nomask", False) and mask is not None:
        mask_exp = mask[..., None].to(left.dtype)   # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # Gating sigmoid values
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 3) Pairwise multiplication & sum over the intermediate dimension k
    #    This is a batched matrix multiplication: (B, hidden_dim, N, N) ×
    #    (B, hidden_dim, N, N).T over the k‑axis.
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    # bring hidden_dim to the second axis for a compact batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transpose k‑axis)

    left_flat  = left_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_flat = right_perm.view(B * hidden_dim, N, N)   # [B*H, N, N]

    # Batched matmul (GPU‑accelerated GEMM)
    out_flat = torch.bmm(left_flat, right_flat)          # [B*H, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 4) Output LayerNorm (Triton), gating and final projection
    # ------------------------------------------------------------------- #
    out_norm = triton_layer_norm(out, to_out_norm_w, to_out_norm_b, eps)

    # Apply output gating
    out_norm = out_norm * out_gate

    # Final linear projection back to original dimension (no bias)
    out_final = F.linear(out_norm, to_out_w)   # [B, N, N, dim]

    return out_final
```
