def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   pid0 = tl.program_id(0)
   pid1 = tl.program_id(1)
   pid2 = tl.program_id(2)

   # compute batch & hidden indices
   bh = pid0
   batch = bh // H
   hidden = bh % H

   # compute offsets
   i_start = pid1 * BLOCK_M
   j_start = pid2 * BLOCK_N

   # block range arrays
   offs_i = i_start + tl.arange(0, BLOCK_M)
   offs_j = j_start + tl.arange(0, BLOCK_N)
   # mask to avoid out-of-bound
   mask_i = offs_i < N
   mask_j = offs_j < N

   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # base pointers for left, right, out
   left_base = left_ptr + batch * stride_left_b + hidden * stride_left_h
   right_base = right_ptr + batch * stride_right_b + hidden * stride_right_h
   out_base = out_ptr + batch * stride_out_b + hidden * stride_out_h

   # loop over k dimension
   for k in range(0, tl.cdiv(N, BLOCK_K)):
       k_start = k * BLOCK_K
       offs_k = k_start + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # load left block: shape (BLOCK_M, BLOCK_K)
       left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
       left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

       acc += tl.dot(left, right)

"""
tri_mul_outgoing.py

Implementation of the “outgoing” TriMul operator (AlphaFold‑3) using Triton.
The heavy tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is performed by a custom Triton kernel (tiled matrix‑multiply over the
sequence dimension). All other operations (LayerNorm, linear projections,
gate sigmoids, masking) are executed with plain PyTorch for simplicity.

The kernel processes each (batch, hidden‑channel) pair as an independent
matrix‑multiply of shape (N×N)·(N×N)ᵀ, using block sizes 64×64×64 and
Tensor‑Core‑compatible `tl.dot`. The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the multiplicative update:  out = Σ_k left[...,k] * right[...,k]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,               # const *float   (B,N,N,H)
    right_ptr,              # const *float   (B,N,N,H)
    out_ptr,                # *float         (B,N,N,H)
    B, N, H,                # int64 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs: (batch*hidden, block_i, block_j)
    # --------------------------------------------------------------
    pid_bh = tl.program_id(0)
    pid_i  = tl.program_id(1)
    pid_j  = tl.program_id(2)

    # decode batch and hidden‑channel
    b = pid_bh // H          # batch index 0 .. B-1
    h = pid_bh %  H          # hidden index 0 .. H-1

    # --------------------------------------------------------------
    # Offsets for the i‑ and j‑tiles
    # --------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = j_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Base pointers for the current (b,h) pair
    # --------------------------------------------------------------
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # --------------------------------------------------------------
    # Main reduction over k
    # --------------------------------------------------------------
    for k_tile in range(num_k):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile: shape (BLOCK_M, BLOCK_K) -----------------
        left_ptrs = left_base + offs_i[:, None] * stride_left_i \
                                 + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ---- load right tile (transposed view): shape (BLOCK_K, BLOCK_N) ----
        # right[j,k] is stored as right[i=j,k] in memory, therefore we
        # use stride_right_i for the j‑offset and stride_right_k for k.
        right_ptrs = right_base + offs_k[:, None] * stride_right_k \
                                   + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # ---- dot product (BLOCK_M × BLOCK_K) × (BLOCK_K × BLOCK_N) -----
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # Store the result tile
    # --------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_out_i \
                         + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim]
        - mask_tensor  : torch.Tensor   [B, N, N]  (may be None)
        - weights_dict : dict of torch.Tensors containing the model weights
        - config_dict  : dict with keys "dim", "hidden_dim", ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # retrieve weight tensors
    # ------------------------------------------------------------------
    norm_weight           = weights["norm.weight"]
    norm_bias             = weights["norm.bias"]
    left_proj_weight      = weights["left_proj.weight"]
    right_proj_weight     = weights["right_proj.weight"]
    left_gate_weight      = weights["left_gate.weight"]
    right_gate_weight     = weights["right_gate.weight"]
    out_gate_weight       = weights["out_gate.weight"]
    to_out_norm_weight    = weights["to_out_norm.weight"]
    to_out_norm_bias      = weights["to_out_norm.bias"]
    to_out_weight         = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) on the original input
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – broadcast over the hidden dimension
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel: out = Σ_k left[...,k] * right[...,k]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape          # left shape = (B, N, N, hidden_dim)

    out = torch.empty_like(left)     # allocate output [B,N,N,hidden_dim]

    # block sizes – chosen to fit H100 Tensor‑Core resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # grid dimensions: (batch*hidden, num_i_tiles, num_j_tiles)
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # strides in *elements* (torch returns element‑wise strides)
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    tri_mul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # reasonable warp count for H100
    )

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm + final gating
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Linear projection back to the original dimension (dim)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
