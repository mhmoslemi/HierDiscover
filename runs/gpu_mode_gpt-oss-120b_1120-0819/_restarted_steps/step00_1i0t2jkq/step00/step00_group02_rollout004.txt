def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(axis=0)
    # compute total number of i tiles and j tiles:
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_batches = B
    total_tiles = num_batches * num_i * num_j
    # decode pid
    b = pid // (num_i * num_j)
    tile_id = pid % (num_i * num_j)
    i_tile = tile_id // num_j
    j_tile = tile_id % num_j

    # compute offset ranges for i and j
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # create accumulator for each D tile, we will accumulate sum across K and then write out later.
    # We'll compute per D tile inside loop, but to reduce registers we allocate acc for one D tile at a time.
    # Instead we allocate a full accumulator for BLOCK_D
    # However we need to accumulate across K => zero init.
    # We'll create an accumulator 3d for each d tile.
    # Actually we need to output for all hidden dims; we will loop over d_tiles.
    # We'll compute out_tile shape (BLOCK_M, BLOCK_N, BLOCK_D) and store after full K loop.
    # So create accumulator:
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)
    
    # Compute number of k tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # Loop over d_tiles
    for d_tile in range(0, H, BLOCK_D):
        # Reset accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)
        
        for k_tile in range(num_k):
            k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k_offset < N

            # Compute pointer offsets: left_ptr + b*stride_left_batch + i_offset*stride_left_i + k_offset*stride_left_k + d_offset*stride_left_h

            d_offset = d_tile + tl.arange(0, BLOCK_D)
            # Compute mask for d dimension: d_offset < H
            mask_d = d_offset < H

            # compute combined mask for each dimension?

            # Trick: Use broadcast to retrieve 3D block: offset for each (i,k,d). Use tl.load with pointer computed using tl.arange?
            # left_ptrs shape (BLOCK_M, BLOCK_K, BLOCK_D) -> we can compute using i_offset[:, None, None] * stride_left_i + k_offset[None, :, None]* stride_left_k + d_offset[None, None, :] * stride_left_h + b*stride_left_batch
            # Use tl.load(left_ptr + offset, mask=mask) which expects address per element.
            # Build offsets using tl.arange's broadcasting:
            i_idx = i_offset[:, None, None]  # (M,1,1)
            k_idx = k_offset[None, :, None]  # (1,K,1)
            d_idx = d_offset[None, None, :]  # (1,1,D)
            # Compute linear address:

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # for layernorm.

    # Fetch weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # compute normalized input
    # Option: torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Cast to fp16 for further compute to reduce memory
    x_fp16 = x.to(torch.float16)

    # Projections
    left = torch.nn.functional.linear(x_fp16, left_proj_weight)  # shape B,N,N,H
    right = torch.nn.functional.linear(x_fp16, right_proj_weight)

    # Gates
    left_gate = torch.nn.functional.linear(x_fp16, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_weight).sigmoid()

    # Apply mask if provided and not nomask (we need config["nomask"] perhaps)
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).to(x_fp16.dtype)
        left = left * mask_
        right = right * mask_

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute contraction via Triton
    out = tri_mul_torch_triton(left, right)   # returns float32

    # Apply to_out_norm (layernorm across hidden_dim)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # multiply by out_gate (shape B,N,N,H)
    out_norm = out_norm * out_gate

    # Final linear to original dim
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # weight shape (dim, hidden_dim)

def tri_mul_torch_triton(left, right):
    assert left.dtype == torch.float16
    out = torch.empty_like(left, dtype=torch.float32)
    B,N,N,H = left.shape

    # Get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Determine block sizes. Choose constants.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 32  # hidden dim tile. Must be <= H. If H < BLOCK_D, we adjust.

    # compile-time constants
    # compute grid
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_i * num_j,)

    # Launch
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
        num_warps=8,  # maybe 8
    )
    return out

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    # compute tile indices
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_b = B
    # Flatten
    tiles_per_batch = num_i * num_j
    b = pid // tiles_per_batch
    rem = pid % tiles_per_batch
    i_tile = rem // num_j
    j_tile = rem % num_j

    # Compute offsets
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for range
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Number of k tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # Loop over hidden dim tiles
    for d_tile in range(0, H, BLOCK_D):
        # Compute d offsets and mask
        d_offset = d_tile + tl.arange(0, BLOCK_D)
        mask_d = d_offset < H

        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # Loop over K dimension
        for k_tile in range(num_k):
            k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k_offset < N

            # Pointers and loads

"""
TriMul "outgoing" kernel (AlphaFold3 style)

The heavy part of the TriMul forward pass is the 4‑D contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]          (1)

where left/right have shape (B, N, N, H)  (H = hidden_dim).
A custom Triton kernel computes (1) efficiently:
* The (i, j) output plane is tiled (BLOCK_M × BLOCK_N).
* The reduction over k is tiled (BLOCK_K).
* The hidden dimension is tiled (BLOCK_D) and processed inside one
  program (so we keep a (M,N,D) accumulator in registers).
* Masks are applied to avoid out‑of‑bounds loads.
* The kernel works for any batch size ≤ 2, any sequence length
  (N = 128…1024) and hidden dimensions up to 384.

All other operations (LayerNorms, linear projections, sigmoid gates,
final LayerNorm & linear) are performed with PyTorch (they are cheap
compared to the O(B·H·N³) contraction).  The kernel returns FP32
output; the surrounding code works in FP16 to reduce memory traffic.

The entry point `custom_kernel` follows the specification:
    input_tensor, mask, weights, config = data
and returns the final Tensor of shape [B, N, N, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel implementing the contraction (1).
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].

    The kernel is launched with a 1‑D grid of size
    B * ceil(N/BLOCK_M) * ceil(N/BLOCK_N).
    """
    pid = tl.program_id(0)

    # ------------------------------
    # Decode program id into (b, i_tile, j_tile)
    # ------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_i * num_j

    b = pid // tiles_per_batch
    rem = pid % tiles_per_batch
    i_tile = rem // num_j
    j_tile = rem % num_j

    # ------------------------------
    # Offsets within the current tiles
    # ------------------------------
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # ------------------------------
    # Loop over the hidden dimension in tiles of BLOCK_D
    # ------------------------------
    for d_tile in range(0, H, BLOCK_D):
        d_offset = d_tile + tl.arange(0, BLOCK_D)
        mask_d = d_offset < H

        # accumulator: (BLOCK_M, BLOCK_N, BLOCK_D)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # --------------------------
        # Reduction over k
        # --------------------------
        num_k = tl.cdiv(N, BLOCK_K)
        for k_tile in range(num_k):
            k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k_offset < N

            # ----- load left block (M, K, D) -----
            # ptr = left_ptr + b*stride_b + i*stride_i + k*stride_k + d*stride_h
            left_ptrs = (
                left_ptr
                + b * stride_left_b
                + i_offset[:, None, None] * stride_left_i
                + k_offset[None, :, None] * stride_left_k
                + d_offset[None, None, :] * stride_left_h
            )
            mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            left = tl.load(left_ptrs, mask=mask_left, other=tl.float16(0.0))

            # ----- load right block (N, K, D) -----
            right_ptrs = (
                right_ptr
                + b * stride_right_b
                + j_offset[:, None, None] * stride_right_j
                + k_offset[None, :, None] * stride_right_k
                + d_offset[None, None, :] * stride_right_h
            )
            mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            right = tl.load(right_ptrs, mask=mask_right, other=tl.float16(0.0))

            # ----- inner product for every d in this tile -----
            # left[:, :, dh] : (M, K)  ;  right[:, :, dh] : (N, K)
            # We need Σₖ left[i,k,d] * right[j,k,d] → dot(left_slice,
            #                                            right_slice, trans_b=True)
            for dh in tl.static_range(BLOCK_D):
                # the mask already zero‑padded out‑of‑range d, so we can safely compute
                acc[:, :, dh] += tl.dot(
                    left[:, :, dh],          # (M, K)
                    right[:, :, dh],         # (N, K)
                    trans_b=True,            # treat right as (K, N)
                )
        # end k_tile loop

        # ------------------------------
        # Write the tile to global memory
        # ------------------------------
        out_ptrs = (
            out_ptr
            + b * stride_out_b
            + i_offset[:, None, None] * stride_out_i
            + j_offset[None, :, None] * stride_out_j
            + d_offset[None, None, :] * stride_out_h
        )
        mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel.
# ----------------------------------------------------------------------
def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : (B, N, N, H)  FP16 tensors
    Returns out    : (B, N, N, H)  FP32 tensor
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    # Allocate output (float32)
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # Strides as Python ints (must be int64 for Triton)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Choose tiling sizes (tuned for H100 & the given ranges)
    BLOCK_M = 32  # rows of output tile
    BLOCK_N = 32  # cols of output tile
    BLOCK_K = 32  # reduction tile
    BLOCK_D = 8   # hidden‑dim tile (must divide H reasonably)

    grid = (B * ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N),)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=8,  # enough to keep the cores busy on H100
    )
    return out

# Main entry point required by the problem statement.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operation.

    data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast everything to half‑precision to save bandwidth.
    # (weights are stored as float32 in the checkpoint.)
    # ------------------------------------------------------------------
    device = input_tensor.device
    # LayerNorm (first)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))           # (B,N,N,H)
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # Sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )

    # ------------------------------------------------------------------
    # Optional mask (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Heavy contraction using Triton (output is float32)
    # ------------------------------------------------------------------
    out = _tri_mul_triton(left, right)                     # (B,N,N,H) FP32

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden dim, gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)                 # broadcast over hidden dim

    # Final projection back to `dim` (no bias)
    out = F.linear(out, weights["to_out.weight"].to(out.dtype))   # (B,N,N,dim)

    return out
