def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_A_batch, stride_Am, stride_Ak,
    stride_B_batch, stride_Bk, stride_Bn,
    stride_C_batch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 1e-6, # not needed
):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)
    # compute offsets for k loop
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N
    # pointers for A,B,C for this block
    a_ptrs = A_ptr + pid_b * stride_A_batch + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_B_batch + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
    c_ptrs = C_ptr + pid_b * stride_C_batch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # iterate on k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)
        # compute
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk
    # store result
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing kernel (AlphaFold3 style)

The forward pass consists of:
1. Layer‑norm over the input channel dimension.
2. Two linear projections (left / right) + linear gates.
3. Optional pair‑wise mask applied to the projected tensors.
4. A batched GEMM:
       out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
   which is equivalent to a batch of matrix‑multiplications
   `left @ rightᵀ`.  This O(N³) part is the only place where we
   use a custom Triton kernel (float‑16 inputs, float‑32 accumulation).
5. Layer‑norm over the hidden dimension, gating, and a final linear
   projection back to the original channel size.

The kernel works on a flattened batch‑head dimension (B*H) and
processes tiles of size BLOCK_M×BLOCK_N with a K‑tile of size BLOCK_K.
All operations are performed on the GPU; the rest of the network
components are kept in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B, where A: (M,K), B: (K,N)
# Input tensors are stored as [batch_head, M, K] and [batch_head, K, N].
# All tensors are contiguous in memory; we pass explicit strides.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr,               # *float16
    B_ptr,               # *float16
    C_ptr,               # *float32
    M, N, K,             # int32
    stride_A_batch, stride_Am, stride_Ak,
    stride_B_batch, stride_Bk, stride_Bn,
    stride_C_batch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # batch_head index

    #---------------------------------
    # Compute offsets for this tile
    #---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current tile
    a_ptrs = (A_ptr + pid_b * stride_A_batch
              + offs_m[:, None] * stride_Am
              + offs_k[None, :] * stride_Ak)
    b_ptrs = (B_ptr + pid_b * stride_B_batch
              + offs_k[:, None] * stride_Bk
              + offs_n[None, :] * stride_Bn)
    c_ptrs = (C_ptr + pid_b * stride_C_batch
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_blocks):
        # Load A and B tiles (float16)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K)[None, :],
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        # Compute tile multiplication and accumulate in float32
        acc += tl.dot(a, b).to(tl.float32)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Store the result tile (float32)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    custom_kernel(data) -> torch.Tensor

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask        : torch.Tensor or None, shape [B, N, N] (optional)
        - weights     : dict of torch.Tensors containing all model parameters
        - config      : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    device = x.device
    dtype = x.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if present (mask is broadcast on the hidden dimension)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Batched GEMM in Triton: out = left @ rightᵀ   (hidden‑wise)
    # ------------------------------------------------------------------
    # Bring hidden dimension before the two sequence dimensions
    # shape -> [B, hidden, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right for the matmul (k dimension becomes the leading one)
    right_t = right_perm.transpose(2, 3).contiguous()   # [B, hidden, N, N]

    # Cast to float16 to reduce memory traffic (kernel will accumulate in FP32)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    Bsz, H, N, _ = left_fp16.shape
    batch_head = Bsz * H

    # Flatten the batch + head dimension → [batch_head, N, N]
    A = left_fp16.view(batch_head, N, N)
    B = right_fp16.view(batch_head, N, N)
    C = torch.empty((batch_head, N, N), dtype=torch.float32, device=device)

    # Strides (in number of elements, not bytes)
    stride_A_batch = A.stride(0)
    stride_Am = A.stride(1)
    stride_Ak = A.stride(2)

    stride_B_batch = B.stride(0)
    stride_Bk = B.stride(1)
    stride_Bn = B.stride(2)

    stride_C_batch = C.stride(0)
    stride_Cm = C.stride(1)
    stride_Cn = C.stride(2)

    # Tile sizes – these work well on an H100 (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # grid dim for rows
        (N + BLOCK_N - 1) // BLOCK_N,   # grid dim for cols
        batch_head                      # one program per batch‑head
    )

    _batched_gemm_kernel[grid](
        A, B, C,
        M=N, N=N, K=N,
        stride_A_batch, stride_Am, stride_Ak,
        stride_B_batch, stride_Bk, stride_Bn,
        stride_C_batch, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before we touch C in PyTorch
    torch.cuda.synchronize()

    # Reshape back to [B, hidden, N, N] then to original layout [B, N, N, hidden]
    out_hidden = C.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 4) Output LayerNorm + gating + final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_gated = out_norm * out_gate
    out = F.linear(out_gated, weights["to_out.weight"])   # -> [B, N, N, dim]

    return out
```
