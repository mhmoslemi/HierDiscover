@triton.jit
def trigemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute batch, tile_i, tile_j
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    total_tiles_per_batch = num_tiles_i * num_tiles_j

    batch_idx = pid // total_tiles_per_batch
    tile_idx = pid % total_tiles_per_batch
    i_tile = tile_idx // num_tiles_j
    j_tile = tile_idx % num_tiles_j

    # compute offsets
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # create masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # compute base pointers
    left_base = left_ptr + batch_idx * stride_l_batch
    right_base = right_ptr + batch_idx * stride_r_batch
    out_base = out_ptr + batch_idx * stride_o_batch

    # pointers for block
    a_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for _k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        acc += tl.dot(a, b)  # both (BLOCK_M, BLOCK_K)*(BLOCK_K,BLOCK_N) -> (BLOCK_M,BLOCK_N)
        # increment pointers
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k
    # write output
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

@triton.jit
def trigemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute indices
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    total_tiles = num_tiles_i * num_tiles_j
    batch_idx = pid // total_tiles
    tile_idx = pid % total_tiles
    i_tile = tile_idx // num_tiles_j
    j_tile = tile_idx % num_tiles_j

    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # base pointers for this batch
    left_batch_ptr = left_ptr + batch_idx * stride_l_batch
    right_batch_ptr = right_ptr + batch_idx * stride_r_batch
    out_batch_ptr = out_ptr + batch_idx * stride_o_batch

    # a: left block
    a_ptrs = left_batch_ptr + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    # b: transposed right block
    b_ptrs = right_batch_ptr + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        acc += tl.dot(a, b)
        # slice increment
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k
    # store
    out_ptrs = out_batch_ptr + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm over dim (last dimension)
    eps = 1e-6
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gate projections + sigmoid
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply mask (if provided)
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute/reshape for batch GEMM: [B, hidden_dim, N, N] -> [B*hidden_dim, N, N]
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    batch_total = B * hidden_dim
    left_mat = left_perm.view(batch_total, N, N)
    right_mat = right_perm.view(batch_total, N, N)

    # Allocate output
    out_mat = torch.empty((batch_total, N, N), dtype=torch.float32, device=device)

    # Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_total * num_tiles_m * num_tiles_n, )

    @triton.jit
    def trigemm_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_j,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid = tl.program_id(0)
        num_tiles_i = tl.cdiv(N, BLOCK_M)
        num_tiles_j = tl.cdiv(N, BLOCK_N)
        total_tiles = num_tiles_i * num_tiles_j

        batch_idx = pid // total_tiles
        tile_idx = pid % total_tiles
        i_tile = tile_idx // num_tiles_j
        j_tile = tile_idx % num_tiles_j

        offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_k = tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        left_batch = left_ptr + batch_idx * stride_l_batch
        right_batch = right_ptr + batch_idx * stride_r_batch
        out_batch = out_ptr + batch_idx * stride_o_batch

        a_ptrs = left_batch + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        b_ptrs = right_batch + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for _ in range(0, tl.cdiv(N, BLOCK_K)):
            a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
            acc += tl.dot(a, b)
            a_ptrs += BLOCK_K * stride_l_k
            b_ptrs += BLOCK_K * stride_r_k

        out_ptrs = out_batch + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])

    trigemm_kernel[grid](
        left_ptr=left_mat,
        right_ptr=right_mat,
        out_ptr=out_mat,
        B=batch_total,
        N=N,
        stride_l_batch=left_mat.stride(0),
        stride_l_i=left_mat.stride(1),
        stride_l_k=left_mat.stride(2),
        stride_r_batch=right_mat.stride(0),
        stride_r_k=right_mat.stride(1),
        stride_r_j=right_mat.stride(2),
        stride_o_batch=out_mat.stride(0),
        stride_o_i=out_mat.stride(1),
        stride_o_j=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]

    # to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply output gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel
==============

Implements the *outgoing* TriMul operation from AlphaFold‑3 in a highly‑optimized
way:

1. **LayerNorm** over the last feature dimension (``dim``).
2. Linear projections to a hidden dimension ``c_z`` (``left_proj``/``right_proj``)  
   and three gating projections (``left_gate``, ``right_gate``, ``out_gate``).
3. Optional masking of the pairwise tensors.
4. **Gated pairwise multiplication**:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]      (h = hidden dim)
   This is a batch of ``c_z`` independent matrix‑multiplications (size N×N).
   It is performed by a custom Triton kernel that treats the hidden dimension
   as an extra batch dimension and computes a tiled GEMM with half‑precision
   inputs and fp32 accumulation.
5. LayerNorm over the hidden dimension, multiplicative output gate and a final
   linear projection back to the original feature dimension.

The kernel follows the classic “tiling + dot‑product” pattern (same as the
attention kernels in the Triton repo) and is tuned for the H100 (BLOCK_M/N=64,
BLOCK_K=32).  The rest of the work (norms, linear layers, sigmoid) is done with
PyTorch which is already highly optimized on the GPU.

The function signature matches the problem statement:
    ``custom_kernel(data) -> torch.Tensor`` where ``data`` is a tuple
    ``(input, mask, weights, config)``.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM computing Σₖ left[i,k] * right[j,k] for many
# independent “heads’’ (the hidden dimension).  The hidden dimension is folded
# into the batch dimension (batch_total = B * hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def _trigram_matmul_kernel(
    left_ptr,    # const __half *
    right_ptr,   # const __half *
    out_ptr,     # float *
    B, N,        # int32 scalars
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- tile indices -------------------------------------------------
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_i * num_tiles_j

    batch_idx = pid // tiles_per_batch
    tile_idx  = pid % tiles_per_batch
    i_tile = tile_idx // num_tiles_j
    j_tile = tile_idx % num_tiles_j

    # ---- offsets within the tile ---------------------------------------
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # ---- base pointers for this batch ----------------------------------
    left_batch  = left_ptr  + batch_idx * stride_l_batch
    right_batch = right_ptr + batch_idx * stride_r_batch
    out_batch   = out_ptr   + batch_idx * stride_o_batch

    # ---- pointer arithmetic --------------------------------------------
    # left : [M, K]  (M == i, K == k)
    a_ptrs = left_batch + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
    # right: transposed view of [K, N] (row = k, col = j)
    b_ptrs = right_batch + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

    # ---- accumulation --------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)           # half‑precision
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)           # half‑precision
        acc += tl.dot(a, b)                # fp32 accumulation
        a_ptrs += BLOCK_K * stride_l_k
        b_ptrs += BLOCK_K * stride_r_k

    # ---- write result --------------------------------------------------
    out_ptrs = out_batch + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [bs, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])          # [bs,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3) Optional masking (mask is 0/1)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # ensure mask dtype matches computation dtype
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)          # [bs,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Reduce over the pairwise axis using Triton
    #    left/right are cast to fp16 to reduce bandwidth.
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # reshape to [bs*hidden_dim, N, N] -> treat each hidden dim as an extra batch
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [bs, hidden, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    batch_total = bs * hidden_dim

    left_mat = left_perm.view(batch_total, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(batch_total, N, N)

    out_mat = torch.empty((batch_total, N, N), dtype=torch.float32, device=device)

    # launch parameters (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    n_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    n_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_total * n_tiles_m * n_tiles_n,)

    _trigram_matmul_kernel[grid](
        left_ptr=left_mat,
        right_ptr=right_mat,
        out_ptr=out_mat,
        B=batch_total,
        N=N,
        stride_l_batch=left_mat.stride(0),
        stride_l_i=left_mat.stride(1),
        stride_l_k=left_mat.stride(2),
        stride_r_batch=right_mat.stride(0),
        stride_r_k=right_mat.stride(1),   # row stride (k dimension after transpose)
        stride_r_j=right_mat.stride(2),   # column stride (j dimension after transpose)
        stride_o_batch=out_mat.stride(0),
        stride_o_i=out_mat.stride(1),
        stride_o_j=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [bs, N, N, hidden_dim]
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)   # [bs,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden_dim, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [bs,N,N,dim]

    return out
