"""
custom_kernel - Outgoing TriMul (AlphaFold3) implemented with Triton + cuBLAS

Algorithm (high‑level):
1️⃣  Row‑wise LayerNorm on the last dimension (float32→float16).
2️⃣  Fuse the five linear projections/gates into ONE GEMM
    X_norm @ W_fusedᵀ → [R, 5·H] (R = B·N·N).
3️⃣  Split the fused tensor, optionally apply the pair‑wise mask,
    evaluate sigmoid gates and store the three tensors directly in
    layout [B, H, N, N] (left, right, out_gate) (all fp16).
4️⃣  Batched half‑precision GEMM (cuBLAS) per hidden slice:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
5️⃣  Row‑wise mean & rsqrt‑variance of `out` (fp16 → fp32 stats).
6️⃣  Fuse the second LayerNorm, the element‑wise out_gate and the final
    linear projection into one Triton kernel.  The kernel normalises
    `out`, applies γ/β of the second LN, multiplies by `out_gate` and
    finally performs the hidden‑dim dot‑product with the transposed output
    weight.  Result is returned as float32 [B, N, N, dim].

All heavy element‑wise work is done in Triton kernels; the two large
matrix multiplications are delegated to cuBLAS (torch.bmm) for maximal
throughput on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # const *[rows, dim]      (fp32)
    y_ptr,   # mutable *[rows, dim]    (fp16)
    w_ptr,   # const *[dim]            (fp32) γ
    b_ptr,   # const *[dim]            (fp32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- mean / variance (fp32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- write normalized values (fp16) -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)      # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)      # β (fp32)

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates → [B, H, N, N] layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]        (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # Decode (b, i, k) from flattened index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Load mask (if present) and broadcast over hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # -> [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden slices
    # Base offset for the first hidden element of (b,i,k)
    base_offset = b * (H * N * N) + i * N + k          # scalar offset

    # Process hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection (row‑major, columns = hidden)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # Load the five H‑wide vectors
        left_raw      = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid in fp32 → fp16 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate_raw * mask_val * out_gate   # note: out_gate_raw not used elsewhere

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# 3️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # const *[B, H, N, N] (fp16)
    mu_ptr,           # mutable *[R]       (float32)
    rstd_ptr,         # mutable *[R]       (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of hidden‑zero slice

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [M, K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# 4️⃣ Fuse second LayerNorm, out_gate, and final linear projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # const *[B, H, N, N] (fp16)
    gate_ptr,         # const *[B, H, N, N] (fp16)   (out_gate)
    mu_ptr,           # const *[R]                (float32)
    rstd_ptr,         # const *[R]                (float32)
    ln_w_ptr,         # const *[H]                (float32) γ of to_out_norm
    ln_b_ptr,         # const *[H]                (float32) β of to_out_norm
    w_out_ptr,        # const *[H, dim]           (fp16)   (transposed weight)
    out_final_ptr,    # mutable *[B, N, N, dim]   (float32)
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)       # row tile (positions)
    pid_n = tl.program_id(1)       # column tile (output dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # Decode (b, i, j) for each position
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of hidden‑zero entry

    # Load per‑position statistics (fp32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # ----- loop over hidden dimension -----
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out and gate slices (fp16 → fp32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # Load LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32

        # Normalisation + γ/β + out‑gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]                # fp32
        norm = diff * gamma[None, :] + beta[None, :]                    # fp32
        term = norm * gate_vals                                         # fp32
        term_fp16 = term.to(tl.float16)                                 # fp16 for dot

        # Load a tile of the transposed output weight: [K, dim_tile]
        w_ptr = w_out_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                     # fp16

        # Accumulate dot product (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # ----- write final result (row‑major B,N,N,dim) -----
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

    """

    # --------------------------------------------------------------

    hidden_dim = cfg["hidden_dim"]

    device = x.device

    R = B * N * N                     # total (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------

    BLOCK_D_LN = 64

        x_flat,
        x_norm,
        w["norm.weight"],   # γ (float32)
        w["norm.bias"],     # β (float32)
        R,

        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=8,
    )

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # fp16, shape [R, 5·H]

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                         # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                        # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # fp16, [B·H, N, N]
    out = out.view(B, H, N, N)                                 # back to (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean / rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final linear projection
    # --------------------------------------------------------------
    # Transpose final weight for hidden‑major locality
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows (positions) per program
    BLOCK_K_OUT = 64           # hidden‑dim tile
    BLOCK_N_OUT = 128          # output‑dim tile
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # Return result (float32) – shape [B, N, N, dim]
    # --------------------------------------------------------------
    return final
```
