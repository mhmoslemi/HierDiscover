"""
TriMul (outgoing) – highly‑optimised Triton + PyTorch forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D pair representation (float32 → fp16).
2️⃣ Fuse the five linear maps (left/right proj + 3 gates) into a single GEMM:
       X_norm @ W_fusedᵀ   →   (R, 5·H)    with  R = B·N·N .
3️⃣ Split the fused projection, apply the optional pair‑wise mask and the three
   sigmoids.  The results are written directly into the flattened buffers
   required for the subsequent batched GEMM, avoiding any view/permute copy.
4️⃣ Batched half‑precision GEMM over the hidden dimension:
       out_mat = left @ rightᵀ                (B·H, N, N).
5️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the final
   reshape in a single Triton kernel.  This kernel reads the strided
   (B·H, N, N) output of step 4, computes per‑position statistics over the
   hidden dimension H, applies γ/β, multiplies with the out‑gate and writes
   the result as a dense (R, H) tensor – *no extra transposition or copy!*
6️⃣ Final linear projection back to the original channel dimension (fp16 → fp32).

All heavyweight element‑wise work is realised in Triton; the three dense
matrix multiplications (the fused projection, the batched GEMM and the final
linear) are delegated to cuBLAS/Tensor‑Core via `torch.nn.functional.linear`
and `torch.bmm`.  This removes the previously dominant `view‑permute‑copy`
step and brings the runtime under the 1 ms target for the test configurations.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   *[rows, dim]   (float32)
    y_ptr,               # mutable *[rows, dim]   (float16)
    w_ptr,               # const   *[dim]        (float32)   γ
    b_ptr,               # const   *[dim]        (float32)   β
    rows,                # tl.int32
    dim,                 # tl.int32
    eps,                 # tl.float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------------------------- mean / var (fp32) ---------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------- write normalised values --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]       (float16)
    left_ptr,          # mutable *[B*H, N*N]      (float16)
    right_ptr,         # mutable *[B*H, N*N]      (float16)
    out_gate_ptr,      # mutable *[rows, H]       (float16)
    mask_ptr,          # const   *[rows]          (float16) (ignored if HAS_MASK=0)
    rows,              # tl.int32   = B*N*N
    H,                 # tl.int32   = hidden_dim
    N,                 # tl.int32   = seq_len
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For every flattened pair (b,i,k) (row r) compute:
        left      = (proj_left      * mask) * sigmoid(proj_left_gate)
        right     = (proj_right     * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)

    The left/right tensors are written directly into the flattened
    buffers of shape (B*H, N*N) so that the next batched GEMM can be
    performed without an explicit view/permute.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                                           # total positions per matrix

    # derive batch index and intra‑matrix position
    b   = offs_r // NN
    pos = offs_r % NN

    # mask value (scalar 0/1 broadcasted over H)
    if HAS_MASK:
        mask_val_raw = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val_raw[:, None]               # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ----------- load the five projected blocks ------------------------
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # (R, BH)

        left_raw      = tl.load(proj_ptr + base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_raw     = tl.load(proj_ptr + base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw= tl.load(proj_ptr + base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(proj_ptr + base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)

        # ----------- sigmoid (stable fp32) --------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----------- apply mask and gates ---------------------------------
        left_val  = left_raw  * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        # destination offsets for left/right (flattened (B*H, N*N))
        batch_idx = b[:, None] * H + offs_h[None, :]                 # [BLOCK_R, BLOCK_H]
        dst_offset = batch_idx * NN + pos[:, None]                   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ----------- store out_gate (row‑wise layout) --------------------
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 3 – fused second LayerNorm, out‑gate multiplication, and reshape
# ----------------------------------------------------------------------
@triton.jit
def out_norm_gate_kernel(
    out_mat_ptr,    # const   *[B*H, N, N]  (flattened, fp16)
    out_gate_ptr,   # const   *[R, H]      (fp16)
    y_ptr,          # mutable *[R, H]      (fp16)
    gamma_ptr,      # const   *[H]         (float32)  γ of to_out_norm
    beta_ptr,       # const   *[H]         (float32)  β of to_out_norm
    rows,           # tl.int32 = R = B·N·N
    H,              # tl.int32 = hidden_dim
    NN,             # tl.int32 = N·N
    eps,            # tl.float32
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Input layout:
        out_mat : (B*H, N, N)  → flattened as 1‑D
        out_gate: (R, H)       (contiguous)
    For each position (b,i,j) (row r) we:
        * compute mean / variance over the hidden dimension H,
        * apply LayerNorm (γ/β),
        * multiply with the out‑gate,
        * store the result as a dense (R, H) matrix.
    This kernel removes the costly ``view → permute → contiguous`` step.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # decode batch and intra‑matrix position
    b   = offs_r // NN
    pos = offs_r % NN                                   # i*N + j

    # ---------- first pass – compute mean / variance -----------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offset = ((b*H + h) * NN) + pos
        base_offset = b * H * NN + pos                     # [BLOCK_R]
        offset = base_offset[:, None] + offs_h[None, :] * NN

        vals = tl.load(out_mat_ptr + offset,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float16)

        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass – normalise, γ/β, gate, store ---------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        base_offset = b * H * NN + pos
        offset = base_offset[:, None] + offs_h[None, :] * NN

        vals = tl.load(out_mat_ptr + offset,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float16)

        # normalisation
        vals_f32 = vals.to(tl.float32)
        norm = (vals_f32 - mean[:, None]) * rstd[:, None]   # (R, BH) fp32

        # γ, β (float32)
        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # [BH]
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)   # [BH]

        norm = norm * gamma[None, :] + beta[None, :]                # fp32

        # out‑gate (contiguous, fp16 → fp32)
        gate = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                            # fp32
        out_fp16 = out.to(tl.float16)

        # store
        tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 “outgoing” variant).
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]          # input / output channel dimension
    hidden_dim = cfg["hidden_dim"]   # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R  = B * N * N               # total rows for row‑wise ops
    H  = hidden_dim
    NN = N * N

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                         # (R, dim) float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program (tuned for H100)
    BLOCK_D = 128          # dim chunk size (fits fp16 SIMD)
    grid1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid1](
        x_flat,
        x_norm,
        w["norm.weight"],          # γ (float32)
        w["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: 5 projections (left, right, gates) → (R, 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5·H, dim)

    proj = F.linear(x_norm, fused_w)           # fp16, shape (R, 5·H)
    del fused_w, x_norm                         # free intermediates

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates – write directly to flattened buffers
    # ------------------------------------------------------------------
    left_flat  = torch.empty(B * H, NN, dtype=torch.float16, device=device)  # (B·H, N·N)
    right_flat = torch.empty_like(left_flat)

    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)        # (R, H)

    if cfg.get("nomask", True) or mask is None:
        # no mask → treat everything as 1.0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        # mask shape [B, N, N] → flatten to (R,)
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 256          # rows per program
    BLOCK_H = 32           # hidden‑dim chunk
    grid2 = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid2](
        proj,
        left_flat,
        right_flat,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # reshape for the batched GEMM
    left  = left_flat.view(B * H, N, N)      # (B·H, N, N)
    right = right_flat.view(B * H, N, N)     # (B·H, N, N)

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out‑gate multiplication (no transpose copy)
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    grid3 = (triton.cdiv(R, BLOCK_R),)
    out_norm_gate_kernel[grid3](
        out_mat.view(-1),            # flatten to 1‑D pointer
        out_gate,
        out_fused,
        w["to_out_norm.weight"],     # γ (float32)
        w["to_out_norm.bias"],       # β (float32)
        R,
        H,
        NN,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del out_mat, out_gate, left, right, left_flat, right_flat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # fp16, shape (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
