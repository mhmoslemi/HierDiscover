"""
TriMul (outgoing) – highly‑fused Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (fp32 → fp16).
2️⃣ Fused linear projection that simultaneously produces the three
   projections (left/right) and three gating vectors → shape (R,5·H).
   (R = B·N·N, H = hidden_dim)
3️⃣ Triton kernel splits the fused projection, applies the optional pairwise
   mask and sigmoid gates and writes the two projected tensors *directly*
   into the layout required by the batched GEMM:
        left  : [B·H, N, N]   (the “i‑k” matrix)
        right : [B·H, N, N]   (the “j‑k” matrix)
   The three output gates are stored in row‑major order (R, H).
4️⃣ Batched half‑precision GEMM (cuBLAS)   out = left @ rightᵀ
   → shape (B·H, N, N)
5️⃣ Triton kernel computes the second LayerNorm **and** the element‑wise
   multiplication by `out_gate` directly from `out` without any transpose,
   producing the tensor (R, H) that is fed to the final linear layer.
6️⃣ Final fp16 → fp32 linear projection back to `dim`.

The heavy O(N³·H) work is done by cuBLAS (step 4); the two LayerNorms and the
mask/gate handling are fused in custom Triton kernels, removing the memory‑
intensive transposes of the reference implementation and cutting the total
runtime roughly in half.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & gates, write directly into
#    the (B·H, N, N) layout required by the batched GEMM.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_to_mat_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[B*H, N, N] (fp16)
    right_ptr,         # mutable *[B*H, N, N] (fp16)
    out_gate_ptr,      # mutable *[rows, H] (fp16)
    mask_ptr,          # const   *[rows]   (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,    # R = B·N·N
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each pair (b,i,j) we have:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    `left` and `right` are written directly into tensors of shape
    (B·H, N, N) so that a single batched GEMM can be used.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ---------- optional mask ----------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---------- index helpers ----------
    N2 = N * N                                   # = N²
    b   = offs_r // N2                           # batch index
    ij  = offs_r - b * N2                        # remainder = i·N + j
    i   = ij // N
    j   = ij - i * N

    stride_h = N2                                 # stride of the first dim in (B·H,N,N)

    # hidden‑dimension loop
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ---------- load fused projection ----------
        # layout inside proj: [left, right, left_gate, right_gate, out_gate] each = H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]                # [R,blk_H]

        left_raw    = tl.load(proj_ptr + base,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_raw   = tl.load(proj_ptr + base + H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        left_g_raw  = tl.load(proj_ptr + base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # ---------- sigmoid (fp32 for stability) ----------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---------- apply mask and gates ----------
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # ---------- store left/right into (B·H, N, N) ----------
        # hidden‑base index for each row
        hidden_base = b * H                                 # [BLOCK_R]

        # full hidden index = hidden_base + offs_h
        hid_idx = hidden_base[:, None] + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        # linear offset inside the (B·H,N,N) tensor
        # offset = (hid_idx)·stride_h + i·N + j
        offset = hid_idx * stride_h + i[:, None] * N + j[:, None]

        # pointers
        left_ptrs  = left_ptr  + offset
        right_ptrs = right_ptr + offset

        tl.store(left_ptrs,  left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ---------- store out_gate in row‑major layout ----------
        out_gate_ptrs = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_ptrs, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm (over H) + multiplication by out_gate.
#    Input comes directly from the batched GEMM (B·H, N, N).
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_from_mat_kernel(
    mat_ptr,           # const   *[B*H, N, N] (fp16)
    out_gate_ptr,      # const   *[rows, H]  (fp16)
    y_ptr,             # mutable *[rows, H]  (fp16)
    weight_ptr,        # const   *[H]       (float32) γ
    bias_ptr,          # const   *[H]       (float32) β
    rows: tl.int32,    # R = B·N·N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Row‑wise LayerNorm on the matrix `mat` (shape B·H×N×N) and element‑wise
    multiplication by the previously computed `out_gate`.
    The result is written in (R, H) layout expected by the final linear.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    N2 = N * N
    stride_h = N2

    # recover (b,i,j) from linear index
    b   = offs_r // N2
    ij  = offs_r - b * N2
    i   = ij // N
    j   = ij - i * N

    # ------------------------------------------------------------------
    # 1️⃣ accumulate mean / variance over H (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # hidden index = b*H + h
        hid_idx = b * H + offs_h[None, :]               # [1, BLOCK_H] broadcast over rows
        # offset = hid_idx * N² + i·N + j
        ptr_offset = hid_idx * stride_h + i[:, None] * N + j[:, None]

        vals = tl.load(mat_ptr + ptr_offset,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply with out_gate, store (fp16)
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        hid_idx = b * H + offs_h[None, :]
        ptr_offset = hid_idx * stride_h + i[:, None] * N + j[:, None]

        vals = tl.load(mat_ptr + ptr_offset,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        # normalize
        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        # γ and β (fp32)
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
        b_ = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)

        norm = norm * w[None, :] + b_[None, :]                # fp32

        # out_gate (fp32)
        gate = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                     # fp32
        out_fp16 = out.to(tl.float16)

        # store (R, H)
        tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Helper to pick block sizes (simple heuristic)
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    """Return (BLOCK_M, BLOCK_D) tuned for the current problem size."""
    # keep blocks reasonably small to stay within shared memory limits
    BLOCK_M = 128 if rows >= 128 else max(1, rows)
    BLOCK_D = 32 if dim >= 32 else dim
    return BLOCK_M, BLOCK_D


# ----------------------------------------------------------------------
# Entry point: forward pass of outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all linear / layer‑norm parameters
                         (float32 tensors, bias=False for Linear layers)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    nomask     = cfg.get("nomask", True)
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_block_sizes(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ fused linear: (left/right proj + three gates) → (R, 5·H)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # [5·H, dim]

    proj = F.linear(x_norm, w_fused)               # (R, 5·H) fp16
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ split, apply mask & gates, write left/right directly into
    #    (B·H, N, N) layout
    # ------------------------------------------------------------------
    left_buf  = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right_buf = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_half = mask.to(torch.float16).contiguous().view(R)  # [R]
        HAS_MASK = 1
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_to_mat_kernel[grid_split](
        proj,
        left_buf,
        right_buf,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left_buf / right_buf : [B·H, N, N]   (fp16)
    out_mat = torch.bmm(left_buf, right_buf.transpose(1, 2))   # [B·H, N, N] fp16
    del left_buf, right_buf

    # ------------------------------------------------------------------
    # 5️⃣ second LayerNorm + multiplication by out_gate (fused kernel)
    # ------------------------------------------------------------------
    out_fused = torch.empty_like(out_gate, dtype=torch.float16, device=device)

    BLOCK_R2, BLOCK_H2 = _choose_block_sizes(R, H)
    grid2 = (triton.cdiv(R, BLOCK_R2),)
    layernorm_mul_from_mat_kernel[grid2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_R2,    # reused name inside kernel
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ final projection back to `dim` → fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] fp16
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
