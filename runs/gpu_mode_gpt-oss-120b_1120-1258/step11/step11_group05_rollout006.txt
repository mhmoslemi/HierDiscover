"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================

The module receives a 4‑D tensor *x* ∈ ℝ^{B×N×N×C} and an optional pair‑wise mask.
The computation (see the PyTorch reference) is:

1️⃣ Row‑wise LayerNorm on the last dimension (C) → FP16.
2️⃣ One fused linear projection that computes
   left_proj, right_proj, left_gate, right_gate, out_gate
   in a single GEMM (R = B·N·N rows, 5·H columns).
3️⃣ A Triton kernel splits the fused result, applies the (optional) mask,
   evaluates the three sigmoids (in FP32 for stability) and writes three
   tensors: ``left``, ``right`` and ``out_gate``.
4️⃣ Batched half‑precision matrix multiplication:
   for every hidden dimension *h* we compute
   out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}
   i.e. a strided‑batched GEMM of shape (B·H, N, N) × (B·H, N, N)^T.
5️⃣ A second Triton kernel fuses a LayerNorm on the intermediate
   ``out`` tensor with element‑wise multiplication by ``out_gate``.
6️⃣ Final linear projection back to ``dim`` (FP16 → FP32).

All heavy matrix‑multiplications use cuBLAS/Tensor‑Cores (torch.matmul /
torch.nn.functional.linear).  The three custom Triton kernels handle the two
LayerNorms and the mask / gate logic, keeping the overall memory traffic
low and exploiting half‑precision throughout.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    inp_ptr,       # const   *[rows, dim]      (float32)
    out_ptr,       # mutable *[rows, dim]      (float16)
    weight_ptr,    # const   *[dim]           (float32) γ
    bias_ptr,      # const   *[dim]           (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean & var (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = inp_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (fp16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = inp_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr_i = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr_i, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """For each row r:
           left  = (proj_left  * mask) * sigmoid(proj_left_gate)
           right = (proj_right * mask) * sigmoid(proj_right_gate)
           out_gate = sigmoid(proj_out_gate)
       All ops in fp16, sigmoid evaluated in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # broadcast mask (1.0 if no mask)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r,
                       mask=mask_r,
                       other=1.0).to(tl.float16)          # [BLOCK_R]
        mval = mval[:, None]                                 # [BLOCK_R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout = [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw    = tl.load(proj_ptr + base,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_raw   = tl.load(proj_ptr + base + H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        left_g_raw  = tl.load(proj_ptr + base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(proj_ptr + base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        out_g_raw   = tl.load(proj_ptr + base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        tl.store(left_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

    mask_m = offs_m < rows

    # ---- mean & var (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = inp_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β, multiply by gate ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x (fp16 → fp32)
        x_vals = tl.load(inp_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]        # fp32

        # γ,β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # fp32

        out = norm * g                                            # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim       = cfg["dim"]
    hidden    = cfg["hidden_dim"]
    eps       = 1e-5
    device    = x.device
    B, N, _, _ = x.shape
    R = B * N * N                     # total rows = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 32           # columns per program
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],          # γ (float32)
        w["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden outputs)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: [5*H, dim]) and cast to fp16
    w_fused = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                            # [5*H, dim]

    # GEMM: (R, dim) @ (5*H, dim).T → (R, 5*H)  (FP16)
    proj = F.linear(x_norm, w_fused)                               # [R, 5*H]
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, evaluate sigmoids (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, hidden, dtype=torch.float16, device=device)
    right     = torch.empty(R, hidden, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, hidden, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        # flatten mask to [R] and keep in half‑precision
        mask_flat = mask.to(torch.float16).reshape(R)

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        hidden,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension:
    #     out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    # Reshape to [B·H, N, N] for a single strided‑batched matmul
    left  = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.reshape(B * hidden, N, N)      # [B·H, N, N]
    right_mat = right.reshape(B * hidden, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)  (FP16, Tensor‑Core)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))

    # Restore original 4‑D layout: [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiply with out_gate (fused)
    # ------------------------------------------------------------------
    out_flat = out.reshape(R, hidden).contiguous()                 # [R, H] fp16
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate.view(R, hidden),
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (fp16 → fp32)
    # ------------------------------------------------------------------
    # Weight shape: [dim, hidden] ; use fp16 matmul then cast to fp32
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, hidden]
    )                                           # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
