def custom_kernel(data):
    x, mask, weights, cfg = data.
    B,N,N,dim = x.shape
    hidden_dim = cfg['hidden_dim']

    # 1st layernorm: flatten to (B*N*N, dim), convert to half for speed.
    x_flat = x.view(-1, dim).to(torch.float16)  # shape (R,dim)
    # torch.nn.functional.layer_norm expects same dtype as input, and weight,bias same dtype.
    # So we need to convert weight,bias to half.
    norm_weight = weights['norm.weight'].to(torch.float16)
    norm_bias = weights['norm.bias'].to(torch.float16)
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(x_flat, (dim,), norm_weight, norm_bias, eps) # still half.

    # 2. fused linear
    fused_weight = torch.cat([
        weights['left_proj.weight'],
        weights['right_proj.weight'],
        weights['left_gate.weight'],
        weights['right_gate.weight'],
        weights['out_gate.weight'],
    ], dim=0).to(torch.float16)  # shape (5*H, dim)

    proj = torch.nn.functional.linear(x_norm, fused_weight) # (R,5H) half

    # split
    splits = torch.split(proj, hidden_dim, dim=1)
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = splits

    # mask
    if cfg.get('nomask', True) or mask is None:
        mask_half = None
    else:
        mask_half = mask.to(torch.float16).view(-1, 1)  # (R,1)

    # compute gates sigmoid
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)   # (R,H)

    # apply mask and gating to left and right
    if mask_half is not None:
        left = left_raw * mask_half * left_gate
        right = right_raw * mask_half * right_gate
    else:
        left = left_raw * left_gate
        right = right_raw * right_gate

    # reshape left/right to (B, H, N, N)
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous() # (B,H,N,N)
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    # now left/right are half
    # GEMM via bmm
    # reshape to (B*H, N, N)
    left_b = left.view(B * hidden_dim, N, N)
    right_b = right.view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_b, right_b.transpose(1,2)) # (B*H,N,N) half

    # reshape to (R, H)
    out_flat = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().view(-1, hidden_dim) # (R,H)

    # 5. second layernorm + gate using kernel
    # Need layernorm weight: to_out_norm.weight dtype float32; we will keep as float32 but kernel loads as float32; kernel expects weight pointer to float32.
    # to_out_norm.bias similarly.
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=x.device)

    BLOCK_M = 256   # may tune
    BLOCK_D = 128   # hidden_dim maybe 128.
    grid3 = (triton.cdiv(out_flat.shape[0], BLOCK_M),)
    layernorm_mul_kernel[grid3](
        out_flat,
        out_gate,
        out_fused,
        weights['to_out_norm.weight'],
        weights['to_out_norm.bias'],
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # final linear
    final_half = torch.nn.functional.linear(out_fused, weights['to_out.weight'].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul (outgoing) – fast Triton + Torch forward pass.
---------------------------------------------------
Algorithm
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16 (Triton).
2️⃣ One fused linear projection produces five tensors
   (left, right, left_gate, right_gate, out_gate) of shape (R, H) with H = hidden_dim.
3️⃣ Apply the optional pair‑wise mask and sigmoid gates with plain PyTorch
   (vectorised element‑wise ops).  The result is reshaped to
   (B·H, N, N) for the batched GEMM.
4️⃣ Batched half‑precision GEMM (Tensor‑Core) implements the cubic
   “∑ₖ left[i,k]·right[j,k]” update:
      out_mat = left @ rightᵀ   → (B·H, N, N)
5️⃣ Row‑wise LayerNorm + element‑wise multiplication with `out_gate`
   (single Triton kernel).  The result is fp16.
6️⃣ Final linear projection back to `dim` (fp16 → fp32).

Only the two LayerNorm kernels are written in Triton; all dense
matrix multiplications and the masking/gating logic use PyTorch
high‑performance kernels.  This keeps the launch overhead low while
still satisfying the “must use a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   *[rows, dim]   (float32)
    y_ptr,               # mutable *[rows, dim]   (float16)
    w_ptr,               # const   *[dim]        (float32)   γ
    b_ptr,               # const   *[dim]        (float32)   β
    rows,                # tl.int32
    dim,                 # tl.int32
    eps,                 # tl.float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalised values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – row‑wise LayerNorm + element‑wise multiplication with gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)   (out_gate)
    y_ptr,            # mutable *[rows, dim]   (float16)
    w_ptr,            # const   *[dim]        (float32)   γ
    b_ptr,            # const   *[dim]        (float32)   β
    rows,             # tl.int32
    dim,              # tl.int32
    eps,              # tl.float32
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean & variance (fp32) -----------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β, multiply with gate -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_val - mean[:, None]) * rstd[:, None]           # fp32

        # γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)      # fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)      # fp32
        norm = norm * w[None, :] + b[None, :]                    # fp32

        # gate value (fp16 → fp32)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        out = norm * gate_val                                    # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

        BLOCK_D=BLOCK_D,

            w["out_gate.weight"],

        mask_half = None
    else:
        mask_half = mask.to(torch.float16).view(R, 1)   # (R,1)

    # sigmoid gates (fp16)
    left_gate   = torch.sigmoid(left_gate_raw)
    right_gate  = torch.sigmoid(right_gate_raw)
    out_gate    = torch.sigmoid(out_gate_raw)          # (R, H)

    # apply mask and gates
    if mask_half is not None:
        left  = left_raw  * mask_half * left_gate
        right = right_raw * mask_half * right_gate
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for batched GEMM (B·H, N, N)  – outgoing Tri‑multiplicative update
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_b  = left.view(B * H, N, N)      # (B·H, N, N)
    right_b = right.view(B * H, N, N)     # (B·H, N, N)

    # batched half‑precision GEMM (Tensor‑Core)
    out_mat = torch.bmm(left_b, right_b.transpose(1, 2))   # (B·H, N, N)

    # reshape to (R, H) for the second LayerNorm
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().view(R, H)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid3 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid3](
        out_flat,
        out_gate,
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate, out_mat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim), fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
