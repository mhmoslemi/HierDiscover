"""
TriMul (outgoing) – fast Triton‑+‑PyTorch forward.
-------------------------------------------------
Algorithm
1️⃣ Row‑wise LayerNorm on the input (float32 → float16) – done with the
   highly‑optimised PyTorch `F.layer_norm` implementation.
2️⃣ Fuse the five linear maps (left/right projection + three gates) into a
   single half‑precision GEMM:  X_norm @ W_fusedᵀ  →  (R, 5·H) with
   R = B·N·N.
3️⃣ Triton kernel `split_fuse_gate_kernel`  
   * reads the fused projection,
   * optionally applies the pair‑wise mask,
   * evaluates the three sigmoid gates,
   * writes the masked/gated left/right tensors directly into the
     flattened buffers `(B·H, N·N)` and stores the output‑gate.
4️⃣ Batched half‑precision GEMM (`torch.bmm`) over the hidden dimension:
        out_mat = left @ rightᵀ                # (B·H, N, N)
5️⃣ LayerNorm + gating for the hidden dimension – performed with
   PyTorch `F.layer_norm` on a tensor of shape (B, N, N, H) followed by
   element‑wise multiplication with `out_gate`.
6️⃣ Final linear projection back to `dim` (half‑precision) and reshape to
   (B, N, N, dim).  The output is returned in float32.

Only step 3 is written in Triton; all dense matrix multiplications are
delegated to cuBLAS (Tensor‑Core) which gives the best possible throughput
on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 3 – split fused projection, apply optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[B*H, N*N]  (fp16)
    right_ptr,         # mutable *[B*H, N*N]  (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – ignored if HAS_MASK=0
    rows,              # tl.int32  = B*N*N
    H,                 # tl.int32  = hidden_dim
    N,                 # tl.int32  = seq_len
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flattened pair (b,i,k) → row r and each hidden index h:
        left  = (proj_left  * mask[r]) * sigmoid(proj_left_gate)
        right = (proj_right * mask[r]) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    The results are written directly into the flattened (B·H, N·N) buffers,
    eliminating any view/permute copy before the batched GEMM.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                                           # positions per matrix

    # split r → batch b and position pos (i·N + k)
    b = offs_r // NN
    pos = offs_r % NN

    # optional mask (scalar 0/1 broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # fp16
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dim in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # base offset into the fused projection matrix
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # (R, BH)

        left_raw  = tl.load(proj_ptr + base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        right_raw = tl.load(proj_ptr + base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # sigmoid – performed in FP32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left_val  = left_raw  * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        # destination index for flattened left/right buffers
        #   batch index = b * H + h
        batch_idx = b[:, None] * H + offs_h[None, :]                 # (R, BH)
        dst_offset = batch_idx * NN + pos[:, None]                    # (R, BH)

        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate (kept in original row layout)
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 “outgoing”
    variant).  Heavy element‑wise work (projection split & gating) is done
    in Triton; the three dense GEMM operations use cuBLAS (Tensor‑Core).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]          # input / output channel dimension (C)
    hidden_dim = cfg["hidden_dim"]   # hidden dimension (H)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N           # total rows for the row‑wise ops
    H = hidden_dim
    NN = N * N

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16) – use PyTorch kernel
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # (R, dim) float32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.half()                        # (R, dim) fp16
    del x_flat, x_norm_fp32

    # --------------------------------------------------------------
    # 2️⃣ Fused linear: five projections → (R, 5·H)
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # (5·H, dim)

    proj = F.linear(x_norm, fused_weight)               # (R, 5·H) fp16
    del fused_weight, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates – write directly to flattened buffers
    # --------------------------------------------------------------
    left_flat  = torch.empty(B * H, NN, dtype=torch.float16, device=device)  # (B·H, N·N)
    right_flat = torch.empty_like(left_flat)

    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)         # (R, H)

    if cfg.get("nomask", True) or mask is None:
        # No mask → treat everything as 1.0
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # mask [B,N,N] -> flatten to (R,)
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 256      # rows per program (tuned)
    BLOCK_H = 32       # hidden‑dim chunk
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left_flat,
        right_flat,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # reshape left/right for the batched GEMM
    left  = left_flat.view(B * H, N, N)       # (B·H, N, N)
    right = right_flat.view(B * H, N, N)      # (B·H, N, N)

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) + element‑wise multiply with out_gate
    # --------------------------------------------------------------
    # bring hidden dim to the last axis: (B, N, N, H)
    out_perm = out_mat.permute(0, 2, 3, 1)                # (B·H→B, N, N, H) – non‑contiguous
    # LayerNorm on the hidden dimension
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )                                                    # (B, N, N, H) fp16

    # reshape out_gate to the same layout and apply the gate
    out_gate_view = out_gate.view(B, N, N, H)            # (B, N, N, H) fp16
    gated = out_norm * out_gate_view                     # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        gated.view(R, H),                               # (R, H)
        weights["to_out.weight"].to(torch.float16),    # (dim, H)
    )                                                    # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)  # (B, N, N, dim) float32
    return final
