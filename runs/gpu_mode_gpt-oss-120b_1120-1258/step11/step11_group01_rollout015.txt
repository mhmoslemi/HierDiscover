"""
High‑performance Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3 / Protenix).

Algorithm
---------
1️⃣  Fast row‑wise LayerNorm (torch.nn.functional.layer_norm) on the
    4‑D input → fp16.

2️⃣  Concatenated linear projection (left/right + 3 gates) with cuBLAS
    (torch.nn.functional.linear).  The weight matrix has shape
    (5·hidden_dim, dim) and is stored in fp16.

3️⃣  Triton kernel `split_fuse_gate_fused_kernel`  
    * unfolds the flat `(B·N·N, 5·hidden_dim)` projection,  
    * applies the optional pair‑wise mask,  
    * evaluates sigmoid gates (fp32 → fp16),  
    * writes the *left* and *right* tensors directly in the layout required
      by the batched GEMM `(B·hidden_dim, N, N)`,  
    * stores the `out_gate` tensor (shape `(B·N·N, hidden_dim)`).

4️⃣  Batched half‑precision GEMM (cuBLAS):
        out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)

5️⃣  Triton kernel `layernorm_mul_strided_kernel`  
    * row‑wise LayerNorm over the hidden dimension (stride = N²),  
    * multiplies the normalized values with the pre‑computed `out_gate`.

6️⃣  Final linear projection back to `dim` (half → float32).

Only kernels 3) and 5) are written in Triton; the rest relies on
highly tuned cuBLAS / torch kernels.  This reduces the number of kernel
launches to two and eliminates the temporary storage of the normalized
input, giving a noticeable speed‑up on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates,
#    write left/right in (B·H, N, N) layout and store out_gate
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]     (fp16)
    left_ptr,          # mutable *[B·H, N, N]   (fp16)
    right_ptr,         # mutable *[B·H, N, N]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]     (fp16)
    mask_ptr,          # const   *[rows]        (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """\
    For each position (b,i,j) (flattened as row index `r`):
        left      = (proj_left  * mask) * sigmoid(proj_left_gate)
        right     = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)
    The left/right tensors are stored as (B·H, N, N) so that a single
    batched GEMM can later compute the TriMul interaction.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ----- decode (b,i,j) from the flat row index -----
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # ----- optional mask (broadcast across hidden dim) -----
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----- loop over hidden dimension blocks -----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # write left/right in (B·H, N, N) layout.
        bh = b_idx[:, None] * H + offs_h[None, :]               # [R, H]
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R, H]

        tl.store(left_ptr  + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate (still (rows, H))
        tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm (over hidden dim) + element‑wise multiplication
#    with the previously computed out_gate.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_strided_kernel(
    x_ptr,          # const   *[B·H, N, N]   (fp16)   GEMM result
    gate_ptr,       # const   *[rows, H]    (fp16)   out_gate
    y_ptr,          # mutable *[rows, H]    (fp16)   fused output
    weight_ptr,     # const   *[H]         (fp32)   γ
    bias_ptr,       # const   *[H]         (fp32)   β
    rows: tl.int32,       # = B·N·N
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """\
    For each (b,i,j) (flattened as `r`), the hidden dimension is stored
    with a stride of N².  This kernel:
        1. computes mean/var over H (fp32),
        2. normalizes, applies γ/β,
        3. multiplies by out_gate,
        4. writes the result as contiguous (rows, H) fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N

    # decode (b,i,j) from flat row index
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # base offset of the first hidden slice (h = 0)
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # --------------------------------------------------------------
    # 1️⃣ accumulate mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalize, γ/β, multiply with out_gate, store (fp16)
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # raw values again
        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        # normalize
        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        # γ, β (fp32)
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # [Hblk]
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # [Hblk]

        norm = norm * w[None, :] + b[None, :]               # fp32

        # out_gate (fp16 → fp32)
        gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``.
                         May also contain other flags.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b,i,j) positions
    H = hidden_dim
    seqsq = N * N

    # ------------------------------------------------------------------
    # 1️⃣ Fast LayerNorm on the 4‑D input (torch kernel, well‑optimised)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension.
    # It returns float32; we immediately cast to fp16 to reduce bandwidth.
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                       # shape [B,N,N,dim] fp16

    # flatten for the subsequent linear projection
    x_norm_flat = x_norm.view(R, dim).contiguous()   # [R, dim] fp16
    del x_norm    # free fp16 tensor early

    # ------------------------------------------------------------------
    # 2️⃣ Concatenated linear projection (left/right + 3 gates)
    # ------------------------------------------------------------------
    # Build a single weight matrix of shape [5*H, dim] in fp16.
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5*H, dim]

    proj = F.linear(x_norm_flat, w_concat)   # (R, 5*H) fp16
    del x_norm_flat, w_concat

    # ------------------------------------------------------------------
    # 3️⃣ Split / gate / mask – write left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask broadcasted over hidden dimension, flatten to [R]
        mask_tensor = mask.to(torch.float16).reshape(R)
        has_mask = 1

    BLOCK_R = 128                     # rows per program – tuned for H100
    BLOCK_H = 32                      # hidden‑dim chunk size
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension:
    #    out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M = 256                     # rows per program for LN2
    BLOCK_H2 = 32                     # hidden‑dim chunk for LN2 (reuse BLOCK_D size)
    grid_lnorm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_strided_kernel[grid_lnorm2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32, [H])
        weights["to_out_norm.bias"],     # β (float32, [H])
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] → fp16
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
