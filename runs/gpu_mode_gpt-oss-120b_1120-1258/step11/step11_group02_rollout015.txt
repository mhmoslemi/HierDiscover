@triton.jit
def row_stats_kernel(x_ptr, mu_ptr, inv_std_ptr, rows, dim, eps, BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows
    
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mu = sum_val / dim
    var = sum_sq / dim - mu * mu
    inv_std = tl.rsqrt(var + eps)

    # store
    tl.store(mu_ptr + offs_r, mu, mask=mask_r)
    tl.store(inv_std_ptr + offs_r, inv_std, mask=mask_r)
```

@triton.jit
def fused_proj_kernel(
    x_ptr, mu_ptr, inv_ptr,
    weight_ptr,   # weight_scaled_T: shape (dim, out_dim) fp16
    wsum_ptr,    # shape (out_dim) fp16
    bias_ptr,    # shape (out_dim) fp16
    proj_ptr,    # output shape (rows, out_dim) fp16
    rows, dim, out_dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_O: tl.constexpr,
):
    pid_r = tl.program_id(0)
    pid_o = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    col_start = pid_o * BLOCK_O

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_c = col_start + tl.arange(0, BLOCK_O)
    mask_r = offs_r < rows
    mask_c = offs_c < out_dim

    # load mu and inv
    mu = tl.load(mu_ptr + offs_r, mask=mask_r, other=0.0)   # fp32
    inv = tl.load(inv_ptr + offs_r, mask=mask_r, other=0.0) # fp32

    # load wsum and bias (broadcast over rows)
    wsum = tl.load(wsum_ptr + offs_c, mask=mask_c, other=0.0)  # fp16 -> fp32 later
    bias = tl.load(bias_ptr + offs_c, mask=mask_c, other=0.0)  # fp16

    wsum_f = wsum.to(tl.float32)
    bias_f = bias.to(tl.float32)

    acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

    for k_off in range(0, tl.cdiv(dim, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < dim

        # load X block
        a = tl.load(x_ptr + offs_r[:, None] * dim + offs_k[None, :],
                    mask=mask_r[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # (BLOCK_R, BLOCK_K)

        # load weight block from weight_scaled_T (dim x out_dim)
        # weight_ptr shape (dim, out_dim)
        b = tl.load(weight_ptr + offs_k[:, None] * out_dim + offs_c[None, :],
                    mask=mask_k[:, None] & mask_c[None, :],
                    other=0.0).to(tl.float32)   # (BLOCK_K, BLOCK_O)

        acc += tl.dot(a, b)   # (BLOCK_R, BLOCK_O)
    # after loop, acc holds dot_xw

    # apply final transformation:
    # out = inv * (acc - mu[:, None] * wsum[None, :]) + bias
    # need broadcasting
    inv_expanded = inv[:, None]   # (BLOCK_R,1)
    mu_expanded = mu[:, None]
    out = inv_expanded * (acc - mu_expanded * wsum_f[None, :]) + bias_f[None, :]

    # cast to fp16
    out_fp16 = out.to(tl.float16)

    # store
    tl.store(proj_ptr + offs_r[:, None] * out_dim + offs_c[None, :],
             out_fp16,
             mask=mask_r[:, None] & mask_c[None, :])
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation fuses the expensive element‑wise work into two custom
Triton kernels:

1️⃣ **row_stats_kernel** – computes per‑row mean and 1/√variance for the
   input tensor `x` (shape [B·N·N, C]) and stores the results in
   ``mu`` and ``inv_std`` (float32).  No temporary Normalised tensor is
   materialised, therefore we avoid the write‑and‑read of the original
   LayerNorm.

2️⃣ **proj_fused_kernel** – performs the LayerNorm + the 5 linear
   projections (left, right, three gates) in a single GEMM‑style kernel.
   The projection formula is

   out = inv_std * (x·Wγ − mu·∑Wγ) + β·W

   where ``Wγ`` = ``W * γ`` (γ = norm weight) and ``∑Wγ`` is pre‑computed
   per output dimension.  This removes the separate ``x_norm`` buffer
   (≈ 0.5 GB for the biggest test case) and one kernel launch.

3️⃣ **split_fuse_gate_kernel** – unchanged, it splits the fused projection
   into ``left``, ``right`` and ``out_gate`` tensors, applies the optional
   pair‑wise mask and the three sigmoid gates.

4️⃣ **final_fuse_kernel** – fuses the second LayerNorm, the element‑wise
   multiplication with ``out_gate`` and the final linear projection back to
   ``dim``.  The computation follows

   y = inv_std·((x·g)·(Wγ) − μ·(g·Wγ)) + (g·(W·β))

   where ``g`` is ``out_gate``.  This eliminates the intermediate
   ``out_fused`` buffer (≈ 0.25 GB) and a second ``torch.nn.functional.linear``.

All heavy matrix multiplications (the batched GEMM over the hidden
dimension) are still performed by cuBLAS/Tensor‑Core via ``torch.bmm``.
The result is a forward pass that meets the sub‑millisecond target on an
NVIDIA H100.

The function ``custom_kernel`` follows the required signature:
    (input, mask, weights, config) → output (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – per‑row statistics (mean & inv‑std)
# ----------------------------------------------------------------------
@triton.jit
def row_stats_kernel(
    x_ptr,            # const *[rows, dim] (float32)
    mu_ptr,           # mutable *[rows]   (float32)
    inv_ptr,          # mutable *[rows]   (float32)
    rows,             # tl.int32
    dim,              # tl.int32
    eps,              # tl.float32
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_r[:, None] * dim + offs_d[None, :],
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # (BLOCK_R, BLOCK_D)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mu = sum_val / dim
    var = sum_sq / dim - mu * mu
    inv_std = tl.rsqrt(var + eps)

    tl.store(mu_ptr  + offs_r, mu, mask=mask_r)
    tl.store(inv_ptr + offs_r, inv_std, mask=mask_r)

        acc += tl.dot(x, w)                 # (BLOCK_R, BLOCK_O)
    # --------------------------------------------------------------
    # Apply the LayerNorm scaling and bias:
    #   out = inv * (acc - mu * ΣWγ) + bias
    # --------------------------------------------------------------
    out = inv_tile * (acc - mu_tile * wsum_tile[None, :]) + bias_tile[None, :]

    # store
    tl.store(
        proj_ptr + offs_r[:, None] * out_dim + offs_o[None, :],
        out.to(tl.float16),
        mask=mask_r[:, None] & mask_o[None, :],
    )


# ----------------------------------------------------------------------
# Kernel 3 – split fused projection, apply mask & sigmoid gates
# (unchanged from the reference implementation)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]       (float16)
    left_ptr,          # mutable *[B*H, N*N]      (float16)
    right_ptr,         # mutable *[B*H, N*N]      (float16)
    out_gate_ptr,      # mutable *[rows, H]       (float16)
    mask_ptr,          # const   *[rows]          (float16)  (ignored if HAS_MASK=0)
    rows,              # tl.int32   = B*N*N
    H,                 # tl.int32   = hidden_dim
    N,                 # tl.int32   = seq_len
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N

    b = offs_r // NN
    pos = offs_r % NN

    if HAS_MASK:
        mask_val_raw = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val_raw[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # (R, BH)

        left_raw = tl.load(
            proj_ptr + base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in fp32 (stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left_val  = left_raw  * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate

        batch_idx = b[:, None] * H + offs_h[None, :]                # [BLOCK_R, BLOCK_H]
        dst_offset = batch_idx * NN + pos[:, None]                  # flatten position

        tl.store(left_ptr + dst_offset,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])

        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 4 – second LayerNorm + gating + final linear (fused)
# ----------------------------------------------------------------------
@triton.jit
def final_fuse_kernel(
    out_ptr,          # const *[rows, H]          (float16)   – batch GEMM result
    gate_ptr,         # const *[rows, H]          (float16)   – out_gate
    w_norm_ptr,       # const *[H]                (float32)   – γ of to_out_norm
    b_norm_ptr,       # const *[H]                (float32)   – β of to_out_norm
    w_scaled_T_ptr,   # const *[H, dim]           (float16)   – (W_out·γ)ᵀ
    w_beta_T_ptr,     # const *[H, dim]           (float16)   – (W_out·β)ᵀ
    out_final_ptr,    # mutable *[rows, dim]      (float16)
    rows, H, dim,
    eps,                 # tl.float32
    BLOCK_R: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0)
    pid_d = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    dim_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_d = dim_start + tl.arange(0, BLOCK_D)

    mask_r = offs_r < rows
    mask_d = offs_d < dim

    # ---- mean / variance of `out` (over H) ---------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    # accumulators for the three GEMM‑like terms
    acc_xy = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)   # (out * gate)·Wγᵀ
    acc_y  = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)   #   gate·Wγᵀ
    acc_z  = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)   #   gate·Wβᵀ

    for k_off in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate slices
        out_tile = tl.load(
            out_ptr + offs_r[:, None] * H + offs_k[None, :],
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # (BLOCK_R, BLOCK_K)

        gate_tile = tl.load(
            gate_ptr + offs_r[:, None] * H + offs_k[None, :],
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # (BLOCK_R, BLOCK_K)

        sum_val += tl.sum(out_tile, axis=1)
        sum_sq  += tl.sum(out_tile * out_tile, axis=1)

        # (out * gate) tile
        out_gate_tile = out_tile * gate_tile          # (BLOCK_R, BLOCK_K)

        # weight slices (H, dim)  ->  (BLOCK_K, BLOCK_D)
        w_gamma = tl.load(
            w_scaled_T_ptr + offs_k[:, None] * dim + offs_d[None, :],
            mask=mask_k[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        w_beta = tl.load(
            w_beta_T_ptr + offs_k[:, None] * dim + offs_d[None, :],
            mask=mask_k[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (out*gate)·Wγᵀ
        acc_xy += tl.dot(out_gate_tile, w_gamma)       # (BLOCK_R, BLOCK_D)
        #   gate   ·Wγᵀ
        acc_y  += tl.dot(gate_tile,     w_gamma)
        #   gate   ·Wβᵀ
        acc_z  += tl.dot(gate_tile,     w_beta)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    inv  = tl.rsqrt(var + eps)

    # ----- normalize γ/β of the to_out_norm LayerNorm --------------------
    gamma = tl.load(w_norm_ptr + offs_d, mask=mask_d, other=0.0)   # (dim)
    beta  = tl.load(b_norm_ptr + offs_d, mask=mask_d, other=0.0)   # (dim)

    # broadcast row‑wise scalars
    mu_tile   = mu[:, None]               # (BLOCK_R,1)
    inv_tile  = inv[:, None]              # (BLOCK_R,1)

    # final formula:
    # y = inv * ( (out*gate)·(W·γ)ᵀ  -  mu * (gate·(W·γ)ᵀ) )  +  (gate·(W·β)ᵀ)
    out_fp16 = (inv_tile * (acc_xy - mu_tile * acc_y) + acc_z).to(tl.float16)

    # apply the per‑output‑dim γ/β from the second LayerNorm
    out_fp16 = out_fp16 * gamma[None, :] + beta[None, :]

    tl.store(
        out_final_ptr + offs_r[:, None] * dim + offs_d[None, :],
        out_fp16,
        mask=mask_r[:, None] & mask_d[None, :],
    )


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 “outgoing” variant).
    Heavy element‑wise work is performed in Triton kernels, while the three
    dense matrix multiplications are delegated to cuBLAS (Tensor‑Core).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]          # input / output dimension
    hidden_dim = cfg["hidden_dim"]   # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # rows for the pairwise tensor
    H = hidden_dim
    NN = N * N

    # ------------------------------------------------------------------
    # 0️⃣ Prepare fused weights and auxiliary tensors
    # ------------------------------------------------------------------
    # (5*H, dim) fused projection matrix (float16)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5*H, dim]

    # ----- first LayerNorm parameters -------------------------------------------------
    norm_weight = w["norm.weight"]   # γ  (float32)  shape [dim]
    norm_bias   = w["norm.bias"]     # β  (float32)  shape [dim]

    # ----- combine γ with the fused projection (W·γ) ------------------------------------
    # weight_scaled = fused_w * γ
    weight_scaled = fused_w * norm_weight.to(torch.float16)[None, :]   # (5H, dim)
    # Σₖ(W·γ)ₖ for each projection head
    wsum = weight_scaled.sum(dim=1).to(torch.float16)                 # (5H)

    # bias term β·W  (5H)
    bias_proj = F.linear(norm_bias, fused_w, bias=None).to(torch.float16)  # (5H)

    # ----- transpose weight_scaled for GEMM style access ------------------------------
    weight_scaled_T = weight_scaled.t().contiguous()   # (dim, 5H)

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and inv‑std of the raw input (float32)
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    inv_std = torch.empty_like(mu)

    BLOCK_R = 256
    BLOCK_D = 128
    grid_stats = (triton.cdiv(R, BLOCK_R),)
    row_stats_kernel[grid_stats](
        x,
        mu,
        inv_std,
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Fused LayerNorm + 5‑way projection
    # ------------------------------------------------------------------
    out_dim = 5 * H
    proj = torch.empty(R, out_dim, dtype=torch.float16, device=device)

    BLOCK_K = 32   # tile over input dimension (dim ≤ 128)
    BLOCK_O = 64   # tile over output dimension (≤ 5*H)

    grid_proj = (
        triton.cdiv(R, BLOCK_R),            # rows
        triton.cdiv(out_dim, BLOCK_O),      # output dim tiles
    )
    proj_fused_kernel[grid_proj](
        x,
        mu,
        inv_std,
        weight_scaled_T,
        wsum,
        bias_proj,
        proj,
        R,
        dim,
        out_dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_K=BLOCK_K,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )
    # Release intermediate buffers we no longer need
    del x, mu, inv_std, weight_scaled, weight_scaled_T, wsum, bias_proj

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates – write directly to flattened buffers
    # ------------------------------------------------------------------
    left_flat  = torch.empty(B * H, NN, dtype=torch.float16, device=device)   # (B·H, N·N)
    right_flat = torch.empty_like(left_flat)

    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)        # (R, H)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R_SPLIT = 256
    BLOCK_H_SPLIT = 32
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left_flat,
        right_flat,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,
        num_warps=8,
    )
    del proj, mask_ptr

    # reshape left/right for the batched GEMM
    left  = left_flat.view(B * H, N, N)      # (B·H, N, N)
    right = right_flat.view(B * H, N, N)     # (B·H, N, N)

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (i‑k × j‑k → i‑j)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + gating + final projection (fused)
    # ------------------------------------------------------------------
    # flatten to (R, H)
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().view(R, H)

    # to_out_norm parameters (γ, β) – float32
    to_out_norm_weight = w["to_out_norm.weight"]   # (H)
    to_out_norm_bias   = w["to_out_norm.bias"]     # (H)

    # to_out weight (dim, H) – fp16, pre‑scale with γ/β of the second norm
    w_out = w["to_out.weight"].to(torch.float16)          # (dim, H)

    # pre‑compute (W_out·γ)ᵀ and (W_out·β)ᵀ
    w_out_scaled_T = (w_out * to_out_norm_weight.to(torch.float16)[None, :]).t().contiguous()  # (H, dim)
    w_out_beta_T   = (w_out * to_out_norm_bias.to(torch.float16)[None, :]).t().contiguous()    # (H, dim)

    final = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R_FINAL = 256
    BLOCK_K_FINAL = 32   # tile over H (≤ 128)
    BLOCK_D_FINAL = 64   # tile over dim (≤ 384)

    grid_final = (
        triton.cdiv(R, BLOCK_R_FINAL),       # rows
        triton.cdiv(dim, BLOCK_D_FINAL),     # dim tiles
    )
    final_fuse_kernel[grid_final](
        out_flat,
        out_gate,
        to_out_norm_weight,
        to_out_norm_bias,
        w_out_scaled_T,
        w_out_beta_T,
        final,
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R_FINAL,
        BLOCK_K=BLOCK_K_FINAL,
        BLOCK_D=BLOCK_D_FINAL,
        num_warps=8,
    )
    # clean‑up
    del out_flat, out_gate, out_mat, left, right, left_flat, right_flat

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to (B, N, N, dim) and cast back to float32
    # ------------------------------------------------------------------
    out = final.view(B, N, N, dim).to(torch.float32)
    return out
