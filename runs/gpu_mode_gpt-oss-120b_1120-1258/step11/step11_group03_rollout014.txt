@triton.jit
def fused_norm_outgate_linear_kernel(
    out_ptr, gate_ptr, weight_norm_ptr, bias_norm_ptr,
    weight_out_ptr,
    y_ptr,
    R, H, D,
    eps,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_r = tl.program_id(0)
    pid_d = tl.program_id(1)
    row_start = pid_r * BLOCK_R
    dim_start = pid_d * BLOCK_D
    offs_r = row_start + tl.arange(0, BLOCK_R)
    offs_d = dim_start + tl.arange(0, BLOCK_D)
    mask_r = offs_r < R
    mask_d = offs_d < D

    # compute mean and var across H per row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        h_start = h_off * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out slice
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)  # [R, Hc]
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulator for final output
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    # load norm gamma and beta
    gamma = tl.load(weight_norm_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0).to(tl.float32)  # [H] (gamma)
    beta = tl.load(bias_norm_ptr + tl.arange(0, H), mask=tl.arange(0, H) < H, other=0.0).to(tl.float32)    # [H] (beta)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        h_start = h_off * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate slices
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # normalize
        norm = (out_vals - mean[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]

        # apply gate
        gated = norm * gate_vals

        # load weight_out slice transposed (Hc, Dblock)
        # offs_d limited by mask_d; we need full block even if partial
        w = tl.load(weight_out_ptr + offs_h[:, None] + (dim_start + offs_d[None, :]) * H,
                    mask=mask_h[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)  # shape [Hc, Dblock]

        # dot product: gated (R x Hc) dot w (Hc x Dblock) -> (R, Dblock)
        acc += tl.dot(gated, w)

    # store result
    ptr_out = y_ptr + offs_r[:, None] * D + offs_d[None, :]
    tl.store(ptr_out,
             acc.to(tl.float32),
             mask=mask_r[:, None] & mask_d[None, :])

"""
custom_kernel
==============

Highly‑optimized forward implementation of the *outgoing* TriMul operator
used in AlphaFold‑3, Chai, Protenix, …  The implementation fuses as many
operations as possible into Triton kernels to minimise memory traffic:

1️⃣ Row‑wise LayerNorm on the input (fp32 → fp16).
2️⃣ Single half‑precision GEMM that computes the five linear projections
   (left, right, left‑gate, right‑gate, out‑gate) in one call.
3️⃣ Triton kernel ``split_fuse_gate_kernel`` extracts the three
   projections, applies the optional pair‑wise mask, computes the three
   σ‑gates and writes the tensors ``left``, ``right`` and ``out_gate`` in
   layout ``[B, H, N, N]`` (no extra transposes).
4️⃣ Batched half‑precision matrix multiplication:
        out[b·H+i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]ᵀ
   executed with ``torch.bmm`` (Tensor‑cores).
5️⃣ **Fused** kernel ``norm_gate_linear_kernel`` performs the second
   LayerNorm over the hidden dimension, multiplies by ``out_gate`` and
   projects back to ``dim`` with the final linear weight in a *single*
   pass, eliminating the intermediate storage used in the reference
   implementation.
6️⃣ Reshape to ``[B, N, N, dim]`` and return ``float32`` tensor.

All kernels target Triton 3.3.1 and an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean / variance (accumulate in fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (fp16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & σ‑gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B, H, N, N] (float16)
    right_ptr,         # mutable *[B, H, N, N] (float16)
    out_gate_ptr,      # mutable *[B, H, N, N] (float16)
    mask_ptr,          # const   *[rows]      (float16) – ignored if HAS_MASK=0
    rows: tl.int32,    # R = B·N·N
    N: tl.int32,       # sequence length
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each (b,i,k) row:
        left  = (proj_left * mask)   * σ(proj_left_gate)
        right = (proj_right* mask)   * σ(proj_right_gate)
        out_gate = σ(proj_out_gate)
    Results are written directly in layout [B, H, N, N].
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- decode (b,i,k) from linear row id ----
    seq_sq = N * N
    b = offs_r // seq_sq
    rem = offs_r - b * seq_sq
    i = rem // N
    k = rem - i * N

    # ---- optional pair‑wise mask (scalar per row) ----
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- loop over hidden dimension in chunks ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, Hc]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw    = tl.load(left_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_raw   = tl.load(right_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        left_g_raw  = tl.load(left_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(right_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        out_g_raw   = tl.load(out_gate_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # ---- sigmoid (fp32 for stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # ---- destination offset for layout [B, H, N, N] ----
        # flat = ((b * H + h) * N + i) * N + k
        b_exp = b[:, None]             # [R,1]
        i_exp = i[:, None]
        k_exp = k[:, None]
        h_exp = offs_h[None, :]        # [1,Hc]

        dst_off = ((b_exp * H + h_exp) * N + i_exp) * N + k_exp   # [R,Hc]

        # ---- store results ----
        tl.store(left_ptr      + dst_off, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, out_gate,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fused LayerNorm + out‑gate multiplication + final linear projection
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_linear_kernel(
    out_ptr,                 # const   *[R, H]            (float16)  – result of BMM
    gate_ptr,                # const   *[R, H]            (float16)
    norm_weight_ptr,         # const   *[H]               (float32) γ of to_out_norm
    norm_bias_ptr,           # const   *[H]               (float32) β of to_out_norm
    proj_weight_ptr,         # const   *[dim, H]          (float16)  – to_out.weight
    out_final_ptr,           # mutable *[R, dim]          (float32)
    R: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # output‑dim per program
    BLOCK_H: tl.constexpr,   # hidden‑dim chunk
):
    """
    For each row r (corresponding to a distinct (b,i,j) pair):
        1) Layer‑norm over its hidden vector (size H)
        2) Element‑wise multiplication by the out‑gate
        3) Linear projection with weight ‑> (dim)
    All operations are fused to avoid the intermediate tensor
    of shape (R, H).
    """
    pid_r = tl.program_id(0)
    pid_d = tl.program_id(1)

    row_start = pid_r * BLOCK_R
    dim_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    offs_d = dim_start + tl.arange(0, BLOCK_D)          # [BLOCK_D]

    mask_r = offs_r < R
    mask_d = offs_d < dim

    # ------------------------------------------------------------------
    # 1) Compute mean / variance of each row over the hidden dim
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        h_start = h_off * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [Rblk, Hc]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) Normalise, apply γ/β, multiply with out‑gate and accumulate
    #    the final projection (dot with weight matrix)
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        h_start = h_off * BLOCK_H
        offs_h = h_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # load raw values and gate
        out_vals  = tl.load(out_ptr  + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)      # [Rblk, Hc]
        gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)      # [Rblk, Hc]

        # normalise
        norm = (out_vals - mean[:, None]) * rstd[:, None]   # [Rblk, Hc]

        # load γ/β for this chunk
        gamma_h = tl.load(norm_weight_ptr + offs_h,
                          mask=mask_h, other=0.0).to(tl.float32)   # [Hc]
        beta_h  = tl.load(norm_bias_ptr   + offs_h,
                          mask=mask_h, other=0.0).to(tl.float32)   # [Hc]

        norm = norm * gamma_h[None, :] + beta_h[None, :]    # [Rblk, Hc]

        # apply out‑gate
        gated = norm * gate_vals                           # [Rblk, Hc]

        # load slice of final linear weight (transposed: [Hc, Dblk])
        w = tl.load(proj_weight_ptr + offs_h[:, None] +
                    (dim_start + offs_d[None, :]) * H,
                    mask=mask_h[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)             # [Hc, Dblk]

        # accumulate contribution for this hidden chunk
        acc += tl.dot(gated, w)                            # (Rblk, Dblk)

    # ------------------------------------------------------------------
    # 3) Store final result
    # ------------------------------------------------------------------
    out_ptr_store = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(out_ptr_store,
             acc,
             mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Helper to pick modest block sizes
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    """Simple heuristic for block sizes used by the LayerNorm kernels."""
    BLOCK_M = 256 if rows >= 8192 else 128
    BLOCK_D = 64 if dim >= 64 else dim
    return BLOCK_M, BLOCK_D


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    nomask     = cfg.get("nomask", True)
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total rows for row‑wise kernels
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_block_sizes(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, w_fused)        # (R, 5*H) fp16
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and σ‑gates → left, right, out_gate in [B, H, N, N]
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if nomask:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        HAS_MASK = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)  # [R] 0/1
        HAS_MASK = 1

    BLOCK_R = 128       # rows per program (tunable)
    BLOCK_H = 32        # hidden‑dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        N,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] – no extra transpose needed
    left_mat  = left.view(B * H, N, N)               # (B·H, N, N) fp16
    right_mat = right.view(B * H, N, N)              # (B·H, N, N) fp16

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N) fp16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, multiplication by out_gate and final linear
    # ------------------------------------------------------------------
    # bring tensors to shape [R, H]  (R = B·N·N)
    out      = out_mat.view(B, H, N, N)                     # [B, H, N, N]
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # (R, H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R, H)

    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_R2, BLOCK_D2 = _choose_block_sizes(R, dim)   # reuse heuristic
    # Ensure hidden‑dim chunk size is reasonable for the fused kernel
    BLOCK_H2 = 32
    grid_fused = (triton.cdiv(R, BLOCK_R2), triton.cdiv(dim, BLOCK_D2))
    norm_gate_linear_kernel[grid_fused](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        weights["to_out.weight"].to(torch.float16),   # [dim, H] fp16
        final,
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_D=BLOCK_D2,
        BLOCK_H=BLOCK_H2,
        num_warps=4,
    )
    del out_flat, out_gate_flat, out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to [B, N, N, dim] and return
    # ------------------------------------------------------------------
    final = final.view(B, N, N, dim)   # (B, N, N, dim)  float32
    return final
