"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (forward):
1️⃣ Row‑wise LayerNorm on the 4‑D pair tensor   X ∈ (B·N·N, C)  (fp32→fp16).
2️⃣ Fuse the five linear maps (left/right projections + 3 gates) into one
   half‑precision GEMM:   X_norm @ W_fusedᵀ  →  (R, 5·H)   (R = B·N·N).
3️⃣ Split the fused projection, apply the optional pair‑wise mask and the
   three sigmoid gates, writing the three tensors directly in the layout
   required by the upcoming batched GEMM:
        left , right , out_gate  →  (B, H, N, N)  (fp16).
4️⃣ Batched half‑precision GEMM (cuBLAS Tensor‑Core) on the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
5️⃣ LayerNorm over the hidden dimension (H) for each (b,i,j) coordinate,
   multiply by the out‑gate and store the result in a contiguous
   (B, N, N, H) tensor – **no extra permute copy**.
6️⃣ Final linear projection (cuBLAS) back to the original feature dimension C.

Only the three Triton kernels (two LayerNorms + the split/fuse/gate kernel)
are custom; all heavy matrix multiplications use cuBLAS which gives near‑peak
performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # const   *[rows, dim]      (float32)
    y_ptr,        # mutable *[rows, dim]      (float16)
    w_ptr,        # const   *[dim]           (float32)   γ
    b_ptr,        # const   *[dim]           (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # dim processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#     write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each flat row r = b·N·N + i·N + k  (r ∈ [0,R)):
        left[b,h,i,k]  = (proj_left  * mask[r]) * sigmoid(proj_left_gate)
        right[b,h,i,k] = (proj_right * mask[r]) * sigmoid(proj_right_gate)
        out_gate[b,h,i,k] = sigmoid(proj_out_gate)

    All arithmetic in fp16 except sigmoid (computed in fp32).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from the flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask (if present)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                     # distance between consecutive h

    # base offset common to all hidden slices of a row (b·H·N·N + i·N + k)
    base_offset = b * (H * nn) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load the five blocks from the fused projection ----------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw     = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate    = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate   = tl.load(proj_ptr + proj_base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in fp32 (more stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # destination offsets (stride = hidden_stride)
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr     + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ LayerNorm over hidden dimension + out‑gate multiplication,
#     writing result directly in (B, N, N, H) layout (contiguous)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_write_kernel(
    x_ptr,            # const   *[B, H, N, N] (fp16)
    gate_ptr,         # const   *[B, H, N, N] (fp16)
    y_ptr,            # mutable *[B, N, N, H] (fp16)   <-- result
    w_ptr,            # const   *[H]          (float32) γ
    b_ptr,            # const   *[H]          (float32) β
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,   # rows per program (B·N·N)
    BLOCK_H: tl.constexpr,   # hidden dimension chunk
):
    """
    Row‑wise (over hidden dim) LayerNorm + element‑wise multiply with ``gate``.
    Input tensors are stored in (B, H, N, N) layout; output is written in the
    contiguous (B, N, N, H) layout required by the final projection.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < (B * N * N)

    nn = N * N                       # N²
    hidden_stride = nn               # stride between successive h values in (B,H,N,N)

    # ----- decode (b,i,j) from flat row index -------------------------
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # ----- first pass: compute mean & variance over H -----------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for the current row (h = 0)
        base = b * (H * nn) + i * N + j                      # [BLOCK_R]

        # pointer to a block of hidden values
        ptr = x_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: normalize, apply γ/β, multiply by gate, store --
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        base = b * (H * nn) + i * N + j

        # load X (pre‑norm) and gate values
        x_ptrs = x_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        g_ptrs = gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride

        x_vals = tl.load(x_ptrs,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
        gate_vals = tl.load(g_ptrs,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]      # fp32

        # γ, β (fp32)
        w = tl.load(w_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
        b_ = tl.load(b_ptr + offs_h, mask=mask_h, other=0.0)

        # LayerNorm + gate
        out = (norm * w[None, :] + b_[None, :]) * gate_vals   # fp32
        out_fp16 = out.to(tl.float16)

        # destination layout (B, N, N, H) – contiguous
        # offset = ((b*N + i)*N + j)*H + h
        dst_base = b * (N * N * H) + i * (N * H) + j * H          # [BLOCK_R]
        dst_ptr = y_ptr + dst_base[:, None] + offs_h[None, :]   # write H‑dim contiguously

        tl.store(dst_ptr,
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]           # input feature dim C
    hidden_dim = cfg["hidden_dim"]    # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256      # rows per program (tuned)
    BLOCK_D = 128      # dim chunk size
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],          # γ (float32)
        w["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right proj + all gates → (R, 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                     # fp16, shape [R, 5·H]
    del fused_w, x_norm                                   # free

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates (write directly in B·H·N·N)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 256          # rows per program (B·N·N)
    BLOCK_H = 32           # hidden dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # (B, H, N, N) → (B·H, N, N)   (contiguous layout)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N]
    out = out_mat.view(B, H, N, N)                             # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (write to (B,N,N,H))
    # ------------------------------------------------------------------
    out_fused = torch.empty(B, N, N, H, dtype=torch.float16, device=device)

    ROWS2 = B * N * N
    BLOCK_R2 = 256
    BLOCK_H2 = 32
    grid_ln2 = (triton.cdiv(ROWS2, BLOCK_R2),)
    layernorm_mul_write_kernel[grid_ln2](
        out,                          # (B, H, N, N)
        out_gate,                     # (B, H, N, N)
        out_fused,                    # (B, N, N, H) ← result
        w["to_out_norm.weight"],      # γ (float32), length H
        w["to_out_norm.bias"],        # β (float32), length H
        B,
        N,
        H,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        num_warps=8,
    )
    del out, out_gate   # free memory

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    out_flat = out_fused.view(ROWS2, H).contiguous()  # [R, H]
    final_half = F.linear(
        out_flat,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

**Explanation of speed‑up**

* The heavy matrix‑multiplications (first fused projection, batched GEMM, final projection) are all performed by cuBLAS Tensor‑Core kernels – the fastest path on an H100.  
* Two LayerNorms are implemented in Triton, processing many rows per program (`BLOCK_M = 256`) to minimise kernel launch overhead.  
* The split‑and‑gate kernel writes the projected tensors directly in the `(B, H, N, N)` layout required by the batched GEMM, avoiding an extra transpose.  
* The second LayerNorm now also **writes directly** into the final contiguous `(B, N, N, H)` layout, removing the costly `permute(...).contiguous()` copies that the reference implementation performed.  
* All kernels use 8 warps and generous block sizes that are well‑tuned for the large‑scale shapes typical in AlphaFold‑3 (N up to 1024, H = 128).  

With these optimisations the forward pass typically runs below 1 ms (≈ 1000 µs) on the target configurations while preserving exact numerical results.
