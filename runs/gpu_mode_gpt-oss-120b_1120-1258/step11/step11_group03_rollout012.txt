"""
Highly‑optimized forward implementation of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16).
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single half‑precision GEMM (torch.nn.functional.linear).
3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * extracts the five slices from the fused GEMM,
    * applies the optional pair‑wise mask,
    * computes three σ‑gates,
    * writes the three tensors **left**, **right**, **out_gate** directly in
      layout **[B, H, N, N]** (no extra transposes later).
4️⃣  Batched half‑precision matrix multiplication:
        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]ᵀ
   performed with a single `torch.bmm` call (cuBLAS Tensor‑cores).
5️⃣  Row‑wise LayerNorm + element‑wise multiplication with `out_gate`
   (fused in `layernorm_mul_kernel`).
6️⃣  Final linear projection back to the model dimension.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm  (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean / variance (accumulate in fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (fp16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & σ‑gates, write in [B, H, N, N]
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B, H, N, N] (float16)
    right_ptr,         # mutable *[B, H, N, N] (float16)
    out_gate_ptr,      # mutable *[B, H, N, N] (float16)
    mask_ptr,          # const   *[rows]      (float16) – ignored if HAS_MASK=0
    rows: tl.int32,    # R = B·N·N
    N: tl.int32,       # sequence length
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For every (b,i,k) row:
        left  = (proj_left  * mask) * σ(proj_left_gate)
        right = (proj_right * mask) * σ(proj_right_gate)
        out_gate = σ(proj_out_gate)

    The three outputs are stored directly in layout [B, H, N, N]
    (no extra transposes later).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- decode (b,i,k) from linear row id ----
    seq_sq = N * N
    b = offs_r // seq_sq
    rem = offs_r - b * seq_sq
    i = rem // N
    k = rem - i * N

    # ---- optional pair‑wise mask (scalar per row) ----
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- loop over hidden dimension in chunks ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate]
        # each slice has size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, Hc]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw    = tl.load(left_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_raw   = tl.load(right_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        left_g_raw  = tl.load(left_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(right_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        out_g_raw   = tl.load(out_gate_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # ---- sigmoid (computed in fp32 for numerical stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----
        left  = left_raw  * m_val * left_gate   # [R, Hc]
        right = right_raw * m_val * right_gate # [R, Hc]

        # ---- compute flat destination offsets for [B, H, N, N] ----
        # flat = ((b * H + h) * N + i) * N + k
        b_exp = b[:, None]             # [R,1]
        i_exp = i[:, None]
        k_exp = k[:, None]
        h_exp = offs_h[None, :]        # [1,Hc]

        dst_off = ((b_exp * H + h_exp) * N + i_exp) * N + k_exp   # [R,Hc]

        # ---- store ----
        tl.store(left_ptr      + dst_off, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, out_gate,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise LayerNorm + multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)
    y_ptr,            # mutable *[rows, dim]   (float16)
    weight_ptr,       # const   *[dim]        (float32) γ
    bias_ptr,         # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean / variance (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β, multiply with gate, store (fp16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        norm = norm * w[None, :] + b[None, :]                     # fp32

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                                        # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Helper: pick modest block sizes (tuned for the tested problem sizes)
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    """Return (BLOCK_M, BLOCK_D) for row‑wise kernels."""
    BLOCK_M = 256 if rows >= 8192 else 128
    BLOCK_D = 64 if dim >= 64 else dim
    return BLOCK_M, BLOCK_D


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    eps         = 1e-5
    nomask      = cfg.get("nomask", True)
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_block_sizes(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R, dim, eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + 3 gates)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices into a single [5*H, dim] matrix
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, w_fused)       # (R, 5*H) fp16
    del w_fused, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, compute σ‑gates  →  left / right / out_gate in [B, H, N, N]
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if nomask or mask is None:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy
        HAS_MASK = 0
    else:
        # mask is shape [B, N, N]; flatten to [R] (0/1)
        mask_flat = mask.to(torch.float16).reshape(R)
        HAS_MASK = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        N,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to [B*H, N, N] – no extra transpose needed
    left_mat  = left.view(B * H, N, N)               # (B·H, N, N)
    right_mat = right.view(B * H, N, N)              # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # back to [B, H, N, N]
    out = out_mat.view(B, H, N, N)
    del left_mat, right_mat, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication by out_gate (fused)
    # ------------------------------------------------------------------
    # flatten to [R, H] for the kernel
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # (B,N,N,H) → (R,H)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)

    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2, BLOCK_D2 = _choose_block_sizes(R, H)
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate_flat, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H] fp16
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

if __name__ == "__main__":
    torch.manual_seed(0)

    # small random test
    cfg = {"dim": 128, "hidden_dim": 128, "nomask": False}
    B, N = 2, 64
    x = torch.randn(B, N, N, cfg["dim"], dtype=torch.float32, device="cuda")
    mask = (torch.randn(B, N, N) > 0).to(torch.float16).to("cuda")

    # dummy weights (bias=False for all Linear layers as in the reference)
    weights = {
        "norm.weight": torch.ones(cfg["dim"], device="cuda"),
        "norm.bias": torch.zeros(cfg["dim"], device="cuda"),
        "left_proj.weight": torch.randn(cfg["hidden_dim"], cfg["dim"], device="cuda"),
        "right_proj.weight": torch.randn(cfg["hidden_dim"], cfg["dim"], device="cuda"),
        "left_gate.weight": torch.randn(cfg["hidden_dim"], cfg["dim"], device="cuda"),
        "right_gate.weight": torch.randn(cfg["hidden_dim"], cfg["dim"], device="cuda"),
        "out_gate.weight": torch.randn(cfg["hidden_dim"], cfg["dim"], device="cuda"),
        "to_out_norm.weight": torch.ones(cfg["hidden_dim"], device="cuda"),
        "to_out_norm.bias": torch.zeros(cfg["hidden_dim"], device="cuda"),
        "to_out.weight": torch.randn(cfg["dim"], cfg["hidden_dim"], device="cuda"),
    }

    out_ref = None
    # reference implementation (slow, for verification)
    class TriMulRef(torch.nn.Module):
        def __init__(self, dim, hidden_dim):
            super().__init__()
            self.norm = torch.nn.LayerNorm(dim)
            self.left_proj = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.right_proj = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.left_gate = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.right_gate = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.out_gate = torch.nn.Linear(dim, hidden_dim, bias=False)
            self.to_out_norm = torch.nn.LayerNorm(hidden_dim)
            self.to_out = torch.nn.Linear(hidden_dim, dim, bias=False)

            # load weights
            self.norm.weight.data = weights["norm.weight"]
            self.norm.bias.data = weights["norm.bias"]
            self.left_proj.weight.data = weights["left_proj.weight"]
            self.right_proj.weight.data = weights["right_proj.weight"]
            self.left_gate.weight.data = weights["left_gate.weight"]
            self.right_gate.weight.data = weights["right_gate.weight"]
            self.out_gate.weight.data = weights["out_gate.weight"]
            self.to_out_norm.weight.data = weights["to_out_norm.weight"]
            self.to_out_norm.bias.data = weights["to_out_norm.bias"]
            self.to_out.weight.data = weights["to_out.weight"]

        def forward(self, x, mask):
            x = self.norm(x)
            left = self.left_proj(x)
            right = self.right_proj(x)

            left = left * mask.unsqueeze(-1)
            right = right * mask.unsqueeze(-1)

            left_gate = torch.sigmoid(self.left_gate(x))
            right_gate = torch.sigmoid(self.right_gate(x))
            out_gate = torch.sigmoid(self.out_gate(x))

            left = left * left_gate
            right = right * right_gate

            out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)
            out = self.to_out_norm(out)
            out = out * out_gate
            out = self.to_out(out)
            return out

    ref_mod = TriMulRef(cfg["dim"], cfg["hidden_dim"]).cuda()
    with torch.no_grad():
        out_ref = ref_mod(x, mask)

    out_opt = custom_kernel((x, mask, weights, cfg))

    print("max abs diff:", (out_ref - out_opt).abs().max().item())
