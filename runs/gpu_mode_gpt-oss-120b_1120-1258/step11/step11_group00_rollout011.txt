def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; eps = 1e-5
    B,N,N2, dim2 = x.shape; (maybe rename)
    rows = B * N * N
    H = hidden_dim.

    # Step 1: Input layernorm using torch
    x_norm_fp32 = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm_fp32.to(torch.float16)

    # Step 2: Fused projection
    w_fused = torch.cat([...], dim=0).to(torch.float16)
    proj = F.linear(x_norm, w_fused)   # [rows, 5*H], fp16

    # Step 3: Split, mask & gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device= device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device= device)
    out_gate = torch.empty(rows, H, dtype=torch.float16, device=device)
    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(rows)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(rows, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        rows,
        H,
        B,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj

    # Step 4: Batched gemm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    del left_mat, right_mat, left, right

    # Step 5: Second LayerNorm + gate
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    out_flat = out.view(rows, H)
    del out

    to_out_norm_weight_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias_fp16 = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(out_flat, (H,), weight=to_out_norm_weight_fp16, bias=to_out_norm_bias_fp16, eps=eps)

    gated_norm = out_norm * out_gate
    del out_norm, out_gate

    # Step 6: final projection
    final_half = F.linear(gated_norm, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
We implement the forward of the “outgoing” TriMul operator.  
Only the *split‑and‑gate* step is written in Triton; the rest uses
high‑performance PyTorch kernels (LayerNorm, GEMM, Linear).  

Algorithm
---------
1️⃣  Input LayerNorm (FP32 → FP16) – torch.nn.functional.layer_norm.  
2️⃣  One fused Linear produces the five tensors  
    (left_proj, right_proj, left_gate, right_gate, out_gate) in FP16.  
3️⃣  `split_fuse_gate_kernel` writes the two projected tensors directly
    into the layout required for the batched GEMM ([B, H, N, N]),
    applies the optional mask and the sigmoid gates, and stores the
    out‑gate (shape [B·N·N, H]).            ← **Triton kernel**
4️⃣  Batched GEMM (`torch.bmm`) computes the tri‑multiplicative update:
    `out = left @ rightᵀ`  (shape [B·H, N, N]).  
5️⃣  Second LayerNorm + out‑gate multiplication – torch.nn.functional.layer_norm.  
6️⃣  Final Linear projection back to `dim` (FP16 GEMM) and cast to FP32.

All heavy O(N³) work is done by cuBLAS (Tensor‑Cores); the Triton kernel
only performs the mask/gate fusion, which dramatically reduces memory
traffic and eliminates two costly custom LayerNorm kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates, write `left`/`right`
#     directly into [B, H, N, N] layout and store `out_gate`
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]  (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,    # rows = B * N * N
    H: tl.int32,
    B: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    NN = N * N                     # N²
    stride_h = NN                  # distance between two consecutive hidden slices
    stride_b = H * NN              # distance between two consecutive batches

    # ---- decode (b, i, k) from the flat row index ----
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # ---- optional mask (broadcast over hidden) ----
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process the hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ---- contiguous slices inside the fused projection ----
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ---- sigmoid (computed in fp32 for stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- store left/right into [B, H, N, N] layout ----
        # flat offset for (b,i,k) in the first hidden slice:
        #   offset = b * stride_b + i * N + k
        base0 = b * stride_b + i * N + k                     # [BLOCK_R]

        left_dst_ptr  = left_ptr  + base0[:, None] + offs_h[None, :] * stride_h
        right_dst_ptr = right_ptr + base0[:, None] + offs_h[None, :] * stride_h

        tl.store(left_dst_ptr,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ---- store out_gate (still flat [rows, H]) ----
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst_ptr,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys (torch.float32 tensors)
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    rows = B * N * N          # total number of (b,i,k) triples
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # PyTorch’s native LayerNorm is highly optimized for the large
    # batch of rows we have, so we use it instead of a custom Triton kernel.
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                            # [B, N, N, dim]  fp32
    x_norm = x_norm_fp32.to(torch.float16)       # cast once for the rest of the pipeline
    del x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ One fused Linear → [rows, 5*H]  (FP16)
    # ------------------------------------------------------------------
    # Concatenate the five projection matrices and perform a single GEMM.
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*H, dim]
    proj = F.linear(x_norm, w_fused)        # [rows, 5*H]  fp16
    del w_fused, x_norm                     # free intermediate memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid gates, write `left`/`right` and `out_gate`
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(rows, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(torch.float16).reshape(rows)
        mask_ptr = mask_fp16

    # Tunable tile sizes – work well for the supported dimensions
    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(rows, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        rows,
        H,
        B,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM:  out = left @ rightᵀ   (B*H, N, N)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B*H, N, N]
    right_mat = right.view(B * H, N, N)                    # [B*H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (both in FP16)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] → flatten to [rows, H] for LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(rows, H)                                        # [rows, H]
    del out, out_mat

    # Convert LN parameters to FP16 (input is now FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                                            # [rows, H]  fp16

    gated_norm = out_norm * out_gate                # element‑wise gate
    del out_norm, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final Linear projection back to `dim`  (FP16 GEMM) → FP32 output
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(gated_norm, to_out_w)            # [rows, dim]  fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
