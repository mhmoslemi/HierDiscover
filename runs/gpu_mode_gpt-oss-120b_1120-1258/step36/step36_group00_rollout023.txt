"""
Outgoing TriMul (TriMul‑out) – Triton‑accelerated inference

The forward pass consists of:
1️⃣ Row‑wise LayerNorm (float32 → float16) over the last dimension `dim`.
2️⃣ One fused linear projection that computes the five hidden‑dim
   maps (left, right, left_gate, right_gate, out_gate) in a single
   torch.mm (FP16) call.
3️⃣ Triton kernel `split_fuse_gate_kernel` that
   * reshapes the fused projection,
   * applies the optional pairwise mask,
   * computes the three sigmoid gates,
   * stores `left`, `right` and `out_gate` (all FP16).
4️⃣ Batched GEMM (`torch.bmm`) evaluates the core einsum  
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
5️⃣ Triton kernel `final_norm_proj_kernel` that **fuses**:
   * per‑position mean / variance across the hidden dimension `H`,
   * the second LayerNorm (`to_out_norm`),
   * multiplication by `out_gate`,
   * projection back to the original dimension `dim`.

Both passes over the intermediate tensor `out` are performed in a
single kernel: the first pass writes the values to shared memory while
accumulating the statistics; the second pass reads them from shared
memory, normalises, applies the gate, and performs a fused GEMM with
the transposed `to_out` weight.  This halves the global‑memory traffic
relative to the reference implementation.

All kernels run in mixed‑precision (FP16 for the bulk of the arithmetic,
FP32 for the reductions) and are launch‑parameterised for the
hardware‑friendly block sizes used on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, D]  FP32 input
    y_ptr,    # [R, D]  FP16 output
    w_ptr,    # [D]    γ (FP32)
    b_ptr,    # [D]    β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- first pass: mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalised values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r -> (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride of one hidden plane in dest tensors
    base_offset = b * (H * hidden_stride) + i * N + k   # start offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # fused layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (stable, in FP32)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ + 6️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_norm_proj_kernel(
    out_ptr,        # [B, H, N, N] FP16  (output of the einsum)
    gate_ptr,       # [B, H, N, N] FP16  (out_gate)
    ln_w_ptr,       # [H] FP32            (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32            (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16       (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, N, dim, eps,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)        # tiles over rows (R = B·N·N)
    pid_n = tl.program_id(1)        # tiles over output dimension

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b, i, j) from flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # ------------------------------------------------------------------
    #  Shared memory buffers for a full hidden vector of each row.
    #  H ≤ 128 in all configs → we can allocate the whole vector.
    # ------------------------------------------------------------------
    sh_out = tl.alloc_shared([BLOCK_M, H], dtype=tl.float16)
    sh_gate = tl.alloc_shared([BLOCK_M, H], dtype=tl.float16)

    # ------------------------------------------------------------------
    #  First pass: load `out` & `gate` into shared memory and compute
    #  per‑row mean / variance.
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # global pointers
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)          # FP16
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)          # FP16

        # store to shared memory
        sh_out_ptr  = sh_out + offs_m[:, None] * H + offs_k[None, :]
        sh_gate_ptr = sh_gate + offs_m[:, None] * H + offs_k[None, :]
        tl.store(sh_out_ptr,  vals_out,  mask=mask_m[:, None] & mask_k[None, :])
        tl.store(sh_gate_ptr, vals_gate, mask=mask_m[:, None] & mask_k[None, :])

        # accumulate statistics (in FP32)
        vals_f32 = vals_out.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    #  Second pass: normalise, apply the output gate, LayerNorm (γ,β)
    #  and fuse with the final linear projection.
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load from shared memory (fast)
        vals_out  = tl.load(sh_out  + offs_m[:, None] * H + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)
        vals_gate = tl.load(sh_gate + offs_m[:, None] * H + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        out_f32  = vals_out.to(tl.float32)
        gate_f32 = vals_gate.to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # FP32

        # LayerNorm over hidden dim + gate
        norm = ((out_f32 - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_f32                     # still FP32
        term_fp16 = term.to(tl.float16)           # back to FP16 for dot

        # weight tile (H x dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # FP16

        acc += tl.dot(term_fp16, w_tile)          # FP32 result
    # ------------------------------------------------------------------
    #  Store final tensor (B, N, N, dim) in FP32
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

    Forward pass of the outgoing TriMul operator (inference only).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all model weights (float32)
            - config_dict  : must contain keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape          # x: [B, N, N, dim]
    R = B * N * N                  # total (b,i,k) pairs
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N).contiguous()          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N).contiguous()        # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N).contiguous()                # [B, H, N, N] FP16
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ + 6️⃣ Fused second LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64   # rows per program (fits shared memory for H=128)
    BLOCK_K_OUT = 32   # hidden‑dim tile
    BLOCK_N_OUT = 128  # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_norm_proj_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        N,
        dim,
        eps,
        H=H,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # clean up
    del out, out_gate, w_out_T

    return final
```
