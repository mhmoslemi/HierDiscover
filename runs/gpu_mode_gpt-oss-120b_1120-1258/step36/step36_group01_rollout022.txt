import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # base offset for hidden dim in destination tensors
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]   # (BLOCK_R,1)
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 for stability
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))

        left = left_raw * mask_val * left_gate.to(tl.float16)
        right = right_raw * mask_val * right_gate.to(tl.float16)
        outg = out_gate.to(tl.float16)

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul kernel (AlphaFold‑3 style).

    1️⃣  Layer‑norm on the input (last dimension) using torch.nn.functional.layer_norm.
    2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into one GEMM.
    3️⃣  Triton kernel splits the fused projection, applies the optional mask,
        computes sigmoid gates and stores ``left``, ``right`` and ``out_gate`` tensors.
    4️⃣  Batched GEMM (`torch.bmm`) computes the core Einsum:
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
    5️⃣  A per‑position LayerNorm across the hidden dimension H is done with torch ops.
    6️⃣  The gated, normalised tensor is linearly projected back to the model dimension D
        using a final ``torch.nn.functional.linear``.

    All operations are inference‑only and use mixed‑precision (FP16) where possible.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (float32 -> float16)
    # ------------------------------------------------------------------
    # flatten [B,N,N,dim] -> [R, dim]
    x_flat = x.view(R, dim)
    # LayerNorm on last dimension
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)   # cast to fp16 for the following GEMM

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection (5·H outputs)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()   # shape (5·H, dim)

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask and sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.reshape(R).to(torch.float16).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy pointer

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # release memory we no longer need
    del proj, x_norm, mask_flat if 'mask_flat' in locals() else None

    # ------------------------------------------------------------------
    # 4️⃣  Core contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N), FP16
    out = out.view(B, H, N, N)   # (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣  Per‑position LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    # mean & variance over H (dim=1)
    mu = out.mean(dim=1, keepdim=True)               # (B,1,N,N), FP16 → upcast to FP32 in next step
    var = out.var(dim=1, unbiased=False, keepdim=True)  # (B,1,N,N)
    rstd = torch.rsqrt(var + eps)

    # normalise and apply learned affine parameters
    ln_w = w["to_out_norm.weight"]   # (H,) FP32
    ln_b = w["to_out_norm.bias"]     # (H,) FP32

    out_norm = (out - mu) * rstd                      # (B,H,N,N) FP16 (auto‑cast to FP32 then back)
    out_norm = out_norm * ln_w[None, :, None, None] + ln_b[None, :, None, None]

    # ------------------------------------------------------------------
    # 6️⃣  Gating and final projection back to model dimension
    # ------------------------------------------------------------------
    gated = out_norm * out_gate   # (B, H, N, N) FP16

    # reshape to (B, N, N, H) for the final linear
    gated_perm = gated.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    final_fp16 = F.linear(gated_perm, w["to_out.weight"].to(torch.float16))   # (B, N, N, dim), FP16
    final = final_fp16.to(torch.float32)

    return final

"""
Outgoing TriMul (AlphaFold‑3 style) – fast inference implementation.

Algorithm
---------
1️⃣  Input LayerNorm (last dim) – torch.nn.functional.layer_norm (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into one GEMM:  (R, dim) × (dim, 5·H) → (R, 5·H) (FP16).
3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional mask, evaluates sigmoid gates and writes the
    three tensors *left*, *right* and *out_gate* in layout [B, H, N, N] (FP16).
4️⃣  Batched GEMM (torch.bmm) evaluates the core einsum:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]      (FP16).
5️⃣  Per‑position LayerNorm over the hidden dimension H is performed with
    torch.mean/torch.var (FP32) and the learned affine parameters
    `to_out_norm.{weight,bias}`.
6️⃣  The normalized tensor is multiplied by `out_gate`, then linearly
    projected back to the model dimension D with a single `torch.nn.functional.linear`.
    The final result is cast to float32.

All heavy work is done in FP16/Tensor‑Core GEMMs; only the lightweight
normalisation steps use FP32 for numeric stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣  Split fused projection, mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r → (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcasted over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Strides for destination tensors (H‑plane is N·N)
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k                # start of hidden plane 0

    # Loop over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = offs_h < H

        # Linear view: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]     # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (inference only).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing model weights (float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape               # x: [B, N, N, dim]
    R = B * N * N                       # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim)                         # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                             # [R, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (produces 5·H hidden vectors)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()                # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                 # (R, 5·H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Split projection, apply mask & sigmoid gates (Triton)
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.view(R).to(torch.float16).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free memory we no longer need
    del proj, x_norm
    if HAS_MASK:
        del mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Contract via batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                   # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Per‑position LayerNorm over the hidden dimension (FP32)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)                      # cast once
    mu = out_fp32.mean(dim=1, keepdim=True)               # (B,1,N,N)
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)

    # affine parameters of the second LayerNorm
    ln_w = w["to_out_norm.weight"]   # (H,) FP32
    ln_b = w["to_out_norm.bias"]     # (H,) FP32

    # normalisation + learned scale/shift
    out_norm = (out_fp32 - mu) * rstd                     # (B,H,N,N) FP32
    out_norm = out_norm * ln_w[None, :, None, None] + ln_b[None, :, None, None]

    # --------------------------------------------------------------
    # 6️⃣ Gating and final projection back to model dimension
    # --------------------------------------------------------------
    # out_gate is still FP16 – promote to FP32 for the element‑wise product
    gated = out_norm * out_gate.to(torch.float32)          # (B,H,N,N) FP32

    # reshape to (..., H) so that torch.nn.functional.linear can be used
    gated_perm = gated.permute(0, 2, 3, 1).contiguous()    # (B,N,N,H)

    # final linear projection (H → dim). Weight is kept in FP32 for best accuracy.
    final_fp32 = F.linear(gated_perm, w["to_out.weight"]) # (B,N,N,dim) FP32

    return final_fp32
