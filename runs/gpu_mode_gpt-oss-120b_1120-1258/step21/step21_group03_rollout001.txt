"""TriMul (outgoing) – fast forward‐only implementation.

The kernel chain follows the “outgoing” TriMul from AlphaFold‑3 but only
covers the forward pass (no gradients).  The computation is split into
four Triton kernels:

1️⃣  Row‑wise LayerNorm over the flattened pair‑wise tensor (FP32 → FP16).
2️⃣  A fused linear projection that produces the five hidden tensors
    (left, right, left_gate, right_gate, out_gate).  The projection is
    performed with a single torch‑linear on FP16, then a Triton kernel
    splits the result, applies the optional pair‑wise mask and the three
    sigmoid gates.
3️⃣  Row‑wise statistics (mean + rsqrt‑variance) for the contracted tensor
    `out = left @ rightᵀ`.  The contraction itself is performed with
    cuBLAS (torch.bmm) because it is already near the theoretical peak
    on H100.
4️⃣  Final LayerNorm + out_gate multiplication + output projection are
    fused into a single Triton kernel that writes the result in FP32.

All kernels are written for Triton ≥ 3.3.1 and tuned for the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, C]  input  (fp32)
    y_ptr,          # [R, C]  output (fp16)
    w_ptr,          # [C]    gamma  (fp32)
    b_ptr,          # [C]    beta   (fp32)
    rows, cols,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BR]
    mask_r = offs_r < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(cols, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)    # [BC]
        mask_c = offs_c < cols

        ptr = x_ptr + offs_r[:, None] * cols + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)          # [BR, BC]

        sum_val += tl.sum(vals, axis=1)                     # [BR]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BR]

    mean = sum_val / cols
    var  = sum_sq / cols - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for c_off in range(0, tl.cdiv(cols, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < cols

        ptr = x_ptr + offs_r[:, None] * cols + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)          # [BR, BC]

        norm = (vals - mean[:, None]) * rstd[:, None]       # [BR, BC]

        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)  # fp32
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)  # fp32

        out = norm * gamma[None, :] + beta[None, :]         # fp32
        tl.store(y_ptr + offs_r[:, None] * cols + offs_c[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_c[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Split fused projection + optional mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5*H]  (fp16)
    left_ptr,        # [B, H, N, N] (fp16)
    right_ptr,       # [B, H, N, N] (fp16)
    out_gate_ptr,    # [B, H, N, N] (fp16)
    mask_ptr,        # [R] (fp16) – ignored if HAS_MASK==0
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BR]
    mask_r = offs_r < R

    # flat index → (b,i,k)
    nn = N * N
    b  = offs_r // nn
    rem = offs_r - b * nn
    i  = rem // N
    k  = rem - i * N

    # mask broadcast over the hidden dimension
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        m_val = m_val[:, None]                                      # [BR,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset in the hidden layout
    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + k               # [BR]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BH]
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_off = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BR, BH]

        left_raw       = tl.load(proj_ptr + proj_off,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_off + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_off + 2*H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_off + 3*H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_off + 4*H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32, back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [BR,BH]

        tl.store(left_ptr     + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N] (fp16)
    mu_ptr,           # [R] (fp32)
    rstd_ptr,         # [R] (fp32)
    total_rows, H, N,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BR]
    mask_r = offs_r < total_rows

    nn = N * N
    b  = offs_r // nn
    rem = offs_r - b * nn
    i  = rem // N
    j  = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # first hidden slice

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)         # [BK]
        mask_k = offs_k < H

        ptr = out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)      # [BR, BK]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mu,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)


# ----------------------------------------------------------------------
# 4️⃣  Fuse final LayerNorm, out_gate and output projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,           # [B, H, N, N]  (fp16)
    gate_ptr,          # [B, H, N, N]  (fp16)
    mu_ptr,            # [R]           (fp32)
    rstd_ptr,          # [R]           (fp32)
    ln_w_ptr,          # [H]           (fp32)
    ln_b_ptr,          # [H]           (fp32)
    w_out_T_ptr,       # [H, dim]      (fp16)  – transposed weight
    out_final_ptr,     # [B, N, N, dim] (fp32)
    total_rows, H, N, dim,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0)          # row tile
    pid_d = tl.program_id(1)          # output‑dim tile

    row_start = pid_r * BLOCK_R
    dim_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BR]
    offs_d = dim_start + tl.arange(0, BLOCK_D)          # [BD]

    mask_r = offs_r < total_rows
    mask_d = offs_d < dim

    nn = N * N
    b  = offs_r // nn
    rem = offs_r - b * nn
    i  = rem // N
    j  = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics (broadcast across hidden dim)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)    # [BR]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BR]

    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)                # [BK]
        mask_k = offs_k < H

        # load hidden activation and gate
        ptr_out  = out_ptr  + base_off[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [BR, BK]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)    # [BR, BK]

        # LayerNorm parameters for the hidden dim (γ,β) – FP32
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [BK]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [BK]

        # (out - μ) * rstd → γ· + β → * gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]           # [BR, BK]
        norm = diff * gamma[None, :] + beta[None, :]              # [BR, BK]
        term = norm * gate_vals                                   # [BR, BK]

        term_fp16 = term.to(tl.float16)                           # FP16 for dot

        # weight slice: (K, D)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_d[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_d[None, :],
                         other=0.0)                                 # [BK, BD]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)          # [BR, BD]

    # write final tensor [B, N, N, dim] (FP32)
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_r[:, None] & mask_d[None, :])

# 5️⃣  Entry point – orchestrates the entire forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor  [B, N, N] (bool/float) or None
        - weights_dict : dict(str → torch.Tensor)   – model parameters (float32)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_C = 64
    grid_ln = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden_dim, dim])
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5*H, dim]
    proj = F.linear(x_norm, fused_w)   # (R, 5*H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask and apply sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM (left @ rightᵀ) using cuBLAS – already near‑optimal
    # ------------------------------------------------------------------
    # View as [B*H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣  Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣  Final LayerNorm + out_gate + projection (all fused)
    # ------------------------------------------------------------------
    # Transpose the final projection weight once for coalesced reads

    BLOCK_K_OUT = 32
    BLOCK_D_OUT = 128
    grid_out = (triton.cdiv(R, BLOCK_R_OUT), triton.cdiv(dim, BLOCK_D_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # gamma (fp32)
        w["to_out_norm.bias"],     # beta  (fp32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_D=BLOCK_D_OUT,
        num_warps=4,
    )

    return final
