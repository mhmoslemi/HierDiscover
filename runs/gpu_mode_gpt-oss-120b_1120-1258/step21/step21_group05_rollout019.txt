Goal: Input x_flat: shape (R, dim) FP32. Output proj_fp16: shape (R, 5*H) FP16.

We need to compute per row:

- mean = sum_d x[d] / dim
- var = sum_d x[d]^2 / dim - mean^2
- rstd = rsqrt(var + eps)

Then for each hidden output h, compute linear projection: proj_h = sum_d ((x[d] - mean) * rstd) * w_fused[h, d]

Note: No bias in projection (weights linear have no bias). That's similar.

Thus algorithm:

for each block of rows (BLOCK_M) and each block of hidden dimension (BLOCK_H):
   # First pass: compute mean and var for each row in block.
   sum_val = 0; sum_sq=0 across dim.
   for d in range(0, dim, BLOCK_D):
        load x row segment as FP32
        sum_val += sum(x)
        sum_sq += sum(x*x)
   mean = sum_val / dim
   var = sum_sq/dim - mean^2
   rstd = rsqrt(var + eps)

   # second pass: compute projection for hidden dims in current block
   acc = zeros([BLOCK_M, BLOCK_H], dtype=fp32)
   for d in range(0, dim, BLOCK_D):
        load x segment as FP32
        x_centered = (x - mean[:, None]) * rstd[:, None]   # shape [BLOCK_M, BLOCK_D] in fp32
        load weight segment w_fused: shape [BLOCK_H, BLOCK_D] (FP16) and convert to fp32.

        # tl.dot expects shape (M, K) * (K, N) -> (M, N)
        # Thus we need to build weight tiles of shape [BLOCK_D, BLOCK_H] or [BLOCK_H, BLOCK_D].

        # So either we load weight transposed from memory: store weight in [5H, dim] but we need weight for each hidden output, each vector across dim.

Define kernel:

@triton.jit
def layernorm_linear_fused_kernel(
    x_ptr,                    # [R, dim] FP32
    out_ptr,                  # [R, hidden_total] FP16
    w_ptr,                    # [hidden_total, dim] FP16
    rows, dim, hidden_total,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [M]
    mask_m = offs_m < rows

    # Compute mean & var per row (FP32)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)  # shape [M]

    # For each hidden tile, accumulate projection
    # We'll loop over hidden dimension chunk indices outermost for readability
    # Also need to adjust block counts: num_hidden_tiles = (hidden_total + BLOCK_H - 1)//BLOCK_H
    num_hidden_tiles = tl.cdiv(hidden_total, BLOCK_H)
    for h_tile in range(0, num_hidden_tiles):
        h_start = h_tile * BLOCK_H
        # compute per row accumulator for this tile
        acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load X segment
            x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            x_vals = tl.load(x_ptrs,
                             mask=mask_m[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)  # [M, D]

            # normalize: (x - mean) * rstd
            x_centered = (x_vals - mean[:, None]) * rstd[:, None]

            # load weight segment for this hidden tile: shape [BLOCK_H, BLOCK_D]
            # w_ptr layout: hidden_total rows (Hdim) and dim columns.
            w_ptrs = w_ptr + (h_start + tl.arange(0, BLOCK_H))[:, None] * dim + offs_d[None, :]
            w_vals = tl.load(w_ptrs,
                             mask=tl.arange(0, BLOCK_H)[:, None] < (hidden_total - h_start) & mask_d[None, :],
                             other=0.0).to(tl.float32)  # [H, D]

            # perform dot: out = x_centered (M,D) dot w_vals.T (D,H) -> (M,H)
            # tl.dot expects dimensions (M, K), (K, N)
            acc += tl.dot(x_centered, w_vals)  # K = D; N = H

        # write out: cast to fp16
        # destination offset: row*hidden_total + h_start + hidden_index
        dst_ptrs = out_ptr + offs_m[:, None] * hidden_total + (h_start + tl.arange(0, BLOCK_H))[None, :]
        tl.store(dst_ptrs,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & (tl.arange(0, BLOCK_H)[None, :] < (hidden_total - h_start))
        )

TriMul (outgoing) – fast forward pass (no backward).

Algorithm (high‑level)
----------------------
1️⃣ Row‑wise LayerNorm on the flattened pairwise tensor (R = B·N·N) in FP32,
   output stored in FP16.  (custom kernel)

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single half‑precision GEMM:
        proj = x_norm @ W_fusedᵀ                → (R, 5·H)
   (standard torch linear – cuBLAS tensor‑core kernel).

3️⃣ Decode ``proj`` into three tensors (left, right, out_gate).  Apply the
   optional binary mask and the three sigmoid gates inside a kernel.

4️⃣ Batched GEMM over the hidden dimension (cuBLAS):
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   (torch.bmm on the view (B·H, N, N)).

   row‑wise reduction kernel (FP32).

   projection (to_out) into a single kernel:
        out_norm = ((out – μ) * rstd) * γ + β

All kernels are written for H100 (TPU‑core‑friendly block sizes, mixed‑precision,

# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,           # [R, dim]  input  FP32
    y_ptr,           # [R, dim]  output FP16
    w_ptr,           # [dim]    gamma   FP32
    b_ptr,           # [dim]    beta    FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [M, D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  fp16
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R]      fp16  (ignored if HAS_MASK==0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional binary mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + k           # begin of H‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout: [left | right | left_gate | right_gate | out_gate] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw .to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor (second LN)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # [B, H, N, N]  fp16
    mu_ptr,       # [R]           fp32
    rstd_ptr,     # [R]           fp32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j           # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   fp16
    gate_ptr,         # [B, H, N, N]   fp16
    mu_ptr,           # [R]            fp32
    rstd_ptr,         # [R]            fp32
    ln_w_ptr,         # [H]            fp32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            fp32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       fp16   (transposed to_out.weight)
    out_final_ptr,    # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)           # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)           # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j           # start of hidden‑plane 0

    # Load per‑row statistics (μ, rstd)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # Iterate over hidden dimension in chunks
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load hidden tensor and gate (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters (γ,β) – FP32
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out - μ) * rstd) → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * gate_vals                                        # [M, K]

        term_fp16 = term.to(tl.float16)

        # Load slice of final projection matrix Wᵀ : (K, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, TILE_N]

        # dot product (FP16·FP16 → FP32) and accumulate
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Write final result [B, N, N, dim]  (float32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    # --------------------------------------------------------------

    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)

    device = x.device

    R = B * N * N                # total pairwise positions
    H = hidden_dim

    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------

    BLOCK_D = 64

        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,

        BLOCK_D=BLOCK_D,

    )

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim] (FP16)

    proj = F.linear(x_norm, fused_w)   # FP16

    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]  FP16

    # out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # free left/right early
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (second LayerNorm)
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # gamma (FP32)
        w["to_out_norm.bias"],     # beta  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
```
