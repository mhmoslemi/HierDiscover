    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat   # free

    fused_w = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16).contiguous()  # shape (5*H, dim)

    del fused_w, x_norm

    proj = proj.view(R, 5, H)

    left_gate = proj[:,2,:]
    right_gate = proj[:,3,:]
    out_gate = proj[:,4,:]

    right = right.view(B, N, N, H).permute(0,3,1,2).contiguous()
    left_gate = left_gate.view(B, N, N, H).permute(0,3,1,2).contiguous()
    right_gate = right_gate.view(B, N, N, H).permute(0,3,1,2).contiguous()

"""
Outgoing TriMul (AlphaFold3) – fast forward pass (no backward).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) implemented as a small Triton kernel.
2️⃣ Fuse the five linear transforms (left, right, left‑gate, right‑gate, out‑gate)
   into one half‑precision GEMM:   proj = x_norm @ W_fusedᵀ   → (R, 5·H).
3️⃣ Split `proj` into the five tensors, reshape to [B, H, N, N],
   apply the optional pairwise mask, compute sigmoid gates and obtain
   `left`, `right`, `out_gate` (all FP16).
4️⃣ Batched half‑precision GEMM (cuBLAS):   out = left @ rightᵀ
   → shape [B, H, N, N]  (contraction over the “k” dimension).
5️⃣ Row‑wise mean & rsqrt‑variance of `out` (second LayerNorm) using
   native torch operations.
6️⃣ Apply the second LayerNorm, multiply by `out_gate` and finish with the
   output projection (another cuBLAS GEMM).  The final result is
   float‑32 tensor [B, N, N, dim].

Only the first LayerNorm is written in Triton; the remaining work uses
high‑performance PyTorch primitives which already call cuBLAS/​CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  input FP32
    y_ptr,          # [R, dim]  output FP16
    w_ptr,          # [dim]    gamma (FP32)
    b_ptr,          # [dim]    beta  (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [M, D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping of weight names to torch.Tensor (float32)
                required keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) via Triton
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates
    # --------------------------------------------------------------
    # shape (R, 5, H)
    proj = proj.view(R, 5, H)

    left      = proj[:, 0, :]   # (R, H)
    right     = proj[:, 1, :]
    left_gate = proj[:, 2, :]
    right_gate= proj[:, 3, :]
    out_gate  = proj[:, 4, :]

    del proj

    # reshape to [B, H, N, N]
    left      = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right     = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right_gate= right_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate  = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # optional per‑pair mask (broadcast over H)
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).view(B, 1, N, N)  # [B,1,N,N]
        left  = left  * mask_fp16
        right = right * mask_fp16

    # sigmoid gates (still FP16)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # apply gates
    left  = left  * left_gate
    right = right * right_gate

    # free the gate tensors we no longer need
    del left_gate, right_gate

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)              # (B·H, N, N)
    right_mat = right.view(B * H, N, N)             # (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                     # (B, H, N, N) FP16

    del left, right, left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm statistics (mean & rsqrt‑variance) – torch
    # --------------------------------------------------------------
    out_fp32 = out.float()
    mu   = out_fp32.mean(dim=1)                   # (B, N, N)   population mean
    var  = out_fp32.var(dim=1, unbiased=False)    # (B, N, N)   population variance
    rstd = 1.0 / torch.sqrt(var + eps)            # (B, N, N)

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate multiplication and final projection
    # --------------------------------------------------------------
    # Normalize (out - mu) * rstd   – keep in FP32 for stability
    mu   = mu.unsqueeze(1)       # (B,1,N,N)
    rstd = rstd.unsqueeze(1)     # (B,1,N,N)
    norm = (out_fp32 - mu) * rstd   # (B,H,N,N) FP32

    # Apply learned scale & bias of the second LayerNorm (γ,β)
    ln_w = w["to_out_norm.weight"]   # (H,)
    ln_b = w["to_out_norm.bias"]     # (H,)
    norm = norm * ln_w[None, :, None, None] + ln_b[None, :, None, None]   # FP32

    # Multiply by output gate (cast gate to FP32)
    norm = norm * out_gate.float()   # (B,H,N,N) FP32

    # Final linear projection: (R, H) @ (dim, H)ᵀ → (R, dim)
    hidden_flat = norm.view(R, H)                # (R, H) FP32
    weight_out = w["to_out.weight"].to(torch.float16).contiguous()   # (dim, H) FP16

    # Linear in FP16 (input casted) → output FP16, then back to FP32
    out_fp16 = F.linear(hidden_flat.to(torch.float16), weight_out)   # (R, dim) FP16
    final = out_fp16.to(torch.float32).view(B, N, N, dim)           # (B,N,N,dim) FP32

    return final
