"""
Outgoing TriMul (AlphaFold3) – forward pass implemented with Triton.

The implementation follows the algorithm described in the prompt:
1️⃣ Row‑wise LayerNorm on the flattened pairwise tensor (FP32 → FP16).
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single half‑precision GEMM.
3️⃣ Split the fused projection, apply an optional pair‑wise mask, compute sigmoid
   gates and store the tensors `left`, `right`, `out_gate` (shape [B, H, N, N],
   dtype FP16) with a Triton kernel.
4️⃣ Batched half‑precision GEMM (cuBLAS) computes the contraction:
   out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k].
5️⃣ Row‑wise mean & rsqrt‑variance of `out` via a Triton kernel.
6️⃣ Fuse the second LayerNorm, the output gate, and the final linear projection
   into a single Triton kernel, producing the final [B, N, N, dim] tensor
   (float‑32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  (FP32)
    y_ptr,          # [R, dim]  (FP16)
    w_ptr,          # [dim]    gamma (FP32)
    b_ptr,          # [dim]    beta  (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: compute mean and variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)               # [M, D]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        y = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            y.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,         # [R, 5·H]  (FP16)
    left_ptr,         # [B, H, N, N] (FP16)
    right_ptr,        # [B, H, N, N] (FP16)
    out_gate_ptr,     # [B, H, N, N] (FP16)
    mask_ptr,         # [R]   (FP16) – used only if HAS_MASK==1
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mval = mval[:, None]                                       # [BLOCK_R, 1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + k                # start of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (compute in FP32 for stability)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N]  (FP16)
    mu_ptr,           # [R]           (FP32)
    rstd_ptr,         # [R]           (FP32)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, output gate & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   (FP16)
    gate_ptr,         # [B, H, N, N]   (FP16)
    mu_ptr,           # [R]            (FP32)
    rstd_ptr,         # [R]            (FP32)
    ln_w_ptr,         # [H]            (FP32)  – gamma of to_out_norm
    ln_b_ptr,         # [H]            (FP32)  – beta  of to_out_norm
    w_out_T_ptr,      # [H, dim]       (FP16)  – transposed final weight
    out_final_ptr,    # [B, N, N, dim] (FP32)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden‑plane 0

    # load per‑row mean/std
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden tensor and gate (FP16 → FP32)
        out_vals = tl.load(
            out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        gate_vals = tl.load(
            gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        # LayerNorm parameters (γ,β) for hidden dim
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd -> γ· + β  (second LN)
        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by output gate
        gated = norm * gate_vals

        # cast to FP16 for the dot product
        gated_fp16 = gated.to(tl.float16)

        # load a tile of the final weight matrix (H, dim)
        w_tile = tl.load(
            w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # [K, TILE_N]

        # accumulate (FP16·FP16 → FP32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # write result: [B, N, N, dim] (float32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Arguments:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, no bias for the first five linear layers)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim) – no bias
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5·H)   FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Prepare mask pointer (if required)
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # gamma (FP32)
        w["to_out_norm.bias"],     # beta  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )

    return final
