Now define fused_output_kernel:

```
@triton.jit
def fused_output_kernel(
    out_ptr,       # [B, H, N, N] fp16
    gate_ptr,      # [B, H, N, N] fp16
    ln_w_ptr,      # [H] fp32
    ln_b_ptr,      # [H] fp32
    w_out_T_ptr,   # [H, dim] fp16
    out_final_ptr, # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # Decode b,i,j from offs_m:
    nn = N * N
    b = (offs_m // nn)
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # start of hidden plane 0

    # ------------------- First pass: compute row-wise sum and sumsq -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_offset in range(0, H, BLOCK_K):
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr_out,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ------------------- Second pass: compute final output -------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_offset in range(0, H, BLOCK_K):
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out and gate
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        out_vals = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M, K]
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # Load layernorm gamma and beta
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)

        # Normalize
        diff = (out_vals - mu[:, None]) * rstd[:, None]       # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]          # [M, K]

        term = norm * gate_vals                              # [M, K]
        term_fp16 = term.to(tl.float16)

        # Load weight tile
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                         # [K, TILE_N]

        # Accumulate dot
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold3) – fast forward only.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) using torch.nn.functional.layer_norm.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single half‑precision GEMM (`proj = x_norm @ W_fusedᵀ`).
3️⃣  Reshape/split the projection, apply the optional pair‑wise mask,
    compute sigmoid gates and obtain `left`, `right` and `out_gate`
    tensors in layout [B, H, N, N] (FP16).
4️⃣  Batched half‑precision GEMM (cuBLAS)
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]   (FP16)
5️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the final
    linear projection into ONE Triton kernel:
        * First pass – compute per‑position mean µ and variance σ² of `out`
          across the hidden dimension H.
        * Second pass – for each hidden slice compute the normalized term,
          apply the gate, and immediately accumulate the dot‑product with the
          transposed output weight `W_outᵀ`.  The three required accumulators
          implement:
                Σₕ outₕ·γₕ·gateₕ·Wₕ   (term 1)
                Σₕ γₕ·gateₕ·Wₕ        (term 2)
                Σₕ βₕ·gateₕ·Wₕ        (term 3)
        * Final result:
                out = rstd·term1 – µ·rstd·term2 + term3   (float‑32)

Only the second‑layer‑norm + projection is written in Triton – all other
operations rely on highly‑optimized PyTorch kernels.  This reduces memory
traffic (no intermediate µ/rstd tensors) and eliminates a separate
split‑gate kernel, yielding a faster forward pass on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton fused kernel: second LayerNorm + out_gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_output_kernel(
    out_ptr,          # [B, H, N, N]   fp16   (result of left @ rightᵀ)
    gate_ptr,         # [B, H, N, N]   fp16   (out_gate after sigmoid)
    ln_w_ptr,         # [H]           fp32   (to_out_norm.weight)
    ln_b_ptr,         # [H]           fp32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]      fp16   (transposed to_out.weight)
    out_final_ptr,    # [B, N, N, dim] fp32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)            # rows  (flattened B·N·N)
    pid_n = tl.program_id(1)            # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # ---- decode (b,i,j) from the flattened row index ----
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N                     # stride between hidden slices
    base_offset = b * (H * N * N) + i * N + j # start of hidden‑plane 0 for (b,i,j)

    # ------------------------------------------------------------------
    # 1️⃣ First pass: compute mean µ and variance σ² across the hidden dim.
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [M, K]

        sum_val += tl.sum(vals,  axis=1)               # Σₕ outₕ
        sum_sq  += tl.sum(vals * vals, axis=1)        # Σₕ outₕ²

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                        # 1/√(σ²+ε)

    # ------------------------------------------------------------------
    # 2️⃣ Second pass: LayerNorm, out_gate, and final projection.
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # final output tile

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out (pre‑norm) and gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # LayerNorm parameters γ (weight) and β (bias)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # Normalisation: ((out‑µ)*rstd)·γ + β
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]

        # Apply out‑gate
        term = norm * gate_vals                                    # [M, K]
        term_fp16 = term.to(tl.float16)                           # cast for dot

        # Load a tile of the final weight matrix W_outᵀ (shape [H, dim])
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, TILE_N]

        # Accumulate dot‑product (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # Write result  – layout [B, N, N, dim] (contiguous)
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint – outgoing TriMul forward pass (no backward)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no backward).
    All heavy linear work stays in FP16; the only Triton kernel fuses the
    second LayerNorm, the ``out_gate`` multiplication and the final projection.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]                 # output dimension (C)
    hidden_dim = cfg["hidden_dim"]   # hidden dimension (H)
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – use the highly‑optimised
    #    torch implementation.
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim] → FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into one GEMM.
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # (5·H, dim)

    # (R, dim) @ (dim, 5·H)ᵀ  → (R, 5·H)  (FP16)
    proj = F.linear(x_norm.view(R, dim), fused_w)

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates.
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)                # (R, 5, H)

    left_flat  = proj[:, 0, :]               # (R, H)
    right_flat = proj[:, 1, :]               # (R, H)
    left_gate  = proj[:, 2, :].sigmoid()     # (R, H)
    right_gate = proj[:, 3, :].sigmoid()     # (R, H)
    out_gate   = proj[:, 4, :].sigmoid()     # (R, H)

    if (not nomask) and (mask is not None):
        # mask is broadcast over the hidden dimension
        mask_flat = mask.view(R).to(torch.float16).unsqueeze(-1)  # (R,1)
        left_flat  = left_flat  * mask_flat
        right_flat = right_flat * mask_flat

    # Apply the pair‑wise gates
    left_flat  = left_flat  * left_gate
    right_flat = right_flat * right_gate

    # Reshape to [B, H, N, N] (contiguous layout)
    left = left_flat.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right_flat.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: out = left @ rightᵀ   (FP16)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                # [B·H, N, N]
    right_mat = right.view(B * H, N, N)              # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N).contiguous()          # [B, H, N, N]  (FP16)

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate and final projection.
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Tuned tile sizes – chosen to keep register pressure modest while
    # exposing enough parallelism on H100.
    BLOCK_M_OUT = 64      # rows per program (tiles of B·N·N)
    BLOCK_K_OUT = 64      # hidden‑dim chunk
    BLOCK_N_OUT = 128     # output‑dim chunk

    grid = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))

    fused_output_kernel[grid](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,               # H100 handles 8‑warp kernels efficiently
    )

    return final
