@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,
    left_ptr,
    right_ptr,
    out_gate_ptr,
    mask_ptr,
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    seqsq = N * N
    # decode (b,i,j)
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = m[:, None]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Loop hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        # offsets for hidden within the projection
        off_hidden = h_off * BLOCK_H
        # mask for valid hidden entries
        mask_h = tl.arange(0, BLOCK_H) < (H - off_hidden)

        # load contiguous block of size 5*BLOCK_H for each row
        block_size = 5 * BLOCK_H
        seg_idx = tl.arange(0, block_size)
        # compute pointers: each row offset + off_hidden + seg offset
        base = offs_r[:, None] * (5 * H) + off_hidden + seg_idx[None, :]
        # mask for data: within H bound (for each segment)
        valid_mask = mask_r[:, None] & (seg_idx[None, :] < 5 * (H - off_hidden))

        proj_block = tl.load(proj_ptr + base,
                             mask=valid_mask,
                             other=0.0)  # fp16

        # split
        left_raw      = proj_block[:, 0:BLOCK_H]
        right_raw     = proj_block[:, BLOCK_H:2*BLOCK_H]
        left_gate_raw = proj_block[:, 2*BLOCK_H:3*BLOCK_H]
        right_gate_raw= proj_block[:, 3*BLOCK_H:4*BLOCK_H]
        out_gate_raw  = proj_block[:, 4*BLOCK_H:5*BLOCK_H]

        # compute sigmoid in fp32
        l_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))
        r_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))
        o_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))
        # cast to fp16
        l_gate = l_gate.to(tl.float16)
        r_gate = r_gate.to(tl.float16)
        o_gate = o_gate.to(tl.float16)

        # apply mask and gates
        left = left_raw * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # write left/right
        bh = b_idx[:, None] * H + (off_hidden + tl.arange(0, BLOCK_H))[None, :]
        # However bh needs hidden index idx; after loop, we need to write each slice to corresponding hidden slice.
        # Alternative: compute bh = b_idx * H + offs_h where offs_h = off_hidden + tl.arange(0, BLOCK_H)
        offs_h = off_hidden + tl.arange(0, BLOCK_H)
        bh = b_idx[:, None] * H + offs_h[None, :]   # shape [R, BLOCK_H]

        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate (R, H)
        # we need to store out_gate for these hidden indices
        gate_off = offs_r[:, None] * H + (off_hidden + tl.arange(0, BLOCK_H))[None, :]
        tl.store(out_gate_ptr + gate_off,
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

   * optionally applies the pair‑wise mask,
   * evaluates sigmoid gates (fp32 → fp16),
   * writes **left** and **right** directly in the layout required for the
     batched GEMM: (B·H, N, N),
   * stores the **out_gate** tensor (R, H) for later use.
   The five required slices are loaded with a *single* memory transaction
   per hidden‑block (instead of five separate loads).
4️⃣ Batched half‑precision GEMM (Tensor‑Cores):
        out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
5️⃣ Triton kernel `layernorm_mul_strided_kernel`:
   * computes a second row‑wise LayerNorm over the hidden dimension of the
     GEMM result (the dimension is strided by N²),
   * multiplies the normalized values by the previously computed `out_gate`,
   * writes the fused result (R, H) in fp16.
6️⃣ Final linear projection back to the original dimension C.

All heavy linear algebra (steps 2, 4, 6) uses cuBLAS/Tensor‑Cores; only the
two LayerNorms and the split‑/gate‑fusion are written in Triton to eliminate
extra memory traffic.  Block sizes have been tuned for the H100 (max. 1024
threads per block) to minimise kernel launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                 # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values (fp16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]           # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split / fuse projection, mask and sigmoid gates (single load)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B·H, N, N] (float16)
    right_ptr,         # mutable *[B·H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]      (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # decode (b,i,j) from flat row index
    seqsq = N * N
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # optional pair‑wise mask (broadcasted over hidden)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        off_hidden = h_off * BLOCK_H                     # start index inside H
        rem_h = H - off_hidden                           # remaining valid hidden entries
        # valid mask for this block (fp16)
        mask_h = tl.arange(0, BLOCK_H) < rem_h

        # ---- load the 5‑slice packed projection in ONE transaction ----
        block_sz = 5 * BLOCK_H
        seg_idx = tl.arange(0, block_sz)                 # [5*BLOCK_H]
        # mask for out‑of‑range columns of the concatenated matrix
        col_mask = seg_idx < 5 * rem_h
        base = offs_r[:, None] * (5 * H) + off_hidden + seg_idx[None, :]   # [R,5*BLOCK_H]

        proj_block = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & col_mask[None, :],
                             other=0.0)                                   # fp16

        # split the packed block
        left_raw      = proj_block[:, 0:BLOCK_H]
        right_raw     = proj_block[:, BLOCK_H:2*BLOCK_H]
        left_gate_raw = proj_block[:, 2*BLOCK_H:3*BLOCK_H]
        right_gate_raw= proj_block[:, 3*BLOCK_H:4*BLOCK_H]
        out_gate_raw  = proj_block[:, 4*BLOCK_H:5*BLOCK_H]

        # ---- sigmoid (computed in fp32) ----
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # hidden indices for this block
        offs_h = off_hidden + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        # write left / right in layout (B·H, N, N)
        bh = b_idx[:, None] * H + offs_h[None, :]       # [R,BLOCK_H]
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]   # [R,BLOCK_H]

        tl.store(left_ptr  + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate (still (rows, H))
        gate_off = offs_r[:, None] * H + offs_h[None, :]   # [R,BLOCK_H]
        tl.store(out_gate_ptr + gate_off,
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_strided_kernel(
    x_ptr,          # const   *[B·H, N, N]   (float16)   (GEMM result)
    gate_ptr,       # const   *[rows, H]    (float16)   (out_gate)
    y_ptr,          # mutable *[rows, H]    (float16)   (fused output)
    weight_ptr,     # const   *[H]         (float32)   γ
    bias_ptr,       # const   *[H]         (float32)   β
    rows: tl.int32,       # = B·N·N
    N: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_r = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_r = offs_r < rows

    seqsq = N * N

    # decode (b,i,j) from flat row index
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # base offset of the first hidden slice (h = 0) inside the strided tensor
    base0 = b_idx * H * seqsq + i_idx * N + j_idx      # [BLOCK_M]

    # ---------- first pass: mean / variance over H (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalize, apply γ/β, multiply by gate ----------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(x_ptr + base0[:, None] + offs_h[None, :] * seqsq,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # [M, Hblk]

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β (fp32)

        norm = norm * w[None, :] + b[None, :]               # fp32

        gate = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)            # fp32

        out = norm * gate                                   # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim`` and ``hidden_dim`` and a bool
                         ``nomask`` (True → no mask).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]            # C
    hidden_dim = cfg["hidden_dim"]    # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise positions
    H = hidden_dim
    seqsq = N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, C] (fp32)
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512      # rows per program (tuned for H100)
    BLOCK_D = 64       # dim chunk size
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (float32, [C])
        weights["norm.bias"],        # β (float32, [C])
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and three gates
    # ------------------------------------------------------------------
    # concatenate the five weight matrices: (5·H, C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, C]

    proj = F.linear(x_norm, concat_w)       # (R, 5·H)  fp16
    del concat_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split + mask + sigmoid gates, write left/right in (B·H, N, N)
    # ------------------------------------------------------------------
    left      = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16).reshape(R)   # [R] 0/1
        has_mask = 1

    BLOCK_R = 256
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (Tensor‑Core)
    # ------------------------------------------------------------------
    # left/right are (B·H, N, N) fp16 → use torch.bmm
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_fused = torch.empty(R, H, dtype=torch.float16, device=device)

    grid_lnorm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_strided_kernel[grid_lnorm2](
        out_mat,
        out_gate,
        out_fused,
        weights["to_out_norm.weight"],   # γ (float32, [H])
        weights["to_out_norm.bias"],     # β (float32, [H])
        R,
        N,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_D,                 # reuse same block size for hidden dim
        num_warps=8,
    )
    del out_mat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
