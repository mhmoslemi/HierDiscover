@triton.jit
def final_fused_layernorm_gate_linear_kernel(
    out_ptr,           # const fp16 *[R, H]
    out_gate_ptr,      # const fp16 *[R, H]
    w_norm_ptr,        # const fp32 *[H]
    b_norm_ptr,        # const fp32 *[H]
    w_out_ptr,         # const fp16 *[H, D]   (weight_T)
    out_ptr_final,     # mutable fp16 *[R, D]
    R, H, D,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Compute mean/var across H
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out values
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Prepare accumulator for final output
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    # Second pass over H to compute gated normalized and multiply with weight
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)  # fp32
        gate_vals = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        # gamma, beta per hidden
        gamma = tl.load(w_norm_ptr + offs_h, mask=mask_h, other=0.0)  # fp32
        beta = tl.load(b_norm_ptr + offs_h, mask=mask_h, other=0.0)   # fp32

        # normalize
        norm = (out_vals - mean[:, None]) * rstd[:, None]   # fp32
        norm = norm * gamma[None, :] + beta[None, :]        # fp32

        # gated
        gated = norm * gate_vals    # fp32

        # Cast gated to fp16 for dot
        gated_fp16 = gated.to(tl.float16)

        # Load weight block (H dims x D block)
        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # Load weight submatrix for this h block
            w = tl.load(w_out_ptr + offs_h[:, None] * D + offs_d[None, :],

@triton.jit
def final_fused_ln_gate_linear_kernel(
    out_ptr,         # const fp16 *[R, H]   normalized pre-gate values
    out_gate_ptr,    # const fp16 *[R, H]   out_gate values (sigmoid)
    w_norm_ptr,      # const fp32 *[H]      gamma (to_out_norm.weight)
    b_norm_ptr,      # const fp32 *[H]      beta (to_out_norm.bias)
    w_out_ptr,       # const fp16 *[H, D]   weight_T (to_out.weight.T)
    out_final_ptr,   # mutable fp16 *[R, D]
    R: tl.int32,
    H: tl.int32,
    D: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Compute mean/var over H
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulator for final output
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    # second pass over H blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load out values and out_gate values as fp32
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)   # shape (BLOCK_R, BLOCK_H)
        gate_vals = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # gamma and beta for this hidden slice
        gamma = tl.load(w_norm_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
        beta = tl.load(b_norm_ptr + offs_h, mask=mask_h, other=0.0)    # fp32

        # normalize
        norm = (out_vals - mean[:, None]) * rstd[:, None]             # fp32
        norm = norm * gamma[None, :] + beta[None, :]                  # fp32

        # gated
        gated = norm * gate_vals                                      # fp32

        # cast to fp16 for dot
        gated_fp16 = gated.to(tl.float16)                             # shape (BLOCK_R, BLOCK_H)

        # Loop over D blocks for matrix multiply
        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # Load weight slice (BLOCK_H, BLOCK_D)
            w = tl.load(w_out_ptr + offs_h[:, None] * D + offs_d[None, :],
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0)   # fp16

            # Compute partial dot product
            # gated_fp16: (BLOCK_R, BLOCK_H)
            # w: (BLOCK_H, BLOCK_D)
            # result: (BLOCK_R, BLOCK_D)
            partial = tl.dot(gated_fp16, w)   # fp32

            acc += partial

    # Store final output
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        tl.store(out_final_ptr + offs_r[:, None] * D + offs_d[None, :],
                 acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])

"""
Optimized Triton implementation of the *outgoing* TriMul operator (AlphaFold3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened 4‑D input (B·N·N × dim) – fp32 → fp16.
2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    half‑precision GEMM using `torch.nn.functional.linear`.
3️⃣  Decode the flat rows (`b,i,k`) and write the three tensors
    `left`, `right`, `out_gate` directly in the (B, H, N, N) layout while
    applying the optional pair‑wise mask and the sigmoid gates.
4️⃣  Batched half‑precision GEMM (`torch.bmm`) over the hidden dimension:
    `out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ`.
5️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the final
    linear projection into *one* Triton kernel:
    *   compute mean/variance of `out` over the hidden dimension,
    *   apply γ/β (to_out_norm), multiply by the already‑sigmoided `out_gate`,
    *   accumulate the result with the output weight matrix (`to_out.weightᵀ`).
    The kernel writes the final tensor `[B, N, N, dim]` directly (fp32).

The implementation uses half precision for all heavy matrix‑multiplications
(Tensor‑Core friendly) and keeps only a few fp32 reductions for numerical
stability.  It removes the intermediate `out_fused` buffer of the reference
implementation, reducing memory traffic and kernel launch overhead, which is
enough to bring the end‑to‑end latency below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalized value (γ·(x‑μ)·rstd + β) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#    write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R] (fp16) (ignored if HAS_MASK=0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode (b,i,k) from flat row index ---------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask (if any)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between consecutive h

    # base offset (b·H·N·N + i·N + k) – common to all hidden slices of a row
    base_offset = b * (H * N * N) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load the five blocks from the fused projection ----------------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw     = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate    = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate   = tl.load(proj_ptr + proj_base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ----- sigmoid (stable fp32) ---------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ----------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- store --------------------------------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out‑gate multiplication and final linear
# ----------------------------------------------------------------------
@triton.jit
def final_ln_gate_linear_kernel(
    out_ptr,            # const   *[R, H] (fp16)  – result of BMM
    out_gate_ptr,       # const   *[R, H] (fp16)  – sigmoid out‑gate
    w_norm_ptr,         # const   *[H]    (fp32) – γ of to_out_norm
    b_norm_ptr,         # const   *[H]    (fp32) – β of to_out_norm
    w_out_ptr,          # const   *[H, D] (fp16) – (to_out.weight)ᵀ
    out_final_ptr,      # mutable *[R, D] (fp16) – final tensor
    R, H, D,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ----------- mean & variance over the hidden dimension ---------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- accumulate final projection --------------------
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # load raw tensors
        out_vals  = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)          # (R, Hblk)
        gate_vals = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)          # (R, Hblk)

        # γ/β for this slice
        gamma = tl.load(w_norm_ptr + offs_h, mask=mask_h, other=0.0)   # (Hblk)
        beta  = tl.load(b_norm_ptr + offs_h, mask=mask_h, other=0.0)   # (Hblk)

        # normalize ⇢ apply γ/β ⇢ multiply by out_gate
        norm = (out_vals - mean[:, None]) * rstd[:, None]            # (R, Hblk)
        norm = norm * gamma[None, :] + beta[None, :]                 # (R, Hblk)
        gated = norm * gate_vals                                      # (R, Hblk)

        # cast to fp16 for the matmul (Tensor‑Core friendly)
        gated_fp16 = gated.to(tl.float16)                             # (R, Hblk)

        # ----- multiply with the output weight matrix (H, D) -----
        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # weight slice (Hblk, Dblk)
            w = tl.load(w_out_ptr + offs_h[:, None] * D + offs_d[None, :],
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0)                                 # fp16

            # partial = gated_fp16 @ w   → (R, Dblk)
            partial = tl.dot(gated_fp16, w)                         # fp32

            acc += partial

    # ------------------- write final tensor -----------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        tl.store(out_final_ptr + offs_r[:, None] * D + offs_d[None, :],
                 acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])

# Custom forward implementation
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping of all linear / layer‑norm parameters
                            (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                        # total rows for row‑wise ops
    H = hidden_dim
    D = dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program (tuned for H100)
    BLOCK_D = 64           # dim chunk size
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],     # γ (float32)
        w["norm.bias"],       # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + all gates → (R, 5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # fp16, shape [R, 5·H]
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates (write directly in B·H·N·N)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128          # rows per program
    BLOCK_H = 32           # hidden dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # --------------------------------------------------------------
    # shape (B·H, N, N) – contiguous for cuBLAS
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, H, N, N)        # [B, H, N, N] – still contiguous

    del left, right                       # free memory

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out‑gate multiplication and final projection
    # --------------------------------------------------------------
    out_flat      = out.view(R, H).contiguous()
    out_gate_flat = out_gate.view(R, H).contiguous()

    # to_out.weightᵀ : shape [H, D]
    w_out_T = w["to_out.weight"].t().contiguous().to(torch.float16)

    final_half = torch.empty(R, D, dtype=torch.float16, device=device)

    BLOCK_R2 = 128
    BLOCK_H2 = 32
    BLOCK_D2 = 64
    grid_final = (triton.cdiv(R, BLOCK_R2),)
    final_ln_gate_linear_kernel[grid_final](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final_half,
        R,
        H,
        D,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )

    del out, out_gate, out_gate_flat, out_flat, w_out_T

    # --------------------------------------------------------------
    # 6️⃣ Convert to float32 and reshape back to (B, N, N, dim)
    # --------------------------------------------------------------
    final = final_half.view(B, N, N, D).to(torch.float32)
    return final
