        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj

    # 4. Batched matmul (left @ right^T) using torch.bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # shape (B*H, N, N)
    out = out.view(B, H, N, N)  # shape (B, H, N, N)
    del left, right

    # 5. Second LayerNorm across hidden dim using torch.nn.functional.layer_norm
    # Reorder to (B,N,N,H)
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    out_norm = torch.nn.functional.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )
    # back to (B, H, N, N)
    out_norm = out_norm.permute(0, 3, 1, 2).contiguous()
    # 6. Apply out_gate
    out_gated = out_norm * out_gate
    del out_norm, out_gate

    # 7. Final linear projection
    # flatten to (R, H)
    out_gated_flat = out_gated.permute(0, 2, 3, 1).contiguous().view(R, H)
    final_half = torch.nn.functional.linear(
        out_gated_flat,
        w["to_out.weight"].to(torch.float16),
    )
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) using PyTorch's
   highly‑optimized CUDA kernel (float32 → float16).

2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single half‑precision GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)   (R = B·N·N).

3️⃣ Split the fused projection, optionally apply a pair‑wise mask, evaluate the
   sigmoid gates and store the three tensors *directly* in the (B, H, N, N)
   layout required by the batched GEMM (custom Triton kernel).

4️⃣ Batched matrix multiplication (torch.bmm → cuBLAS Tensor‑Core):
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
   – produces a tensor of shape (B, H, N, N).

5️⃣ Second LayerNorm (again with PyTorch) across the hidden dimension H for
   each (b,i,j) location, followed by element‑wise multiplication with the
   already‑sigmoided ``out_gate``.

6️⃣ Final linear projection back to the original ``dim`` (half‑precision GEMM
   + cast to float32).

Only step 3 is a custom Triton kernel – all heavy matrix multiplications are
delegated to cuBLAS, giving near‑peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat row index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask handling (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                         # distance between successive h

    # base offset for the (b,i,k) entry in the (B,H,N,N) layout
    base_offset = b * (H * nn) + i * N + k      # [BLOCK_R]

    # iterate over hidden dimension in chunks of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ---- load fused projection blocks ---------------------------------
        # proj layout: [R, 5*H]   order = [left, right, left_gate, right_gate, out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ---- sigmoid (stable, compute in fp32) ---------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- compute destination offsets -----------------------------------
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        # store results
        tl.store(left_ptr      + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16) using PyTorch kernel
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim)                     # [R, dim] float32
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                           # cast to fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear: left/right proj + three gates → (R, 5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                            # [5·H, dim]

    proj = F.linear(x_norm, fused_w)               # fp16, shape [R, 5·H]
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates (write directly in B·H·N·N layout)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Prepare mask pointer (fp16). If no mask -> dummy tensor and HAS_MASK=0
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    # Launch the split‑fuse‑gate kernel
    BLOCK_R = 256
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out.view(B, H, N, N)                            # back to (B, H, N, N)
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm across hidden dim (PyTorch) + out_gate mult.
    # --------------------------------------------------------------
    # Transform to shape (B, N, N, H) so that we can normalize over H
    out_perm = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # Cast LayerNorm parameters to fp16 (compatible with half‑precision input)
    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )                                                    # still fp16

    # back to (B, H, N, N)
    out_norm = out_norm.permute(0, 3, 1, 2).contiguous()
    # element‑wise out‑gate multiplication
    out_gated = out_norm * out_gate
    del out_norm, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    # flatten to (R, H) -> (B·N·N, H)
    out_gated_flat = out_gated.permute(0, 2, 3, 1).contiguous().view(R, H)

    final_half = F.linear(
        out_gated_flat,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
