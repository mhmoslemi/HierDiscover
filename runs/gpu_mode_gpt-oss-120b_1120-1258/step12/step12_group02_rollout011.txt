"""
High‑performance Triton implementation of the *outgoing* TriMul
operator (AlphaFold‑3).  The kernel performs:

1️⃣  Row‑wise LayerNorm over the channel dimension (dim) → fp16.
2️⃣  One fused linear projection that produces the two tensors
    (left/right) and the three gating vectors in a single matmul.
3️⃣  Triton kernel `split_fuse_gate_fused_kernel`:
    * extracts the five slices (left, right, left‑gate, right‑gate,
      out‑gate) from the projected tensor,
    * optionally masks the pairwise tensors,
    * evaluates sigmoid gates (fp32 for stability),
    * writes *left* and *right* directly in the layout required for the
      later batched GEMM ((B·H, N, N) row‑major),
    * stores the output‑gate tensor for the second LayerNorm.
4️⃣  Batched GEMM (cuBLAS) over the hidden dimension:
        out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
5️⃣  Standard PyTorch layer‑norm over the hidden dimension,
    followed by element‑wise multiplication with the stored out‑gate.
6️⃣  Final linear projection back to the original feature dimension.

Only the two kernels (LayerNorm and split‑fuse‑gate) are written in
Triton; the remaining dense ops use highly‑optimized PyTorch / cuBLAS
calls.  All heavy arithmetic is done in fp16 (with fp32 where
necessary for numerical stability).  The function returns a
float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32) γ
    bias_ptr,       # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ accumulate mean & variance in fp32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]    # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized values (γ·(x‑μ)·rstd + β) in fp16
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]    # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split‑fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_fused_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[B·H, N, N] (float16)
    right_ptr,         # mutable *[B·H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]      (float16) (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    seqsq = N * N                              # N²  (int32)
    # decode (b,i,j) from flat index
    b_idx = offs_r // seqsq
    rem   = offs_r - b_idx * seqsq
    i_idx = rem // N
    j_idx = rem - i_idx * N

    # optional mask (broadcast across hidden dim)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r,
                    mask=mask_r,
                    other=1.0).to(tl.float16)          # [BLOCK_R]
        mask_val = m[:, None]                          # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)

    # iterate over hidden dimension blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # ------------------- load the five slices -------------------
        # layout per row: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, Hblk]

        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # ------------------- sigmoid (fp32) -------------------
        l_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        r_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        o_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ------------------- apply mask & gates -------------------
        left  = left_raw  * mask_val * l_gate
        right = right_raw * mask_val * r_gate

        # ------------------- write left/right in (B·H, N, N) layout ---
        # compute flat index in the (B·H, N, N) tensor
        bh = b_idx[:, None] * H + offs_h[None, :]                # [R,Hblk]
        dst_off = bh * seqsq + i_idx[:, None] * N + j_idx[:, None]  # [R,Hblk]

        tl.store(left_ptr  + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # ------------------- store out_gate (rows, H) -------------------
        tl.store(out_gate_ptr + (offs_r[:, None] * H + offs_h[None, :]),
                 o_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim`` and ``hidden_dim``. May also
                         contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total (b,i,j) positions
    H = hidden_dim
    seqsq = N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # channel chunk
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (float32, [dim])
        weights["norm.bias"],        # β (float32, [dim])
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free early

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (left/right + 3 gates)
    # ------------------------------------------------------------------
    # concatenate the five projection matrices: [5*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # shape [5*H, dim]

    proj = F.linear(x_norm, w_concat)   # (R, 5*H) fp16
    del w_concat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split fused result, apply optional mask & sigmoid gates
    #    and write left/right into (B·H, N, N) layout
    # ------------------------------------------------------------------
    left  = torch.empty(B * H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_tensor = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask expected to be 0/1 (bool or float); convert to half
        mask_tensor = mask.to(torch.float16).reshape(R)
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_fused_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_tensor,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_tensor

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension:
    #    out(b,h,i,j) = Σₖ left(b,h,i,k) · right(b,h,j,k)
    # ------------------------------------------------------------------
    # left/right are already in (B·H, N, N) layout
    out_mat = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) for layer‑norm over the hidden dim
    out_reshaped = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    out_norm = torch.nn.functional.layer_norm(
        out_reshaped,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # (B,N,N,H) fp16

    out_gate_view = out_gate.view(B, N, N, H)  # (B,N,N,H) fp16
    gated = out_norm * out_gate_view

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = torch.nn.functional.linear(
        gated,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (B,N,N,dim) fp16

    final = final_half.to(torch.float32)  # convert back to FP32 as required
    return final
