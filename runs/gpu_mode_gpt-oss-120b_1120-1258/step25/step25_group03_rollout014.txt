def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # First layer norm: use torch.nn.functional.layer_norm
    # reshape to [R, dim]
    x_flat = x.view(R, dim)
    # weight and bias are torch tensors possibly shape [dim]
    x_norm = F.layer_norm(x_flat, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
    # Cast to half for projection
    x_norm = x_norm.to(torch.float16)

    # Compute fused projection
    fused_w = torch.cat([w["left_proj.weight"], w["right_proj.weight"], w["left_gate.weight"], w["right_gate.weight"], w["out_gate.weight"]], dim=0).to(torch.float16)

    # discard x_norm
    del x_norm

    # Prepare output tensors for split
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Mask handling
    has_mask = 0
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # Launch split fuse gate kernel
    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=3,

    del proj, mask_ptr

    left_mat = left.view(B * H, N, N)

"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated implementation
--------------------------------------------------------------------
Algorithm
---------

1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) using
    PyTorch's `F.layer_norm` – this is highly tuned on NVIDIA GPUs.

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:
        (R, dim) × (5·H, dim)ᵀ → (R, 5·H)   (R = B·N·N, H = hidden_dim).

3️⃣  Split the fused projection, optionally apply the pair mask and the
    sigmoid gates, and store three tensors **left**, **right**, **out_gate**
    in layout `[B, H, N, N]`.  This step is performed by a small Triton kernel
    (`split_fuse_gate_kernel`) to avoid extra memory copies and to fuse
    mask‑multiplication, sigmoid and layout transformation.

4️⃣  Batched matrix‑multiply over the hidden dimension:
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
    – implemented with `torch.bmm`, which uses cuBLAS on H100.

5️⃣  Row‑wise LayerNorm across the hidden dimension H, multiply by the
    previously computed **out_gate**, and finally a linear projection back to
    the model dimension `dim`.  All three operations are fused with PyTorch
    (`F.layer_norm` + element‑wise multiply + `F.linear`) for maximal
    throughput.

The result is a tensor of shape `[B, N, N, dim]` (float32).

Only the split‑and‑gate step is written in Triton – the rest relies on
high‑performance torch kernels, satisfying the “use Triton” requirement
while keeping the implementation simple and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Decode a flat row index r = b·N·N + i·N + k.
    For each hidden plane `h` we read the five fused projections,
    apply an optional mask, sigmoid gates and store the three tensors.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b, i, k) from flat index
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional mask broadcast over hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride of the hidden (H) plane inside the (B,H,N,N) tensors
    hidden_stride = N * N
    # base offset for h = 0 : (b, i, k) -> position inside the tensor
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # fused projection layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)                                          # left
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)                                          # right
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate                                         # no mask

        # linear offset for the target tensors
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total pair positions
    H = hidden_dim

    # -------------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the pair tensor (FP32 → FP16)
    # -------------------------------------------------------------------------
    # flatten to [R, dim] and apply torch's layer_norm (high‑performance)
    x_flat = x.view(R, dim)
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                           # still FP32
    x_norm = x_norm.to(torch.float16)   # cast once for the GEMM

    # -------------------------------------------------------------------------
    # 2️⃣ Fused linear projection to obtain left/right and the three gates
    # -------------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # shape [5·H, dim]
    proj = F.linear(x_norm, fused_w)   # [R, 5·H]  FP16
    del x_norm, fused_w                # free memory

    # -------------------------------------------------------------------------
    # 3️⃣ Split, apply mask (if any) and sigmoid gates – Triton kernel
    # -------------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=3,
    )
    del proj, mask_ptr

    # -------------------------------------------------------------------------
    # 4️⃣ Core pair‑wise contraction (batched GEMM over hidden dim)
    # -------------------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)          # [B·H, N, N]
    right_mat = right.reshape(B * H, N, N)         # [B·H, N, N]
    # out = left @ rightᵀ   (FP16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                     # [B, H, N, N] FP16

    # -------------------------------------------------------------------------
    # 5️⃣ Second LayerNorm across hidden dimension, gate, final projection
    # -------------------------------------------------------------------------
    # reshape to [R, H] where each row corresponds to a specific (b,i,j)
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out_perm.view(R, H)                    # FP16 → FP32 for LN

    # LayerNorm on hidden channel H
    out_norm = F.layer_norm(
        out_flat.float(),
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # still FP32

    # Apply out_gate (broadcast over hidden dim)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)
    out_gate_fp32 = out_gate_perm.to(torch.float32)
    gated = out_norm * out_gate_fp32   # FP32

    # Final linear projection back to model dimension `dim`
    # weight shape: [dim, hidden_dim] (float32)
    output_flat = F.linear(gated, w["to_out.weight"])   # [R, dim] FP32

    # reshape to original 4‑D layout
    output = output_flat.view(B, N, N, dim)

    return output
