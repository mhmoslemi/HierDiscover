"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑Norm on the input pair representation (FP32 → FP16) using
    `torch.nn.functional.layer_norm` – a highly‑optimized CUDA kernel.
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM (`torch.nn.functional.linear`).  The result is reshaped
    to a flat `[R,5·H]` tensor where `R = B·N·N`.
3️⃣  Triton kernel `split_fuse_gate_kernel` extracts the three tensors
    `left`, `right` and `out_gate` from the fused projection, applies the optional
    mask, computes the sigmoid gates and writes the results in layout
    `[B, H, N, N]` (FP16).
4️⃣  The core contraction is performed with a batched GEMM:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    using `torch.bmm` (cuBLAS) → `[B, H, N, N]` (FP16).
5️⃣  Second Layer‑Norm over the hidden dimension `H` is done with the
    built‑in `layer_norm` (again in FP32 → FP16), followed by element‑wise
    multiplication with `out_gate`.
6️⃣  Final projection is a single matrix multiplication:
        (B·N·N, H) × (H, dim) → (B·N·N, dim)   (FP16)
    and the result is cast back to `float32`.

Only step 3 is a custom Triton kernel; the remaining steps use highly
optimized PyTorch/CUDA primitives, giving a sizable speed‑up over the
original reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16  fused projection
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R]          FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # start offset for h = 0
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor  [B, N, N] (bool/float) or None
            - weights_dict : {
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"
              }  (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # layer_norm works over the last dimension (dim)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # still FP32
    x_norm = x_norm.to(torch.float16)  # FP16 for the following GEMM

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5 × hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # (H, dim)
            w["right_proj.weight"],  # (H, dim)
            w["left_gate.weight"],   # (H, dim)
            w["right_gate.weight"],  # (H, dim)
            w["out_gate.weight"],   # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                         # (5·H, dim)

    # tensor shape after linear: (B, N, N, 5·H)
    proj = F.linear(x_norm, fused_w)           # FP16
    proj = proj.contiguous().view(R, 5 * H)    # flatten to [R,5·H]

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (custom Triton kernel)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free no‑longer‑needed tensors
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # view as [B·H, N, N] matrices for bmm
    left_mat  = left.view(B * H, N, N)                    # (B·H, N, N)
    right_mat = right.view(B * H, N, N)                   # (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # (B·H, N, N)
    out = out.view(B, H, N, N)                           # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate
    # --------------------------------------------------------------
    # LayerNorm over hidden dimension H (row‑wise).  Use built‑in kernel.
    # Permute to bring H to the last dimension, then permute back.
    out_perm = out.permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"].to(torch.float16),
        bias=w["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )                                               # (B, N, N, H) FP16
    out_norm = out_norm.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Apply the gate (element‑wise)
    out_gated = out_norm * out_gate  # FP16

    # --------------------------------------------------------------
    # 6️⃣ Final projection (FP16 GEMM) → float32 output
    # --------------------------------------------------------------
    # reshape to a 2‑D matrix: (B·N·N, H)
    out_flat = out_gated.view(B * N * N, H)   # FP16
    # to_out weight: (dim, H) → transpose → (H, dim)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # (H, dim)

    # matrix multiply (FP16·FP16 → FP16) then cast to FP32
    out_final = torch.matmul(out_flat, w_out_T)          # (B·N·N, dim) FP16
    out_final = out_final.view(B, N, N, dim).to(torch.float32)

    return out_final
```
