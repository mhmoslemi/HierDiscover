@triton.jit
def fused_proj_gate_kernel(
    x_ptr,    # [R, dim] fp16 (normalized)
    mask_ptr, # [R] fp16 optional
    left_ptr, right_ptr, out_gate_ptr,   # [B, H, N, N] fp16
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # scalar per row
        mask_val = mask_val[:, None]  # expand to [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R,1], 1.0, dtype=tl.float16)

    # loop over hidden tile
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Allocate accumulators in fp32
        acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_ogate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # iterate over dim in blocks
        for d_start in range(0, dim, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load x tile (Rblock, Dblock)
            x_ptrs = x_ptr + (offs_r[:, None] * dim + offs_d[None, :])
            x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            # load weight tiles for each projection
            # left_w: shape (dim, H)
            left_w = tl.load(
                left_w_ptr + (offs_d[:, None] * H + offs_h[None, :]),
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            right_w = tl.load(
                right_w_ptr + (offs_d[:, None] * H + offs_h[None, :]),
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            left_gate_w = tl.load(
                left_gate_w_ptr + (offs_d[:, None] * H + offs_h[None, :]),
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            right_gate_w = tl.load(
                right_gate_w_ptr + (offs_d[:, None] * H + offs_h[None, :]),
                mask=mask_d[:, None] & mask_h[None, :],

@triton.jit
def fused_proj_gate_kernel(
    x_ptr,  # [R, dim] FP16 normalized input
    mask_ptr, # [R] optional FP16 mask
    left_ptr, right_ptr, out_gate_ptr, # [B, H, N, N] FP16
    left_w_ptr, right_w_ptr, left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr, # [dim, H] FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    # load mask per row if applicable
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        mask_val_f32 = mask_val.to(tl.float32)[:, None]  # [BLOCK_R, 1]
    else:
        mask_val_f32 = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    # iterate over hidden blocks
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators in FP32
        acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_ogate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # loop over dim
        for d_off in range(0, dim, BLOCK_D):
            offs_d = d_off + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load x tile
            x_ptrs = x_ptr + (offs_r[:, None] * dim + offs_d[None, :])
            x = tl.load(
                x_ptrs,
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            # load weight tiles (dim, H)
            w_off = offs_d[:, None] * H + offs_h[None, :]
            left_w = tl.load(
                left_w_ptr + w_off,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            right_w = tl.load(
                right_w_ptr + w_off,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fused projection & gating.
-----------------------------------------------------------------
Algorithm (mixed‑precision, H100):
1️⃣ Row‑wise LayerNorm (FP32 → FP16)  →  x_norm [R, dim]
2️⃣ Fused kernel:
   - Computes the five linear maps (left, right, left_gate, right_gate, out_gate)
     on the normalized tensor.
   - Applies the optional pair mask, the three sigmoid gates, and writes the
     three result tensors directly in layout [B, H, N, N] (FP16).
   - This replaces the separate `F.linear` + split‑gate kernel and avoids
     materialising the large intermediate (R,5·H) tensor.
3️⃣ Batched GEMM (cuBLAS) computes out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k].
4️⃣ Row‑wise mean & rsqrt‑variance of `out`.
5️⃣ Final kernel fuses the second LayerNorm, the out‑gate multiplication,
   and the projection to the output dimension (FP32).

All kernels target Triton 3.3.1 and an NVIDIA H100 (mixed‑precision, 8 warps).
The final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]  FP32 input
    y_ptr,   # [R, dim]  FP16 output
    w_ptr,   # [dim]    LN weight (γ) FP32
    b_ptr,   # [dim]    LN bias   (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gates (produces left, right, out_gate)
# ----------------------------------------------------------------------
@triton.jit

    x_ptr,          # [R, dim]  FP16 (already layer‑normed)

    out_gate_ptr,   # [B, H, N, N] FP16
    left_w_ptr,     # [dim, H]  FP16 (transposed weight)
    right_w_ptr,    # [dim, H]  FP16
    left_gate_w_ptr,# [dim, H]  FP16
    right_gate_w_ptr,#[dim, H] FP16
    out_gate_w_ptr, # [dim, H]  FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k  # offset for h = 0

    # optional mask (scalar per pair)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val_f32 = mask_val.to(tl.float32)[:, None]                # [BLOCK_R,1]
    else:
        mask_val_f32 = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    # ------------------------------------------------------------------
    # hidden‑dimension tiles
    # ------------------------------------------------------------------
    for h_start in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_start * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators in FP32
        acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_ogate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        # reduction over the input dimension
        # ------------------------------------------------------------------
        for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_start * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # x_tile  : [BLOCK_R, BLOCK_D]
            x = tl.load(
                x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                mask=mask_r[:, None] & mask_d[None, :],
                other=0.0,
            ).to(tl.float32)

            # weight tile: [BLOCK_D, BLOCK_H] (transposed layout)
            w_off = offs_d[:, None] * H + offs_h[None, :]

            left_w = tl.load(
                left_w_ptr + w_off,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            right_w = tl.load(
                right_w_ptr + w_off,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            left_gate_w = tl.load(
                left_gate_w_ptr + w_off,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            right_gate_w = tl.load(
                right_gate_w_ptr + w_off,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            out_gate_w = tl.load(
                out_gate_w_ptr + w_off,
                mask=mask_d[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)

            # GEMV (dot) for each of the five projections
            acc_left += tl.dot(x, left_w)       # [BLOCK_R, BLOCK_H]
            acc_right += tl.dot(x, right_w)
            acc_lgate += tl.dot(x, left_gate_w)
            acc_rgate += tl.dot(x, right_gate_w)
            acc_ogate += tl.dot(x, out_gate_w)

        # ------------------------------------------------------------------
        # apply sigmoid gates and pair mask
        # ------------------------------------------------------------------
        left_gate = 1.0 / (1.0 + tl.exp(-acc_lgate))
        right_gate = 1.0 / (1.0 + tl.exp(-acc_rgate))
        out_gate = 1.0 / (1.0 + tl.exp(-acc_ogate))

        left = acc_left * mask_val_f32 * left_gate   # FP32
        right = acc_right * mask_val_f32 * right_gate
        outg = out_gate                               # no mask

        # store results (cast to fp16)
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(
            left_ptr + dst_off,
            left.to(tl.float16),
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_off,
            right.to(tl.float16),
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_off,
            outg.to(tl.float16),
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j  # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # Load per‑row statistics (mu, rstd)
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)     # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0) # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load hidden‑plane values
        ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        # LayerNorm parameters (γ,β) per hidden index
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)    # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * vals_gate                                      # [M, K]
        term_fp16 = term.to(tl.float16)                               # back to FP16 for dot

        # Load a slice of the final projection weight
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # Accumulate dot product (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"      (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of (b,i,k) pairs
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (produces left, right, out_gate)
    # --------------------------------------------------------------
    # transpose weights to shape (dim, hidden_dim) for efficient dot‑product
    left_w_T   = w["left_proj.weight"].to(torch.float16).t().contiguous()
    right_w_T  = w["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_w_T  = w["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_w_T = w["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_w_T   = w["out_gate.weight"].to(torch.float16).t().contiguous()

    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        # dummy tensor – never accessed when HAS_MASK == 0
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64     # hidden‑dim tile (fits inside registers for H=128)
    BLOCK_D = 64
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    fused_proj_gate_kernel[grid_proj](
        x_norm,
        mask_ptr,
        left,
        right,
        out_gate,
        left_w_T,
        right_w_T,
        left_gate_w_T,
        right_gate_w_T,
        out_gate_w_T,
        R,
        H,
        N,
        dim,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                # hidden_dim ≤ 768 fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
