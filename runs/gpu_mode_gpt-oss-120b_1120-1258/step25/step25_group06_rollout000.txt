"""
TriMul (outgoing) – fully‑fused forward pass.

The implementation follows the reference PyTorch module but moves the heavyweight
operations into Triton kernels:

1. **LayerNorm** over the last dimension (FP32 → FP16) – one‑program per row.
2. **Fused linear projection** of the 5 weight matrices (left, right, left_gate,
   right_gate, out_gate).  The projection is performed by a regular
   `torch.nn.functional.linear` in FP16 and the result is split in a Triton
   kernel while simultaneously applying the pair‑wise mask (if present) and the
   sigmoid gates.
3. **Batched GEMM** over the hidden dimension `H`:  
   `out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]`.  
   This is the expensive N³ term and is executed with cuBLAS via `torch.bmm`.
4. **Row‑wise statistics** (mean & rsqrt‑variance) of `out` over the hidden
   dimension – required for the second LayerNorm.
5. **Final LayerNorm + out‑gate + projection** fused into a single Triton kernel
   that writes the result `[B, N, N, dim]` in FP32.

All kernels use mixed‑precision (FP16 for the bulk of the work, FP32 for
reductions) and are tuned for H100 (8 warps, block sizes chosen to keep shared
memory < 64 KB).  The code works for any `B ∈ {1,2}`, `N ∈ {128,…,1024}`,
`dim ∈ {128,384}` and `hidden_dim` up to 128 (the values used in the tests).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm  (FP32 → FP16)
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, D]   float32 input
    y_ptr,    # [R, D]   float16 output
    w_ptr,    # [D]      gamma (float32)
    b_ptr,    # [D]      beta  (float32)
    rows, D, eps,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ---- first pass: mean & variance ----
    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_R,), dtype=tl.float32)

    for d0 in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d0 * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        ptr = x_ptr + offs_r[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d0 in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d0 * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_r[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_r[:, None] * D + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])


# -------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# -------------------------------------------------------------
@triton.jit
def split_gate_kernel(
    proj_ptr,        # [R, 5*H] FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b * N*N + i * N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional scalar mask per (b,i,k)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mval = mval[:, None]                                       # [BLOCK_R,1]
    else:
        mval = tl.full((BLOCK_R, 1), 1.0, tl.float16)

    # layout helpers
    hid_stride = N * N                               # distance between hidden planes
    base_offset = b * (H * hid_stride) + i * N + k   # offset for hidden h=0

    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # each column in proj contains five groups of size H (contiguous)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]                # [R, H_tile]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 → cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hid_stride   # [R, H_tile]

        tl.store(left_ptr  + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# -------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# -------------------------------------------------------------
@triton.jit
def stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    R, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hid_stride = N * N
    base_off = b * (H * hid_stride) + i * N + j   # hidden plane 0

    sum_val = tl.zeros((BLOCK_R,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_R,), dtype=tl.float32)

    for k0 in range(0, H, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_off[:, None] + offs_k[None, :] * hid_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)     # [M, K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd,  mask=mask_r)


# -------------------------------------------------------------
# 4️⃣ Final LayerNorm + out_gate + projection (fused)
# -------------------------------------------------------------
@triton.jit
def final_kernel(
    out_ptr,          # [B, H, N, N] FP16
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    mu_ptr,           # [R] FP32
    rstd_ptr,         # [R] FP32
    ln_w_ptr,         # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] FP16   (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim] FP32
    R, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_r = tl.program_id(0)   # rows (R)
    pid_d = tl.program_id(1)   # dim chunk

    row_start = pid_r * BLOCK_R
    dim_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)   # [M]
    offs_d = dim_start + tl.arange(0, BLOCK_D)   # [D]

    mask_r = offs_r < R
    mask_d = offs_d < dim

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hid_stride = N * N
    base_off = b * (H * hid_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [M]

    acc = tl.zeros((BLOCK_R, BLOCK_D), dtype=tl.float32)

    for k0 in range(0, H, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden plane values (FP16 → FP32)
        ptr_out = out_ptr  + base_off[:, None] + offs_k[None, :] * hid_stride
        ptr_gate= gate_ptr + base_off[:, None] + offs_k[None, :] * hid_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # second LayerNorm (γ,β) – broadcast over rows
        ln_w = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        ln_b = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * ln_w[None, :] + ln_b[None, :]
        term = norm * gate_vals                                     # [M, K]

        term_fp16 = term.to(tl.float16)                             # [M, K]

        # weight tile: [K, D] from transposed weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_d[None, :]   # [K, D]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_d[None, :],
                         other=0.0)                                      # [K, D]

        # accumulate dot products (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final result [B, N, N, dim] (FP32)
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_r[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"          (all float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over last dimension (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim)
    # ------------------------------------------------------------------
    # concatenate weight matrices: shape [5*H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5*H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split / mask / gates
    # ------------------------------------------------------------------
    left        = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right       = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # optional mask
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        # dummy tensor (never accessed when HAS_MASK==0)
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core tensor contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)            # [B*H, N, N]
    right_mat = right.view(B * H, N, N)           # [B*H, N, N]

    # out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean & rstd) over hidden dimension
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 32    # hidden_dim ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out_gate + projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128          # rows per program (tiles over R)
    BLOCK_K_OUT = 32           # hidden‑dim chunk
    BLOCK_D_OUT = 128          # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_D_OUT),
    )
    final_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],    # γ (float32)
        w["to_out_norm.bias"],      # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_D=BLOCK_D_OUT,
        num_warps=8,
    )

    return final
