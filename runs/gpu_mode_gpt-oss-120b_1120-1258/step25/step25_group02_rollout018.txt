def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    eps = 1e-5
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    device = x.device

    # 1 rownorm: flatten and run 2-pass kernel
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty_like(x_flat, dtype=torch.float16)
    # choose block size
    BLOCK_M = 256
    BLOCK_D = 128 if dim > 192 else 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # Linear mix
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, fused_weight) # shape R,5*H
    # Free x_norm
    del x_norm

    # Reshape & split
    proj = proj.view(R, 5, H) # shape (R,5,H)
    left = proj[:,0]   # (R,H)
    right = proj[:,1]
    left_gate = proj[:,2]
    right_gate = proj[:,3]
    out_gate_raw = proj[:,4]

    # mask
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R, 1)  # (R,1)
        left = left * mask_flat
        right = right * mask_flat
    # gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate_raw)
    # apply gating
    left = left * left_gate
    right = right * right_gate

    # reshape to B,H,N,N
    left = left.view(B, N, N, H).permute(0,3,1,2).contiguous()
    right = right.view(B, N, N, H).permute(0,3,1,2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0,3,1,2).contiguous()

    # core gemm: batch matmul
    left_mat = left.view(B*H, N, N) # [B*H, N, N]
    right_mat = right.view(B*H, N, N)  # same
    out = torch.bmm(left_mat, right_mat.transpose(1,2)) # [B*H,N,N]
    out = out.view(B, H, N, N)

    # Compute rowwise mean and rstd along hidden dimension H
    # Out shape (B,H,N,N)
    # Flatten first 3 dims: B,N,N => R
    out_flat = out.permute(0,2,3,1).contiguous().view(R, H) # shape (R,H)
    mu = out_flat.mean(dim=1)   # shape(R)
    var = out_flat.var(dim=1, unbiased=False)   # shape(R)
    rstd = torch.rsqrt(var + eps)                # shape(R)
    # Reshape back
    mu = mu.view(B, 1, N, N)
    rstd = rstd.view(B, 1, N, N)

    # apply second LN, out_gate and final proj
    gamma = weights["to_out_norm.weight"].view(1, H, 1, 1)  # [1,H,1,1]
    beta = weights["to_out_norm.bias"].view(1, H, 1, 1)    # [1,H,1,1]
    # Normalize
    out_norm = (out - mu) * rstd
    out_norm = out_norm * gamma + beta
    out_norm = out_norm * out_gate

    # Final projection: flatten to (R,H) and apply weight
    final_flat = out_norm.permute(0,2,3,1).contiguous().view(R, H)  # (R,H)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final

"""
Triton‑accelerated implementation of the outgoing TriMul operator.

Algorithm:
1. Row‑wise LayerNorm (FP32 → FP16) on the flattened pair representation.
2. Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a single GEMM.
3. Split the fused projection, apply optional mask, compute sigmoid gates and compose left/right tensors.
4. Batched matrix‑multiply over the hidden dimension (cuBLAS) implements the core einsum.
5. Compute per‑row mean / rsqrt‑variance of the contracted tensor using torch reductions.
6. Apply the second LayerNorm, the out‑gate, and a final linear projection (cuBLAS) in pure torch
   (the numerically dominant kernels are already cuBLAS‑optimized; only the custom LayerNorm
   is written in Triton).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implemented with a small Triton LayerNorm
    and the rest of the pipeline in highly‑optimized torch primitives.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",

        torch.Tensor: [B, N, N, dim] (float32)
    """

    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                      # flattened pair dimension
    H = hidden_dim

    device = x.device

    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_D = 128 if dim > 192 else 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)          # [R, 5·H] FP16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)                 # [R, 5, H]
    left      = proj[:, 0, :]                 # [R, H]
    right     = proj[:, 1, :]                 # [R, H]
    left_gate = proj[:, 2, :]                 # [R, H]
    right_gate= proj[:, 3, :]                 # [R, H]
    out_gate_raw = proj[:, 4, :]              # [R, H]

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R, 1)   # [R,1]
        left  = left  * mask_flat
        right = right * mask_flat

    # Sigmoids (computed in FP32 for stability)
    left_gate  = torch.sigmoid(left_gate.to(torch.float32)).to(torch.float16)
    right_gate = torch.sigmoid(right_gate.to(torch.float32)).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw.to(torch.float32)).to(torch.float16)

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # Reshape to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right= right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_gate, right_gate, out_gate_raw

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out.view(B, H, N, N)                         # [B, H, N, N]

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance (FP32)
    # --------------------------------------------------------------
    # Work in FP32 for the statistics – conversion is cheap compared to the GEMM.
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H]

    mu   = out_flat.mean(dim=1, keepdim=True)               # [R, 1]
    var  = out_flat.var(dim=1, unbiased=False, keepdim=True)  # [R, 1]
    rstd = torch.rsqrt(var + eps)                           # [R, 1]

    mu   = mu.view(B, 1, N, N)      # [B,1,N,N]
    rstd = rstd.view(B, 1, N, N)    # [B,1,N,N]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate, final projection (all torch)
    # --------------------------------------------------------------
    # Normalisation (γ,β are FP32)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1]
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)     # [1,H,1,1]

    # (out - μ) * rstd   ->   FP32
    norm = (out_fp32 - mu) * rstd
    norm = norm * gamma + beta

    # Multiply by the out‑gate (cast gate to FP32)
    norm = norm * out_gate.to(torch.float32)

    # Cast back to FP16 for the final linear (cuBLAS heavily optimised for FP16)
    norm_fp16 = norm.to(torch.float16)

    # Final linear projection  (R, H) @ (dim, H).T  → (R, dim)
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final_flat = norm_fp16.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
    final_fp16 = F.linear(final_flat, w_out)                               # [R, dim]

    # Reshape to [B, N, N, dim] and promote to FP32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final

    right = right * right_gate

    # Reshape to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_gate, right_gate, out_gate_raw

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out.view(B, H, N, N)                         # [B, H, N, N]

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance (FP32)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R,H]

    mu   = out_flat.mean(dim=1, keepdim=True)               # [R,1]
    var  = out_flat.var(dim=1, unbiased=False, keepdim=True)  # [R,1]
    rstd = torch.rsqrt(var + eps)                           # [R,1]

    mu   = mu.view(B, 1, N, N)      # [B,1,N,N]
    rstd = rstd.view(B, 1, N, N)    # [B,1,N,N]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out‑gate, final projection (torch)
    # --------------------------------------------------------------
    # Normalise (γ,β are FP32)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1]
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)     # [1,H,1,1]

    norm = (out_fp32 - mu) * rstd
    norm = norm * gamma + beta

    # Multiply by out_gate (cast gate to FP32)
    norm = norm * out_gate.to(torch.float32)

    # Cast back to FP16 for the final linear (cuBLAS best with FP16)
    norm_fp16 = norm.to(torch.float16)

    # Final linear projection (R, H) @ (dim, H).T → (R, dim)
    w_out = w["to_out.weight"].to(torch.float16)          # [dim, H]
    final_flat = norm_fp16.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R,H]
    final_fp16 = F.linear(final_flat, w_out)                               # [R,dim]

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
