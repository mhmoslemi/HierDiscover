    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    H_MAX: tl.constexpr, # compile-time max hidden dimension, maybe =128
):

    pid_n = tl.program_id(1)

    # compute row indices...
    # allocate shared memory:
    out_shmem = tl.shared_memory([BLOCK_M * H_MAX], dtype=tl.float16)
    gate_shmem = tl.shared_memory([BLOCK_M * H_MAX], dtype=tl.float16)

    # First pass: load out and gate into shared memory, accumulate sum and sumsq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_K):
        h_off = h_start + tl.arange(0, BLOCK_K)
        mask_h = h_off < H

        # load out values
        out_vals = tl.load(out_ptr + base_offset[:, None] + h_off[None,:] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None,:], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + h_off[None,:] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None,:], other=0.0).to(tl.float32)

        # store to shared memory (convert to fp16)
        out_fp16 = out_vals.to(tl.float16)
        gate_fp16 = gate_vals.to(tl.float16)

        # compute offset in shmem: row_idx * H_MAX + h_off

        tl.store(out_shmem + offs_shmem, out_fp16, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(gate_shmem + offs_shmem, gate_fp16, mask=mask_m[:, None] & mask_h[None, :])

        # accumulate for stats using fp32 values (out_vals)
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq  += tl.sum(out_vals * out_vals, axis=1)

"""
Outgoing TriMul (AlphaFold‑3) – fused row‑wise statistics + final projection

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into one GEMM:  (R, dim) × (5·H, dim)ᵀ → (R, 5·H)  (FP16).
3️⃣  Split the fused projection, optionally mask it, apply sigmoid gates
    and store the three tensors *left*, *right* and *out_gate* in layout
    [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, k, j]
    producing **out** ∈ FP16 of shape [B, H, N, N].
5️⃣  One Triton kernel (``final_fused_kernel``) loads the hidden‑plane
    values of *out* and *out_gate*, computes the per‑row mean/variance,
    applies the second LayerNorm, multiplies by *out_gate* and finally
    projects to the model dimension (FP16⋅FP16 → FP32).  The kernel
    stores the result in [B, N, N, dim] (float32) and eliminates the
    separate row‑wise‑stats kernel, saving a memory round‑trip.

All kernels are written for Triton 3.3.1 and target an NVIDIA H‑100.
Mixed‑precision (FP16 bulk, FP32 for reductions) is used throughout.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]  FP32 input
    y_ptr,          # [R, dim]  FP16 output
    w_ptr,          # [dim]    γ (FP32)
    b_ptr,          # [dim]    β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass – compute mean & variance --------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass – write normalized values --------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        lx = tl.load(proj_ptr + proj_base,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        rx = tl.load(proj_ptr + proj_base + H,
                     mask=mask_r[:, None] & mask_h[None, :],
                     other=0.0)
        lgate = tl.load(proj_ptr + proj_base + 2 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        rgate = tl.load(proj_ptr + proj_base + 3 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)
        ogate = tl.load(proj_ptr + proj_base + 4 * H,
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0)

        # sigmoid in FP32 → back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate.to(tl.float32)))).to(tl.float16)
        ogate = (1.0 / (1.0 + tl.exp(-ogate.to(tl.float32)))).to(tl.float16)

        left  = lx * mask_val * lgate
        right = rx * mask_val * rgate
        outg  = ogate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# --------------------------------------------------------------
# 3️⃣ Fuse row‑wise statistics, second LayerNorm, out_gate and
#    final projection into a single kernel
# --------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # [B, H, N, N]   FP16   (contracted tensor)
    gate_ptr,         # [B, H, N, N]   FP16   (out_gate)
    ln_w_ptr,         # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16   (transposed to_out weight)
    out_final_ptr,    # [B, N, N, dim] FP32   (final output)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
    H_MAX: tl.constexpr,
):
    """
    For each flattened pair index (b,i,j) we:
        * load all hidden‑plane values of `out` and `out_gate` into shared memory;
        * compute the per‑row mean / rstd (row‑wise statistics);
        * apply the second LayerNorm (γ/β), multiply by `out_gate`;
        * accumulate the dot‑product with the final projection matrix.
    The kernel writes the result directly to the [B, N, N, dim] tensor.
    """
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N                            # distance between hidden planes
    base_offset = b * (H * hidden_stride) + i * N + j

    # ----------------------------------------------------------
    # Shared memory: store the full hidden vector for each row
    # ----------------------------------------------------------
    # H_MAX must be a compile‑time constant ≥ hidden_dim (128 in all tests)
    out_shmem  = tl.shared_memory(BLOCK_M * H_MAX, dtype=tl.float16)
    gate_shmem = tl.shared_memory(BLOCK_M * H_MAX, dtype=tl.float16)

    # ----------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance while caching values
    # ----------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # load a tile of the contracted tensor and the gate
        out_tile = tl.load(out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gate_tile = tl.load(gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # store FP16 copies in shared memory for the second pass
        out_fp16  = out_tile.to(tl.float16)
        gate_fp16 = gate_tile.to(tl.float16)

        # linear index into the 1‑D shared memory buffer
        shmem_off = offs_m[:, None] * H_MAX + offs_h[None, :]
        tl.store(out_shmem  + shmem_off, out_fp16,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(gate_shmem + shmem_off, gate_fp16,
                 mask=mask_m[:, None] & mask_h[None, :])

        # accumulate sums for mean / variance
        sum_val += tl.sum(out_tile, axis=1)
        sum_sq  += tl.sum(out_tile * out_tile, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ----------------------------------------------------------
    # 2️⃣  Apply LayerNorm, out_gate and final projection
    # ----------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)
        mask_h = offs_h < H

        # reload the cached values from shared memory
        shmem_off = offs_m[:, None] * H_MAX + offs_h[None, :]
        out_fp16 = tl.load(out_shmem + shmem_off,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)
        gate_fp16 = tl.load(gate_shmem + shmem_off,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0)

        out_f32   = out_fp16.to(tl.float32)
        gate_f32  = gate_fp16.to(tl.float32)

        # LayerNorm γ/β for hidden channels
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # (out - μ)·rstd → γ· + β → * out_gate
        norm = ((out_f32 - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_f32                     # FP32

        # cast back to FP16 for the dot with the final weight
        term_fp16 = term.to(tl.float16)

        # tile of the final projection weight (H, dim)
        w_tile = tl.load(w_out_T_ptr + offs_h[:, None] * dim + offs_n[None, :],
                         mask=mask_h[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, TILE_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)   # [M, N]

    # write the fused result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# 4️⃣ Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pair positions
    H = hidden_dim

    # ----------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ----------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                         # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)  # FP16 result

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ----------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ----------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)          # FP16, shape [R, 5·H]
    del fused_w, x_norm

    # ----------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # ----------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ----------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ----------------------------------------------------------------------
    # reshape to [B·H, N, N] and perform bmm
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16, shape [B·H, N, N]
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ----------------------------------------------------------------------
    # 5️⃣ Fuse row‑wise stats, 2nd LayerNorm, out_gate & final proj
    # ----------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64            # rows per program (tiles over R)
    BLOCK_K_OUT = 32            # hidden‑dim chunk
    BLOCK_N_OUT = 128           # output‑dim chunk
    H_MAX = hidden_dim          # compile‑time constant (≤128 in all tests)

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        H_MAX=H_MAX,
        num_warps=8,
    )

    return final
