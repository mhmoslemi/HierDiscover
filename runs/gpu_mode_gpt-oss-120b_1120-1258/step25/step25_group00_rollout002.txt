"""
Outgoing TriMul (AlphaFold3) – forward‑only Triton implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair representation (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) producing a tensor of shape (R, 5·H) where
    R = B·N·N and H = hidden_dim.
3️⃣  Split the fused projection, apply the optional mask, compute the sigmoid
    gates and store the three tensors `left`, `right`, `out_gate` in layout
    [B, H, N, N] (FP16) via a Triton kernel.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).
5️⃣  Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H via Triton.
6️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the final
    projection (`to_out`) into a single Triton kernel, writing the final
    `[B, N, N, dim]` tensor in FP32.

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed‑precision
(FP16 for the bulk of the work, FP32 for the reductions).  The
implementation follows the reference PyTorch code but keeps the
memory‑traffic‑heavy steps on the GPU to obtain > 1 µs per token
throughput on the benchmark shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, D] FP32 input
    y_ptr,   # [R, D] FP16 output
    w_ptr,   # [D]   LN weight (γ) FP32
    b_ptr,   # [D]   LN bias (β)   FP32
    rows, D, eps,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < rows

    # --- first pass: mean & variance ---
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        ptr = x_ptr + offs_r[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_R, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --- second pass: write normalized values ---
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        ptr = x_ptr + offs_r[:, None] * D + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_r[:, None] * D + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_r[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]  FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask loading
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # strides for the destination layout [B, H, N, N] (H is the innermost dim)
    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + k          # start of hidden slice 0

    # iterate over hidden-dim tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: [left|right|l_gate|r_gate|out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 (more accurate) then back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # already masked in step 1 (mask only applies to left/right)

        # destination offset (flattened)
        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    R, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # [BLOCK_R, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd,  mask=mask_r)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16  (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    R, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0)       # rows dimension (R)
    pid_d = tl.program_id(1)       # output‑dim tiles (dim)

    row_start = pid_r * BLOCK_R
    col_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    offs_d = col_start + tl.arange(0, BLOCK_D)   # [BLOCK_D]

    mask_r = offs_r < R
    mask_d = offs_d < dim

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # start of hidden slice 0

    # per‑row LN stats (FP32)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]

    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    # iterate over hidden dimension chunks
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)                # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden‑plane values (FP16 → FP32)
        ptr_out = out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(
            ptr_out,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_K]

        vals_gate = tl.load(
            ptr_gate,
            mask=mask_r[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_K]

        # LayerNorm params for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # LN: (out - μ) * rstd  →  *γ + β
        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # Apply out_gate
        gated = norm * vals_gate                               # [BLOCK_R, BLOCK_K]

        # Convert to FP16 for GEMM with final projection weight
        gated_fp16 = gated.to(tl.float16)

        # Load weight slice [K, BLOCK_D] (row‑major in the transposed weight)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_d[None, :]   # [BLOCK_K, BLOCK_D]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_d[None, :],
            other=0.0,
        )                                                             # [BLOCK_K, BLOCK_D]

        # Accumulate (M,K)·(K,D) → (M,D)   (FP16·FP16 → FP32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # store final tensor (row‑major)
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_r[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]
    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # mask is [B, N, N] -> flatten to [R] and cast to FP16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R_SPLIT = 64
    BLOCK_H_SPLIT = 64
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # view as [B·H, N, N] for batched matmul
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean & rsqrt‑variance) of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 128          # hidden_dim ≤ 128 → fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gate and projection (fused)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128
    BLOCK_K_OUT = 64      # hidden‑dim tile
    BLOCK_D_OUT = 128     # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_D_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ FP32
        w["to_out_norm.bias"],     # β FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_D=BLOCK_D_OUT,
        num_warps=8,
    )

    return final
