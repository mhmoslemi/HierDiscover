@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows, dim,
    stride_row, stride_dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # compute mean & variance
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * stride_row + offs_d[None, :] * stride_dim
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized output
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * stride_row + offs_d[None, :] * stride_dim
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

```python
def custom_kernel(data):
    """
    Fast outgoing TriMul forward pass (AlphaFold‑3 style).

    The kernel:
    1️⃣  PyTorch layer‑norm on the input (float32 → fp16).
    2️⃣  A single fused half‑precision linear layer producing left/right projections
        and their sigmoid gates.
    3️⃣  Optional mask application.
    4️⃣  Batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ using torch.matmul.
    5️⃣  Row‑wise LayerNorm on the N³ update (implemented with a custom Triton kernel
        that works on strided data, i.e. no extra permute / copy).
    6️⃣  Element‑wise output‑gate multiplication.
    7️⃣  Final half‑precision projection to the original ``dim`` using a
        vectorised ``einsum`` to avoid an extra permute+contiguous copy.

    All heavy linear algebra is performed in fp16; the final result is cast back
    to float32 for compatibility with the reference implementation.
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    total_rows = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use torch's highly‑optimised implementation)
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension ``dim``
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=weights["norm.weight"],
                          bias=weights["norm.bias"],
                          eps=eps)                 # float32
    x_norm = x_norm.to(torch.float16)                # fp16
    x_norm_flat = x_norm.view(total_rows, dim)       # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # ------------------------------------------------------------------
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)  # [4·H, dim]

    fused = F.linear(x_norm_flat, w_4h)               # fp16, [R, 4·H]

    left_raw      = fused[:, 0:H]                     # [R, H]
    right_raw     = fused[:, H:2*H]                  # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]                # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]               # [R, H]

    out_gate_raw = F.linear(
        x_norm_flat,
        weights["out_gate.weight"].to(torch.float16)  # [H, dim]
    )                                                # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H]
    right = right_raw * right_gate  # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to [B, H, N, N] (contiguous) for the N³ update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # (B·H, N, N) @ (B·H, N, N)ᵀ  ->  (B·H, N, N)
    out_mat = torch.matmul(left, right.transpose(-1, -2))   # fp16, shape [B, H, N, N]

    BLOCK_D = 32

    stride_row = 1                        # rows are packed consecutively in memory
    stride_dim = N * N                    # distance between two consecutive hidden elements

    layernorm_kernel[grid](
        out_mat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_ln2,
        H,
        stride_row,
        stride_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection (einsum avoids extra permutes)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                       # element‑wise fp16

    weight_out = weights["to_out.weight"].to(torch.float16)  # [dim, H]
    final_fp16 = torch.einsum('b h i j, d h -> b i j d', out_norm, weight_out)  # fp16

    return final_fp16.to(torch.float32)

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Input LayerNorm – use PyTorch’s highly‑optimised `nn.functional.layer_norm`
    (float32 → float16) – keeps the implementation simple and fast.

2️⃣  Fuse the four linear layers (left/right projections + their two gates) into a
    single half‑precision GEMM:   [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    where R = B·N·N and H = hidden_dim.  The result is split into
    left_proj, right_proj, left_gate, right_gate.

3️⃣  Optional mask application, sigmoid gates and element‑wise gating.

4️⃣  Reshape to [B, N, N, H] and permute to [B, H, N, N] (contiguous) so that a
    single batched GEMM computes the triangular multiplicative update:
        out = left @ rightᵀ   (B·H independent N×N matmuls).

5️⃣  Row‑wise LayerNorm on the hidden dimension (the only Triton kernel).  The
    kernel works for arbitrary strides, allowing us to normalise directly on the
    strided layout produced by the GEMM – no extra permute / copy is needed.

6️⃣  Multiply by the output‑gate (already in the same layout).

7️⃣  Final projection back to `dim` with a vectorised `einsum`.  Because the
    hidden dimension is the second axis (`[B, H, N, N]`) the einsum can write the
    result directly in the required `[B, N, N, dim]` layout, avoiding another
    transpose.

All heavy linear algebra runs in FP16 (Tensor‑Core accelerated on H100);
the final result is cast to FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: generic row‑wise LayerNorm (supports strided data)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const   * [total_rows, dim] (float16 input)
    y_ptr,                # mutable * [total_rows, dim] (float16 output)
    weight_ptr,           # const   * [dim] (float32 γ)
    bias_ptr,             # const   * [dim] (float32 β)
    total_rows,           # int (runtime)
    dim,                  # int (runtime)
    stride_row,           # int (distance between successive rows)
    stride_dim,           # int (distance between successive elements inside a row)
    eps,                  # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x − μ) / sqrt(σ² + eps)) * γ + β

    The tensor may be strided: element (r, d) is located at
        x_ptr + r * stride_row + d * stride_dim
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer = base + row_offset*stride_row + col_offset*stride_dim
        ptrs = x_ptr + offs_m[:, None] * stride_row + offs_d[None, :] * stride_dim
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)               # fp16 → fp32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * stride_row + offs_d[None, :] * stride_dim
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ  (float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β  (float32)

        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * stride_row + offs_d[None, :] * stride_dim
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use PyTorch implementation – fast and fully‑fused)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                 # float32
    x_norm = x_norm.to(torch.float16) # fp16
    x_flat = x_norm.view(R, dim).contiguous()   # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their two gates (4·H outputs)
    # ------------------------------------------------------------------
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                 # [4·H, dim]

    fused = F.linear(x_flat, w_4h)      # fp16, shape [R, 4·H]

    left_raw      = fused[:, 0:H]               # [R, H]
    right_raw     = fused[:, H:2*H]            # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]          # [R, H]
    right_gate_raw= fused[:, 3*H:4*H]          # [R, H]

    out_gate_raw = F.linear(
        x_flat,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                             # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates, and apply them
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H]   fp16
    right = right_raw * right_gate  # [R, H]   fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to [B, N, N, H] (contiguous) – prepare for the N³ update
    # ------------------------------------------------------------------
    left_view  = left.view(B, N, N, H)   # [B, N, N, H]
    right_view = right.view(B, N, N, H)  # [B, N, N, H]
    out_gate_view = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # Permute to [B, H, N, N] and make contiguous for the batched GEMM
    left_mat   = left_view.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat  = right_view.permute(0, 3, 1, 2).contiguous()
    out_gate_mat = out_gate_view.permute(0, 3, 1, 2).contiguous()  # for later use

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-1, -2))   # fp16, [B, H, N, N]

    # ------------------------------------------------------------------
    # 6️⃣ Row‑wise LayerNorm over the hidden dimension (stride‑aware kernel)
    # ------------------------------------------------------------------
    out_norm = torch.empty_like(out_mat, dtype=torch.float16)

    BLOCK_M = 256
    BLOCK_D = 32
    rows_ln = B * N * N                     # total_rows for LN
    stride_row = 1                          # rows are packed contiguously
    stride_dim = N * N                      # distance between two hidden elements

    grid = (triton.cdiv(rows_ln, BLOCK_M),)
    layernorm_kernel[grid](
        out_mat,                     # x_ptr (fp16)
        out_norm,                    # y_ptr (fp16)
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        rows_ln,
        H,
        stride_row,
        stride_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    # out_norm : [B, H, N, N]   fp16
    # out_gate_mat : [B, H, N, N] fp16 (same layout)
    out_norm = out_norm * out_gate_mat                     # element‑wise fp16

    # Final linear projection via einsum – avoids an extra transpose+contiguous
    weight_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_fp16 = torch.einsum('b h i j, d h -> b i j d', out_norm, weight_out)  # fp16

    return final_fp16.to(torch.float32)
