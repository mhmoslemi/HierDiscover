"""
Highly‑optimized Triton implementation of the outgoing TriMul operator
(as described in AlphaFold‑3).  The heavy 4‑D tensor flow is:

1️⃣  Row‑wise LayerNorm on the input (float32 → float16) and simultaneous
    projection of the normalised tensor to four linear branches
    (left, right, left‑gate, right‑gate) plus the output‑gate.
    The LN γ (weight) is folded into the linear weight, the LN β (bias)
    is folded into a pre‑computed bias vector.  The four raw branches are
    immediately gated (σ) and the left/right branches are multiplied by
    their gates, producing the tensors needed for the N³ multiplicative
    update.

2️⃣  Outgoing mask (if present) is applied to the gated left/right tensors.

3️⃣  The core N³ update (batched GEMM):
    out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
    Implemented with a single torch.bmm on a view (B·H, N, N).

4️⃣  A fused kernel that
    – computes per‑row mean / variance of the hidden‑dim tensor,
    – applies the second LayerNorm (γ/β),
    – multiplies by the previously computed output‑gate,
    – and directly projects to the model dimension with a GEMM.
    The kernel emits the final float‑32 tensor, eliminating the
    intermediate storage of the gated hidden tensor.

All kernels run in two passes (statistics → normalisation) and use
block‑wise FP32 accumulation for numerical stability.  The implementation
covers all test configurations listed in the problem description.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fused LayerNorm + four projections + gating
# ----------------------------------------------------------------------
@triton.jit
def proj_ln_gate_fused(
    x_ptr,                         # const fp32  [R, D]
    left_out_ptr, right_out_ptr,   # mutable fp16 [R, H]
    out_gate_ptr,                  # mutable fp16 [R, H]
    w_left_ptr, w_right_ptr, w_lgate_ptr,
    w_rgate_ptr, w_outgate_ptr,   # const fp16  [D, H] (transposed & γ‑folded)
    b_left_ptr, b_right_ptr, b_lgate_ptr,
    b_rgate_ptr, b_outgate_ptr,   # const fp32  [H]   (β‑folded)
    total_rows, dim, hidden_dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ statistics (mean / variance) of the input rows
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ projection + bias + gate (four branches)
    # ------------------------------------------------------------------
    sigmoid = lambda x: 1.0 / (1.0 + tl.exp(-x))

    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # accumulators for the current hidden‑dim block
        left_acc   = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        lgate_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        rgate_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        outgate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        # loop over the feature dimension D
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # ---- load a tile of the input and normalise it ----
            x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            x = tl.load(x_ptrs,
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)               # [BM, BD]
            norm = (x - mean[:, None]) * rstd[:, None]            # FP32

            # ---- load the transposed weight tiles (shape [BD, BH]) ----
            wL = tl.load(w_left_ptr   + offs_d[:, None] * hidden_dim + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wR = tl.load(w_right_ptr  + offs_d[:, None] * hidden_dim + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wLG = tl.load(w_lgate_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wRG = tl.load(w_rgate_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            wOG = tl.load(w_outgate_ptr + offs_d[:, None] * hidden_dim + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

            # ---- GEMM accumulation for the five branches ----
            left_acc   += tl.dot(norm, wL)
            right_acc  += tl.dot(norm, wR)
            lgate_acc  += tl.dot(norm, wLG)
            rgate_acc  += tl.dot(norm, wRG)
            outgate_acc+= tl.dot(norm, wOG)

        # ---- add β‑folded bias vectors (broadcasted over rows) ----
        bL = tl.load(b_left_ptr   + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        bR = tl.load(b_right_ptr  + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        bLG = tl.load(b_lgate_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        bRG = tl.load(b_rgate_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        bOG = tl.load(b_outgate_ptr+ offs_h, mask=mask_h, other=0.0).to(tl.float32)

        left_acc   += bL[None, :]
        right_acc  += bR[None, :]
        lgate_acc  += bLG[None, :]
        rgate_acc  += bRG[None, :]
        outgate_acc+= bOG[None, :]

        # ---- sigmoid gates and final left/right values ----
        lgate = sigmoid(lgate_acc)
        rgate = sigmoid(rgate_acc)
        left  = left_acc   * lgate
        right = right_acc  * rgate
        out_gate = sigmoid(outgate_acc)                     # already sigmoid‑ed

        # ---- store results (FP16) ----
        left_ptrs  = left_out_ptr   + offs_m[:, None] * hidden_dim + offs_h[None, :]
        right_ptrs = right_out_ptr  + offs_m[:, None] * hidden_dim + offs_h[None, :]
        out_ptrs   = out_gate_ptr   + offs_m[:, None] * hidden_dim + offs_h[None, :]

        tl.store(left_ptrs,  left.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_ptrs,   out_gate.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 2: fused second LayerNorm + output‑gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def final_layernorm_proj_kernel(
    a_ptr,           # const fp16 [R, H]   (TriMul output)
    gate_ptr,        # const fp16 [R, H]   (out_gate after sigmoid)
    w_ptr,           # const fp16 [H, D]   (to_out.weight.T)
    gamma_ptr,       # const fp32 [H]      (to_out_norm.weight)
    beta_ptr,        # const fp32 [H]      (to_out_norm.bias)
    out_ptr,         # mutable fp32 [R, D]
    total_rows, hidden_dim, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ compute mean / variance over the hidden dimension
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        ptr = a_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ for each output‑dim block compute (LN → gate → linear)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        out_acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            a_vals   = tl.load(a_ptr   + offs_m[:, None] * hidden_dim + offs_h[None, :],
                               mask=mask_m[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            gate_vals= tl.load(gate_ptr+ offs_m[:, None] * hidden_dim + offs_h[None, :],
                               mask=mask_m[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

            # normalise the hidden vector
            norm = (a_vals - mean[:, None]) * rstd[:, None]

            # apply the second LayerNorm (γ,β) and the output‑gate
            gamma_h = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            beta_h  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            norm2 = norm * gamma_h[None, :] + beta_h[None, :]
            gated = norm2 * gate_vals                         # [BM, BH]

            # weight block for this hidden slice and current output‑dim slice
            w_block = tl.load(w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)          # [BH, BD]

            out_acc += tl.dot(gated, w_block)                 # [BM, BD]

        # write the final result for this output‑dim slice
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_acc,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor, shape [B, N, N] (optional)
        weights_dict : model parameters (LayerNorms, Linear weights)
        config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 0️⃣ pre‑process: fold LN γ into linear weights and fold β into biases
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]       # γ  shape [dim]
    norm_bias   = weights["norm.bias"]         # β  shape [dim]

    # helper to create (weight * γ).T  and bias = weight·β
    def prepare_linear(w_name):
        w = weights[w_name]                     # [H, dim] float32
        w_fp16 = w.to(torch.float16)
        w_scaled_T = (w_fp16 * norm_weight).t().contiguous()   # [dim, H]
        b = (w.float() * norm_bias).sum(dim=1)                # [H] float32
        return w_scaled_T, b

    w_left_T,    b_left    = prepare_linear("left_proj.weight")
    w_right_T,   b_right   = prepare_linear("right_proj.weight")
    w_lgate_T,   b_lgate   = prepare_linear("left_gate.weight")
    w_rgate_T,   b_rgate   = prepare_linear("right_gate.weight")
    w_outgate_T, b_outgate = prepare_linear("out_gate.weight")

    # ------------------------------------------------------------------
    # 1️⃣ fused LayerNorm + four projections + gating
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, D] fp32

    left_gated  = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    right_gated = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    out_gate    = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    BLOCK_M = 64            # rows per program
    BLOCK_D = 64            # feature‑dim tile
    BLOCK_H = 32            # hidden‑dim tile

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    proj_ln_gate_fused[grid](
        x_flat,
        left_gated,
        right_gated,
        out_gate,
        w_left_T,
        w_right_T,
        w_lgate_T,
        w_rgate_T,
        w_outgate_T,
        b_left,
        b_right,
        b_lgate,
        b_rgate,
        b_outgate,
        total_rows,
        dim,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ optional mask (applied after gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_gated  = left_gated * mask_flat
        right_gated = right_gated * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left_gated .view(B, N, N, H)          # [B, N, N, H] fp16
    right = right_gated.view(B, N, N, H)          # [B, N, N, H] fp16

    # bring hidden dim into the batch axis
    left_perm  = left .permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left_perm .view(B * H, N, N)               # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)               # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16
    out_flat = out.view(total_rows, H)                               # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ fused second LayerNorm + out‑gate + final projection
    # ------------------------------------------------------------------
    # final linear weight (to_out): shape [dim, H] → transpose to [H, dim]
    w_to_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()

    # allocate the final output tensor (float32)
    final_out = torch.empty(total_rows, dim, dtype=torch.float32, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    final_layernorm_proj_kernel[grid2](
        out_flat,                     # a_ptr
        out_gate,                     # gate_ptr (already sigmoid‑ed)
        w_to_out_T,                   # weight ptr [H, D]
        weights["to_out_norm.weight"],   # γ (fp32) [H]
        weights["to_out_norm.bias"],     # β (fp32) [H]
        final_out,                    # output ptr
        total_rows,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 5️⃣ reshape to the original 4‑D layout and return
    # ------------------------------------------------------------------
    final_out = final_out.view(B, N, N, dim)   # [B, N, N, dim] float32
    return final_out
