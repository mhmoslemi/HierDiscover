"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input (B·N·N × dim) using a custom
    Triton kernel (pure FP32 – identical to nn.LayerNorm).

2️⃣  Fuse the four linear layers (left/right projections + their gates) into a
    single GEMM by concatenating the weight matrices:
        X_norm @ W_proj_gateᵀ → [R, 4·H]   (R = B·N·N, H = hidden_dim)

3️⃣  Compute the output‑gate with a second GEMM.

4️⃣  Apply the mask (if present), compute sigmoid gates and multiply them
    with the projected tensors.

5️⃣  Reshape the tensors to (B, H, N, N) and perform the N³ multiplicative
    update with a batched matrix multiplication:
        out[b, h] = left[b, h] @ right[b, h]ᵀ

6️⃣  Second LayerNorm over the hidden dimension (again via Triton).

7️⃣  Apply the output‑gate, a final linear projection back to `dim`,
    and return a FP32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float32)
    weight_ptr,     # const   * [dim]               (float32) gamma
    bias_ptr,       # const   * [dim]               (float32) beta
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # load a tile of shape [BLOCK_M, BLOCK_D]
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)                    # ∑_d x
        sum_sq  += tl.sum(vals * vals, axis=1)             # ∑_d x²

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                              # 1 / √(var+ε)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized output y = (x-mean)*rstd * gamma + beta
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # [BLOCK_D]
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # [BLOCK_D]

        out = norm * w[None, :] + b[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out,
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 128,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim] (float32)
    → returns a float32 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty_like(x, dtype=torch.float32, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N  # R

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP32
    # --------------------------------------------------------------
    # flatten to [R, dim] (float32) and normalize
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma (float32)
        weights["norm.bias"],     # beta  (float32)
        eps=eps,
    )   # shape [R, dim], dtype float32

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection / gate weight matrices
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    )  # shape [4·H, dim], dtype float32

    # single GEMM: X_norm @ W_proj_gateᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, w_proj_gate)   # fp32

    # split the concatenated result
    left_proj_raw   = proj_gate[:, :H]                 # [R, H]
    right_proj_raw  = proj_gate[:, H:2 * H]            # [R, H]
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]        # [R, H]
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]        # [R, H]

    # output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"],   # [H, dim]
    )                                 # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gating, element‑wise gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] (float32) broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float32).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid for gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # apply gates
    left = left_proj_raw * left_gate       # [R, H]
    right = right_proj_raw * right_gate    # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # move hidden dimension to axis 1 → shape (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_bh = torch.matmul(left_bh,
                          right_bh.transpose(-2, -1))   # [B, H, N, N]

    # back to [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) → FP32
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        eps=eps,
    )                                    # [R, H] fp32
    out_norm = out_norm.view(B, N, N, H)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Output gate & final projection back to dim
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # [B,N,N,H]
    final = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"],                # [dim, H]
    )                                            # [R, dim]

    final = final.view(B, N, N, dim).to(torch.float32)
    return final
