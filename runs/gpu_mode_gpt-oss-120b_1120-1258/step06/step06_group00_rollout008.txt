"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (B·N·N × dim) – custom
   Triton kernel that works in fp32 for statistics and writes fp16.

2️⃣ Fuse the four linear projections (left, right) and their two gates
   together with the output‑gate linear layer into a *single* fp16 GEMM:
   (R, dim) @ (5·H, dim).T → (R, 5·H).  The result is sliced into
   left_proj, right_proj, left_gate, right_gate and out_gate.

3️⃣ (Optional) Apply the pair‑wise mask to left/right before gating.
   Apply sigmoid in‑place to the three gate tensors and multiply
   left/right by their gates (in‑place).

4️⃣ Reshape to (B, H, N, N) and perform the N³ multiplicative update
   with a single batched gemm:
       out = left @ rightᵀ          # shape (B, H, N, N)
   No extra copies – we use `torch.matmul` on 4‑D tensors.

5️⃣ Second LayerNorm over the hidden dimension (H) – use the highly‑
   tuned cuDNN/torch implementation (`F.layer_norm`).  This replaces
   a second Triton kernel and is faster on H100.

6️⃣ Multiply by the output‑gate, then a final linear projection back
   to `dim`.  All linear layers run in fp16; the result is cast to
   fp32 for the model output.

The Triton kernel is the only required custom kernel – the rest
leverages PyTorch’s fast kernels and Tensor‑Core GEMMs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 statistics → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    n_iter = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, n_iter):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # flat pointer arithmetic: row * dim + col
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, n_iter):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → no mask
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total rows for the flattened tensors
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,                     # H100 – more warps give higher occupancy
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections, their gates and the out‑gate
    #    into a single fp16 GEMM: (R, dim) @ (5·H, dim).T → (R, 5·H)
    # ------------------------------------------------------------------
    # Concatenate all five weight matrices (each [H, dim]) and cast once.
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)          # shape (5·H, dim)

    # One fused linear → (R, 5·H) in fp16
    fused = F.linear(x_norm_fp16, w_fused)            # fp16

    # Slice the fused result
    left_raw   = fused[:, 0:H]
    right_raw  = fused[:, H:2*H]
    left_gate_raw = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]
    out_gate_raw = fused[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any), sigmoid gates, and combine
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_h = mask.to(dtype=torch.float16, device=device).reshape(R, 1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # In‑place sigmoid to avoid extra temporaries
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates – in‑place to reuse memory
    left_raw.mul_(left_gate)    # left = left_raw after this line
    right_raw.mul_(right_gate)  # right = right_raw after this line

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update
    #    left/right: (B, N, N, H) → (B, H, N, N)
    #    out = left @ rightᵀ   (batched GEMM)
    # ------------------------------------------------------------------
    left = left_raw.view(B, N, N, H).transpose(1, 3)   # (B, H, N, N)
    right = right_raw.view(B, N, N, H).transpose(1, 3) # (B, H, N, N)

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ
    out = torch.matmul(left, right.transpose(-2, -1))  # (B, H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out.transpose(1, 3)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension H
    #    Use the highly‑optimized torch implementation.
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype=torch.float16, device=device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(dtype=torch.float16, device=device)

    out_norm = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # still fp16

    # Apply the output gate (reshape to broadcast)
    out_gate = out_gate.view(B, N, N, H)
    out_norm = out_norm * out_gate   # element‑wise fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(dtype=torch.float16, device=device)  # [dim, H]
    final = F.linear(out_norm.view(R, H), to_out_weight)   # (R, dim) fp16
    final = final.view(B, N, N, dim).to(dtype=torch.float32)  # cast to fp32

    return final
