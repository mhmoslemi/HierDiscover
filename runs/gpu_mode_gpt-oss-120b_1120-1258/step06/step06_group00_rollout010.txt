```
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute block start indices
    offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load left tile: [BLOCK_M, BLOCK_K]
        left_ptrs = left_ptr + batch_idx * stride_left_batch + \
                    offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile: right shape [batch, K, N]; we need tile [BLOCK_K, BLOCK_N]
        right_ptrs = right_ptr + batch_idx * stride_right_batch + \
                    offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product across k dimension: left (M x K) dot right (K x N)
        acc += tl.dot(left, right)

    # write back
    out_ptrs = out_ptr + batch_idx * stride_out_batch + \
               offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
Highly‑optimized Triton implementation of the outgoing TriMul operator
(AlphaFold‑3).  The forward pass consists of:

1️⃣  Row‑wise LayerNorm on the input (FP32 → FP16) – custom Triton kernel.
2️⃣  Fuse the four projection/gate linear layers into a single GEMM
    (R = B·N·N rows, dim → 5·H columns) and split the result.
3️⃣  Optional masking, σ(sigmoid) on the three gate tensors and elementwise
    gating of the left/right tensors.
4️⃣  Batched matrix‑multiply `left @ rightᵀ` for the N³ multiplicative update.
    Implemented as a custom Triton GEMM kernel that works on the
    (B·H, N, N) tensors in FP16 (accumulate in FP32 → Tensor‑Core friendly).
5️⃣  Second LayerNorm over the hidden dimension (FP32 → FP16) – same kernel.
6️⃣  Elementwise multiplication with the output‑gate and a final linear
    projection back to the original feature dimension.

All heavy normalisation steps are done in Triton, the large matrix
multiplications (steps 2 and 4) use fused FP16 GEMM kernels that fully
exploit H100 Tensor‑Cores.  The function returns a float‑32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, num_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, num_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ multiplicative update
# Computes out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
# left  : [B·H, M, K]   (fp16)
# right : [B·H, K, N]   (fp16)  ← transposed version of the original right tensor
# out   : [B·H, M, N]   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,   # const   * [batch, M, K]   (fp16)
    right_ptr,  # const   * [batch, K, N]   (fp16)
    out_ptr,    # mutable * [batch, M, N]   (fp16)

    batch,      # int (runtime)
    M,          # int (runtime)
    N,          # int (runtime)
    K,          # int (runtime)

    stride_l_batch, stride_l_m, stride_l_k,   # i64
    stride_r_batch, stride_r_k, stride_r_n,   # i64
    stride_o_batch, stride_o_m, stride_o_n,   # i64

    BLOCK_M: tl.constexpr,   # rows per block
    BLOCK_N: tl.constexpr,   # cols per block
    BLOCK_K: tl.constexpr,   # reduction tile
):
    b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)      # tile row index
    pid_j = tl.program_id(2)      # tile col index

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in float32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Reduction over K dimension (loop over tiles)
    # --------------------------------------------------------------
    for k_base in range(0, K, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: [BLOCK_M, BLOCK_K]
        ptrs_l = left_ptr + \
                 b * stride_l_batch + \
                 offs_m[:, None] * stride_l_m + \
                 offs_k[None, :] * stride_l_k
        a = tl.load(ptrs_l,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                 # fp16

        # right tile (already transposed): [BLOCK_K, BLOCK_N]
        ptrs_r = right_ptr + \
                 b * stride_r_batch + \
                 offs_k[:, None] * stride_r_k + \
                 offs_n[None, :] * stride_r_n
        b_tile = tl.load(ptrs_r,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)               # fp16

        # dot product (fp16·fp16 → fp32 accumulation)
        acc += tl.dot(a, b_tile)

    # --------------------------------------------------------------
    # Write result back (fp16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
               b * stride_o_batch + \
               offs_m[:, None] * stride_o_m + \
               offs_n[None, :] * stride_o_n
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)   # True → ignore mask
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse all projection / gate linear layers (5 * hidden_dim outputs)
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    # One fused GEMM: [R, dim] @ [5·H, dim].T → [R, 5·H]
    fused_all = F.linear(x_norm_fp16, w_fused)          # fp16, no bias

    # Split the fused result
    (left_raw,
     right_raw,
     left_gate_raw,
     right_gate_raw,
     out_gate_raw) = torch.split(fused_all, H, dim=1)

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoids and gating
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → reshape to [R, 1] (fp16)
        mask_flat = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid for the three gate tensors
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates to the projected tensors
    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # From [R, H] → [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move head dimension into the batch axis → [B·H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)               # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)              # [B·H, N, N]

    # Transpose the right matrix once (K‑N swap) → [B·H, N, N]
    right_mat_t = right_mat.transpose(1, 2).contiguous()  # [B·H, N, N] (K,N)

    # Output buffer for the batched GEMM
    out_mat = torch.empty_like(left_mat)   # fp16, shape [B·H, N, N]

    # --------------------------------------------------------------
    # 4️⃣ Triton batched GEMM (the N³ update)
    # --------------------------------------------------------------
    BLOCK_M_GEMM = 128
    BLOCK_N_GEMM = 128
    BLOCK_K_GEMM = 32

    grid = (
        B * H,                                   # batch dimension
        triton.cdiv(N, BLOCK_M_GEMM),            # i‑tiles
        triton.cdiv(N, BLOCK_N_GEMM),            # j‑tiles
    )

    tri_mul_gemm_kernel[grid](
        left_mat,
        right_mat_t,
        out_mat,
        B * H,                         # batch
        N,                             # M
        N,                             # N
        N,                             # K (same as M & N)
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat_t.stride(0), right_mat_t.stride(1), right_mat_t.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M_GEMM,
        BLOCK_N=BLOCK_N_GEMM,
        BLOCK_K=BLOCK_K_GEMM,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)          # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                       # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)     # [dim, H]
    )                                                   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
