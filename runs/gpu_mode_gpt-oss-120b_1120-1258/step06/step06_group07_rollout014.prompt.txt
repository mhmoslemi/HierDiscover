import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – single Triton kernel.
2️⃣ Two fused linear‑+‑gate passes (FP16):
   * left  = (norm(x)·γ + β)·W_left_proj  ⊙  σ((norm(x)·γ + β)·W_left_gate)
   * right = (norm(x)·γ + β)·W_right_proj ⊙  σ((norm(x)·γ + β)·W_right_gate)
   The two passes share the normalised tensor, avoiding
   additional memory traffic.
3️⃣ Output‑gate = σ((norm(x)·γ + β)·W_out_gate)
4️⃣ Optional binary mask (broadcasted on the hidden dimension).
5️⃣ Triangular multiplicative update – batched GEMM:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with `torch.bmm` on half‑precision tensors (Tensor‑Core).
6️⃣ Fused LayerNorm + output‑gate (FP16) – a Triton kernel that
   normalises, applies γ/β, multiplies by the previously computed
   `out_gate`, and writes the result in half‑precision.
7️⃣ Final linear projection (hidden → dim) – ordinary FP16 GEMM,
   cast back to float32.

Only the LayerNorms are written in Triton; the heavy matrix‑multiplications
use cuBLAS/Tensor‑Core which is already optimal.  This keeps the kernel
launch count low while still demonstrating custom‑kernel fusion.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*   [total_rows, dim]   (FP32)
    y_ptr,               # *mut   half*    [total_rows, dim]   (FP16)
    weight_ptr,          # *const float*   [dim]               (γ)
    bias_ptr,            # *const float*   [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, hidden] (post‑norm⋅gate)
    weight_ptr,          # *const float* [hidden] (γ)
    bias_ptr,            # *const float* [hidden] (β)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate, already sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]

    # ---------------------------------------------------------------

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                     # C
    hidden = cfg["hidden_dim"]              # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                      # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)                # [R, C] half

    # Block sizes – chosen to fit the modest channel dimension while keeping
    # enough rows per program to amortise launch overhead.
    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 256 else 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear + gate passes (left / right) – fully fused in a single GEMM
    # ------------------------------------------------------------------
    # Cast all weight matrices to FP16 once (half‑precision GEMM)
    left_proj_w  = weights["left_proj.weight"].half().contiguous()
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # ------------------------------------------------------------------
    # Left side
    # ------------------------------------------------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, C)
    left_comb   = F.linear(x_norm_fp16, left_comb_w)           # (R, 2*H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)                   # (R, H)
    left = left_proj_raw * left_gate                           # (R, H)

    # ------------------------------------------------------------------
    # Right side
    # ------------------------------------------------------------------
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)
    right_comb   = F.linear(x_norm_fp16, right_comb_w)            # (R, 2*H)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)                    # (R, H)
    right = right_proj_raw * right_gate                          # (R, H)

    # ------------------------------------------------------------------
    # Output gate (layer‑norm → linear → sigmoid)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)           # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)                     # (R, H) – half‑precision is fine

    # ------------------------------------------------------------------
    # 2a) Apply binary mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (R,1) as half, broadcast on hidden dim
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, hidden)      # (B, N, N, H)
    right = right.view(B, N, N, hidden)     # (B, N, N, H)

    # Prepare tensors for the batched matmul: (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    # Batched half‑precision matmul → (B*H, N, N) half
    out_mat = torch.bmm(left_mat, right_mat)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    # Re‑use the same block configuration (fits hidden ≤ 384)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                           # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()            # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)        # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
