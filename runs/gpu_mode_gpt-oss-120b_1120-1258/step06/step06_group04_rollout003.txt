"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated.

The implementation follows the reference PyTorch module but fuses many
operations:

1️⃣  Row‑wise LayerNorm on the 4‑D pair representation (FP32 → FP16) with a
    custom Triton kernel.

2️⃣  A *single* fused linear projection produces the left/right
    projections, their gates, **and** the output gate in one GEMM:
        (dim → 5·hidden) = [left_proj, left_gate, right_proj,
                            right_gate, out_gate].

3️⃣  Optional mask is applied to the left/right projections, then the
    sigmoid gates are applied.

4️⃣  The tri‑multiplicative update
        out_h(i,j) = Σ_k left_h(i,k)·right_h(j,k)
    is a batched GEMM over the sequence dimension.  We flatten the
    hidden dimension into the batch axis and use a single
    `torch.bmm` (mapped to a strided‑batched cuBLAS call).

5️⃣  A second LayerNorm (over `hidden_dim`) with the same Triton kernel.

6️⃣  Multiply by the output gate and a final linear projection back to
    `dim`.  The final tensor is returned in FP32.

All heavy arithmetic is performed in FP16, the only FP32 work is the
LayerNorm statistics.  At least one custom kernel (LayerNorm) is used,
as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------- Pass 1: compute mean & variance -------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var+eps)

    # ----------- Pass 2: write normalized (γ·x̂+β) ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be ignored)
        weights_dict : mapping with keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        config_dict  : {"dim": int,
                       "hidden_dim": int,
                       "nomask": bool (optional, default True)}

    Returns:
        torch.Tensor[float32] shape [B, N, N, dim]
    """
    # ----------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # ----------------------------------------------------------
    # 0️⃣ Cast/prepare weights once (FP16)
    # ----------------------------------------------------------
    # NOTE: all weight tensors are stored as (out_features, in_features)
    # and have no bias (biases are separate entries where present).
    norm_weight = weights["norm.weight"].to(dtype=torch.float16, device=device)
    norm_bias   = weights["norm.bias"].to(dtype=torch.float16, device=device)

    left_proj_w  = weights["left_proj.weight"].to(dtype=torch.float16, device=device)
    left_gate_w  = weights["left_gate.weight"].to(dtype=torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(dtype=torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(dtype=torch.float16, device=device)
    out_gate_w   = weights["out_gate.weight"].to(dtype=torch.float16, device=device)

    to_out_norm_w = weights["to_out_norm.weight"].to(dtype=torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype=torch.float16, device=device)

    to_out_w = weights["to_out.weight"].to(dtype=torch.float16, device=device)

    # ----------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ----------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Heuristic block sizes (dim ≤ 384 → 3 blocks max)
    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        norm_weight,
        norm_bias,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # Shape back to [B, N, N, dim] (still FP16)
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # ----------------------------------------------------------
    # 2️⃣ Fused projection + gates + out_gate (single GEMM)
    # ----------------------------------------------------------
    # Concatenate the five weight matrices: left_proj, left_gate,
    # right_proj, right_gate, out_gate → shape (5·hidden, dim)
    combined_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )
    # Linear: (rows, dim) @ (5·hidden, dim).T → (rows, 5·hidden)
    rows = total_rows
    x_norm_flat = x_norm_fp16.view(rows, dim)  # already FP16
    combined_out = F.linear(x_norm_flat, combined_weight)  # FP16

    # Slice the concatenated output
    h = hidden_dim
    left_proj    = combined_out[:, 0:h]
    left_gate_pre = combined_out[:, h:2 * h]
    right_proj   = combined_out[:, 2 * h:3 * h]
    right_gate_pre = combined_out[:, 3 * h:4 * h]
    out_gate_pre = combined_out[:, 4 * h:5 * h]

    # ----------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ----------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ----------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ----------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)   # shape (rows, hidden)

    left_h = left_proj * left_gate   # (rows, hidden)
    right_h = right_proj * right_gate

    # ----------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # ----------------------------------------------------------
    # Reshape to [B, N, N, hidden]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Merge hidden into batch dimension → [B·hidden, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM: left_mat @ right_matᵀ → (B·hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16

    # Reshape back to [B, N, N, hidden]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16

    # ----------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (FP16)
    # ----------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        to_out_norm_w,
        to_out_norm_b,
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # FP16

    # ----------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # ----------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim)   # FP16
    gated = out_norm * out_gate                     # FP16

    # Final linear projection (hidden → dim)
    gated_flat = gated.view(rows_h, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_w)    # FP16
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
