"""
custom_kernel
==============

Forward pass of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm (high‑level)
----------------------
1️⃣ Row‑wise LayerNorm on the flattened 4‑D input (`[B, N, N, C]`) → FP16.
2️⃣ Fuse the four linear projections (left/right + their gates) into one
   half‑precision GEMM, then split the result.
3️⃣ Compute the output‑gate with a second half‑precision GEMM.
4️⃣ Apply the optional mask, sigmoid‑gating and element‑wise multiplication
   to obtain the gated tensors **left** and **right**.
5️⃣ Perform the N³ multiplicative update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix‑multiply ``(B·H, N, N) @ (B·H, N, N)ᵀ`` which
   leverages cuBLAS Tensor‑Core kernels.
6️⃣ Second LayerNorm on the hidden dimension (FP16).
7️⃣ Apply the output‑gate and a final linear projection back to ``dim``.
   The result is cast to FP32 (the API of the reference PyTorch module).

Only the row‑wise LayerNorms are implemented in Triton; all large dense
matrix multiplications use PyTorch’s highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (fp32)
    y_ptr,          # mutable * [rows, dim] (fp16)
    weight_ptr,     # const   * [dim]       (fp32)  gamma
    bias_ptr,       # const   * [dim]       (fp32)  beta
    rows,           # i32 total rows
    dim,            # i32 hidden dimension
    eps,            # fp32 epsilon (compile‑time const)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                      # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized result: (x-μ)/σ * γ + β (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    block_m: int = 256,
    block_d: int = 32,
) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim]
    (float32) → half‑precision tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of all linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N  # R = B·N·N

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma  (FP32)
        weights["norm.bias"],     # beta   (FP32)
        eps=eps,
    )  # -> FP16, shape [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim

    # concatenate weight matrices for left/right proj + their gates
    # each weight: [H, dim] (FP32) → cast to FP16 once
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [4·H, dim]

    # single GEMM: (R, dim) @ (dim, 4·H).T → (R, 4·H)
    proj_gate = F.linear(x_norm, w_proj_gate)       # FP16

    # split into four tensors of shape (R, H)
    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2 * H]
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]

    # output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),  # [H, dim]
    )  # (R, H), FP16

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gating, element‑wise multiplication
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (FP16) and broadcast over hidden dim
        mask_flat = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # in‑place sigmoid to stay memory‑efficient
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # apply gates
    left = left_proj_raw * left_gate      # (R, H), FP16
    right = right_proj_raw * right_gate   # (R, H), FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # shape back to 4‑D tensors
    left = left.view(B, N, N, H)           # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # move hidden dimension to second axis → (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()
    right_bh = right.permute(0, 3, 1, 2).contiguous()

    # batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ  → (B, H, N, N)
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))

    # restore original layout [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H), FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) → FP16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma (FP32)
        weights["to_out_norm.bias"],     # beta  (FP32)
        eps=eps,
    )  # (R, H) in FP16
    out_norm = out_norm.view(B, N, N, H)   # (B, N, N, H), FP16

    # --------------------------------------------------------------
    # 6️⃣ Output gate & final linear projection back to dim
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # gating, FP16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )  # (R, dim), FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
