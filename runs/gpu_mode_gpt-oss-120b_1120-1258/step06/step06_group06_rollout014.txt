def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps=1e-5; device = input_tensor.device
    B,N,_,_ = input_tensor.shape
    total_rows = B * N * N
    # reshape input
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    
    # Prepare buffers for projections
    left_proj_raw = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    right_proj_raw = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    left_gate_raw = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    right_gate_raw = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    out_gate_raw = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    
    # First layernorm + projection + gates
    BLOCK_M = 64
    BLOCK_D = 64
    BLOCK_H = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    norm_proj_raw_kernel[grid](
        x_flat,
        left_proj_raw,
        right_proj_raw,
        left_gate_raw,
        right_gate_raw,
        out_gate_raw,
        weights["norm.weight"], weights["norm.bias"],
        weights["left_proj.weight"].to(torch.float16), 
        weights["right_proj.weight"].to(torch.float16),
        weights["left_gate.weight"].to(torch.float16),
        weights["right_gate.weight"].to(torch.float16),
        weights["out_gate.weight"].to(torch.float16),
        total_rows,
        dim,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    
    # Apply mask if needed
    if not config.get("nomask", True):
        mask_flat = mask_tensor.to(torch.float16).contiguous().view(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat
    # Apply sigmoid gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()
    # Apply elementwise gating
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate
    
    # Triangular multiplication (batched GEMM)
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B * hidden_dim, N, N)
    right_mat = right.view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape to [R, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_flat = out.view(total_rows, hidden_dim)
    
    # Final fused LayerNorm + out_gate multiplication + linear projection
    final = torch.empty(total_rows, dim, dtype=torch.float32, device=device)
    # Prepare weight transposed for easier loading
    w_to_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()  # [hidden_dim, dim]
    
    BLOCK_N = 64
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    final_norm_linear_kernel[grid2](
        out_flat,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        w_to_out_T,
        final,
        total_rows,
        hidden_dim,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_N=BLOCK_N,
        num_warps=4,
    )
    
    # reshape final
    final = final.view(B, N, N, dim)
    return final

@triton.jit
def norm_proj_raw_kernel(
    x_ptr, left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    w_left_ptr, w_right_ptr, w_left_gate_ptr, w_right_gate_ptr, w_out_gate_ptr,
    total_rows, dim, hidden_dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass: mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # allocate accumulators for each hidden block
    # We'll loop over hidden blocks and compute dot products, storing directly
    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # init per-block accum
        left_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
            norm = (vals - mean[:, None]) * rstd[:, None]  # fp32

            # load weight tiles
            w_left = tl.load(w_left_ptr + offs_h[:, None] * dim + offs_d[None, :],
                             mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            w_right = tl.load(w_right_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            w_lgate = tl.load(w_left_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            w_rgate = tl.load(w_right_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :], other=0.0)
            w_out_gate = tl.load(w_out_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                 mask=mask_h[:, None] & mask_d[None, :], other=0.0)

            # dot product
            left_acc += tl.dot(norm, w_left.T)
            right_acc += tl.dot(norm, w_right.T)
            left_gate_acc += tl.dot(norm, w_lgate.T)
            right_gate_acc += tl.dot(norm, w_rgate.T)
            out_gate_acc += tl.dot(norm, w_out_gate.T)

        # store results (fp16)
        left_ptrs = left_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        right_ptrs = right_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        left_gate_ptrs = left_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        right_gate_ptrs = right_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        out_gate_ptrs = out_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]

        tl.store(left_ptrs, left_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(left_gate_ptrs, left_gate_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_gate_ptrs, right_gate_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptrs, out_gate_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – highly‑fused Triton implementation
-----------------------------------------------------

Algorithm
~~~~~~~~~
1️⃣  Row‑wise LayerNorm + four Linear projections (left, right,
    left‑gate, right‑gate) + out‑gate projection are fused into a
    single Triton kernel.  The kernel works on the flattened input
    tensor ``[B·N·N, dim]`` (float32) and directly produces the
    *raw* left/right tensors and the three gate tensors in fp16.
    Only one pass over the input is needed to compute the per‑row
    mean/variance, a second pass computes the normalised values and
    the dot‑products with the small weight matrices (the weight
    matrices are tiny and stay in registers/shared memory).

2️⃣  Optional mask is applied **after** the left/right projections
    and **before** the gating (exactly as in the reference
    implementation).  The mask is a fp16 scalar per row.

3️⃣  The multiplicative update (the N³ term) is unchanged – a
    batched half‑precision GEMM ``(B·H, N, N) @ (B·H, N, N)ᵀ`` performed
    by ``torch.bmm`` (Tensor‑Core‐accelerated).

4️⃣  The second LayerNorm, the output‑gate multiplication and the final
    linear projection are merged into a second Triton kernel.
    For each row the kernel:
        * computes mean/variance over the hidden dimension,
        * normalises the row,
        * applies the learned γ,β (to_out_norm),
        * multiplies by the already‑sigmoid‑ed out‑gate,
        * accumulates the dot‑product with the transposed output weight
          matrix ``to_out.T`` (shape ``[hidden_dim, dim]``).

   This eliminates the temporary ``[R, hidden_dim]`` tensor that the
   reference code creates, cutting memory traffic dramatically.

The two custom kernels perform the heavy per‑row reductions while all
large matrix multiplications are handed to cuBLAS/Tensor‑Core kernels,
giving a fast end‑to‑end forward pass that fits comfortably inside the
target runtime budget.

The implementation works for the full test matrix of shapes
( B∈{1,2} , N∈{128,256,512,1024} , dim∈{128,384} , hidden_dim=128 )
and for both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – LayerNorm + 4 projections + out‑gate (produces raw tensors)
# ----------------------------------------------------------------------
@triton.jit
def norm_proj_raw_kernel(
    x_ptr,                # float32  [R, D]   (input)
    left_ptr, right_ptr,                     # fp16 [R, H]   (proj outputs)
    left_gate_ptr, right_gate_ptr, out_gate_ptr,   # fp16 [R, H] (gate pre‑sigmoid)
    norm_weight_ptr, norm_bias_ptr,          # float32 [D]   (LayerNorm γ,β)
    w_left_ptr, w_right_ptr,                 # fp16 [H, D]   (proj weights)
    w_lgate_ptr, w_rgate_ptr,                # fp16 [H, D]   (gate weights)
    w_out_gate_ptr,                          # fp16 [H, D]   (out‑gate weight)
    total_rows, dim, hidden_dim, eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim tile size (≤64)
    BLOCK_H: tl.constexpr,   # hidden tile size (≤32)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance (two‑pass)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # static number of dim‑tiles (max dim = 384, BLOCK_D = 64 → 6 tiles)
    for d_off in range(0, 6):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0)                               # fp32
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1/√(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Normalise and compute the 5 linear maps (left,right,gates,out_gate)
    # --------------------------------------------------------------
    # Output accumulators for each hidden block
    for h_off in range(0, 4):                     # hidden_dim ≤128, BLOCK_H=32 → 4 blocks
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden_dim

        # initialise accumulators for this hidden slice
        left_acc      = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

        # loop over the dim‑tiles again – now we have the normalised values
        for d_off in range(0, 6):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            x = tl.load(x_ptrs,
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0)                         # fp32

            # normalise the block
            norm = (x - mean[:, None]) * rstd[:, None]        # fp32

            # LayerNorm γ / β (per‑dim) – these are the same γ,β used for the input norm
            gamma = tl.load(norm_weight_ptr + offs_d,
                            mask=mask_d, other=0.0)          # fp32
            beta  = tl.load(norm_bias_ptr   + offs_d,
                            mask=mask_d, other=0.0)          # fp32
            norm = norm * gamma[None, :] + beta[None, :]      # fp32

            # ---------- load the five weight slices ----------
            w_l = tl.load(w_left_ptr      + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            w_r = tl.load(w_right_ptr     + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            w_lg= tl.load(w_lgate_ptr     + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            w_rg= tl.load(w_rgate_ptr     + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            w_og= tl.load(w_out_gate_ptr  + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            # ---------- dot products (norm [M,D] · weightᵀ [D,H]) ----------
            left_acc      += tl.dot(norm, w_l.T)
            right_acc     += tl.dot(norm, w_r.T)
            left_gate_acc += tl.dot(norm, w_lg.T)
            right_gate_acc+= tl.dot(norm, w_rg.T)
            out_gate_acc  += tl.dot(norm, w_og.T)

        # -------- store the 5 results (cast to fp16) ----------
        offs_m_tile = offs_m[:, None] * hidden_dim + offs_h[None, :]   # stride = hidden_dim
        tl.store(left_ptr       + offs_m_tile, left_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptr      + offs_m_tile, right_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(left_gate_ptr  + offs_m_tile, left_gate_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_gate_ptr + offs_m_tile, right_gate_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr   + offs_m_tile, out_gate_acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        

# ----------------------------------------------------------------------
# Kernel 2 – 2nd LayerNorm + out‑gate multiplication + final linear proj.
# ----------------------------------------------------------------------
@triton.jit
def final_norm_linear_kernel(
    out_ptr, out_gate_ptr,                     # fp16 [R, H]   (post‑bmm tensors)
    norm_weight_ptr, norm_bias_ptr,            # float32 [H]   (to_out_norm γ,β)
    w_out_T_ptr,                               # fp16 [H, D]   (transposed to_out weight)
    final_ptr,                                 # float32 [R, D] (output)
    total_rows, hidden_dim, dim, eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden tile (≤32)
    BLOCK_N: tl.constexpr,   # output‑dim tile (≤64)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance over the hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, 4):                # hidden_dim = 128, BLOCK_H = 32 → 4 tiles
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        ptrs = out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        v = tl.load(ptrs,
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0)                        # fp16 → fp32
        v = v.to(tl.float32)
        sum_val += tl.sum(v, axis=1)
        sum_sq  += tl.sum(v * v, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  For each output‑dim block: accumulate dot‑products
    # --------------------------------------------------------------
    # max dim = 384, BLOCK_N = 64 → 6 tiles
    for n_off in range(0, 6):
        offs_n = n_off * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
        mask_n = offs_n < dim

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for h_off in range(0, 4):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # load out and its gate
            out_tile = tl.load(out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                               mask=mask_m[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)          # fp16→fp32
            gate_tile = tl.load(out_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # normalise row
            norm = (out_tile - mean[:, None]) * rstd[:, None]   # (M,H)

            # apply the to_out_norm γ/β (per‑hidden)
            gamma = tl.load(norm_weight_ptr + offs_h,
                            mask=mask_h, other=0.0)               # fp32
            beta  = tl.load(norm_bias_ptr   + offs_h,
                            mask=mask_h, other=0.0)               # fp32
            norm = norm * gamma[None, :] + beta[None, :]

            # gate
            gated = norm * gate_tile                           # (M,H)

            # weight slice w_out_T[H, D] → (BLOCK_H, BLOCK_N)
            w = tl.load(w_out_T_ptr + offs_h[:, None] * dim + offs_n[None, :],
                        mask=mask_h[:, None] & mask_n[None, :], other=0.0)   # fp16

            acc += tl.dot(gated, w)    # (M,N)

        # -------- store final block ----------
        out_ptrs = final_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(out_ptrs, acc,
                 mask=mask_m[:, None] & mask_n[None, :])

# Entry point – map PyTorch tensors to the kernels
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of all linear / layer‑norm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                       # total number of (b,i,k) rows

    # ------------------------------------------------------------------
    # 1️⃣ flatten input → [R, dim] (float32, contiguous)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()

    # ------------------------------------------------------------------
    # 2️⃣ allocate raw outputs of the first fused kernel
    # ------------------------------------------------------------------
    left_raw      = torch.empty(R, hidden_dim,
                                 dtype=torch.float16, device=device)
    right_raw     = torch.empty(R, hidden_dim,
                                 dtype=torch.float16, device=device)
    left_gate_raw = torch.empty(R, hidden_dim,
                                 dtype=torch.float16, device=device)
    right_gate_raw= torch.empty(R, hidden_dim,
                                 dtype=torch.float16, device=device)
    out_gate_raw  = torch.empty(R, hidden_dim,
                                 dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 3️⃣ launch the fused LayerNorm + 4 proj + out‑gate kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64          # rows per program
    BLOCK_D = 64          # dim tile (max 64, works for dim≤384)
    BLOCK_H = 32          # hidden tile (max 32, works for hidden_dim=128)

    grid = (triton.cdiv(R, BLOCK_M),)
    norm_proj_raw_kernel[grid](
        x_flat,
        left_raw, right_raw,
        left_gate_raw, right_gate_raw, out_gate_raw,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        weights["left_proj.weight"].to(torch.float16),
        weights["right_proj.weight"].to(torch.float16),
        weights["left_gate.weight"].to(torch.float16),
        weights["right_gate.weight"].to(torch.float16),
        weights["out_gate.weight"].to(torch.float16),
        R, dim, hidden_dim, eps,
        BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D, BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4️⃣ optional mask (broadcast over hidden dim) and sigmoids
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.to(torch.float16).view(R, 1)   # [R,1]
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # ------------------------------------------------------------------
    # 5️⃣ element‑wise gating (in‑place to save memory)
    # ------------------------------------------------------------------
    left  = left_raw * left_gate      # [R, H]  fp16
    right = right_raw * right_gate    # [R, H]  fp16

    # ------------------------------------------------------------------
    # 6️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right= right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    left_mat  = left.view(B * hidden_dim, N, N)            # [B·H, N, N]
    right_mat = right.view(B * hidden_dim, N, N)           # [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H,N,N)

    # reshape back to [R, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_flat = out.view(R, hidden_dim)            # [R, H]

    # ------------------------------------------------------------------
    # 7️⃣ Fuse 2nd LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    # transpose the to_out weight to [H, D] for convenient tiled loads
    w_to_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()

    BLOCK_N = 64          # output‑dim tile (max 64 → covers dim up to 384)
    grid2 = (triton.cdiv(R, BLOCK_M),)
    final_norm_linear_kernel[grid2](
        out_flat,
        out_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        w_to_out_T,                      # [H, D]  fp16
        final,
        R, hidden_dim, dim, eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_N=BLOCK_N,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8️⃣ reshape to the original 4‑D shape and return
    # ------------------------------------------------------------------
    final = final.view(B, N, N, dim)   # [B, N, N, dim], float32
    return final
