@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_z // num_h_blocks
    h_start = (pid_z % num_h_blocks) * BLOCK_H

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    offs_h = h_start + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k blocks
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_start * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (I, K, H)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed (K, J, H)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_k[:, None, None] * stride_right_k + \
            offs_j[None, :, None] * stride_right_j + \
            offs_h[None, None, :] * stride_right_h

        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                        other=0.0).to(tl.float32)

        # accumulate dot products per hidden slice
        for h_idx in range(0, BLOCK_H):
            # slice (I, K)
            left_slice = left[:, :, h_idx]   # shape (BLOCK_I, BLOCK_K)
            # slice (K, J)
            right_slice = right[:, :, h_idx] # shape (BLOCK_K, BLOCK_J)
            acc[:, :, h_idx] += tl.dot(left_slice, right_slice)   # (BLOCK_I, BLOCK_J)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h

    # cast to fp16 before store
    out_fp16 = acc.to(tl.float16)
    tl.store(out_ptrs,
             out_fp16,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

    grid2 = (triton.cdiv(R, BLOCK_M),)

        out_flat,

        to_out_norm_b,
        R,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)

    # apply out gate
    out_norm = out_norm * out_gate  # broadcast (B,N,N,H)*(B,N,N,H)

    # final linear projection
    final = F.linear(out_norm.reshape(R, hidden_dim), to_out_w)  # fp16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

* Row‑wise LayerNorm is implemented in pure Triton (fp32 → fp16) and
  reused for the two normalisation steps.
* The cubic‑time multiplicative update is reduced to a batched
  half‑precision GEMM by reshaping the tensors and calling
  ``torch.bmm`` (cuBLAS / Tensor‑cores).  This is significantly faster
  than the reference ``torch.einsum`` while preserving numerical
  correctness.
* All heavy work after the first normalisation is carried out in
  fp16; the final result is cast back to fp32 to match the original
  implementation.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [rows, dim]   (float32)
    y_ptr,               # mutable * [rows, dim]   (float16)
    weight_ptr,          # const   * [dim]        (float32)  γ
    bias_ptr,            # const   * [dim]        (float32)  β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance in fp32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Write normalised values (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool or float)
        - weights_dict : model parameters (see keys in skeleton)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm → fp16 (row‑wise)
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] float32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 128                # works for dim ≤ 384
    grid = (math.ceil(total_rows / BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ  (float32)
        weights["norm.bias"],            # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ fused projection + gates (single GEMM)
    # ------------------------------------------------------------------
    #   concat left_proj, right_proj, left_gate, right_gate → [4·H, dim]
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [4*H, dim]

    proj_gate = F.linear(x_norm, w_4h)         # [R, 4*H] fp16

    left_proj_raw  = proj_gate[:, 0:hidden_dim]
    right_proj_raw = proj_gate[:, hidden_dim:2 * hidden_dim]
    left_gate_raw  = proj_gate[:, 2 * hidden_dim:3 * hidden_dim]
    right_gate_raw = proj_gate[:, 3 * hidden_dim:4 * hidden_dim]

    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                           # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [R,1] (broadcast over hidden dim)
        mask = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask
        right_proj_raw = right_proj_raw * mask

    left_gate = left_gate_raw.sigmoid_()      # in‑place sigmoid (fp16)
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate           # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ triangular multiplicative update (batched GEMM)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    #   (B,H,N,N) layout required for a regular batched GEMM
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_bh = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N) – swap last two axes

    # flatten batch & head dimensions → (B·H, N, N)
    left_mat = left_bh.view(B * hidden_dim, N, N)
    right_mat = right_bh.view(B * hidden_dim, N, N)

    # batched matrix multiply (Tensor‑cores, fp16)
    out_mat = torch.bmm(left_mat, right_mat)          # (B·H, N, N) fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6️⃣ second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, hidden_dim).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (math.ceil(total_rows / BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)      # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # [B,N,N,H] fp16

    final = F.linear(
        out_norm.reshape(total_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16)     # [dim, H]
    )                                                   # [R, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)  # cast to fp32

    return final
