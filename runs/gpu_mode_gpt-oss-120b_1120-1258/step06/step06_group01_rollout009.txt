"""
custom_kernel
==============

Forward pass of the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input `[B, N, N, C]` → fp16.  
   Implemented with a tiny Triton kernel that reduces per row in fp32
   and writes the normalized result in fp16.

2️⃣ Fuse the four linear layers (`left_proj`, `right_proj`,
   `left_gate`, `right_gate`) into one GEMM:
   ```
   [R, C] @ [4·H, C]ᵀ → [R, 4·H]         R = B·N·N
   ```
   Afterwards the tensor is split to obtain the projected tensors and
   their gates.  The output‑gate (`out_gate`) is computed with a separate
   linear layer.

3️⃣ Optional mask (applied *after* the two projections, before gating).

4️⃣ Apply the three sigmoid gates (in‑place, half‑precision) and obtain
   the gated tensors `left` and `right`.

5️⃣ **Multiplicative update** – compute the N³ term.
   The gated tensors are reshaped to `[B·H, N, N]` and a batched
   half‑precision GEMM (`torch.bmm`) evaluates

        out_mat = left_mat @ right_matᵀ      # shape [B·H, N, N]

   This call uses cuBLAS Tensor‑Cores on H100 and is the dominant cost.

6️⃣ Second row‑wise LayerNorm over the hidden dimension (`H`) → fp16
   (same Triton kernel as in step 1).

7️⃣ Apply the output‑gate, then a final linear projection back to `C`
   (`dim`).  The projection is a regular FP16 GEMM, after which the
   result is cast to `float32`.

All heavy linear algebra (steps 2, 5, 7) is performed with cuBLAS,
which is optimal on modern GPUs.  Two Triton kernels (steps 1 and 6)
handle the LayerNorms, guaranteeing at least one custom kernel
as required.

The implementation works for all configurations listed in the
statement, supports optional masking, and returns a `float32`
tensor of shape `[B, N, N, C]`.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32, output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]  (float32)
    y_ptr,          # mutable * [total_rows, dim]  (float16)
    w_ptr,          # const   * [dim]              (float32) γ
    b_ptr,          # const   * [dim]              (float32) β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The reduction (mean / variance) uses fp32, the final result is fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)            # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized result γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    ----------

        - config_dict  : must contain "dim" and "hidden_dim"; optional "nomask"

    """

    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()   # [R, C]  fp32
    x_norm_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Triton launch parameters – tuned for H100
    BLOCK_M = 512                     # rows per program
    BLOCK_D = 64                      # columns per tile
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],          # γ (fp32)
        weights["norm.bias"],            # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # 2️⃣ Fuse left/right projections + gate projections
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection weight matrices (no bias)
    # Shape: [4·H, C]  (fp16)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Fused GEMM:  [R, C] @ [4·H, C]ᵀ → [R, 4·H]   (fp16)
    proj_gate = F.linear(x_norm_half, w_proj_gate)   # fp16

    left_proj_raw  = proj_gate[:, 0:H]                # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]            # [R, H]

    # Output‑gate linear (separate weight)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # [H, C]

    # --------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):

        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H]  fp16
    right = right_proj_raw * right_gate        # [R, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ update
    # --------------------------------------------------------------
    # [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Bring head dimension into the batch axis (no extra copy)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Merge batch and head dimensions → batched GEMM inputs
    left_mat  = left_perm.view(B * H, N, N)    # [B·H, N, N]   fp16
    right_mat = right_perm.view(B * H, N, N)   # [B·H, N, N]   fp16

    # --------------------------------------------------------------
    # 5️⃣ Batched half‑precision GEMM (Tensor‑Cores)
    # --------------------------------------------------------------
    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.reshape(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B, N, N, H]  fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [C, H]
    )  # → [R, C] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
