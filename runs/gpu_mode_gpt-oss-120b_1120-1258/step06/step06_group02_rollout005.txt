"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16) using a Triton kernel.
2️⃣ Fuse the four linear projections (left/right + their gates) into a single
   half‑precision GEMM:  x_norm @ W_4Hᵀ  →  [left_proj, right_proj,
   left_gate, right_gate]  (shape [R, 4·H]).
3️⃣ Compute the output‑gate projection separately.
4️⃣ (Optional) apply the pairwise mask to left/right projections.
5️⃣ Apply sigmoid to the three gate tensors and multiply them with the
   corresponding projected values.
6️⃣ Reshape left/right to [B, N, N, H] and perform the cubic update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply:  out = left @ rightᵀ for each head.
7️⃣ Row‑wise LayerNorm on the hidden‑dim output (still fp16).
8️⃣ Multiply by the output‑gate, then a final linear projection back to `dim`.
9️⃣ Cast the result back to float32 (as required by the reference implementation).

All heavy linear work stays in fp16 so Tensor Cores on an H100 are fully
utilised. Only the two LayerNorms are custom Triton kernels; everything
else uses fused PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (float32)
    y_ptr,               # mutable * [total_rows, dim]   (float16)
    weight_ptr,          # const   * [dim]               (float32)
    bias_ptr,            # const   * [dim]               (float32)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1. Compute per‑row mean / variance in fp32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2. Write normalised values (fp16) = ((x-μ)/σ) * γ + β
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : mapping of weight name → torch.Tensor (float32)
        - config_dict  : must contain
            * "dim"         – input feature dimension
            * "hidden_dim" – hidden dimension H
            * "nomask"      – True → ignore mask, False → use mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    H   = config["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(total_rows, dim).contiguous()          # [R, dim] float32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)                     # [R, dim] fp16

    BLOCK_M = 256               # rows per block (tuned for H100)
    BLOCK_D = 128               # columns per inner loop (covers up to dim=384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear layers (half‑precision GEMM)
    # ------------------------------------------------------------------
    # Concatenate the four projection matrices [left_proj; right_proj;
    # left_gate; right_gate] → shape (4*H, dim)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [4·H, dim]

    proj_gate = F.linear(x_norm, w_4h)                    # [R, 4·H] fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out‑gate projection (separate linear)
    out_gate_raw = F.linear(x_norm,
                            weights["out_gate.weight"].to(torch.float16))   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] broadcast over hidden dim
        mask_h = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to the original 4‑D layout
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # Out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # This can be done as a batched matmul: left @ rightᵀ for each head.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for both tensors
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)        # [B·H, N, N]

    # batched matmul with the transpose of right
    out_mat = torch.bmm(left_mat, right_mat.transpose(-1, -2))       # [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()               # [R, H]
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)       # [R, H]

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)          # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                # [B,N,N,H] fp16 (gated)

    # Final linear projection without bias
    final = F.linear(out_norm.reshape(total_rows, H),
                     weights["to_out.weight"].to(torch.float16))   # [R, dim] fp16

    final = final.reshape(B, N, N, dim).to(torch.float32)      # back to float32
    return final
