"""
TriMul (outgoing) – fast Triton+PyTorch implementation
=====================================================

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input  (B·N·N × dim) → fp16.
2️⃣ Fuse the four linear layers (left/right projection + left/right gate)
   into a single half‑precision GEMM:
        [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]
   where  R = B·N·N  and  H = hidden_dim.
3️⃣ Separate the result into left_proj, right_proj, left_gate, right_gate.
4️⃣ Compute the output gate with a second half‑precision GEMM.
5️⃣ (Optional) apply the pair‑wise mask to the *projected* tensors.
6️⃣ Apply sigmoid to the three gate tensors and multiply:
        left  = left_proj  * sigmoid(left_gate)
        right = right_proj * sigmoid(right_gate)
7️⃣ Tri‑multiplicative update (batched GEMM):
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a bmm on shape (B·H, N, N).
8️⃣ Row‑wise LayerNorm over the hidden dimension H → fp16.
9️⃣ Multiply by the output‑gate, apply the final linear projection
   back to `dim`, and cast to float32.

Only the two row‑wise LayerNorms are written as Triton kernels; all
large matrix multiplications use cuBLAS (through torch.nn.functional.linear
and torch.bmm), which is already optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]    (float32)
    y_ptr,          # mutable *[total_rows, dim]    (float16)
    weight_ptr,     # const   *[dim]                (float32)
    bias_ptr,       # const   *[dim]                (float32)
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β
    Output is fp16; accumulates in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------- 1️⃣ mean & variance -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)          # sum over D
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------- 2️⃣ write normalized output -----------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (M,D)

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]         # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : dict of all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack --------------------------------------------------------
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) -----------------------------
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program (can be tuned)
    BLOCK_D = 64           # cols per iteration (tuned for dim ∈ {128,384})
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates -----------------------
    # --------------------------------------------------------------
    H = hidden_dim

    # Concatenate the four weight matrices: [4·H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # -> fp16

    # Half‑precision GEMM: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, w_concat)   # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )   # fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present) & sigmoid gates -------------------------
    # --------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask: [B,N,N] → [R, 1] (fp16)
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM) -------------------
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring head dimension to batch for bmm: [B, H, N, N]
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # collapse batch & head: [B·H, N, N]
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)

    # out_mat[b·H, i, j] = Σₖ left_mat[b·H, i, k] * right_mat[b·H, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16) -------------
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate + final linear projection ----------------
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate          # [B,N,N,H] fp16

    # final projection: [R, H] @ [dim, H]ᵀ → [R, dim] (fp16)
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    # Cast back to float32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
