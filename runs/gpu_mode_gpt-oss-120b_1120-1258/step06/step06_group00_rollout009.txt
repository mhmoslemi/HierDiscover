"""
Fast "outgoing" TriMul (AlphaFold‑3) forward pass.

Algorithm Overview
------------------
1️⃣ Row‑wise LayerNorm (custom Triton) – works for float32 or float16 inputs.
   The kernel reads each element **once** from global memory and writes the
   normalized result to fp16.

2️⃣ Fuse the four linear projects (left/right + their gates) into a single
   GEMM:   X_norm @ W_fused.T   →   [R, 4·H]  (R = B·N·N).
   The output is split into left‑proj, right‑proj, left‑gate, right‑gate.

3️⃣ Compute the output‑gate with a second (small) GEMM.

4️⃣ Optional mask is applied to the two projections, then the sigmoid gates
   are applied in‑place.

5️⃣ Triangular multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with a **single** `torch.einsum`, which avoids the costly
   `permute → contiguous` copy required for a batched `bmm`.

6️⃣ Second LayerNorm (same Triton kernel) over the hidden dimension H.

7️⃣ Element‑wise multiply by the output‑gate and a final linear projection.
   All heavy linear layers use FP16 tensors to exploit Tensor‑Cores.

The only Triton kernels are the two LayerNorm passes; all large matrix
multiplications stay on cuBLAS (the fastest implementation on H100).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (supports fp32 or fp16 input)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # * const  [total_rows, dim]   (float32 or float16)
    y_ptr,            # * mutable [total_rows, dim]   (float16)
    w_ptr,            # * const  [dim]               (float32)
    b_ptr,            # * const  [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    x may be fp32 or fp16; it is cast to fp32 for the statistics.
    The output y is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance (single pass over dim)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dim in chunks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer arithmetic
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load – the pointer dtype may be fp16 or fp32 → cast to fp32
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]       # (x‑μ)/σ
        # apply learned γ,β (float32)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                  # fp32
        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the *outgoing* TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Args:
        data : tuple of (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
            - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)   # True → no mask
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of rows (pairwise positions)
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] float32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # launch Triton LN kernel
    BLOCK_M = 1024
    BLOCK_D = 128
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],        # γ  (float32)
        weights["norm.bias"],          # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate the four projection weight matrices
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                 # [4·H, dim]

    # One fused GEMM: (R, dim) @ (4·H, dim).T → (R, 4·H)
    fused = F.linear(x_norm_fp16, w_fused)             # fp16, shape [R, 4·H]

    # Split the fused result
    left_raw      = fused[:, 0*H:1*H]          # [R, H]
    right_raw     = fused[:, 1*H:2*H]          # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]          # [R, H]
    right_gate_raw= fused[:, 3*H:4*H]          # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                    # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (optional) and sigmoid gates
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw  * mask_h
        right_raw = right_raw * mask_h

    # in‑place sigmoid to avoid temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)        # [B, N, N, H]
    right = right.view(B, N, N, H)       # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)

    # Triangular multiplicative update via a single einsum (no permute copies)
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H) → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] fp16
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)       # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply the output‑gate and final projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                   # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast back to fp32
    return final
