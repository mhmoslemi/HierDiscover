```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def rowwise_layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # accumulate sum and sumsq per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized values
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N

    # ---------- First LayerNorm (row-wise) ----------
    # flatten input
    x_flat = input_tensor.view(R, dim).contiguous()
    # allocate output half
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ---------- Fuse four linear projections ----------
    H = hidden_dim
    # Concatenate the four weight matrices: left_proj, right_proj, left_gate, right_gate
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [4*H, dim]

    proj_gate = F.linear(x_norm, w_proj_gate)  # [R, 4*H]

    left_proj_raw = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out gate linear
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)  # [H, dim]
    )  # [R, H]

    # ---------- Optional mask ----------
    if not nomask:
        # mask shape [B, N, N] -> flatten to [R, 1]
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ---------- Sigmoid gates ----------
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # ---------- Apply gates ----------
    left = left_proj_raw * left_gate   # [R, H]
    right = right_proj_raw * right_gate  # [R, H]

    # ---------- Reshape to [B, N, N, H] ----------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # ---------- N^3 multiplicative update ----------
    # Use einsum to avoid costly permute+contiguous copies
    out = torch.einsum('bikd,bjkd->bijd', left, right)  # [B, N, N, H], fp16

    # ---------- Second LayerNorm ----------
    out_flat = out.reshape(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # ---------- Apply output gate ----------
    out_norm = out_norm * out_gate.view(B, N, N, H)

    # ---------- Final linear projection ----------
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)  # [dim, H]
    )  # [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```

"""
Custom Triton‑accelerated forward pass for the *outgoing* TriMul operator
(AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input  [B·N·N, dim]  (fp32 → fp16)
2️⃣ Fuse the four linear projections (left/right + their gates) into a single
   GEMM:  [R, dim] @ [4·H, dim]ᵀ → [R, 4·H] (fp16)
3️⃣ Separate the output‑gate linear layer.
4️⃣ Optional mask multiplication (broadcasted over the hidden dim) –
   then in‑place sigmoid gating.
5️⃣ Reshape the gated tensors to [B, N, N, H] and compute the N³
   multiplicative update with a single `einsum`, avoiding costly
   permute+contiguous copies:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   (executed by cuBLAS under the hood).
6️⃣ Second row‑wise LayerNorm over the hidden dimension (fp16).
7️⃣ Apply the output‑gate, project back to `dim` with a final linear layer,
   and cast the result to float32.

Only the two LayerNorm steps are written in Triton; all dense matrix
multiplications use cuBLAS via `torch.nn.functional.linear` / `torch.einsum`,
which is optimal on H100.  The implementation works for any
`dim ∈ {128,384}` and `hidden_dim = 128`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    w_ptr,          # const   * [dim]         (float32)   γ
    b_ptr,          # const   * [dim]         (float32)   β
    rows,           # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean & variance per row (fp32 reduction)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim

        # ptrs: [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)           # sum across dim
        sum_sq  += tl.sum(vals * vals, axis=1)    # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                    # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Normalise + apply γ,β → store as fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]              # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor   shape [B, N, N]    (bool/float)
        - weights_dict : dict of torch.Tensor parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for the first LN
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (single GEMM)
    # --------------------------------------------------------------
    # concatenate the four linear weight matrices → [4·H, dim]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # fp16

    proj_gate = F.linear(x_norm, w_proj_gate)   # [R, 4·H] fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out‑gate linear (separate weight)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask & sigmoid gates
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate       # [R, H] fp16
    right = right_proj_raw * right_gate     # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to [B, N, N, H] and compute the N³ update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)          # [B, N, N, H]

    # einsum implements: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('bikd,bjkd->bijd', left, right)   # fp16, shape [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # gating (fp16)

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
