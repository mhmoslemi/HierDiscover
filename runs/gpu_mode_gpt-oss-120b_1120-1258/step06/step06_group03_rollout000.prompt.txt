import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input (B·N·N × dim) using a
    custom Triton kernel (fp32 → fp16).

2️⃣  Fuse the four linear layers (left/right projections + their gates)
    into a single half‑precision GEMM:
        X_norm_half @ W_proj_gate.T → [R, 4·H]
    where R = B·N·N and H = hidden_dim.
    The result is split into left_proj, right_proj, left_gate, right_gate.

3️⃣  Compute the output‑gate with a second GEMM (still half‑precision).

4️⃣  Optional mask‑application, sigmoid gating (in‑place) and element‑wise
    multiplication to obtain the gated tensors *left* and *right*.

5️⃣  Tri‑multiplicative update: for each head h we need
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
    This is a batched half‑precision GEMM of shape (B·H, N, N) @
    (B·H, N, N)ᵀ, performed with `torch.matmul` which maps to cuBLAS
    Tensor‑Core kernels.

6️⃣  Second LayerNorm over the hidden dimension (again Triton, fp16).

7️⃣  Apply the output‑gate, a final linear projection back to `dim`
    and cast the result to fp32 (the API of the reference model).

All heavy per‑row normalisations are done in Triton, while large dense
matrix multiplications leverage cuBLAS for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)  gamma
    bias_ptr,       # const   * [dim]               (float32)  beta
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)             # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output y = (x-mean)*rstd * gamma + beta  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # beta
        out = norm * w[None, :] + b[None, :]                 # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim] (float32)
    -> returns half‑precision tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N  # R

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # flatten to [R, dim] (float32) and normalize
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_half = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma (float32)
        weights["norm.bias"],     # beta  (float32)
        eps=eps,
    )                                            # fp16, shape [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gates (4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection / gate weight matrices
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)               # [4·H, dim]

    # single GEMM: X_norm_half @ W.T → [R, 4·H]
    proj_gate = F.linear(x_norm_half, w_proj_gate)   # fp16

    # split
    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16),  # [H, dim]
    )                                                  # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gating, element‑wise gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] (fp16) broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid (no extra buffers)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # apply gates
    left = left_proj_raw * left_gate       # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # view back to 4‑D tensors
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # move head dimension to axis 1 → shape (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # batched matrix multiplication (B·H, N, N) @ (B·H, N, N)ᵀ
    # torch.matmul broadcasts the batch dimensions
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))   # [B, H, N, N]

    # back to [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_half = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        eps=eps,
    )                                    # fp16

    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Output gate & final projection back to dim
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
