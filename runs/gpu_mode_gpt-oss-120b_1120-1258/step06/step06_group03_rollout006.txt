"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input (B·N·N × dim) – custom Triton
    kernel, reading float32, writing float16.

2️⃣  Fuse the five linear layers (left/right projections + their gates +
    the output gate) into a single half‑precision GEMM:
        X_norm_half @ W_all.T  →  (B·N·N, 5·H)
    where H = hidden_dim and W_all stacks the five weight matrices.
    The result is sliced into left_proj, right_proj, left_gate,
    right_gate and out_gate.

3️⃣  Optional mask multiplication (only on the two projection tensors).

4️⃣  In‑place sigmoid on the gate tensors and element‑wise gating:
        left  = left_proj  * sigmoid(left_gate)
        right = right_proj * sigmoid(right_gate)

5️⃣  Reshape to (B, H, N, N) and perform the N³ multiplicative update
    with a single batched half‑precision GEMM:
        out = left @ rightᵀ                       # shape (B, H, N, N)

6️⃣  Second LayerNorm over the hidden dimension (again Triton, fp16).

7️⃣  Apply the output‑gate, a final linear projection back to `dim` and
    cast the result to float32 (the public API of the reference module).

All heavy per‑row normalisations are done in Triton, while the large dense
matrix multiplications leverage cuBLAS Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]         (float32) γ
    bias_ptr,       # const   * [dim]         (float32) β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_blocks_d = tl.cdiv(dim, BLOCK_D)
    for d_off in range(num_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output y = (x-mean)*rstd * γ + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(num_blocks_d):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                 # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim] (float32)
    → returns half‑precision tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : contains "dim", "hidden_dim" and "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N           # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()    # [R, dim] float32
    x_norm_half = _layernorm_triton(
        x_flat,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        eps=eps,
        block_m=256,
        block_d=64 if dim > 128 else 32,
    )                                            # [R, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (left/right proj + 2 gates + out gate)
    # ------------------------------------------------------------------
    # Stack weights: each is [H, dim] → final shape [5*H, dim]
    w_all = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                    # [5*H, dim]

    # Single GEMM: (R, dim) @ (dim, 5*H).T → (R, 5*H)
    proj_all = F.linear(x_norm_half, w_all)                 # fp16

    # Slice the concatenated result
    left_proj_raw   = proj_all[:, 0:H]                # [R, H]
    right_proj_raw  = proj_all[:, H:2*H]              # [R, H]
    left_gate_raw   = proj_all[:, 2*H:3*H]            # [R, H]
    right_gate_raw  = proj_all[:, 3*H:4*H]            # [R, H]
    out_gate_raw    = proj_all[:, 4*H:5*H]            # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gating (in‑place) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()      # [R, H] fp16
    right_gate = right_gate_raw.sigmoid_()     # [R, H] fp16
    out_gate   = out_gate_raw.sigmoid_()       # [R, H] fp16

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)               # [B, N, N, H]
    right = right.view(B, N, N, H)              # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)        # [B, N, N, H]

    # Move hidden dimension to the batch axis: (B, H, N, N)
    left_bh  = left.permute(0, 3, 1, 2).contiguous()
    right_bh = right.permute(0, 3, 1, 2).contiguous()

    # Batched half‑precision GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ
    # This uses cuBLAS Tensor‑Core kernels.
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))   # [B, H, N, N]

    # Back to original layout
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    out_norm_half = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
        block_m=256,
        block_d=64 if H > 128 else 32,
    )                                    # [R, H] fp16

    out_norm = out_norm_half.view(B, N, N, H)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # fp16

    # Cast to float32 for the API contract
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
