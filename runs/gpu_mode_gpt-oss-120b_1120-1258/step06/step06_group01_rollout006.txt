        sum_right_gate = tl.zeros(...)

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (B·N·N × dim).  
    Implemented in Triton (fp16 output, fp32 accumulation).

2️⃣  Fuse the four linear projections (left/right and their gates) into a
    single GEMM:  X @ Wᵀ → [R, 4·H]  where R = B·N·N.
    Split the result into left_proj, right_proj, left_gate, right_gate.

3️⃣  Compute the output‑gate linear layer separately.

4️⃣  Optional mask (applied before gating), sigmoid gates and element‑wise
    gating of left/right tensors.

5️⃣  Reshape tensors to (B·H, N, N) and perform the N³ multiplicative update
    with a single batched half‑precision GEMM:
        out = left @ rightᵀ        # shape (B·H, N, N)

6️⃣  Row‑wise LayerNorm over the hidden dimension (again Triton).

7️⃣  Apply the output‑gate, project back to the original `dim` and return
    a float‑32 tensor.

Only the layer‑norm steps use a custom Triton kernel; all dense linear
operations rely on PyTorch’s cuBLAS‑backed GEMM (the fastest path on H100).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP16 output, FP32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    w_ptr,          # const  * [dim]            (float32) gamma
    b_ptr,          # const  * [dim]            (float32) beta
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance (FP32 reduction)
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # load tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # fp32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized result γ·(x−μ)·rstd + β  (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor       shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor or None shape [B, N, N] (bool/float)
        - weights_dict : dict of torch.Tensors holding all layer‑norm
                         and linear parameters.
        - config_dict  : must contain "dim", "hidden_dim" and optional
                         "nomask" (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise, fp16 output)
    # --------------------------------------------------------------
    # flatten input to [R, dim] (contiguous, fp32)
    x_flat = input_tensor.view(R, dim).contiguous()
    # allocate output buffer (fp16)
    x_norm_half = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # launch Triton LN kernel
    BLOCK_M = 256                     # rows per program
    BLOCK_D = 32                      # columns per program
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],          # gamma (float32)
        weights["norm.bias"],            # beta  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection matrices → [4·H, dim] (fp16)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # fused GEMM: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H] (fp16)
    proj_gate = F.linear(x_norm_half, w_proj_gate)   # shape [R, 4·H]

    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]           # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]           # [R, H]

    # output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # shape [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    apply_mask = not config.get("nomask", True)
    if apply_mask:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid (half precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # gated tensors
    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring head dimension into batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & head → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (row‑wise, fp16)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise gating (fp16)

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # shape [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
