@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute block indices: pid = program_id(0)
    # Each program handles a (batch_hidden) * (i_block, j_block)
    # Determine batch and hidden ids: total batch_hidden = B*H
    pid = tl.program_id(0)
    total_batch = B * H
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // (num_n * total_batch)
    tmp = pid % (num_n * total_batch)
    pid_n = tmp // total_batch
    pid_batch = tmp % total_batch  # maps to batch*H id
    
    # map pid_batch to b (batch) and h (hidden)
    b = pid_batch // H
    h = pid_batch % H

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for M,N dims
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape [BLOCK_M, BLOCK_K] from left[b,i,k,h]
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block: shape [BLOCK_K, BLOCK_N] from right[b,j,k,h] (j is column)
        # Note right index: stored as [b, j, k, h] so we need i->j and k same.

@triton.jit
def fused_norm_gate_linear_kernel(
    out_ptr,  # half
    out_gate_ptr,  # half
    to_out_norm_weight_ptr, # float
    to_out_norm_bias_ptr,   # float
    to_out_weight_ptr, # float (dim*hidden)
    final_ptr, # float32
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr (maybe same as BLOCK_D).
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_R
   offs_r = row_start + tl.arange(0, BLOCK_R)
   mask_r = offs_r < total_rows

   # compute per row mean/var across hidden dim (like earlier)
   sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
   for h in range(0, tl.cdiv(hidden, BLOCK_H)):
       offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
       mask_h = offs_h < hidden
       ptrs = out_ptr + offs_r[:, None] * hidden + offs_h[None, :]
       val = tl.load(ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
       sum_val += tl.sum(val, axis=1)
       sum_sq += tl.sum(val * val, axis=1)

   mean = sum_val / hidden
   var = sum_sq / hidden - mean * mean
   rstd = tl.rsqrt(var + eps)

   # Now for each dim tile compute matmul
   for d in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim

       # accumulate result for each row in block
       acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

       for h in range(0, tl.cdiv(hidden, BLOCK_H)):
           offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
           mask_h = offs_h < hidden

           # load out, gate
           out_vals = tl.load(out_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)   # (BLOCK_R, BLOCK_H)
           gate_vals = tl.load(out_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

           # normalize
           norm = (out_vals - mean[:, None]) * rstd[:, None]
           # apply gamma, beta
           w = tl.load(to_out_norm_weight_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # (1, BLOCK_H)
           b = tl.load(to_out_norm_bias_ptr + offs_h, mask=mask_h, other=0.0)[None, :]     # (1, BLOCK_H)
           g = (norm * w + b) * gate_vals   # (BLOCK_R, BLOCK_H)

           # load to_out weight slice for this hidden block (dim x hidden)
           w_out = tl.load(to_out_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0)   # shape (BLOCK_H, BLOCK_D)
           # Multiply: g (BLOCK_R, BLOCK_H) dot w_out (BLOCK_H, BLOCK_D) -> (BLOCK_R, BLOCK_D)
           acc += tl.dot(g, w_out)

       # store final
       tl.store(final_ptr + offs_r[:, None] * dim + offs_d[None, :],
                acc,
                mask=mask_r[:, None] & mask_d[None, :])

def custom_kernel(data):
   inp, mask, weights, cfg = data
   device = inp.device
   dim = cfg["dim"]
   hidden = cfg["hidden_dim"]
   eps = cfg.get("eps", 1e-5)
   nomask = cfg.get("nomask", True)
   B,N,_,_ = inp.shape
   total_rows = B * N * N

   # input LayerNorm: produce x_norm_fp16
   input_flat = inp.view(total_rows, dim).contiguous()
   x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)
   BLOCK_M = 128
   BLOCK_D = 64 if dim <= 256 else 128
   grid = (triton.cdiv(total_rows, BLOCK_M),)
   layernorm_fp16_kernel[grid](input_flat, x_norm_fp16,
         weights["norm.weight"], weights["norm.bias"], total_rows, dim, eps,
         BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D)

   # Linear + gate for left, right
   left_proj_w = weights["left_proj.weight"].half()
   left_gate_w = weights["left_gate.weight"].half()
   right_proj_w = weights["right_proj.weight"].half()
   right_gate_w = weights["right_gate.weight"].half()
   out_gate_w = weights["out_gate.weight"].half()

   # left
   left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden, dim)
   left_comb = torch.nn.functional.linear(x_norm_fp16, left_comb_w) # (R, 2*hidden)
   left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
   left_gate = torch.sigmoid(left_gate_raw)
   left = left_proj_raw * left_gate

   # right
   right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)
   right_comb = torch.nn.functional.linear(x_norm_fp16, right_comb_w)
   right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
   right_gate = torch.sigmoid(right_gate_raw)
   right = right_proj_raw * right_gate

   # out_gate
   out_gate_raw = torch.nn.functional.linear(x_norm_fp16, out_gate_w)
   out_gate = torch.sigmoid(out_gate_raw)

   # mask
   if not nomask:
       mask_flat = mask.view(total_rows, 1).to(torch.float16)
       left = left * mask_flat
       right = right * mask_flat

   # reshape
   left = left.view(B, N, N, hidden).contiguous()
   right = right.view(B, N, N, hidden).contiguous()

   stride_left = left.stride()

"""
Highly‑optimized "outgoing" TriMul implementation (AlphaFold‑3).

Algorithm
---------
1️⃣ Input LayerNorm (FP32 → FP16) – a tiny Triton kernel.
2️⃣ Two fused Linear+Gate passes (half‑precision, cuBLAS):
   left  = (norm·W_left_proj) ⊙ σ(norm·W_left_gate)
   right = (norm·W_right_proj) ⊙ σ(norm·W_right_gate)
   (mask is applied here, if present)
3️⃣ Output‑gate = σ(norm·W_out_gate)   – half‑precision.
4️⃣ Triangular multiplicative update (`out[b,i,j,h] = Σ_k left[b,i,k,h]·right[b,j,k,h]`)
   – a custom Triton batched GEMM (no extra memory copies, tile‑wise Tensor‑Core).
5️⃣ Fused LayerNorm + out‑gate + final projection:
   y = ( (out−μ)/σ·γ + β ) ⊙ out_gate
   out = y·W_to_outᵀ                (produces the final float32 tensor)

Only the two small LayerNorm kernels and the heavy N³ multiplication
are written in Triton; all linear projections use cuBLAS (already optimal).
The final LayerNorm + output‑gate + projection are fused into a single
kernel to cut the extra memory traffic of an intermediate tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*   [total_rows, dim]   (FP32)
    y_ptr,               # *mut   half*    [total_rows, dim]   (FP16)
    weight_ptr,          # *const float*   [dim]               (γ)
    bias_ptr,            # *const float*   [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass – mean + variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass – normalise, γ/β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------

    # block offsets for i/j
    offs_i = i_blk_id * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_blk_id * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: [BLOCK_M, BLOCK_K]
        left_ptrs = left_ptr \
                    + b * stride_left_b \
                    + h * stride_left_h \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile **transposed**: want shape [BLOCK_K, BLOCK_N]
        right_ptrs = right_ptr \
                     + b * stride_right_b \
                     + h * stride_right_h \
                     + offs_k[:, None] * stride_right_i \
                     + offs_j[None, :] * stride_right_k
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0).to(tl.float32)

        acc += tl.dot(a, b_tile)          # (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_ptr \
               + b * stride_out_b \
               + h * stride_out_h \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 3) Fused LayerNorm + out‑gate + final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,               # *const half*   [R, H]    (post‑mul update)
    gate_ptr,            # *const half*   [R, H]    (out_gate, already sigmoid)
    norm_w_ptr,          # *const float*  [H]      (γ)
    norm_b_ptr,          # *const float*  [H]      (β)
    proj_w_ptr,          # *const float*  [H, D]   (W_to_outᵀ)  (H rows, D columns)
    out_ptr,             # *mut   float*  [R, D]
    R: tl.int32,
    H: tl.constexpr,
    D: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim tile
    BLOCK_D: tl.constexpr,   # output‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # ---------------------------------------------------------------
    # 1️⃣ mean / variance (FP32) over the hidden dimension H
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptrs = x_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # 2️⃣ For each output‑dim tile, compute
    #      ( (x‑µ)/σ·γ + β ) ⊙ gate  @  W_to_outᵀ
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        for h in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load x and gate
            x_vals = tl.load(x_ptr + offs_r[:, None] * H + offs_h[None, :],
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)          # (BR, BH)
            g_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)          # (BR, BH)

            # normalise + γ/β + gate
            norm = (x_vals - mean[:, None]) * rstd[:, None]
            w = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # (1, BH)
            b = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)[None, :]   # (1, BH)
            gated = (norm * w + b) * g_vals                                 # (BR, BH)

            # weight slice (H×D) – we stored the **transposed** weight so that
            # it is indexed as [hidden, dim]
            w_out = tl.load(proj_w_ptr + offs_h[:, None] * D + offs_d[None, :],
                            mask=mask_h[:, None] & mask_d[None, :],
                            other=0.0)                                      # (BH, BD)

            # accumulate
            acc += tl.dot(gated, w_out)                                    # (BR, BD)

        # write final result (float32)
        out_ptrs = out_ptr + offs_r[:, None] * D + offs_d[None, :]
        tl.store(out_ptrs,
                 acc,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim   = cfg["dim"]                # C
    hidden = cfg["hidden_dim"]        # H
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                # flattened rows for any (B,N,N,*) tensor

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 256 else 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear + gate (left / right) – fused weight matrices
    # --------------------------------------------------------------
    # Cast static weights once (half‑precision)
    left_proj_w = weights["left_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # left side
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, C)
    left_comb = F.linear(x_norm_fp16, left_comb_w)              # (R, 2H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)
    left = left_proj_raw * left_gate                               # (R, H)

    # right side
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, C)
    right_comb = F.linear(x_norm_fp16, right_comb_w)               # (R, 2H)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)
    right = right_proj_raw * right_gate                             # (R, H)

    # output gate (used later)
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)               # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)                         # (R, H)

    # --------------------------------------------------------------
    # 2a) Optional binary mask
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.half)
        left = left * mask_flat
        right = right * mask_flat

    # --------------------------------------------------------------
    # 3️⃣ Reshape for the TriMul GEMM
    # --------------------------------------------------------------
    left = left.view(B, N, N, hidden).contiguous()
    right = right.view(B, N, N, hidden).contiguous()
    out = torch.empty_like(left)          # placeholder for the product

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update – custom Triton GEMM
    # --------------------------------------------------------------
    stride_left = left.stride()            # (B*N*N*H, N*H, H, 1)
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_MM = 64   # tile sizes tuned for H100 Tensor‑Core
    BLOCK_NN = 64
    BLOCK_KK = 32

    n_i = triton.cdiv(N, BLOCK_MM)
    n_j = triton.cdiv(N, BLOCK_NN)
    grid = (B * hidden * n_i * n_j,)

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_MM,
        BLOCK_N=BLOCK_NN,
        BLOCK_K=BLOCK_KK,
    )

    # --------------------------------------------------------------
    # 5️⃣ Fuse final LayerNorm, out‑gate and projection to dim
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()
    out_gate_flat = out_gate.view(total_rows, hidden).contiguous()

    # Transpose final weight so we can index as [hidden, dim]
    to_out_w_T = weights["to_out.weight"].t().contiguous()   # (H, D)

    final = torch.empty(total_rows, dim, dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_R = 64          # rows per program
    BLOCK_H = 32          # hidden‑dim tile (≤ 384)
    BLOCK_D = 64          # output‑dim tile (≤ 384)

    grid_final = (triton.cdiv(total_rows, BLOCK_R) *
                  triton.cdiv(dim, BLOCK_D),)

    fused_norm_gate_linear_kernel[grid_final](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        to_out_w_T,
        final,
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape to the expected output layout
    # --------------------------------------------------------------
    final = final.view(B, N, N, dim)
    return final
