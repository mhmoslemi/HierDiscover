"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) using a small Triton
   kernel (mean / variance computed in fp32, output in fp16).

2️⃣ Fuse the four linear projections/gates (left/right + left/right‑gate) into a
   single half‑precision GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N)

3️⃣ Separate the result into left/right projections and their gates; compute the
   output gate with a second GEMM.

4️⃣ Apply the optional pair‑wise mask, sigmoid the gates and multiply them:
        left  = left_proj_raw  * sigmoid(left_gate_raw)
        right = right_proj_raw * sigmoid(right_gate_raw)

5️⃣ Tri‑multiplicative update (batched matrix multiplication):
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a single `torch.bmm` over the reshaped tensors
   (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N).

6️⃣ Second LayerNorm over the hidden dimension (H) – another Triton kernel.

7️⃣ Multiply by the output‑gate, project back to `dim` with a final linear
   layer and cast to float32.

All heavy arithmetic is performed in fp16 (Tensor‑Core friendly);
only the two LayerNorms use a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]    (float32)
    y_ptr,          # mutable *[total_rows, dim]    (float16)
    weight_ptr,     # const   *[dim]                (float32)  (γ)
    bias_ptr,       # const   *[dim]                (float32)  (β)
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # feature‑wise tile size
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / √(σ²+eps)) * γ + β
    Mean/variance are accumulated in fp32, output is stored in fp16.
    """
    pid = tl.program_id(0)                     # each program works on a tile of rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # load a (BLOCK_M, BLOCK_D) tile
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized result γ·(x‑μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping containing every weight/bias used by the module
                         (all tensors are float32)
        - config_dict  : must contain
                         * "dim"         – input channel dimension
                         * "hidden_dim"  – hidden dimension H
                         * "nomask"      – True ⇒ no mask is applied

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into one GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate the four weight matrices (shape: [4·H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16

    # (R, dim) @ (4·H, dim)ᵀ → (R, 4·H)
    proj_gate = F.linear(x_norm, w_concat)   # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )   # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if present), sigmoid gates and apply them
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask: [B,N,N] → [R,1] (fp16)
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) – no copy, just a view
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring head dimension into batch dimension: (B, H, N, N)
    left   = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right  = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate        # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
