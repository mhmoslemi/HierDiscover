"""
TriMul (outgoing) forward pass – highly‑optimized Triton + torch implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B,N,N,dim).  
   The input is flattened to (R,dim) where R = B·N·N, normalized in FP32
   and the result is written in FP16 (kernel `layernorm_kernel`).

2️⃣ Fuse the four linear projections/gates (left_proj, right_proj,
   left_gate, right_gate) into a single FP16 GEMM:
        X_norm_half @ W_proj_gate.T   →   (R, 4·H)
   The weight matrix is built once by concatenation.  The result is split
   into left_proj, right_proj, left_gate and right_gate.

3️⃣ Compute the output‑gate with a second GEMM (still FP16).

4️⃣ Apply optional mask (broadcast over the hidden dimension),
   sigmoid the gates (in‑place) and multiply the projections by their
   gates → `left` and `right` tensors in FP16.

5️⃣ Reshape to (B, H, N, N) and perform the N³ multiplicative update
   with a batched half‑precision matrix multiplication:
        out_bh = left_bh @ right_bhᵀ
   This call maps to cuBLAS Tensor‑Core kernels (the dominant cost).

6️⃣ Second LayerNorm over the hidden dimension (FP16).

7️⃣ Apply the output‑gate, a final FP16 linear projection back to `dim`,
   and cast the result to FP32.

All heavy per‑row normalisations are done with a custom Triton kernel,
while the large dense matrix products use cuBLAS for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input FP32 → output FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)   gamma
    bias_ptr,       # const   * [dim]               (float32)   beta
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    """
    Row‑wise LayerNorm: y = (x - μ) / √(σ²+ε) * γ + β
    Input  : x (float32) – shape [total_rows, dim]
    Output : y (float16) – same shape
    """
    pid = tl.program_id(0)                     # 1‑D grid over rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the column dimension in blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(σ²+ε)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 64) -> torch.Tensor:
    """
    Apply `layernorm_kernel` to a 2‑D FP32 tensor ``x`` → FP16 tensor.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Main entry point – outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or
                         ``None`` when ``config['nomask']`` is True
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain the keys
                         "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()   # (R, dim) FP32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        eps=eps,
    )                                      # (R, dim) FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused projections + gates (produces 4·H channels)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices [H, dim] → [4·H, dim]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                 # (4·H, dim)

    # Single GEMM: (R, dim) @ (4·H, dim).T → (R, 4·H)
    proj_gate = F.linear(x_norm, w_proj_gate)   # FP16

    # Split the concatenated result
    left_proj_raw  = proj_gate[:, 0:H]               # (R, H)
    right_proj_raw = proj_gate[:, H:2*H]             # (R, H)
    left_gate_raw  = proj_gate[:, 2*H:3*H]           # (R, H)
    right_gate_raw = proj_gate[:, 3*H:4*H]           # (R, H)

    # Output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # (H, dim)
    )                                                    # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gating, element‑wise gating
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B,N,N] → [R,1] broadcast over H
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (no extra buffers)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left = left_proj_raw * left_gate          # (R, H) FP16
    right = right_proj_raw * right_gate       # (R, H) FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # Back to 4‑D layout
    left = left.view(B, N, N, H)          # (B, N, N, H)
    right = right.view(B, N, N, H)        # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)  # (B, N, N, H)

    # Move the hidden dimension to the batch axis → (B·H, N, N)
    left_bh = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_bh = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))  # (B·H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → FP16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
    )                                   # (R, H) FP16
    out_norm = out_norm.view(B, N, N, H)   # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 6️⃣ Output gate & final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # (B,N,N,H) FP16
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),    # (dim, H)
    )                                                   # (R, dim) FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
