def sigmoid_kernel(x_ptr, y_ptr, total, H, BLOCK_M, BLOCK_H):
    pid = tl.program_id(0)
    rows = tl.arange(0, BLOCK_M) + pid * BLOCK_M
    cols = tl.arange(0, BLOCK_H)
    mask_rows = rows < total
    mask_cols = cols < H
    ptrs = x_ptr + rows[:, None] * H + cols[None, :]
    vals = tl.load(ptrs, mask=mask_rows[:, None] & mask_cols[None, :], other=0.0)
    sigmoid = 1.0 / (1.0 + tl.exp(-vals))
    out_ptrs = y_ptr + rows[:, None] * H + cols[None, :]
    tl.store(out_ptrs, sigmoid, mask=mask_rows[:, None] & mask_cols[None, :])

def sigmoid_kernel(
    x_ptr, y_ptr,
    total_rows, total_cols,
    stride_row,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,):

@triton.jit
def three_sigmoid_kernel(
    a_ptr, b_ptr, c_ptr, # inputs
    a_out_ptr, b_out_ptr, c_out_ptr,
    total_rows, total_cols,
    stride_row,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    mask_m = offs_m < total_rows
    mask_n = offs_n < total_cols

    # load
    a = tl.load(a_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)
    b = tl.load(b_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)
    c = tl.load(c_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)

    a_f = a.to(tl.float32)
    b_f = b.to(tl.float32)
    c_f = c.to(tl.float32)

    a_sig = 1.0 / (1.0 + tl.exp(-a_f))
    b_sig = 1.0 / (1.0 + tl.exp(-b_f))
    c_sig = 1.0 / (1.0 + tl.exp(-c_f))

    a_sig = a_sig.to(tl.float16)
    b_sig = b_sig.to(tl.float16)
    c_sig = c_sig.to(tl.float16)

    tl.store(a_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                a_sig,
                mask=mask_m[:, None] & mask_n[None, :])
    tl.store(b_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                b_sig,
                mask=mask_m[:, None] & mask_n[None, :])
    tl.store(c_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                c_sig,
                mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def three_sigmoid_kernel(
    a_ptr, b_ptr, c_ptr,
    a_out_ptr, b_out_ptr, c_out_ptr,
    total_rows, total_cols,
    stride_row,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < total_cols

    # load
    a = tl.load(a_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)
    b = tl.load(b_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)
    c = tl.load(c_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)

    a_f = a.to(tl.float32)
    b_f = b.to(tl.float32)
    c_f = c.to(tl.float32)

    a_sig = 1.0 / (1.0 + tl.exp(-a_f))
    b_sig = 1.0 / (1.0 + tl.exp(-b_f))
    c_sig = 1.0 / (1.0 + tl.exp(-c_f))

    a_sig = a_sig.to(tl.float16)
    b_sig = b_sig.to(tl.float16)
    c_sig = c_sig.to(tl.float16)

    tl.store(a_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                a_sig,
                mask=mask_m[:, None] & mask_n[None, :])
    tl.store(b_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                b_sig,

    tl.store(c_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def rowwise_layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    total_rows,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # accumulate sums
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over column blocks
    for d_off in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM

        # load tile
        ptrs = x_ptr + (offs_m[:, None] * DIM) + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / DIM
    var = sum_sq / DIM - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized output
    for d_off in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM

        ptrs = x_ptr + (offs_m[:, None] * DIM) + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # gamma
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # beta

        out = norm * w[None, :] + b[None, :]  # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * DIM) + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

@triton.jit
def three_sigmoid_kernel(
    a_ptr, b_ptr, c_ptr,
    a_out_ptr, b_out_ptr, c_out_ptr,
    total_rows, total_cols,
    stride_row,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < total_cols

    a = tl.load(a_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)
    b = tl.load(b_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)
    c = tl.load(c_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                mask=mask_m[:, None] & mask_n[None, :], other=0.0)

    a_f = a.to(tl.float32)
    b_f = b.to(tl.float32)
    c_f = c.to(tl.float32)

    a_sig = 1.0 / (1.0 + tl.exp(-a_f))
    b_sig = 1.0 / (1.0 + tl.exp(-b_f))
    c_sig = 1.0 / (1.0 + tl.exp(-c_f))

    a_sig = a_sig.to(tl.float16)
    b_sig = b_sig.to(tl.float16)
    c_sig = c_sig.to(tl.float16)

    tl.store(a_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                a_sig,
                mask=mask_m[:, None] & mask_n[None, :])
    tl.store(b_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                b_sig,
                mask=mask_m[:, None] & mask_n[None, :])
    tl.store(c_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :],
                c_sig,
                mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N  # total rows for the first LayerNorm

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()
    x_norm_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],      # gamma, fp32
        weights["norm.bias"],        # beta, fp32
        R,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        DIM=dim,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into one GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)
    # Compute [R, 4*H] tensor
    proj_gate = F.linear(x_norm_half, w_proj_gate)  # fp16

    left_proj_raw = proj_gate[:, :H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)  # [H, dim]
    )  # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)  # [R,1]
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates (fused Triton kernel)
    # ------------------------------------------------------------------
    left_gate = torch.empty_like(left_gate_raw)
    right_gate = torch.empty_like(right_gate_raw)
    out_gate = torch.empty_like(out_gate_raw)

    # stride for contiguous 2‑D tensor = total_cols (H)
    stride = H
    G_BLOCK_M = 256
    G_BLOCK_N = 64
    grid_gate = (triton.cdiv(R, G_BLOCK_M),)
    three_sigmoid_kernel[grid_gate](
        left_gate_raw,
        right_gate_raw,
        out_gate_raw,
        left_gate,
        right_gate,
        out_gate,
        R,
        H,
        stride,
        BLOCK_M=G_BLOCK_M,
        BLOCK_N=G_BLOCK_N,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Apply gating
    # ------------------------------------------------------------------
    left = left_proj_raw * left_gate   # [R, H] fp16
    right = right_proj_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Reshape for N³ update and batched GEMM
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H) # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # bring hidden dimension to batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.view(B * H, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(B * H, N, N) # [B*H, N, N]

    # batched half‑precision GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # gamma fp32
        weights["to_out_norm.bias"],     # beta fp32
        R,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        DIM=H,
        num_warps=4,
    )
    out_norm = out_norm_half.view(B, N, N, H)  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate  # fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)  # [dim, H]
    )  # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) written in Triton.
2️⃣ Fuse the four projection/gate Linear layers into a single GEMM.
3️⃣ Compute the three sigmoid gates with a single Triton kernel
   (processes `left_gate`, `right_gate`, `out_gate` in one launch).
4️⃣ Optional mask is applied *before* gating.
5️⃣ The N³ multiplicative update is a batched half‑precision GEMM
   (cuBLAS Tensor‑Cores) – the fastest primitive for this operation.
6️⃣ Second LayerNorm (over the hidden dimension) – again Triton.
7️⃣ Apply the output‑gate, final Linear projection and cast back to fp32.

All heavy reductions (mean/var, sigmoid) run on‑device; the dense
matrix multiplications rely on cuBLAS which is optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,               # const   * [total_rows, DIM]  (float32)
    y_ptr,               # mutable * [total_rows, DIM]  (float16)
    w_ptr,               # const   * [DIM]               (float32) γ
    b_ptr,               # const   * [DIM]               (float32) β
    total_rows,          # int (runtime)
    eps,                 # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    DIM: tl.constexpr,   # channel dimension (compile‑time)
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Reduce to obtain mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < DIM

        ptrs = x_ptr + (offs_m[:, None] * DIM) + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / DIM
    var  = sum_sq / DIM - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(σ² + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(DIM, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < DIM

        ptrs = x_ptr + (offs_m[:, None] * DIM) + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * DIM) + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: three sigmoids in one launch
# ----------------------------------------------------------------------
@triton.jit
def three_sigmoid_kernel(
    a_ptr, b_ptr, c_ptr,          # input tensors   (fp16)
    a_out_ptr, b_out_ptr, c_out_ptr,  # output tensors (fp16, can alias inputs)
    total_rows, total_cols,
    stride_row,                   # row stride = total_cols (contiguous)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    """
    Compute sigmoid(x) for three tensors in parallel.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    offs_n = tl.arange(0, BLOCK_N)                  # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < total_cols
    mask = mask_m[:, None] & mask_n[None, :]

    # load
    a = tl.load(a_ptr + offs_m[:, None] * stride_row + offs_n[None, :], mask=mask, other=0.0)
    b = tl.load(b_ptr + offs_m[:, None] * stride_row + offs_n[None, :], mask=mask, other=0.0)
    c = tl.load(c_ptr + offs_m[:, None] * stride_row + offs_n[None, :], mask=mask, other=0.0)

    a_f = a.to(tl.float32)
    b_f = b.to(tl.float32)
    c_f = c.to(tl.float32)

    a_sig = 1.0 / (1.0 + tl.exp(-a_f))
    b_sig = 1.0 / (1.0 + tl.exp(-b_f))
    c_sig = 1.0 / (1.0 + tl.exp(-c_f))

    a_sig = a_sig.to(tl.float16)
    b_sig = b_sig.to(tl.float16)
    c_sig = c_sig.to(tl.float16)

    tl.store(a_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :], a_sig, mask=mask)
    tl.store(b_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :], b_sig, mask=mask)
    tl.store(c_out_ptr + offs_m[:, None] * stride_row + offs_n[None, :], c_sig, mask=mask)


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : dict with all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input / output channel dimension
    hidden_dim = config["hidden_dim"] # hidden dimension (H)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                      # total rows for the first LayerNorm

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()  # [R, dim] fp32
    x_norm_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],    # γ  (fp32)
        weights["norm.bias"],      # β  (fp32)
        R,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        DIM=dim,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate weight matrices (all are [H, dim] without bias)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [4*H, dim] fp16

    # fused GEMM: [R, dim] @ [4*H, dim]^T → [R, 4*H]
    proj_gate = F.linear(x_norm_half, w_proj_gate)   # fp16

    # split the concatenated result
    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]           # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]           # [R, H]

    # out‑gate Linear (separate weight)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )  # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)  # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates – one Triton launch for three tensors
    # ------------------------------------------------------------------
    # In‑place: input and output pointers are the same tensors
    G_BLOCK_M = 256
    G_BLOCK_N = 64
    grid_gate = (triton.cdiv(R, G_BLOCK_M),)
    three_sigmoid_kernel[grid_gate](
        left_gate_raw, right_gate_raw, out_gate_raw,
        left_gate_raw, right_gate_raw, out_gate_raw,
        R,
        H,
        H,                     # row stride (contiguous)
        BLOCK_M=G_BLOCK_M,
        BLOCK_N=G_BLOCK_N,
        num_warps=4,
    )
    # after the kernel the three tensors now contain their sigmoids
    left_gate  = left_gate_raw
    right_gate = right_gate_raw
    out_gate   = out_gate_raw

    # ------------------------------------------------------------------
    # 5️⃣ Apply gating to the projection tensors
    # ------------------------------------------------------------------
    left  = left_proj_raw * left_gate   # [R, H] fp16
    right = right_proj_raw * right_gate # [R, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # bring the hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(B * H, N, N)  # [B*H, N, N]

    # batched half‑precision GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        DIM=H,
        num_warps=4,
    )
    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate   # fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )  # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
