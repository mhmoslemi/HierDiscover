"""
TriMul (outgoing) forward pass – high‑performance implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (over the last channel) on the flattened input.
   Implemented with a Triton kernel that works on fp32 inputs and writes fp16.

2️⃣ Fuse the four linear layers (left/right projections + left/right gates)
   into a single half‑precision GEMM:
        [R, dim] × [4·H, dim]ᵀ  →  [R, 4·H]
   where R = B·N·N and H = hidden_dim.
   The result is split into *left_proj*, *right_proj*, *left_gate*, *right_gate*.

3️⃣ Compute the output‑gate with a separate half‑precision GEMM.

4️⃣ (Optional) Apply the binary mask to the left/right projections.
   Apply the sigmoid to the three gate tensors (in‑place) and fuse them:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)

5️⃣ Triangular multiplicative update (batched GEMM):
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   Implemented as a batched half‑precision matrix multiplication
   (B·H, N, N) @ (B·H, N, N)ᵀ using torch.bmm (cuBLAS Tensor‑Cores).

6️⃣ Second LayerNorm over the hidden dimension (H) → fp16 via the same Triton kernel.

7️⃣ Multiply by the output‑gate, apply the final linear projection,
   and cast the result back to fp32.

All heavy linear algebra is performed in FP16 to leverage Tensor Cores,
while the two LayerNorms are done with a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the channel dimension in tiles
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # address: row * dim + col
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    Forward pass of the outgoing TriMul operator.

        x_flat,
        x_norm_fp16,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)

        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

            weights["left_gate.weight"],   # [H, dim]

    fused = F.linear(x_norm_fp16, w_fused)                  # fp16

    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and combine
    # --------------------------------------------------------------
    if not nomask:

        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid (fp16) to minimise temporaries
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H]  fp16
    right = right_raw * right_gate  # [R, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring hidden dimension into the batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch & heads → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched GEMM (FP16) – each head does a full N×N matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)      # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                 # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                               # [R, dim] fp16

    # Cast back to float32 for the model’s expected dtype
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
