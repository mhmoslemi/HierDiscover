import triton.language as tl

@triton.jit
def rowwise_layernorm_kernel(x_ptr, y_ptr, w_ptr, b_ptr, total_rows, dim, eps,
                             BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < total_rows
   # accumulate sum and sumsq
   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim
       ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
       vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
       vals_f32 = vals.to(tl.float32)
       sum_val += tl.sum(vals_f32, axis=1)
       sum_sq += tl.sum(vals_f32 * vals_f32, axis=1)

   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim
       ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
       vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
       vals_f32 = vals.to(tl.float32)

       norm = (vals_f32 - mean[:, None]) * rstd[:, None]  # fp32

       w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
       b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

       out = norm * w[None, :] + b[None, :]
       out_fp16 = out.to(tl.float16)
       tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N

    # Flatten input for first LayerNorm
    x_flat = input_tensor.view(R, dim).contiguous()
    x_norm_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Launch Rowwise LN kernel
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],   # float32
        weights["norm.bias"],     # float32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # Fuse projections and gate weights
    H = hidden_dim
    # Concatenate weights (out_features x in_features)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    proj_gate = F.linear(x_norm_half, w_proj_gate)   # [R, 4*H] fp16

    left_proj_raw = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # output gate
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)
    )   # [R, H]

    # Apply mask if needed
    if not config.get("nomask", True):
        # mask to fp16 shape [R,1]
        mask_flat = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # Sigmoid gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate   # [R, H]
    right = right_proj_raw * right_gate # [R, H]

    # Reshape for batched GEMM
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Fuse batch and head dimensions for torch.bmm
    left_mat = left_perm.view(B * H, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(B * H, N, N) # [B*H, N, N]

    # Batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm over hidden dimension H
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm_half.view(B, N, N, H)

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection to dim
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )  # [R, dim]
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
custom_kernel
==============

Forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the last dimension (`dim`) of the input tensor.
    Implemented in Triton and writes the result in FP16 (FP32 accumulation).

2️⃣  Fuse the four linear layers (`left_proj`, `right_proj`,
    `left_gate`, `right_gate`) into a single GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N, H = hidden_dim)

3️⃣  Compute the separate output‑gate linear layer (`out_gate`) also with a GEMM.

4️⃣  (Optional) Apply the pair‑wise mask to the *proj* tensors.
    Apply sigmoid to the three gate tensors and multiply the projections
    with their respective gates.

5️⃣  Reshape to `[B, N, N, H]` and perform the N³ multiplicative update
    as a batched half‑precision matrix multiplication:
        out[b, h] = left[b, h] @ right[b, h]ᵀ

6️⃣  Second row‑wise LayerNorm over the hidden dimension (`H`), again in Triton.

7️⃣  Multiply with the output gate, project back to `dim` with a final linear
    layer and cast the result to FP32.

Only the two LayerNorm steps are written in Triton; all dense matrix
multiplications use PyTorch’s highly‑optimized cuBLAS kernels (FP16/Tensor‑Core).
The implementation works for all shapes listed in the problem statement and
supports the presence or absence of a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# --------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]  (float32)
    y_ptr,          # mutable * [total_rows, dim]  (float16)
    w_ptr,          # const   * [dim]              (float32)  γ
    b_ptr,          # const   * [dim]              (float32)  β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ² + eps)) * γ + β
       Output is FP16, reduction is performed in FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------
    # 1️⃣  Compute per‑row mean & variance
    # --------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of shape [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # --------------------------------------------------
    # 2️⃣  Write normalized result γ·(x−μ)·rstd + β  (FP16)
    # --------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # FP32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total number of rows for row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()      # [R, dim] FP32
    x_norm_half = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections (single GEMM)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection/gate weight matrices (shape [4·H, dim])
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)      # FP16

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H] (FP16)
    proj_gate = F.linear(x_norm_half, w_proj_gate)

    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2 * H]
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]

    # Output‑gate linear layer (separate weight matrix)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H] (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1] (FP16)
        mask_fp16 = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()    # in‑place sigmoid (FP16)
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate       # [R, H] FP16
    right = right_proj_raw * right_gate     # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ N³ Multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # Move hidden dimension into the batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch × head for a single torch.bmm call
    left_mat  = left_perm.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)  # [B·H, N, N]

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → FP16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm_half.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B, N, N, H] FP16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim] FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
