"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------

1. **Row‑wise LayerNorm** – implemented in Triton (fp32 statistics → fp16 output).
2. **Fused linear projections + gates** – a single `F.linear` with a
   concatenated weight matrix produces left/right projections and left/right
   gates in one half‑precision GEMM.
3. **Masking & sigmoid gating** – optional elementwise mask on the projections,
   followed by in‑place sigmoid for the three gate tensors.
4. **Triangular multiplicative update** – for each hidden dimension `h`
   we need `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is exactly a batched matrix multiplication:
   `out_h = left_h @ right_h.T`.  All `H` heads are processed together
   as a single batched GEMM (`torch.bmm`) in fp16, which is far faster
   than the original `torch.einsum`.
5. **Second LayerNorm** – again with the Triton kernel (fp16 output).
6. **Output gating & final projection** – elementwise gate, then a half‑precision
   linear layer back to the model dimension, finally cast to `float32`.

All heavy tensor work stays in `float16` to fully exploit Tensor Cores on an
H100.  The only Triton kernel required is the row‑wise LayerNorm; the cubic
update is performed by a highly‑optimised batched GEMM.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (float32)
    y_ptr,               # mutable * [total_rows, dim]   (float16)
    weight_ptr,          # const   * [dim]               (float32)
    bias_ptr,            # const   * [dim]               (float32)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = ((x-μ)/√(σ²+eps))*γ + β
    μ,σ are computed in fp32, the result is stored in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------
    # 1) compute mean / variance per row (fp32)
    # ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # ----------
    # 2) write normalised values (fp16)
    # ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool / float)
        - weights_dict : model parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5
    device     = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] float32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 128                     # works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],       # γ (float32)
        weights["norm.bias"],         # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ fused projection & gates (single GEMM)
    # --------------------------------------------------------------
    #   [4*H, dim] weight matrix:
    #   rows 0:H         → left_proj
    #   rows H:2H        → right_proj
    #   rows 2H:3H       → left_gate
    #   rows 3H:4H       → right_gate
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [4*H, dim]

    # Linear in fp16
    proj_gate = F.linear(x_norm, w_4h)           # [R, 4*H] fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    out_gate_raw = F.linear(x_norm,
                            weights["out_gate.weight"].to(torch.float16))   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not nomask:
        # mask has shape [B, N, N]; broadcast over hidden dim
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ reshape to 4‑D tensors
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ triangular multiplicative update via batched GEMM
    # --------------------------------------------------------------
    # Convert to shape [B*H, N, N] -> treat each hidden slice as a batch
    left_perm  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_perm = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [B*H, N, N]

    # out_h = left_h @ right_h.T   (batch GEMM)
    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))   # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_perm.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm over hidden dim → fp16
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ output gate + final linear projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                       # elementwise gate, fp16
    final = F.linear(out_norm.reshape(total_rows, H),
                     weights["to_out.weight"].to(torch.float16))   # fp16 -> [R, dim]
    final = final.reshape(B, N, N, dim).to(torch.float32)

    return final
