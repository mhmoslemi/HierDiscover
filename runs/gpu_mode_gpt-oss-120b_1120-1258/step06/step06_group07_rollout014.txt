    gate_ptr,       # out_gate (R, hidden) half

    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

    # Compute mean and variance across hidden dimension:
    # need to accumulate sum and sum sq across hidden before normalizing
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # First pass to compute sum and sumsq
    for k in range(0, tl.cdiv(hidden, BLOCK_K)):

        mask_k = offs_k < hidden

                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0).to(tl.float32)   # (BLOCK_M, BLOCK_K)
        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)

    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop again over hidden to compute normalized*gate and multiply with weight_out
    for k in range(0, tl.cdiv(hidden, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

"""
Highly‑optimized Triton forward for the **outgoing** TriMul operator
(AlphaFold‑3).  

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor  `[B,N,N,C] → (B·N·N, C)`  
   (FP32 → FP16) – a single Triton kernel.

2️⃣ A *single* fused linear pass builds the three required streams
   (left_proj/left_gate, right_proj/right_gate, out_gate) from the
   normalised tensor.  This saves two extra reads of the normalised
   input.

3️⃣ Optional binary mask is applied (broadcasted on the hidden dim).

4️⃣ Triangular multiplicative update – batched half‑precision GEMM
   using `torch.bmm` (the dominant O(N³) work).

5️⃣ A fused kernel performs **LayerNorm + out‑gate** on the result of the
   triangular update *and* the final linear projection back to `dim`
   (hidden → dim).  The kernel computes the mean/variance of each row,
   normalises, multiplies by the pre‑computed `out_gate`, and accumulates
   the product with the weight matrix `to_out` (stored transposed for
   efficient access).  This removes an extra memory round‑trip and one
   large GEMM.

All kernels run in FP16 (except the reductions that use FP32 for numerical
stability).  The final output is cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm  (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*   [total_rows, dim]   (FP32)
    y_ptr,               # *mut   half*    [total_rows, dim]   (FP16)
    w_ptr,               # *const float*   [dim]               (γ)
    b_ptr,               # *const float*   [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # first pass – compute mean & variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # dim fits in a single block (BLOCK_D = dim) → only one iteration
    offs_d = tl.arange(0, BLOCK_D)                     # [BLOCK_D]
    mask_d = offs_d < dim
    ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
    vals = tl.load(ptrs,
                   mask=mask_m[:, None] & mask_d[None, :],
                   other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

    sum_val += tl.sum(vals, axis=1)
    sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # second pass – normalise, apply γ/β and store as FP16
    # ---------------------------------------------------------------
    w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
    b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
    norm = (vals - mean[:, None]) * rstd[:, None]       # (x-µ)/σ
    out = norm * w[None, :] + b[None, :]                # FP32
    tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out.to(tl.float16),
             mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) Fused LayerNorm + out‑gate + final linear   (hidden → dim)
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]  (pre‑norm)
    gate_ptr,            # *const half*  [total_rows, hidden]  (out_gate, sigmoid’ed)
    y_ptr,               # *mut   half*  [total_rows, dim]    (final output)
    ln_w_ptr,            # *const float* [hidden]             (γ of to_out_norm)
    ln_b_ptr,            # *const float* [hidden]             (β of to_out_norm)
    w_out_ptr,           # *const half*  [hidden, dim]        (to_out weight, transposed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # 1) Compute mean / var across hidden  (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden
        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)                # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # 2) Normalise, apply ln‑γ/β, multiply by out_gate and accumulate
    #    the final projection (hidden → dim) via a tiled dot product.
    # ---------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

    # dim offset (static)
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    for k in range(0, tl.cdiv(hidden, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        # ----- load a tile of the normalised input -----
        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)                # [BLOCK_M, BLOCK_K]

        # normalise
        norm = (x - mean[:, None]) * rstd[:, None]            # (x‑µ)/σ

        # apply LayerNorm weight / bias (γ,β) on the hidden dim
        ln_w = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # γ
        ln_b = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # β
        norm = norm * ln_w[None, :] + ln_b[None, :]

        # multiply by the already‑sigmoid’ed out‑gate
        g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        norm = norm * g

        # ----- load the corresponding slice of the output weight -----
        # w_out shape : (hidden, dim)  →  tile [BLOCK_K, BLOCK_D]
        w = tl.load(
            w_out_ptr + offs_k[:, None] * dim + offs_d[None, :],
            mask=mask_k[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)                       # [BLOCK_K, BLOCK_D]

        # accumulate mat‑mul: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_D)
        acc += tl.dot(norm, w)                              # [BLOCK_M, BLOCK_D]

    # store final result
    tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_d[None, :])

    dim     = cfg["dim"]                     # C
    hidden  = cfg["hidden_dim"]              # H
    eps     = cfg.get("eps", 1e-5)
    nomask  = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()               # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)                       # [R, C] half

    BLOCK_M = 256                         # rows per program (chosen to cut launch overhead)
    BLOCK_D = dim                         # one iteration over the channel dimension

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear + gating for left / right streams and out_gate
    # ------------------------------------------------------------------
    # All linear weights are stored as FP16 for Tensor‑Core GEMM
    left_proj_w  = weights["left_proj.weight"].half().contiguous()
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # fuse the 5 linear maps into a single GEMM
    comb_w = torch.cat(
        [left_proj_w, left_gate_w,
         right_proj_w, right_gate_w,
         out_gate_w], dim=0)                        # (5*H, C)

    comb = F.linear(x_norm_fp16, comb_w)                # (R, 5*H)

    # split and apply gates
    split = torch.split(comb, hidden, dim=1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = split
    left_gate = torch.sigmoid(left_gate_raw)            # (R, H)
    right_gate = torch.sigmoid(right_gate_raw)          # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)              # (R, H)

    left  = left_proj_raw * left_gate   # (R, H)
    right = right_proj_raw * right_gate # (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional binary mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update  (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden)       # (B, N, N, H)
    right = right.view(B, N, N, hidden)      # (B, N, N, H)

    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                         # (B*H, N, N) half
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out‑gate + final linear
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                             device=device)                      # (R, dim)

    # weight of the *second* LayerNorm (to_out_norm) – stays FP32
    # final linear weight is transposed for easier access inside the kernel
    w_out_T = weights["to_out.weight"].t().contiguous().half()  # (H, dim)

    BLOCK_D_OUT = dim               # one block in the output feature dimension
    BLOCK_K = 32                    # tile size for the hidden dimension
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_norm_gate_linear_kernel[grid2](
        out_flat,
        out_gate,
        final_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        w_out_T,
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D_OUT,
        BLOCK_K=BLOCK_K,
    )

    # cast back to float32 for the model
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
