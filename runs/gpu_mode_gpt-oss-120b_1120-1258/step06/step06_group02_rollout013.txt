"""
High‑performance implementation of the outgoing TriMul operator (AlphaFold‑3).

The algorithm follows the reference PyTorch implementation but fuses many
operations:

1️⃣  Row‑wise LayerNorm (fp32 statistics → fp16 output) implemented in Triton.
2️⃣  A single fused linear projection that produces the two projections
    (left/right) and the three gating tensors (left‑gate, right‑gate, out‑gate)
    in one GEMM.
3️⃣  Optional mask‑application (broadcasted over the hidden dimension).
4️⃣  Element‑wise sigmoid gates applied in‑place.
5️⃣  The cubic multiplicative update is expressed as a batched matrix
    multiplication: ``out = left @ rightᵀ`` for each hidden channel.
    This maps to cuBLAS batched GEMM (Tensor‑Core accelerated on H100).
6️⃣  Second row‑wise LayerNorm (fp16) on the cubic result.
7️⃣  Element‑wise output gate, final linear projection back to ``dim`` and
    conversion to ``float32``.

All heavy linear work stays in half‑precision, only the LayerNorm
statistics are computed in fp32 to preserve numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 mean/var → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The mean and variance are computed in fp32, the normalized result is
    written out as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row sum and sum‑of‑squares (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a BLOCK_M × BLOCK_D tile
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Normalise and apply γ, β (store fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored
        - weights_dict : model parameters (all float32)
        - config_dict  : must contain
            "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5                     # same as torch.nn.LayerNorm default
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # Flatten to [R, dim]
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 128                     # Works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (fp32)
        weights["norm.bias"],        # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + all three gates in a single GEMM
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (proj left/right + gate left/right + out_gate)
    w_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # shape (5*H, dim)

    # Linear in half‑precision: [R, dim] @ [dim, 5*H]ᵀ → [R, 5*H]
    proj_gate = F.linear(x_norm, w_all)               # fp16

    # Split into the individual tensors
    left_raw   = proj_gate[:, 0:H]                     # [R, H]
    right_raw  = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]
    out_gate_raw   = proj_gate[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_flat = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate            # [R, H]  fp16
    right = right_raw * right_gate          # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)            # [B, N, N, H]
    right = right.view(B, N, N, H)           # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)     # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Transform to [B, H, N, N] to use batched GEMM
    left_perm  = left.permute(0, 3, 1, 2)     # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2)    # [B, H, N, N]

    # out_perm[b, h, i, j] = Σ_k left_perm[b, h, i, k] * right_perm[b, h, j, k]
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # [B, H, N, N]

    # Back to [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1)       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate           # element‑wise gating, fp16

    final = F.linear(
        out_norm.reshape(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                            # fp16, shape [R, dim]

    final = final.view(B, N, N, dim).to(torch.float32)
    return final
