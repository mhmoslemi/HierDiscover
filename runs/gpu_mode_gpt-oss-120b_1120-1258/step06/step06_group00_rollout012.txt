"""
fast_outgoing_trimul.py
-----------------------

Implementation of the *outgoing* TriMul operator (AlphaFold‑3) that
uses a single Triton kernel for the second LayerNorm + gating + final
projection.  The heavy N³ multiplicative update is performed with a
batched fp16 GEMM (torch.matmul) which exploits the H100 Tensor‑Core
engine.  The first LayerNorm is delegated to the highly‑optimised
torch.nn.functional.layer_norm implementation, while the output
normalisation+linear projection are fused in the custom kernel to
reduce memory traffic.

The kernel processes a tile of rows (`BLOCK_M`) and a tile of output
features (`BLOCK_O`).  For each row it:
    1) computes the row‑wise mean/variance over the hidden dimension `H`,
    2) normalises the `out` values, applies `to_out_norm` scale/shift
       and the element‑wise output‑gate,
    3) performs a dot‑product with the transposed final projection
       matrix `W_outᵀ : (H, dim)`.
All arithmetic is carried out in fp32 inside the kernel, while the
input, weights and output are stored in fp16 for maximum Tensor‑Core
throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: fused LayerNorm + out‑gate + final linear projection
# ---------------------------------------------------------------
@triton.jit
def final_proj_layernorm_kernel(
    out_ptr,            # fp16  *[total_rows, H]      (pre‑norm result)
    out_gate_ptr,       # fp16  *[total_rows, H]      (output‑gate)
    norm_w_ptr,         # fp32  *[H]                  (to_out_norm.weight)
    norm_b_ptr,         # fp32  *[H]                  (to_out_norm.bias)
    w_out_T_ptr,        # fp16  *[H, dim]             (to_out.weightᵀ)
    out_fp16_ptr,       # fp16  *[total_rows, dim]    (final result)
    total_rows,         # int runtime
    dim,                # int runtime
    H,                  # int runtime (hidden_dim)
    eps,                # float constant
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # reduction tile over hidden dim
    BLOCK_O: tl.constexpr,   # output‑feature tile
):
    """
    For each row r ∈ [0, total_rows) compute
        y = ((out[r] - μ) / √(σ²+eps)) * γ + β        (LayerNorm)
        y = y * out_gate[r]                          (output‑gate)
        out_fp16[r] = y @ W_outᵀ                     (final linear)
    where μ,σ² are the mean/variance of out[r] over the H hidden
    dimensions, γ,β are the to_out_norm parameters and
    W_out has shape (dim, H).  All heavy work is done in fp16/ fp32
    using Tensor‑Cores.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------
    # 1️⃣ Compute row‑wise mean and variance over the hidden dim H
    # -----------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < H
        # out_ptr layout: [row, H] → stride = H
        ptrs = out_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # -----------------------------------------------------------
    # 2️⃣ Outer loop over output‑feature tiles (dim)
    # -----------------------------------------------------------
    for o_off in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)    # [BLOCK_O]
        mask_o = offs_o < dim

        # accumulator for the current tile (fp32)
        acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

        # -------------------------------------------------------
        #   Inner reduction over hidden dimension H
        # -------------------------------------------------------
        for d_off in range(0, tl.cdiv(H, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
            mask_d = offs_d < H

            # Load out and out_gate
            out_ptrs = out_ptr + offs_m[:, None] * H + offs_d[None, :]
            out_vals = tl.load(out_ptrs,
                               mask=mask_m[:, None] & mask_d[None, :],
                               other=0.0).to(tl.float32)

            gate_ptrs = out_gate_ptr + offs_m[:, None] * H + offs_d[None, :]
            gate_vals = tl.load(gate_ptrs,
                                mask=mask_m[:, None] & mask_d[None, :],
                                other=0.0).to(tl.float32)

            # Layer‑Norm transformation (γ,β are fp32)
            gamma = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
            beta  = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)   # β
            norm = (out_vals - mean[:, None]) * rstd[:, None]              # (out-μ)/σ
            norm = norm * gamma[None, :] + beta[None, :]                    # γ·norm+β

            # Apply output‑gate
            norm_gate = norm * gate_vals                                   # (γ·norm+β)·gate

            # Load the corresponding slice of W_outᵀ (shape H×dim)
            #   layout of w_out_T: [H, dim] → stride dim
            w_ptrs = w_out_T_ptr + offs_d[:, None] * dim + offs_o[None, :]
            w_vals = tl.load(w_ptrs,
                             mask=mask_d[:, None] & mask_o[None, :],
                             other=0.0).to(tl.float32)                     # [BLOCK_D, BLOCK_O]

            # Accumulate dot‑product over the hidden dim
            #   norm_gate: [BLOCK_M, BLOCK_D]
            #   w_vals   : [BLOCK_D, BLOCK_O]
            acc += tl.dot(norm_gate, w_vals)                               # [BLOCK_M, BLOCK_O]

        # -------------------------------------------------------
        #   Store the tile (fp16)
        # -------------------------------------------------------
        out_fp_ptrs = out_fp16_ptr + offs_m[:, None] * dim + offs_o[None, :]
        tl.store(out_fp_ptrs,
                 acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_o[None, :])


# --------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)   # True → no masking
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (row‑wise) – use torch's highly tuned kernel
    # --------------------------------------------------------------
    # Normalise over the last dimension (dim)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # fp16 for the GEMMs

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # --------------------------------------------------------------
    # concatenate the four weight matrices and cast once to fp16
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # fused linear: (R, dim) @ (4·H, dim).T → (R, 4·H)
    fused = F.linear(x_norm, w_4h)          # fp16, shape [R, 4·H]

    left_raw   = fused[:, 0:H]               # [R, H]
    right_raw  = fused[:, H:2*H]              # [R, H]
    left_gate_raw   = fused[:, 2*H:3*H]       # [R, H]
    right_gate_raw  = fused[:, 3*H:4*H]       # [R, H]

    # separate linear for the output‑gate
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates, and element‑wise multiplication
    # --------------------------------------------------------------
    if not nomask:
        # mask → float16, shape [R, 1] for broadcasting
        mask_h = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw  * mask_h
        right_raw = right_raw * mask_h

    # in‑place sigmoid (saves temporaries)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_raw * left_gate          # [R, H] fp16
    right = right_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (N³) using a batched fp16 GEMM
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into the batch dimension
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # (B, H, N, N) views – non‑contiguous but supported by torch.matmul
    left_perm  = left.permute(0, 3, 1, 2)        # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)       # (B, H, N, N)

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))   # (B, H, N, N)

    # Bring back to (B, N, N, H) and make it contiguous for the kernel
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, output‑gate and final projection
    # --------------------------------------------------------------
    # flatten tensors for the kernel (row‑major layout)
    out_flat      = out.view(total_rows, H).contiguous()          # fp16
    out_gate_flat = out_gate.view(total_rows, H).contiguous()    # fp16

    # pre‑transpose final projection weight → shape (H, dim)
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()

    # allocate output buffer (fp16) – will be cast to fp32 at the end
    final_fp16 = torch.empty(total_rows, dim,
                             dtype=torch.float16,
                             device=device)

    # Triton launch configuration
    BLOCK_M = 128      # rows per program
    BLOCK_D = 32       # reduction tile over hidden dim
    BLOCK_O = 64       # output‑feature tile

    grid = (triton.cdiv(total_rows, BLOCK_M),)

    final_proj_layernorm_kernel[grid](
        out_flat,                     # out_ptr
        out_gate_flat,                # out_gate_ptr
        weights["to_out_norm.weight"],   # norm weight (fp32)
        weights["to_out_norm.bias"],     # norm bias   (fp32)
        w_out_T,                         # W_outᵀ (fp16)
        final_fp16,                      # output buffer (fp16)
        total_rows,
        dim,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )

    # reshape to the original 4‑D layout and cast back to fp32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
