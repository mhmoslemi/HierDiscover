import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the [B,N,N,dim] tensor (FP32 → FP16) using a
    custom Triton kernel.
2️⃣ Fuse each (proj + gate) pair for the left and right branches:
        (dim → hidden) + (dim → hidden) → (dim → 2·hidden) via a single
    `torch.nn.functional.linear`.  Apply the sigmoid gates and an
    optional mask (broadcast over the hidden dimension).
3️⃣ Compute the tri‑multiplicative update
        out_h(i,j) = Σ_k left_h(i,k)·right_h(j,k)
    as a batched GEMM on FP16 tensors (`torch.bmm`).  The hidden
    dimension is flattened into the batch axis (size B·hidden).
4️⃣ Second LayerNorm (over `hidden_dim`) with the same Triton kernel.
5️⃣ Multiply by the output gate, apply the final linear projection
    (hidden → dim) and cast back to FP32.

All heavy arithmetic is performed in FP16; the final result matches
the reference PyTorch module’s dtype (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var+eps)

    # ---------- Pass 2: write normalized values (γ·x̂+β) ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be ignored)
        weights_dict : {
            "norm.weight":          Tensor[dim],
            "norm.bias":            Tensor[dim],
            "left_proj.weight":     Tensor[hidden_dim, dim],
            "right_proj.weight":    Tensor[hidden_dim, dim],
            "left_gate.weight":     Tensor[hidden_dim, dim],
            "right_gate.weight":    Tensor[hidden_dim, dim],
            "out_gate.weight":      Tensor[hidden_dim, dim],
            "to_out_norm.weight":   Tensor[hidden_dim],
            "to_out_norm.bias":     Tensor[hidden_dim],
            "to_out.weight":        Tensor[dim, hidden_dim],
        }
        config_dict  : {"dim": int,
                       "hidden_dim": int,
                       "nomask": bool (optional, default True)}

    Returns:
        torch.Tensor[float32] shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Tuned block sizes (dim ≤ 384 ⇒ 2‑3 iterations)
    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    x_norm = x_norm_fp16.view(B, N, N, dim)               # FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left/right branches
    # --------------------------------------------------------------
    # Concatenate (proj, gate) weights: (2·hidden) × dim
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    rows = total_rows
    # Linear: (rows, dim) @ (2·H, dim).T → (rows, 2·H)
    left_concat = F.linear(x_norm.view(rows, dim), left_w)   # FP16
    right_concat = F.linear(x_norm.view(rows, dim), right_w)

    # Split into proj / pre‑gate
    left_proj = left_concat[:, :hidden_dim]          # (rows, H)
    left_gate_pre = left_concat[:, hidden_dim:]      # (rows, H)
    right_proj = right_concat[:, :hidden_dim]
    right_gate_pre = right_concat[:, hidden_dim:]

    # Output gate (shared across positions)
    out_gate = torch.sigmoid(
        F.linear(
            x_norm.view(rows, dim),
            weights["out_gate.weight"].to(dtype=torch.float16,
                                          device=device,
                                          non_blocking=True)
        )
    )   # (rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16,
                                         device=device,
                                         non_blocking=True)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate          # (rows, H)
    right_h = right_proj * right_gate       # (rows, H)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Merge H into batch dimension → [B·H, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # GEMM over the sequence dimension (Tensor‑Core enabled)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16, shape [B·H, N, N]

    # Reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) → FP16
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # --------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim)       # FP16
    gated = out_norm * out_gate                         # FP16

    final_fp16 = F.linear(
        gated.view(rows_h, hidden_dim),
        weights["to_out.weight"].to(dtype=torch.float16,
                                    device=device,
                                    non_blocking=True)
    )   # (rows_h, dim) FP16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)   # cast back

    return final
