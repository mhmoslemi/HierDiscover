"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened `[B·N·N, dim]` tensor (FP32 → FP16)
   using a custom Triton kernel.  This avoids the costly `torch.nn.LayerNorm`
   overhead and produces half‑precision data for the remaining ops.

2️⃣ Fuse *all* linear projections (left/right, their gates, and the output‑gate)
   into a **single** half‑precision GEMM:
        proj = X_norm_half @ W_all.T        # shape [R, 5·H]
   where `W_all` concatenates the five weight matrices
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`).

3️⃣ Split the result into the four projections, two gate pre‑activations and the
   output‑gate pre‑activation.  Apply the optional mask, sigmoid‐gates and
   element‑wise multiplication to obtain `left` and `right`.

4️⃣ Compute the multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched half‑precision GEMM of shape
   `(B·H, N, N) @ (B·H, N, N)ᵀ` performed with `torch.matmul` (cuBLAS/Tensor‑Core).

5️⃣ Second row‑wise LayerNorm (FP16) on the result.

6️⃣ Apply the output‑gate, a final half‑precision linear projection back to
   `dim`, and cast the output to FP32.

All heavy per‑row normalisations are done in Triton, while the large dense
matrix multiplications leverage cuBLAS Tensor‑Core kernels.  The implementation
uses mixed‑precision (FP16) throughout the compute‑heavy part and returns a
float32 tensor as required by the reference model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]          (float32)  gamma
    bias_ptr,       # const   * [dim]          (float32)  beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # Write normalized output y = (x-mean)*rstd * gamma + beta  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # beta
        out = norm * w[None, :] + b[None, :]                 # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim] (float32)
    → returns a half‑precision tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y

    w_all = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # FP16

    # Single half‑precision linear: [R, dim] @ [5·H, dim].T → [R, 5·H]
    proj_all = F.linear(x_norm, w_all)      # FP16

    H = hidden_dim
    # Split tensor into constituent parts
    left_proj_raw   = proj_all[:, 0 * H : 1 * H]   # [R, H]
    right_proj_raw  = proj_all[:, 1 * H : 2 * H]   # [R, H]
    left_gate_raw   = proj_all[:, 2 * H : 3 * H]   # [R, H]
    right_gate_raw  = proj_all[:, 3 * H : 4 * H]   # [R, H]
    out_gate_raw    = proj_all[:, 4 * H : 5 * H]   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gating and element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] (FP16) broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left = left_proj_raw * left_gate      # [R, H] FP16
    right = right_proj_raw * right_gate   # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # Back to 4‑D tensors
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # Move hidden dimension to axis 1 → shape (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ
    # Collapse batch dimension for a single torch.matmul call
    left_flat = left_bh.view(B * H, N, N)
    right_flat = right_bh.view(B * H, N, N)
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))   # [B·H, N, N]

    # Restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) → FP16
    # --------------------------------------------------------------
    out_flat2 = out.view(R, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat2,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        eps=eps,
    )                                   # FP16 [R, H]
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] (FP16)

    # --------------------------------------------------------------
    # 6️⃣ Output gate & final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # gated hidden state, FP16
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H] (FP16)
    )                                                   # FP16 [R, dim]

    # Cast back to FP32 as required by the reference model
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
