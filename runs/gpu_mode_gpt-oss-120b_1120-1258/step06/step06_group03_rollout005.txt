"""
TriMul (outgoing) forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened 4‑D input (B·N·N × dim) using a
   custom Triton kernel.  The kernel works on FP32 data and writes the
   normalized result as FP16.

2️⃣ Fuse the four linear layers (left/right projections + left/right gates)
   into a single half‑precision GEMM:
        X_norm_fp16 @ W_proj_gate.T → [R, 4·H]
   where R = B·N·N and H = hidden_dim.  The result is split into the four
   tensors `left_proj_raw`, `right_proj_raw`, `left_gate_raw`, `right_gate_raw`.

3️⃣ Compute the output‑gate with a second GEMM (same precision).

4️⃣ Apply the optional mask (broadcast over the hidden dimension), the
   sigmoid gates and combine the projections:
        left  = left_proj_raw  * sigmoid(left_gate_raw)
        right = right_proj_raw * sigmoid(right_gate_raw)

5️⃣ Multiplicative update: for each head `h` we need
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched GEMM:
        (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
   performed with `torch.bmm` (cuBLAS Tensor‑Core).

6️⃣ Second LayerNorm over the hidden dimension (again Triton, fp16 → fp16).

7️⃣ Apply the output‑gate, a final linear projection back to `dim`,
   and cast the result to float32 (the public API of the reference model).

Only the two LayerNorms are implemented in Triton; the large matrix
multiplications reuse highly‑optimised cuBLAS kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input can be float32 or float16,
# output is always float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const   * [rows, dim]   (float32 or float16)
    y_ptr,                # mutable * [rows, dim]   (float16)
    weight_ptr,           # const   * [dim]        (float32) gamma
    bias_ptr,             # const   * [dim]        (float32) beta
    rows,                 # int32   (runtime)
    dim,                  # int32   (runtime)
    eps,                  # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- accumulate sum & sum of squares ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dim in blocks
    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile of the input (fp16 or fp32) and cast to fp32
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ---------- write normalized output ----------
    for d_off in range(0, num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # (x - mean) * rstd
        norm = (x - mean[:, None]) * rstd[:, None]

        # gamma / beta are FP32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # FP32 result
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm implemented in Triton.

    Parameters
    ----------
    x : torch.Tensor
        Input of shape [rows, dim]. dtype can be float32 or float16.
    weight, bias : torch.Tensor
        LayerNorm parameters (float32) of shape [dim].
    eps : float
        Numerical stability constant.
    block_m, block_d : int
        Tile sizes for the kernel.

    Returns
    -------
    torch.Tensor
        Normalized output of shape [rows, dim] and dtype float16.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim,
                    dtype=torch.float16,
                    device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear / LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]           # input / output channel dimension
    hidden_dim   = config["hidden_dim"]    # hidden dimension (H)
    eps          = 1e-5
    device       = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total number of rows (B·N·N)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    # flatten to [R, dim] (float32)
    x_flat = input_tensor.view(R, dim).contiguous()
    x_norm_half = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # gamma FP32
        weights["norm.bias"],     # beta  FP32
        eps=eps,
    )                               # FP16, shape [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear layers
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection / gate weight matrices (each [H, dim])
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)  # shape [4·H, dim]

    # X_norm (R, dim) @ W_proj_gate.T (dim, 4·H) → (R, 4·H)
    proj_gate = F.linear(x_norm_half, w_proj_gate)     # FP16

    # split the concatenated output
    left_proj_raw   = proj_gate[:, 0:H]               # (R, H)
    right_proj_raw  = proj_gate[:, H:2 * H]           # (R, H)
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]       # (R, H)
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]       # (R, H)

    # output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16),   # (H, dim)
    )                                                   # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present), sigmoid gating, apply gates
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1] (FP16) broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid to avoid extra allocations
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # apply gating
    left = left_proj_raw * left_gate          # (R, H) FP16
    right = right_proj_raw * right_gate       # (R, H) FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # back‑to 4‑D shape
    left  = left.view(B, N, N, H)          # (B, N, N, H)
    right = right.view(B, N, N, H)         # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # (B, N, N, H)

    # bring hidden dimension to the batch axis: (B, H, N, N)
    left_bh  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_bh = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # fuse batch and head dimensions for a single bmm call
    LH = B * H
    left_bh  = left_bh.view(LH, N, N)          # (B·H, N, N)
    right_bh = right_bh.view(LH, N, N)         # (B·H, N, N)

    # batched matrix multiplication -> (B·H, N, N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # restore original ordering: (B, N, N, H)
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (H) → FP16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # gamma FP32
        weights["to_out_norm.bias"],     # beta  FP32
        eps=eps,
    )                                   # FP16, shape [R, H]

    out_norm = out_norm_half.view(B, N, N, H)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # (B, N, N, H) FP16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )                                                   # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
