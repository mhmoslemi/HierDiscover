"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (float32 → float16) using a Triton kernel.
2️⃣ Fuse the four linear projections (left/right proj & left/right gate) into a
   single GEMM (`F.linear`).  The `out_gate` projection is a second GEMM.
3️⃣ Apply optional pairwise mask (broadcast over the hidden dimension).
4️⃣ Apply sigmoid to the three gate tensors and combine:
        left  = (xW_left  * mask) * sigmoid(xW_left_gate)
        right = (xW_right * mask) * sigmoid(xW_right_gate)
        out_gate = sigmoid(xW_out_gate)
5️⃣ Reshape to [B, N, N, H] (H = hidden_dim) and compute the cubic
   multiplicative update with a *batched* matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to `left @ right.transpose(-2,-1)` for every head.
6️⃣ Row‑wise LayerNorm on the result (fp16) with a second Triton kernel.
7️⃣ Multiply by `out_gate`, project back to the original `dim` with a final
   linear layer and cast to float32.

All heavy linear work stays in half‑precision (Tensor‑Core friendly).  Only
the two LayerNorms are implemented in Triton; the rest uses highly‑optimised
PyTorch kernels (`F.linear`, `torch.bmm`).  The implementation works for the
full range of shapes used in the tests (B∈{1,2}, N up to 1024, dim∈{128,384},
hidden_dim=128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 input → float16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    w_ptr,          # const   * [dim]               (float32)  γ
    b_ptr,          # const   * [dim]               (float32)  β
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute per‑row mean / variance in fp32 and store the normalised result
    in fp16:   y = ((x-μ)/√(σ²+eps))*γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)             # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute sum and sum‑of‑squares for mean / var (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalised values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                 # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights_dict : dictionary with all model parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)
    eps         = 1e-5
    device      = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N           # total number of “rows” for layer‑norms
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16,
                         device=device)                       # [R, dim] fp16

    BLOCK_M = 256               # rows per program
    BLOCK_D = 64                # columns per program (good up to dim=384)
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear layers (one GEMM)
    # ------------------------------------------------------------------
    # stack the four projection/gate weight matrices → shape [4*H, dim]
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [4*H, dim]

    proj_gate = F.linear(x_norm, w_4h)  # [R, 4*H] fp16

    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2*H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]

    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                          # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → broadcast over hidden dim
        mask_h = mask_tensor.reshape(R, 1).to(torch.float16)  # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update via batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    ⇒ for each head h: out_h = left_h @ right_h^T
    # ------------------------------------------------------------------
    # bring head dimension to the batch axis
    left_4d   = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_4d  = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [B*H, N, N]

    # right needs to be transposed on the last two dims
    out_4d = torch.bmm(left_4d, right_4d.transpose(1, 2))        # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_4d.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dim → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)            # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate & final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                    # gating, fp16

    final = F.linear(
        out_norm.view(R, H),                      # [R, H]
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                             # [R, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)  # cast to fp32

    return final
