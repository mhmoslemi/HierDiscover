"""
TriMul outgoing kernel (AlphaFold‑3) – forward pass only.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16) – Triton.
2️⃣ Fuse the four linear projections/gates 
      (left_proj, right_proj, left_gate, right_gate)  
   into a single half‑precision GEMM:   [R,dim] @ [4·H,dim]ᵀ → [R,4·H].
3️⃣ Compute the output‑gate with a second half‑precision GEMM.
4️⃣ (Optional) apply the pair‑wise mask, then sigmoids for the gates.
5️⃣ Form the gated tensors `left` and `right` (shape B,N,N,H) and
   perform the triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   Implemented as a batched GEMM (torch.bmm) on half‑precision data.
6️⃣ Second LayerNorm over the hidden dimension – Triton.
7️⃣ Multiply by the output‑gate and project back to `dim` with a final
   half‑precision linear layer; cast the result to float32.

Only the two row‑wise LayerNorms are written in Triton; all large matrix
multiplications use cuBLAS (torch.linear / torch.bmm) which is optimal on
H100. The implementation supports both masked / un‑masked variants and
arbitrary `dim` / `hidden_dim` (including the 384‑dim case used in the
tests).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 input → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]    (float32)
    y_ptr,          # mutable *[total_rows, dim]    (float16)
    weight_ptr,     # const   *[dim]                (float32)   γ
    bias_ptr,       # const   *[dim]                (float32)   β
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β
    The reduction is done in fp32; the output is written as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized values γ·(x‑μ)·rstd + β (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – may be unused
        - weights_dict : mapping with all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total rows for row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)  – row‑wise
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates (single GEMM)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices: [left, right, left_gate, right_gate]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                               # [4·H, dim]

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (half‑precision GEMM)
    proj_gate = F.linear(x_norm, w_concat)          # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )  # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if required) and sigmoids for all gates
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [R,1] (fp16) for broadcasting
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H]  fp16
    right = right_proj_raw * right_gate    # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # move head dimension into the batch axis for bmm
    left  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # reshape out_gate to match
    out_gate = out_gate.view(B, N, N, H).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16) – row‑wise
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)      # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate            # [B,N,N,H]  fp16

    final_half = F.linear(
        out_norm.view(R, H),                     # [R, H]
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )  # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)  # back to fp32
    return final
