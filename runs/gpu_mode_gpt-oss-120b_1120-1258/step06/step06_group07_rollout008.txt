    ln_weight_ptr,      # const float* [H]
    ln_bias_ptr,        # const float* [H]
    w_out_T_ptr,        # const float* [H, D]  (transposed)
    out_fp16_ptr,       # *mut half* [R, D]
    total_rows: tl.int32,
    hidden: tl.constexpr,
    output_dim: tl.constexpr,

    BLOCK_M: tl.constexpr,

    BLOCK_D: tl.constexpr,

Implement algorithm:

```python
pid = tl.program_id(0)
row_start = pid * BLOCK_M
offs_m = row_start + tl.arange(0, BLOCK_M)
mask_m = offs_m < total_rows

# Compute mean & variance per row
sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
for k in range(0, tl.cdiv(hidden, BLOCK_K)):
    offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = offs_k < hidden
    # Load out half -> convert to float
    vals = tl.load(out_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                   mask=mask_m[:, None] & mask_k[None, :],
                   other=0.0).to(tl.float32)
    sum_val += tl.sum(vals, axis=1)
    sum_sq += tl.sum(vals * vals, axis=1)
mean = sum_val / hidden
var = sum_sq / hidden - mean * mean
rstd = tl.rsqrt(var + eps)

# Initialize accumulator for final dim
acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

for k in range(0, tl.cdiv(hidden, BLOCK_K)):
    offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = offs_k < hidden

    # Load out and gate
    out_vals = tl.load(out_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
    gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

    # Normalize
    norm = (out_vals - mean[:, None]) * rstd[:, None]   # [BLOCK_M, BLOCK_K]

    # Apply layernorm weight & bias
    w_ln = tl.load(ln_weight_ptr + offs_k, mask=mask_k, other=0.0)
    b_ln = tl.load(ln_bias_ptr   + offs_k, mask=mask_k, other=0.0)
    norm = norm * w_ln[None, :] + b_ln[None, :]

    # Apply gate
    norm = norm * gate_vals

    # Load weight slice for linear
    w_to = tl.load(w_out_T_ptr + offs_k[:, None] * output_dim + tl.arange(0, BLOCK_D)[None, :],
                   mask=mask_k[:, None] & (tl.arange(0, BLOCK_D)[None, :] < output_dim),
                   other=0.0)   # shape [BLOCK_K, BLOCK_D]

    # Multiply and accumulate
    # Use dot product: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_D) -> (BLOCK_M, BLOCK_D)
    acc += tl.dot(norm, w_to)

"""
Highly‑optimised Triton + PyTorch forward for the **outgoing** TriMul
operator (AlphaFold‑3).  

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input → FP16 (custom Triton kernel).
2️⃣  Fused linear + gating for the *left* and *right* arms (single GEMM each)
    – all weights are stored in FP16.
3️⃣  Optional binary mask applied to the hidden projections.
4️⃣  Batched GEMM that implements
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    (uses torch.bmm – cuBLAS/Tensor‑Core on H100).
5️⃣  Output‑gate = σ(x_norm·W_out_gate) (already in FP16).
6️⃣  **Fused** LayerNorm + output‑gate + final linear projection:
    – a single Triton kernel computes per‑row mean/var,
      normalises, applies `to_out_norm` γ/β, multiplies by the gate,
      and immediately performs the dense projection `to_out`.
    – This eliminates an extra write‑read of the intermediate tensor.
7️⃣  Cast the result back to float32.

All kernels are written for H100 (Tensor‑Core FP16) and keep the
hidden dimension ≤ 384, output dimension ≤ 384, so small static blocks
fit into registers.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*   [total_rows, dim]   (FP32)
    y_ptr,               # *mut   half*    [total_rows, dim]   (FP16)
    weight_ptr,          # *const float*   [dim]               (γ)
    bias_ptr,            # *const float*   [dim]               (β)
    total_rows: tl.int32,
    dim: tl.constexpr,               # static (≤ 384)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: compute mean & variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 6) Fused LayerNorm + output‑gate + final linear projection
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    # inputs
    out_ptr,                # *const half*   [total_rows, hidden]   (post‑mul)
    gate_ptr,               # *const half*   [total_rows, hidden]   (out_gate)
    ln_weight_ptr,          # *const float*  [hidden]               (γ of to_out_norm)
    ln_bias_ptr,            # *const float*  [hidden]               (β of to_out_norm)
    w_out_T_ptr,            # *const half*   [hidden, dim]          (to_out weight transposed)
    # output
    out_fp16_ptr,           # *mut   half*   [total_rows, dim]     (final output)
    # scalars
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    # block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    # 2‑D grid: rows × output‑dim tiles
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_D

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    offs_d = col_start + tl.arange(0, BLOCK_D)          # [BLOCK_D]
    mask_d = offs_d < dim

    # -----------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance over the hidden axis
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        vals = tl.load(out_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2️⃣  Accumulate final projection (norm → gate → linear)
    # -----------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden

        # load raw values & gate
        out_vals = tl.load(out_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_K]

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_K]

        # normalise
        norm = (out_vals - mean[:, None]) * rstd[:, None]             # FP32

        # apply to_out_norm γ/β
        w_ln = tl.load(ln_weight_ptr + offs_k, mask=mask_k, other=0.0)   # γ
        b_ln = tl.load(ln_bias_ptr   + offs_k, mask=mask_k, other=0.0)   # β
        norm = norm * w_ln[None, :] + b_ln[None, :]                     # FP32

        # apply output gate
        norm = norm * gate_vals                                        # FP32

        # load slice of the transposed final weight (hidden → dim)
        w_out = tl.load(
            w_out_T_ptr + offs_k[:, None] * dim + offs_d[None, :],
            mask=mask_k[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)                                 # [BLOCK_K, BLOCK_D]

        # accumulate matmul contribution
        acc += tl.dot(norm, w_out)                                    # [BLOCK_M, BLOCK_D]

    # -----------------------------------------------------------------
    # 3️⃣  Write result (cast back to FP16)
    # -----------------------------------------------------------------
    tl.store(out_fp16_ptr + offs_m[:, None] * dim + offs_d[None, :],
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                     # C
    hidden = cfg["hidden_dim"]              # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                # rows when flattening the 4‑D tensor

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()     # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)               # [R, C] half

    BLOCK_M = 128
    BLOCK_D = 64 if dim <= 256 else 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear + gate (left / right) – fused GEMMs
    # ------------------------------------------------------------------
    # Cast all projection / gate matrices to FP16 once
    left_proj_w  = weights["left_proj.weight"].half().contiguous()
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # ---- left arm ----------------------------------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)          # (2*H, C)
    left_comb   = F.linear(x_norm_fp16, left_comb_w)                   # (R, 2*H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw).half()
    left = left_proj_raw * left_gate                                   # (R, H)

    # ---- right arm ---------------------------------------------------
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)      # (2*H, C)
    right_comb   = F.linear(x_norm_fp16, right_comb_w)                # (R, 2*H)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw).half()
    right = right_proj_raw * right_gate                                 # (R, H)

    # ------------------------------------------------------------------
    # 2a) Optional binary mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, hidden)       # (B, N, N, H)
    right = right.view(B, N, N, hidden)      # (B, N, N, H)

    # prepare for batched bmm: (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)    # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)   # (B*H, N, N)

    # batched half‑precision matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat)                         # (B*H, N, N), half

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Output gate (already from the *normed* input)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)                 # (R, H)
    out_gate = torch.sigmoid(out_gate_raw).half()                    # (R, H)

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + gate + final linear projection
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()            # (R, H) half
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                             device=device)                         # (R, dim) half

    # transposed final weight: (hidden, dim) for contiguous column access
    w_out_T = weights["to_out.weight"].transpose(0, 1).contiguous().half()

    # block configuration – fits hidden ≤ 384 and dim ≤ 384
    BLOCK_K = 32                         # tile over hidden dimension
    BLOCK_D = 64                         # tile over output dim
    grid_rows = triton.cdiv(total_rows, BLOCK_M)
    grid_cols = triton.cdiv(dim, BLOCK_D)
    grid = (grid_rows, grid_cols)

    fused_norm_gate_linear_kernel[grid](
        out_flat,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        w_out_T,
        final_fp16,
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Cast back to float32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
