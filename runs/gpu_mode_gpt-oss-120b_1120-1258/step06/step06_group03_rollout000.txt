"""
custom_kernel
==============

Forward pass of the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (float32 → float16) over the flattened input
   tensor `[B·N·N, dim]`.  Implemented with a fast Triton kernel.

2️⃣ Fuse the four linear projections/gates (`left_proj`, `right_proj`,
   `left_gate`, `right_gate`) into a single half‑precision GEMM:
   `X_norm_half @ W_proj_gate.T  →  [R, 4·H]`
   where `R = B·N·N` and `H = hidden_dim`.  The result is split into the
   four tensors.

3️⃣ Compute the output‑gate with a second GEMM.

4️⃣ (Optional) apply the pairwise mask, run the sigmoid‑gates and
   multiply the projected tensors → `left`, `right`.

5️⃣ Reshape to `[B, N, N, H]`, permute to `[B, H, N, N]` and perform the
   batched matrix multiplication
   `out[b, h] = left[b, h] @ right[b, h].T`   (`N³` update).

6️⃣ Second LayerNorm over the hidden dimension (`H`) – again a Triton
   kernel.

7️⃣ Apply the output‑gate, a final half‑precision linear projection back
   to `dim` and cast the result to float32.

All heavy per‑row normalisations are done in Triton; large dense matrix
multiplications reuse cuBLAS Tensor‑Core kernels via `torch.nn.functional`
operations.  The implementation works for the full test matrix of
configs (different sequence lengths, dims, hidden dims and optional mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 input → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32) – γ
    bias_ptr,       # const   * [dim]               (float32) – β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute per‑row mean and variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+ε)

    # --------------------------------------------------------------
    # 2️⃣  Write normalized output y = (x-μ)/σ * γ + β  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Apply row‑wise LayerNorm to a 2‑D tensor ``x`` of shape [rows, dim]
    (float32) → returns half‑precision tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear / LayerNorm parameters:
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight"
            * "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict : must contain "dim", "hidden_dim" and "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                         # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # reshape to [R, dim] (contiguous)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_half = _layernorm_triton(
        x_flat,
        weights["norm.weight"],          # γ   (float32)
        weights["norm.bias"],            # β   (float32)
        eps=eps,
    )                                    # -> [R, dim] (float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates (four heads → 4·H outputs)
    # ------------------------------------------------------------------
    # Concatenate the four projection/gate weight matrices (shape H×dim)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # Single GEMM: X_norm_half @ W.T → [R, 4·H]  (half‑precision)
    proj_gate = F.linear(x_norm_half, w_proj_gate)   # fp16

    # Split the result
    left_proj_raw   = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]           # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]           # [R, H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                  # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gating, element‑wise multiplication
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw  * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (works for fp16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate       # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    # Back to 4‑D layout
    left = left.view(B, N, N, H)            # [B, N, N, H]
    right = right.view(B, N, N, H)          # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)    # [B, N, N, H]

    # Move hidden dim to the second axis → [B, H, N, N] (contiguous)
    left_bh = left.permute(0, 3, 1, 2).contiguous()
    right_bh = right.permute(0, 3, 1, 2).contiguous()

    # Batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_bh = torch.matmul(left_bh, right_bh.transpose(-2, -1))   # [B, H, N, N]

    # Restore original layout
    out = out_bh.permute(0, 2, 3, 1).contiguous()               # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension) → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_half = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        eps=eps,
    )                                    # [R, H] fp16

    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output gate & final projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
