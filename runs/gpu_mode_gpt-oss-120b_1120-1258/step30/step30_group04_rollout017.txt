"""
TriMul (outgoing) – fast Triton implementation
================================================

The operator takes a 4‑D pairwise embedding tensor X ∈ ℝ^{B×N×N×C}
and produces an output of the same shape.  The forward pass consists of

1️⃣  Row‑wise LayerNorm over the last channel (C)                     →  FP16
2️⃣  A *single* linear projection that produces the five tensors used
    in the update (left_proj, right_proj, left_gate, right_gate, out_gate)
    – all kept in FP16.
3️⃣  Fuse the optional mask, the sigmoid gates and store the three
    tensors  L, R, G in layout [B, H, N, N] (H = hidden_dim).
4️⃣  Batched matrix multiplication L·Rᵀ (equivalent to the einsum
    “... i k d, ... j k d → … i j d”).  This is the N³ work‑horse.
5️⃣  Row‑wise mean / rsqrt‑var of the contracted tensor over the hidden
    dimension H (still in FP16, reduction in FP32).
6️⃣  Final LayerNorm + out‑gate multiplication + output projection
    (H → C).  This whole step is fused into one Triton kernel so that
    the hidden dimension is never written back to memory a second time.

All heavy work is done in FP16 with FP32 reductions, which lets the
kernel fully exploit the Tensor‑Core units of an H100.  The implementation
still respects the optional `mask` (broadcasted over the hidden dimension)
and works for any batch size {1,2}, sequence length {128…1024} and
channel dimensions required by the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # [R, C] input FP32, output FP16
    w_ptr, b_ptr,               # [C] LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)    # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized result ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fused_proj_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask broadcasted over hidden dim
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                        # [BLOCK_R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride for a single hidden plane
    # base offset of the (b,i,k) element for h = 0
    base_offset = b * (H * hidden_stride) + i * N + k           # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        # view into fused projection: concat([L,R,LG,RG,OG])
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]            # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise (over H) mean / rsqrt‑var of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N] FP16
    mu_ptr, rstd_ptr, # [R] FP32   (R = B·N·N)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j        # start of h=0 plane

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Final output:  LayerNorm(H) → out‑gate → Linear(H→C)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N] FP16   (contracted tensor O)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    mu_ptr, rstd_ptr, # [R] FP32
    ln_w_ptr, ln_b_ptr,                # [H] FP32  (to_out_norm weight & bias)
    w_out_T_ptr,                       # [H, C] FP16 (transposed to_out weight)
    out_final_ptr,                     # [B, N, N, C] FP32
    total_rows, H, N, C,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (flattened B×N×N)
    pid_n = tl.program_id(1)   # output‑channel tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)    # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)    # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < C

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j    # start of h=0 for (b,i,j)

    # per‑row mean / rsqrt‑var (FP32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor O   (FP16→FP32)
        out_tile = tl.load(out_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)                 # [BM, BK]

        # load out‑gate (FP16→FP32)
        gate_tile = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)               # [BM, BK]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BK]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BK]

        # ((out‑μ)·rstd·γ + β)   –   element‑wise across hidden dim
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by out‑gate
        term = norm * gate_tile                     # [BM, BK]

        # cast to FP16 for tensor‑core dot
        term_fp16 = term.to(tl.float16)

        # weight slice (H, C) → (BK, BN)
        w_ptr = w_out_T_ptr + offs_k[:, None] * C + offs_n[None, :]   # [BK, BN]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [BK, BN]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)               # [BM, BN]

    # write final FP32 result
    dst_ptr = out_final_ptr + offs_m[:, None] * C + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    Returns
    -------
    torch.Tensor – shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into one matmul (R × dim) × (dim × 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, C]

    proj = F.linear(x_norm, fused_w)                       # [R, 5·H] FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask & sigmoid gates → L, R, G
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fused_proj_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core N³ contraction (batched GEMM L · Rᵀ)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑var over the hidden dimension H
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 → whole hidden dim in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out‑gate + projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, C]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows per program (tiles over R)
    BLOCK_K_OUT = 32           # hidden‑dim chunk per iteration (≤ H)
    BLOCK_N_OUT = 128          # output‑channel chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
