@triton.jit
def apply_ln_gate_kernel(
    out_ptr,      # pointer to (R, H) FP16
    mu_ptr,
    rstd_ptr,
    gate_ptr,
    ln_w_ptr,
    ln_b_ptr,
    outg_ptr,
    R, H,
    eps,                 # maybe not needed
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < R

    # Load mu and rstd for this block (broadcast later)
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)      # fp32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)  # fp32

    # Repeat for hidden tiles
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < H

        # Load out values (FP16)
        out_val = tl.load(out_ptr + offs_m[:, None] * H + offs_h[None, :],
                          mask=mask_m[:, None] & mask_h[None, :],
                          other=0.0)  # [BLOCK_M, BLOCK_H]
        # Load gate (FP16)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0)

        # Broadcast mu/rstd to tile
        mu_tile = mu[:, None]   # shape [BLOCK_M, 1]
        rstd_tile = rstd[:, None]

        # Compute normalized
        out_fp32 = out_val.to(tl.float32)  # cast
        norm_fp32 = (out_fp32 - mu_tile) * rstd_tile

        # Load ln weight and bias
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)  # fp32
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # fp32

        norm_ln_fp32 = norm_fp32 * gamma[None, :] + beta[None, :]

        # Multiply gate (cast to fp32)
        gate_fp32 = gate_val.to(tl.float32)
        gated_fp32 = norm_ln_fp32 * gate_fp32

        # Cast back to fp16
        gated_fp16 = gated_fp32.to(tl.float16)

        # Store result
        tl.store(outg_ptr + offs_m[:, None] * H + offs_h[None, :],
                 gated_fp16,
                 mask=mask_m[:, None] & mask_h[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only kernel implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣  Fuse the five linear mappings (left, right, left_gate, right_gate, out_gate)
    into a single GEMM → (R, 5·H) where R = B·N·N.
3️⃣  Decode the fused projection, apply the optional pair‑mask,
    compute sigmoid gates and write the three tensors `left`, `right`,
    `out_gate` in layout [B, H, N, N] (FP16) via a Triton kernel.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → [B, H, N, N] (FP16)
5️⃣  Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
    (Triton kernel) → µ and rstd per (b,i,j) element.
6️⃣  Fuse the second LayerNorm, the `out_gate` multiplication and the final
    projection into:
        hidden_gated = ((out‑µ)·rstd·γ + β) ⊙ out_gate   (FP16)
    This is done by a small Triton kernel.
7️⃣  The huge mat‑mul hidden_gated @ to_out.weight (cuBLAS) yields the
    final tensor of shape [B, N, N, dim] (float32).

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed‑precision
(FP32 for statistics, FP16 for the bulk of the work).  No gradients are
computed – this is a forward‑only implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]  FP32 input
    y_ptr,   # [R, dim]  FP16 output
    w_ptr,   # [dim]     γ  (FP32)
    b_ptr,   # [dim]     β  (FP32)
    rows,    # scalar R
    dim,     # scalar dim
    eps,     # scalar eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for h = 0 in the target tensors
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # Index into the fused projection: [left|right|lg|rg|og]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection (pre‑linear)
# ----------------------------------------------------------------------
@triton.jit
def apply_ln_gate_kernel(
    out_ptr,       # [R, H] FP16      (pairwise contracted tensor)
    mu_ptr,        # [R]    FP32
    rstd_ptr,      # [R]    FP32
    gate_ptr,      # [R, H] FP16     (out_gate)
    ln_w_ptr,      # [H]    FP32     (to_out_norm.weight)
    ln_b_ptr,      # [H]    FP32     (to_out_norm.bias)
    outg_ptr,      # [R, H] FP16     (output – gated & normalized)
    R, H,
    eps,                     # not used, kept for API compatibility
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    # Load µ and rstd for the whole block (broadcast later)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # FP32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # FP32

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)                # [BLOCK_H]
        mask_h = offs_h < H

        # Load the raw out and gate values (FP16)
        out_val = tl.load(
            out_ptr + offs_m[:, None] * H + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        gate_val = tl.load(
            gate_ptr + offs_m[:, None] * H + offs_h[None, :],
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Broadcast µ / rstd across the hidden‑dim tile
        mu_tile   = mu[:, None]    # [BLOCK_M,1]
        rstd_tile = rstd[:, None]  # [BLOCK_M,1]

        # Normalization (FP32)
        out_f32 = out_val.to(tl.float32)
        norm_f32 = (out_f32 - mu_tile) * rstd_tile

        # LayerNorm parameters (γ, β)
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        norm_ln_f32 = norm_f32 * gamma[None, :] + beta[None, :]

        # Apply out_gate (sigmoid already applied in previous kernel)
        gate_f32 = gate_val.to(tl.float32)
        gated_f32 = norm_ln_f32 * gate_f32

        # Store back as FP16
        tl.store(
            outg_ptr + offs_m[:, None] * H + offs_h[None, :],
            gated_f32.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)         # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 → fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and produce gated hidden tensor
    # ------------------------------------------------------------------
    # Permute to [B, N, N, H] → flatten to (R, H)
    out_perm     = out.permute(0, 2, 3, 1).contiguous()          # (B, N, N, H)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    hidden_gated = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUTG = 128
    BLOCK_H_OUTG = 64
    grid_gate = (triton.cdiv(R, BLOCK_M_OUTG),)
    apply_ln_gate_kernel[grid_gate](
        out_perm,
        mu,
        rstd,
        out_gate_perm,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        hidden_gated,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_OUTG,
        BLOCK_H=BLOCK_H_OUTG,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS)
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16).contiguous()   # [dim, H]
    final_fp16 = torch.nn.functional.linear(hidden_gated, w_out)   # (R, dim) FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)        # (B, N, N, dim) FP32

    return final
