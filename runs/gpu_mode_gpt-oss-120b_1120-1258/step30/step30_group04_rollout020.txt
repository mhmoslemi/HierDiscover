3️⃣  Split + mask + sigmoid gating (produces L, R and G)
4️⃣  Batched GEMM   L @ Rᵀ                              → O ∈ ℝ^{B×H×N×N}
5️⃣  **Fused** per‑row statistics (μ, σ) and the second LayerNorm,
    the out‑gate multiplication and the final projection
    to the original channel dimension C.

Only the heaviest work is done on the GPU; all reductions are kept in‑register,
and the final projection uses Tensor‑Core‑accelerated `tl.dot`.  The separate
row‑wise‑stats kernel from the reference implementation is eliminated, which
cuts one global‑memory pass over the O tensor.

The entry point is `custom_kernel(data)` where `data` is a tuple
(input, mask, weights, config).  The returned tensor is float32
[B, N, N, C].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # [R, C] input FP32, output FP16
    w_ptr, b_ptr,               # [C] LN weight & bias (FP32)
    rows, C, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(C, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < C

        ptr = x_ptr + offs_m[:, None] * C + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized tensor ----
    for d_off in range(0, tl.cdiv(C, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < C

        ptr = x_ptr + offs_m[:, None] * C + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * C + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   fused projection (FP16)
    left_ptr,          # [B, H, N, N]   ← L (FP16)
    right_ptr,         # [B, H, N, N]   ← R (FP16)
    out_gate_ptr,      # [B, H, N, N]   ← G (FP16)
    mask_ptr,          # [R] (FP16) – optional, broadcasted over H
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flattened index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                       # [BLOCK_R, 1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride between two H‑planes in the final layout
    hidden_stride = N * N
    # base offset of the (b,i,k) element for h = 0
    base_offset = b * (H * hidden_stride) + i * N + k        # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view into the fused projection: [L,R,LG,RG,OG] concatenated along H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        # destination offset for each (b,i,k) plane
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Final fused kernel:
#    – per‑row mean / rsqrt‑var of O,
#    – second LayerNorm (γ,β),
#    – multiplication by out‑gate,
#    – projection to the original channel dimension.
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,               # [B, H, N, N]   O = L @ Rᵀ   (FP16)
    gate_ptr,              # [B, H, N, N]   out_gate    (FP16)
    w_gamma_ptr,           # [H, dim]       γ·W_out     (FP16)
    w_beta_ptr,            # [H, dim]       β·W_out     (FP16)
    out_final_ptr,         # [B, N, N, dim] result      (FP32)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)          # rows tile
    pid_n = tl.program_id(1)          # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from flattened row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    # start of hidden‑plane h = 0 for this (b,i,j)
    base = b * (H * hidden_stride) + i * N + j

    # accumulators
    sum_o   = tl.zeros([BLOCK_M], dtype=tl.float32)   # Σ O_h
    sum_o2  = tl.zeros([BLOCK_M], dtype=tl.float32)   # Σ O_h²

    acc1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ (O·gate)·(γ·W)
    acc2 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ   gate   ·(γ·W)
    acc3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ   gate   ·(β·W)

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # O and gate slices
        o_tile = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=0.0)                     # [BLOCK_M, BLOCK_H]  FP16
        g_tile = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=0.0)                     # [BLOCK_M, BLOCK_H]  FP16

        # update mean / var components (in FP32)
        o_fp32 = o_tile.to(tl.float32)
        sum_o  += tl.sum(o_fp32, axis=1)
        sum_o2 += tl.sum(o_fp32 * o_fp32, axis=1)

        # left matrices for the three dot‑products
        left1 = o_tile * g_tile               # (O·gate)
        left2 = g_tile                         # gate

        # weight tiles (γ·W) and (β·W)
        w_gamma_tile = tl.load(w_gamma_ptr + offs_h[:, None] * dim + offs_n[None, :],
                               mask=mask_h[:, None] & mask_n[None, :],
                               other=0.0)          # [BLOCK_H, BLOCK_N] FP16
        w_beta_tile  = tl.load(w_beta_ptr +  offs_h[:, None] * dim + offs_n[None, :],
                               mask=mask_h[:, None] & mask_n[None, :],
                               other=0.0)          # [BLOCK_H, BLOCK_N] FP16

        # three fused GEMM‑style updates
        acc1 += tl.dot(left1, w_gamma_tile).to(tl.float32)   # Σ (O·gate)·(γ·W)
        acc2 += tl.dot(left2, w_gamma_tile).to(tl.float32)   # Σ   gate   ·(γ·W)
        acc3 += tl.dot(left2, w_beta_tile ).to(tl.float32)   # Σ   gate   ·(β·W)

    # per‑row statistics
    mu   = sum_o / H
    var  = sum_o2 / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # Y = rstd·(acc1 - mu·acc2) + acc3
    y = rstd[:, None] * (acc1 - mu[:, None] * acc2) + acc3

    # write result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, y, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                   # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection for the five heads (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                         # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoids → left, right, out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (left·rightᵀ) – core O tensor
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                         # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)                       # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate + final projection (fused)
    # --------------------------------------------------------------
    # pre‑compute γ·W_out and β·W_out  (both FP16)
    gamma = w["to_out_norm.weight"]                            # [H] FP32
    beta  = w["to_out_norm.bias"]                              # [H] FP32
    w_out = w["to_out.weight"].t().contiguous().to(torch.float16)   # [H, dim]

    w_gamma = (w_out * gamma[:, None].to(torch.float16)).contiguous()
    w_beta  = (w_out * beta [:, None].to(torch.float16)).contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128         # rows per program
    BLOCK_H_OUT = 32          # hidden‑dim chunk per iteration
    BLOCK_N_OUT = 128         # output‑dim chunk

    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w_gamma,
        w_beta,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_H=BLOCK_H_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
```
