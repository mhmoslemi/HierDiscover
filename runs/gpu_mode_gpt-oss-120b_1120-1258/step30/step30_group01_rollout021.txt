def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = bool(cfg.get("nomask", True))
    eps=1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm + cast to half
    x_flat = x.reshape(R, dim)
    x_norm_fp32 = torch.nn.functional.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)

    # 2. Fuse projection
    # concat five weight matrices of shape [H, dim] each
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [5*H, dim]

    proj = F.linear(x_norm, fused_w)   # [R, 5*H] half

    # 3. Split / mask / gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4. Batch GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)  # fp16

    # 5. per-row mean/var using torch
    mu = out.mean(dim=1).reshape(-1).to(torch.float32)   # [R]
    var = out.var(dim=1, unbiased=False).reshape(-1).to(torch.float32)
    rstd = (var + eps).rsqrt()

    # 6. final output
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],      # gamma
        w["to_out_norm.bias"],        # beta
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final

"""
Outgoing TriMul (AlphaFold‑3) – highly‑fused forward‑only implementation.

Algorithm
---------
1️⃣  Pairwise LayerNorm (FP32→FP16) on the flattened [B·N·N, C] tensor using
    torch.nn.functional.layer_norm (highly‑optimized cuDNN).

2️⃣  Fuse the five linear maps (left‑proj, right‑proj, left‑gate, right‑gate,
    out‑gate) into a single GEMM (cuBLAS).  The result has shape
    [R, 5·H] where   R = B·N·N  and  H = hidden_dim.

3️⃣  Triton kernel ``split_fuse_gate_kernel``:
    * decodes the flat index (b,i,k) → (b,i,j) layout,
    * applies the optional pair mask,
    * evaluates the sigmoid gates,
    * writes three tensors ``left``, ``right`` and ``out_gate`` in layout
      [B, H, N, N] (FP16).

4️⃣  Batched GEMM (cuBLAS) implements the core einsum:
    out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → [B, H, N, N] (FP16).

5️⃣  Second LayerNorm on the hidden dimension is performed with
    ``torch.nn.functional.layer_norm`` after a cheap permutation to
    [B, N, N, H].  This yields the normalized tensor in FP16.

6️⃣  Gating (multiply by ``out_gate``) and final projection
    (Linear (H→C) without bias) are carried out with a single
    ``torch.nn.functional.linear`` call.  The result is cast back to
    FP32 and reshaped to [B, N, N, C].

Only the split‑&‑gate step is written in Triton; the remaining
operations exploit highly‑optimized CUDA kernels from PyTorch / cuBLAS,
giving a very fast forward pass while remaining fully functional.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split / mask / sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]      FP16 fused projection
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R]           FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,  # rows of the flattened pair tensor
    BLOCK_H: tl.constexpr,  # hidden‑dim tiles
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast across H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                       # stride between hidden planes
    # base address of the (i,k) element inside a hidden plane
    base = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # address for the current hidden slice
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,  left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – forward pass only
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys (all float32,
                              already on the same device as inputs):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # flattened pair count
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Pairwise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim)                                # [R, C]
    x_norm_fp32 = torch.nn.functional.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                          # FP32
    x_norm = x_norm_fp32.to(torch.float16)                     # FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)            # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)  # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64   # rows per program (adjustable)
    BLOCK_H = 64   # hidden‑dim tiles per program
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (core einsum) – left @ rightᵀ
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    # permute to put H at the last dimension for torch.layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP16
    out_norm = torch.nn.functional.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Gate * final projection (Fusion of out_gate & to_out)
    # ------------------------------------------------------------------
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP16
    gated = out_norm * gate_perm                               # FP16

    # flatten to (R, H) for the final linear projection
    gated_flat = gated.reshape(R, H)

    # final linear: hidden → dim (no bias)
    to_out_w = w["to_out.weight"].to(torch.float16)            # [dim, H]
    final_flat = F.linear(gated_flat, to_out_w)               # FP16

    # reshape to the required output layout and cast to FP32
    final = final_flat.to(torch.float32).reshape(B, N, N, dim)

    return final
