"""
TriMul (outgoing) – mixed‑precision Triton + PyTorch implementation
===================================================================

The operator transforms a 4‑D pairwise tensor X ∈ ℝ^{B×N×N×C}
into an output of the same shape using

1️⃣  Row‑wise LayerNorm on X (performed in a Triton kernel, FP32 → FP16).
2️⃣  A single fused linear projection that computes the five
    linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
    in one matmul (FP16).
3️⃣  Masking (if provided) and sigmoid gates applied in FP16.
4️⃣  Batched GEMM   (L @ Rᵀ)  using torch.bmm → O(N³·H) but fully
    Tensor‑Core accelerated (FP16).
5️⃣  Second LayerNorm (over the hidden dimension H) performed in FP32
    with ``torch.nn.functional.layer_norm``.
6️⃣  Element‑wise out‑gate multiplication and final linear projection
    back to the original channel dimension C (FP32).

Only the first LayerNorm is written in Triton; the rest of the
pipeline is expressed with highly‑optimized torch operations that keep
the data in FP16 as long as possible.  The final result is a
float‑32 tensor of shape [B, N, N, C].

The implementation works for all configurations listed in the task
statement and supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # [R, C] input FP32, output FP16
    w_ptr, b_ptr,               # [C] weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile [BLOCK_M, BLOCK_D]
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized data ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    out_gate = out_gate.to(torch.float32)          # (B, N, N, H) → will be broadcasted
    out_gate = out_gate.permute(0, 3, 1, 2)        # (B, H, N, N)

    gated = core_norm * out_gate                    # (B, H, N, N)

    # final projection: (B, N, N, H) → (B, N, N, C)
    gated_flat = gated.permute(0, 2, 3, 1).reshape(-1, H)   # (B·N·N, H)
    out_flat = F.linear(gated_flat, w["to_out.weight"])    # (B·N·N, C)  FP32
    out = out_flat.view(B, N, N, dim)                     # (B, N, N, C)

    return out
```
