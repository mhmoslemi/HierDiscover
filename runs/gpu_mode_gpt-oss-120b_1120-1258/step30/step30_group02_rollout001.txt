"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32→FP16) on the flattened pair tensor using
    `torch.nn.functional.layer_norm`.  This is highly optimized in the
    PyTorch/ATen library and eliminates a custom kernel launch.

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (`torch.nn.functional.linear` with a 5·H × dim matrix).
    A tiny Triton kernel (`split_fuse_gate_kernel`) then:
        * splits the fused projection,
        * applies the optional pair‑mask,
        * evaluates the three sigmoid gates,
        * writes the three tensors (`left`, `right`, `out_gate`) in the
          `[B, H, N, N]` layout (FP16).

3️⃣  Core “einsum” is performed as a batched matrix‑multiply:
        out[b, h] = left[b, h] @ right[b, h].T   →   [B·H, N, N] (FP16).

4️⃣  Second LayerNorm + out‑gate multiplication + final projection are all
    expressed with stock PyTorch ops:
        • `torch.nn.functional.layer_norm` on `out` (flattened to [R, H]),
        • element‑wise multiplication with `out_gate`,
        • a final linear layer (`to_out`) that maps `H → dim`.

Only the split‑and‑gate step is written in Triton; everything else
leverages highly‑optimized CUDA kernels from PyTorch (cuBLAS, cuDNN,
etc.).  The implementation supports optional pair‑masking and works for
all test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, and compute sigmoid
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,           # [R, 5*H]  FP16 fused projection (contiguous)
    left_ptr,           # [B, H, N, N] FP16
    right_ptr,          # [B, H, N, N] FP16
    out_gate_ptr,       # [B, H, N, N] FP16
    mask_ptr,           # [R] FP16 (optional, broadcast on H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Decode the flat index `r = b·N·N + i·N + k` and write the three slices
    left/right/out_gate into a `[B, H, N, N]` layout.  The kernel
    operates on a tile of rows (`BLOCK_R`) and a tile of hidden channels
    (`BLOCK_H`).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r → (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional pair‑mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride of one hidden‑plane in the output layout
    # Base offset for the first hidden plane of this (b,i,k) location
    base = b * (H * hidden_stride) + i * N + k

    # Loop over hidden channels in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # 5 consecutive blocks of size H store [left | right | lg | rg | og]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Sigmoid in FP32 (more accurate) and cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offsets inside the 4‑D tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with the required weight tensors (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # flattened pair dimension
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    # Reshape to [R, dim] and apply a row‑wise LayerNorm.
    # `torch.nn.functional.layer_norm` is heavily optimized and runs on
    # the GPU without any custom kernel.
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # [R, dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, gates) into a single GEMM
    # --------------------------------------------------------------
    # Build a concatenated weight matrix of shape (5·H, dim)
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # FP16
    # One GEMM: (R, dim) @ (dim, 5·H)^T → (R, 5·H)
    proj = F.linear(x_norm, fused_weight)   # [R, 5·H] FP16
    del fused_weight, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy pointer

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM
    # --------------------------------------------------------------
    # Reshape to (B·H, N, N) and multiply left @ rightᵀ
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden_dim), out‑gate, final projection
    # --------------------------------------------------------------
    # Flatten the tensor to [R, H] for LayerNorm.
    out_flat = out.reshape(R, H).to(torch.float32)   # cast to FP32 for LN
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )

    # Multiply by the out‑gate (also cast to FP32)
    out_gate_flat = out_gate.reshape(R, H).to(torch.float32)
    out_norm = out_norm * out_gate_flat

    # Final linear projection (H → dim) – performed with a standard GEMM.
    # `to_out.weight` is (dim, H); we need (H, dim) for the GEMM, so we
    # transpose it once (the transpose is cheap because the tensor is
    # already on GPU and contiguous).
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]
    # `out_norm` is FP32; we keep the multiply in FP32 for numerical stability.
    out_final = F.linear(out_norm, w_out_T)   # [R, dim] FP32

    # Reshape to the original batch‑sequence layout.
    out_final = out_final.view(B, N, N, dim)   # [B, N, N, dim] FP32

    return out_final
```
