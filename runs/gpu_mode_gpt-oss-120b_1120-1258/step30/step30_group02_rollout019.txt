"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) using torch.nn.functional.layer_norm.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM → (R,5·H) tensor, where R = B·N·N.
3️⃣  Split the fused projection, optionally apply the pair‑mask,
    compute the three sigmoid gates and write the tensors
    `left`, `right`, `out_gate` (FP16) via a Triton kernel.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → [B, H, N, N] (FP16)
5️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the
    final projection into a single Triton kernel that produces a
    (R, H) tensor `out_normed`.  Afterwards a single GEMM
    (FP16·FP16 → FP32) with the transposed output weight yields the
    final `[B, N, N, dim]` tensor.
All kernels are written for Triton 3.3.1 and run on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride of one hidden plane inside the (B,H,N,N) tensor
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k        # start of first hidden plane

    # iterate over hidden dimension in one tile (H ≤ 128)
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)               # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,    mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out‑gate and final projection (pre‑matmul)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,        # [R, H]  FP16   (will be overwritten)
    gate_ptr,       # [R, H]  FP16
    ln_w_ptr,       # [H]    FP32   (γ)
    ln_b_ptr,       # [H]    FP32   (β)
    H, eps,
    total_rows,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    # ----------- first pass: mean & variance -------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zero([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------- second pass: LN + out_gate -------------
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # load raw values
        ptr_out  = out_ptr  + offs_r[:, None] * H + offs_h[None, :]
        vals_out = tl.load(ptr_out,  mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        ptr_gate = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals_gate = tl.load(ptr_gate, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # layer‑norm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # FP32

        # (out - μ) * rstd → γ· + β
        norm = (vals_out - mean[:, None]) * rstd[:, None]
        norm = norm * gamma[None, :] + beta[None, :]

        gated = norm * vals_gate                     # FP32

        tl.store(ptr_out, gated.to(tl.float16), mask=mask_r[:, None] & mask_h[None, :])

    """

    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"     (all float32)
        - config_dict must contain
            "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # flattened pair dimension
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # flatten to [R, dim] (contiguous)
    x_flat = x.view(R, dim).contiguous()               # FP32
    # built‑in layer‑norm (highly tuned)
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)             # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5·H, dim]
    proj = F.linear(x_norm, fused_w)        # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64     # rows per program (tunable)
    BLOCK_H = 128    # hidden dimension (fits exactly)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16
    # out_gate already has shape [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate multiplication, then project
    # ------------------------------------------------------------------
    # bring hidden dim to the last axis → shape (R, H)
    out_perm      = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]
    out_flat      = out_perm.view(R, H)                       # FP16
    out_gate_flat = out_gate_perm.view(R, H)                  # FP16

    # fused LayerNorm + out_gate
    BLOCK_R_NORM = 128
    BLOCK_H_NORM = 128                                   # == H (≤128)
    grid_norm = (triton.cdiv(R, BLOCK_R_NORM),)
    norm_gate_kernel[grid_norm](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],        # γ  FP32
        w["to_out_norm.bias"],          # β  FP32
        H,
        eps,
        R,
        BLOCK_R=BLOCK_R_NORM,
        BLOCK_H=BLOCK_H_NORM,
        num_warps=8,
    )
    # after the kernel `out_flat` holds the LN‑gated hidden vectors

    # final linear projection (FP16·FP16 → FP32)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16
    final_fp16 = torch.matmul(out_flat, w_out_T)                      # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)           # [B, N, N, dim] FP32

    return final
```
