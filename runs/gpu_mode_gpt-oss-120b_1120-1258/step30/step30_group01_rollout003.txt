"""
Outgoing TriMul (AlphaFold‑3) – forward‑only Triton implementation.

Algorithm Overview
------------------
1️⃣ Row‑wise LayerNorm over the flattened [B·N·N, dim] tensor.
   * Input: FP32, Output: FP16.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS) → (R, 5·H) where R = B·N·N.
3️⃣ Split the fused projection, apply the optional pair‑mask, compute the
   sigmoid gates and write the three tensors *left*, *right* and *out_gate*
   in layout [B, H, N, N] (FP16) with a Triton kernel.
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
      out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → shape [B, H, N, N] (FP16).
5️⃣ Row‑wise mean & rsqrt‑variance of ``out`` over the hidden dimension H.
   → two FP32 scalars (µ, rstd) per (b,i,j) element (Triton).
6️⃣ Fuse the second LayerNorm, the *out_gate* multiplication and the final
   projection (to_out) into a single Triton kernel, writing the final
   [B, N, N, dim] tensor (FP32).

All kernels target Triton 3.3.1, use mixed precision (FP32 for statistics,
FP16 for bulk work) and run on an NVIDIA H100.  No backward pass is needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]  FP32 input
    y_ptr,   # [R, dim]  FP16 output
    w_ptr,   # [dim]     γ (FP32)
    b_ptr,   # [dim]     β (FP32)
    rows,    # scalar R
    dim,     # scalar dim
    eps,     # scalar eps
    BLOCK_R: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # dim chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ---------- pass 1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_r[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R,5·H]   FP16 fused projection
    left_ptr,      # [B,H,N,N] FP16
    right_ptr,     # [B,H,N,N] FP16
    outg_ptr,      # [B,H,N,N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair mask (broadcast over hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base address of the slice for h==0 (within left/right/outg tensors)
    base = b * (H * hidden_stride) + i * N + k                      # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # fused projection layout: [left|right|lg|rg|og] repeated per hidden dim
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R,BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        outg_raw = tl.load(proj_ptr + proj_base + 4 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        # sigmoid in FP32 => back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        outg_gate  = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = outg_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(outg_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B,H,N,N] FP16
    mu_ptr,    # [R]       FP32
    rstd_ptr,  # [R]       FP32
    total_rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_R, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)

# 6️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B,H,N,N] FP16   (contracted tensor)
    gate_ptr,       # [B,H,N,N] FP16   (out_gate)
    mu_ptr,         # [R]       FP32
    rstd_ptr,       # [R]       FP32
    ln_w_ptr,       # [H]       FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H]       FP32   (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]  FP16   (to_out weight transposed)
    out_final_ptr,  # [B,N,N,dim] FP32 output
    total_rows, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_r = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_r * BLOCK_R
    col_start = pid_n * BLOCK_N

    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    offs_c = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_r = offs_r < total_rows
    mask_c = offs_c < dim

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden‑plane 0

    # per‑row mean & rstd
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]

    acc = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)                # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden plane values and gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_r[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_R, BLOCK_K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)         # [BLOCK_R, BLOCK_K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # (out - µ) * rstd  →  γ· + β  →  * out_gate
        diff   = (vals_out - mu[:, None]) * rstd[:, None]          # [R,K]
        norm   = diff * gamma[None, :] + beta[None, :]             # [R,K]
        term   = norm * vals_gate                                 # [R,K]
        term_f16 = term.to(tl.float16)                            # fp16 for dot

        # final projection slice (H, dim) → (K, TILE_N)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_c[None, :]   # [K,TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_c[None, :],
                         other=0.0)                                    # [K,TILE_N]

        # accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_f16, w_tile).to(tl.float32)

    # write final tensor (flattened first three dimensions → row “R”)
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_c[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_r[:, None] & mask_c[None, :])

def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple):
            (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
            - mask_tensor  : torch.Tensor  [B, N, N]      (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, already on the correct device)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = bool(cfg.get("nomask", True))
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (row‑wise) – FP32 → FP16
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 256
    BLOCK_D = 128
    grid_ln = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps via a single GEMM (cuBLAS)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],    # [H, dim]
            w["right_proj.weight"],   # [H, dim]
            w["left_gate.weight"],    # [H, dim]
            w["right_gate.weight"],   # [H, dim]
            w["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                      # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)         # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    #    left @ rightᵀ → [B·H, N, N] (FP16)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)                # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 32            # H ≤ 128 ⇒ a few tiles suffice
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128    # rows of R per program
    BLOCK_K_OUT = 16     # hidden dim chunk
    BLOCK_N_OUT = 128    # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
