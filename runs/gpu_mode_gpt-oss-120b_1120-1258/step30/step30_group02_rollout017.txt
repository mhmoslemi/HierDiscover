def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. Input layernorm using torch (fp32) then cast to fp16
    x_flat = x.reshape(R, dim)
    x_norm = F.layer_norm(x_flat, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)

    # 2. Fuse five linear maps.
    fused_w = torch.cat(
        [w["left_proj.weight"],
         w["right_proj.weight"],
         w["left_gate.weight"],
         w["right_gate.weight"],
         w["out_gate.weight"]], dim=0
    ).to(torch.float16)  # [5*H, dim]
    proj = F.linear(x_norm, fused_w)   # (R,5*H) FP16
    # free x_norm & fused_w
    del x_norm, fused_w

    # 3. Split+mask+gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4. Batch GEMM: left @ right^T
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))   # [B*H,N,N] fp16
    out = out.view(B, H, N, N)  # [B,H,N,N] fp16

"""
Outgoing TriMul (forward‑only) – highly‑optimized implementation.

Algorithm
---------
1️⃣  LayerNorm on the flattened pair representation – done with the
    highly‑tuned PyTorch `torch.nn.functional.layer_norm` (FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS).  The result has shape (R, 5·H) in FP16.
3️⃣  Triton kernel `split_fuse_gate_kernel`:
        * decodes the flat index (b,i,k) ↔ (b,i,j,k)   (R = B·N·N)
        * splits the fused projection into `left`, `right` and `out_gate`
        * applies the optional pair mask and the three sigmoids.
4️⃣  Batched matrix‑multiply (cuBLAS) implements the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → FP16.
5️⃣  Per‑row mean / variance of `out` are computed with PyTorch (`mean`,
    `var`).  The statistics are kept in FP32 (`mu`, `rstd`).
6️⃣  Triton kernel `final_output_kernel` fuses:
        * the second LayerNorm over the hidden dimension,
        * multiplication by the pre‑computed `out_gate`,
        * the final linear projection (`to_out`).
   Everything is performed in mixed precision (FP16 bulk, FP32
   accumulation) and the result is written as FP32 `[B, N, N, dim]`.

Only the forward pass is required; all kernels run on an H100 and
exploit the large hidden dimension (≤ 128) by processing it in a single
tile.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride between hidden planes
    # start of the first hidden plane for this (b,i,k)
    base = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # each row of the fused projection stores 5 blocks of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst, outg,      mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Final LayerNorm → out‑gate → projection (single kernel)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N]   FP16
    gate_ptr,         # [B, H, N, N]   FP16 (out_gate)
    mu_ptr,           # [R]            FP32
    rstd_ptr,         # [R]            FP32
    ln_w_ptr,         # [H]            FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H]            FP32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim]       FP16 (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j            # start of hidden plane 0

    # per‑row statistics (mu, rstd) are already pre‑computed
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] FP32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] FP32

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # Whole hidden dimension is processed in a single tile (BLOCK_K == H)
    offs_k = tl.arange(0, BLOCK_K)               # [BLOCK_K]
    mask_k = offs_k < H

    # ----- load raw values -------------------------------------------------
    ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
    vals_out = tl.load(ptr_out,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)               # [M,K]

    ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride
    vals_gate = tl.load(ptr_gate,
                        mask=mask_m[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)               # [M,K]

    # ----- second LayerNorm (γ,β) -----------------------------------------
    gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
    beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

    diff = (vals_out - mu[:, None]) * rstd[:, None]            # [M,K]
    norm = diff * gamma[None, :] + beta[None, :]                # [M,K]

    term = norm * vals_gate                                      # [M,K]
    term_fp16 = term.to(tl.float16)                              # fp16 for dot

    # ----- final projection (H → dim) ---------------------------------------
    w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
    w_tile = tl.load(w_ptr,
                     mask=mask_k[:, None] & mask_n[None, :],
                     other=0.0)                                      # [K,N]

    acc += tl.dot(term_fp16, w_tile).to(tl.float32)               # [M,N]

    # ----- write output ----------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with the following FP32 keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict must contain
            "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # flattened pair count
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16) using the highly‑optimized torch kernel
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim)
    x_norm = F.layer_norm(x_flat,
                          (dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps)
    x_norm = x_norm.to(torch.float16)   # fp16 for all subsequent ops
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM (cuBLAS)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # shape [5·H, dim]
    proj = F.linear(x_norm, fused_w)    # (R, 5·H)  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum implemented as a batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)           # FP16
    right_mat = right.view(B * H, N, N)          # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                   # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Per‑row mean / rstd (float32) – PyTorch version is very fast
    # ------------------------------------------------------------------
    mu   = out.mean(dim=1, keepdim=False).float()          # [B, N, N] FP32
    var  = out.var(dim=1, unbiased=False, keepdim=False).float()
    rstd = torch.rsqrt(var + eps)                          # [B, N, N] FP32
    mu   = mu.reshape(R)                                   # flatten to [R]
    rstd = rstd.reshape(R)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows processed per program
    BLOCK_K_OUT = H            # process the whole hidden dim in one tile
    BLOCK_N_OUT = 128          # output‑dim tile size

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
