"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1️⃣  First LayerNorm (over the channel dimension) is performed with the
    PyTorch `torch.nn.functional.layer_norm` primitive (FP32 → FP16).

2️⃣  All five linear maps (left, right, left_gate, right_gate, out_gate) are
    fused into a single GEMM `proj = x_norm @ W_fused` (FP16).

3️⃣  A small Triton kernel (`split_fuse_gate_kernel`) splits the fused projection
    into the three tensors `left`, `right` and `out_gate`, applies the optional
    mask, computes the sigmoid gates (in FP32) and stores the results in
    contiguous `[B, H, N, N]` FP16 buffers.

4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   (FP16).

5️⃣  The second LayerNorm (over the hidden dimension `H`) is again performed
    with `torch.nn.functional.layer_norm`.  The result is multiplied by the
    `out_gate` tensor.

6️⃣  The final linear projection to the model dimension `dim` is carried out
    with a half‑precision GEMM (`torch.nn.functional.linear`).  The result is
    cast to FP32 and reshaped to `[B, N, N, dim]`.

All heavy compute (the split + gating step) is executed in a Triton kernel,
while the remaining operations use highly‑optimized PyTorch / cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16   (optional, broadcasted over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # Decode flat row index r -> (batch, i, k)
    # ------------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    # Load optional mask (broadcasted over the hidden dimension H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Destination stride for one hidden plane (N×N)
    hidden_stride = N * N
    # Base offset for (b, i, k) with h = 0
    base_offset = b * (H * hidden_stride) + i * N + k

    # ------------------------------------------------------------------
    # Process the hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Linear layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid in FP32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # out_gate is not masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, on the same device as the inputs)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm over the channel dimension (FP32 → FP16)
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension.
    # It internally uses a highly optimized CUDA kernel.
    x_ln = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection: 5·H outputs, keep everything in FP16
    # --------------------------------------------------------------
    # Build the fused weight matrix [5·H, dim] in FP16.
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    # Linear projection (R, dim) → (R, 5·H) in half‑precision.
    proj = F.linear(x_ln.view(R, dim).to(torch.float16), fused_w)   # [R, 5·H] FP16
    del x_ln  # free memory

    # --------------------------------------------------------------
    # 3️⃣ Split the fused projection, apply mask & sigmoid gates (Triton)
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy tensor

    BLOCK_R = 64   # rows per program (adjustable)
    BLOCK_H = 64   # hidden‑dim tiles per program
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr  # release references

    # --------------------------------------------------------------
    # 4️⃣ Core pairwise contraction (batched GEMM, FP16)
    # --------------------------------------------------------------
    # left/right are [B, H, N, N]. Collapse B‑H into one dimension.
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]  FP16

    # out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (FP32)
    # --------------------------------------------------------------
    # Rearrange to (..., H) layout expected by torch.layer_norm.
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]  FP16 → FP32
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]  float32

    # --------------------------------------------------------------
    # 6️⃣ Apply out_gate and final linear projection
    # --------------------------------------------------------------
    # out_gate has shape [B, H, N, N]; bring it to the same layout.
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] FP16

    # Combine LN output with the gate (both in FP32 for stability)
    gated = out_norm * out_gate_perm.to(out_norm.dtype)       # [B, N, N, H] FP32

    # Final projection: hidden → dim. Use half‑precision GEMM for speed.
    w_out_fp16 = w["to_out.weight"].to(torch.float16)  # [dim, H] FP16

    # Linear expects input of shape (..., H)
    gated_flat = gated.view(R, H).to(torch.float16)    # [R, H] FP16
    final_fp16 = F.linear(gated_flat, w_out_fp16)     # [R, dim] FP16

    # Cast back to FP32 and reshape to the original 4‑D layout.
    final = final_fp16.to(torch.float32).view(B, N, N, dim)  # [B, N, N, dim] FP32

    return final
