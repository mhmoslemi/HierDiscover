def fused_ln_proj_kernel(...):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R

    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute mean,var
    sum = tl.zeros([BLOCK_R], dtype=tl.float32)
    sumsq = tl.zeros([BLOCK_R], dtype=tl.float32)

    # pass over dim blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + (offs_r[:, None] * dim + offs_d[None, :])
        x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum += tl.sum(x, axis=1)
        sumsq += tl.sum(x * x, axis=1)

    mean = sum / dim
    var = sumsq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute projection
    # We'll allocate accumulator for 5*H dim, maybe process BLOCK_H chunk at a time.
    # We need weight_T pointer shape [dim, 5*H] (FP16). We'll load weight block for hidden chunk.
    # For each hidden block offset:
    for h_off in range(0, tl.cdiv(5*H, BLOCK_H)):
        # Note: we are processing 5*H output dims; we also need to later split into left/right/gates.
        # We store partial results in a local accumulator block: acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        # iterate over dim blocks
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load input values again
            x_ptrs = x_ptr + (offs_r[:, None] * dim + offs_d[None, :])
            x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            # normalize
            x_norm = (x - mean[:, None]) * rstd[:, None]

            # load weight block: weight_T shape [dim, 5*H]
            w_ptrs = w_ptr + (offs_d[:, None] * (5 * H) + (h_off + tl.arange(0, BLOCK_H))[None, :])
            # note: offs_d is size BLOCK_D, h_block size BLOCK_H
            # Need to broadcast dims appropriately.
            w = tl.load(w_ptrs, mask=mask_d[:, None] & (h_off + tl.arange(0, BLOCK_H) < 5*H), other=0.0).to(tl.float32)

            # accumulate
            # x_norm shape [BLOCK_R, BLOCK_D], w shape [BLOCK_D, BLOCK_H]
            acc += tl.dot(x_norm, w) # (BLOCK_R, BLOCK_H) result

        # after processing all dim, store output
        # store to global out: out_ptr shape [R, 5*H]
        out_ptrs = out_ptr + (offs_r[:, None] * (5*H) + (h_off + tl.arange(0, BLOCK_H))[None, :])
        tl.store(out_ptrs, acc.to(tl.float16), mask=mask_r[:, None] & (h_off + tl.arange(0, BLOCK_H) < 5*H))

    # End.

  x, mask, w, cfg = data
  dim = cfg["dim"]
  hidden_dim = cfg["hidden_dim"]
  eps = 1e-5
  device = x.device
  B,N,_,_ = x.shape
  R = B*N*N
  H = hidden_dim

  # Step 1: LayerNorm with torch.nn.functional.layer_norm
  x_flat = x.reshape(R, dim)  # FP32
  # Use weight and bias
  x_norm = F.layer_norm(x_flat, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
  x_norm = x_norm.to(torch.float16).contiguous()  # FP16

  # Step 2: fused linear (5*H) with torch.nn.functional.linear
  fused_w = torch.cat([...], dim=0).to(torch.float16)  # shape [5*H, dim]
  proj = F.linear(x_norm, fused_w)  # shape [R,5*H] FP16

  # Step 3: split/fuse/gates
  left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
  right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
  out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
  if not cfg.get("nomask", True) and mask is not None:
      mask_flat = mask.to(torch.float16).reshape(R).contiguous()
      HAS_MASK = 1
      mask_ptr = mask_flat
  else:
      HAS_MASK = 0
      mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

  BLOCK_R = 128
  BLOCK_H = 64
  grid_split = (triton.cdiv(R, BLOCK_R),)
  split_fuse_gate_kernel[grid_split](
      proj, left, right, out_gate,
      mask_ptr,
      R, H, N,
      HAS_MASK=HAS_MASK,
      BLOCK_R=BLOCK_R,
      BLOCK_H=BLOCK_H,
      num_warps=8,
  )
  del proj

  # Step 4: batched GEMM (left x right^T)
  # left shape [B,H,N,N] and right similar
  left_mat = left.view(B*H, N, N)
  right_mat = right.view(B*H, N, N)
  out = torch.bmm(left_mat, right_mat.transpose(1,2)).view(B, H, N, N)

  # Step 5: Rowwise stats
  mu = torch.empty(R, dtype=torch.float32, device=device)
  rstd = torch.empty(R, dtype=torch.float32, device=device)
  BLOCK_M_STATS = 256
  BLOCK_K_STATS = 128
  grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
  rowwise_stats_kernel[grid_stats](

"""
Outgoing TriMul (TriMul‑out) – fully‑fused forward implementation.

Algorithm (all heavy work stays on the GPU):
-----------------------------------------------------------------
1️⃣  Row‑wise LayerNorm on the flattened [B·N·N, C] tensor (torch.nn.functional.layer_norm).
2️⃣  Single Linear projection with a fused weight matrix of shape (5·H, C) → gives
    [R, 5·H] (R = B·N·N).  No intermediate FP32→FP16 copy is written to device memory.
3️⃣  Triton kernel `split_fuse_gate_kernel`:
      * reshapes the fused projection into three tensors  (left, right, out_gate)
        of layout [B, H, N, N] (FP16);
      * optionally applies the mask (broadcast over the hidden dimension);
      * computes the three sigmoid gates (left_gate, right_gate, out_gate).
4️⃣  Batched GEMM (torch.bmm) computes the core einsum
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]      → [B, H, N, N] (FP16).
5️⃣  Triton kernel `rowwise_stats_kernel` computes per‑position (b,i,j) mean &
    rsqrt‑variance of `out` across the hidden dimension H (producing μ, rstd).
6️⃣  Remaining element‑wise work (second LayerNorm, out_gate, final Linear)
    is done with fast PyTorch kernels:
        norm   = (out – μ)·rstd
        norm   = norm·γ_out + β_out                (γ_out / β_out = to_out_norm)
        norm   = norm·out_gate
        final  = norm @ W_out.T                    (W_out = to_out.weight)
    The result is reshaped back to [B, N, N, C] (float32).

Only two Triton kernels are needed (split/gate + row‑wise stats) while the
memory‑intensive LayerNorm and the final projection are delegated to highly‑optimised
PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)    # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over H
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # Base offset for the destination tensors (hidden‑plane 0)
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid (stable, compute in FP32 then cast)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N]  FP16
    mu_ptr,    # [R]            FP32
    rstd_ptr,  # [R]            FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total (b,i,j) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP16) – use the highly‑optimised torch kernel
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim)                       # [R, C]  FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                # FP32
    x_norm = x_norm.to(torch.float16).contiguous()   # FP16, ready for the GEMM

    # --------------------------------------------------------------
    # 2️⃣ Fused Linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, C]  FP16
    proj = F.linear(x_norm, fused_w)                  # [R, 5·H]  FP16
    del x_norm, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (einsum equivalent)
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] and perform (i,k)·(j,k)ᵀ
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128            # H ≤ 128 ⇒ single tile in the inner loop
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate, final linear projection (pure PyTorch)
    # --------------------------------------------------------------
    # reshape `out` and `out_gate` to [R, H] where R = B·N·N
    out_flat      = out.permute(0, 2, 3, 1).contiguous().view(R, H)        # FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)   # FP16

    # move to FP32 for the remaining arithmetic (more stable)
    out_fp32 = out_flat.to(torch.float32)
    gate_fp32 = out_gate_flat.to(torch.float32)

    # broadcast μ and rstd to shape [R, H]
    mu_b   = mu[:,   None]      # [R,1]
    rstd_b = rstd[:, None]      # [R,1]

    # second LayerNorm (to_out_norm)
    norm = (out_fp32 - mu_b) * rstd_b                     # (out‑μ)/σ
    norm = norm * w["to_out_norm.weight"] + w["to_out_norm.bias"]

    # apply out‑gate
    norm = norm * gate_fp32

    # final projection (to_out) – a standard Linear (no bias)
    final_fp32 = torch.nn.functional.linear(norm, w["to_out.weight"])   # [R, dim] FP32
    final = final_fp32.view(B, N, N, dim)                               # [B,N,N,dim]

    return final
```
