"""
TriMul (outgoing) – fast forward‑only implementation with Triton + Torch.

Algorithm
---------
1️⃣  LayerNorm on the last dimension (dim) – use torch.nn.functional.layer_norm
    (highly tuned CUDA kernel).

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,
    out_gate) into one GEMM (cuBLAS) producing a tensor of shape (R,5·H)
    in FP16, where R = B·N·N and H = hidden_dim.

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    • Decode the flat index (b,i,k) → writes left[b,h,i,k] and right[b,h,j,k].
    • Optionally applies the pair‑wise mask (broadcast over H).
    • Computes sigmoid gates (left_gate, right_gate, out_gate) in FP32 → FP16.
    • Stores three FP16 tensors: left, right, out_gate in layout [B,H,N,N].

4️⃣  Batched GEMM (torch.bmm, Tensor‑cores) computes the core einsum:
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    Result shape = [B,H,N,N] (FP16).

5️⃣  Second LayerNorm over the hidden dimension (H):
    • Compute mean/variance per (b,i,j) row with torch.mean/torch.var on the
      FP32 view of `out` (fast reduction kernels).
    • Apply learned γ/β (to_out_norm.weight/bias) and the sigmoid out_gate.
    • All operations are fused in a few torch kernels (no extra Triton launch).

6️⃣  Final linear projection (to_out):
        final = out_norm_gate @ to_out.weight.T
    Performed by a single `torch.nn.functional.linear` call (FP32).

The only Triton kernel used is the split/fuse/gate kernel, which gives us
the required custom‑kernel component while the rest of the work stays in
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 – fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask broadcast over hidden dimension
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                     # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride of one hidden plane in dest tensors
    base_offset = b * (H * hidden_stride) + i * N + k  # start of hidden plane 0 for (b,i,k)

    # Iterate over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Linear offset inside the fused projection: [left|right|l_gate|r_gate|out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Stable sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offsets for the three tensors
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor   : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor    : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict   : mapping with the model's parameters (all float32)
            - config_dict    : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total (b,i,k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use torch's highly‑optimized kernel)
    # ------------------------------------------------------------------
    # x shape: [B, N, N, dim]
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)      # cast to half for the following GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (cuBLAS, FP16)
    # ------------------------------------------------------------------
    # Build fused weight [5·H, dim] (FP16) – concatenated row‑wise
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)     # (5·H, dim)

    # GEMM: (R, dim) @ (dim, 5·H).T  → (R, 5·H)
    proj = F.linear(x_norm, fused_weight)   # FP16, shape [R, 5·H]

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # mask values are 0/1 (cast to half) and flattened to [R]
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # rows per program (tuned for large N²)
    BLOCK_H = 64   # hidden‑dim tile
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # proj & mask no longer needed
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (Tensor‑cores via cuBLAS)
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                     # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over the hidden dimension H) – Torch kernels
    # ------------------------------------------------------------------
    # Compute mean & variance per (b,i,j) over H in FP32 for numerical stability
    out_fp32 = out.to(torch.float32)               # [B, H, N, N] FP32
    # `torch.var_mean` returns (var, mean); we want both
    var, mean = torch.var_mean(out_fp32, dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                  # [B,1,N,N]

    # Apply learned weight & bias (to_out_norm) – broadcasting over (b,i,j)
    w_norm = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1] FP32
    b_norm = w["to_out_norm.bias"].view(1, H, 1, 1)     # [1,H,1,1] FP32

    # Normalized tensor (still FP32), then cast back to FP16 for out_gate mul
    out_norm_fp32 = ((out_fp32 - mean) * rstd) * w_norm + b_norm   # [B,H,N,N] FP32
    out_norm = out_norm_fp32.to(torch.float16)                     # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Multiply by out_gate and final linear projection (to_out)
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                                     # FP16, [B,H,N,N]

    # Move hidden dim to last axis and flatten batch/spatial dims → [B·N·N, H]
    gated_flat = gated.permute(0, 2, 3, 1).reshape(-1, H)          # FP16

    # Final linear: hidden -> dim (float32 output)
    final = F.linear(gated_flat, w["to_out.weight"])              # [B·N·N, dim] FP32
    final = final.view(B, N, N, dim)                               # [B,N,N,dim] FP32

    return final
