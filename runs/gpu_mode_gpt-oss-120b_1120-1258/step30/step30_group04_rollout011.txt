TriMul (outgoing) – Triton‑accelerated forward pass

The operator transforms a 4‑D pairwise tensor X ∈ ℝ^{B×N×N×C}

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, C]  FP32 input
    y_ptr,            # [R, C]  FP16 output (normed)
    w_ptr, b_ptr,    # [C]    LN weight & bias (FP32)
    rows, cols, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute per‑row mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(cols, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < cols

        # load a tile of shape [BLOCK_M, BLOCK_D]
        ptr = x_ptr + offs_m[:, None] * cols + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / cols
    var  = sum_sq / cols - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized output ----------
    for d_off in range(0, tl.cdiv(cols, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < cols

        ptr = x_ptr + offs_m[:, None] * cols + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * cols + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]    FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flattened index (b,i,k) from r
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        m_val = m_val[:, None]                                     # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn        # distance between consecutive hidden planes

    # Base offset for the (b, *, i, k) element (h = 0)
    base_offset = b * (H * hidden_stride) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: concat([L,R,LG,RG,OG]) each of shape [R, H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw * m_val * left_gate
        right = right_raw * m_val * right_gate

        # destination offsets for each hidden channel
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise reduction (mean & rsqrt‑var) over hidden dimension H
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,                # [B, H, N, N] FP16
    mu_ptr, rstd_ptr,       # [R] FP32   (R = B·N·N)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j                # start of h=0 slice

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Final output: LN(H) → out_gate → projection → FP32 write
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,              # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,             # [B, H, N, N] FP16   (out_gate)
    mu_ptr, rstd_ptr,     # [R] FP32
    ln_w_ptr, ln_b_ptr,   # [H] FP32  (to_out_norm weight & bias)
    w_out_T_ptr,          # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,        # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)    # rows tile (R)
    pid_n = tl.program_id(1)    # output‑dim tile (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)                # [BLOCK_N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j                # start of h=0 plane

    # per‑row μ/σ (FP32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor and gate (FP16 → FP32)
        out_tile = tl.load(out_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        gate_tile = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        # LayerNorm over hidden dim: ((out-μ)*rstd)*γ + β
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by out‑gate
        term = norm * gate_tile                         # [BLOCK_M, BLOCK_K]

        # cast to FP16 for tensor‑core dot
        term_fp16 = term.to(tl.float16)

        # weight slice (H, dim) transposed → (BLOCK_K, BLOCK_N)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [BLOCK_K, BLOCK_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                # [BLOCK_M, BLOCK_N]

    # write final FP32 result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (5·H × dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)      # [5·H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, compute sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core N³ einsum using batched GEMM (L @ Rᵀ)
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)        # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                 # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑var over hidden dimension H
    # --------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128   # H ≤ 128 in the supported configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out_gate + projection to original dim
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows per program (tiles over R)
    BLOCK_K_OUT = 32           # hidden‑dim chunk per iteration
    BLOCK_N_OUT = 128          # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
