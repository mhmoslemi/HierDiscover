"""
TriMul (outgoing) – fused‑kernel Triton implementation
====================================================

This implementation follows the reference PyTorch ``TriMul`` module but
moves the heavy work onto the GPU:

1️⃣  Row‑wise LayerNorm on the flatten pairwise tensor (FP32 → FP16).
2️⃣  A single ``torch.nn.functional.linear`` computes the five linear
    projections (left, right, left‑gate, right‑gate, out‑gate) in FP16.
3️⃣  ``split_fuse_gate_kernel`` splits the fused projection, applies the
    optional mask, and evaluates the sigmoid gates, writing three tensors
    ``left``, ``right`` and ``out_gate`` in layout ``[B, H, N, N]``.
4️⃣  The core ``N×N`` pairwise interaction is performed with a batched
    GEMM (``torch.bmm``) on ``left`` and ``right`` → ``out``.
5️⃣  ``final_fused_kernel`` fuses the second LayerNorm, the out‑gate,
    and the final projection.  It loads the entire hidden dimension
    (``H = hidden_dim``) for a block of rows, computes per‑row mean &
    rsqrt‑variance, builds the normalized term, and performs a single
    ``tl.dot`` with the output weight matrix.  This removes the separate
    mean/variance reduction and halves the global‑memory traffic.

All kernels use mixed‑precision (FP16 compute, FP32 reduction) and are
tuned for H100 (Triton 3.3.1).  No gradients are computed – only the
forward pass is needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # [R, dim] input FP32, output FP16
    w_ptr, b_ptr,               # [dim] LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)    # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: accumulate mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # linear offset
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized result ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # linear offset
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]    FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flatten index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask broadcasted over H
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                         # [BLOCK_R, 1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                                   # stride between H‑planes
    # offset of the (b,i,k) element for h = 0
    base_offset = b * (H * hidden_stride) + i * N + k        # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # view into fused projection: [L,R,LG,RG,OG] concatenated
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]            # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        # destination offset for each (b,i,k) / (b,j,k) plane
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Final output: fused second LayerNorm + out‑gate + projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,                # [B, H, N, N] FP16 (contracted tensor)
    gate_ptr,               # [B, H, N, N] FP16 (out_gate)
    ln_w_ptr, ln_b_ptr,    # [H] FP32 (to_out_norm weight & bias)
    w_out_T_ptr,            # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,          # [B, N, N, dim] FP32
    total_rows, N, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    H: tl.constexpr,
):
    pid_m = tl.program_id(0)    # rows tile
    pid_n = tl.program_id(1)    # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)                 # [BLOCK_N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b, i, j) from linear row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j          # start of h=0 plane for (b,i,j)

    # Guard against out‑of‑bounds rows (set base to 0 when masked)
    base = tl.where(mask_m, base, 0)

    # ------------------------------------------------------------------
    # Load the entire hidden dimension for the row tile (H is a constexpr)
    # ------------------------------------------------------------------
    h_range = tl.arange(0, H)                               # [H]

    out_vals = tl.load(
        out_ptr + base[:, None] + h_range[None, :] * hidden_stride,
        mask=mask_m[:, None],
        other=0.0)                                            # FP16, shape [BLOCK_M, H]

    gate_vals = tl.load(
        gate_ptr + base[:, None] + h_range[None, :] * hidden_stride,
        mask=mask_m[:, None],
        other=0.0)                                            # FP16, shape [BLOCK_M, H]

    # ------------------------------------------------------------------
    # Compute per‑row mean & rsqrt‑variance (FP32)
    # ------------------------------------------------------------------
    out_f32 = out_vals.to(tl.float32)
    sum_val = tl.sum(out_f32, axis=1)                       # [BLOCK_M]
    sum_sq  = tl.sum(out_f32 * out_f32, axis=1)             # [BLOCK_M]
    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Build the normalized term: ((out - μ)·rstd·γ + β)·gate
    # ------------------------------------------------------------------
    mu_f16   = mu[:, None].to(tl.float16)                  # [BLOCK_M, 1]
    rstd_f16 = rstd[:, None].to(tl.float16)                # [BLOCK_M, 1]

    gamma_f16 = tl.load(ln_w_ptr + h_range).to(tl.float16)    # [H]
    beta_f16  = tl.load(ln_b_ptr + h_range).to(tl.float16)    # [H]

    gamma_f16 = gamma_f16[None, :]                          # [1, H]
    beta_f16  = beta_f16[None, :]                           # [1, H]

    norm = ((out_vals - mu_f16) * rstd_f16) * gamma_f16 + beta_f16   # FP16
    term = norm * gate_vals                                            # FP16

    # ------------------------------------------------------------------
    # Dot with the output projection (H × dim)
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    w_tile = tl.load(
        w_out_T_ptr + h_range[:, None] * dim + offs_n[None, :],
        mask=mask_n[None, :],
        other=0.0)                                            # [H, BLOCK_N]

    # tensor‑core friendly fused dot (FP16 × FP16 → FP32 accumulation)
    acc += tl.dot(term, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # Write final FP32 tensor
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point ---------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # = H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                           # total pairwise tokens

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for the five heads (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                       # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left·rightᵀ) – core pairwise interaction
    # ------------------------------------------------------------------
    left_mat = left.view(B * hidden_dim, N, N)                       # [B·H, N, N] FP16
    right_mat = right.view(B * hidden_dim, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))            # [B·H, N, N] FP16
    out = out.view(B, hidden_dim, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Final fused LayerNorm + out‑gate + to_out projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64          # rows per program (tiles over R)
    BLOCK_N_OUT = 128         # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
        H=hidden_dim,              # compile‑time constant
    )
    return final
