Pseudo Triton kernel:

@triton.jit
def fused_proj_gate_kernel(
    x_ptr,    # [R, dim] FP16
    left_ptr, right_ptr, out_gate_ptr,  # output [B, H, N, N] FP16
    mask_ptr,   # [R] FP16 optional
    left_proj_weight_ptr, right_proj_weight_ptr,
    left_gate_weight_ptr, right_gate_weight_ptr,
    out_gate_weight_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # compute base offset for output location at h=0
    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + k  # [BLOCK_R]

    # Load mask if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # shape [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R,1], 1.0, dtype=tl.float16)

    # iterate over hidden blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
            mask_d = offs_d < dim

            # load x block
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)  # [BLOCK_R,BLOCK_D]

            # load weight slices for each projection/gate
            # each weight matrix shape [H, dim]; stride_h = dim
            w_left_proj = tl.load(left_proj_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)

            w_right_proj = tl.load(right_proj_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                   mask=mask_h[:, None] & mask_d[None, :],
                                   other=0.0).to(tl.float32)

            w_left_gate = tl.load(left_gate_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                  mask=mask_h[:, None] & mask_d[None, :],
                                  other=0.0).to(tl.float32)

            w_right_gate = tl.load(right_gate_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                   mask=mask_h[:, None] & mask_d[None, :],
                                   other=0.0).to(tl.float32)

            w_out_gate = tl.load(out_gate_weight_ptr + offs_h[:, None] * dim + offs_d[None, :],
                                 mask=mask_h[:, None] & mask_d[None, :],
                                 other=0.0).to(tl.float32)

            # dot products
            left_acc += tl.dot(x, w_left_proj)      # [BLOCK_R,BLOCK_H]
            right_acc += tl.dot(x, w_right_proj)
            left_gate_acc += tl.dot(x, w_left_gate)
            right_gate_acc += tl.dot(x, w_right_gate)
            out_gate_acc += tl.dot(x, w_out_gate)

        # apply sigmoid to gate values (in FP32) and cast to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

        # convert left/right accum to FP16
        left = left_acc.to(tl.float16) * left_gate * mask_val
        right = right_acc.to(tl.float16) * right_gate * mask_val

        # compute destination offsets per hidden block
        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fused‑projection Triton implementation
========================================================

The forward pass consists of:

1️⃣ Row‑wise LayerNorm on the input X ∈ ℝ^{R×C} (R = B·N·N, C = dim).
   The kernel works in FP32 for the reduction and writes FP16.

2️⃣ *Fused* linear projection + gating + optional mask.
   A single Triton kernel computes the five linear maps
   (left_proj, right_proj, left_gate, right_gate, out_gate) directly from
   the normalized tensor, applies the sigmoid gates and the mask
   and stores three tensors L,R,G of shape [B,H,N,N] (FP16).  This removes the
   intermediate (R×5H) projection matrix used in the reference implementation
   and saves one full‑tensor write‑read pass.

3️⃣ Batched matrix multiplication L·Rᵀ (B·H independent GEMMs) using
   `torch.bmm`.  This is heavily optimized in cuBLAS on H100.

4️⃣ Row‑wise mean/variance over the hidden dimension H (µ,σ⁻¹) with a small
   Triton kernel.

5️⃣ Final kernel fuses the second LayerNorm, the out‑gate multiplication and
   the output projection (H→dim) into a single pass, reading the contracted
   tensor only once.

All kernels run in mixed precision (FP16 for the bulk of the work,
FP32 for reductions) and are tuned for H100 (Triton 3.3.1).  No gradients are
computed – only the forward pass is required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # [R, C] input FP32, output FP16
    w_ptr, b_ptr,               # [C] LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1st pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2nd pass: write normalized tensor ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# 2️⃣ Fused projection + gating (+ optional mask)
# --------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,                     # [R, dim] FP16 (output of LN)
    left_ptr, right_ptr, out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,                  # [R] FP16 (optional)
    w_left_proj_ptr,
    w_right_proj_ptr,
    w_left_gate_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    For each (b,i,k) flatten index r:
        left  = (x @ W_left)      * sigmoid(x @ W_left_gate)  * mask
        right = (x @ W_right)     * sigmoid(x @ W_right_gate) * mask
        out_gate = sigmoid(x @ W_out_gate)
    stores the three tensors in layout [B, H, N, N].
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    ## decode (b, i, k) from flat index r
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    hidden_stride = NN                     # stride between consecutive H‑planes
    base_off = b * (H * hidden_stride) + i * N + k   # offset for h = 0

    # optional mask (broadcasted over H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators (FP32 for accuracy)
        acc_left      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_left_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right_gate= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_out_gate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
            mask_d = offs_d < dim

            # x tile: [BLOCK_R, BLOCK_D]
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

            # weight tiles: [BLOCK_H, BLOCK_D] each
            w_lproj = tl.load(w_left_proj_ptr  + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            w_rproj = tl.load(w_right_proj_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            w_lgate = tl.load(w_left_gate_ptr  + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            w_rgate = tl.load(w_right_gate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)
            w_ogate = tl.load(w_out_gate_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)

            # dot products -> contributes to the five linear maps
            acc_left      += tl.dot(x, w_lproj)       # [BLOCK_R, BLOCK_H]
            acc_right     += tl.dot(x, w_rproj)
            acc_left_gate += tl.dot(x, w_lgate)
            acc_right_gate+= tl.dot(x, w_rgate)
            acc_out_gate  += tl.dot(x, w_ogate)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-acc_left_gate))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_right_gate))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-acc_out_gate))).to(tl.float16)

        # apply mask and gates, cast to FP16 for storage
        left  = acc_left .to(tl.float16) * left_gate  * m
        right = acc_right.to(tl.float16) * right_gate * m

        # destination offsets for this hidden tile
        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# 3️⃣ Row‑wise stats (mean & rsqrt‑var) over hidden dimension H
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N] FP16
    mu_ptr, rstd_ptr, # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = NN
    base = b * (H * hidden_stride) + i * N + j          # start of h=0 plane

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

# 4️⃣ Final output: LN (2nd) + out‑gate + projection
# --------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    mu_ptr, rstd_ptr, # [R] FP32
    ln_w_ptr, ln_b_ptr,                # [H] FP32 (to_out_norm weight/bias)
    w_out_T_ptr,                       # [H, dim] FP16 (transposed to_out)
    out_final_ptr,                     # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)    # rows tile (R)
    pid_n = tl.program_id(1)    # output‑dim tile (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)    # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)    # [BLOCK_N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = NN
    base = b * (H * hidden_stride) + i * N + j    # start of plane h=0 for (b,i,j)

    # load μ, σ⁻¹
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m,  other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m,  other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # contracted tensor O and gate G (FP16 → FP32)
        out_tile = tl.load(out_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)            # [BLOCK_M, BLOCK_K]

        gate_tile = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)            # [BLOCK_M, BLOCK_K]

        # LayerNorm parameters (γ, β) for this slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [BLOCK_K]

        # ((out-μ)·rstd·γ + β)   –   element‑wise across hidden dim
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by the out‑gate
        term = norm * gate_tile                     # [BLOCK_M, BLOCK_K]
        term_fp16 = term.to(tl.float16)             # FP16 for tensor‑core dot

        # weight slice (H, dim) transposed → (BLOCK_K, BLOCK_N)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [BLOCK_K, BLOCK_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                # [BLOCK_M, BLOCK_N]

    # write final FP32 result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    Args:

            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the same keys as the reference module
                             (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gating (+ optional mask)
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # optional mask (broadcast over H)
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # cast all weight matrices to FP16 once
    w_left_proj   = w["left_proj.weight"].to(torch.float16).contiguous()
    w_right_proj  = w["right_proj.weight"].to(torch.float16).contiguous()
    w_left_gate   = w["left_gate.weight"].to(torch.float16).contiguous()
    w_right_gate  = w["right_gate.weight"].to(torch.float16).contiguous()
    w_out_gate    = w["out_gate.weight"].to(torch.float16).contiguous()

    BLOCK_R = 64
    BLOCK_H = 64               # tile over hidden dimension
    BLOCK_D = 32               # tile over input feature dimension
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    fused_proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        w_left_proj,
        w_right_proj,
        w_left_gate,
        w_right_gate,
        w_out_gate,
        R,
        H,
        N,
        dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, mask_ptr, w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate

    # --------------------------------------------------------------
    # 3️⃣ Core N³ contraction (batched GEMM)  →  out ∈ FP16
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                 # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑var over H
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 → whole hidden dim
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out‑gate + projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows per program (tiles over R)
    BLOCK_K_OUT = 32           # hidden‑dim chunk per iteration
    BLOCK_N_OUT = 128          # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
