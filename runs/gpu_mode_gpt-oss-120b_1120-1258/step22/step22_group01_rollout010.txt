"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass (no backward).

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the flattened pairwise tensor using
    torch.nn.functional.layer_norm (highly‑optimised cuDNN kernel).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into
    a single half‑precision GEMM (torch.nn.functional.linear).

3️⃣  Split the fused projection, mask (optional) and sigmoid gates in a single
    Triton kernel → tensors `left`, `right` and `out_gate` in layout
    [B, H, N, N] (FP16).

4️⃣  Batched half‑precision GEMM (cuBLAS) over the hidden dimension:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k].

5️⃣  Row‑wise mean / variance on `out` (H‑dim) → μ, rstd (FP32).

6️⃣  Apply the second LayerNorm (to_out_norm), multiply by `out_gate`,
    then a single GEMM (H × dim) over all pairwise positions:
        final = ((out – μ) * rstd * γ + β) * out_gate  @  W_outᴛ .
    The massive GEMM is performed with torch.matmul, which uses cuBLAS
    and is significantly faster than a hand‑written Triton reduction.

Only the split‑&‑gate kernel is written in Triton – the rest leverages
high‑performance PyTorch / cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Split‑fuse‑gate kernel (produces left/right/out_gate in FP16)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,                     # [R] FP16 (used only if HAS_MASK==1)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑row mask (broadcast over the hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]          # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # start of hidden‑dim 0 for this (b,i,k) in the output tensors
    base_offset = b * (H * hidden_stride) + i * N + k

    # loop over hidden dimension blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw      = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr      + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
#  Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"

            - config_dict : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    R = B * N * N                         # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – use cuDNN implementation
    # ------------------------------------------------------------------
    # flatten to [R, dim] and cast to half
    x_flat_fp16 = x.view(R, dim).to(torch.float16)
    # half‑precision layer‑norm parameters
    ln_w_fp16 = w["norm.weight"].to(torch.float16)
    ln_b_fp16 = w["norm.bias"].to(torch.float16)
    x_norm_fp16 = F.layer_norm(x_flat_fp16,
                               normalized_shape=(dim,),
                               weight=ln_w_fp16,
                               bias=ln_b_fp16,
                               eps=eps)                        # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5 × hidden_dim) → (R, 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm_fp16, fused_w)   # (R, 5·H)  FP16
    del x_norm_fp16, fused_w                 # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate (FP16)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = (not nomask) and (mask is not None)
    if has_mask:
        mask_fp16 = mask.to(torch.float16).view(R).contiguous()
        mask_ptr = mask_fp16
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=int(has_mask),
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    #    left[b,h,i,k] • right[b,h,j,k]  →  out[b,h,i,j]
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                             # (B, H, N, N) FP16
    del left, right, left_mat, right_mat                 # free memory

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over H)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)                       # (B, H, N, N) FP32
    del out                                                # free half version

    mu   = out_fp32.mean(dim=1, keepdim=True)             # (B,1,N,N)
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    rstd = torch.rsqrt(var + eps)                         # (B,1,N,N)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    # layer‑norm parameters for the hidden dimension (float32)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)       # (1,H,1,1)
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)        # (1,H,1,1)

    # In‑place normalisation
    out_fp32.sub_(mu).mul_(rstd)          # now (out - μ) * rstd
    out_fp32 = out_fp32 * gamma + beta   # apply γ,β   → still FP32

    # multiply by out_gate (FP16 → FP32)
    out_gate_fp32 = out_gate.to(torch.float32)
    out_fp32 = out_fp32 * out_gate_fp32
    del out_gate, out_gate_fp32, mu, var, rstd, gamma, beta

    # flatten to [R, H] for the final GEMM
    gated_flat = out_fp32.permute(0, 2, 3, 1).contiguous().view(R, H)  # FP32
    del out_fp32

    # final projection matrix (H × dim) in half‑precision
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)

    # cast gated features to half for the massive GEMM (cuBLAS)
    gated_half = gated_flat.to(torch.float16)

    final_half = torch.matmul(gated_half, w_out_T)                     # (R, dim) FP16
    final = final_half.to(torch.float32).view(B, N, N, dim)           # (B,N,N,dim) FP32

    return final
```
