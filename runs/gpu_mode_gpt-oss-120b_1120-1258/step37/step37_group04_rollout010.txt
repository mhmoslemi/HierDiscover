@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    M, D, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # Compute mean & variance across D
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Process D in blocks
    for d_offset in range(0, D, BLOCK_D):
        offs_d = d_offset + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Normalize and apply weight/bias
    for d_offset in range(0, D, BLOCK_D):
        offs_d = d_offset + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        ptr = x_ptr + offs_m[:, None] * D + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]
        tl.store(y_ptr + offs_m[:, None] * D + offs_d[None, :], out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])
