"""
TriMul (outgoing) – fully‑fused Triton forward pass
==================================================
The implementation follows the reference PyTorch module but all heavy
tensor work is done with custom Triton kernels:

1️⃣  Row‑wise LayerNorm over the flattened input (FP32 → FP16).
2️⃣  Fuse the five linear projections (left, right, left_gate,
    right_gate, out_gate) into one matrix‑multiply (cuBLAS) producing a
    (R, 5·H) tensor (`R = B·N·N`).
3️⃣  Split the fused projection, apply the optional mask, compute the
    sigmoid gates and store three tensors:
       left   [B, H, N, N], right  [B, H, N, N] and
       out_gate [B, H, N, N]   (all FP16).
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    resulting in `out` of shape [B, H, N, N] (FP16).
5️⃣  Row‑wise statistics (mean & rsqrt‑variance) of `out` over the
    hidden dimension H are computed by a Triton kernel.
6️⃣  Final kernel fuses the second LayerNorm, the out‑gate, and the
    output linear projection (H → C).  The result is written as a
    float‑32 tensor of shape [B, N, N, C].

All kernels are written for Triton 3.3.1 and target an H100 (mixed‑
precision, FP16 for bulk work, FP32 for reductions).  The public entry
point is `custom_kernel`, which receives a tuple
`(input, mask, weights, config)` and returns the float‑32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, C]   FP32 input
    y_ptr,      # [R, C]   FP16 output
    w_ptr,      # [C]      LayerNorm weight (γ) FP32
    b_ptr,      # [C]      LayerNorm bias   (β) FP32
    rows, C, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: compute mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(C, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < C

        x = tl.load(
            x_ptr + offs_m[:, None] * C + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(x, axis=1)                       # [BLOCK_M]
        sum_sq  += tl.sum(x * x, axis=1)                   # [BLOCK_M]
    # mean & rsqrt‑variance
    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values -----
    for d_off in range(0, tl.cdiv(C, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < C

        x = tl.load(
            x_ptr + offs_m[:, None] * C + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        y = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * C + offs_d[None, :],
            y.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R]  FP16 (optional; dummy if not used)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat index r
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑position mask (broadcast over hidden dim)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride for a single hidden plane (N×N)
    hidden_stride = N * N
    # base offset for hidden‑plane 0
    base = b * (H * hidden_stride) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offset inside the fused projection:
        #   [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw     = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for better accuracy
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N]  FP16
    mu_ptr,    # [R]           FP32
    rstd_ptr,  # [R]           FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # [BLOCK_M]

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)       # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N]  FP16
    gate_ptr,       # [B, H, N, N]  FP16  (out_gate)
    mu_ptr,         # [R]           FP32
    rstd_ptr,       # [R]           FP32
    ln_w_ptr,       # [H]           FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]           FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]      FP16 (transpose of to_out.weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # [BLOCK_M]

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # hidden‑plane values
        out_vals = tl.load(
            out_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                       # [M, K]

        gate_vals = tl.load(
            gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                       # [M, K]

        # LayerNorm over hidden dim (γ,β)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd  →  normalized hidden vector
        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by out‑gate
        term = norm * gate_vals                                 # [M, K]
        term_fp16 = term.to(tl.float16)                         # back to fp16 for dot

        # slice of final weight matrix (H, dim)
        w_tile = tl.load(
            w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                        # [K, BLOCK_N]

        # accumulate: (M,K)·(K,N) → (M,N)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final output
    dst = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul (no gradient)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32 tensors on the same device as the inputs)
            - config_dict must contain:
                "dim" (input channel C),
                "hidden_dim" (H),
                "nomask" (bool, True → no mask applied)

    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    # concatenate the five weight matrices (each [hidden_dim, dim])
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, C]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] for the two‑sided bmm
    left_mat = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)             # [B·H, N, N] FP16

    # out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                         # H ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    # transpose final weight once (shape [H, dim])
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
