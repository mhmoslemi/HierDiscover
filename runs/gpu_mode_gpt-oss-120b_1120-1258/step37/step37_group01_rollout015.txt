"""
Outgoing TriMul (TriMul‑out) – Triton‑accelerated forward pass (inference only)

The kernel implements the “outgoing” variant of the TriMul operator described in
AlphaFold‑3.  The computation follows the PyTorch reference:

    x          = LayerNorm(x)                               # [B,N,N,D]
    left,right = Linear(x) (two separate projections)       # [B,N,N,H]
    left_gate  = sigmoid(Linear(x))                         # [B,N,N,H]
    right_gate = sigmoid(Linear(x))                         # [B,N,N,H]
    out_gate   = sigmoid(Linear(x))                         # [B,N,N,H]

    left  = left  * mask * left_gate
    right = right * mask * right_gate

    # core pairwise update:
    out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]   (H independent GEMMs)

    # second LayerNorm over the hidden dimension H
    out = LayerNorm(out)          # (mean / var computed per (b,i,j) location)
    out = out * out_gate
    out = Linear(out)             # project back to original dim D

All heavy‑weight work is fused in Triton kernels:

1️⃣ Row‑wise LayerNorm (float32 → float16) over D.
2️⃣ One fused linear projection (5·H outputs) computed in PyTorch,
   then split & gated in a Triton kernel (including optional mask).
3️⃣ Batched GEMM (cuBLAS) for the einsum.
4️⃣ Triton kernel computes per‑position mean / rsqrt‑variance across H.
5️⃣ Final kernel fuses the second LayerNorm, out‑gate multiplication and
   the projection back to D (gemm with the transposed to_out weight).

The implementation works for arbitrary batch size (B≤2), sequence length N,
model dimension D∈{128,384} and hidden dimension H=128 (as used in the tests).
Masking is optional (controlled by `config["nomask"]`).
The output is a float32 tensor of shape [B,N,N,D].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, D]  FP32 input
    y_ptr,      # [R, D]  FP16 output
    w_ptr,      # [D]    γ (FP32)
    b_ptr,      # [D]    β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Layer‑norm each of `rows` rows of length `dim`."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --- first pass: compute mean & variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --- second pass: write normalized values --------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16   (output)
    right_ptr,     # [B, H, N, N] FP16   (output)
    out_gate_ptr,  # [B, H, N, N] FP16   (output)
    mask_ptr,      # [R] FP16 (optional, broadcast)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Extract left/right slices, apply mask, sigmoid gates."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r -> (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional per‑pair mask (broadcast over hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride for one hidden plane
    base_offset = b * (H * hidden_stride) + i * N + k   # start offset for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Linear layout of the fused projection:
        #   [ left | right | left_gate | right_gate | out_gate ]   each of shape (R, H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid in FP32 for stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute per‑row mean and rsqrt‑variance across the hidden dimension."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)       # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m,  mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """Normalize `out`, multiply by `out_gate`, apply LN and linear proj."""
    pid_m = tl.program_id(0)   # rows tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]  FP32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]  FP32

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)            # [K]
        mask_k = offs_k < H

        # Load hidden‑plane values and gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm over hidden dim (γ, β)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd  →  normalized hidden vector
        diff = (vals_out - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]

        # Multiply by out_gate
        term = norm * vals_gate                                    # [M, K]

        # Cast to FP16 for the GEMM with the final projection weight
        term_fp16 = term.to(tl.float16)

        # Load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # Accumulate GEMM contribution
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)   # (M,K)·(K,N) → (M,N)

    # Write final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Triton entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all model weights (float32)
                Required keys:
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"
            - config_dict  : must contain keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]                # D (e.g. 128 or 384)
    hidden_dim = cfg["hidden_dim"]  # H (always 128 in the provided tests)
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape          # x: [B, N, N, D]
    R = B * N * N                  # total number of (b,i,k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # reshape to [R, D] to treat every (b,i,k) as an independent row
    x_flat = x.reshape(R, dim).contiguous()                     # [R, D] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128   # rows per program
    BLOCK_D = 64    # D‑wise tile
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # free original fp32 input as it is no longer needed
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    # Concatenate the five projection matrices (each [H, D]) -> [5*H, D]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, D]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16
    del fused_w, x_norm                 # release memory we no longer need

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # flatten mask to shape [R] and cast to half
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never accessed when HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64   # rows per program (over R)
    BLOCK_H = 64   # hidden dimension tile
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # clean up intermediate projection
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right have shape [B, H, N, N]; we want to compute
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # which is equivalent to a batch of H independent matrix multiplications.
    left_mat = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                   # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2)) # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256   # rows per program for stats
    BLOCK_K_STATS = 128   # hidden‑dim chunk (H ≤ 128)
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    # to_out weight has shape [dim, H]; we need its transpose [H, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk (tuned for H ≤ 128)
    BLOCK_N_OUT = 128   # column (output‑dim) chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
