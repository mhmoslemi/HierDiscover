Pseudo-code:

@triton.jit
def proj_gate_kernel(
    x_ptr,         # [R, dim] FP16
    mask_ptr,      # [R] FP16
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # same
    out_gate_ptr,  # same
    weight_ptr,    # [5, H, dim] FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k for each row index r
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Compute base offset in left/right/out_gate tensor for hidden h=0:
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k  # each hidden plane offset added

    # Loops over hidden dimension in blocks
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Process dot product over dim
        # We'll load weight slices for each projection (5).
        # weight_ptr layout: offset = proj_id * H * dim + offs_h[:, None] * dim + d_off

        # We'll accumulate result in registers
        left_proj = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_proj = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        for d_off in range(0, dim, BLOCK_D):
            offs_d = d_off + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load x slice
            x_ptrs = x_ptr + (offs_r[:, None] * dim + offs_d[None, :])
            x_vals = tl.load(x_ptrs, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            # load weight slices: for each projection id p
            # weight offset formula: p * H * dim + (offs_h[:, None] * dim) + offs_d[None, :]
            # we can loop over each p separately.
            # left_proj weight:
            w_left_ptr = weight_ptr + (0 * H * dim) + (offs_h[:, None] * dim) + offs_d[None, :]
            w_left = tl.load(w_left_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            left_proj += tl.dot(x_vals, w_left)  # returns [BLOCK_R, BLOCK_H]

            # right weight:
            w_right_ptr = weight_ptr + (1 * H * dim) + (offs_h[:, None] * dim) + offs_d[None, :]
            w_right = tl.load(w_right_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            right_proj += tl.dot(x_vals, w_right)

            # left_gate weight:
            w_lgate_ptr = weight_ptr + (2 * H * dim) + (offs_h[:, None] * dim) + offs_d[None, :]
            w_lgate = tl.load(w_lgate_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            left_gate += tl.dot(x_vals, w_lgate)

            # right_gate weight:
            w_rgate_ptr = weight_ptr + (3 * H * dim) + (offs_h[:, None] * dim) + offs_d[None, :]
            w_rgate = tl.load(w_rgate_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            right_gate += tl.dot(x_vals, w_rgate)

            # out_gate weight:
            w_outgate_ptr = weight_ptr + (4 * H * dim) + (offs_h[:, None] * dim) + offs_d[None, :]
            w_outgate = tl.load(w_outgate_ptr, mask=mask_h[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            out_gate += tl.dot(x_vals, w_outgate)

        # after dim loop: compute sigmoid gates and mask left/right
        # left_gate sigmoid
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

        # apply mask if needed
        if HAS_MASK:
            # load mask values for rows
            mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # shape [BLOCK_R]
            # broadcast
            mask_val = mask_val[:, None]  # [BLOCK_R,1]
            left_proj = left_proj * mask_val
            right_proj = right_proj * mask_val

        # multiply by gates
        left = left_proj * left_gate
        right = right_proj * right_gate
        # out_gate stays as is

        # convert to fp16 for storage
        left_fp16 = left.to(tl.float16)
        right_fp16 = right.to(tl.float16)
        out_gate_fp16 = out_gate.to(tl.float16)

        # compute destination offset for each hidden index
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        # store
        tl.store(left_ptr + dst_offset, left_fp16, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right_fp16, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, out_gate_fp16, mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fused‑kernel implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened [B·N·N, C] input (FP32 → FP16).
2️⃣ A single Triton kernel fuses the five linear maps (left, right, left_gate,
   right_gate, out_gate), the optional mask, and the sigmoid gates, writing the
   three tensors `left`, `right` and `out_gate` directly in layout [B, H, N, N].
3️⃣ Batched GEMM (torch.bmm) computes the core contraction
        out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]            (FP16)
4️⃣ A second Triton kernel fuses:
   * row‑wise mean / variance of `out`  (LayerNorm statistics)
   * LayerNorm over the hidden dimension (γ,β)
   * multiplication with `out_gate`
   * the final linear projection `to_out` (H → C)
   The kernel reads `out` and `out_gate` once, performs the statistics,
   normalisation and the final GEMM in a single launch → reduced memory traffic
   and fewer kernel launches.
All arithmetic is mixed‑precision: FP16 for the bulk of the work, FP32 for the
accumulators to retain numerical stability.  The final tensor has dtype
torch.float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, C]  FP32 input
    y_ptr,    # [R, C]  FP16 output
    w_ptr,    # [C]    γ  (FP32)
    b_ptr,    # [C]    β  (FP32)
    rows, C, eps,
    BLOCK_M: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)   # [BLOCK_C]
        mask_c = offs_c < C

        ptr = x_ptr + offs_m[:, None] * C + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_C]

        sum_val += tl.sum(vals, axis=1)                    # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)             # [BLOCK_M]
    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values -------------------------
    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C

        ptr = x_ptr + offs_m[:, None] * C + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * C + offs_c[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_c[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,            # [R, C]               FP16 (layer‑normed input)

    right_ptr,        # [B, H, N, N]         FP16
    out_gate_ptr,     # [B, H, N, N]         FP16
    w_fused_ptr,      # [5, C, H]            FP16  (proj & gate weights, transposed)
    R, H, N, C,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat index r = b·N·N + i·N + k
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # base offset for hidden plane 0 in destination tensors
    hidden_stride = NN
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # optional mask broadcast
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float32)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                              # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Accumulators (FP32)
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ----- dot over the feature dimension C -------------------------
        for d_off in range(0, C, BLOCK_D):
            offs_d = d_off + tl.arange(0, BLOCK_D)               # [BLOCK_D]
            mask_d = offs_d < C

            # load a tile of x (R×C)
            x_tile = tl.load(x_ptr + offs_r[:, None] * C + offs_d[None, :],
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)           # [BLOCK_R, BLOCK_D]

            # helper to load weight slice for projection p (0‑4)
            # layout of w_fused: [p, C, H] row‑major → offset = p*C*H + d*H + h
            base_w = 0                                   # we will add p*C*H later

            # left_proj (p=0)
            w_left  = tl.load(w_fused_ptr
                               + (0 * C * H) + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)      # [BLOCK_D, BLOCK_H]
            left_acc      += tl.dot(x_tile, w_left)                     # [R,H]

            # right_proj (p=1)
            w_right = tl.load(w_fused_ptr
                               + (1 * C * H) + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            right_acc     += tl.dot(x_tile, w_right)

            # left_gate (p=2)
            w_lgate = tl.load(w_fused_ptr
                               + (2 * C * H) + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            left_gate_acc += tl.dot(x_tile, w_lgate)

            # right_gate (p=3)
            w_rgate = tl.load(w_fused_ptr
                               + (3 * C * H) + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            right_gate_acc+= tl.dot(x_tile, w_rgate)

            # out_gate (p=4)
            w_ogate = tl.load(w_fused_ptr
                               + (4 * C * H) + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            out_gate_acc  += tl.dot(x_tile, w_ogate)

        # sigmoid on gate tensors (FP32 → FP16 later)
        left_gate_sig  = 1.0 / (1.0 + tl.exp(-left_gate_acc))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate_acc))
        out_gate_sig   = 1.0 / (1.0 + tl.exp(-out_gate_acc))

        # apply optional mask on left/right before gate multiplication
        left  = (left_acc  * mask_val) * left_gate_sig   # [R,H]
        right = (right_acc * mask_val) * right_gate_sig
        outg  = out_gate_sig                               # no mask

        # cast to FP16 for storage
        left_fp  = left.to(tl.float16)
        right_fp = right.to(tl.float16)
        outg_fp  = outg.to(tl.float16)

        # destination offset for hidden‑plane block
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R,H]

        tl.store(left_ptr      + dst_off, left_fp,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_off, right_fp,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_off, outg_fp,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Final LayerNorm + out_gate + projection (merged with stats)
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,          # [B, H, N, N] FP16   – result of the contract
    out_gate_ptr,     # [B, H, N, N] FP16
    ln_w_ptr,         # [H]               FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H]               FP32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, C]            FP16 (to_out weight transposed)
    out_final_ptr,    # [B, N, N, C]      FP32
    total_rows, H, N, C,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < C

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = NN
    base_offset = b * (H * hidden_stride) + i * N + j      # start of hidden plane 0

    # ------------------------------------------------------------------
    # Phase 1: compute mean & variance across H for each (b,i,j)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)   # [M,K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                     # [BLOCK_M]

    # ------------------------------------------------------------------
    # Phase 2: normalized value * gate * final projection
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < H

        # load out & out_gate values
        ptr_out    = out_ptr      + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate   = out_gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # LayerNorm parameters (γ,β) per hidden channel
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)   # [K]

        # Normalisation: ((out - μ)·rstd)·γ + β
        mu_tile   = mu[:, None]
        rstd_tile = rstd[:, None]

        norm = ((out_vals - mu_tile) * rstd_tile) * gamma[None, :] + beta[None, :]

        term = norm * gate_vals                     # [M,K]

        # Cast to FP16 for the final GEMM (fp16·fp16 → fp32)
        term_fp16 = term.to(tl.float16)

        # Load a tile of the final weight (H, C) – stored transposed as [H, C]
        w_ptr = w_out_T_ptr + offs_k[:, None] * C + offs_n[None, :]          # [K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                         # [K, BLOCK_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                     # [M,N]

    # Write final output
    dst_ptr = out_final_ptr + offs_m[:, None] * C + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]               # C
    hidden_dim = cfg["hidden_dim"] # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                    # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 128
    BLOCK_C_LN = 64
    grid_ln = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_C=BLOCK_C_LN,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + mask + sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # stack weights as (5, C, H)   (transposed for dot‑product)
    w_fused = torch.stack(
        [
            w["left_proj.weight"].t(),
            w["right_proj.weight"].t(),
            w["left_gate.weight"].t(),
            w["right_gate.weight"].t(),
            w["out_gate.weight"].t(),
        ],
        dim=0,
    ).contiguous().to(torch.float16)        # [5, C, H]

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy tensor

    BLOCK_R = 64
    BLOCK_H = 64
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_proj](
        x_norm,
        mask_ptr,
        left,
        right,
        out_gate,
        w_fused,
        R,
        H,
        N,
        dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, w_fused, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 4️⃣ Final LayerNorm + out_gate + projection (merged kernel)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, C] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
```

This implementation follows the reference TriMul computation while
* fusing the five linear projections, mask handling and sigmoid gates into a single Triton kernel,
* merging the per‑position mean/variance computation, second LayerNorm, out‑gate multiplication and final projection into one kernel,
* keeping the heavy pair‑wise contraction (`left @ rightᵀ`) as a highly‑optimised cuBLAS batched GEMM.

The reduced number of kernel launches and lower memory traffic generally improve
runtime on H100 GPUs while preserving exact numerical results (final output is
float‑32, matching the PyTorch reference).
