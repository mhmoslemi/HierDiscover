"""
TriMul (outgoing) kernel.

Algorithm:
1. LayerNorm the input tensor (float32) across the feature dimension.
2. Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into one batched GEMM (FP16).
3. Triton kernel splits the fused projection into left/right tensors,
   applies optional pair mask, and computes sigmoid gates.
4. Batched half‑precision GEMM computes the core contraction
   (left @ right^T) per hidden channel.
5. The contracted tensor is LayerNorm‑ed across the hidden channel,
   multiplied by the out‑gate, and finally projected back to the
   original feature dimension using a single GEMM.
Everything is forward‑only; only the split/fuse kernel is written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair mask (scalar per r)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base index of the first hidden channel for this (b,i,k) position
    base = b * (H * hidden_stride) + i * N + k                # start of channel 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [R,H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for better precision, cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [R,H]

        tl.store(left_ptr     + dst, left,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,     mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the same keys as the reference module
            - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input layer‑norm (FP32 → FP16)
    # --------------------------------------------------------------
    # reshape to [R, dim] and apply LayerNorm in FP32, then cast to FP16
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = F.layer_norm(x_flat, (dim,), weight=w["norm.weight"],
                          bias=w["norm.bias"], eps=eps).to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)       # [5·H, dim]
    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    # free intermediate tensors
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split projections, apply (optional) pair mask and sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # flatten mask to [R] and cast to fp16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM: compute Σₖ left[...,k,:]·right[...,k,:]^T
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # (B·H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                           # (B, H, N, N) FP16

    # free the raw projections – keep only what we need later
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Per‑pair LayerNorm across the hidden dimension,
    #    element‑wise gate, and final projection.
    # --------------------------------------------------------------
    # bring hidden dimension to the innermost axis: (B, N, N, H)
    out_perm = out.permute(0, 2, 3, 1).contiguous()           # (B, N, N, H)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    del out, out_gate

    out_flat = out_perm.view(R, H)          # (R, H) FP16
    gate_flat = out_gate_perm.view(R, H)    # (R, H) FP16

    # LayerNorm on the hidden dimension (FP32 for statistics, keeping FP16 values)
    out_norm = F.layer_norm(out_flat, (H,), weight=w["to_out_norm.weight"],
                            bias=w["to_out_norm.bias"], eps=eps)

    # apply out‑gate
    gated = out_norm * gate_flat   # (R, H) FP16

    # final linear projection back to `dim`
    final = F.linear(gated, w["to_out.weight"])   # (R, dim) FP16
    final = final.float().view(B, N, N, dim)      # (B, N, N, dim) FP32

    return final

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (left ⋅ rightᵀ)  → out  [B, H, N, N]  FP16
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # (B·H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                           # (B, H, N, N) FP16
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate + final projection
    # --------------------------------------------------------------
    # Move hidden dimension to the innermost position so that we can
    # treat each (b,i,j) pair as a row of length H.
    out_perm      = out.permute(0, 2, 3, 1).contiguous()          # (B, N, N, H)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    del out, out_gate

    out_flat = out_perm.view(R, H)          # (R, H) FP16
    gate_flat = out_gate_perm.view(R, H)    # (R, H) FP16
    del out_perm, out_gate_perm

    # LayerNorm across the hidden dimension (statistics in FP32, keep FP16)
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                      # (R, H) FP16

    # Apply the out‑gate
    gated = out_norm * gate_flat            # (R, H) FP16

    # Final linear projection back to `dim` (use half‑precision weight)
    to_out_w = w["to_out.weight"].to(torch.float16)   # (dim, H) FP16
    final = F.linear(gated, to_out_w)                # (R, dim) FP16
    final = final.float().view(B, N, N, dim)         # (B, N, N, dim) FP32

    return final
```
