"""
Outgoing TriMul (forward‑only) – Triton implementation
====================================================

Algorithm
---------
1️⃣ **Layer‑norm** over the last dimension of the input tensor.
   The kernel works on the flattened view `[R, C]` where `R = B·N·N`.
   Statistics (mean / variance) are computed per‑row in FP32 and the
   normalized result is written in FP16.

2️⃣ **Fused projection** – the five linear maps
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`)
   are concatenated into a single weight matrix `[5·H, C]` (FP16) and
   applied with a single `torch.nn.functional.linear`.  The result
   has shape `[R, 5·H]` in FP16.

3️⃣ **Split‑fuse‑gate** – a Triton kernel decodes the flat row index
   `r = b·N·N + i·N + k` and writes three tensors:
   `left[b, h, i, k]`, `right[b, h, i, k]`, `out_gate[b, h, i, k]`
   (all FP16).  An optional mask is applied before gating, and the
   sigmoid gates are computed in FP32 then cast back to FP16.

4️⃣ **Core einsum** – the operation
   `out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]`
   is performed with a batched FP16 GEMM (`torch.bmm`), yielding a
   tensor `[B, H, N, N]`.

5️⃣ **Row‑wise statistics** – a second Triton kernel computes per‑row
   (i.e. per `(b,i,j)`) mean and rsqrt‑variance over the hidden
   dimension `H`.  The statistics are stored in FP32 buffers `mu` and
   `rstd`.

6️⃣ **Final LayerNorm + out‑gate + projection** – a fused Triton kernel
   evaluates the second LayerNorm (`to_out_norm`), multiplies by the
   output gate, and projects back to the original dimension `C` with
   the transposed weight `W_outᵀ`.  The kernel processes a tile of rows
   (`BLOCK_M`) and a tile of output channels (`BLOCK_N`) at a time,
   using `tl.dot` (Tensor‑Core) on FP16 intermediate values and writes
   the final result in FP32.

All kernels are written for Triton 3.3.1 and target NVIDIA H100.
Only the forward pass is implemented – no autograd support is required.

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, C]  input FP32
    y_ptr,            # [R, C]  output FP16
    w_ptr, b_ptr,    # [C]    LN weight & bias (FP32)
    rows, cols, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ---------- pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(cols, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < cols

        ptr = x_ptr + offs_r[:, None] * cols + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / cols
    var  = sum_sq / cols - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalized output ----------
    for c_off in range(0, tl.cdiv(cols, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < cols

        ptr = x_ptr + offs_r[:, None] * cols + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_r[:, None] * cols + offs_c[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_c[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16  fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index -> (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑pair mask (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mval = mval[:, None]                                       # [BLOCK_R, 1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k          # start of h=0 plane

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # each of the five projections occupies a contiguous chunk of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_g_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(proj_ptr + proj_base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        outg_raw    = tl.load(proj_ptr + proj_base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid (computed in FP32 for numerical stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑var over hidden dimension H
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N] FP16
    mu_ptr, rstd_ptr, # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of h=0 plane

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride   # [R, K]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)


# ----------------------------------------------------------------------
# 4️⃣ Final output: 2nd LayerNorm + out‑gate + projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N] FP16
    gate_ptr,         # [B, H, N, N] FP16
    mu_ptr, rstd_ptr, # [R] FP32
    ln_w_ptr, ln_b_ptr,                # [H] FP32 (to_out_norm weight/bias)
    w_out_T_ptr,                       # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,                     # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0)    # rows (R) tile
    pid_d = tl.program_id(1)    # output‑dim (dim) tile

    row_start = pid_r * BLOCK_R
    dim_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)    # [BLOCK_R]
    offs_d = dim_start + tl.arange(0, BLOCK_D)    # [BLOCK_D]
    mask_r = offs_r < total_rows
    mask_d = offs_d < dim

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j    # start of h=0 plane

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r,  other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r,  other=0.0)   # [BLOCK_R]

    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # (out - mu) * rstd  ->  normalized hidden tensor   (FP32)
        out_tile  = tl.load(out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [R, K]
        gate_tile = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [R, K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out‑µ)·rstd·γ + β)
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by the output gate
        term = norm * gate_tile                     # [R, K]  FP32
        term_fp16 = term.to(tl.float16)            # cast for Tensor‑Core dot

        # weight tile (H×dim) – note we store W_outᵀ so we can load contiguous rows
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_d[None, :]   # [K, D]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_d[None, :],
                         other=0.0)                                 # [K, D]

        # accumulate partial result
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                # [R, D]
    # write final FP32 result
    dst_ptr = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_r[:, None] & mask_d[None, :])

    Args

    """

    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R_LN = 256
    BLOCK_C_LN = 64
    grid_ln = (triton.cdiv(R, BLOCK_R_LN),)

        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R_LN,
        BLOCK_C=BLOCK_C_LN,
        num_warps=8,
    )
    del x_flat

    # 2️⃣ Fused projection (5×hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, C]

    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy buffer

    BLOCK_R_SPLIT = 256
    BLOCK_H_SPLIT = 64
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,
        num_warps=8,
    )

    # 4️⃣ Core einsum: left·rightᵀ (batched GEMM)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑var over hidden dimension H
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 64   # H ≤ 128 in all supported configurations
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out_gate + projection back to C
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, C]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 256   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk processed per iteration
    BLOCK_D_OUT = 128   # output‑dim (C) chunk per program

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_D_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ   (FP32)
        w["to_out_norm.bias"],     # β   (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_D=BLOCK_D_OUT,
        num_warps=8,
    )
    # --------------------------------------------------------------
    return final
```
