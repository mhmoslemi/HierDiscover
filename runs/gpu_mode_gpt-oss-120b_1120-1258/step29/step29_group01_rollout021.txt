def norm_gate_kernel(out_ptr, mu_ptr, rstd_ptr, gamma_ptr, beta_ptr, gate_ptr, gated_ptr, total_rows,R, H, N, eps, BLOCK_M,BLOCK_H):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < total_rows

   nn = N*N
   b = offs_m // nn
   rem = offs_m - b * nn
   i = rem // N
   j = rem - i * N

   # compute offsets to hidden dimension base
   hidden_stride = N * N
   base_offset = b * (H * hidden_stride) + i * N + j

   # load mu, rstd per row
   mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # shape [BLOCK_M]
   rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

   # broadcast to hidden dimension, we process a chunk of H each loop.
   for h_start in range(0, H, BLOCK_H):
       offs_h = h_start + tl.arange(0, BLOCK_H)
       mask_h = offs_h < H

       # loads out values
       ptr_out = out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
       vals_out = tl.load(ptr_out, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

       # load gate values
       ptr_gate = gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
       vals_gate = tl.load(ptr_gate, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

       # gamma and beta
       gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)
       beta = tl.load(beta_ptr + offs_h, mask=mask_h, other=0.0)

       # compute norm = ((out - mu) * rstd) * gamma + beta  # all fp32
       normed = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None,:]

       gated = normed * vals_gate

       # store gated to gated_ptr (same layout as out)
       dst_ptr = gated_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
       tl.store(dst_ptr, gated.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fast Triton‑only forward implementation.

The operator works on a 4‑D tensor X ∈ ℝ^{B·N·N·C} and produces Y ∈ ℝ^{B·N·N·C}.
The computation is

1️⃣  Row‑wise LayerNorm over the channel dimension C (FP32→FP16).
2️⃣  Fused Linear (weight shape 5·H×C) → (left, right, left_gate, right_gate, out_gate).
3️⃣  Apply optional pair‑wise mask, sigmoid the three gates and write
    left/right/out_gate tensors of shape [B, H, N, N] (FP16).
4️⃣  Batched GEMM (cuBLAS) computes
       out_{b,i,j,h} = Σ_k left_{b,i,k,h}·right_{b,j,k,h}   (FP16).
5️⃣  Row‑wise mean/variance of out over the hidden dimension H (μ, rstd, FP32).
6️⃣  Kernel normalises out, applies the second LayerNorm (γ, β),
    multiplies with out_gate and writes a gated tensor G.
7️⃣  Final projection with a regular torch.nn.functional.linear
    (G  *  W_out) → Y  (FP32).  No gradients are required.

All heavy work (normalisations, mask handling, gate computation) is done in
Triton kernels; the two large matrix‑multiplications are delegated to cuBLAS.
The implementation supports both masked / un‑masked inputs and the
configuration space required by the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,               # [R, 5·H]  (FP16)
    left_ptr, right_ptr, out_gate_ptr,   # [B, H, N, N] (FP16)
    mask_ptr,               # [R] (FP16) – optional
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)    # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcasted over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of h‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Linear offset inside the fused projection: [left|right|l_gate|r_gate|out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 for stability, back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # H ≤ 128 in all test configs → a single iteration suffices
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Normalise out, apply second LayerNorm, gate and store G
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr, gate_ptr,               # [B, H, N, N] FP16
    mu_ptr, rstd_ptr,               # [R] FP32
    ln_w_ptr, ln_b_ptr,             # [H] FP32  (γ, β of to_out_norm)
    gated_ptr,                      # [B, H, N, N] FP16  (output of this kernel)
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # load raw out values
        ptr_out  = out_ptr  + base_offset[:, None] + offs_h[None, :] * hidden_stride
        val_out  = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)     # [BLOCK_M, BLOCK_H]

        # load gate values
        ptr_gate = gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
        val_gate = tl.load(ptr_gate,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)     # [BLOCK_M, BLOCK_H]

        # layer‑norm parameters γ, β (FP32)
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        # (out - μ) * rstd   →   LayerNorm →  * gate
        normed = ((val_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        gated  = normed * val_gate

        # store gated tensor (still FP16)
        dst_ptr = gated_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(dst_ptr,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments:
        data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the keys used in the reference module
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat, x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R, dim, eps,
        BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·H) – produces left/right + gates
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim] (FP16)

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left, right, out_gate,
        mask_ptr,
        R, H, N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R, BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # shape: [B, H, N, N] → view as [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 fits in a single tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R, H, N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Normalise out, apply to_out_norm, gate → G
    # ------------------------------------------------------------------
    gated = torch.empty_like(out)   # same layout as out: [B, H, N, N] FP16

    BLOCK_M_G = 128
    BLOCK_H_G = 64
    grid_norm_gate = (triton.cdiv(R, BLOCK_M_G),)
    norm_gate_kernel[grid_norm_gate](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],    # γ (FP32)
        w["to_out_norm.bias"],      # β (FP32)
        gated,
        R, H, N,
        BLOCK_M=BLOCK_M_G,
        BLOCK_H=BLOCK_H_G,
        num_warps=8,
    )
    # μ, rstd, and the intermediate tensors `out` / `out_gate` are no longer needed.
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final projection (G → Y) – use cuBLAS linear for speed.
    # ------------------------------------------------------------------
    # rearrange G to [R, H] (contiguous H‑last layout)
    gated_perm = gated.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]
    gated_flat = gated_perm.view(R, H)                     # [R, H] FP16

    # linear projection (FP16 → FP32 output)
    # weight shape: [dim, H] (float32)
    final_flat = F.linear(gated_flat.float(), w["to_out.weight"])   # [R, dim] float32
    final = final_flat.view(B, N, N, dim)                           # [B,N,N,dim]

    return final
