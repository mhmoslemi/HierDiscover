@triton.jit
def fused_proj_gate_kernel(
    x_ptr,  # [R, dim] FP16
    left_ptr, right_ptr, out_gate_ptr,
    w_left_ptr, w_right_ptr,
    w_lgate_ptr, w_rgate_ptr, w_outgate_ptr,
    mask_ptr,   # [R] FP16 optional
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BR]
    mask_r = offs_r < R

    # decode indices to b,i,k for left/right storage
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # broadcast H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset for storing results in layout [B, H, N, N]
    hidden_stride = N * N  # per hidden dim
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Compute left projection
        # compute dot between x_norm rows and weight slice
        # accumulator for left_proj, right_proj, etc.
        acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)
        acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)
        acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)
        acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)
        acc_outgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)

        for d_start in range(0, dim, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load x rows (R x D)
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0)  # [BR, BD]

            # load weight slices for each group
            w_left = tl.load(w_left_ptr + offs_d[:, None] * H + offs_h[None, :],
                            mask=mask_d[:, None] & mask_h[None, :],
                            other=0.0)  # [BD, BH]
            w_right = tl.load(w_right_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0)
            w_lgate = tl.load(w_lgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0)
            w_rgate = tl.load(w_rgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0)
            w_outgate = tl.load(w_outgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0)

            # dot: acc_left += x @ w_left
            acc_left += tl.dot(x, w_left)   # [BR, BH]
            acc_right += tl.dot(x, w_right)
            acc_lgate += tl.dot(x, w_lgate)
            acc_rgate += tl.dot(x, w_rgate)
            acc_outgate += tl.dot(x, w_outgate)

        # sigmoid for gates
        left_gate = (1.0 / (1.0 + tl.exp(-acc_lgate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_rgate.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-acc_outgate.to(tl.float32)))).to(tl.float16)

        left = acc_left * mask_val * left_gate
        right = acc_right * mask_val * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride  # [BR, BH]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])
```

"""
TriMul (outgoing) – fast Triton forward implementation
====================================================

Algorithm (forward‑only)
------------------------
1️⃣ Row‑wise LayerNorm on the flattened [B·N·N, C] tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single Triton kernel.
   The kernel reads the normalised input, multiplies it with the
   (transposed) weight matrices and directly writes the three tensors
   `left`, `right` and `out_gate` in layout [B, H, N, N] (FP16),
   applying the optional mask and the sigmoid gates.
3️⃣ Batched GEMM (torch.bmm) computes the core einsum
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   producing an intermediate tensor `[B, H, N, N]` (FP16).
4️⃣ Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
   (Triton kernel) → μ and rstd per (b,i,j) in FP32.
5️⃣ Final Triton kernel fuses the second LayerNorm, the out‑gate
   multiplication and the output projection.  For each row (b,i,j) it
   computes
        out_fp32 = Σₕ ( ((out‑μ)·rstd·γ + β) * gate )·W_out[h,d]
   and writes the final tensor `[B, N, N, C]` (float32).

All kernels target Triton 3.3.1 (H100) and use mixed‑precision
(FP16 for the heavy inner loops, FP32 for the reductions and the
final result).  The implementation works for all configurations
used in the test suite and honours the optional mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,          # [R, dim] input FP32, output FP16
    w_ptr, b_ptr,          # [dim] LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    mask_m = offs_m < rows

    # ---------- first pass: accumulate mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized result ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gating (left, right, out_gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,                     # [R, dim]                FP16
    left_ptr, right_ptr, out_gate_ptr,   # [B, H, N, N]  FP16
    w_left_ptr, w_right_ptr,                # [dim, H]  FP16
    w_lgate_ptr, w_rgate_ptr, w_outgate_ptr,   # [dim, H]  FP16
    mask_ptr,                 # [R]   FP16 (optional)
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BR]
    mask_r = offs_r < R

    # decode (b,i,k) for the storage layout
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BR,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset for (b, h, i, k) in the flattened layout
    hidden_stride = N * N               # stride between successive hidden dims
    base_offset = b * (H * hidden_stride) + i * N + k     # start of h=0 plane

    # iterate over hidden dimension in tiles
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulators for the five linear maps
        acc_left   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)
        acc_right  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)
        acc_lgate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)
        acc_rgate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)
        acc_outgate= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float16)

        # loop over the input feature dimension
        for d_start in range(0, dim, BLOCK_D):
            offs_d = d_start + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load a tile of the normalised input
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0)                     # [BR, BD]

            # load corresponding weight tiles (transposed layout)
            wL = tl.load(w_left_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0)                     # [BD, BH]
            wR = tl.load(w_right_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0)
            wLG = tl.load(w_lgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0)
            wRG = tl.load(w_rgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0)
            wOG = tl.load(w_outgate_ptr + offs_d[:, None] * H + offs_h[None, :],
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0)

            # dot products for each projection
            acc_left    += tl.dot(x, wL)          # [BR, BH]
            acc_right   += tl.dot(x, wR)
            acc_lgate   += tl.dot(x, wLG)
            acc_rgate   += tl.dot(x, wRG)
            acc_outgate += tl.dot(x, wOG)

        # sigmoid for gates (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-acc_lgate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-acc_rgate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-acc_outgate.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = acc_left  * mval * left_gate
        right = acc_right * mval * right_gate

        # destination offsets for the three outputs
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BR, BH]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑var over hidden dimension H
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N] FP16
    mu_ptr, rstd_ptr, # [R] FP32  (R = B·N·N)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of h=0 plane

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m,   mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Final output: second LayerNorm + out‑gate + projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    mu_ptr, rstd_ptr, # [R] FP32
    ln_w_ptr, ln_b_ptr,                # [H] FP32 (to_out_norm weight/bias)
    w_out_T_ptr,                       # [H, dim] FP16 (transposed to_out)
    out_final_ptr,                     # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)    # rows tile (R)
    pid_n = tl.program_id(1)    # output‑dim tile (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)    # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)    # [N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j    # start of plane h=0 for (b,i,j)

    # per‑row μ, rstd
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # contracted tensor and gate (FP16 → FP32)
        out_tile = tl.load(out_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)            # [M, K]

        gate_tile = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)            # [M, K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out‑μ)·rstd·γ + β)
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by out_gate
        term = norm * gate_tile                       # [M, K]
        term_fp16 = term.to(tl.float16)               # FP16 for tensor‑core dot

        # weight slice (K, BLOCK_N) – w_out_T is transposed (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                # [M, N]

    # store final FP32 result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gating (left, right, out_gate)
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # transpose weights so that they are [dim, H] (better for dot)
    w_left_T = w["left_proj.weight"].to(torch.float16).t().contiguous()
    w_right_T = w["right_proj.weight"].to(torch.float16).t().contiguous()
    w_lgate_T = w["left_gate.weight"].to(torch.float16).t().contiguous()
    w_rgate_T = w["right_gate.weight"].to(torch.float16).t().contiguous()
    w_outgate_T = w["out_gate.weight"].to(torch.float16).t().contiguous()

    if (not nomask) and (mask is not None):
        # mask broadcast over hidden dimension, cast to FP16 (0.0 / 1.0)
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_D = 32
    grid_fused = (triton.cdiv(R, BLOCK_R),)
    fused_proj_gate_kernel[grid_fused](
        x_norm,
        left,
        right,
        out_gate,
        w_left_T,
        w_right_T,
        w_lgate_T,
        w_rgate_T,
        w_outgate_T,
        mask_ptr,
        R,
        H,
        N,
        dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, mask_ptr, w_left_T, w_right_T, w_lgate_T, w_rgate_T, w_outgate_T

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM (left·rightᵀ) – core einsum
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)             # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                      # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑var over hidden dimension H
    # --------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128               # H ≤ 128 → fits in a tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out_gate + to_out projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows per program (tiles over R)
    BLOCK_K_OUT = 32           # hidden‑dim chunk per iteration
    BLOCK_N_OUT = 128          # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
