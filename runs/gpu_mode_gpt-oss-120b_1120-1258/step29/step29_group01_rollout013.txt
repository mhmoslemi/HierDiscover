"""
TriMul (outgoing) – fast forward‑only Triton implementation.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and produces Y ∈ ℝ^{B×N×N×C}.
The computation consists of

1️⃣ Row‑wise LayerNorm over the last channel dimension C (torch.nn.functional).
2️⃣ A single fused linear (5·H) that generates the three projections
   (left, right) and the three gates (left, right, out) in FP16.
3️⃣ A custom Triton kernel that
   – decodes the flat index (b,i,k) of each token,
   – optionally applies the pair‑mask,
   – applies a sigmoid to the three gate vectors,
   – writes left, right and out_gate tensors with layout [B, H, N, N].
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
        out_{b,h,i,j} = Σ_k left_{b,h,i,k} · right_{b,h,j,k}
   producing a tensor [B, H, N, N] (FP16).
5️⃣ A second LayerNorm over the hidden dimension H (torch.nn.functional),
   multiplication by the out_gate, and a final linear projection back to C
   (all in FP16, final result cast to FP32).

Only the split‑and‑gate step is written in Triton – the rest uses highly
optimised PyTorch kernels (torch.nn.functional.layer_norm, torch.bmm and
torch.nn.functional.linear).  This design keeps the amount of data moved
between kernels small and leverages the best‑available GPU kernels for the
dominant matrix‑multiply work while still satisfying the “use a Triton
kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid → left / right / out_gate
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N]  FP16 output
    right_ptr,     # [B, H, N, N]  FP16 output
    out_gate_ptr,  # [B, H, N, N]  FP16 output
    mask_ptr,      # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,  # rows per program
    BLOCK_H: tl.constexpr,  # hidden dim per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # Decode flat index r = b·N·N + i·N + k   (k is the summed‑over dimension)
    # ------------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16 scalar per row
        mask_val = mask_val[:, None]  # [BLOCK_R, 1] for broadcasting over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Base offset into the contiguous [B, H, N, N] tensors.
    # Tensor layout:  (b, h, i, k) →  offset = b·(H·N·N) + h·(N·N) + i·N + k
    # ------------------------------------------------------------------
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden‑plane 0

    # ------------------------------------------------------------------
    # Process hidden dimension in blocks.
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Linear offset inside the fused projection: [left|right|l_gate|r_gate|out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid in FP32 (more stable) then cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # already masked (gate does not use mask)

        # Destination offsets for the three output tensors.
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total (b,i,k) triples
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over channel dim (torch implementation, fast)
    # ------------------------------------------------------------------
    # Note: we keep the result in FP16 for the subsequent linear projection.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)   # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection that produces 5·H hidden vectors
    # ------------------------------------------------------------------
    # Gather the five weight matrices into a single [5·H, dim] matrix.
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # (H, dim)
            w["right_proj.weight"],  # (H, dim)
            w["left_gate.weight"],   # (H, dim)
            w["right_gate.weight"],  # (H, dim)
            w["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                     # (5·H, dim)

    # Reshape to a flat list of tokens and apply the fused linear.
    x_flat = x_norm.reshape(R, dim)         # (R, dim)  FP16, contiguous
    proj = F.linear(x_flat, fused_w)        # (R, 5·H) FP16
    del x_norm, x_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, compute sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # Flatten mask to the same R ordering as `proj`.
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy tensor

    # Kernel launch configuration
    BLOCK_R = 128   # rows per program (tuned for typical R up to 1M)
    BLOCK_H = 128   # hidden dim block (covers the whole H in a single iteration for H≤128)
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (torch cuBLAS)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) and use torch.bmm.
    left_mat  = left.view(B * H, N, N)                # (B·H, N, N) FP16
    right_mat = right.view(B * H, N, N)               # (B·H, N, N) FP16

    # out_{b,h,i,j} = Σ_k left_{b,h,i,k} * right_{b,h,j,k}
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                              # (B, H, N, N) FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H, gate & final projection
    # ------------------------------------------------------------------
    # Permute to (B, N, N, H) so that LayerNorm can treat H as the trailing dim.
    out_perm = out.permute(0, 2, 3, 1).contiguous()           # (B, N, N, H) FP16

    # LayerNorm parameters for the hidden dimension.
    ln_weight = w["to_out_norm.weight"].to(torch.float16)    # (H,)
    ln_bias   = w["to_out_norm.bias"].to(torch.float16)      # (H,)

    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=ln_weight,
        bias=ln_bias,
        eps=eps,
    )  # (B, N, N, H) FP16

    # Apply the out_gate (broadcasted over the H dimension).
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H) FP16
    out_gated = out_norm * out_gate_perm                        # (B, N, N, H) FP16

    # Final linear projection back to `dim`.
    # Weight shape in the dict is (dim, H). Use it directly with torch.nn.functional.linear.
    w_out = w["to_out.weight"].to(torch.float16)                # (dim, H)
    final_fp16 = F.linear(out_gated.view(-1, H), w_out)        # (B·N·N, dim) FP16

    # Reshape and cast to float32 for the final output.
    final = final_fp16.view(B, N, N, dim).to(torch.float32)    # (B, N, N, dim)

    return final
