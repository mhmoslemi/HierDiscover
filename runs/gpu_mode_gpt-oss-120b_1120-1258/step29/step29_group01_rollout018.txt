"""
TriMul (outgoing) – fast forward‑only Triton implementation.

Algorithm
---------
1️⃣ **Row‑wise LayerNorm** (FP32 → FP16) on the flattened
   [B·N·N, C] tensor.  The mean / variance are computed in‐place
   without extra synchronisation.

2️⃣ **Fused projection + gating**  
   A single matmul (FP16) produces 5·H hidden features
   (left, right, left‑gate, right‑gate, out‑gate).  A Triton kernel
   splits them, applies the optional pair mask, computes the sigmoid
   of the three gates and stores the three tensors
   ``left``, ``right`` and ``out_gate`` (shape [B,H,N,N]).

3️⃣ **Batched GEMM (cuBLAS)**  
   ``out = left @ rightᵀ`` for every (b,h) pair, yielding
   ``out`` of shape [B,H,N,N] (FP16).

4️⃣ **Re‑layout** – we transpose ``out`` and ``out_gate`` to
   [B,N,N,H] so that the hidden dimension becomes the innermost
   (contiguous) one.  This dramatically improves memory locality for
   the following kernels.

5️⃣ **Row‑wise statistics** – a Triton kernel computes the mean and
   rsqrt‑variance of ``out`` over the hidden dimension H, producing
   ``mu`` and ``rstd`` (shape [B·N·N]).

6️⃣ **Final LayerNorm + out‑gate + linear projection**  
   A fused Triton kernel normalises ``out`` using ``mu``/``rstd``,
   applies the learnt LayerNorm (γ,β) and the ``out_gate``, then
   performs a batched matmul with the transposed ``to_out`` weight
   (shape [H, C]).  The result is written as float32 tensor
   ``output`` of shape [B,N,N,C].

All heavy work is done in FP16; only the two LayerNorm statistics
are accumulated in FP32 for numerical stability.  The implementation
covers the full configuration space required by the tests and
maintains the original functional semantics of the PyTorch reference. 
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, C]  FP32 input
    y_ptr,   # [R, C]  FP16 output
    w_ptr,   # [C]    γ (FP32)
    b_ptr,   # [C]    β (FP32)
    rows, C, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- first pass: mean & variance -----------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)      # [BLOCK_C]
        mask_c = offs_c < C

        # Load a tile of the input, shape [BLOCK_R, BLOCK_C]
        ptr = x_ptr + offs_r[:, None] * C + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals,  axis=1)      # sum over C
        sum_sq  += tl.sum(vals * vals, axis=1)  # sum of squares
    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: normalise and apply γ/β ------------------------
    for c_off in range(0, tl.cdiv(C, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < C

        ptr = x_ptr + offs_r[:, None] * C + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_r[:, None] * C + offs_c[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_c[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H] FP16 (output of fused linear)
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]   FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                 # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask (scalar per pair) – broadcast over hidden
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                         # distance between
    # Base offset for hidden‑plane 0 in the destination tensors
    base_off = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # linear offset inside the fused projection: [left|right|lg|rg|og]
        proj_off = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_off,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_off + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_off + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_off + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_off + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 for stability, then back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                   # no mask on out_gate

        # destination offsets (stride = hidden_stride)
        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,            # [R, H] FP16  (contiguous over H)
    mu_ptr,             # [R] FP32
    rstd_ptr,           # [R] FP32
    R, H,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # Load hidden slice for each row
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)          # sum over H
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,            # [R, H] FP16
    gate_ptr,           # [R, H] FP16   (out_gate)
    mu_ptr,             # [R] FP32
    rstd_ptr,           # [R] FP32
    ln_w_ptr,           # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,           # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,        # [H, C] FP16 (transposed to_out weight)
    out_final_ptr,      # [R, C] FP32
    R, H, C,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid_r = tl.program_id(0)          # row tile
    pid_c = tl.program_id(1)          # column tile

    row_start = pid_r * BLOCK_R
    col_start = pid_c * BLOCK_C

    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    offs_c = col_start + tl.arange(0, BLOCK_C)          # [BLOCK_C]

    mask_r = offs_r < R
    mask_c = offs_c < C

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # [BLOCK_R]

    # accumulator for the final matrix‑multiply
    acc = tl.zeros([BLOCK_R, BLOCK_C], dtype=tl.float32)

    # loop over hidden dimension in tiles of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # load hidden values (contiguous over H)
        out_tile  = tl.load(out_ptr  + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)           # [R, H_tile]
        gate_tile = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # LayerNorm over hidden dimension
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [H_tile]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [H_tile]

        # (out - mu) * rstd  -> normalised hidden vector
        norm = (out_tile - mu[:, None]) * rstd[:, None]             # [R, H_tile]
        norm = norm * gamma[None, :] + beta[None, :]               # LN

        # multiply by out_gate (the gating tensor)
        term = norm * gate_tile                                    # [R, H_tile]

        # cast back to fp16 for the dot (fp16·fp16 → fp32)
        term_fp16 = term.to(tl.float16)

        # load a slice of the final weight (H, C)
        w_ptr = w_out_T_ptr + offs_h[:, None] * C + offs_c[None, :]   # [H_tile, C_tile]
        w_tile = tl.load(w_ptr,
                         mask=mask_h[:, None] & mask_c[None, :],
                         other=0.0)                                    # [H_tile, C_tile]

        # accumulate: (R,H_tile)·(H_tile,C_tile) → (R,C_tile)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final result
    dst_ptr = out_final_ptr + offs_r[:, None] * C + offs_c[None, :]
    tl.store(dst_ptr, acc, mask=mask_r[:, None] & mask_c[None, :])

    """

    Parameters
    ----------

            "to_out.weight"
          (all float32)
        - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim          = cfg["dim"]            # C
    hidden_dim   = cfg["hidden_dim"]     # H
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # flattened pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                      # [R, C]
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R   = 256    # rows per program for the LN kernel
    BLOCK_C   = 128    # columns per program for the LN kernel
    grid_ln   = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
        num_warps=8,
    )
    del x_flat  # free original fp32 tensor

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H)  – matmul in torch (cuBLAS)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, C]

    proj = F.linear(x_norm, fused_w)          # (R, 5·H)  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK  = 1
        mask_ptr  = mask_flat
    else:
        HAS_MASK  = 0
        mask_ptr  = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R_SPLIT = 64
    BLOCK_H_SPLIT = 64
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) : left @ rightᵀ
    # ------------------------------------------------------------------
    # Reshape left/right to [B·H, N, N] for bmm
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                 # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Transpose to make H the innermost dimension (contiguous)
    # ------------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out.view(R, H)           # [R, H]
    gate_flat = out_gate.view(R, H)     # [R, H]

    # ------------------------------------------------------------------
    # 6️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_H_STATS = 64   # H ≤ 128, a single tile works well
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out_flat,
        mu,
        rstd,
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_H=BLOCK_H_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm + out_gate + projection to original dim
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, C] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Tiling parameters – tuned for H100
    BLOCK_R_OUT = 128    # rows per program
    BLOCK_H_OUT = 64     # hidden‑dim tile
    BLOCK_C_OUT = 64     # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),        # rows
        triton.cdiv(dim, BLOCK_C_OUT),      # columns (output dim)
    )
    final_output_kernel[grid_out](
        out_flat,
        gate_flat,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final.view(R, dim),        # flatten final tensor to [R, C]
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_H=BLOCK_H_OUT,
        BLOCK_C=BLOCK_C_OUT,
        num_warps=8,
    )

    # reshape back to original 4‑D layout
    final = final.view(B, N, N, dim)
    return final
```
