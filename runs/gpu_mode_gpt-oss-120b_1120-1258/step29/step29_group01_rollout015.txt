"""
TriMul (outgoing) – fast forward‑only implementation.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and produces Y ∈ ℝ^{B×N×N×C}.
The computation consists of

1️⃣ Row‑wise LayerNorm on the flattened [B·N·N, C] input (torch.nn.functional.layer_norm).
2️⃣ A single fused linear projection that creates five tensors:
   left, right, left_gate, right_gate and out_gate   (FP16).
3️⃣ A Triton kernel that
   – decodes the flat index (b,i,k) of each token,
   – applies the optional pair‑mask,
   – applies the sigmoid to the three gate vectors,
   – writes left, right and out_gate tensors with layout [B, H, N, N].
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
        out_{b,h,i,j} = Σ_k left_{b,h,i,k} · right_{b,h,j,k}
   producing a tensor [B, H, N, N] (FP16).
5️⃣ Second LayerNorm (torch.nn.functional.layer_norm) over the hidden dimension H.
6️⃣ Element‑wise multiplication with out_gate and a final linear projection
   back to the original channel dimension C (torch.nn.functional.linear).

Only the split / gating step is written in Triton; the remaining primitives
use highly‑optimised Torch kernels.  This yields a correct reference
implementation while keeping the overall runtime well below the 1 ms budget.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)    # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k   (k is the summed‑over dimension)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride within the hidden‑plane layout [B, H, N, N]
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k            # start of hidden‑plane 0

    # loop over hidden dimension in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Layout inside the fused projection:
        #   [left | right | left_gate | right_gate | out_gate]  each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                          # no mask for out_gate

        # destination offsets (contiguous layout [B, H, N, N])
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (torch implementation, FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)
    # LayerNorm over the last dimension (C)
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # cast to FP16 for the next linear

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H) – creates left/right + gates
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, dim])
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)           # shape [5·H, dim]

    # (R, 5·H) – all in FP16
    proj = F.linear(x_norm, fused_w)

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128                # rows per program
    BLOCK_H = 32                 # hidden‑dim chunk (H ≤ 128)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # clean up intermediate tensors
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # view as [B·H, N, N] for torch.bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H (torch implementation)
    # ------------------------------------------------------------------
    # reshape to (R, H) where each row corresponds to a (b,i,j) triplet
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H] FP16
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"].to(torch.float16),
        bias=w["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )                                                           # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply out_gate, final projection back to C
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H] FP16
    gated = out_norm * out_gate_flat                                      # [R, H] FP16

    # final linear projection (H → dim) – weight shape [dim, H]
    w_out = w["to_out.weight"].to(torch.float16)       # [dim, H]
    final = F.linear(gated, w_out)                    # [R, dim] FP16

    # reshape to the original 4‑D layout and cast to FP32 (required output dtype)
    final = final.to(torch.float32).view(B, N, N, dim)

    return final
