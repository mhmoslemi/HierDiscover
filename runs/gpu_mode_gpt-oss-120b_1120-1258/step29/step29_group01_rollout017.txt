def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B*N*N
    H = hidden

    # 1. First LayerNorm (custom)
    x_flat = x.reshape(R, dim)
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](x_flat, x_norm,
                                                weights["norm.weight"], weights["norm.bias"], R, dim, eps,
                                                BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D, num_warps=8)
    del x_flat

    # 2. Linear projection fused
    fused_w = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"],
                         weights["left_gate.weight"], weights["right_gate.weight"],
                         weights["out_gate.weight"]], dim=0).to(torch.float16) # [5*H, dim]
    proj = F.linear(x_norm, fused_w)  # [R, 5*H]   FP16
    del x_norm, fused_w

    # 3. Split, mask, sigmoid
    # split into 5 chunks
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = proj.chunk(5, dim=1)  # each [R, H]
    # apply mask (if any)
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R,1)  # broadcast
        left_raw = left_raw * mask_flat
        right_raw = right_raw * mask_flat
    # apply sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)
    # apply gates to left/right
    left = left_raw * left_gate
    right = right_raw * right_gate
    # reshape to (B,H,N,N)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    del left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw, left_gate, right_gate

    # 4. Batched GEMM: left @ right^T over k dim (the "k" is the summed dimension)
    # left shape (B*H, N, N); right shape similarly
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H, N, N]
    out = out.view(B, H, N, N)   # FP16
    del left, right

    # 5. Second LayerNorm + out_gate + projection using torch ops
    # flatten rows
    out_flat = out.permute(0,2,3,1).reshape(R, H)   # [R, H]
    out_gate_flat = out_gate.permute(0,2,3,1).reshape(R, H)  # [R, H]
    # compute mean and var in float32
    out_fp32 = out_flat.float()
    mu = out_fp32.mean(dim=1, keepdim=True)  # [R,1]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)   # [R,1]
    rstd = torch.rsqrt(var + eps)
    # layernorm weight & bias
    gamma = weights["to_out_norm.weight"]  # [H] float32
    beta = weights["to_out_norm.bias"]     # [H] float32
    # normalize and apply weight/bias
    out_norm = (out_fp32 - mu) * rstd   # [R,H] FP32
    out_norm = out_norm * gamma + beta   # broadcast [H]
    # apply out_gate (original out_gate is FP16, convert to FP32)
    out_norm = out_norm * out_gate_flat.float()
    # final projection

    # multiply out_norm (FP32) with w_out (FP16) maybe cast to FP32 for matmul

    w_out_fp32 = w_out.float()  # [H, dim]
    y = out_norm @ w_out_fp32   # [R, dim] FP32
    y = y.view(B, N, N, dim)  # final shape
    return y

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr,        # *[rows, dim] input FP32
    y_ptr,        # *[rows, dim] output FP16
    w_ptr,        # *[dim] weight (gamma) FP32
    b_ptr,        # *[dim] bias (beta) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < rows

    # first pass: compute sum and sum of squares
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [BLOCK_M, BLOCK_D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized rows
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass, fully fused and written in fp16 where possible.
    The implementation follows the AlphaFold3 formulation:
      1. Row‑wise LayerNorm over the channel dimension C.
      2. One fused linear projection that simultaneously produces the left/right
         projections and the three gating vectors.
      3. Optional pair‑masking and sigmoid gates (handled by PyTorch for simplicity).
      4. Batched GEMM (cuBLAS) to compute the core 4‑D einsum.
      5. LayerNorm over the hidden dimension H, multiplication by the out‑gate,
         and final linear projection back to the original channel dimension.
    The first LayerNorm is performed with a Triton kernel; all other steps
    use highly‑optimized PyTorch kernels.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over last dimension (FP32 → FP16)
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection: generate left/right + three gates together
    # ------------------------------------------------------------------
    # concatenate weight matrices (each [H, dim]) → [5*H, dim]
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]

    # Linear projection (R, 5*H) in fp16
    proj = F.linear(x_norm, fused_w)  # no bias
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, and sigmoid gates
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = proj.chunk(5, dim=1)

    if not nomask and mask is not None:
        # mask shape [B, N, N] → flat [R, 1] for broadcasting
        mask_flat = mask.to(torch.float16).reshape(R, 1)
        left_raw = left_raw * mask_flat
        right_raw = right_raw * mask_flat
    # sigmoid gates (in fp16, torch internally promotes to fp32 for stability)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # apply gates
    left = left_raw * left_gate
    right = right_raw * right_gate
    # shape to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core pairwise contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # reshape to [B*H, N, N] for torch.bmm
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16
    out = out.view(B, H, N, N)  # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out‑gate, and final projection (pure PyTorch)
    # ------------------------------------------------------------------
    # flatten rows: [B, H, N, N] → [R, H]
    out_flat = out.permute(0, 2, 3, 1).reshape(R, H)          # FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).reshape(R, H)  # FP16

    # compute mean / variance in float32 for numerical stability
    out_fp32 = out_flat.float()
    mu = out_fp32.mean(dim=1, keepdim=True)                  # [R,1]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True) # [R,1]
    rstd = torch.rsqrt(var + eps)

    gamma = w["to_out_norm.weight"]  # [H] float32
    beta = w["to_out_norm.bias"]     # [H] float32

    # layer‑norm over hidden dimension
    out_norm = (out_fp32 - mu) * rstd               # [R,H]
    out_norm = out_norm * gamma + beta               # broadcasting over H

    # multiply by the out‑gate (cast gate to float32)
    out_norm = out_norm * out_gate_flat.float()

    # final linear projection back to `dim`
    # weight shape: [dim, hidden_dim]; torch.nn.functional.linear expects (out_features, in_features)
    final = F.linear(out_norm, w["to_out.weight"], bias=None)  # [R, dim] FP32

    final = final.view(B, N, N, dim)  # [B, N, N, dim]
    return final

Outgoing TriMul forward (AlphaFold‑3 style).

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and produces Y ∈ ℝ^{B×N×N×C}.
Steps:

1️⃣ Row‑wise LayerNorm over the channel dimension C (FP32 → FP16) – implemented as a
   Triton kernel `layernorm_kernel`.

2️⃣ One fused linear projection (weight matrix of shape (5·H, C)) creates the three
   projections (left/right) and the three gates (left‑gate/right‑gate/out‑gate).

3️⃣ Split the fused projection, optionally apply the pair‑mask, and apply the
   sigmoid to the three gate vectors (pure PyTorch – the heavy work is already
   done in the fused projection).

4️⃣ Batched GEMM (cuBLAS) evaluates the core einsum
   out_{b,h,i,j} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}  → shape (B, H, N, N).

5️⃣ Second LayerNorm over the hidden dimension H, multiplication by the out‑gate,
   and a final linear projection back to the original channel dimension C.
   This part is done with highly‑vectorised PyTorch operations (mean/var,
   broadcasting, and `torch.nn.functional.linear`).

    # 2️⃣ Fused projection producing left/right + three gates (FP16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # shape [5*H, dim]

    proj = F.linear(x_norm, fused_w)          # (R, 5*H) FP16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional pair mask, and sigmoid gates
    # ------------------------------------------------------------------
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = proj.chunk(5, dim=1)

    if not nomask and mask is not None:
        # mask → [R, 1] for broadcasting over the hidden dimension
        mask_flat = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat
        # NOTE: out_gate is *not* masked (same as reference implementation)

    left_gate  = torch.sigmoid(left_gate_raw)   # FP16
    right_gate = torch.sigmoid(right_gate_raw)  # FP16
    out_gate   = torch.sigmoid(out_gate_raw)    # FP16

    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # reshape to [B, H, N, N] (contiguous layout)
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    del left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left .view(B * H, N, N)   # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)   # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm → out‑gate → final projection (pure PyTorch)
    # ------------------------------------------------------------------
    # flatten rows: [B, H, N, N] → [R, H]
    out_flat = out.permute(0, 2, 3, 1).reshape(R, H)          # FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).reshape(R, H)  # FP16

    # mean / variance in float32 for stability
    out_fp32 = out_flat.float()
    mu = out_fp32.mean(dim=1, keepdim=True)                 # [R,1]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True) # [R,1]
    rstd = torch.rsqrt(var + eps)

    # to_out_norm parameters (γ, β) are FP32 vectors of length H
    gamma = w["to_out_norm.weight"]   # [H]
    beta  = w["to_out_norm.bias"]     # [H]

    # LayerNorm over hidden dimension H
    out_norm = (out_fp32 - mu) * rstd                 # [R,H]
    out_norm = out_norm * gamma + beta                # broadcast over H

    # multiply by the out‑gate (cast gate to FP32)
    out_norm = out_norm * out_gate_flat.float()        # [R,H]

    # final linear projection back to the original channel dimension C
    # weight shape: [dim, hidden_dim]; bias is None (bias=False in reference)
    final = F.linear(out_norm, w["to_out.weight"], bias=None)   # [R, dim] FP32

    final = final.view(B, N, N, dim)                  # [B, N, N, dim]
    return final
```
