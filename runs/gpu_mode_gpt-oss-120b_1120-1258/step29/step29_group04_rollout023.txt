    pid_row = tl.program_id(0)
    pid_dim = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    offs_row = row_start + tl.arange(0, BLOCK_M)  # shape [BLOCK_M]

    nn = N * N

    rem = offs_row - b * nn
    i = rem // N
    j = rem - i * N

    # compute hidden_stride = N * N

    hidden_stride = N * N

    gate_sh = tl.shared_memory((BLOCK_M * H), dtype=tl.float16)
    # allocate registers for sum and sum_sq per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over heads in chunks
    for h_start in range(0, H, BLOCK_H):
        cur_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # index = b * (H * hidden_stride) + cur_h[None, :] * hidden_stride + i[:, None] * N + j[:, None]

        # For each row r, offset for head cur_h: base_offset[r] + cur_h * hidden_stride

"""
Outgoing TriMul (forward‑only) – highly‑fused Triton implementation

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into one GEMM (torch.nn.functional.linear) → (R, 5·H) in FP16.
3️⃣ Split the fused projection, apply the optional mask, evaluate the
   sigmoid gates and write three tensors `left`, `right`, `out_gate`
   in the layout [B, H, N, N] (FP16).  (already fused in the original code)
4️⃣ Batched GEMM (torch.bmm) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → `out` is [B, H, N, N] (FP16).
5️⃣ **Fused** kernel reads `out` and `out_gate` only **once**, builds the
   per‑head values in shared memory, computes the per‑position mean/variance,
   applies the second LayerNorm, the output‑gate, and finally multiplies with
   the `to_out` weight.  The result `[B, N, N, dim]` is written in FP32.
   This eliminates the separate “rowwise‑stats” and “final‑output” kernels
   and halves the global‑memory traffic for the massive `[B·H·N·N]` tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, dim] input FP32
    y_ptr,            # [R, dim] output FP16
    w_ptr, b_ptr,    # [dim] LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- pass 1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]  FP16  fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                                       # [R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k          # start of h=0 plane

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # each projection occupies a contiguous chunk of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_g_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(proj_ptr + proj_base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        outg_raw    = tl.load(proj_ptr + proj_base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fused kernel: mean/var + 2nd LayerNorm + out‑gate + to_out
# ----------------------------------------------------------------------
@triton.jit
def fused_output_kernel(
    out_ptr,               # [B, H, N, N]  FP16   (result of left·rightᵀ)
    gate_ptr,              # [B, H, N, N]  FP16   (out_gate)
    ln_w_ptr, ln_b_ptr,    # [H]           FP32   (to_out_norm weight/bias)
    w_out_T_ptr,           # [H, dim]      FP16   (transposed to_out weight)
    final_ptr,             # [B, N, N, dim] FP32
    total_rows, N, dim, eps,
    BLOCK_M: tl.constexpr,   # rows per program (tile over B·N·N)
    BLOCK_N: tl.constexpr,   # dim tile
    BLOCK_H: tl.constexpr,   # head‑chunk size (compile‑time)
    H: tl.constexpr,         # hidden_dim (compile‑time)
):
    pid_m = tl.program_id(0)          # rows
    pid_n = tl.program_id(1)          # output‑dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_row = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_col = col_start + tl.arange(0, BLOCK_N)   # [N]
    mask_row = offs_row < total_rows
    mask_col = offs_col < dim

    # decode (b,i,j) from linear row index
    nn = N * N
    b = offs_row // nn
    rem = offs_row - b * nn
    i = rem // N
    j = rem - i * N

    # hidden stride = N*N
    hidden_stride = nn

    # base offset for head‑0 (per row)
    base_offset = b * (H * hidden_stride) + i * N + j          # [M]

    # ------------------------------------------------------------------
    # Shared memory for per‑head values (FP16)
    # shape: BLOCK_M * H
    # ------------------------------------------------------------------
    out_sh  = tl.shared_memory((BLOCK_M * H), dtype=tl.float16)
    gate_sh = tl.shared_memory((BLOCK_M * H), dtype=tl.float16)

    # ------------------------------------------------------------------
    # First pass: load out/gate, compute µ and σ², store values in SM
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_H):
        cur_h = h_start + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = cur_h < H

        # linear offset = base + head*hidden_stride
        offs = base_offset[:, None] + cur_h[None, :] * hidden_stride   # [M, BLOCK_H]

        out_tile  = tl.load(out_ptr + offs,
                            mask=mask_row[:, None] & mask_h[None, :],
                            other=0.0)                               # FP16
        gate_tile = tl.load(gate_ptr + offs,
                            mask=mask_row[:, None] & mask_h[None, :],
                            other=0.0)                               # FP16

        # store into shared memory (flattened)
        row_idx = tl.arange(0, BLOCK_M) * H
        sh_idx = row_idx[:, None] + cur_h[None, :]                         # [M, BLOCK_H]
        tl.store(out_sh + sh_idx, out_tile,
                 mask=mask_row[:, None] & mask_h[None, :])
        tl.store(gate_sh + sh_idx, gate_tile,
                 mask=mask_row[:, None] & mask_h[None, :])

        # accumulate for mean / variance
        out_f32 = out_tile.to(tl.float32)
        sum_val += tl.sum(out_f32, axis=1)
        sum_sq  += tl.sum(out_f32 * out_f32, axis=1)

    # compute mean and rsqrt‑variance per row
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalization, out‑gate, LayerNorm, projection
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)  # final output accumulator

    for h_start in range(0, H, BLOCK_H):
        cur_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = cur_h < H

        # load stored out / gate from SM
        row_idx = tl.arange(0, BLOCK_M) * H
        sh_idx = row_idx[:, None] + cur_h[None, :]
        out_tile  = tl.load(out_sh + sh_idx,
                            mask=mask_row[:, None] & mask_h[None, :])
        gate_tile = tl.load(gate_sh + sh_idx,
                            mask=mask_row[:, None] & mask_h[None, :])

        out_f32  = out_tile.to(tl.float32)            # [M, BLOCK_H]
        gate_f32 = gate_tile.to(tl.float32)           # [M, BLOCK_H]

        # LayerNorm parameters (γ,β) per head
        gamma = tl.load(ln_w_ptr + cur_h, mask=mask_h, other=0.0)  # [BLOCK_H] FP32
        beta  = tl.load(ln_b_ptr + cur_h, mask=mask_h, other=0.0)  # [BLOCK_H] FP32

        # broadcast mean / rstd over head dimension
        mean_b = mean[:, None]    # [M,1]
        rstd_b = rstd[:, None]    # [M,1]

        # ((out‑µ)·rstd·γ + β) * gate
        term = ((out_f32 - mean_b) * rstd_b) * gamma[None, :] + beta[None, :]
        term = term * gate_f32                                 # [M, BLOCK_H]

        # weight slice for these heads   w_out_T[h, d]
        w = tl.load(w_out_T_ptr + cur_h[:, None] * dim + offs_col[None, :],
                    mask=mask_h[:, None] & mask_col[None, :],
                    other=0.0)                                 # [BLOCK_H, BLOCK_N]
        w_f32 = w.to(tl.float32)

        # accumulate contribution: Σₕ termₕ·Wₕ
        acc += tl.dot(term, w_f32)                             # [M, BLOCK_N]

    # write final FP32 result
    dst_ptr = final_ptr + offs_row[:, None] * dim + offs_col[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_row[:, None] & mask_col[None, :])

                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[layernorm_grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for the five heads (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, dim]

    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64                     # inner split within the kernel
    split_grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[split_grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left·rightᵀ) – core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Fused stats + 2nd LayerNorm + out_gate + output projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 32     # rows per program (tile over B·N·N)
    BLOCK_N_OUT = 128    # output‑dim tile
    BLOCK_H_OUT = 32     # head‑chunk size for the kernel
    out_grid = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_output_kernel[out_grid](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_N=BLOCK_N_OUT,
        BLOCK_H=BLOCK_H_OUT,
        H=H,                       # compile‑time constant
        num_warps=8,
    )
    # clean‑up
    del out, out_gate, w_out_T

    return final
```
