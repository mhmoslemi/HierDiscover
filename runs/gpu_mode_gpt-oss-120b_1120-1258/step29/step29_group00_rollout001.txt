"""
Outgoing TriMul (TriMul‑out) – Triton‑accelerated forward pass (inference only)

The operator processes a 4‑D tensor `x` of shape [B, N, N, D] and returns a tensor
with the same shape.  The computation follows the “outgoing” formulation of the
AlphaFold‑3 TriMul:

1️⃣  Row‑wise LayerNorm on the last dimension (float32 → float16).
2️⃣  A *single* fused linear projection that produces the concatenation of the
    five hidden‑dim linear maps (left, right, left‑gate, right‑gate, out‑gate).
3️⃣  Triton kernel splits the fused projection, applies the optional pairwise
    mask, and evaluates the sigmoid gates.
4️⃣  Batched GEMM (torch.bmm) computes the core einsum
       out[b,i,j] = Σₖ left[b,i,k] • right[b,j,k]   →  shape [B, H, N, N] (fp16).
5️⃣  Triton kernel computes per‑position mean and rsqrt‑variance of the
    contracted tensor over the hidden dimension H.
6️⃣  A final Triton kernel fuses the second LayerNorm, the out‑gate, and the
    final projection back to the original dimension D.

All operations are performed without gradients and use mixed‑precision arithmetic
( fp32 for statistics, fp16 for the bulk of the tensor math ) to achieve maximal
throughput on an H100 GPU.  The final output is returned in float32.

The implementation keeps the memory‑traffic low by:
* fusing the five linear maps into one weight matrix,
* performing mask & gate fusion inside a single kernel,
* re‑using the per‑position statistics for the second LayerNorm,
* tiling the final projection with a small hidden‑dim block (≈32) which fits
  nicely into registers on H100.

The code has been tuned (block sizes, warp count) for the test configurations
listed in the problem statement, but it works for any B∈{1,2}, N∈{128,…,1024},
D∈{128,384,768}, H∈{128}.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, D]   FP32 input
    y_ptr,          # [R, D]   FP16 output
    w_ptr,          # [D]      LN weight (γ) FP32
    b_ptr,          # [D]      LN bias  (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1st pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Loop over the feature dimension in chunks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # [M, D]

        sum_val += tl.sum(vals, axis=1)        # [M]
        sum_sq  += tl.sum(vals * vals, axis=1) # [M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2nd pass: write normalized tensor ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # [M, D]

        norm = (vals - mean[:, None]) * rstd[:, None]          # [M, D]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # [D] FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # [D] FP32

        out = norm * gamma[None, :] + beta[None, :]               # [M, D]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask and compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused linear output
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)    # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r -> (b, i, k)
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Optional mask (broadcasted over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)    # [R]
        m_val = m_val[:, None]                                        # [R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Destination stride for a single hidden plane (N*N) inside the 4‑D tensors
    hidden_stride = N * N
    dst_base = b * (H * hidden_stride) + i * N + k                     # start for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # Linear layout of the fused projection:
        # [left | right | left_gate | right_gate | out_gate] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [R, H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid (computed in fp32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # Compute flat offsets into the 4‑D destination tensors
        dst_off = dst_base[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Row‑wise statistics (mean + rsqrt‑variance) of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride   # [M, K]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [M, K]

        sum_val += tl.sum(vals, axis=1)                  # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [M]

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    # 2‑D grid: rows (R) × output‑dim (dim)
    pid_m = tl.program_id(0)   # rows tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_off = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # Load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)  # [M]

    # Accumulator in fp32
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # Load hidden‑plane values and the out‑gate
        ptr_out  = out_ptr  + base_off[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_off[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # Second LayerNorm: (x-μ)/σ * γ + β
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]   # [M, K]

        # Apply out‑gate
        term = norm * vals_gate                                 # [M, K] (fp32)

        # Load slice of the final weight matrix (H, dim) – stored transposed
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N).  Use fp16 dot for inner product → fp32 accumulation.
        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # Write final result (row‑major [R, dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all model weights (float32)
            - config_dict  : must contain keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape               # x: [B, N, N, D]
    R = B * N * N                       # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, D] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128                # # rows per program
    BLOCK_D = 64                 # feature tile size
    grid_ln = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    # Weight matrix [5·H, D] stored in FP16 for speed
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, D]

    proj = F.linear(x_norm, fused_w)                     # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # Broadcast mask over the hidden dimension and flatten
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask   = 1
        mask_ptr   = mask_flat
    else:
        has_mask = 0
        # Dummy tensor – will never be read because HAS_MASK=0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64    # rows tile for split kernel
    BLOCK_H = 64    # hidden‑dim tile
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) and compute left·rightᵀ via torch.bmm
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256    # rows tile for statistics kernel
    BLOCK_K_STATS = 128    # hidden‑dim tile (covers H ≤ 128)
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)

    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    # Transpose to_out weight for easier tiling (shape [H, dim])
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    out_final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim tile (nice register usage on H100)
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )

    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        out_final,
        R,
        H,
        N,
