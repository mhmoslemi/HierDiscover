"""
TriMul – outgoing version (forward only)

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the flattened [B·N·N, C] tensor.
2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into one matrix‑multiply (R,5·H) and split
    the result with a tiny Triton kernel that also applies the optional
    mask and the sigmoid gates, producing three tensors
        left   : [B, H, N, N]   FP16
        right  : [B, H, N, N]   FP16
        out_g  : [B, H, N, N]   FP16
3️⃣  Batched GEMM (torch.bmm) computes the core einsum
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
    → tensor `[B, H, N, N]` (FP16).
4️⃣  Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
    (Triton kernel) → μ, rstd per (b,i,j) in FP32.
5️⃣  Final fused kernel:
        (out‑μ)·rstd·γ + β   (LayerNorm over H)
        × out_g             (gate)
        × W_out (H→C)       (matrix‑multiply to the original dim)
    The kernel emits the final tensor `[B, N, N, C]` (FP32).

All kernels target Triton 3.3.1 and use mixed‑precision (FP16 for the
large pair‑wise ops, FP32 for the statistics).  The implementation
requires only a forward pass – no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # [R, C] input  (float32)
    y_ptr,               # [R, C] output (float16)
    w_ptr,               # [C]    γ (float32)
    b_ptr,               # [C]    β (float32)
    rows, cols, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, tl.cdiv(cols, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < cols

        ptr = x_ptr + offs_r[:, None] * cols + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / cols
    var  = sum_sq / cols - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized result ----------
    for c_off in range(0, tl.cdiv(cols, BLOCK_C)):
        offs_c = c_off * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = offs_c < cols

        ptr = x_ptr + offs_r[:, None] * cols + offs_c[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_c[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_c, mask=mask_c, other=0.0)
        beta  = tl.load(b_ptr + offs_c, mask=mask_c, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_r[:, None] * cols + offs_c[None, :],
                 out.to(tl.float16),
                 mask=mask_r[:, None] & mask_c[None, :])

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16

    mask_ptr,          # [R] FP16 (optional)

    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over H
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                # broadcast over H
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of h=0 plane for this (b,i,k)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view into fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (FP32 → FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑var over hidden dimension H
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,            # [B, H, N, N] FP16
    mu_ptr, rstd_ptr,   # [R] FP32     (R = B·N·N)
    total_rows, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < total_rows

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of h=0 plane for (b,i,j)

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_r, mean,  mask=mask_r)
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)


# ----------------------------------------------------------------------
# 4️⃣ Final LayerNorm + out_gate + projection (fused GEMM)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,           # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,          # [B, H, N, N] FP16   (out_gate)
    mu_ptr, rstd_ptr,  # [R] FP32
    ln_w_ptr, ln_b_ptr,# [H] FP32   (to_out_norm weight & bias)
    w_out_T_ptr,       # [H, dim] FP16  (transpose of to_out.weight)
    out_ptr_final,     # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)    # tile over rows (R)
    pid_n = tl.program_id(1)    # tile over output dim (dim)

    row_start = pid_m * BLOCK_R
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_R)    # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)    # [N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j    # start of h=0 plane for (b,i,j)

    # per‑row statistics μ, rstd
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    # accumulator for the final GEMM (FP32)
    acc = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor and gate (FP16 → FP32)
        out_tile = tl.load(out_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)      # [M, K]

        gate_tile = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)      # [M, K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out‑μ)·rstd·γ + β) × gate
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_tile                         # [M, K]

        # weight slice (H, dim) – note that we stored the weight transposed
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, N]

        # GEMM for this hidden chunk
        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)      # [M, N]

    # store final result (FP32)
    dst_ptr = out_ptr_final + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point – exported as required by the environment
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]    (float32)
        - mask_tensor  : torch.Tensor [B, N, N]        (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32)
        - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total (b,i,k) tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_C = 64
    grid_ln = (triton.cdiv(R, BLOCK_R),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_C=BLOCK_C,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (five heads) + split + mask + gates
    # ------------------------------------------------------------------
    # Stack the five weight matrices (no bias in the reference implementation)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]

    # Linear projection of the normalized input (R,5·H) – still FP16
    proj = F.linear(x_norm, fused_w)                       # (R, 5·H)
    del fused_w, x_norm

    # Allocate tensors for the three streams that will be used later
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # mask is broadcasted over the hidden dimension
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R_SPLIT = 64
    BLOCK_H_SPLIT = 64
    grid_split = (triton.cdiv(R, BLOCK_R_SPLIT),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R_SPLIT,
        BLOCK_H=BLOCK_H_SPLIT,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Core einsum via batched GEMM (left·rightᵀ)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) and use torch.bmm which dispatches to an
    # extremely fast cuBLAS GEMM (TensorCores on H100)
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                  # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑var over hidden dimension H
    # ------------------------------------------------------------------
    mu    = torch.empty(R, dtype=torch.float32, device=device)
    rstd  = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_R_STATS = 256
    BLOCK_K_STATS = 128                # H ≤ 128 → fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_R_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out_gate + to_out projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_R_OUT = 128          # rows per program (tiles over R)
    BLOCK_K_OUT = 32           # hidden‑dim chunk per iteration
    BLOCK_N_OUT = 128          # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_R_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_R=BLOCK_R_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
