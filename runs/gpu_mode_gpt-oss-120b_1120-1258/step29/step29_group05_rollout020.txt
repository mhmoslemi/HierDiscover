@triton.jit
def final_fused_kernel(
    out_ptr, gate_ptr, ln_w_ptr, ln_b_ptr, w_out_T_ptr, out_final_ptr,
    R, H, N, dim, eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr
):
    pid = tl.program_id(0)
    col = tl.program_id(1)
    row_start = pid * BLOCK_R
    col_start = col * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_R)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # decode indices: r = b*N*N + i*N + j (since shape [B,N,N,H] flattened across B,N,N dimension)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # compute base offset for hidden dimension stride = 1 (since hidden last)
    hidden_stride = 1
    # base offset = ((b* N + i)* N + j) * H
    base = (b * N * N + i * N + j) * H

    # accumulate sum and sum_sq on the out hidden per row:
    sum_out = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    # accumulators for term1, term2, term3 per tile of dim
    acc1 = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)
    acc2 = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)
    acc3 = tl.zeros([BLOCK_R, BLOCK_N], dtype=tl.float32)

    # iterate over hidden in steps of BLOCK_K
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out hidden values and gate (scalar per hidden)
        out_vals = tl.load(out_ptr + base[:, None] + offs_k[None, :] * hidden_stride, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)  # shape [M,K]
        gate_vals = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)  # [M,K]

        sum_out += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

        # Load layernorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0) # [K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0) # [K]

        # Compute left operands for term1, term2, term3
        # term1 left = out_vals * gamma * gate_vals
        left1 = out_vals * gamma[None, :] * gate_vals  # [M,K] (fp32)
        # term2 left = gamma * gate_vals
        left2 = gamma[None, :] * gate_vals  # [M,K]
        # term3 left = beta * gate_vals
        left3 = beta[None, :] * gate_vals  # [M,K]

        # weight tile (K,N)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]  # [K,N]
        w_tile = tl.load(w_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)  # convert to fp32

        # Perform dot products
        acc1 += tl.dot(left1, w_tile)  # (M,N)
        acc2 += tl.dot(left2, w_tile)
        acc3 += tl.dot(left3, w_tile)

    # after loop compute mu and rstd
    mu = sum_out / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # compute final
    # final = rstd * acc1 - mu * rstd * acc2 + acc3
    # broadcast mu,rstd over dim axis
    mu_vec = mu[:, None]  # [M,1]
    rstd_vec = rstd[:, None]

    final_tile = rstd_vec * acc1 - mu_vec * rstd_vec * acc2 + acc3

    # store
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, final_tile, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]; hidden_dim = cfg["hidden_dim"]; nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. Initial LayerNorm (rowwise)
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 128; BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat, x_norm,
        w["norm.weight"], w["norm.bias"],
        R, dim, eps,
        BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D,
        num_warps=8
    )

    # 2. fused linear projection
    fused_w = torch.cat([...], dim=0).to(torch.float16) # shape [5*H, dim]
    proj = torch.nn.functional.linear(x_norm, fused_w)  # [R, 5*H] fp16

    # 3. split and apply mask and sigmoid gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64; BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R, H, N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8
    )
    del proj

    # 4. Compute left @ right^T across k dimension (einsum)

        BLOCK_N=BLOCK_N_OUT,

"""
TriMul (outgoing) – fused‑pass implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS) → tensor (R, 5·H) where R = B·N·N.
3️⃣ Split the fused projection, apply the optional pair‑mask, and compute the
   sigmoid gates (left, right, out_gate) – all in a Triton kernel.
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]   → [B, H, N, N].
   The result is immediately permuted to [B, N, N, H] so the hidden dimension
   becomes the fastest‑moving axis (contiguous reads later).
5️⃣ One‑pass Triton kernel fuses **row‑wise statistics** (mean / rsqrt‑var),
   the second LayerNorm, the out_gate multiplication and the final linear
   projection.  For each (b,i,j) the kernel accumulates:
     * Σₕ outₕ               → μ
     * Σₕ outₕ²              → σ²
     * Σₕ outₕ·γₕ·gₕ·Wₕ  → A₁  (γ = LN weight, g = out_gate, W = to_outᵀ)
     * Σₕ γₕ·gₕ·Wₕ          → A₂
     * Σₕ βₕ·gₕ·Wₕ          → A₃  (β = LN bias)
   The final output is
        rstd·A₁  –  μ·rstd·A₂  +  A₃
   where rstd = rsqrt(σ² + eps).  Everything is kept in registers; the
   hidden dimension is accessed contiguously, halving global‑memory traffic.
6️⃣ The kernel writes the `[B, N, N, dim]` FP32 tensor.

The implementation is forward‑only, uses mixed‑precision (FP16 work, FP32
statistics) and is tuned for the H100 (Triton 3.3.1).  It supports the
“nomask” and “masked” variants of the TriMul operator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]   FP32 input
    y_ptr,   # [R, dim]   FP16 output
    w_ptr,   # [dim]      γ (FP32)
    b_ptr,   # [dim]      β (FP32)
    rows,    # scalar    R
    dim,     # scalar    dim
    eps,     # scalar    eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: sum and sum‑of‑squares ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # FP32

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]          # FP32 → FP16
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r,
                        mask=mask_r,
                        other=1.0)          # [BLOCK_R]
        m_val = m_val[:, None]                # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k        # start of hidden plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # FP32 sigmoid for accuracy, then cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr      + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ One‑pass kernel: stats, LayerNorm, out_gate, final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,        # [B, N, N, H] FP16
    gate_ptr,       # [B, N, N, H] FP16
    ln_w_ptr,       # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32   (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16  (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    R, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)     # rows (= B·N·N)
    pid_n = tl.program_id(1)     # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # each row contains a contiguous hidden vector of length H
    base = offs_m * H                                   # start of hidden slice

    # accumulators
    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    acc1 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out·γ·g·W
    acc2 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ γ·g·W
    acc3 = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ β·g·W

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load hidden values (contiguous)
        out_vals = tl.load(out_ptr + base[:, None] + offs_k[None, :],
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M,K]
        gate_vals = tl.load(gate_ptr + base[:, None] + offs_k[None, :],
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        sum_out += tl.sum(out_vals, axis=1)
        sum_sq  += tl.sum(out_vals * out_vals, axis=1)

        # LayerNorm parameters (γ, β)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        # left‑hand sides for the three dot‑products (all FP32)
        left1 = out_vals * gamma[None, :] * gate_vals      # out·γ·g
        left2 = gamma[None, :] * gate_vals                  # γ·g
        left3 = beta[None, :] * gate_vals                   # β·g

        # weight tile (K, BLOCK_N) – up‑cast to FP32 once
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)          # [K,N]

        acc1 += tl.dot(left1, w_tile)   # Σ out·γ·g·W
        acc2 += tl.dot(left2, w_tile)   # Σ γ·g·W
        acc3 += tl.dot(left3, w_tile)   # Σ β·g·W

    # row‑wise mean & rsqrt‑variance
    mu   = sum_out / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # combine the three accumulated terms:
    #   final = rstd·acc1  -  mu·rstd·acc2  +  acc3
    final_tile = rstd[:, None] * acc1 \
                 - mu[:, None] * rstd[:, None] * acc2 \
                 + acc3

    # store result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, final_tile,
             mask=mask_m[:, None] & mask_n[None, :])

            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None

                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """

    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------

    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_D = 64

        w["norm.bias"],
        R,
        dim,

        BLOCK_D=BLOCK_D,

    )

    # ------------------------------------------------------------------

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)       # [5·H, dim]
    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right : [B, H, N, N]  →  reshape to [B·H, N, N] for cuBLAS
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))       # [B·H, N, N] FP16
    out = out.view(B, H, N, N).contiguous()                    # [B, H, N, N]

    # --------------------------------------------------------------
    # 5️⃣ Permute to make the hidden dimension contiguous
    # --------------------------------------------------------------
    # shape becomes [B, N, N, H] – optimal for the fused kernel
    out = out.permute(0, 2, 3, 1).contiguous()
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()
    del left, right  # free intermediate tensors

    # --------------------------------------------------------------
    # 6️⃣ One‑pass fused kernel (stats + LN + gating + final proj)
    # --------------------------------------------------------------
    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    BLOCK_M_OUT = 128          # rows per program
    BLOCK_K_OUT = 32           # hidden‑dim tile
    BLOCK_N_OUT = 128          # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),      # rows dimension
        triton.cdiv(dim, BLOCK_N_OUT),    # output‑dim dimension
    )
    final_fused_kernel[grid_out](
        out,                                   # [B, N, N, H]
        out_gate,                              # [B, N, N, H]
        w["to_out_norm.weight"],               # γ (FP32)
        w["to_out_norm.bias"],                 # β (FP32)
        w_out_T,                               # [H, dim] (FP16)
        final,                                 # [B, N, N, dim] (FP32)
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
```
