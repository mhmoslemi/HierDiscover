def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N
    H = hidden_dim

    # step1: layernorm using torch
    x_flat = input_tensor.reshape(R, dim).contiguous()
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(x_flat, (dim,), weight=norm_weight, bias=norm_bias, eps=eps).to(torch.float16)

    # step2: fused projection
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]
    proj = F.linear(x_norm, fused_weight)  # (R, 5*H)

    # step3: split and gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not config.get("nomask", True) and mask_tensor is not None:
        mask_flat = mask_tensor.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128  # maybe larger
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # step4: batched GEMM
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)

    # step5: layernorm on out, gating, final projection
    # reshape to (R, H)
    out_perm = out.permute(0, 2, 3, 1).contiguous()
    out_flat = out_perm.view(R, H)   # FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # FP16

    # per-row stats
    out_f = out_flat.float()
    mu = out_f.mean(dim=1)                     # (R,)
    var = out_f.var(dim=1, unbiased=False)    # (R,)
    rstd = (var + eps).rsqrt()                # (R,)

    # normalize
    ln_w = weights["to_out_norm.weight"]   # (H,)
    ln_b = weights["to_out_norm.bias"]     # (H,)
    # broadcasting: (R, H)
    normalized = ((out_f - mu.unsqueeze(1)) * rstd.unsqueeze(1)) * ln_w + ln_b

    # gate
    gated = normalized * out_gate_flat.float()

    # final projection
    final_flat = gated @ weights["to_out.weight"].t()   # (R, dim) float32
    final = final_flat.view(B, N, N, dim)

    return final
```

"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm (high‑level)
----------------------
1️⃣  Row‑wise LayerNorm on the flattened input (torch.nn.functional.layer_norm).
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) → (R, 5·H) in FP16.
3️⃣  Split the fused projection, apply the optional mask and sigmoid gates
    (Triton kernel) → tensors   left, right, out_gate  of shape [B,H,N,N].
4️⃣  Batched GEMM (torch.bmm) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   (FP16).
5️⃣  Row‑wise LayerNorm on `out` (torch), multiply by `out_gate`,
    and a final linear projection to the original channel dimension
    (all performed with vectorised torch kernels → cuBLAS).
The only Triton kernel needed is the split‑&‑gate kernel,
which dramatically cuts the amount of Python‑level work and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k   (k is the summed‑over dimension)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask broadcast over H
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                      # distance between two hidden planes
    # Base offset for hidden‑plane 0 (h = 0) in the destination tensors
    base_offset = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    # Loop over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = offs_h < H

        # Linear offset inside the fused projection: [left|right|l_gate|r_gate|out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]     # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Stable sigmoid in FP32, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                     # no mask on out_gate

        # Destination offsets for each hidden plane
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (use torch's highly‑optimised kernel)
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                         # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                             # (R, 5·H) FP16
    del fused_w, x_norm                                          # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask_tensor is not None):
        mask_flat = mask_tensor.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # tile sizes – chosen to balance occupancy and register usage
    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm, out‑gate, final projection (all torch)
    # ------------------------------------------------------------------
    # bring hidden dimension to the last axis for easy flattening
    out_perm = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]
    out_flat = out_perm.view(R, H)                         # FP16

    out_gate_flat = (
        out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)
    )                                                        # FP16

    # per‑row mean / variance (FP32 accumulator)
    out_f = out_flat.float()
    mu = out_f.mean(dim=1)                                 # (R,)
    var = out_f.var(dim=1, unbiased=False)                # (R,)
    rstd = (var + eps).rsqrt()                             # (R,)

    # second LayerNorm (γ,β) – shapes (H,)
    ln_w = weights["to_out_norm.weight"]
    ln_b = weights["to_out_norm.bias"]

    # normalisation + gating
    normalized = ((out_f - mu.unsqueeze(1)) * rstd.unsqueeze(1)) * ln_w + ln_b
    gated = normalized * out_gate_flat.float()            # (R, H) FP32

    # final projection to original channel dimension
    final_flat = gated @ weights["to_out.weight"].t()      # (R, dim) FP32
    final = final_flat.view(B, N, N, dim)                 # [B, N, N, dim]

    return final
```

This implementation keeps the heavy linear‑projection fusion and the mask/gate split in a Triton kernel while delegating all large‑scale dense operations (batched GEMM, statistics, final projection) to PyTorch’s highly optimised cuBLAS‑based kernels. The extra memory pass over the contracted tensor is eliminated, yielding a faster forward pass without sacrificing numerical correctness.
