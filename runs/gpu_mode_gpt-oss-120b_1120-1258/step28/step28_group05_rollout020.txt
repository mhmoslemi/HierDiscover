"""
Outgoing TriMul (AlphaFold‑3) – forward only.

Algorithm (high‑level, mixed‑precision)
---------------------------------------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor  (FP32 → FP16).
2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
   with the optional pair‑mask and the sigmoid gates in a single Triton kernel.
   The kernel computes the three tensors `left`, `right` (already gated) and
   `out_gate` in the canonical layout `[B, H, N, N]` (FP16) – no huge
   intermediate (R × 5H) matrix is materialised.
3️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] • right[b, h, j, k]
   → `[B, H, N, N]` (FP16).
4️⃣ Triton kernel computes row‑wise mean & rsqrt‑variance of `out`
   (over the hidden dimension H) → μ, rstd per (b,i,j).
5️⃣ Final Triton kernel fuses the second LayerNorm, the `out_gate` multiplication,
   and the final projection (`to_out`) into a single kernel that writes the
   result `[B, N, N, dim]` (float32).

All kernels target Triton 3.3.1 on an H100 and use mixed‑precision:
  * FP32 for statistics,
  * FP16 for the bulk of the computation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, dim]   FP32 input
    y_ptr,      # [R, dim]   FP16 output
    w_ptr,      # [dim]      γ (FP32)
    b_ptr,      # [dim]      β (FP32)
    rows,       # scalar     R
    dim,        # scalar     dim
    eps,        # scalar     epsilon
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1st pass – compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2nd pass – write normalized values
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,                # [R, dim]          FP16 normalized input
    left_ptr,             # [B, H, N, N]      FP16 (gated)
    right_ptr,            # [B, H, N, N]      FP16 (gated)
    out_gate_ptr,         # [B, H, N, N]      FP16
    mask_ptr,             # [R]               FP16 (optional)
    left_w_ptr,           # [dim, H]          FP16 (left_proj.T)
    right_w_ptr,          # [dim, H]          FP16 (right_proj.T)
    left_gate_w_ptr,      # [dim, H]          FP16
    right_gate_w_ptr,     # [dim, H]          FP16
    out_gate_w_ptr,       # [dim, H]          FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k          # start of hidden‑plane 0

    # ------------------------------------------------------------------
    # Iterate over hidden tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators (FP32) – one per projection / gate
        sum_left   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        sum_right  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        sum_lgate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        sum_rgate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        sum_ogate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        # Dot over the channel dimension
        # ------------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
            mask_d = offs_d < dim

            # load a tile of the normalized input
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                # [R, D]

            # load transposed weight tiles  (shape: [BLOCK_D, BLOCK_H])
            # each weight matrix is stored as (dim, H) → stride_dim = H, stride_H = 1
            w_left   = tl.load(left_w_ptr   + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)            # [D, H]
            w_right  = tl.load(right_w_ptr  + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            w_lgate  = tl.load(left_gate_w_ptr   + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            w_rgate  = tl.load(right_gate_w_ptr  + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            w_ogate  = tl.load(out_gate_w_ptr   + offs_d[:, None] * H + offs_h[None, :],
                               mask=mask_d[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

            # (R,D) @ (D,H) → (R,H)
            sum_left  += tl.dot(x, w_left)
            sum_right += tl.dot(x, w_right)
            sum_lgate += tl.dot(x, w_lgate)
            sum_rgate += tl.dot(x, w_rgate)
            sum_ogate += tl.dot(x, w_ogate)

        # ------------------------------------------------------------------
        # Apply sigmoid gates (computed in FP32 then cast to FP16)
        # ------------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-sum_lgate))).to(tl.float16)   # [R, H]
        right_gate = (1.0 / (1.0 + tl.exp(-sum_rgate))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-sum_ogate))).to(tl.float16)

        # ------------------------------------------------------------------
        # Optional pair mask (broadcast over H)
        # ------------------------------------------------------------------
        if HAS_MASK:
            mask_val = tl.load(mask_ptr + offs_r,
                               mask=mask_r,
                               other=1.0)                         # [R] FP16
        else:
            mask_val = tl.full([BLOCK_R], 1.0, dtype=tl.float16)       # [R]

        mask_val = mask_val[:, None]                                   # [R,1] – broadcast over H
        mask_fp32 = mask_val.to(tl.float32)

        # ------------------------------------------------------------------
        # Final gated values (store as FP16)
        # ------------------------------------------------------------------
        left_fp16  = (sum_left  * mask_fp32 * left_gate .to(tl.float32)).to(tl.float16)
        right_fp16 = (sum_right * mask_fp32 * right_gate.to(tl.float32)).to(tl.float16)

        dst = base[:, None] + offs_h[None, :] * hidden_stride        # [R, H]

        tl.store(left_ptr     + dst, left_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right_fp16,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N]  FP16
    mu_ptr,    # [R]            FP32
    rstd_ptr,  # [R]            FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [M,K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N]  FP16
    gate_ptr,       # [B, H, N, N]  FP16 (out_gate)
    mu_ptr,         # [R]            FP32
    rstd_ptr,       # [R]            FP32
    ln_w_ptr,       # [H]            FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]            FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]       FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # load row‑wise statistics (FP32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # load hidden activation and its gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M,K]

        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)         # [M,K]

        # LayerNorm parameters for the hidden dimension
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M,K]
        term = norm * vals_gate                                       # [M,K]
        term_fp16 = term.to(tl.float16)                               # back to FP16 for matmul

        # Load a tile of the final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, dim]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                     # [K, dim]

        # Accumulate (M,K)·(K,dim) → (M,dim)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, already on the target device)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack everything
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                              # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm  (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 128
    BLOCK_D_LN = 64
    grid_ln = (triton.cdiv(R, BLOCK_M_LN),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=8,
    )
    del x_flat   # free the FP32 view

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps, apply mask & sigmoid gates
    # --------------------------------------------------------------
    # Transpose the weight matrices so we can load them as (dim, H)
    left_w_T     = w["left_proj.weight"].to(torch.float16).t().contiguous()
    right_w_T    = w["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_w_T  = w["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_w_T = w["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_w_T   = w["out_gate.weight"].to(torch.float16).t().contiguous()

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    BLOCK_D = 64
    grid_fuse = (triton.cdiv(R, BLOCK_R),)
    fused_proj_gate_kernel[grid_fuse](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_w_T,
        right_w_T,
        left_gate_w_T,
        right_gate_w_T,
        out_gate_w_T,
        R,
        H,
        N,
        dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, left_w_T, right_w_T, left_gate_w_T, right_gate_w_T, out_gate_w_T, mask_ptr

    # --------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                # H ≤ 128, fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim tile
    BLOCK_N_OUT = 128   # output‑dim tile (dim)

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
